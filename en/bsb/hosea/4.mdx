<span className="v" id="v17" data-v="17" data-xref="Psalms 81:12; Revelation 22:11; Hosea 13:2"> <span className="pl pl-1">Ephraim</span><sup className="fn"><a href="#fn-e" id="fnref-e" data-fn="That is, the northern kingdom of Israel">e</a></sup><br /><span className="pl pl-1">is joined to idols;</span><br /><span className="pl pl-2">leave him alone!</span></span>
