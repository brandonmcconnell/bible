<span className="v" id="v1" data-v="1" data-xref="नीतिवचन 6:15; 1 शमूएल 2:25; नीतिवचन 1:24–31"> <span className="pl pl-1">जो बार बार डाँटे जाने पर भी हठ करता है, वह अचानक नष्ट हो जाएगा29:1 अचानक नष्ट हो जाएगा: दीर्घ काल से विलम्बित दण्ड की आकस्मिकता पर बल दिया गया है।</span><br /><span className="pl pl-1">और उसका कोई भी उपाय काम न आएगा।</span><br /></span>
<span className="v" id="v2" data-v="2" data-xref="नीतिवचन 28:12; नीतिवचन 11:10; नीतिवचन 28:28"> <span className="pl pl-1">जब धर्मी लोग शिरोमणि होते हैं, तब प्रजा आनन्दित होती है;</span><br /><span className="pl pl-1">परन्तु जब दुष्ट प्रभुता करता है तब प्रजा हाय-हाय करती है।</span><br /></span>
<span className="v" id="v3" data-v="3" data-xref="नीतिवचन 27:11; नीतिवचन 10:1; नीतिवचन 15:20"> <span className="pl pl-1">जो बुद्धि से प्रीति रखता है, वह अपने पिता को आनन्दित करता है,</span><br /><span className="pl pl-1">परन्तु वेश्याओं की संगति करनेवाला धन को उड़ा देता है। (लूका 15:13)</span><br /></span>
<span className="v" id="v4" data-v="4" data-xref="नीतिवचन 29:14; नीतिवचन 20:8; दानिय्येल 11:20"> <span className="pl pl-1">राजा न्याय से देश को स्थिर करता है,</span><br /><span className="pl pl-1">परन्तु जो बहुत घूस लेता है उसको उलट देता है।</span><br /></span>
<span className="v" id="v5" data-v="5" data-xref="होशे 5:1; अय्यूब 17:5; भजन संहिता 5:9"> <span className="pl pl-1">जो पुरुष किसी से चिकनी चुपड़ी बातें करता है,</span><br /><span className="pl pl-1">वह उसके पैरों के लिये जाल लगाता है।</span><br /></span>
<span className="v" id="v6" data-v="6" data-xref="सभोपदेशक 9:12; निर्गमन 15:1; 2 तीमुथियुस 2:26"> <span className="pl pl-1">बुरे मनुष्य का अपराध उसके लिए फंदा होता है,</span><br /><span className="pl pl-1">परन्तु धर्मी आनन्दित होकर जयजयकार करता है।</span><br /></span>
<span className="v" id="v7" data-v="7" data-xref="अय्यूब 29:16; भजन संहिता 41:1; नीतिवचन 21:13"> <span className="pl pl-1">धर्मी पुरुष कंगालों के मुकद्दमे में मन लगाता है;</span><br /><span className="pl pl-1">परन्तु दुष्ट जन उसे जानने की समझ नहीं रखता।</span><br /></span>
<span className="v" id="v8" data-v="8" data-xref="नीतिवचन 11:11; यिर्मयाह 15:1; याकूब 5:15–18"> <span className="pl pl-1">ठट्ठा करनेवाले लोग नगर को फूँक देते हैं,</span><br /><span className="pl pl-1">परन्तु बुद्धिमान लोग क्रोध को ठंडा करते हैं।</span><br /></span>
<span className="v" id="v9" data-v="9" data-xref="नीतिवचन 26:4; सभोपदेशक 10:13; मत्ती 7:6"> <span className="pl pl-1">जब बुद्धिमान मूर्ख के साथ वाद-विवाद करता है,</span><br /><span className="pl pl-1">तब वह मूर्ख क्रोधित होता और ठट्ठा करता है, और वहाँ शान्ति नहीं रहती।</span><br /></span>
<span className="v" id="v10" data-v="10" data-xref="यूहन्ना 15:18–19; 1 यूहन्ना 3:12–13; उत्पत्ति 4:5–8"> <span className="pl pl-1">हत्यारे लोग खरे पुरुष से बैर रखते हैं,</span><br /><span className="pl pl-1">और सीधे लोगों के प्राण की खोज करते हैं।</span><br /></span>
<span className="v" id="v11" data-v="11" data-xref="नीतिवचन 19:11; नीतिवचन 12:16; मीका 7:5"> <span className="pl pl-1">मूर्ख अपने सारे मन की बात खोल देता है,</span><br /><span className="pl pl-1">परन्तु बुद्धिमान अपने मन को रोकता, और शान्त कर देता है।</span><br /></span>
<span className="v" id="v12" data-v="12" data-xref="नीतिवचन 20:8; 2 राजाओं 10:6–7; भजन संहिता 52:2–4"> <span className="pl pl-1">जब हाकिम झूठी बात की ओर कान लगाता है,</span><br /><span className="pl pl-1">तब उसके सब सेवक दुष्ट हो जाते हैं29:12 उसके सब सेवक दुष्ट हो जाते हैं: वे जानते हैं कि किस बात से प्रसन्नता होगी, वे दूसरों की बुराई करनेवाले बन जाते है।।</span><br /></span>
<span className="v" id="v13" data-v="13" data-xref="मत्ती 5:45; नीतिवचन 22:2; भजन संहिता 13:3"> <span className="pl pl-1">निर्धन और अंधेर करनेवाले व्यक्तियों में एक समानता है;</span><br /><span className="pl pl-1">यहोवा दोनों की आँखों में ज्योति देता है।</span><br /></span>
<span className="v" id="v14" data-v="14" data-xref="नीतिवचन 16:12; यशायाह 11:4; भजन संहिता 72:2–4"> <span className="pl pl-1">जो राजा कंगालों का न्याय सच्चाई से चुकाता है,</span><br /><span className="pl pl-1">उसकी गद्दी सदैव स्थिर रहती है।</span><br /></span>
<span className="v" id="v15" data-v="15" data-xref="नीतिवचन 10:1; नीतिवचन 17:25; नीतिवचन 29:17"> <span className="pl pl-1">छड़ी और डाँट से बुद्धि प्राप्त होती है,</span><br /><span className="pl pl-1">परन्तु जो लड़का ऐसे ही छोड़ा जाता है वह अपनी माता की लज्जा का कारण होता है।</span><br /></span>
<span className="v" id="v16" data-v="16" data-xref="भजन संहिता 37:36; भजन संहिता 91:8; भजन संहिता 58:10"> <span className="pl pl-1">दुष्टों के बढ़ने से अपराध भी बढ़ता है;</span><br /><span className="pl pl-1">परन्तु अन्त में धर्मी लोग उनका गिरना देख लेते हैं।</span><br /></span>
<span className="v" id="v17" data-v="17" data-xref="नीतिवचन 10:1; नीतिवचन 29:15; नीतिवचन 23:13–14"> <span className="pl pl-1">अपने बेटे की ताड़ना कर, तब उससे तुझे चैन मिलेगा;</span><br /><span className="pl pl-1">और तेरा मन सुखी हो जाएगा।</span><br /></span>
<span className="v" id="v18" data-v="18" data-xref="याकूब 1:25; लूका 11:28; यूहन्ना 13:17"> <span className="pl pl-1">जहाँ दर्शन की बात नहीं होती, वहाँ लोग निरंकुश हो जाते हैं,</span><br /><span className="pl pl-1">परन्तु जो व्यवस्था को मानता है वह धन्य होता है।</span><br /></span>
<span className="v" id="v19" data-v="19" data-xref="नीतिवचन 26:3; अय्यूब 19:16; नीतिवचन 30:22"> <span className="pl pl-1">दास बातों ही के द्वारा सुधारा नहीं जाता,</span><br /><span className="pl pl-1">क्योंकि वह समझकर भी नहीं मानता।</span><br /></span>
<span className="v" id="v20" data-v="20" data-xref="याकूब 1:19; नीतिवचन 26:12; नीतिवचन 14:29"> <span className="pl pl-1">क्या तू बातें करने में उतावली करनेवाले मनुष्य को देखता है?</span><br /><span className="pl pl-1">उससे अधिक तो मूर्ख ही से आशा है।</span><br /></span>
<span className="v" id="v21" data-v="21"> <span className="pl pl-1">जो अपने दास को उसके लड़कपन से ही लाड़-प्यार से पालता है,</span><br /><span className="pl pl-1">वह दास अन्त में उसका बेटा बन बैठता है।</span><br /></span>
<span className="v" id="v22" data-v="22" data-xref="नीतिवचन 15:18; नीतिवचन 17:19; नीतिवचन 26:21"> <span className="pl pl-1">क्रोध करनेवाला मनुष्य झगड़ा मचाता है</span><br /><span className="pl pl-1">और अत्यन्त क्रोध करनेवाला अपराधी भी होता है।</span><br /></span>
<span className="v" id="v23" data-v="23" data-xref="मत्ती 23:12; लूका 14:11; नीतिवचन 11:2"> <span className="pl pl-1">मनुष्य को गर्व के कारण नीचा देखना पड़ता है,</span><br /><span className="pl pl-1">परन्तु नम्र आत्मावाला महिमा का अधिकारी होता है। (मत्ती 23:12)</span><br /></span>
<span className="v" id="v24" data-v="24" data-xref="लैव्यव्यवस्था 5:1; नीतिवचन 1:11–19; नीतिवचन 8:36"> <span className="pl pl-1">जो चोर की संगति करता है वह अपने प्राण का बैरी होता है;</span><br /><span className="pl pl-1">शपथ खाने पर भी वह बात को प्रगट नहीं करता।</span><br /></span>
<span className="v" id="v25" data-v="25" data-xref="भजन संहिता 118:8; लूका 12:4; नीतिवचन 18:10"> <span className="pl pl-1">मनुष्य का भय खाना फंदा हो जाता है,</span><br /><span className="pl pl-1">परन्तु जो यहोवा पर भरोसा रखता है उसका स्थान ऊँचा किया जाएगा।</span><br /></span>
<span className="v" id="v26" data-v="26" data-xref="नीतिवचन 19:6; नीतिवचन 19:21; भजन संहिता 20:9"> <span className="pl pl-1">हाकिम से भेंट करना बहुत लोग चाहते हैं,</span><br /><span className="pl pl-1">परन्तु मनुष्य का न्याय यहोवा ही करता है29:26 मनुष्य का न्याय यहोवा ही करता है: प्रशासकों पर भरोसा करना रेत पर घर बनाना है। सब गलतियों को सुधारने का सही निर्णय यहोवा ही से प्राप्त होता है।।</span><br /></span>
<span className="v" id="v27" data-v="27" data-xref="1 यूहन्ना 3:13; यूहन्ना 15:17–19; भजन संहिता 139:21"> <span className="pl pl-1">धर्मी लोग कुटिल मनुष्य से घृणा करते हैं</span><br /><span className="pl pl-1">और दुष्ट जन भी सीधी चाल चलनेवाले से घृणा करता है।</span></span>
