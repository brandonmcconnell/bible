<span className="v" id="v2" data-v="2" data-xref="2 कुरिन्थियों 10:18; नीतिवचन 25:27; 2 कुरिन्थियों 10:12"> <span className="pl pl-1">तेरी प्रशंसा और लोग करें तो करें, परन्तु तू आप न करना;</span><br /><span className="pl pl-1">दूसरा तुझे सराहे तो सराहे, परन्तु तू अपनी सराहना न करना।</span><br /></span>
<span className="v" id="v3" data-v="3" data-xref="दानिय्येल 3:19; 1 यूहन्ना 3:12; उत्पत्ति 49:7"> <span className="pl pl-1">पत्थर तो भारी है और रेत में बोझ है,</span><br /><span className="pl pl-1">परन्तु मूर्ख का क्रोध, उन दोनों से भी भारी है।</span><br /></span>
<span className="v" id="v4" data-v="4" data-xref="नीतिवचन 6:34; नीतिवचन 14:30; याकूब 3:14–16"> <span className="pl pl-1">क्रोध की क्रूरता और प्रकोप की बाढ़,</span><br /><span className="pl pl-1">परन्तु ईर्ष्या के सामने कौन ठहर सकता है?</span><br /></span>
<span className="v" id="v5" data-v="5" data-xref="नीतिवचन 28:23; 1 तीमुथियुस 5:20; गलातियों 2:14"> <span className="pl pl-1">खुली हुई डाँट गुप्त प्रेम से उत्तम है।</span><br /></span>
<span className="v" id="v6" data-v="6" data-xref="भजन संहिता 141:5; प्रकाशितवाक्य 3:19; अय्यूब 5:17–18"> <span className="pl pl-1">जो घाव मित्र के हाथ से लगें वह विश्वासयोग्य हैं</span><br /><span className="pl pl-1">परन्तु बैरी अधिक चुम्बन करता है।</span><br /></span>
<span className="v" id="v7" data-v="7" data-xref="लूका 15:16–17; गिनती 11:18–20; गिनती 21:5"> <span className="pl pl-1">सन्तुष्ट होने पर मधु का छत्ता भी फीका लगता है,</span><br /><span className="pl pl-1">परन्तु भूखे को सब कड़वी वस्तुएँ भी मीठी जान पड़ती हैं।</span><br /></span>
<span className="v" id="v8" data-v="8" data-xref="नीतिवचन 21:16; यशायाह 16:2; योना 1:10–17"> <span className="pl pl-1">स्थान छोड़कर घूमनेवाला मनुष्य उस चिड़िया के समान है,</span><br /><span className="pl pl-1">जो घोंसला छोड़कर उड़ती फिरती है।</span><br /></span>
<span className="v" id="v9" data-v="9" data-xref="नीतिवचन 16:23–24; नीतिवचन 16:21; भजन संहिता 23:5"> <span className="pl pl-1">जैसे तेल और सुगन्ध से,</span><br /><span className="pl pl-1">वैसे ही मित्र के हृदय की मनोहर सम्मति से मन आनन्दित होता है।</span><br /></span>
<span className="v" id="v10" data-v="10" data-xref="नीतिवचन 18:24; नीतिवचन 17:17; 1 राजाओं 12:6–8"> <span className="pl pl-1">जो तेरा और तेरे पिता का भी मित्र हो उसे न छोड़ना;</span><br /><span className="pl pl-1">और अपनी विपत्ति के दिन, अपने भाई के घर न जाना।</span><br /><span className="pl pl-1">प्रेम करनेवाला पड़ोसी, दूर रहनेवाले भाई से कहीं उत्तम है27:10 प्रेम करनेवाला पड़ोसी, दूर रहनेवाले भाई से कहीं उत्तम है: वास्तव में, मन और आत्मा के निकट रहनेवाला पड़ोसी उससे बेहतर है जो रिश्ते में भाई तो है परन्तु भावनाओं में दूर है।।</span><br /></span>
<span className="v" id="v11" data-v="11" data-xref="नीतिवचन 10:1; भजन संहिता 119:42; नीतिवचन 23:15–16"> <span className="pl pl-1">हे मेरे पुत्र, बुद्धिमान होकर27:11 हे मेरे पुत्र, बुद्धिमान होकर: अपने सच्चे शिष्य के लिए शिक्षक का वचन, वह उससे याचना करता है कि विद्वान की खराई उसके गुरू के चरित्र या शिक्षाओं पर किए गए कटाक्षों का सबसे सच्चा उत्तर होगा। मेरा मन आनन्दित कर,</span><br /><span className="pl pl-1">तब मैं अपने निन्दा करनेवाले को उत्तर दे सकूँगा।</span><br /></span>
<span className="v" id="v12" data-v="12" data-xref="नीतिवचन 22:3; नीतिवचन 18:10; इब्रानियों 11:7"> <span className="pl pl-1">बुद्धिमान मनुष्य विपत्ति को आती देखकर छिप जाता है;</span><br /><span className="pl pl-1">परन्तु भोले लोग आगे बढ़े चले जाते और हानि उठाते हैं।</span><br /></span>
<span className="v" id="v13" data-v="13" data-xref="नीतिवचन 20:16; नीतिवचन 6:1–4; नीतिवचन 22:26–27"> <span className="pl pl-1">जो पराए का उत्तरदायी हो उसका कपड़ा,</span><br /><span className="pl pl-1">और जो अनजान का उत्तरदायी हो उससे बन्धक की वस्तु ले ले।</span><br /></span>
<span className="v" id="v14" data-v="14" data-xref="2 शमूएल 16:16–19; प्रेरितों के काम 12:22–23; यिर्मयाह 28:2–4"> <span className="pl pl-1">जो भोर को उठकर अपने पड़ोसी को ऊँचे शब्द से आशीर्वाद देता है,</span><br /><span className="pl pl-1">उसके लिये यह श्राप गिना जाता है।</span><br /></span>
<span className="v" id="v15" data-v="15" data-xref="नीतिवचन 21:9; नीतिवचन 19:13; नीतिवचन 21:19"> <span className="pl pl-1">झड़ी के दिन पानी का लगातार टपकना,</span><br /><span className="pl pl-1">और झगड़ालू पत्नी दोनों एक से हैं;</span><br /></span>
<span className="v" id="v16" data-v="16" data-xref="यूहन्ना 12:3"> <span className="pl pl-1">जो उसको रोक रखे, वह वायु को भी रोक रखेगा और दाहिने हाथ से वह तेल पकड़ेगा।</span><br /></span>
<span className="v" id="v17" data-v="17" data-xref="इब्रानियों 10:24; नीतिवचन 27:9; अय्यूब 4:3–4"> <span className="pl pl-1">जैसे लोहा लोहे को चमका देता है,</span><br /><span className="pl pl-1">वैसे ही मनुष्य का मुख अपने मित्र की संगति से चमकदार हो जाता है।</span><br /></span>
<span className="v" id="v18" data-v="18" data-xref="1 कुरिन्थियों 9:7; 2 तीमुथियुस 2:6; श्रेष्ठगीत 8:12"> <span className="pl pl-1">जो अंजीर के पेड़ की रक्षा करता है वह उसका फल खाता है,</span><br /><span className="pl pl-1">इसी रीति से जो अपने स्वामी की सेवा करता उसकी महिमा होती है।</span><br /></span>
<span className="v" id="v19" data-v="19" data-xref="याकूब 1:22–25; भजन संहिता 33:15; मरकुस 7:21"> <span className="pl pl-1">जैसे जल में मुख की परछाई मुख को प्रगट करती है,</span><br /><span className="pl pl-1">वैसे ही मनुष्य का मन मनुष्य को प्रगट करती है।</span><br /></span>
<span className="v" id="v20" data-v="20" data-xref="सभोपदेशक 1:8; 1 यूहन्ना 2:16; नीतिवचन 30:15–16"> <span className="pl pl-1">जैसे अधोलोक और विनाशलोक,</span><br /><span className="pl pl-1">वैसे ही मनुष्य की आँखें भी तृप्त नहीं होती।</span><br /></span>
<span className="v" id="v21" data-v="21" data-xref="नीतिवचन 17:3; जकर्याह 13:9; भजन संहिता 66:10"> <span className="pl pl-1">जैसे चाँदी के लिये कुठाली और सोने के लिये भट्ठी हैं,</span><br /><span className="pl pl-1">वैसे ही मनुष्य के लिये उसकी प्रशंसा है।</span><br /></span>
<span className="v" id="v22" data-v="22" data-xref="यिर्मयाह 5:3; नीतिवचन 23:35; यशायाह 1:5"> <span className="pl pl-1">चाहे तू मूर्ख को अनाज के बीच ओखली में डालकर मूसल से कूटे,</span><br /><span className="pl pl-1">तो भी उसकी मूर्खता नहीं जाने की।</span><br /></span>
<span className="v" id="v23" data-v="23" data-xref="1 पतरस 5:2; यूहन्ना 21:15–17; 2 इतिहास 26:10"> <span className="pl pl-1">अपनी भेड़-बकरियों की दशा भली भाँति मन लगाकर जान ले,</span><br /><span className="pl pl-1">और अपने सब पशुओं के झुण्डों की देख-भाल उचित रीति से कर;</span><br /></span>
<span className="v" id="v24" data-v="24" data-xref="नीतिवचन 23:5; 1 तीमुथियुस 6:17–18; सपन्याह 1:18"> <span className="pl pl-1">क्योंकि सम्पत्ति सदा नहीं ठहरती;</span><br /><span className="pl pl-1">और क्या राजमुकुट पीढ़ी से पीढ़ी तक बना रहता है?</span><br /></span>
<span className="v" id="v25" data-v="25" data-xref="नीतिवचन 10:5; भजन संहिता 104:14"> <span className="pl pl-1">कटी हुई घास उठा ली जाती और नई घास दिखाई देती है</span><br /><span className="pl pl-1">और पहाड़ों की हरियाली काटकर इकट्ठी की जाती है;</span><br /></span>
<span className="v" id="v26" data-v="26" data-xref="अय्यूब 31:20"> <span className="pl pl-1">तब भेड़ों के बच्चे तेरे वस्त्र के लिये होंगे,</span><br /><span className="pl pl-1">और बकरों के द्वारा खेत का मूल्य दिया जाएगा;</span><br /></span>
<span className="v" id="v27" data-v="27" data-xref="मत्ती 6:33; नीतिवचन 30:8–9"> <span className="pl pl-1">और बकरियों का इतना दूध होगा कि तू अपने घराने समेत पेट भरकर पिया करेगा,</span><br /><span className="pl pl-1">और तेरी दासियों का भी जीवन निर्वाह होता रहेगा।</span></span>
