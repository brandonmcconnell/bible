<div className="bp poetry">

</div>
<div className="bp poetry">
<span className="v" id="v2" data-v="2" data-xref="1 राजा 15:28; 1 राजा 16:8–29; 2 राजा 15:8–31"> <span className="pl pl-1">राष्ट्र में अराजकता फैलने पर अनेक शासक उठ खड़े होते हैं,</span><br /><span className="pl pl-2">किंतु बुद्धिमान शासक के शासन में स्थायी सुव्यवस्था बनी रहती है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v3" data-v="3" data-xref="मत्तियाह 18:28–30"> <span className="pl pl-1">वह शासक, जो निर्धनों को उत्पीड़ित करता है,</span><br /><span className="pl pl-2">ऐसी घनघोर वृष्टि-समान है, जो समस्त उपज को नष्ट कर जाती है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v4" data-v="4" data-xref="रोमियों 1:32; इफ़ेसॉस 5:11; मत्तियाह 14:4"> <span className="pl pl-1">कानून को नहीं मानने वाला व्यक्ति दुर्जनों की प्रशंसा करते नहीं थकते,</span><br /><span className="pl pl-2">किंतु वे, जो सामाजिक सुव्यवस्था का निर्वाह करते हैं, ऐसों का प्रतिरोध करते हैं.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v5" data-v="5" data-xref="स्तोत्र 92:6; याकोब 1:5; 1 योहन 2:27"> <span className="pl pl-1">दुष्ट लोग न्याय का मूल्य नहीं समझ सकते,</span><br /><span className="pl pl-2">किंतु याहवेह के अभिलाषी इसे उत्तम रीति से पहचानते हैं.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v6" data-v="6" data-xref="सूक्ति संग्रह 19:1; सूक्ति संग्रह 28:18; लूकॉस 16:19–23"> <span className="pl pl-1">खराई का चलनेवाला निर्धन उस धनी से कहीं उत्तम है</span><br /><span className="pl pl-2">जिसकी जीवनशैली कुटिल है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v7" data-v="7" data-xref="1 पेतरॉस 4:3–4; सूक्ति संग्रह 29:3; सूक्ति संग्रह 3:1–35"> <span className="pl pl-1">नियमों का पालन करता है बुद्धिमान संतान,</span><br /><span className="pl pl-2">किंतु पेटू का साथी अपने पिता को लज्जा लाता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v8" data-v="8" data-xref="अय्योब 27:16–17; सूक्ति संग्रह 13:22; उद्बोधक 2:26"> <span className="pl pl-1">जो कोई अपनी संपत्ति की वृद्धि अतिशय ब्याज लेकर करता है,</span><br /><span className="pl pl-2">वह इसे उस व्यक्ति के लिए संचित कर रहा होता है, जो निर्धनों को उदारतापूर्वक देता रहता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v9" data-v="9" data-xref="स्तोत्र 66:18; 2 तिमोथियॉस 4:3–4; ज़करयाह 7:11–13"> <span className="pl pl-1">जो व्यक्ति नियम-व्यवस्था का परित्याग करता है,</span><br /><span className="pl pl-2">उसकी प्रार्थना भी परमेश्वर के लिए घृणित हो जाती है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v10" data-v="10" data-xref="सूक्ति संग्रह 26:27; स्तोत्र 37:25–26; गलातिया 2:4"> <span className="pl pl-1">जो कोई किसी धर्मी को भटका कर विसंगत चालचलन के लिए उकसाता है</span><br /><span className="pl pl-2">वह अपने ही जाल में फंस जाएगा,</span><br /><span className="pl pl-2">किंतु खरे व्यक्ति का प्रतिफल सुखद होता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v11" data-v="11" data-xref="सूक्ति संग्रह 18:17; उद्बोधक 9:15–17; सूक्ति संग्रह 26:16"> <span className="pl pl-1">अपने ही विचार में धनाढ्य स्वयं को बुद्धिमान मानता है;</span><br /><span className="pl pl-2">जो गरीब और समझदार है, वह देखता है कि धनवान कितना भ्रमित है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v12" data-v="12" data-xref="सूक्ति संग्रह 29:2; सूक्ति संग्रह 11:10; सूक्ति संग्रह 28:28"> <span className="pl pl-1">धर्मी व्यक्ति की विजय पर अतिशय आनंद मनाया जाता है;</span><br /><span className="pl pl-2">किंतु जब दुष्ट उन्‍नत होने लगते हैं, प्रजा छिप जाती है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v13" data-v="13" data-xref="1 योहन 1:8–10; स्तोत्र 32:3–5; अय्योब 31:33"> <span className="pl pl-1">जो अपने अपराध को छिपाए रखता है, वह समृद्ध नहीं हो पाता,</span><br /><span className="pl pl-2">किंतु वह, जो अपराध स्वीकार कर उनका परित्याग कर देता है, उस पर कृपा की जाएगी.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v14" data-v="14" data-xref="सूक्ति संग्रह 23:17; स्तोत्र 2:11; यशायाह 66:2"> <span className="pl pl-1">धन्य होता है वह व्यक्ति जिसके हृदय में याहवेह के प्रति श्रद्धा सर्वदा रहती है,</span><br /><span className="pl pl-2">किंतु जो अपने हृदय को कठोर बनाए रखता है, विपदा में जा पड़ता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v15" data-v="15" data-xref="1 पेतरॉस 5:8; मत्तियाह 2:16; सूक्ति संग्रह 19:12"> <span className="pl pl-1">निर्धनों के प्रति दुष्ट शासक का व्यवहार वैसा ही होता है</span><br /><span className="pl pl-2">जैसा दहाड़ते हुए सिंह अथवा आक्रामक रीछ का.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v16" data-v="16" data-xref="निर्गमन 18:21; येरेमियाह 22:15–17; उद्बोधक 4:1"> <span className="pl pl-1">एक शासक जो समझदार नहीं, अपनी प्रजा को उत्पीड़ित करता है,</span><br /><span className="pl pl-2">किंतु वह, जिसे अनुचित अप्रिय है, आयुष्मान होता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v17" data-v="17" data-xref="उत्पत्ति 9:6; 1 राजा 21:19; निर्गमन 21:14"> <span className="pl pl-1">यदि किसी की अंतरात्मा पर मनुष्य हत्या का बोझ है</span><br /><span className="pl pl-2">वह मृत्युपर्यंत छिपता और भागता रहेगा;</span><br /><span className="pl pl-2">यह उपयुक्त नहीं कि कोई उसकी सहायता करे.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v18" data-v="18" data-xref="सूक्ति संग्रह 10:9; सूक्ति संग्रह 28:6; प्रकाशन 3:3"> <span className="pl pl-1">जिसका चालचलन खराईपूर्ण है, वह विपत्तियों से बचा रहेगा,</span><br /><span className="pl pl-2">किंतु जिसके चालचलन में कुटिलता है, शीघ्र ही पतन के गर्त में जा गिरेगा.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v19" data-v="19" data-xref="सूक्ति संग्रह 12:11; प्रशासक 9:4; सूक्ति संग्रह 27:23–27"> <span className="pl pl-1">जो किसान अपनी भूमि की जुताई-गुड़ाई करता रहता है, उसे भोजन का अभाव नहीं होता,</span><br /><span className="pl pl-2">किंतु जो व्यर्थ कार्यों में समय नष्ट करता है, निर्बुद्धि प्रमाणित होता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v20" data-v="20" data-xref="लूकॉस 16:10–12; सूक्ति संग्रह 13:11; प्रकाशन 2:10"> <span className="pl pl-1">खरे व्यक्ति को प्रचुरता में आशीषें प्राप्‍त होती रहती है,</span><br /><span className="pl pl-2">किंतु जो शीघ्र ही धनाढ्य होने की धुन में रहता है, वह दंड से बच न सकेगा.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v21" data-v="21" data-xref="सूक्ति संग्रह 24:23; सूक्ति संग्रह 18:5; यहेजकेल 13:19"> <span className="pl pl-1">पक्षपात भयावह होता है.</span><br /><span className="pl pl-2">फिर भी यह संभव है कि मनुष्य मात्र रोटी के एक टुकड़े को प्राप्‍त करने के लिए अपराध कर बैठे.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v22" data-v="22" data-xref="सूक्ति संग्रह 23:6; 1 तिमोथियॉस 6:9; सूक्ति संग्रह 28:20"> <span className="pl pl-1">कंजूस व्यक्ति को धनाढ्य हो जाने की उतावली होती है,</span><br /><span className="pl pl-2">जबकि उन्हें यह अन्देशा ही नहीं होता, कि उसका निर्धन होना निर्धारित है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v23" data-v="23" data-xref="सूक्ति संग्रह 27:5–6; मत्तियाह 18:15; स्तोत्र 141:5"> <span className="pl pl-1">अंततः कृपापात्र वही बन जाएगा, जो किसी को किसी भूल के लिए डांटता है,</span><br /><span className="pl pl-2">वह नहीं, जो चापलूसी करता रहता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v24" data-v="24" data-xref="मत्तियाह 15:4–6; सूक्ति संग्रह 18:9; सूक्ति संग्रह 19:26"> <span className="pl pl-1">जो अपने माता-पिता से संपत्ति छीनकर</span><br /><span className="pl pl-2">यह कहता है, “इसमें मैंने कुछ भी अनुचित नहीं किया है,”</span><br /><span className="pl pl-2">लुटेरों का सहयोगी होता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v25" data-v="25" data-xref="सूक्ति संग्रह 11:25; 1 तिमोथियॉस 6:6; सूक्ति संग्रह 13:10"> <span className="pl pl-1">लोभी व्यक्ति कलह उत्पन्‍न करा देता है,</span><br /><span className="pl pl-2">किंतु समृद्ध वह हो जाता है, जिसने याहवेह पर भरोसा रखा है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v26" data-v="26" data-xref="सूक्ति संग्रह 3:5; येरेमियाह 17:9; अय्योब 28:28"> <span className="pl pl-1">मूर्ख होता है वह, जो मात्र अपनी ही बुद्धि पर भरोसा रखता है,</span><br /><span className="pl pl-2">किंतु सुरक्षित वह बना रहता है, जो अपने निर्णय विद्वत्ता में लेता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v27" data-v="27" data-xref="सूक्ति संग्रह 19:17; व्यवस्था 15:7; स्तोत्र 41:1–3"> <span className="pl pl-1">जो निर्धनों को उदारतापूर्वक दान देता है, उसे अभाव कभी नहीं होता,</span><br /><span className="pl pl-2">किंतु वह, जो दान करने से कतराता है अनेक ओर से शापित हो जाता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v28" data-v="28" data-xref="सूक्ति संग्रह 29:2; अय्योब 24:4; प्रेरितों 12:23–24"> <span className="pl pl-1">दुष्टों का उत्थान लोगों को छिपने के लिए विवश कर देता है;</span><br /><span className="pl pl-2">किंतु दुष्ट नष्ट हो जाते हैं, खरे की वृद्धि होने लगती है.</span></span>
</div>
