<span className="v" id="v2" data-v="2" data-xref="व्यवस्था 29:29; रोमियों 11:33–34; अय्योब 11:7–8"> <span className="pl pl-1">परमेश्वर की महिमा इसमें है कि वह किसी विषय को गुप्‍त रख देते हैं;</span><br /><span className="pl pl-2">जबकि राजा की महिमा किसी विषय की गहराई तक खोजने में होती है.</span></span>
<span className="v" id="v3" data-v="3" data-xref="यशायाह 55:9; स्तोत्र 103:11; 1 राजा 4:29"> <span className="pl pl-1">जैसे आकाश की ऊंचाई और पृथ्वी की गहराई,</span><br /><span className="pl pl-2">उसी प्रकार राजाओं का हृदय भी रहस्यमय होता है.</span></span>

<div className="bp poetry">
<span className="v" id="v4" data-v="4" data-xref="2 तिमोथियॉस 2:20–21; यहेजकेल 22:18; सूक्ति संग्रह 17:3"> <span className="pl pl-1">चांदी में से खोट दूर कर दो,</span><br /><span className="pl pl-2">तो चांदीकार के लिए शुद्ध चांदी शेष रह जाती है.</span></span>
<span className="v" id="v5" data-v="5" data-xref="सूक्ति संग्रह 16:12; सूक्ति संग्रह 20:8; स्तोत्र 101:7–8"> <span className="pl pl-1">राजा के सामने से दुष्टों को हटा दो,</span><br /><span className="pl pl-2">तो राज सिंहासन धर्म में प्रतिष्ठित हो जाएगा.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v6" data-v="6" data-xref="सूक्ति संग्रह 27:2; सूक्ति संग्रह 25:27; सूक्ति संग्रह 16:19"> <span className="pl pl-1">न तो राजा के समक्ष स्वयं को सम्मान्य प्रमाणित करो,</span><br /><span className="pl pl-2">और न ही किसी प्रतिष्ठित व्यक्ति का स्थान लेने का प्रयास करो;</span></span>
<span className="v" id="v7" data-v="7" data-xref="लूकॉस 14:7–11; लूकॉस 18:14; 1 पेतरॉस 5:5"> <span className="pl pl-1">क्योंकि उत्तम तो यह होगा कि राजा ही तुम्हें आमंत्रित कर यह कहे, “यहां मेरे पास आओ,”</span><br /><span className="pl pl-2">इसकी अपेक्षा कि तुम्हें सब की दृष्टि में निम्नतर स्थान पर जाने का आदेश दिया जाए.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v8" data-v="8" data-xref="मत्तियाह 5:25; सूक्ति संग्रह 17:14; सूक्ति संग्रह 18:6"> <span className="pl pl-1">मात्र इसलिये कि तुमने कुछ देख लिया है,</span><br /><span className="pl pl-2">मुकदमा चलाने की उतावली न करना.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v9" data-v="9" data-xref="मत्तियाह 18:5–17; सूक्ति संग्रह 11:13; सूक्ति संग्रह 20:19"> <span className="pl pl-1">विवादास्पद विषय पर सीधा उसी व्यक्ति से विचार-विमर्श कर लो,</span><br /><span className="pl pl-2">और किसी अन्य व्यक्ति का रहस्य प्रकाशित न करना,</span></span>
<span className="v" id="v10" data-v="10" data-xref="स्तोत्र 119:39"> <span className="pl pl-1">कहीं ऐसा न हो कि कोई इसे सुन ले, यह तुम्हारे ही लिए लज्जा का कारण हो जाए</span><br /><span className="pl pl-2">और तुम्हारी प्रतिष्ठा स्थायी रूप से नष्ट हो जाए.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v11" data-v="11" data-xref="सूक्ति संग्रह 15:23; यशायाह 50:4; उद्बोधक 12:10"> <span className="pl pl-1">उचित अवसर पर कहा हुआ वचन चांदी के पात्र में</span><br /><span className="pl pl-2">प्रस्तुत स्वर्ण के सेब के समान होता है.</span></span>
<span className="v" id="v12" data-v="12" data-xref="स्तोत्र 141:5; सूक्ति संग्रह 15:31–32; सूक्ति संग्रह 15:5"> <span className="pl pl-1">तत्पर श्रोता के लिए ज्ञानवान व्यक्ति की चेतावनी वैसी ही होती है</span><br /><span className="pl pl-2">जैसे स्वर्ण कर्णफूल अथवा स्वर्ण आभूषण.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v13" data-v="13" data-xref="सूक्ति संग्रह 13:17; सूक्ति संग्रह 25:25; फ़िलिप्पॉय 2:25–30"> <span className="pl pl-1">कटनी के समय की उष्णता में ठंडे पानी के पेय के समान होता है,</span><br /><span className="pl pl-2">प्रेषक के लिए वह दूत, जो विश्वासयोग्य है;</span><br /><span className="pl pl-2">वह अपने स्वामी के हृदय को प्रफुल्लित कर देता है.</span></span>
<span className="v" id="v14" data-v="14" data-xref="सूक्ति संग्रह 20:6; लूकॉस 14:11; 2 पेतरॉस 2:15–19"> <span className="pl pl-1">बारिश के बिना बादलों और हवा की तरह है जो व्यक्ति उपहार तो देता नहीं,</span><br /><span className="pl pl-2">किंतु सबके समक्ष देने की डींग मारता रहता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v15" data-v="15" data-xref="सूक्ति संग्रह 15:1; उद्बोधक 10:4; सूक्ति संग्रह 16:14"> <span className="pl pl-1">धैर्य के द्वारा शासक को भी मनाया जा सकता है,</span><br /><span className="pl pl-2">और कोमलता में कहे गए वचन से हड्डी को भी तोड़ा जा सकता है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v16" data-v="16" data-xref="सूक्ति संग्रह 25:27; इफ़ेसॉस 5:18; सूक्ति संग्रह 24:13–14"> <span className="pl pl-1">यदि तुम्हें कहीं मधु प्राप्‍त हो जाए, तो उतना ही खाना, जितना पर्याप्‍त है,</span><br /><span className="pl pl-2">सीमा से अधिक खाओगे तो, तुम उसे उगल दोगे.</span></span>

</div>
<div className="bp poetry">
<span className="v" id="v18" data-v="18" data-xref="स्तोत्र 57:4; सूक्ति संग्रह 12:18; याकोब 3:6"> <span className="pl pl-1">वह व्यक्ति, जो अपने पड़ोसी के विरुद्ध झूठा साक्षी हो जाता है,</span><br /><span className="pl pl-2">वह युद्ध के लिए प्रयुक्त लाठी, तलवार अथवा बाण के समान है.</span></span>
<span className="v" id="v19" data-v="19" data-xref="यशायाह 36:6; यहेजकेल 29:6–7; अय्योब 6:14–20"> <span className="pl pl-1">विपदा के अवसर पर ऐसे व्यक्ति पर भरोसा रखना, जिस पर विश्वास नहीं किया जा सकता,</span><br /><span className="pl pl-2">वैसा ही होता है, जैसे सड़े दांत अथवा टूटे पैर पर भरोसा रखना.</span></span>
<span className="v" id="v20" data-v="20" data-xref="रोमियों 12:15; उद्बोधक 3:4; यशायाह 58:7"> <span className="pl pl-1">दुःख में डूबे व्यक्ति के समक्ष हर्ष गीत गाने का वैसा ही प्रभाव होता है,</span><br /><span className="pl pl-2">जैसा शीतकाल में किसी को विवस्त्र कर देना</span><br /><span className="pl pl-2">अथवा किसी के घावों पर सिरका मल देना.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v21" data-v="21" data-xref="रोमियों 12:20–21; मत्तियाह 5:44; निर्गमन 23:4–5"> <span className="pl pl-1">यदि तुम्हारा विरोधी भूखा है, उसे भोजन कराओ,</span><br /><span className="pl pl-2">यदि प्यासा है, उसे पीने के लिए जल दो;</span></span>
<span className="v" id="v22" data-v="22" data-xref="2 शमुएल 16:12; मत्तियाह 10:13; 1 कोरिंथ 15:18"> <span className="pl pl-1">इससे तुम उसके सिर पर प्रज्वलित कोयलों का ढेर लगा दोगे,</span><br /><span className="pl pl-2">और तुम्हें याहवेह की ओर से पारितोषिक प्राप्‍त होगा.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v23" data-v="23" data-xref="अय्योब 37:22; स्तोत्र 15:3; रोमियों 1:30"> <span className="pl pl-1">जैसे उत्तरी वायु प्रवाह वृष्टि का उत्पादक होता है,</span><br /><span className="pl pl-2">वैसे ही पीठ पीछे पर निंदा करती जीभ शीघ्र क्रोधी मुद्रा उत्पन्‍न करती है.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v24" data-v="24" data-xref="सूक्ति संग्रह 21:19; सूक्ति संग्रह 21:9; सूक्ति संग्रह 27:15–16"> <span className="pl pl-1">विवादी पत्नी के साथ घर में निवास करने से कहीं अधिक श्रेष्ठ है</span><br /><span className="pl pl-2">छत के एक कोने में रह लेना.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v25" data-v="25" data-xref="सूक्ति संग्रह 15:30; रोमियों 10:15; स्तोत्र 63:1"> <span className="pl pl-1">दूर देश से आया शुभ संदेश वैसा ही होता है,</span><br /><span className="pl pl-2">जैसा प्यासी आत्मा को दिया गया शीतल जल.</span></span>
<span className="v" id="v26" data-v="26" data-xref="यहेजकेल 34:18–19; 1 शमुएल 22:14–18; 1 थेस्सलोनि 2:15"> <span className="pl pl-1">वह धर्मी व्यक्ति, जो दुष्टों के आगे झुक जाता है,</span><br /><span className="pl pl-2">गंदले सोते तथा दूषित कुओं-समान होता है.</span></span>

</div>
<div className="bp poetry">
<span className="v" id="v28" data-v="28" data-xref="सूक्ति संग्रह 16:32; सूक्ति संग्रह 22:24; 2 इतिहास 32:5"> <span className="pl pl-1">वह व्यक्ति, जिसका स्वयं पर कोई नियंत्रण नहीं है, वैसा ही है,</span><br /><span className="pl pl-2">जैसा वह नगर, जिसकी सुरक्षा के लिए कोई दीवार नहीं है.</span></span>
