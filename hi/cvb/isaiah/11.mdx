<div className="bt">
<div className="bp poetry">

<span className="v" id="v2" data-v="2" data-xref="यशायाह 61:1; योहन 16:13; इफ़ेसॉस 1:17–18"> <span className="pl pl-1">याहवेह का आत्मा,</span><br /><span className="pl pl-2">बुद्धि और समझ का आत्मा,</span><br /><span className="pl pl-2">युक्ति और सामर्थ्य का आत्मा,</span><br /><span className="pl pl-2">ज्ञान और समझ की आत्मा—</span></span>
<span className="v" id="v3" data-v="3" data-xref="योहन 7:24; योहन 8:15–16; लूकॉस 2:52"> <span className="pl pl-1">उनकी खुशी याहवेह के प्रति ज्यादा होगी.</span><br /><span className="pl pl-1">वे मुंह देखकर न्याय नहीं करेंगे,</span><br /><span className="pl pl-2">न सुनकर करेंगे;</span></span>
<span className="v" id="v4" data-v="4" data-xref="2 थेस्सलोनि 2:8; प्रकाशन 19:15; मलाकी 4:6"> <span className="pl pl-1">वे तो कंगालों का न्याय धर्म से,</span><br /><span className="pl pl-2">और पृथ्वी के नम्र लोगों का न्याय सच्चाई से करेंगे.</span><br /><span className="pl pl-1">वे अपने मुंह के शब्द से पृथ्वी पर हमला करेंगे;</span><br /><span className="pl pl-2">और अपनी फूंक से दुष्टों का नाश कर देंगे.</span></span>
<span className="v" id="v5" data-v="5" data-xref="इफ़ेसॉस 6:14; 1 योहन 1:9; 2 कोरिंथ 6:7"> <span className="pl pl-1">धर्म उनका कटिबंध</span><br /><span className="pl pl-2">और सच्चाई उनकी कमर होगी.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v6" data-v="6" data-xref="यशायाह 65:25; होशेआ 2:18; यहेजकेल 34:25"> <span className="pl pl-1">भेड़िया मेमने के साथ रहेगा,</span><br /><span className="pl pl-2">चीता बकरी के बच्चों के पास लेटेगा,</span><br /><span className="pl pl-1">बछड़ा, सिंह और एक पुष्ट पशु साथ साथ रहेंगे;</span><br /><span className="pl pl-2">और बालक उनको संभालेगा.</span></span>
<span className="v" id="v7" data-v="7"> <span className="pl pl-1">गाय और रीछ मिलकर चरेंगे,</span><br /><span className="pl pl-2">उनके बच्‍चे पास-पास रहेंगे,</span><br /><span className="pl pl-2">और सिंह बैल समान भूसा खाएगा.</span></span>
