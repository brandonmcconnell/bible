<div className="bp poetry">

</div>
<div className="bp poetry">

<div className="bp poetry">
<span className="v" id="v7" data-v="7" data-xref="उद्बोधक 7:11; मत्तियाह 5:45; सूक्ति संग्रह 15:30"> <span className="pl pl-1">उजाला मनभावन होता है,</span><br /><span className="pl pl-2">और आंखों के लिए सूरज सुखदायी.</span></span>

<span className="v" id="v10" data-v="10" data-xref="2 कोरिंथ 7:1; 2 तिमोथियॉस 2:22; उद्बोधक 12:1"> <span className="pl pl-1">अपने हृदय से क्रोध</span><br /><span className="pl pl-2">और अपने शरीर से बुराई करना छोड़ दो क्योंकि बचपन,</span><br /><span className="pl pl-2">और जवानी भी बेकार ही हैं.</span></span>
