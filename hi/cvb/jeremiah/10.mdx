<div className="bp poetry">

</div>
<div className="bp poetry">
<span className="v" id="v8" data-v="8" data-xref="ज़करयाह 10:2; येरेमियाह 10:14; स्तोत्र 115:8"> <span className="pl pl-1">किंतु वे पूर्णतः निर्बुद्धि एवं मूर्ख हैं;</span><br /><span className="pl pl-2">उनकी शिक्षाएं धोखे के सिवा और कुछ नहीं.</span></span>

<span className="v" id="v12" data-v="12" data-xref="उत्पत्ति 1:1; अय्योब 9:8; सूक्ति संग्रह 3:19"> <span className="pl pl-1">याहवेह ही हैं जिन्होंने अपने सामर्थ्य से पृथ्वी की सृष्टि की;</span><br /><span className="pl pl-2">जिन्होंने विश्व को अपनी बुद्धि द्वारा प्रतिष्ठित किया है,</span><br /><span className="pl pl-2">अपनी सूझ-बूझ से उन्होंने आकाश को विस्तीर्ण कर दिया.</span></span>
<span className="v" id="v13" data-v="13" data-xref="स्तोत्र 135:7; अय्योब 38:22; अय्योब 38:34–35"> <span className="pl pl-1">उनके स्वर उच्चारण से आकाश के जल में हलचल मच जाती है;</span><br /><span className="pl pl-2">वही हैं जो चारों ओर से मेघों का आरोहण बनाया करते हैं.</span><br /><span className="pl pl-1">वह वृष्टि के लिए बिजली को अधीन करते हैं</span><br /><span className="pl pl-2">तथा अपने भण्डारगृह से पवन को चलाते हैं.</span></span>
</div>
<div className="bp poetry">
<span className="v" id="v14" data-v="14" data-xref="स्तोत्र 135:16–18; स्तोत्र 115:4–8; यशायाह 46:7–8"> <span className="pl pl-1">हर एक मनुष्य मूर्ख है—ज्ञानहीन;</span><br /><span className="pl pl-2">हर एक स्वर्णशिल्पी अपनी ही कृति प्रतिमा द्वारा लज्जित किया जाता है.</span><br /><span className="pl pl-1">क्योंकि उसके द्वारा ढाली गई प्रतिमाएं धोखा हैं;</span><br /><span className="pl pl-2">उनमें जीवन-श्वास तो है ही नहीं.</span></span>

<span className="v" id="v16" data-v="16" data-xref="व्यवस्था 32:9; येरेमियाह 32:18; येरेमियाह 31:35"> <span className="pl pl-1">याहवेह जो याकोब की निधि हैं इनके सदृश नहीं हैं,</span><br /><span className="pl pl-2">क्योंकि वे सभी के सृष्टिकर्ता हैं,</span><br /><span className="pl pl-1">इस्राएल उन्हीं के इस निज भाग का कुल है—</span><br /><span className="pl pl-2">उनका नाम है सेनाओं का याहवेह.</span></span>

<div className="bp poetry">
