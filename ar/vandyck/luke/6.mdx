## رب السبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اَلتَّثْنِيَة 23:25; إِنْجِيلُ مَرْقُسَ 2:23–28; إِنْجِيلُ مَتَّى 12:1–8"> وَفِي ٱلسَّبْتِ ٱلثَّانِي بَعْدَ ٱلْأَوَّلِ ٱجْتَازَ بَيْنَ ٱلزُّرُوعِ. وَكَانَ تَلَامِيذُهُ يَقْطِفُونَ ٱلسَّنَابِلَ وَيَأْكُلُونَ وَهُمْ يَفْرُكُونَهَا بِأَيْدِيهِمْ.</span>
<span className="v" id="v2" data-v="2" data-xref="إِنْجِيلُ مَتَّى 12:2; إِنْجِيلُ يُوحَنَّا 5:16; إِنْجِيلُ يُوحَنَّا 5:9–11"> فَقَالَ لَهُمْ قَوْمٌ مِنَ ٱلْفَرِّيسِيِّينَ: «لِمَاذَا تَفْعَلُونَ مَا لَا يَحِلُّ فِعْلُهُ فِي ٱلسُّبُوتِ؟».</span>
<span className="v" id="v3" data-v="3" data-xref="صَمُوئِيلَ ٱلْأَوَّلُ 21:3–6; إِنْجِيلُ مَرْقُسَ 2:25–26; إِنْجِيلُ مَتَّى 12:3–5"> فَأَجَابَ يَسُوعُ وَقَالَ لَهُمْ: «أَمَا قَرَأْتُمْ وَلَا هَذَا ٱلَّذِي فَعَلَهُ دَاوُدُ، حِينَ جَاعَ هُوَ وَٱلَّذِينَ كَانُوا مَعَهُ؟</span>
<span className="v" id="v4" data-v="4" data-xref="اَللَّاوِيِّينَ 24:5–9"> كَيْفَ دَخَلَ بَيْتَ ٱللهِ وَأَخَذَ خُبْزَ ٱلتَّقْدِمَةِ وَأَكَلَ، وَأَعْطَى ٱلَّذِينَ مَعَهُ أَيْضًا، ٱلَّذِي لَا يَحِلُّ أَكْلُهُ إِلَّا لِلْكَهَنَةِ فَقَطْ».</span>
<span className="v" id="v5" data-v="5" data-xref="إِنْجِيلُ مَرْقُسَ 2:27; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 1:10; إِنْجِيلُ مَرْقُسَ 9:7"> وَقَالَ لَهُمْ: «إِنَّ ٱبْنَ ٱلْإِنْسَانِ هُوَ رَبُّ ٱلسَّبْتِ أَيْضًا».</span>
</div>

## شفاء في السبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إِنْجِيلُ مَرْقُسَ 3:1–6; إِنْجِيلُ مَتَّى 12:9–14; إِنْجِيلُ لُوقَا 4:31"> وَفِي سَبْتٍ آخَرَ دَخَلَ ٱلْمَجْمَعَ وَصَارَ يُعَلِّمُ. وَكَانَ هُنَاكَ رَجُلٌ يَدُهُ ٱلْيُمْنَى يَابِسَةٌ،</span>
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ لُوقَا 20:20; اَلْمَزَامِيرُ 38:12; إِنْجِيلُ لُوقَا 13:14"> وَكَانَ ٱلْكَتَبَةُ وَٱلْفَرِّيسِيُّونَ يُرَاقِبُونَهُ هَلْ يَشْفِي فِي ٱلسَّبْتِ، لِكَيْ يَجِدُوا عَلَيْهِ شِكَايَةً.</span>
<span className="v" id="v8" data-v="8" data-xref="إِشَعْيَاءَ 42:4; إِنْجِيلُ مَتَّى 9:4; إِنْجِيلُ لُوقَا 5:22"> أَمَّا هُوَ فَعَلِمَ أَفْكَارَهُمْ، وَقَالَ لِلرَّجُلِ ٱلَّذِي يَدُهُ يَابِسَةٌ: «قُمْ وَقِفْ فِي ٱلْوَسْطِ». فَقَامَ وَوَقَفَ.</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ مَرْقُسَ 3:4; إِنْجِيلُ مَتَّى 12:12–13; إِنْجِيلُ لُوقَا 14:3"> ثُمَّ قَالَ لَهُمْ يَسُوعُ: «أَسْأَلُكُمْ شَيْئًا: هَلْ يَحِلُّ فِي ٱلسَّبْتِ فِعْلُ ٱلْخَيْرِ أَوْ فِعْلُ ٱلشَّرِّ؟ تَخْلِيصُ نَفْسٍ أَوْ إِهْلَاكُهَا؟».</span>
<span className="v" id="v10" data-v="10" data-xref="اَلْخُرُوجُ 4:6–7; إِنْجِيلُ مَرْقُسَ 3:5; إِنْجِيلُ يُوحَنَّا 5:8"> ثُمَّ نَظَرَ حَوْلَهُ إِلَى جَمِيعِهِمْ وَقَالَ لِلرَّجُلِ: «مُدَّ يَدَكَ». فَفَعَلَ هَكَذَا. فَعَادَتْ يَدُهُ صَحِيحَةً كَٱلْأُخْرَى.</span>
<span className="v" id="v11" data-v="11" data-xref="اَلْمَزَامِيرُ 2:1–2; أَعْمَالُ ٱلرُّسُلِ 5:33; اَلْجَامِعَةِ 9:3"> فَٱمْتَلَأُوا حُمْقًا وَصَارُوا يَتَكَالَمُونَ فِيمَا بَيْنَهُمْ مَاذَا يَفْعَلُونَ بِيَسُوعَ.</span>
</div>

## اختيار الرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 5:7; إِنْجِيلُ مَرْقُسَ 6:46; إِنْجِيلُ مَرْقُسَ 1:35"> وَفِي تِلْكَ ٱلْأَيَّامِ خَرَجَ إِلَى ٱلْجَبَلِ لِيُصَلِّيَ. وَقَضَى ٱللَّيْلَ كُلَّهُ فِي ٱلصَّلَاةِ لِلهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="إِنْجِيلُ مَرْقُسَ 3:13–19; إِنْجِيلُ مَتَّى 9:36–4; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 21:14"> وَلَمَّا كَانَ ٱلنَّهَارُ دَعَا تَلَامِيذَهُ، وَٱخْتَارَ مِنْهُمُ ٱثْنَيْ عَشَرَ، ٱلَّذِينَ سَمَّاهُمْ أَيْضًا «رُسُلًا»:</span>
<span className="v" id="v14" data-v="14" data-xref="أَعْمَالُ ٱلرُّسُلِ 1:13; إِنْجِيلُ مَتَّى 4:21; إِنْجِيلُ مَتَّى 4:18"> سِمْعَانَ ٱلَّذِي سَمَّاهُ أَيْضًا بُطْرُسَ وَأَنْدَرَاوُسَ أَخَاهُ. يَعْقُوبَ وَيُوحَنَّا. فِيلُبُّسَ وَبَرْثُولَمَاوُسَ.</span>
<span className="v" id="v15" data-v="15" data-xref="إِنْجِيلُ مَتَّى 9:9; إِنْجِيلُ مَرْقُسَ 3:18; أَعْمَالُ ٱلرُّسُلِ 1:13"> مَتَّى وَتُومَا. يَعْقُوبَ بْنَ حَلْفَى وَسِمْعَانَ ٱلَّذِي يُدْعَى ٱلْغَيُورَ.</span>
<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ يُوحَنَّا 14:22; إِنْجِيلُ يُوحَنَّا 6:70–71; إِنْجِيلُ مَرْقُسَ 3:18"> يَهُوذَا أَخَا يَعْقُوبَ، وَيَهُوذَا ٱلْإِسْخَرْيُوطِيَّ ٱلَّذِي صَارَ مُسَلِّمًا أَيْضًا.</span>
</div>
</div>

## بركات وويلات

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="إِنْجِيلُ مَتَّى 11:21; إِنْجِيلُ مَتَّى 4:23–25; اَلْمَزَامِيرُ 107:17–20"> وَنَزَلَ مَعَهُمْ وَوَقَفَ فِي مَوْضِعٍ سَهْلٍ، هُوَ وَجَمْعٌ مِنْ تَلَامِيذِهِ، وَجُمْهُورٌ كَثِيرٌ مِنَ ٱلشَّعْبِ، مِنْ جَمِيعِ ٱلْيَهُودِيَّةِ وَأُورُشَلِيمَ وَسَاحِلِ صُورَ وَصَيْدَاءَ، ٱلَّذِينَ جَاءُوا لِيَسْمَعُوهُ وَيُشْفَوْا مِنْ أَمْرَاضِهِمْ،</span>
<span className="v" id="v18" data-v="18" data-xref="إِنْجِيلُ مَتَّى 17:15; إِنْجِيلُ مَتَّى 15:22; أَعْمَالُ ٱلرُّسُلِ 5:16"> وَٱلْمُعَذَّبُونَ مِنْ أَرْوَاحٍ نَجِسَةٍ. وَكَانُوا يَبْرَأُونَ.</span>
<span className="v" id="v19" data-v="19" data-xref="إِنْجِيلُ مَتَّى 14:36; إِنْجِيلُ مَرْقُسَ 3:10; أَعْمَالُ ٱلرُّسُلِ 5:15–16"> وَكُلُّ ٱلْجَمْعِ طَلَبُوا أَنْ يَلْمِسُوهُ، لِأَنَّ قُوَّةً كَانَتْ تَخْرُجُ مِنْهُ وَتَشْفِي ٱلْجَمِيعَ.</span>
<span className="v" id="v20" data-v="20" data-xref="رِسَالَةُ يَعْقُوبَ 2:5; إِنْجِيلُ لُوقَا 6:20–24; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 8:9"> وَرَفَعَ عَيْنَيْهِ إِلَى تَلَامِيذِهِ وَقَالَ: «طُوبَاكُمْ أَيُّهَا ٱلْمَسَاكِينُ، لِأَنَّ لَكُمْ مَلَكُوتَ ٱللهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ يُوحَنَّا 6:35; إِنْجِيلُ مَتَّى 5:6; رِسَالَةُ يَعْقُوبَ 1:12"> طُوبَاكُمْ أَيُّهَا ٱلْجِيَاعُ ٱلْآنَ، لِأَنَّكُمْ تُشْبَعُونَ. طُوبَاكُمْ أَيُّهَا ٱلْبَاكُونَ ٱلْآنَ، لِأَنَّكُمْ سَتَضْحَكُونَ.</span>
<span className="v" id="v22" data-v="22" data-xref="إِنْجِيلُ مَتَّى 5:10–12; إِنْجِيلُ مَتَّى 10:22; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 4:12–16"> طُوبَاكُمْ إِذَا أَبْغَضَكُمُ ٱلنَّاسُ، وَإِذَا أَفْرَزُوكُمْ وَعَيَّرُوكُمْ، وَأَخْرَجُوا ٱسْمَكُمْ كَشِرِّيرٍ مِنْ أَجْلِ ٱبْنِ ٱلْإِنْسَانِ.</span>
<span className="v" id="v23" data-v="23" data-xref="أَعْمَالُ ٱلرُّسُلِ 5:41; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 21:7; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 3:12"> اِفْرَحُوا فِي ذَلِكَ ٱلْيَوْمِ وَتَهَلَّلُوا، فَهُوَذَا أَجْرُكُمْ عَظِيمٌ فِي ٱلسَّمَاءِ. لِأَنَّ آبَاءَهُمْ هَكَذَا كَانُوا يَفْعَلُونَ بِٱلْأَنْبِيَاءِ.</span>
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ مَتَّى 6:2; رِسَالَةُ يَعْقُوبَ 5:1–6; اَلْمَزَامِيرُ 49:6–7"> وَلَكِنْ وَيْلٌ لَكُمْ أَيُّهَا ٱلْأَغْنِيَاءُ، لِأَنَّكُمْ قَدْ نِلْتُمْ عَزَاءَكُمْ.</span>
<span className="v" id="v25" data-v="25" data-xref="إِشَعْيَاءَ 65:13; رِسَالَةُ يَعْقُوبَ 4:9; أَمْثَالٌ 14:13"> وَيْلٌ لَكُمْ أَيُّهَا ٱلشَّبَاعَى، لِأَنَّكُمْ سَتَجُوعُونَ. وَيْلٌ لَكُمْ أَيُّهَا ٱلضَّاحِكُونَ ٱلْآنَ، لِأَنَّكُمْ سَتَحْزَنُونَ وَتَبْكُونَ.</span>
<span className="v" id="v26" data-v="26" data-xref="إِنْجِيلُ يُوحَنَّا 15:19; رِسَالَةُ يَعْقُوبَ 4:4; إِنْجِيلُ مَتَّى 7:15"> وَيْلٌ لَكُمْ إِذَا قَالَ فِيكُمْ جَمِيعُ ٱلنَّاسِ حَسَنًا. لِأَنَّهُ هَكَذَا كَانَ آبَاؤُهُمْ يَفْعَلُونَ بِٱلْأَنْبِيَاءِ ٱلْكَذَبَةِ.</span>

## محبة الأعداء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ تَسَالُونِيكِي 5:15; إِنْجِيلُ لُوقَا 6:35; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 12:17–21"> «لَكِنِّي أَقُولُ لَكُمْ أَيُّهَا ٱلسَّامِعُونَ: أَحِبُّوا أَعْدَاءَكُمْ، أَحْسِنُوا إِلَى مُبْغِضِيكُمْ،</span>
<span className="v" id="v28" data-v="28" data-xref="رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 3:9; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 4:12; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 12:14"> بَارِكُوا لَاعِنِيكُمْ، وَصَلُّوا لِأَجْلِ ٱلَّذِينَ يُسِيئُونَ إِلَيْكُمْ.</span>
