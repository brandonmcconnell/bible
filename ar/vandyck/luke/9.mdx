## إرسال الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إِنْجِيلُ يُوحَنَّا 14:12; إِنْجِيلُ لُوقَا 10:19; إِنْجِيلُ مَتَّى 16:19"> وَدَعَا تَلَامِيذَهُ ٱلِٱثْنَيْ عَشَرَ، وَأَعْطَاهُمْ قُوَّةً وَسُلْطَانًا عَلَى جَمِيعِ ٱلشَّيَاطِينِ وَشِفَاءِ أَمْرَاضٍ،</span>
<span className="v" id="v2" data-v="2" data-xref="إِنْجِيلُ مَتَّى 10:7–8; إِنْجِيلُ لُوقَا 10:9; إِنْجِيلُ مَتَّى 3:2"> وَأَرْسَلَهُمْ لِيَكْرِزُوا بِمَلَكُوتِ ٱللهِ وَيَشْفُوا ٱلْمَرْضَى.</span>
<span className="v" id="v3" data-v="3" data-xref="إِنْجِيلُ لُوقَا 22:35; اَلْمَزَامِيرُ 37:3; إِنْجِيلُ لُوقَا 10:4–12"> وَقَالَ لَهُمْ: «لَا تَحْمِلُوا شَيْئًا لِلطَّرِيقِ: لَا عَصًا وَلَا مِزْوَدًا وَلَا خُبْزًا وَلَا فِضَّةً، وَلَا يَكُونُ لِلْوَاحِدِ ثَوْبَانِ.</span>
<span className="v" id="v4" data-v="4" data-xref="إِنْجِيلُ مَتَّى 10:11; إِنْجِيلُ مَرْقُسَ 6:10; إِنْجِيلُ لُوقَا 10:5–8"> وَأَيُّ بَيْتٍ دَخَلْتُمُوهُ فَهُنَاكَ أَقِيمُوا، وَمِنْ هُنَاكَ ٱخْرُجُوا.</span>
<span className="v" id="v5" data-v="5" data-xref="أَعْمَالُ ٱلرُّسُلِ 13:51; أَعْمَالُ ٱلرُّسُلِ 18:6; إِنْجِيلُ مَرْقُسَ 6:11"> وَكُلُّ مَنْ لَا يَقْبَلُكُمْ فَٱخْرُجُوا مِنْ تِلْكَ ٱلْمَدِينَةِ، وَٱنْفُضُوا ٱلْغُبَارَ أَيْضًا عَنْ أَرْجُلِكُمْ شَهَادَةً عَلَيْهِمْ».</span>
<span className="v" id="v6" data-v="6" data-xref="أَعْمَالُ ٱلرُّسُلِ 4:30; إِنْجِيلُ مَرْقُسَ 16:20; أَعْمَالُ ٱلرُّسُلِ 5:15"> فَلَمَّا خَرَجُوا كَانُوا يَجْتَازُونَ فِي كُلِّ قَرْيَةٍ يُبَشِّرُونَ وَيَشْفُونَ فِي كُلِّ مَوْضِعٍ.</span>
</div>

## حيرة هيرودس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ مَرْقُسَ 6:14–28; إِنْجِيلُ مَتَّى 14:1–12; إِنْجِيلُ لُوقَا 3:1"> فَسَمِعَ هِيرُودُسُ رَئِيسُ ٱلرُّبْعِ بِجَمِيعِ مَا كَانَ مِنْهُ، وَٱرْتَابَ، لِأَنَّ قَوْمًا كَانُوا يَقُولُونَ: «إِنَّ يُوحَنَّا قَدْ قَامَ مِنَ ٱلْأَمْوَاتِ».</span>
<span className="v" id="v8" data-v="8" data-xref="إِنْجِيلُ يُوحَنَّا 1:21; إِنْجِيلُ لُوقَا 9:19; إِنْجِيلُ مَرْقُسَ 6:15"> وَقَوْمًا: «إِنَّ إِيلِيَّا ظَهَرَ». وَآخَرِينَ: «إِنَّ نَبِيًّا مِنَ ٱلْقُدَمَاءِ قَامَ».</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ لُوقَا 23:8; إِنْجِيلُ لُوقَا 13:31–32"> فَقَالَ هِيرُودُسُ: «يُوحَنَّا أَنَا قَطَعْتُ رَأْسَهُ. فَمَنْ هُوَ هَذَا ٱلَّذِي أَسْمَعُ عَنْهُ مِثْلَ هَذَا؟». وَكَانَ يَطْلُبُ أَنْ يَرَاهُ.</span>
</div>
</div>

## إشباع الخمسة الآلاف رَجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إِنْجِيلُ مَتَّى 11:21; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 13:17; إِنْجِيلُ مَتَّى 14:13–21"> وَلَمَّا رَجَعَ ٱلرُّسُلُ أَخْبَرُوهُ بِجَمِيعِ مَا فَعَلُوا، فَأَخَذَهُمْ وَٱنْصَرَفَ مُنْفَرِدًا إِلَى مَوْضِعٍ خَلَاءٍ لِمَدِينَةٍ تُسَمَّى بَيْتَ صَيْدَا.</span>
<span className="v" id="v11" data-v="11" data-xref="إِنْجِيلُ لُوقَا 9:2; إِنْجِيلُ لُوقَا 5:31; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 4:16"> فَٱلْجُمُوعُ إِذْ عَلِمُوا تَبِعُوهُ، فَقَبِلَهُمْ وَكَلَّمَهُمْ عَنْ مَلَكُوتِ ٱللهِ، وَٱلْمُحْتَاجُونَ إِلَى ٱلشِّفَاءِ شَفَاهُمْ.</span>
<span className="v" id="v12" data-v="12" data-xref="إِنْجِيلُ يُوحَنَّا 6:1; إِنْجِيلُ يُوحَنَّا 6:5–15; هُوشَع 13:5"> فَٱبْتَدَأَ ٱلنَّهَارُ يَمِيلُ. فَتَقَدَّمَ ٱلِٱثْنَا عَشَرَ وَقَالُوا لَهُ: «ٱصْرِفِ ٱلْجَمْعَ لِيَذْهَبُوا إِلَى ٱلْقُرَى وَٱلضِّيَاعِ حَوَالَيْنَا فَيَبِيتُوا وَيَجِدُوا طَعَامًا، لِأَنَّنَا هَهُنَا فِي مَوْضِعٍ خَلَاءٍ».</span>
<span className="v" id="v13" data-v="13" data-xref="أَمْثَالٌ 11:24–25; إِنْجِيلُ يُوحَنَّا 6:5–9; اَلْمُلُوكِ ٱلثَّانِي 4:42–43"> فَقَالَ لَهُمْ: «أَعْطُوهُمْ أَنْتُمْ لِيَأْكُلُوا». فَقَالُوا: «لَيْسَ عِنْدَنَا أَكْثَرُ مِنْ خَمْسَةِ أَرْغِفَةٍ وَسَمَكَتَيْنِ، إِلَّا أَنْ نَذْهَبَ وَنَبْتَاعَ طَعَامًا لِهَذَا ٱلشَّعْبِ كُلِّهِ».</span>
<span className="v" id="v14" data-v="14" data-xref="إِنْجِيلُ مَرْقُسَ 8:6; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 14:40; إِنْجِيلُ مَرْقُسَ 6:39–40"> لِأَنَّهُمْ كَانُوا نَحْوَ خَمْسَةِ آلَافِ رَجُلٍ. فَقَالَ لِتَلَامِيذِهِ: «أَتْكِئُوهُمْ فِرَقًا خَمْسِينَ خَمْسِينَ».</span>
<span className="v" id="v15" data-v="15"> فَفَعَلُوا هَكَذَا، وَأَتْكَأُوا ٱلْجَمِيعَ.</span>
<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ مَتَّى 14:19; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلْأُولَى إِلَى تِيمُوثَاوُسَ 4:4–5; إِنْجِيلُ مَتَّى 15:36"> فَأَخَذَ ٱلْأَرْغِفَةَ ٱلْخَمْسَةَ وَٱلسَّمَكَتَيْنِ، وَرَفَعَ نَظَرَهُ نَحْوَ ٱلسَّمَاءِ وَبَارَكَهُنَّ، ثُمَّ كَسَّرَ وَأَعْطَى ٱلتَّلَامِيذَ لِيُقَدِّمُوا لِلْجَمْعِ.</span>
<span className="v" id="v17" data-v="17" data-xref="اَلْمُلُوكِ ٱلثَّانِي 4:44; إِنْجِيلُ مَتَّى 15:37–38; إِنْجِيلُ يُوحَنَّا 6:11–13"> فَأَكَلُوا وَشَبِعُوا جَمِيعًا. ثُمَّ رُفِعَ مَا فَضَلَ عَنْهُمْ مِنَ ٱلْكِسَرِ ٱثْنَتَا عَشْرَةَ قُفَّةً.</span>
</div>

## اعتراف بطرس بالمسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إِنْجِيلُ مَتَّى 16:13–16; إِنْجِيلُ لُوقَا 11:1; إِنْجِيلُ مَتَّى 26:36"> وَفِيمَا هُوَ يُصَلِّي عَلَى ٱنْفِرَادٍ كَانَ ٱلتَّلَامِيذُ مَعَهُ. فَسَأَلَهُمْ قَائِلًا: «مَنْ تَقُولُ ٱلْجُمُوعُ أَنِّي أَنَا؟».</span>
<span className="v" id="v19" data-v="19" data-xref="مَلَاخِي 4:5; إِنْجِيلُ لُوقَا 9:7–8; إِنْجِيلُ مَتَّى 14:2"> فَأَجَابُوا وَقَالوا: «يُوحَنَّا ٱلْمَعْمَدَانُ. وَآخَرُونَ: إِيلِيَّا. وَآخَرُونَ: إِنَّ نَبِيًّا مِنَ ٱلْقُدَمَاءِ قَامَ».</span>
<span className="v" id="v20" data-v="20" data-xref="إِنْجِيلُ مَتَّى 16:15–17; إِنْجِيلُ يُوحَنَّا 4:29; إِنْجِيلُ يُوحَنَّا 7:41"> فَقَالَ لَهُمْ: «وَأَنْتُمْ، مَنْ تَقُولُونَ أَنِّي أَنَا؟» فَأَجَابَ بُطْرُسُ وَقَالَ: «مَسِيحُ ٱللهِ!».</span>
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ مَتَّى 16:20; إِنْجِيلُ مَتَّى 17:9; إِنْجِيلُ مَرْقُسَ 8:30–31"> فَٱنْتَهَرَهُمْ وَأَوْصَى أَنْ لَا يَقُولُوا ذَلِكَ لِأَحَدٍ،</span>
<span className="v" id="v22" data-v="22" data-xref="إِشَعْيَاءَ 53:1–12; إِنْجِيلُ لُوقَا 24:7; إِنْجِيلُ لُوقَا 18:31–34"> قَائِلًا: «إِنَّهُ يَنْبَغِي أَنَّ ٱبْنَ ٱلْإِنْسَانِ يَتَأَلَّمُ كَثِيرًا، وَيُرْفَضُ مِنَ ٱلشُّيُوخِ وَرُؤَسَاءِ ٱلْكَهَنَةِ وَٱلْكَتَبَةِ، وَيُقْتَلُ، وَفِي ٱلْيَوْمِ ٱلثَّالِثِ يَقُومُ».</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="إِنْجِيلُ مَتَّى 10:38–39; إِنْجِيلُ لُوقَا 14:26–27; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 8:13"> وَقَالَ لِلْجَمِيعِ: «إِنْ أَرَادَ أَحَدٌ أَنْ يَأْتِيَ وَرَائِي، فَلْيُنْكِرْ نَفْسَهُ وَيَحْمِلْ صَلِيبَهُ كُلَّ يَوْمٍ، وَيَتْبَعْنِي.</span>
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ لُوقَا 17:33; إِنْجِيلُ يُوحَنَّا 12:25; أَعْمَالُ ٱلرُّسُلِ 20:23–24"> فَإِنَّ مَنْ أَرَادَ أَنْ يُخَلِّصَ نَفْسَهُ يُهْلِكُهَا، وَمَنْ يُهْلِكُ نَفْسَهُ مِنْ أَجْلِي فَهَذَا يُخَلِّصُهَا.</span>
<span className="v" id="v25" data-v="25" data-xref="إِنْجِيلُ مَرْقُسَ 8:36; إِنْجِيلُ مَتَّى 16:26; اَلْمَزَامِيرُ 49:6–8"> لِأَنَّهُ مَاذَا يَنْتَفِعُ ٱلْإِنْسَانُ لَوْ رَبِحَ ٱلْعَالَمَ كُلَّهُ، وَأَهْلَكَ نَفْسَهُ أَوْ خَسِرَهَا؟</span>
<span className="v" id="v26" data-v="26" data-xref="إِنْجِيلُ مَتَّى 10:32–33; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى تِيمُوثَاوُسَ 2:12; إِنْجِيلُ لُوقَا 12:8–9"> لِأَنَّ مَنِ ٱسْتَحَى بِي وَبِكَلَامِي، فَبِهَذَا يَسْتَحِي ٱبْنُ ٱلْإِنْسَانِ مَتَى جَاءَ بِمَجْدِهِ وَمَجْدِ ٱلْآبِ وَٱلْمَلَائِكَةِ ٱلْقِدِّيسِينَ.</span>
<span className="v" id="v27" data-v="27" data-xref="إِنْجِيلُ مَتَّى 16:28; إِنْجِيلُ مَرْقُسَ 9:1; إِنْجِيلُ يُوحَنَّا 21:22–23"> حَقًّا أَقُولُ لَكُمْ: إِنَّ مِنَ ٱلْقِيَامِ هَهُنَا قَوْمًا لَا يَذُوقُونَ ٱلْمَوْتَ حَتَّى يَرَوْا مَلَكُوتَ ٱللهِ».</span>
</div>

## التجلي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="إِنْجِيلُ مَتَّى 17:1–13; إِنْجِيلُ مَتَّى 26:37–39; إِنْجِيلُ مَرْقُسَ 14:33–36"> وَبَعْدَ هَذَا ٱلْكَلَامِ بِنَحْوِ ثَمَانِيَةِ أَيَّامٍ، أَخَذَ بُطْرُسَ وَيُوحَنَّا وَيَعْقُوبَ وَصَعِدَ إِلَى جَبَلٍ لِيُصَلِّيَ.</span>
<span className="v" id="v29" data-v="29" data-xref="إِنْجِيلُ مَتَّى 17:2; إِنْجِيلُ مَرْقُسَ 9:2–3; اَلْخُرُوجُ 34:29–35"> وَفِيمَا هُوَ يُصَلِّي صَارَتْ هَيْئَةُ وَجْهِهِ مُتَغَيِّرَةً، وَلِبَاسُهُ مُبْيَضًّا لَامِعًا.</span>
