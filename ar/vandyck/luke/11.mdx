## الصلاة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اَلْمَزَامِيرُ 19:14; إِنْجِيلُ لُوقَا 22:39–45; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 8:26–27"> وَإِذْ كَانَ يُصَلِّي فِي مَوْضِعٍ، لَمَّا فَرَغَ، قَالَ وَاحِدٌ مِنْ تَلَامِيذِهِ: «يَارَبُّ، عَلِّمْنَا أَنْ نُصَلِّيَ كَمَا عَلَّمَ يُوحَنَّا أَيْضًا تَلَامِيذَهُ».</span>
<span className="v" id="v2" data-v="2" data-xref="إِنْجِيلُ مَتَّى 6:6–15; اَلْجَامِعَةِ 5:2; اَلْمَزَامِيرُ 72:18–19"> فَقَالَ لَهُمْ: «مَتَى صَلَّيْتُمْ فَقُولُوا: أَبَانَا ٱلَّذِي فِي ٱلسَّمَاوَاتِ، لِيَتَقَدَّسِ ٱسْمُكَ، لِيَأْتِ مَلَكُوتُكَ، لِتَكُنْ مَشِيئَتُكَ كَمَا فِي ٱلسَّمَاءِ كَذَلِكَ عَلَى ٱلْأَرْضِ.</span>
<span className="v" id="v3" data-v="3" data-xref="إِنْجِيلُ مَتَّى 6:11; إِنْجِيلُ مَتَّى 6:34; أَمْثَالٌ 30:8"> خُبْزَنَا كَفَافَنَا أَعْطِنَا كُلَّ يَوْمٍ،</span>
<span className="v" id="v4" data-v="4" data-xref="إِنْجِيلُ مَتَّى 6:12–15; إِنْجِيلُ مَتَّى 26:41; إِشَعْيَاءَ 43:25–26"> وَٱغْفِرْ لَنَا خَطَايَانَا لِأَنَّنَا نَحْنُ أَيْضًا نَغْفِرُ لِكُلِّ مَنْ يُذْنِبُ إِلَيْنَا، وَلَا تُدْخِلْنَا فِي تَجْرِبَةٍ لَكِنْ نَجِّنَا مِنَ ٱلشِّرِّيرِ».</span>
<span className="v" id="v5" data-v="5" data-xref="إِنْجِيلُ لُوقَا 18:1–8"> ثُمَّ قَالَ لَهُمْ: «مَنْ مِنْكُمْ يَكُونُ لَهُ صَدِيقٌ، وَيَمْضِي إِلَيْهِ نِصْفَ ٱللَّيْلِ، وَيَقُولُ لَهُ يَا صَدِيقُ، أَقْرِضْنِي ثَلَاثَةَ أَرْغِفَةٍ،</span>
<span className="v" id="v6" data-v="6"> لِأَنَّ صَدِيقًا لِي جَاءَنِي مِنْ سَفَرٍ، وَلَيْسَ لِي مَا أُقَدِّمُ لَهُ.</span>
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ لُوقَا 13:25; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ غَلَاطِيَّةَ 6:17; إِنْجِيلُ لُوقَا 7:6"> فَيُجِيبَ ذَلِكَ مِنْ دَاخِلٍ وَيَقُولَ: لَا تُزْعِجْنِي! اَلْبَابُ مُغْلَقٌ ٱلْآنَ، وَأَوْلَادِي مَعِي فِي ٱلْفِرَاشِ. لَا أَقْدِرُ أَنْ أَقُومَ وَأُعْطِيَكَ.</span>
<span className="v" id="v8" data-v="8" data-xref="اَلتَّكْوِينُ 32:26; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ كُولُوسِّي 4:12; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 15:30"> أَقُولُ لَكُمْ: وَإِنْ كَانَ لَا يَقُومُ وَيُعْطِيهِ لِكَوْنِهِ صَدِيقَهُ، فَإِنَّهُ مِنْ أَجْلِ لَجَاجَتِهِ يَقُومُ وَيُعْطِيهِ قَدْرَ مَا يَحْتَاجُ.</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ مَرْقُسَ 11:24; إِنْجِيلُ يُوحَنَّا 16:23–24; إِنْجِيلُ مَتَّى 21:22"> وَأَنَا أَقُولُ لَكُمُ: ٱسْأَلُوا تُعْطَوْا، اُطْلُبُوا تَجِدُوا، اِقْرَعُوا يُفْتَحْ لَكُمْ.</span>
<span className="v" id="v10" data-v="10" data-xref="رِسَالَةُ يَعْقُوبَ 4:3; مَرَاثِي إِرْمِيَا 3:54–58; اَلْمَزَامِيرُ 31:22"> لِأَنَّ كُلَّ مَنْ يَسْأَلُ يَأْخُذُ، وَمَنْ يَطْلُبُ يَجِدُ، وَمَنْ يَقْرَعُ يُفْتَحُ لَهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="إِنْجِيلُ مَتَّى 7:9; إِشَعْيَاءَ 49:15"> فَمَنْ مِنْكُمْ، وَهُوَ أَبٌ، يَسْأَلُهُ ٱبْنُهُ خُبْزًا، أَفَيُعْطِيهِ حَجَرًا؟ أَوْ سَمَكَةً، أَفَيُعْطِيهِ حَيَّةً بَدَلَ ٱلسَّمَكَةِ؟</span>
<span className="v" id="v12" data-v="12" data-xref="رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 9:10; إِنْجِيلُ لُوقَا 10:19; حِزْقِيَال 2:6"> أَوْ إِذَا سَأَلَهُ بَيْضَةً، أَفَيُعْطِيهِ عَقْرَبًا؟</span>
<span className="v" id="v13" data-v="13" data-xref="إِنْجِيلُ مَتَّى 7:11; إِشَعْيَاءَ 44:3–4; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 8:32"> فَإِنْ كُنْتُمْ وَأَنْتُمْ أَشْرَارٌ تَعْرِفُونَ أَنْ تُعْطُوا أَوْلَادَكُمْ عَطَايَا جَيِّدَةً، فَكَمْ بِٱلْحَرِيِّ ٱلْآبُ ٱلَّذِي مِنَ ٱلسَّمَاءِ، يُعْطِي ٱلرُّوحَ ٱلْقُدُسَ لِلَّذِينَ يَسْأَلُونَهُ؟».</span>

## يسوع وبَعلزَبول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="إِنْجِيلُ مَتَّى 12:22–24; إِنْجِيلُ مَتَّى 9:32–34; إِنْجِيلُ مَرْقُسَ 7:32–37"> وَكَانَ يُخْرِجُ شَيْطَانًا، وَكَانَ ذَلِكَ أَخْرَسَ. فَلَمَّا أُخْرِجَ ٱلشَّيْطَانُ تَكَلَّمَ ٱلْأَخْرَسُ، فَتَعَجَّبَ ٱلْجُمُوعُ.</span>
<span className="v" id="v15" data-v="15" data-xref="إِنْجِيلُ مَتَّى 9:34; إِنْجِيلُ مَتَّى 12:24–30; إِنْجِيلُ يُوحَنَّا 8:48"> وَأَمَّا قَوْمٌ مِنْهُمْ فَقَالُوا: «بِبَعْلَزَبُولَ رَئِيسِ ٱلشَّيَاطِينِ يُخْرِجُ ٱلشَّيَاطِينَ».</span>
<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ يُوحَنَّا 6:30; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 1:22; إِنْجِيلُ مَرْقُسَ 8:11–12"> وَآخَرُونَ طَلَبُوا مِنْهُ آيَةً مِنَ ٱلسَّمَاءِ يُجَرِّبُونَهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="إِنْجِيلُ مَرْقُسَ 3:23–27; إِنْجِيلُ مَتَّى 12:25–29; إِشَعْيَاءَ 19:2–3"> فَعَلِمَ أَفْكَارَهُمْ، وَقَالَ لَهُمْ: «كُلُّ مَمْلَكَةٍ مُنْقَسِمَةٍ عَلَى ذَاتِهَا تَخْرَبُ، وَبَيْتٍ مُنْقَسِمٍ عَلَى بَيْتٍ يَسْقُطُ.</span>
<span className="v" id="v18" data-v="18" data-xref="إِنْجِيلُ مَتَّى 12:26; إِنْجِيلُ مَتَّى 12:31–34; إِنْجِيلُ لُوقَا 11:15"> فَإِنْ كَانَ ٱلشَّيْطَانُ أَيْضًا يَنْقَسِمُ عَلَى ذَاتِهِ، فَكَيْفَ تَثْبُتُ مَمْلَكَتُهُ؟ لِأَنَّكُمْ تَقُولُونَ: إِنِّي بِبَعْلَزَبُولَ أُخْرِجُ ٱلشَّيَاطِينَ.</span>
<span className="v" id="v19" data-v="19" data-xref="إِنْجِيلُ لُوقَا 9:49; أَيُّوبَ 15:6; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 3:19"> فَإِنْ كُنْتُ أَنَا بِبَعْلَزَبُولَ أُخْرِجُ ٱلشَّيَاطِينَ، فَأَبْنَاؤُكُمْ بِمَنْ يُخْرِجُونَ؟ لِذَلِكَ هُمْ يَكُونُونَ قُضَاتَكُمْ!</span>
<span className="v" id="v20" data-v="20" data-xref="اَلْخُرُوجُ 8:19; إِنْجِيلُ مَتَّى 12:28; إِنْجِيلُ لُوقَا 10:9"> وَلَكِنْ إِنْ كُنْتُ بِأَصْبِعِ ٱللهِ أُخْرِجُ ٱلشَّيَاطِينَ، فَقَدْ أَقْبَلَ عَلَيْكُمْ مَلَكُوتُ ٱللهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ مَرْقُسَ 3:27; إِنْجِيلُ مَتَّى 12:29"> حِينَمَا يَحْفَظُ ٱلْقَوِيُّ دَارَهُ مُتَسَلِّحًا،تَكُونُ أَمْوَالُهُ فِي أَمَانٍ.</span>
<span className="v" id="v22" data-v="22" data-xref="إِشَعْيَاءَ 49:24–25; إِشَعْيَاءَ 53:12; رِسَالَةُ يُوحَنَّا ٱلرَّسُولِ ٱلْأُولَى 3:8"> وَلَكِنْ مَتَى جَاءَ مَنْ هُوَ أَقْوَى مِنْهُ فَإِنَّهُ يَغْلِبُهُ، وَيَنْزِعُ سِلَاحَهُ ٱلْكَامِلَ ٱلَّذِي ٱتَّكَلَ عَلَيْهِ، وَيُوَزِّعُ غَنَائِمَهُ.</span>
<span className="v" id="v23" data-v="23" data-xref="إِنْجِيلُ مَتَّى 12:30; إِنْجِيلُ لُوقَا 9:50; إِنْجِيلُ مَرْقُسَ 9:40"> مَنْ لَيْسَ مَعِي فَهُوَ عَلَيَّ، وَمَنْ لَا يَجْمَعُ مَعِي فَهُوَ يُفَرِّقُ.</span>
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ مَتَّى 12:43–45; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 5:8; أَيُّوبَ 2:2"> مَتَى خَرَجَ ٱلرُّوحُ ٱلنَّجِسُ مِنَ ٱلْإِنْسَانِ، يَجْتَازُ فِي أَمَاكِنَ لَيْسَ فِيهَا مَاءٌ يَطْلُبُ رَاحَةً، وَإِذْ لَا يَجِدُ يَقُولُ: أَرْجِعُ إِلَى بَيْتِي ٱلَّذِي خَرَجْتُ مِنْهُ.</span>
<span className="v" id="v25" data-v="25" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ تَسَالُونِيكِي 2:9–12; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلثَّانِيَةُ 2:10–19; رِسَالَةُ يَهُوذَا 1:8–13"> فَيَأْتِي وَيَجِدُهُ مَكْنُوسًا مُزَيَّنًا.</span>
<span className="v" id="v26" data-v="26" data-xref="إِنْجِيلُ يُوحَنَّا 5:14; إِنْجِيلُ مَتَّى 12:45; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلثَّانِيَةُ 2:20–22"> ثُمَّ يَذْهَبُ وَيَأْخُذُ سَبْعَةَ أَرْوَاحٍ أُخَرَ أَشَرَّ مِنْهُ، فَتَدْخُلُ وَتَسْكُنُ هُنَاكَ، فَتَصِيرُ أَوَاخِرُ ذَلِكَ ٱلْإِنْسَانِ أَشَرَّ مِنْ أَوَائِلِهِ!».</span>
<span className="v" id="v27" data-v="27" data-xref="إِنْجِيلُ لُوقَا 1:42; إِنْجِيلُ لُوقَا 1:48; إِنْجِيلُ لُوقَا 23:29"> وَفِيمَا هُوَ يَتَكَلَّمُ بِهَذَا، رَفَعَتِ ٱمْرَأَةٌ صَوْتَهَا مِنَ ٱلْجَمْعِ وَقَالَتْ لَهُ: «طُوبَى لِلْبَطْنِ ٱلَّذِي حَمَلَكَ وَٱلثَّدْيَيْنِ ٱللَّذَيْنِ رَضِعْتَهُمَا».</span>
<span className="v" id="v28" data-v="28" data-xref="إِنْجِيلُ لُوقَا 6:47–48; إِنْجِيلُ لُوقَا 8:21; اَلْمَزَامِيرُ 112:1"> أَمَّا هُوَ فَقَالَ: «بَلْ طُوبَى لِلَّذِينَ يَسْمَعُونَ كَلَامَ ٱللهِ وَيَحْفَظُونَهُ».</span>

## آية يونان

<div className="bt" dir="rtl">
<div className="bp">

## توبيخ الفريسيين والكتبة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="إِنْجِيلُ لُوقَا 7:36; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 9:19–23; إِنْجِيلُ لُوقَا 14:1"> وَفِيمَا هُوَ يَتَكَلَّمُ سَأَلَهُ فَرِّيسِيٌّ أَنْ يَتَغَدَّى عِنْدَهُ، فَدَخَلَ وَاتَّكَأَ.</span>
<span className="v" id="v38" data-v="38" data-xref="إِنْجِيلُ مَتَّى 15:2–3; إِنْجِيلُ مَرْقُسَ 7:2–5; إِنْجِيلُ يُوحَنَّا 3:25"> وَأَمَّا ٱلْفَرِّيسِيُّ فَلَمَّا رَأَى ذَلِكَ تَعَجَّبَ أَنَّهُ لَمْ يَغْتَسِلْ أَوَّلًا قَبْلَ ٱلْغَدَاءِ.</span>
<span className="v" id="v39" data-v="39" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى تِيطُسَ 1:15; أَمْثَالٌ 30:12; اَلتَّكْوِينُ 6:5"> فَقَالَ لَهُ ٱلرَّبُّ: «أَنْتُمُ ٱلْآنَ أَيُّهَا ٱلْفَرِّيسِيُّونَ تُنَقُّونَ خَارِجَ ٱلْكَأْسِ وَٱلْقَصْعَةِ، وَأَمَّا بَاطِنُكُمْ فَمَمْلُوءٌ ٱخْتِطَافًا وَخُبْثًا.</span>
<span className="v" id="v40" data-v="40" data-xref="إِنْجِيلُ لُوقَا 12:20; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 15:36; إِنْجِيلُ مَتَّى 23:26"> يَا أَغْبِيَاءُ، أَلَيْسَ ٱلَّذِي صَنَعَ ٱلْخَارِجَ صَنَعَ ٱلدَّاخِلَ أَيْضًا؟</span>
<span className="v" id="v41" data-v="41" data-xref="إِنْجِيلُ لُوقَا 12:33; رِسَالَةُ يَعْقُوبَ 1:27; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى تِيطُسَ 1:15"> بَلْ أَعْطُوا مَا عِنْدَكُمْ صَدَقَةً، فَهُوَذَا كُلُّ شَيْءٍ يَكُونُ نَقِيًّا لَكُمْ.</span>
<span className="v" id="v42" data-v="42" data-xref="إِنْجِيلُ مَتَّى 23:23; إِنْجِيلُ لُوقَا 18:12; أَمْثَالٌ 21:3"> وَلَكِنْ وَيْلٌ لَكُمْ أَيُّهَا ٱلْفَرِّيسِيُّونَ! لِأَنَّكُمْ تُعَشِّرُونَ ٱلنَّعْنَعَ وَٱلسَّذَابَ وَكُلَّ بَقْلٍ، وَتَتَجَاوَزُونَ عَنِ ٱلْحَقِّ وَمَحَبَّةِ ٱللهِ. كَانَ يَنْبَغِي أَنْ تَعْمَلُوا هَذِهِ وَلَا تَتْرُكُوا تِلْكَ.</span>
