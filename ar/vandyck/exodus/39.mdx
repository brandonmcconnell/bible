## الرداء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="اَلْخُرُوجُ 28:6–12; اَللَّاوِيِّينَ 8:7; اَلْخُرُوجُ 25:7"> فَصَنَعَ ٱلرِّدَاءَ مِنْ ذَهَبٍ وَأَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ وَبُوصٍ مَبْرُومٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="اَلْخُرُوجُ 36:8; اَلْخُرُوجُ 26:1"> وَمَدُّوا ٱلذَّهَبَ صَفَائِحَ وَقَدُّوهَا خُيُوطًا لِيَصْنَعُوهَا فِي وَسَطِ ٱلْأَسْمَانْجُونِيِّ وَٱلْأُرْجُوَانِ وَٱلْقِرْمِزِ وَٱلْبُوصِ، صَنْعَةَ ٱلْمُوَشِّي.</span>
<span className="v" id="v4" data-v="4"> وَصَنَعُوا لَهُ كَتِفَيْنِ مَوْصُولَيْنِ. عَلَى طَرَفَيْهِ ٱتَّصَلَ.</span>
<span className="v" id="v5" data-v="5" data-xref="رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 1:13; إِنْجِيلُ مَتَّى 28:20; إِشَعْيَاءَ 11:5"> وَزُنَّارُ شَدِّهِ ٱلَّذِي عَلَيْهِ كَانَ مِنْهُ كَصَنْعَتِهِ مِنْ ذَهَبٍ وَأَسْمَانْجُونِيٍّ وَقِرْمِزٍ وَبُوصٍ مَبْرُومٍ، كَمَا أَمَرَ ٱلرَّبُّ مُوسَى.</span>
<span className="v" id="v6" data-v="6" data-xref="اَلْخُرُوجُ 28:9; حِزْقِيَال 28:13; أَيُّوبَ 28:16"> وَصَنَعُوا حَجَرَيِ ٱلْجَزْعِ مُحَاطَيْنِ بِطَوْقَيْنِ مِنْ ذَهَبٍ مَنْقُوشَيْنِ نَقْشَ ٱلْخَاتِمِ عَلَى حَسَبِ أَسْمَاءِ بَنِي إِسْرَائِيلَ.</span>

## الصُّدرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="اَلْخُرُوجُ 28:4; اَللَّاوِيِّينَ 8:8–9; إِشَعْيَاءَ 59:17"> وَصَنَعَ ٱلصُّدْرَةَ صَنْعَةَ ٱلْمُوَشِّي كَصَنْعَةِ ٱلرِّدَاءِ مِنْ ذَهَبٍ وَأَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ وَبُوصٍ مَبْرُومٍ.</span>
<span className="v" id="v9" data-v="9"> كَانَتْ مُرَبَّعَةً. مَثْنِيَّةً صَنَعُوا ٱلصُّدْرَةَ. طُولُهَا شِبْرٌ وَعَرْضُهَا شِبْرٌ مَثْنِيَّةً.</span>
<span className="v" id="v10" data-v="10" data-xref="اَلْخُرُوجُ 28:21; اَلْخُرُوجُ 28:16–17; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 21:19–21"> وَرَصَّعُوا فِيهَا أَرْبَعَةَ صُفُوفِ حِجَارَةٍ. صَفُّ: عَقِيقٌ أَحْمَرُ وَيَاقُوتٌ أَصْفَرُ وَزُمُرُّدٌ، ٱلصَّفُّ ٱلْأَوَّلُ.</span>
<span className="v" id="v11" data-v="11" data-xref="اَلْخُرُوجُ 28:18; حِزْقِيَال 28:13"> وَٱلصَّفُّ ٱلثَّانِي: بَهْرَمَانُ وَيَاقُوتٌ أَزْرَقُ وَعَقِيقٌ أَبْيَضُ.</span>
<span className="v" id="v12" data-v="12"> وَٱلصَّفُّ ٱلثَّالِثُ: عَيْنُ ٱلْهِرِّ وَيَشْمٌ وَجَمَسْتُ.</span>
<span className="v" id="v13" data-v="13"> وَٱلصَّفُّ ٱلرَّابِعُ: زَبَرْجَدٌ وَجَزْعٌ وَيَشْبٌ. مُحَاطَةٌ بِأَطْوَاقٍ مِنْ ذَهَبٍ فِي تَرْصِيعِهَا.</span>
<span className="v" id="v14" data-v="14" data-xref="رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 21:12"> وَٱلْحِجَارَةُ كَانَتْ عَلَى أَسْمَاءِ بَنِي إِسْرَائِيلَ، ٱثْنَيْ عَشَرَ عَلَى أَسْمَائِهِمْ كَنَقْشِ ٱلْخَاتِمِ. كُلُّ وَاحِدٍ عَلَى ٱسْمِهِ لِلِٱثْنَيْ عَشَرَ سِبْطًا.</span>
<span className="v" id="v15" data-v="15" data-xref="إِنْجِيلُ يُوحَنَّا 10:28; إِنْجِيلُ يُوحَنَّا 17:12; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 1:5"> وَصَنَعُوا عَلَى ٱلصُّدْرَةِ سَلَاسِلَ مَجْدُولَةً صَنْعَةَ ٱلضَّفْرِ مِنْ ذَهَبٍ نَقِيٍّ.</span>
<span className="v" id="v16" data-v="16" data-xref="اَلْخُرُوجُ 25:12"> وَصَنَعُوا طَوْقَيْنِ مِنْ ذَهَبٍ وَحَلْقَتَيْنِ مِنْ ذَهَبٍ، وَجَعَلُوا ٱلْحَلْقَتَيْنِ عَلَى طَرَفَيِ ٱلصُّدْرَةِ.</span>
<span className="v" id="v17" data-v="17"> وَجَعَلُوا ضَفِيرَتَيِ ٱلذَّهَبِ فِي ٱلْحَلْقَتَيْنِ عَلَى طَرَفَيِ ٱلصُّدْرَةِ.</span>
<span className="v" id="v18" data-v="18" data-xref="اَلْخُرُوجُ 28:14; نَشِيدُ ٱلْأَنْشَادِ 1:10"> وَطَرَفَا ٱلضَّفِيرَتَيْنِ جَعَلُوهُمَا فِي ٱلطَّوْقَيْنِ، وَجَعَلُوهُمَا عَلَى كَتِفَيِ ٱلرِّدَاءِ إِلَى قُدَّامِهِ.</span>
<span className="v" id="v19" data-v="19"> وَصَنَعُوا حَلْقَتَيْنِ مِنْ ذَهَبٍ وَوَضَعُوهُمَا عَلَى طَرَفَيِ ٱلصُّدْرَةِ. عَلَى حَاشِيَتِهَا ٱلَّتِي إِلَى جِهَةِ ٱلرِّدَاءِ مِنْ دَاخِلٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="اَلْخُرُوجُ 26:3"> وَصَنَعُوا حَلْقَتَيْنِ مِنْ ذَهَبٍ وَجَعَلُوهُمَا عَلَى كَتِفَيِ ٱلرِّدَاءِ مِنْ أَسْفَلُ مِنْ قُدَّامِهِ عِنْدَ وَصْلِهِ فَوْقَ زُنَّارِ ٱلرِّدَاءِ.</span>
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ مَتَّى 16:24; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 1:25; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 1:27"> وَرَبَطُوا ٱلصُّدْرَةَ بِحَلْقَتَيْهَا إِلَى حَلْقَتَيِ ٱلرِّدَاءِ بِخَيْطٍ مِنْ أَسْمَانْجُونِيٍّ لِيَكُونَ عَلَى زُنَّارِ ٱلرِّدَاءِ، وَلَا تُنْزَعُ ٱلصُّدْرَةُ عَنِ ٱلرِّدَاءِ، كَمَا أَمَرَ ٱلرَّبُّ مُوسَى.</span>
</div>
</div>

## ثياب كهنوتية أخرى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اَلْخُرُوجُ 28:31–35"> وَصَنَعَ جُبَّةَ ٱلرِّدَاءِ صَنْعَةَ ٱلنَّسَّاجِ، كُلَّهَا مِنْ أَسْمَانْجُونِيٍّ.</span>
<span className="v" id="v23" data-v="23"> وَفَتْحَةُ ٱلْجُبَّةِ فِي وَسَطِهَا كَفَتْحَةِ ٱلدِّرْعِ، وَلِفَتْحَتِهَا حَاشِيَةٌ حَوَالَيْهَا. لَا تَنْشَقُّ.</span>
<span className="v" id="v24" data-v="24" data-xref="اَلْخُرُوجُ 28:33; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ غَلَاطِيَّةَ 5:22"> وَصَنَعُوا عَلَى أَذْيَالِ ٱلْجُبَّةِ رُمَّانَاتٍ مِنْ أَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ مَبْرُومٍ.</span>
<span className="v" id="v25" data-v="25" data-xref="اَلْخُرُوجُ 28:33–34; نَشِيدُ ٱلْأَنْشَادِ 4:13; اَلْمَزَامِيرُ 89:15"> وَصَنَعُوا جَلَاجِلَ مِنْ ذَهَبٍ نَقِيٍّ، وَجَعَلُوا ٱلْجَلَاجِلَ فِي وَسَطِ ٱلرُّمَّانَاتِ عَلَى أَذْيَالِ ٱلْجُبَّةِ حَوَالَيْهَا فِي وَسَطِ ٱلرُّمَّانَاتِ.</span>
<span className="v" id="v26" data-v="26" data-xref="اَلْخُرُوجُ 28:34; نَشِيدُ ٱلْأَنْشَادِ 4:13; نَشِيدُ ٱلْأَنْشَادِ 6:7"> جُلْجُلٌ وَرُمَّانَةٌ. جُلْجُلٌ وَرُمَّانَةٌ. عَلَى أَذْيَالِ ٱلْجُبَّةِ حَوَالَيْهَا لِلْخِدْمَةِ، كَمَا أَمَرَ ٱلرَّبُّ مُوسَى.</span>
<span className="v" id="v27" data-v="27" data-xref="حِزْقِيَال 44:18; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 1:13; إِشَعْيَاءَ 61:10"> وَصَنَعُوا ٱلْأَقْمِصَةَ مِنْ بُوصٍ صَنْعَةَ ٱلنَّسَّاجِ لِهَارُونَ وَبَنِيهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="اَلْخُرُوجُ 28:4; حِزْقِيَال 44:18; اَلْخُرُوجُ 28:39"> وَٱلْعِمَامَةَ مِنْ بُوصٍ، وَعَصَائِبَ ٱلْقَلَانِسِ مِنْ بُوصٍ، وَسَرَاوِيلَ ٱلْكَتَّانِ مِنْ بُوصٍ مَبْرُومٍ.</span>
<span className="v" id="v29" data-v="29" data-xref="اَلْخُرُوجُ 28:39"> وَٱلْمِنْطَقَةَ مِنْ بُوصٍ مَبْرُومٍ وَأَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ صَنْعَةَ ٱلطَّرَّازِ، كَمَا أَمَرَ ٱلرَّبُّ مُوسَى.</span>
<span className="v" id="v30" data-v="30" data-xref="رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 5:10; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 7:26; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 1:3"> وَصَنَعُوا صَفِيحَةَ ٱلْإِكْلِيلِ ٱلْمُقَدَّسِ مِنْ ذَهَبٍ نَقِيٍّ، وَكَتَبُوا عَلَيْهَا كِتَابَةَ نَقْشِ ٱلْخَاتِمِ: «قُدْسٌ لِلرَّبِّ».</span>
<span className="v" id="v31" data-v="31"> وَجَعَلُوا عَلَيْهَا خَيْطَ أَسْمَانْجُونِيٍّ لِتُجْعَلَ عَلَى ٱلْعِمَامَةِ مِنْ فَوْقُ، كَمَا أَمَرَ ٱلرَّبُّ مُوسَى.</span>

## موسى يتفحص الخيمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="اَلْخُرُوجُ 39:42–43; إِنْجِيلُ مَتَّى 28:20; اَلْخُرُوجُ 25:40"> فَكَمُلَ كُلُّ عَمَلِ مَسْكَنِ خَيْمَةِ ٱلِٱجْتِمَاعِ. وَصَنَعَ بَنُو إِسْرَائِيلَ بِحَسَبِ كُلِّ مَا أَمَرَ ٱلرَّبُّ مُوسَى. هَكَذَا صَنَعُوا.</span>
<span className="v" id="v33" data-v="33" data-xref="اَلْخُرُوجُ 35:11–19; اَلْخُرُوجُ 31:7–11; اَلْخُرُوجُ 36:1–38"> وَجَاءُوا إِلَى مُوسَى بِٱلْمَسْكَنِ: ٱلْخَيْمَةِ وَجَمِيعِ أَوَانِيهَا، أَشِظَّتِهَا وَأَلْوَاحِهَا وَعَوَارِضِهَا وَأَعْمِدَتِهَا وَقَوَاعِدِهَا،</span>
<span className="v" id="v34" data-v="34"> وَٱلْغِطَاءِ مِنْ جُلُودِ ٱلْكِبَاشِ ٱلْمُحَمَّرَةِ، وَٱلْغِطَاءِ مِنْ جُلُودِ ٱلتُّخَسِ، وَحِجَابِ ٱلسَّجْفِ،</span>
