## فرائض السبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اَلْخُرُوجُ 34:32; اَلْخُرُوجُ 31:1–11; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 2:13"> وَجَمَعَ مُوسَى كُلَّ جَمَاعَةِ بَنِي إِسْرَائِيلَ وَقَالَ لَهُمْ: «هَذِهِ هِيَ ٱلْكَلِمَاتُ ٱلَّتِي أَمَرَ ٱلرَّبُّ أَنْ تُصْنَعَ:</span>
<span className="v" id="v2" data-v="2" data-xref="اَللَّاوِيِّينَ 23:3; إِنْجِيلُ لُوقَا 13:14–15; اَلْخُرُوجُ 20:9–10"> سِتَّةَ أَيَّامٍ يُعْمَلُ عَمَلٌ، وَأَمَّا ٱلْيَوْمُ ٱلْسَّابِعُ فَفِيهِ يَكُونُ لَكُمْ سَبْتُ عُطْلَةٍ مُقَدَّسٌ لِلرَّبِّ. كُلُّ مَنْ يَعْمَلُ فِيهِ عَمَلًا يُقْتَلُ.</span>

## مواد لبناء خيمة الاجتماع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="اَلْخُرُوجُ 25:1–2"> وَكَلَّمَ مُوسَى كُلَّ جَمَاعَةِ بَنِي إِسْرَائِيلَ قَائِلًا: «هَذَا هُوَ ٱلشَّيْءُ ٱلَّذِي أَمَرَ بِهِ ٱلرَّبُّ قَائِلًا:</span>
<span className="v" id="v5" data-v="5" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 9:7; اَلْمَزَامِيرُ 110:3; اَلْخُرُوجُ 25:2–7"> خُذُوا مِنْ عِنْدِكُمْ تَقْدِمَةً لِلرَّبِّ. كُلُّ مَنْ قَلْبُهُ سَمُوحٌ فَلْيَأْتِ بِتَقْدِمَةِ ٱلرَّبِّ: ذَهَبًا وَفِضَّةً وَنُحَاسًا،</span>
<span className="v" id="v6" data-v="6" data-xref="اَلْخُرُوجُ 26:7–14; اَلْخُرُوجُ 26:31; اَلْخُرُوجُ 28:15"> وَأَسْمَانْجُونِيًّا وَأُرْجُوَانًا وَقِرْمِزًا وَبُوصًا وَشَعْرَ مِعْزًى،</span>
<span className="v" id="v7" data-v="7"> وَجُلُودَ كِبَاشٍ مُحَمَّرَةً وَجُلُودَ تُخَسٍ وَخَشَبَ سَنْطٍ،</span>
<span className="v" id="v8" data-v="8" data-xref="اَلْخُرُوجُ 27:20; اَلْخُرُوجُ 25:1–40; اَلْخُرُوجُ 30:23"> وَزَيْتًا لِلضَّوْءِ وَأَطْيَابًا لِدُهْنِ ٱلْمَسْحَةِ وَلِلْبَخُورِ ٱلْعَطِرِ،</span>
<span className="v" id="v9" data-v="9" data-xref="اَلْخُرُوجُ 39:6–14; اَلْخُرُوجُ 28:9; اَلْخُرُوجُ 25:5"> وَحِجَارَةَ جَزْعٍ وَحِجَارَةَ تَرْصِيعٍ لِلرِّدَاءِ وَٱلصُّدْرَةِ.</span>
<span className="v" id="v10" data-v="10" data-xref="اَلْخُرُوجُ 31:1–6; اَلْخُرُوجُ 36:1–4"> وَكُلُّ حَكِيمِ ٱلْقَلْبِ بَيْنَكُمْ فَلْيَأْتِ وَيَصْنَعْ كُلَّ مَا أَمَرَ بِهِ ٱلرَّبُّ:</span>
<span className="v" id="v11" data-v="11" data-xref="اَلْخُرُوجُ 26:1–37; اَلْخُرُوجُ 36:8–34; اَلْخُرُوجُ 31:7–9"> ٱلْمَسْكَنَ وَخَيْمَتَهُ وَغِطَاءَهُ وَأَشِظَّتَهُ وَأَلْوَاحَهُ وَعَوَارِضَهُ وَأَعْمِدَتَهُ وَقَوَاعِدَهُ،</span>
<span className="v" id="v12" data-v="12" data-xref="اَلْخُرُوجُ 25:10–22; اَلْخُرُوجُ 26:31–33; اَلْخُرُوجُ 37:1–9"> وَٱلتَّابُوتَ وَعَصَوَيْهِ، وَٱلْغِطَاءَ وَحِجَابَ ٱلسَّجْفِ،</span>
<span className="v" id="v13" data-v="13" data-xref="اَلْخُرُوجُ 25:23–30; اَللَّاوِيِّينَ 24:5–6; اَلْخُرُوجُ 37:10–16"> وَٱلْمَائِدَةَ وَعَصَوَيْهَا وَكُلَّ آنِيَتِهَا، وَخُبْزَ ٱلْوُجُوهِ،</span>
<span className="v" id="v14" data-v="14" data-xref="اَلْخُرُوجُ 25:31–39; إِنْجِيلُ مَتَّى 5:14–15; اَلْخُرُوجُ 37:17–24"> وَمَنَارَةَ ٱلضَّوْءِ وَآنِيَتَهَا وَسُرُجَهَا وَزَيْتَ ٱلضَّوْءِ،</span>
<span className="v" id="v15" data-v="15" data-xref="اَلْخُرُوجُ 30:1–10; اَلْخُرُوجُ 30:22–38; اَلْخُرُوجُ 37:25–28"> وَمَذْبَحَ ٱلْبَخُورِ وَعَصَوَيْهِ، وَدُهْنَ ٱلْمَسْحَةِ وَٱلْبَخُورَ ٱلْعَطِرَ، وَسَجْفَ ٱلْبَابِ لِمَدْخَلِ ٱلْمَسْكَنِ،</span>
<span className="v" id="v16" data-v="16" data-xref="اَلْخُرُوجُ 27:1–8; اَلْخُرُوجُ 30:18–21; اَلْخُرُوجُ 38:1–8"> وَمَذْبَحَ ٱلْمُحْرَقَةِ وَشُبَّاكَةَ ٱلنُّحَاسِ ٱلَّتِي لَهُ وَعَصَوَيْهِ وَكُلَّ آنِيَتِهِ، وَٱلْمِرْحَضَةَ وَقَاعِدَتَهَا،</span>
<span className="v" id="v17" data-v="17" data-xref="صَمُوئِيلَ ٱلثَّانِي 7:2; اَلْخُرُوجُ 38:9–20; اَلْخُرُوجُ 27:9–19"> وَأَسْتَارَ ٱلدَّارِ وَأَعْمِدَتَهَا وَقَوَاعِدَهَا، وَسَجْفَ بَابِ ٱلدَّارِ،</span>
<span className="v" id="v18" data-v="18" data-xref="اَلْخُرُوجُ 27:19"> وَأَوْتَادَ ٱلْمَسْكَنِ، وَأَوْتَادَ ٱلدَّارِ وَأَطْنَابَهَا،</span>
<span className="v" id="v19" data-v="19" data-xref="اَلْخُرُوجُ 31:10; اَلْعَدَد 4:5–15; اَلْخُرُوجُ 28:1–43"> وَٱلثِّيَابَ ٱلْمَنْسُوجَةَ لِلْخِدْمَةِ فِي ٱلْمَقْدِسِ، وَٱلثِّيَابَ ٱلْمُقَدَّسَةَ لِهَارُونَ ٱلْكَاهِنِ، وَثِيَابَ بَنِيهِ لِلْكَهَانَةِ».</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20"> فَخَرَجَ كُلُّ جَمَاعَةِ بَنِي إِسْرَائِيلَ مِنْ قُدَّامِ مُوسَى،</span>
<span className="v" id="v21" data-v="21" data-xref="اَلْخُرُوجُ 25:2; اَلْخُرُوجُ 35:26; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 29:14"> ثُمَّ جَاءَ كُلُّ مَنْ أَنْهَضَهُ قَلْبُهُ، وَكُلُّ مَنْ سَمَّحَتْهُ رُوحُهُ. جَاءُوا بِتَقْدِمَةِ ٱلرَّبِّ لِعَمَلِ خَيْمَةِ ٱلِٱجْتِمَاعِ وَلِكُلِّ خِدْمَتِهَا وَلِلثِّيَابِ ٱلْمُقَدَّسَةِ.</span>
<span className="v" id="v22" data-v="22" data-xref="نَحَمْيَا 7:70–72; إِنْجِيلُ مَتَّى 2:11; عَزْرَا 2:68–69"> وَجَاءَ ٱلرِّجَالُ مَعَ ٱلنِّسَاءِ، كُلُّ سَمُوحِ ٱلْقَلْبِ، جَاءَ بِخَزَائِمَ وَأَقْرَاطٍ وَخَوَاتِمَ وَقَلَائِدَ، كُلِّ مَتَاعٍ مِنَ ٱلذَّهَبِ. وَكُلُّ مَنْ قَدَّمَ تَقْدِمَةَ ذَهَبٍ لِلرَّبِّ.</span>
<span className="v" id="v23" data-v="23" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 29:8; اَلْخُرُوجُ 25:2–7; اَلْخُرُوجُ 35:6–10"> وَكُلُّ مَنْ وُجِدَ عِنْدَهُ أَسْمَانْجُونِيٌّ وَأُرْجُوانٌ وَقِرْمِزٌ وَبُوصٌ وَشَعْرُ مِعْزًى وَجُلُودُ كِبَاشٍ مُحَمَّرَةٌ وَجُلُودُ تُخَسٍ، جَاءَ بِهَا.</span>
<span className="v" id="v24" data-v="24" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 8:12"> كُلُّ مَنْ قَدَّمَ تَقْدِمَةَ فِضَّةٍ وَنُحَاسٍ جَاءَ بِتَقْدِمَةِ ٱلرَّبِّ. وَكُلُّ مَنْ وُجِدَ عِنْدَهُ خَشَبُ سَنْطٍ لِصَنْعَةٍ مَا مِنَ ٱلْعَمَلِ جَاءَ بِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 16:12; إِنْجِيلُ لُوقَا 8:2–3; أَمْثَالٌ 14:1"> وَكُلُّ ٱلنِّسَاءِ ٱلْحَكِيمَاتِ ٱلْقَلْبِ غَزَلْنَ بِأَيْدِيهِنَّ وَجِئْنَ مِنَ ٱلْغَزْلِ بِٱلْأَسْمَانْجُونِيِّ وَٱلْأُرْجُوَانِ وَٱلْقِرْمِزِ وَٱلْبُوصِ.</span>
<span className="v" id="v26" data-v="26" data-xref="اَلْخُرُوجُ 35:21; اَلْخُرُوجُ 36:8; اَلْخُرُوجُ 35:29"> وَكُلُّ ٱلنِّسَاءِ ٱللَّوَاتِي أَنْهَضَتْهُنَّ قُلُوبُهُنَّ بِٱلْحِكْمَةِ غَزَلْنَ شَعْرَ ٱلْمِعْزَى.</span>
<span className="v" id="v27" data-v="27" data-xref="عَزْرَا 2:68; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 29:6; اَلْخُرُوجُ 35:9"> وَٱلرُّؤَسَاءُ جَاءُوا بِحِجَارَةِ ٱلْجَزْعِ وَحِجَارَةِ ٱلتَّرْصِيعِ لِلرِّدَاءِ وَٱلصُّدْرَةِ،</span>
<span className="v" id="v28" data-v="28" data-xref="اَلْخُرُوجُ 30:23–38; اَلْخُرُوجُ 35:8"> وَبِٱلطِّيبِ وَٱلزَّيْتِ لِلضَّوْءِ وَلِدُهْنِ ٱلْمَسْحَةِ وَلِلْبَخُورِ ٱلْعَطِرِ.</span>
<span className="v" id="v29" data-v="29" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 9:7; اَلْخُرُوجُ 35:21–22; اَلْخُرُوجُ 35:4"> بَنُو إِسْرَائِيلَ، جَمِيعُ ٱلرِّجَالِ وَٱلنِّسَاءِ ٱلَّذِينَ سَمَّحَتْهُمْ قُلُوبُهُمْ أَنْ يَأْتُوا بِشَيْءٍ لِكُلِّ ٱلْعَمَلِ ٱلَّذِي أَمَرَ ٱلرَّبُّ أَنْ يُصْنَعَ عَلَى يَدِ مُوسَى، جَاءُوا بِهِ تَبَرُّعًا إِلَى ٱلرَّبِّ.</span>

## بصلئيل وأهوليآب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="اَلْخُرُوجُ 31:1–6; رِسَالَةُ يَعْقُوبَ 1:17; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 12:11"> وَقَالَ مُوسَى لِبَنِي إِسْرَائِيلَ: «ٱنْظُرُوا. قَدْ دَعَا ٱلرَّبُّ بَصَلْئِيلَ بْنَ أُورِي بْنَ حُورَ مِنْ سِبْطِ يَهُوذَا بِٱسْمِهِ،</span>
<span className="v" id="v31" data-v="31" data-xref="رِسَالَةُ يَعْقُوبَ 1:17; إِشَعْيَاءَ 11:2–5; إِشَعْيَاءَ 61:1–3"> وَمَلَأَهُ مِنْ رُوحِ ٱللهِ بِٱلْحِكْمَةِ وَٱلْفَهْمِ وَٱلْمَعْرِفَةِ وَكُلِّ صَنْعَةٍ،</span>
<span className="v" id="v32" data-v="32"> وَلِٱخْتِرَاعِ مُخْتَرَعَاتٍ، لِيَعْمَلَ فِي ٱلذَّهَبِ وَٱلْفِضَّةِ وَٱلنُّحَاسِ،</span>
<span className="v" id="v33" data-v="33"> وَنَقْشِ حِجَارَةٍ لِلتَّرْصِيعِ، وَنِجَارَةِ ٱلْخَشَبِ، لِيَعْمَلَ فِي كُلِّ صَنْعَةٍ مِنَ ٱلْمُخْتَرَعَاتِ.</span>
<span className="v" id="v34" data-v="34" data-xref="اَلْخُرُوجُ 31:6; أَخْبَارِ ٱلْأَيَّامِ ٱلثَّانِي 2:14; عَزْرَا 7:27"> وَجَعَلَ فِي قَلْبِهِ أَنْ يُعَلِّمَ هُوَ وَأُهُولِيآبُ بْنَ أَخِيسَامَاكَ مِنْ سِبْطِ دَانَ.</span>
<span className="v" id="v35" data-v="35" data-xref="اَلْخُرُوجُ 35:31; اَلْمُلُوكِ ٱلْأَوَّلُ 7:14; إِشَعْيَاءَ 28:26"> قَدْ مَلَأَهُمَا حِكْمَةَ قَلْبٍ لِيَصْنَعَا كُلَّ عَمَلِ ٱلنَّقَّاشِ وَٱلْحَائِكِ ٱلْحَاذِقِ وَٱلطَّرَّازِ فِي ٱلْأَسْمَانْجُونِيِّ وَٱلْأُرْجُوَانِ وَٱلْقِرْمِزِ وَٱلْبُوصِ وَكُلَّ عَمَلِ ٱلنَّسَّاجِ. صَانِعِي كُلِّ صَنْعَةٍ وَمُخْتَرِعِي ٱلْمُخْتَرَعَاتِ.</span>
