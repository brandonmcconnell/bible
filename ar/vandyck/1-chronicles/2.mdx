## بنو يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="اَلتَّكْوِينُ 38:2–10; اَلتَّكْوِينُ 46:12; اَلْعَدَد 26:19"> بَنُو يَهُوذَا: عَيْرُ وَأُونَانُ وَشَيْلَةُ. وُلِدَ ٱلثَّلَاثَةُ مِنْ بِنْتِ شُوعَ ٱلْكَنْعَانِيَّةِ. وَكَانَ عَيْرُ بِكْرُ يَهُوذَا شِرِّيرًا فِي عَيْنَيِ ٱلرَّبِّ فَأَمَاتَهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="إِنْجِيلُ مَتَّى 1:3; اَلتَّكْوِينُ 38:13–30; نَحَمْيَا 11:4"> وَثَامَارُ كَنَّتُهُ وَلَدَتْ لَهُ فَارَصَ وَزَارَحَ. كُلُّ بَنِي يَهُوذَا خَمْسَةٌ.</span>
<span className="v" id="v5" data-v="5" data-xref="اَلتَّكْوِينُ 46:12; رَاعُوث 4:18; اَلْعَدَد 26:21"> اِبْنَا فَارَصَ: حَصْرُونُ وَحَامُولُ.</span>
<span className="v" id="v6" data-v="6" data-xref="اَلْمُلُوكِ ٱلْأَوَّلُ 4:31; يَشُوع 7:17–18; يَشُوع 7:1"> وَبَنُو زَارَحَ: زِمْرِي وَأَيْثَانُ وَهَيْمَانُ وَكَلْكُولُ وَدَارَعُ. ٱلْجَمِيعُ خَمْسَةٌ.</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 6:18; اَلتَّثْنِيَة 7:26; يَشُوع 7:25"> وَٱبْنُ كَرْمِي عَخَارُ مُكَدِّرُ إِسْرَائِيلَ ٱلَّذِي خَانَ فِي ٱلْحَرَامِ.</span>
<span className="v" id="v8" data-v="8"> وَٱبْنُ أَيْثَانَ: عَزَرْيَا.</span>
<span className="v" id="v9" data-v="9" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:42; إِنْجِيلُ مَتَّى 1:3; رَاعُوث 4:19"> وَبَنُو حَصْرُونَ ٱلَّذِينَ وُلِدُوا لَهُ: يَرْحَمْئِيلُ وَرَامُ وَكَلُوبَايُ.</span>
</div>
</div>

## رام بن حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إِنْجِيلُ مَتَّى 1:4; اَلْعَدَد 1:7; اَلْعَدَد 2:3"> وَرَامُ وَلَدَ عَمِّينَادَابَ، وَعَمِّينَادَابُ وَلَدَ نَحْشُونَ رَئِيسَ بَنِي يَهُوذَا،</span>
<span className="v" id="v11" data-v="11" data-xref="رَاعُوث 4:21; إِنْجِيلُ مَتَّى 1:4–5; إِنْجِيلُ لُوقَا 3:32"> وَنَحْشُونُ وَلَدَ سَلْمُوَ، وَسَلْمُو وَلَدَ بُوعَزَ،</span>
<span className="v" id="v12" data-v="12" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 15:12; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 10:14; إِشَعْيَاءَ 11:10"> وَبُوعَزُ وَلَدَ عُوبِيدَ، وَعُوبِيدُ وَلَدَ يَسَّى،</span>
<span className="v" id="v13" data-v="13" data-xref="صَمُوئِيلَ ٱلْأَوَّلُ 17:13; صَمُوئِيلَ ٱلْأَوَّلُ 17:28; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 20:7"> وَيَسَّى وَلَدَ: بِكْرَهُ أَلِيآبَ، وَأَبِينَادَابَ ٱلثَّانِي، وَشِمْعَى ٱلثَّالِثَ،</span>
<span className="v" id="v14" data-v="14"> وَنَثْنِئِيلَ ٱلرَّابِعَ، وَرَدَّايَ ٱلْخَامِسَ،</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئِيلَ ٱلْأَوَّلُ 16:10–11; صَمُوئِيلَ ٱلْأَوَّلُ 17:12–14"> وَأُوصَمَ ٱلسَّادِسَ، وَدَاوُدَ ٱلسَّابِعَ.</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئِيلَ ٱلْأَوَّلُ 26:6; صَمُوئِيلَ ٱلثَّانِي 3:39; صَمُوئِيلَ ٱلثَّانِي 2:18–23"> وَأُخْتَاهُمْ صَرُويَةُ وَأَبِيجَايِلُ. وَبَنُو صَرُويَةَ: أَبْشَايُ وَيُوآبُ وَعَسَائِيلُ، ثَلَاثَةٌ.</span>
<span className="v" id="v17" data-v="17" data-xref="صَمُوئِيلَ ٱلثَّانِي 17:25; اَلْمُلُوكِ ٱلْأَوَّلُ 2:32; صَمُوئِيلَ ٱلثَّانِي 20:4–12"> وَأَبِيجَايِلُ وَلَدَتْ عَمَاسَا، وَأَبُو عَمَاسَا يَثْرُ ٱلْإِسْمَاعِيلِيُّ.</span>
</div>
</div>

## كالب بن حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18"> وَكَالَبُ بْنُ حَصْرُونَ وَلَدَ مِنْ عَزُوبَةَ ٱمْرَأَتِهِ وَمِنْ يَرِيعُوثَ. وَهَؤُلَاءِ بَنُوهَا: يَاشَرُ وَشُوبَابُ وَأَرْدُونُ.</span>
<span className="v" id="v19" data-v="19" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:50; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:24; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 4:4"> وَمَاتَتْ عَزُوبَةُ فَٱتَّخَذَ كَالَبُ لِنَفْسِهِ أَفْرَاتَ فَوَلَدَتْ لَهُ حُورَ.</span>
<span className="v" id="v20" data-v="20" data-xref="اَلْخُرُوجُ 31:2; اَلْخُرُوجُ 36:1–2; اَلْخُرُوجُ 37:1"> وَحُورُ وَلَدَ أُورِيَ، وَأُورِي وَلَدَ بَصَلْئِيلَ.</span>
<span className="v" id="v21" data-v="21" data-xref="اَلْعَدَد 27:1; اَلتَّثْنِيَة 3:15; اَلْعَدَد 26:29"> وَبَعْدُ دَخَلَ حَصْرُونُ عَلَى بِنْتِ مَاكِيرَ أَبِي جِلْعَادَ وَٱتَّخَذَهَا وَهُوَ ٱبْنُ سِتِّينَ سَنَةً فَوَلَدَتْ لَهُ سَجُوبَ.</span>
<span className="v" id="v22" data-v="22" data-xref="يَشُوع 13:30; اَلتَّثْنِيَة 3:14; اَلْعَدَد 32:41"> وَسَجُوبُ وَلَدَ يَائِيرَ، وَكَانَ لَهُ ثَلَاثٌ وَعِشْرُونَ مَدِينَةً فِي أَرْضِ جِلْعَادَ.</span>
<span className="v" id="v23" data-v="23" data-xref="يَشُوع 13:30; اَلتَّثْنِيَة 3:14; اَلْعَدَد 32:41–42"> وَأَخَذَ جَشُورَ وَأَرَامَ حَوُّوثَ يَائِيرَ مِنْهُمْ مَعَ قَنَاةَ وَقُرَاهَا، سِتِّينَ مَدِينَةً. كُلُّ هَؤُلَاءِ بَنُو مَاكِيرَ أَبِي جِلْعَادَ.</span>
<span className="v" id="v24" data-v="24" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 4:5; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:9; صَمُوئِيلَ ٱلْأَوَّلُ 30:14"> وَبَعْدَ وَفَاةِ حَصْرُونَ فِي كَالَبِ أَفْرَاتَةَ، وَلَدَتْ لَهُ أَبِيَّاهُ ٱمْرَأَةُ حَصْرُونَ أَشْحُورَ أَبَا تَقُوعَ.</span>
</div>

## يرحمئيل بن حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25"> وَكَانَ بَنُو يَرْحَمْئِيلَ بِكْرِ حَصْرُونَ: ٱلْبِكْرُ رَامَ، ثُمَّ بُونَةَ وَأَوْرَنَ وَأَوْصَمَ وَأَخِيَّا.</span>
<span className="v" id="v26" data-v="26"> وَكَانَتِ ٱمْرَأَةٌ أُخْرَى لِيَرْحَمْئِيلَ ٱسْمُهَا عَطَارَةُ. هِيَ أُمُّ أُونَامَ.</span>
<span className="v" id="v27" data-v="27"> وَكَانَ بَنُو رَامَ بِكْرِ يَرْحَمْئِيلَ: مَعَصُ وَيَمِينُ وَعَاقَرُ.</span>
<span className="v" id="v28" data-v="28"> وَكَانَ ٱبْنَا أُونَامَ: شَمَّايَ وَيَادَاعَ. وَٱبْنَا شَمَّايَ: نَادَابَ وَأَبِيشُورَ.</span>
<span className="v" id="v29" data-v="29"> وَٱسْمُ ٱمْرَأَةِ أَبِيشُورَ أَبِيحَايِلُ، وَوَلَدَتْ لَهُ أَحْبَانَ وَمُولِيدَ.</span>
<span className="v" id="v30" data-v="30"> وَٱبْنَا نَادَابَ: سَلَدُ وَأَفَّايِمُ. وَمَاتَ سَلَدُ بِلَا بَنِينَ.</span>
<span className="v" id="v31" data-v="31" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:34–35"> وَٱبْنُ أَفَّايِمَ يَشْعِي، وَٱبْنُ يَشْعِي شِيشَانُ، وَٱبْنُ شِيشَانَ أَحْلَايُ.</span>
<span className="v" id="v32" data-v="32"> وَٱبْنَا يَادَاعَ أَخِي شَمَّايَ: يَثَرُ وَيُونَاثَانُ. وَمَاتَ يَثَرُ بِلَا بَنِينَ.</span>
<span className="v" id="v33" data-v="33"> وَٱبْنَا يُونَاثَانَ: فَالَتُ وَزَازَا. هَؤُلَاءِ هُمْ بَنُو يَرْحَمْئِيلَ.</span>
<span className="v" id="v34" data-v="34"> وَلَمْ يَكُنْ لِشِيشَانَ بَنُونَ بَلْ بَنَاتٌ. وَكَانَ لِشِيشَانَ عَبْدٌ مِصْرِيٌّ ٱسْمُهُ يَرْحَعُ،</span>
<span className="v" id="v35" data-v="35"> فَأَعْطَى شِيشَانُ ٱبْنَتَهُ لِيَرْحَعَ عَبْدِهِ ٱمْرَأَةً، فَوَلَدَتْ لَهُ عَتَّايَ.</span>
<span className="v" id="v36" data-v="36" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 11:41"> وَعَتَّايُ وَلَدَ نَاثَانَ، وَنَاثَانُ وَلَدَ زَابَادَ،</span>
<span className="v" id="v37" data-v="37"> وَزَابَادُ وَلَدَ أَفْلَالَ، وَأَفْلَالُ وَلَدَ عُوبِيدَ،</span>
<span className="v" id="v38" data-v="38"> وَعُوبِيدُ وَلَدَ يَاهُوَ، وَيَاهُو وَلَدَ عَزَرْيَا،</span>
<span className="v" id="v39" data-v="39"> وَعَزَرْيَا وَلَدَ حَالَصَ، وَحَالَصُ وَلَدَ إِلْعَاسَةَ،</span>
<span className="v" id="v40" data-v="40"> وَإِلْعَاسَةُ وَلَدَ سِسَمَايَ، وَسِسَمَايُ وَلَدَ شَلُّومَ،</span>
<span className="v" id="v41" data-v="41"> وَشَلُّومُ وَلَدَ يَقَمْيَةَ، وَيَقَمْيَةُ وَلَدَ أَلِيشَمَعَ.</span>

## عشائر كالب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:23–24; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 12:23; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 8:29"> وَبَنُو كَالَبَ أَخِي يَرْحَمْئِيلَ: مِيشَاعُ بِكْرُهُ. هُوَ أَبُو زِيفَ. وَبَنُو مَرِيشَةَ أَبِي حَبْرُونَ.</span>
<span className="v" id="v43" data-v="43"> وَبَنُو حَبْرُونَ: قُورَحُ وَتَفُّوحُ وَرَاقَمُ وَشَامَعُ.</span>
<span className="v" id="v44" data-v="44"> وَشَامَعُ وَلَدَ رَاقَمَ أَبَا يَرُقْعَامَ. وَرَاقَمُ وَلَدَ شَمَّايَ.</span>
<span className="v" id="v45" data-v="45" data-xref="يَشُوع 15:58"> وَٱبْنُ شَمَّايَ مَعُونُ، وَمَعُونُ أَبُو بَيْتِ صُورَ.</span>

<span className="v" id="v47" data-v="47"> وَبَنُو يَهْدَايَ: رَجَمُ وَيُوثَامُ وَجِيشَانُ وَفَلَطُ وَعِيفَةُ وَشَاعَفُ.</span>
<span className="v" id="v48" data-v="48" data-xref="اَلتَّكْوِينُ 25:5–6; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:46"> وَأَمَّا مَعْكَةُ سُرِّيَّةُ كَالَبَ فَوَلَدَتْ: شَبَرَ وَتَرْحَنَةَ.</span>
<span className="v" id="v49" data-v="49" data-xref="يَشُوع 15:31; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 2:42; إِشَعْيَاءَ 10:31"> وَوَلَدَتْ شَاعَفُ أَبَا مَدْمَنَّةَ، وَشَوَا أَبَا مَكْبِينَا وَأَبَا جَبَعَا. وَبِنْتُ كَالَبَ عَكْسَةُ.</span>
