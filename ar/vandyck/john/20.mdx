## القبر الفارغ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إِنْجِيلُ مَرْقُسَ 16:9; إِنْجِيلُ مَتَّى 27:60; إِنْجِيلُ يُوحَنَّا 19:25"> وَفِي أَوَّلِ ٱلْأُسْبُوعِ جَاءَتْ مَرْيَمُ ٱلْمَجْدَلِيَّةُ إِلَى ٱلْقَبْرِ بَاكِرًا، وَٱلظَّلَامُ بَاقٍ. فَنَظَرَتِ ٱلْحَجَرَ مَرْفُوعًا عَنِ ٱلْقَبْرِ.</span>
<span className="v" id="v2" data-v="2" data-xref="إِنْجِيلُ يُوحَنَّا 13:23; إِنْجِيلُ يُوحَنَّا 20:13; إِنْجِيلُ يُوحَنَّا 21:24"> فَرَكَضَتْ وَجَاءَتْ إِلَى سِمْعَانَ بُطْرُسَ وَإِلَى ٱلتِّلْمِيذِ ٱلْآخَرِ ٱلَّذِي كَانَ يَسُوعُ يُحِبُّهُ، وَقَالَتْ لَهُمَا: «أَخَذُوا ٱلسَّيِّدَ مِنَ ٱلْقَبْرِ، وَلَسْنَا نَعْلَمُ أَيْنَ وَضَعُوهُ!».</span>
<span className="v" id="v3" data-v="3" data-xref="إِنْجِيلُ لُوقَا 24:12"> فَخَرَجَ بُطْرُسُ وَٱلتِّلْمِيذُ ٱلْآخَرُ وَأَتَيَا إِلَى ٱلْقَبْرِ.</span>
<span className="v" id="v4" data-v="4" data-xref="صَمُوئِيلَ ٱلثَّانِي 18:23; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 8:12; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 9:24"> وَكَانَ ٱلِٱثْنَانِ يَرْكُضَانِ مَعًا. فَسَبَقَ ٱلتِّلْمِيذُ ٱلْآخَرُ بُطْرُسَ وَجَاءَ أَوَّلًا إِلَى ٱلْقَبْرِ،</span>
<span className="v" id="v5" data-v="5" data-xref="إِنْجِيلُ يُوحَنَّا 19:40; إِنْجِيلُ يُوحَنَّا 20:11; إِنْجِيلُ يُوحَنَّا 11:44"> وَٱنْحَنَى فَنَظَرَ ٱلْأَكْفَانَ مَوْضُوعَةً، وَلَكِنَّهُ لَمْ يَدْخُلْ.</span>
<span className="v" id="v6" data-v="6" data-xref="إِنْجِيلُ يُوحَنَّا 18:17; إِنْجِيلُ يُوحَنَّا 21:7; إِنْجِيلُ مَتَّى 16:15–16"> ثُمَّ جَاءَ سِمْعَانُ بُطْرُسُ يَتْبَعُهُ، وَدَخَلَ ٱلْقَبْرَ وَنَظَرَ ٱلْأَكْفَانَ مَوْضُوعَةً،</span>
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ يُوحَنَّا 11:44"> وَٱلْمِنْدِيلَ ٱلَّذِي كَانَ عَلَى رَأْسِهِ لَيْسَ مَوْضُوعًا مَعَ ٱلْأَكْفَانِ، بَلْ مَلْفُوفًا فِي مَوْضِعٍ وَحْدَهُ.</span>
<span className="v" id="v8" data-v="8" data-xref="إِنْجِيلُ يُوحَنَّا 20:25; إِنْجِيلُ يُوحَنَّا 20:4; إِنْجِيلُ يُوحَنَّا 1:50"> فَحِينَئِذٍ دَخَلَ أَيْضًا ٱلتِّلْمِيذُ ٱلْآخَرُ ٱلَّذِي جَاءَ أَوَّلًا إِلَى ٱلْقَبْرِ، وَرَأَى فَآمَنَ،</span>
<span className="v" id="v9" data-v="9" data-xref="اَلْمَزَامِيرُ 16:10; إِشَعْيَاءَ 25:8; إِشَعْيَاءَ 53:10–12"> لِأَنَّهُمْ لَمْ يَكُونُوا بَعْدُ يَعْرِفُونَ ٱلْكِتَابَ: أَنَّهُ يَنْبَغِي أَنْ يَقُومَ مِنَ ٱلْأَمْوَاتِ.</span>
<span className="v" id="v10" data-v="10" data-xref="إِنْجِيلُ يُوحَنَّا 7:53; إِنْجِيلُ يُوحَنَّا 16:32"> فَمَضَى ٱلتِّلْمِيذَانِ أَيْضًا إِلَى مَوْضِعِهِمَا.</span>

## يسوع يظهر لمريم المجدلية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إِنْجِيلُ يُوحَنَّا 20:5"> أَمَّا مَرْيَمُ فَكَانَتْ وَاقِفَةً عِنْدَ ٱلْقَبْرِ خَارِجًا تَبْكِي. وَفِيمَا هِيَ تَبْكِي ٱنْحَنَتْ إِلَى ٱلْقَبْرِ،</span>
<span className="v" id="v12" data-v="12" data-xref="إِنْجِيلُ لُوقَا 24:3–7; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 3:4; إِنْجِيلُ مَرْقُسَ 16:5–6"> فَنَظَرَتْ مَلَاكَيْنِ بِثِيَابٍ بِيضٍ جَالِسَيْنِ وَاحِدًا عِنْدَ ٱلرَّأْسِ وَٱلْآخَرَ عِنْدَ ٱلرِّجْلَيْنِ، حَيْثُ كَانَ جَسَدُ يَسُوعَ مَوْضُوعًا.</span>
<span className="v" id="v13" data-v="13" data-xref="إِنْجِيلُ يُوحَنَّا 20:15; إِنْجِيلُ يُوحَنَّا 2:4; صَمُوئِيلَ ٱلْأَوَّلُ 1:8"> فَقَالَا لَهَا: «يَا ٱمْرَأَةُ، لِمَاذَا تَبْكِينَ؟». قَالَتْ لَهُمَا: «إِنَّهُمْ أَخَذُوا سَيِّدِي، وَلَسْتُ أَعْلَمُ أَيْنَ وَضَعُوهُ!».</span>
<span className="v" id="v14" data-v="14" data-xref="إِنْجِيلُ لُوقَا 24:16; إِنْجِيلُ يُوحَنَّا 21:4; إِنْجِيلُ مَتَّى 28:9"> وَلَمَّا قَالَتْ هَذَا ٱلْتَفَتَتْ إِلَى ٱلْوَرَاءِ، فَنَظَرَتْ يَسُوعَ وَاقِفًا، وَلَمْ تَعْلَمْ أَنَّهُ يَسُوعُ.</span>
<span className="v" id="v15" data-v="15" data-xref="إِنْجِيلُ يُوحَنَّا 20:13; إِنْجِيلُ يُوحَنَّا 18:7; إِنْجِيلُ يُوحَنَّا 1:38"> قَالَ لَهَا يَسُوعُ: «يَا ٱمْرَأَةُ، لِمَاذَا تَبْكِينَ؟ مَنْ تَطْلُبِينَ؟». فَظَنَّتْ تِلْكَ أَنَّهُ ٱلْبُسْتَانِيُّ، فَقَالَتْ لَهُ: «يَا سَيِّدُ، إِنْ كُنْتَ أَنْتَ قَدْ حَمَلْتَهُ فَقُلْ لِي أَيْنَ وَضَعْتَهُ، وَأَنَا آخُذُهُ».</span>
<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ يُوحَنَّا 10:3; إِنْجِيلُ يُوحَنَّا 1:38; أَعْمَالُ ٱلرُّسُلِ 10:3"> قَالَ لَهَا يَسُوعُ: «يَا مَرْيَمُ». فَٱلْتَفَتَتْ تِلْكَ وَقَالَتْ لَهُ: «رَبُّونِي!» ٱلَّذِي تَفْسِيرُهُ: يَا مُعَلِّمُ.</span>
<span className="v" id="v17" data-v="17" data-xref="إِنْجِيلُ يُوحَنَّا 16:28; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ أَفَسُسَ 4:8–10; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ غَلَاطِيَّةَ 4:6–7"> قَالَ لَهَا يَسُوعُ: «لَا تَلْمِسِينِي لِأَنِّي لَمْ أَصْعَدْ بَعْدُ إِلَى أَبِي. وَلَكِنِ ٱذْهَبِي إِلَى إِخْوَتِي وَقُولِي لَهُمْ: إِنِّي أَصْعَدُ إِلَى أَبِي وَأَبِيكُمْ وَإِلَهِي وَإِلَهِكُمْ».</span>
<span className="v" id="v18" data-v="18" data-xref="إِنْجِيلُ لُوقَا 24:10; إِنْجِيلُ مَتَّى 28:10; إِنْجِيلُ يُوحَنَّا 20:1"> فَجَاءَتْ مَرْيَمُ ٱلْمَجْدَلِيَّةُ وَأَخْبَرَتِ ٱلتَّلَامِيذَ أَنَّهَا رَأَتِ ٱلرَّبَّ، وَأَنَّهُ قَالَ لَهَا هَذَا.</span>
</div>
</div>

## ظهوره للتلاميذ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="إِنْجِيلُ يُوحَنَّا 20:26; إِنْجِيلُ يُوحَنَّا 14:27; إِنْجِيلُ يُوحَنَّا 20:21"> وَلَمَّا كَانَتْ عَشِيَّةُ ذَلِكَ ٱلْيَوْمِ، وَهُوَ أَوَّلُ ٱلْأُسْبُوعِ، وَكَانَتِ ٱلْأَبْوَابُ مُغَلَّقَةً حَيْثُ كَانَ ٱلتَّلَامِيذُ مُجْتَمِعِينَ لِسَبَبِ ٱلْخَوْفِ مِنَ ٱلْيَهُودِ، جَاءَ يَسُوعُ وَوَقَفَ فِي ٱلْوَسْطِ، وَقَالَ لَهُمْ: «سَلَامٌ لَكُمْ!».</span>
<span className="v" id="v20" data-v="20" data-xref="إِنْجِيلُ يُوحَنَّا 16:22; إِنْجِيلُ يُوحَنَّا 16:20; إِنْجِيلُ يُوحَنَّا 20:27"> وَلَمَّا قَالَ هَذَا أَرَاهُمْ يَدَيْهِ وَجَنْبَهُ، فَفَرِحَ ٱلتَّلَامِيذُ إِذْ رَأَوْا ٱلرَّبَّ.</span>
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ يُوحَنَّا 17:18–19; إِنْجِيلُ يُوحَنَّا 13:20; إِنْجِيلُ يُوحَنَّا 3:17"> فَقَالَ لَهُمْ يَسُوعُ أَيْضًا: «سَلَامٌ لَكُمْ! كَمَا أَرْسَلَنِي ٱلْآبُ أُرْسِلُكُمْ أَنَا».</span>
<span className="v" id="v22" data-v="22" data-xref="أَيُّوبَ 33:4; اَلتَّكْوِينُ 2:7; أَعْمَالُ ٱلرُّسُلِ 8:15"> وَلَمَّا قَالَ هَذَا نَفَخَ وَقَالَ لَهُمُ: «ٱقْبَلُوا ٱلرُّوحَ ٱلْقُدُسَ.</span>
<span className="v" id="v23" data-v="23" data-xref="إِنْجِيلُ مَتَّى 18:18; إِنْجِيلُ مَتَّى 16:19; إِنْجِيلُ مَرْقُسَ 2:5–10"> مَنْ غَفَرْتُمْ خَطَايَاهُ تُغْفَرُ لَهُ، وَمَنْ أَمْسَكْتُمْ خَطَايَاهُ أُمْسِكَتْ».</span>
</div>

## ظهوره لتوما

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ يُوحَنَّا 11:16; إِنْجِيلُ يُوحَنَّا 6:66–67; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 10:25"> أَمَّا تُومَا، أَحَدُ ٱلِٱثْنَيْ عَشَرَ، ٱلَّذِي يُقَالُ لَهُ ٱلتَّوْأَمُ، فَلَمْ يَكُنْ مَعَهُمْ حِينَ جَاءَ يَسُوعُ.</span>
<span className="v" id="v25" data-v="25" data-xref="إِنْجِيلُ يُوحَنَّا 6:30; إِنْجِيلُ مَرْقُسَ 16:11; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 3:12"> فَقَالَ لَهُ ٱلتَّلَامِيذُ ٱلْآخَرُونَ: «قَدْ رَأَيْنَا ٱلرَّبَّ!». فَقَالَ لَهُمْ: «إِنْ لَمْ أُبْصِرْ فِي يَدَيْهِ أَثَرَ ٱلْمَسَامِيرِ، وَأَضَعْ إِصْبِعِي فِي أَثَرِ ٱلْمَسَامِيرِ، وَأَضَعْ يَدِي فِي جَنْبِهِ، لَا أُومِنْ».</span>
<span className="v" id="v26" data-v="26" data-xref="إِنْجِيلُ يُوحَنَّا 20:19; إِشَعْيَاءَ 54:10; إِنْجِيلُ يُوحَنَّا 20:21"> وَبَعْدَ ثَمَانِيَةِ أَيَّامٍ كَانَ تَلَامِيذُهُ أَيْضًا دَاخِلًا وَتُومَا مَعَهُمْ. فَجَاءَ يَسُوعُ وَٱلْأَبْوَابُ مُغَلَّقَةٌ، وَوَقَفَ فِي ٱلْوَسْطِ وَقَالَ: «سَلَامٌ لَكُمْ!».</span>
<span className="v" id="v27" data-v="27" data-xref="إِنْجِيلُ يُوحَنَّا 20:25; رِسَالَةُ يُوحَنَّا ٱلرَّسُولِ ٱلْأُولَى 1:1–2; إِنْجِيلُ يُوحَنَّا 20:20"> ثُمَّ قَالَ لِتُومَا: «هَاتِ إِصْبِعَكَ إِلَى هُنَا وَأَبْصِرْ يَدَيَّ، وَهَاتِ يَدَكَ وَضَعْهَا فِي جَنْبِي، وَلَا تَكُنْ غَيْرَ مُؤْمِنٍ بَلْ مُؤْمِنًا».</span>
<span className="v" id="v28" data-v="28" data-xref="إِنْجِيلُ يُوحَنَّا 1:1; إِشَعْيَاءَ 9:6; إِشَعْيَاءَ 7:14"> أَجَابَ تُومَا وَقَالَ لَهُ: «رَبِّي وَإِلَهِي!».</span>
