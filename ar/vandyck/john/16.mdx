## عمل الروح القدس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="إِنْجِيلُ يُوحَنَّا 13:36; إِنْجِيلُ يُوحَنَّا 7:33; إِنْجِيلُ يُوحَنَّا 16:28"> «وَأَمَّا ٱلْآنَ فَأَنَا مَاضٍ إِلَى ٱلَّذِي أَرْسَلَنِي، وَلَيْسَ أَحَدٌ مِنْكُمْ يَسْأَلُنِي: أَيْنَ تَمْضِي؟</span>
<span className="v" id="v6" data-v="6" data-xref="إِنْجِيلُ يُوحَنَّا 14:1; إِنْجِيلُ يُوحَنَّا 16:20–22; إِنْجِيلُ لُوقَا 24:17"> لَكِنْ لِأَنِّي قُلْتُ لَكُمْ هَذَا قَدْ مَلَأَ ٱلْحُزْنُ قُلُوبَكُمْ.</span>
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ يُوحَنَّا 14:26; إِنْجِيلُ يُوحَنَّا 14:16–17; إِنْجِيلُ يُوحَنَّا 15:26"> لَكِنِّي أَقُولُ لَكُمُ ٱلْحَقَّ: إِنَّهُ خَيْرٌ لَكُمْ أَنْ أَنْطَلِقَ، لِأَنَّهُ إِنْ لَمْ أَنْطَلِقْ لَا يَأْتِيكُمُ ٱلْمُعَزِّي، وَلَكِنْ إِنْ ذَهَبْتُ أُرْسِلُهُ إِلَيْكُمْ.</span>
<span className="v" id="v8" data-v="8" data-xref="أَعْمَالُ ٱلرُّسُلِ 2:37; رِسَالَةُ يَهُوذَا 1:15; زَكَريَّا 12:10"> وَمَتَى جَاءَ ذَاكَ يُبَكِّتُ ٱلْعَالَمَ عَلَى خَطِيَّةٍ وَعَلَى بِرٍّ وَعَلَى دَيْنُونَةٍ:</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ يُوحَنَّا 15:22–25; إِنْجِيلُ يُوحَنَّا 8:23–24; إِنْجِيلُ يُوحَنَّا 12:47–48"> أَمَّا عَلَى خَطِيَّةٍ فَلِأَنَّهُمْ لَا يُؤْمِنُونَ بِي،</span>
<span className="v" id="v10" data-v="10" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 5:21; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 3:18; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 10:3–4"> وَأَمَّا عَلَى بِرٍّ فَلِأَنِّي ذَاهِبٌ إِلَى أَبِي وَلَا تَرَوْنَنِي أَيْضًا،</span>
<span className="v" id="v11" data-v="11" data-xref="إِنْجِيلُ يُوحَنَّا 12:31; رِسَالَةُ يُوحَنَّا ٱلرَّسُولِ ٱلْأُولَى 3:8; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 2:14"> وَأَمَّا عَلَى دَيْنُونَةٍ فَلِأَنَّ رَئِيسَ هَذَا ٱلْعَالَمِ قَدْ دِينَ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إِنْجِيلُ مَرْقُسَ 4:33; إِنْجِيلُ يُوحَنَّا 15:15; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 3:1–2"> «إِنَّ لِي أُمُورًا كَثِيرَةً أَيْضًا لِأَقُولَ لَكُمْ، وَلَكِنْ لَا تَسْتَطِيعُونَ أَنْ تَحْتَمِلُوا ٱلْآنَ.</span>
<span className="v" id="v13" data-v="13" data-xref="إِنْجِيلُ يُوحَنَّا 14:26; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 2:10–13; إِنْجِيلُ يُوحَنَّا 14:17"> وَأَمَّا مَتَى جَاءَ ذَاكَ، رُوحُ ٱلْحَقِّ، فَهُوَ يُرْشِدُكُمْ إِلَى جَمِيعِ ٱلْحَقِّ، لِأَنَّهُ لَا يَتَكَلَّمُ مِنْ نَفْسِهِ، بَلْ كُلُّ مَا يَسْمَعُ يَتَكَلَّمُ بِهِ، وَيُخْبِرُكُمْ بِأُمُورٍ آتِيَةٍ.</span>
<span className="v" id="v14" data-v="14" data-xref="رِسَالَةُ يُوحَنَّا ٱلرَّسُولِ ٱلْأُولَى 4:13–14; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 3:14–18; إِنْجِيلُ يُوحَنَّا 15:26"> ذَاكَ يُمَجِّدُنِي، لِأَنَّهُ يَأْخُذُ مِمَّا لِي وَيُخْبِرُكُمْ.</span>

<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ يُوحَنَّا 7:33; أَعْمَالُ ٱلرُّسُلِ 1:3; إِنْجِيلُ يُوحَنَّا 14:18–24"> بَعْدَ قَلِيلٍ لَا تُبْصِرُونَنِي، ثُمَّ بَعْدَ قَلِيلٍ أَيْضًا تَرَوْنَنِي، لِأَنِّي ذَاهِبٌ إِلَى ٱلْآبِ».</span>

## حزن التلاميذ سيتحول إلى فرح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="إِنْجِيلُ يُوحَنَّا 16:16; إِنْجِيلُ مَرْقُسَ 9:32; إِنْجِيلُ يُوحَنَّا 16:5"> فَقَالَ قَوْمٌ مِنْ تَلَامِيذِهِ، بَعْضُهُمْ لِبَعْضٍ: «مَا هُوَ هَذَا ٱلَّذِي يَقُولُهُ لَنَا: بَعْدَ قَلِيلٍ لَا تُبْصِرُونَنِي، ثُمَّ بَعْدَ قَلِيلٍ أَيْضًا تَرَوْنَنِي، وَلِأَنِّي ذَاهِبٌ إِلَى ٱلْآبِ؟».</span>
<span className="v" id="v18" data-v="18" data-xref="إِنْجِيلُ لُوقَا 24:25; إِنْجِيلُ مَتَّى 16:9–11; اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 5:12"> فَقَالُوا: «مَا هُوَ هَذَا ٱلْقَلِيلُ ٱلَّذِي يَقُولُ عَنْهُ؟ لَسْنَا نَعْلَمُ بِمَاذَا يَتَكَلَّمُ!».</span>
<span className="v" id="v19" data-v="19" data-xref="إِنْجِيلُ يُوحَنَّا 16:30; إِنْجِيلُ يُوحَنَّا 16:16; إِنْجِيلُ يُوحَنَّا 14:19"> فَعَلِمَ يَسُوعُ أَنَّهُمْ كَانُوا يُرِيدُونَ أَنْ يَسْأَلُوهُ، فَقَالَ لَهُمْ: «أَعَنْ هَذَا تَتَسَاءَلُونَ فِيمَا بَيْنَكُمْ، لِأَنِّي قُلْتُ: بَعْدَ قَلِيلٍ لَا تُبْصِرُونَنِي، ثُمَّ بَعْدَ قَلِيلٍ أَيْضًا تَرَوْنَنِي</span>
<span className="v" id="v20" data-v="20" data-xref="رِسَالَةُ يَعْقُوبَ 1:2; إِنْجِيلُ مَتَّى 5:4; اَلْمَزَامِيرُ 126:5–6"> اَلْحَقَّ ٱلْحَقَّ أَقُولُ لَكُمْ: إِنَّكُمْ سَتَبْكُونَ وَتَنُوحُونَ وَٱلْعَالَمُ يَفْرَحُ. أَنْتُمْ سَتَحْزَنُونَ، وَلَكِنَّ حُزْنَكُمْ يَتَحَوَّلُ إِلَى فَرَحٍ.</span>
<span className="v" id="v21" data-v="21" data-xref="اَلْمَزَامِيرُ 113:9; إِشَعْيَاءَ 13:8; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ تَسَالُونِيكِي 5:3"> اَلْمَرْأَةُ وَهِيَ تَلِدُ تَحْزَنُ لِأَنَّ سَاعَتَهَا قَدْ جَاءَتْ، وَلَكِنْ مَتَى وَلَدَتِ ٱلطِّفْلَ لَا تَعُودُ تَذْكُرُ ٱلشِّدَّةَ لِسَبَبِ ٱلْفَرَحِ، لِأَنَّهُ قَدْ وُلِدَ إِنْسَانٌ فِي ٱلْعَالَمِ.</span>
<span className="v" id="v22" data-v="22" data-xref="رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 1:8; أَعْمَالُ ٱلرُّسُلِ 13:52; إِنْجِيلُ يُوحَنَّا 16:6"> فَأَنْتُمْ كَذَلِكَ، عِنْدَكُمُ ٱلْآنَ حُزْنٌ. وَلَكِنِّي سَأَرَاكُمْ أَيْضًا فَتَفْرَحُ قُلُوبُكُمْ، وَلَا يَنْزِعُ أَحَدٌ فَرَحَكُمْ مِنْكُمْ</span>
<span className="v" id="v23" data-v="23" data-xref="إِنْجِيلُ يُوحَنَّا 15:7; إِنْجِيلُ يُوحَنَّا 14:13–14; إِنْجِيلُ مَتَّى 21:22"> وَفِي ذَلِكَ ٱلْيَوْمِ لَا تَسْأَلُونَنِي شَيْئًا. اَلْحَقَّ ٱلْحَقَّ أَقُولُ لَكُمْ: إِنَّ كُلَّ مَا طَلَبْتُمْ مِنَ ٱلْآبِ بِٱسْمِي يُعْطِيكُمْ.</span>
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ مَتَّى 7:7–8; رِسَالَةُ يَعْقُوبَ 4:2–3; إِنْجِيلُ يُوحَنَّا 15:11"> إِلَى ٱلْآنَ لَمْ تَطْلُبُوا شَيْئًا بِٱسْمِي. اُطْلُبُوا تَأْخُذُوا، لِيَكُونَ فَرَحُكُمْ كَامِلًا.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إِنْجِيلُ يُوحَنَّا 16:12; إِنْجِيلُ مَرْقُسَ 4:13; إِنْجِيلُ مَتَّى 13:34–35"> «قَدْ كَلَّمْتُكُمْ بِهَذَا بِأَمْثَالٍ، وَلَكِنْ تَأْتِي سَاعَةٌ حِينَ لَا أُكَلِّمُكُمْ أَيْضًا بِأَمْثَالٍ، بَلْ أُخْبِرُكُمْ عَنِ ٱلْآبِ عَلَانِيَةً.</span>
<span className="v" id="v26" data-v="26" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 8:34; إِنْجِيلُ يُوحَنَّا 17:24; إِنْجِيلُ يُوحَنَّا 17:19"> فِي ذَلِكَ ٱلْيَوْمِ تَطْلُبُونَ بِٱسْمِي. وَلَسْتُ أَقُولُ لَكُمْ إِنِّي أَنَا أَسْأَلُ ٱلْآبَ مِنْ أَجْلِكُمْ،</span>
<span className="v" id="v27" data-v="27" data-xref="إِنْجِيلُ يُوحَنَّا 14:21; إِنْجِيلُ يُوحَنَّا 14:23; إِنْجِيلُ يُوحَنَّا 17:23"> لِأَنَّ ٱلْآبَ نَفْسَهُ يُحِبُّكُمْ، لِأَنَّكُمْ قَدْ أَحْبَبْتُمُونِي، وَآمَنْتُمْ أَنِّي مِنْ عِنْدِ ٱللهِ خَرَجْتُ.</span>
<span className="v" id="v28" data-v="28" data-xref="إِنْجِيلُ يُوحَنَّا 13:1; إِنْجِيلُ يُوحَنَّا 13:3; إِنْجِيلُ يُوحَنَّا 8:14"> خَرَجْتُ مِنْ عِنْدِ ٱلْآبِ، وَقَدْ أَتَيْتُ إِلَى ٱلْعَالَمِ، وَأَيْضًا أَتْرُكُ ٱلْعَالَمَ وَأَذْهَبُ إِلَى ٱلْآبِ».</span>
