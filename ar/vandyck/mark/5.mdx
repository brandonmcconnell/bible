## شفاء إنسان به روح نجس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إِنْجِيلُ لُوقَا 8:26–39; إِنْجِيلُ مَتَّى 8:28–34; إِنْجِيلُ مَرْقُسَ 4:35"> وَجَاءُوا إِلَى عَبْرِ ٱلْبَحْرِ إِلَى كُورَةِ ٱلْجَدَرِيِّينَ.</span>
<span className="v" id="v2" data-v="2" data-xref="إِنْجِيلُ مَرْقُسَ 1:23; إِشَعْيَاءَ 65:4; إِنْجِيلُ مَرْقُسَ 5:8"> وَلَمَّا خَرَجَ مِنَ ٱلسَّفِينَةِ لِلْوَقْتِ ٱسْتَقْبَلَهُ مِنَ ٱلْقُبُورِ إِنْسَانٌ بِهِ رُوحٌ نَجِسٌ،</span>
<span className="v" id="v3" data-v="3" data-xref="إِشَعْيَاءَ 65:4; إِنْجِيلُ لُوقَا 8:29; دَانِيآل 4:32–33"> كَانَ مَسْكَنُهُ فِي ٱلْقُبُورِ، وَلَمْ يَقْدِرْ أَحَدٌ أَنْ يَرْبِطَهُ وَلَا بِسَلَاسِلَ،</span>
<span className="v" id="v4" data-v="4" data-xref="رِسَالَةُ يَعْقُوبَ 3:7–8"> لِأَنَّهُ قَدْ رُبِطَ كَثِيرًا بِقُيُودٍ وَسَلَاسِلَ فَقَطَّعَ ٱلسَّلَاسِلَ وَكَسَّرَ ٱلْقُيُودَ، فَلَمْ يَقْدِرْ أَحَدٌ أَنْ يُذَلِّلَهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="أَيُّوبَ 2:7–8; اَلْمُلُوكِ ٱلْأَوَّلُ 18:28; إِنْجِيلُ يُوحَنَّا 8:44"> وَكَانَ دَائِمًا لَيْلًا وَنَهَارًا فِي ٱلْجِبَالِ وَفِي ٱلْقُبُورِ، يَصِيحُ وَيُجَرِّحُ نَفْسَهُ بِٱلْحِجَارَةِ.</span>
<span className="v" id="v6" data-v="6" data-xref="رِسَالَةُ يَعْقُوبَ 2:19; اَلْمَزَامِيرُ 66:3; إِنْجِيلُ لُوقَا 4:41"> فَلَمَّا رَأَى يَسُوعَ مِنْ بَعِيدٍ رَكَضَ وَسَجَدَ لَهُ،</span>
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ مَتَّى 8:29; أَعْمَالُ ٱلرُّسُلِ 16:17; إِنْجِيلُ مَتَّى 26:63"> وَصَرَخَ بِصَوْتٍ عَظِيمٍ وَقَالَ: «مَا لِي وَلَكَ يَا يَسُوعُ ٱبْنَ ٱللهِ ٱلْعَلِيِّ؟ أَسْتَحْلِفُكَ بِٱللهِ أَنْ لَا تُعَذِّبَنِي!».</span>
<span className="v" id="v8" data-v="8" data-xref="إِنْجِيلُ مَرْقُسَ 1:25; إِنْجِيلُ مَرْقُسَ 9:25–26; أَعْمَالُ ٱلرُّسُلِ 16:18"> لِأَنَّهُ قَالَ لَهُ: «ٱخْرُجْ مِنَ ٱلْإِنْسَانِ يَا أَيُّهَا ٱلرُّوحُ ٱلنَّجِسُ».</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ لُوقَا 8:30; إِنْجِيلُ مَرْقُسَ 5:15; إِنْجِيلُ مَتَّى 26:53"> وَسَأَلَهُ: «مَا ٱسْمُكَ؟». فَأَجَابَ قَائِلًا: «ٱسْمِي لَجِئُونُ، لِأَنَّنَا كَثِيرُونَ».</span>
<span className="v" id="v10" data-v="10" data-xref="إِنْجِيلُ مَرْقُسَ 3:22; إِنْجِيلُ مَرْقُسَ 5:13"> وَطَلَبَ إِلَيْهِ كَثِيرًا أَنْ لَا يُرْسِلَهُمْ إِلَى خَارِجِ ٱلْكُورَةِ.</span>
<span className="v" id="v11" data-v="11" data-xref="اَللَّاوِيِّينَ 11:7–8; إِشَعْيَاءَ 65:4; إِشَعْيَاءَ 66:3"> وَكَانَ هُنَاكَ عِنْدَ ٱلْجِبَالِ قَطِيعٌ كَبِيرٌ مِنَ ٱلْخَنَازِيرِ يَرْعَى،</span>
<span className="v" id="v12" data-v="12" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 2:11; أَيُّوبَ 1:10–12; إِنْجِيلُ لُوقَا 22:31–32"> فَطَلَبَ إِلَيْهِ كُلُّ ٱلشَّيَاطِينِ قَائِلِينَ: «أَرْسِلْنَا إِلَى ٱلْخَنَازِيرِ لِنَدْخُلَ فِيهَا».</span>
<span className="v" id="v13" data-v="13" data-xref="اَلْمُلُوكِ ٱلْأَوَّلُ 22:22; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 3:22; أَيُّوبَ 2:6"> فَأَذِنَ لَهُمْ يَسُوعُ لِلْوَقْتِ. فَخَرَجَتِ ٱلْأَرْوَاحُ ٱلنَّجِسَةُ وَدَخَلَتْ فِي ٱلْخَنَازِيرِ، فَٱنْدَفَعَ ٱلْقَطِيعُ مِنْ عَلَى ٱلْجُرْفِ إِلَى ٱلْبَحْرِ. وَكَانَ نَحْوَ أَلْفَيْنِ، فَٱخْتَنَقَ فِي ٱلْبَحْرِ.</span>
<span className="v" id="v14" data-v="14" data-xref="إِنْجِيلُ لُوقَا 8:34; إِنْجِيلُ مَتَّى 8:33"> وَأَمَّا رُعَاةُ ٱلْخَنَازِيرِ فَهَرَبُوا وَأَخْبَرُوا فِي ٱلْمَدِينَةِ وَفِي ٱلضِّيَاعِ. فَخَرَجُوا لِيَرَوْا مَا جَرَى.</span>
<span className="v" id="v15" data-v="15" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ كُولُوسِّي 1:13; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى تِيمُوثَاوُسَ 1:7; صَمُوئِيلَ ٱلْأَوَّلُ 6:20–21"> وَجَاءُوا إِلَى يَسُوعَ فَنَظَرُوا ٱلْمَجْنُونَ ٱلَّذِي كَانَ فِيهِ ٱللَّجِئُونُ جَالِسًا وَلَابِسًا وَعَاقِلًا، فَخَافُوا.</span>
<span className="v" id="v16" data-v="16"> فَحَدَّثَهُمُ ٱلَّذِينَ رَأَوْا كَيْفَ جَرَى لِلْمَجْنُونِ وَعَنِ ٱلْخَنَازِيرِ.</span>
<span className="v" id="v17" data-v="17" data-xref="إِنْجِيلُ مَتَّى 8:34; إِنْجِيلُ مَرْقُسَ 1:24; إِنْجِيلُ لُوقَا 8:37"> فَٱبْتَدَأُوا يَطْلُبُونَ إِلَيْهِ أَنْ يَمْضِيَ مِنْ تُخُومِهِمْ.</span>
<span className="v" id="v18" data-v="18" data-xref="إِنْجِيلُ لُوقَا 8:38–39; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ فِيلِبِّي 1:23–24; إِنْجِيلُ مَرْقُسَ 5:17"> وَلَمَّا دَخَلَ ٱلسَّفِينَةَ طَلَبَ إِلَيْهِ ٱلَّذِي كَانَ مَجْنُونًا أَنْ يَكُونَ مَعَهُ،</span>
<span className="v" id="v19" data-v="19" data-xref="اَلْمَزَامِيرُ 66:16; إِنْجِيلُ يُوحَنَّا 4:29; إِشَعْيَاءَ 38:9–20"> فَلَمْ يَدَعْهُ يَسُوعُ، بَلْ قَالَ لَهُ: «ٱذْهَبْ إِلَى بَيْتِكَ وَإِلَى أَهْلِكَ، وَأَخْبِرْهُمْ كَمْ صَنَعَ ٱلرَّبُّ بِكَ وَرَحِمَكَ».</span>
<span className="v" id="v20" data-v="20" data-xref="إِنْجِيلُ مَرْقُسَ 7:31; إِنْجِيلُ مَتَّى 4:25"> فَمَضَى وَٱبْتَدَأَ يُنَادِي فِي ٱلْعَشْرِ ٱلْمُدُنِ كَمْ صَنَعَ بِهِ يَسُوعُ. فَتَعَجَّبَ ٱلْجَمِيعُ.</span>
</div>

## إقامة ابنة يايرس وشفاء نازفة الدم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ مَتَّى 9:1; إِنْجِيلُ مَرْقُسَ 4:1; إِنْجِيلُ لُوقَا 8:40"> وَلَمَّا ٱجْتَازَ يَسُوعُ فِي ٱلسَّفِينَةِ أَيْضًا إِلَى ٱلْعَبْرِ، ٱجْتَمَعَ إِلَيْهِ جَمْعٌ كَثِيرٌ، وَكَانَ عِنْدَ ٱلْبَحْرِ.</span>
<span className="v" id="v22" data-v="22" data-xref="إِنْجِيلُ لُوقَا 13:14; أَعْمَالُ ٱلرُّسُلِ 18:8; إِنْجِيلُ مَرْقُسَ 5:35–36"> وَإِذَا وَاحِدٌ مِنْ رُؤَسَاءِ ٱلْمَجْمَعِ ٱسْمُهُ يَايِرُسُ جَاءَ. وَلَمَّا رَآهُ خَرَّ عِنْدَ قَدَمَيْهِ،</span>
<span className="v" id="v23" data-v="23" data-xref="إِنْجِيلُ لُوقَا 13:13; إِنْجِيلُ لُوقَا 4:40; إِنْجِيلُ مَرْقُسَ 16:18"> وَطَلَبَ إِلَيْهِ كَثِيرًا قَائِلًا: «ٱبْنَتِي ٱلصَّغِيرَةُ عَلَى آخِرِ نَسَمَةٍ. لَيْتَكَ تَأْتِي وَتَضَعُ يَدَكَ عَلَيْهَا لِتُشْفَى فَتَحْيَا!».</span>
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ مَرْقُسَ 5:31; إِنْجِيلُ مَرْقُسَ 3:20; إِنْجِيلُ لُوقَا 7:6"> فَمَضَى مَعَهُ وَتَبِعَهُ جَمْعٌ كَثِيرٌ وَكَانُوا يَزْحَمُونَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="اَللَّاوِيِّينَ 15:25–27; إِنْجِيلُ لُوقَا 8:43–44; إِنْجِيلُ مَتَّى 9:20–22"> وَٱمْرَأَةٌ بِنَزْفِ دَمٍ مُنْذُ ٱثْنَتَيْ عَشْرَةَ سَنَةً،</span>
<span className="v" id="v26" data-v="26" data-xref="إِرْمِيَا 30:12–13; إِرْمِيَا 8:22; اَلْمَزَامِيرُ 108:12"> وَقَدْ تَأَلَّمَتْ كَثِيرًا مِنْ أَطِبَّاءَ كَثِيرِينَ، وَأَنْفَقَتْ كُلَّ مَا عِنْدَهَا وَلَمْ تَنْتَفِعْ شَيْئًا، بَلْ صَارَتْ إِلَى حَالٍ أَرْدَأَ.</span>
<span className="v" id="v27" data-v="27" data-xref="أَعْمَالُ ٱلرُّسُلِ 19:12; إِنْجِيلُ مَتَّى 14:36; أَعْمَالُ ٱلرُّسُلِ 5:15"> لَمَّا سَمِعَتْ بِيَسُوعَ، جَاءَتْ فِي ٱلْجَمْعِ مِنْ وَرَاءُ، وَمَسَّتْ ثَوْبَهُ،</span>
<span className="v" id="v28" data-v="28"> لِأَنَّهَا قَالَتْ: «إِنْ مَسَسْتُ وَلَوْ ثِيَابَهُ شُفِيتُ».</span>
<span className="v" id="v29" data-v="29" data-xref="إِنْجِيلُ مَرْقُسَ 5:34; اَلْمَزَامِيرُ 30:2; إِنْجِيلُ مَرْقُسَ 3:10"> فَلِلْوَقْتِ جَفَّ يَنْبُوعُ دَمِهَا، وَعَلِمَتْ فِي جِسْمِهَا أَنَّهَا قَدْ بَرِئَتْ مِنَ ٱلدَّاءِ.</span>
<span className="v" id="v30" data-v="30" data-xref="إِنْجِيلُ لُوقَا 6:19; إِنْجِيلُ لُوقَا 8:46; إِنْجِيلُ لُوقَا 5:17"> فَلِلْوَقْتِ ٱلْتَفَتَ يَسُوعُ بَيْنَ ٱلْجَمْعِ شَاعِرًا فِي نَفْسِهِ بِٱلْقُوَّةِ ٱلَّتِي خَرَجَتْ مِنْهُ، وَقَالَ: «مَنْ لَمَسَ ثِيَابِي؟».</span>
<span className="v" id="v31" data-v="31" data-xref="إِنْجِيلُ لُوقَا 8:45; إِنْجِيلُ لُوقَا 9:12"> فَقَالَ لَهُ تَلَامِيذُهُ: «أَنْتَ تَنْظُرُ ٱلْجَمْعَ يَزْحَمُكَ، وَتَقُولُ: مَنْ لَمَسَنِي؟».</span>
<span className="v" id="v32" data-v="32"> وَكَانَ يَنْظُرُ حَوْلَهُ لِيَرَى ٱلَّتِي فَعَلَتْ هَذَا.</span>
<span className="v" id="v33" data-v="33" data-xref="إِنْجِيلُ لُوقَا 8:47; إِنْجِيلُ لُوقَا 1:29; إِنْجِيلُ لُوقَا 1:12"> وَأَمَّا ٱلْمَرْأَةُ فَجَاءَتْ وَهِيَ خَائِفَةٌ وَمُرْتَعِدَةٌ، عَالِمَةً بِمَا حَصَلَ لَهَا، فَخَرَّتْ وَقَالَتْ لَهُ ٱلْحَقَّ كُلَّهُ.</span>
<span className="v" id="v34" data-v="34" data-xref="إِنْجِيلُ لُوقَا 8:48; إِنْجِيلُ مَتَّى 9:22; إِنْجِيلُ لُوقَا 7:50"> فَقَالَ لَهَا: «يَا ٱبْنَةُ، إِيمَانُكِ قَدْ شَفَاكِ، ٱذْهَبِي بِسَلَامٍ وَكُونِي صَحِيحَةً مِنْ دَائِكِ».</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="إِنْجِيلُ لُوقَا 8:49; إِنْجِيلُ يُوحَنَّا 11:39; إِنْجِيلُ مَرْقُسَ 5:22"> وَبَيْنَمَا هُوَ يَتَكَلَّمُ جَاءُوا مِنْ دَارِ رَئِيسِ ٱلْمَجْمَعِ قَائِلِينَ: «ٱبْنَتُكَ مَاتَتْ. لِمَاذَا تُتْعِبُ ٱلْمُعَلِّمَ بَعْدُ؟».</span>

<span className="v" id="v37" data-v="37" data-xref="أَعْمَالُ ٱلرُّسُلِ 9:40; إِنْجِيلُ مَرْقُسَ 14:33; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 13:1"> وَلَمْ يَدَعْ أَحَدًا يَتْبَعُهُ إِلَّا بُطْرُسَ وَيَعْقُوبَ، وَيُوحَنَّا أَخَا يَعْقُوبَ.</span>
