## شفاء في السبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إِنْجِيلُ مَتَّى 12:9–14; إِنْجِيلُ لُوقَا 6:6–11; إِنْجِيلُ مَرْقُسَ 1:21"> ثُمَّ دَخَلَ أَيْضًا إِلَى ٱلْمَجْمَعِ، وَكَانَ هُنَاكَ رَجُلٌ يَدُهُ يَابِسَةٌ.</span>
<span className="v" id="v2" data-v="2" data-xref="اَلْمَزَامِيرُ 37:32; إِنْجِيلُ لُوقَا 14:1; إِنْجِيلُ لُوقَا 6:7"> فَصَارُوا يُرَاقِبُونَهُ: هَلْ يَشْفِيهِ فِي ٱلسَّبْتِ؟ لِكَيْ يَشْتَكُوا عَلَيْهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ فِيلِبِّي 1:28–30; إِشَعْيَاءَ 42:4; إِنْجِيلُ لُوقَا 6:8"> فَقَالَ لِلرَّجُلِ ٱلَّذِي لَهُ ٱلْيَدُ ٱلْيَابِسَةُ: «قُمْ فِي ٱلْوَسْطِ!».</span>
<span className="v" id="v4" data-v="4" data-xref="إِنْجِيلُ لُوقَا 6:9; إِنْجِيلُ مَتَّى 12:10–12; هُوشَع 6:6"> ثُمَّ قَالَ لَهُمْ: «هَلْ يَحِلُّ فِي ٱلسَّبْتِ فِعْلُ ٱلْخَيْرِ أَوْ فِعْلُ ٱلشَّرِّ؟ تَخْلِيصُ نَفْسٍ أَوْ قَتْلٌ؟». فَسَكَتُوا.</span>
<span className="v" id="v5" data-v="5" data-xref="إِنْجِيلُ لُوقَا 6:10; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ أَفَسُسَ 4:18; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 11:25"> فَنَظَرَ حَوْلَهُ إِلَيْهِمْ بِغَضَبٍ، حَزِينًا عَلَى غِلَاظَةِ قُلُوبِهِمْ، وَقَالَ لِلرَّجُلِ: «مُدَّ يَدَكَ». فَمَدَّهَا، فَعَادَتْ يَدُهُ صَحِيحَةً كَٱلْأُخْرَى.</span>
<span className="v" id="v6" data-v="6" data-xref="إِنْجِيلُ مَتَّى 22:16; إِنْجِيلُ مَرْقُسَ 12:13; إِنْجِيلُ مَتَّى 12:14"> فَخَرَجَ ٱلْفَرِّيسِيُّونَ لِلْوَقْتِ مَعَ ٱلْهِيرُودُسِيِّينَ وَتَشَاوَرُوا عَلَيْهِ لِكَيْ يُهْلِكُوهُ.</span>

## الجموع تتبع يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ مَتَّى 4:25; إِنْجِيلُ لُوقَا 6:17; إِنْجِيلُ مَتَّى 12:15"> فَٱنْصَرَفَ يَسُوعُ مَعَ تَلَامِيذِهِ إِلَى ٱلْبَحْرِ، وَتَبِعَهُ جَمْعٌ كَثِيرٌ مِنَ ٱلْجَلِيلِ وَمِنَ ٱلْيَهُودِيَّةِ</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيَال 35:15; إِشَعْيَاءَ 34:5; إِنْجِيلُ مَتَّى 11:21"> وَمِنْ أُورُشَلِيمَ وَمِنْ أَدُومِيَّةَ وَمِنْ عَبْرِ ٱلْأُرْدُنِّ. وَٱلَّذِينَ حَوْلَ صُورَ وَصَيْدَاءَ، جَمْعٌ كَثِيرٌ، إِذْ سَمِعُوا كَمْ صَنَعَ أَتَوْا إِلَيْهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ مَرْقُسَ 5:30; إِنْجِيلُ يُوحَنَّا 6:15"> فَقَالَ لِتَلَامِيذِهِ أَنْ تُلَازِمَهُ سَفِينَةٌ صَغِيرَةٌ لِسَبَبِ ٱلْجَمْعِ، كَيْ لَا يَزْحَمُوهُ،</span>
<span className="v" id="v10" data-v="10" data-xref="إِنْجِيلُ مَتَّى 14:36; إِنْجِيلُ مَرْقُسَ 6:56; إِنْجِيلُ مَتَّى 4:23"> لِأَنَّهُ كَانَ قَدْ شَفَى كَثِيرِينَ، حَتَّى وَقَعَ عَلَيْهِ لِيَلْمِسَهُ كُلُّ مَنْ فِيهِ دَاءٌ.</span>
<span className="v" id="v11" data-v="11" data-xref="إِنْجِيلُ لُوقَا 4:41; إِنْجِيلُ مَرْقُسَ 1:23–24; إِنْجِيلُ لُوقَا 8:28"> وَٱلْأَرْوَاحُ ٱلنَّجِسَةُ حِينَمَا نَظَرَتْهُ خَرَّتْ لَهُ وَصَرَخَتْ قَائِلَةً: «إِنَّكَ أَنْتَ ٱبْنُ ٱللهِ!».</span>
<span className="v" id="v12" data-v="12" data-xref="إِنْجِيلُ مَتَّى 12:16; إِنْجِيلُ مَرْقُسَ 1:25; إِنْجِيلُ مَرْقُسَ 1:34"> وَأَوْصَاهُمْ كَثِيرًا أَنْ لَا يُظْهِرُوهُ.</span>
</div>

## اختيار الرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إِنْجِيلُ لُوقَا 9:1; إِنْجِيلُ لُوقَا 6:12–16; إِنْجِيلُ مَتَّى 10:1–4"> ثُمَّ صَعِدَ إِلَى ٱلْجَبَلِ وَدَعَا ٱلَّذِينَ أَرَادَهُمْ فَذَهَبُوا إِلَيْهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="أَعْمَالُ ٱلرُّسُلِ 1:8; إِنْجِيلُ لُوقَا 9:1–6; إِنْجِيلُ يُوحَنَّا 15:16"> وَأَقَامَ ٱثْنَيْ عَشَرَ لِيَكُونُوا مَعَهُ، وَلِيُرْسِلَهُمْ لِيَكْرِزُوا،</span>
<span className="v" id="v15" data-v="15"> وَيَكُونَ لَهُمْ سُلْطَانٌ عَلَى شِفَاءِ ٱلْأَمْرَاضِ وَإِخْرَاجِ ٱلشَّيَاطِينِ.</span>
<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ يُوحَنَّا 1:42; أَعْمَالُ ٱلرُّسُلِ 1:13; إِنْجِيلُ مَتَّى 16:16–18"> وَجَعَلَ لِسِمْعَانَ ٱسْمَ بُطْرُسَ.</span>
<span className="v" id="v17" data-v="17" data-xref="اَلرِّسَالَةُ إِلَى ٱلْعِبْرَانِيِّينَ 4:12; إِنْجِيلُ مَرْقُسَ 14:33; إِرْمِيَا 23:29"> وَيَعْقُوبَ بْنَ زَبْدِي وَيُوحَنَّا أَخَا يَعْقُوبَ، وَجَعَلَ لَهُمَا ٱسْمَ بُوَانَرْجِسَ أَيِ ٱبْنَيِ ٱلرَّعْدِ.</span>
<span className="v" id="v18" data-v="18" data-xref="أَعْمَالُ ٱلرُّسُلِ 1:13; إِنْجِيلُ مَرْقُسَ 2:14; إِنْجِيلُ مَتَّى 9:9"> وَأَنْدَرَاوُسَ، وَفِيلُبُّسَ، وَبَرْثُولَمَاوُسَ، وَمَتَّى، وَتُومَا، وَيَعْقُوبَ بْنَ حَلْفَى، وَتَدَّاوُسَ، وَسِمْعَانَ ٱلْقَانَوِيَّ،</span>
<span className="v" id="v19" data-v="19" data-xref="إِنْجِيلُ يُوحَنَّا 6:71; إِنْجِيلُ يُوحَنَّا 13:2; أَعْمَالُ ٱلرُّسُلِ 1:16–25"> وَيَهُوذَا ٱلْإِسْخَرْيُوطِيَّ ٱلَّذِي أَسْلَمَهُ. ثُمَّ أَتَوْا إِلَى بَيْتٍ.</span>
</div>

## يسوع وبَعلزَبول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="إِنْجِيلُ مَرْقُسَ 6:31; إِنْجِيلُ يُوحَنَّا 4:31–34; إِنْجِيلُ مَرْقُسَ 9:28"> فَٱجْتَمَعَ أَيْضًا جَمْعٌ حَتَّى لَمْ يَقْدِرُوا وَلَا عَلَى أَكْلِ خُبْزٍ.</span>
<span className="v" id="v21" data-v="21" data-xref="أَعْمَالُ ٱلرُّسُلِ 26:24; إِنْجِيلُ يُوحَنَّا 10:20; إِنْجِيلُ مَرْقُسَ 3:31"> وَلَمَّا سَمِعَ أَقْرِبَاؤُهُ خَرَجُوا لِيُمْسِكُوهُ، لِأَنَّهُمْ قَالُوا: «إِنَّهُ مُخْتَلٌّ!».</span>
<span className="v" id="v22" data-v="22" data-xref="إِنْجِيلُ مَتَّى 15:1; إِنْجِيلُ مَتَّى 10:25; إِنْجِيلُ مَتَّى 9:34"> وَأَمَّا ٱلْكَتَبَةُ ٱلَّذِينَ نَزَلُوا مِنْ أُورُشَلِيمَ فَقَالُوا: «إِنَّ مَعَهُ بَعْلَزَبُولَ! وَإِنَّهُ بِرَئِيسِ ٱلشَّيَاطِينِ يُخْرِجُ ٱلشَّيَاطِينَ».</span>
<span className="v" id="v23" data-v="23" data-xref="إِنْجِيلُ لُوقَا 11:17–23; إِنْجِيلُ مَتَّى 12:25–30; إِنْجِيلُ مَتَّى 13:34"> فَدَعَاهُمْ وَقَالَ لَهُمْ بِأَمْثَالٍ: «كَيْفَ يَقْدِرُ شَيْطَانٌ أَنْ يُخْرِجَ شَيْطَانًا؟</span>
<span className="v" id="v24" data-v="24" data-xref="زَكَريَّا 11:14; إِشَعْيَاءَ 19:2–3; صَمُوئِيلَ ٱلثَّانِي 20:6"> وَإِنِ ٱنْقَسَمَتْ مَمْلَكَةٌ عَلَى ذَاتِهَا لَا تَقْدِرُ تِلْكَ ٱلْمَمْلَكَةُ أَنْ تَثْبُتَ.</span>
<span className="v" id="v25" data-v="25" data-xref="رِسَالَةُ يَعْقُوبَ 3:16; اَلْمَزَامِيرُ 133:1; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ غَلَاطِيَّةَ 5:15"> وَإِنِ ٱنْقَسَمَ بَيْتٌ عَلَى ذَاتِهِ لَا يَقْدِرُ ذَلِكَ ٱلْبَيْتُ أَنْ يَثْبُتَ.</span>
<span className="v" id="v26" data-v="26"> وَإِنْ قَامَ ٱلشَّيْطَانُ عَلَى ذَاتِهِ وَٱنْقَسَمَ لَا يَقْدِرُ أَنْ يَثْبُتَ، بَلْ يَكُونُ لَهُ ٱنْقِضَاءٌ.</span>
<span className="v" id="v27" data-v="27" data-xref="إِشَعْيَاءَ 49:24–26; إِنْجِيلُ مَتَّى 12:29; إِنْجِيلُ لُوقَا 11:21–23"> لَا يَسْتَطِيعُ أَحَدٌ أَنْ يَدْخُلَ بَيْتَ قَوِيٍّ وَيَنْهَبَ أَمْتِعَتَهُ، إِنْ لَمْ يَرْبِطِ ٱلْقَوِيَّ أَوَّلًا، وَحِينَئِذٍ يَنْهَبُ بَيْتَهُ.</span>
<span className="v" id="v28" data-v="28" data-xref="إِنْجِيلُ لُوقَا 12:10; إِنْجِيلُ مَتَّى 12:31–32; رِسَالَةُ يُوحَنَّا ٱلرَّسُولِ ٱلْأُولَى 5:16"> اَلْحَقَّ أَقُولُ لَكُمْ: إِنَّ جَمِيعَ ٱلْخَطَايَا تُغْفَرُ لِبَنِي ٱلْبَشَرِ، وَٱلتَّجَادِيفَ ٱلَّتِي يُجَدِّفُونَهَا.</span>
<span className="v" id="v29" data-v="29" data-xref="إِنْجِيلُ مَتَّى 12:31–32; إِنْجِيلُ لُوقَا 12:10; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ تَسَالُونِيكِي 1:9"> وَلَكِنْ مَنْ جَدَّفَ عَلَى ٱلرُّوحِ ٱلْقُدُسِ فَلَيْسَ لَهُ مَغْفِرَةٌ إِلَى ٱلْأَبَدِ، بَلْ هُوَ مُسْتَوْجِبٌ دَيْنُونَةً أَبَدِيَّةً».</span>
<span className="v" id="v30" data-v="30" data-xref="إِنْجِيلُ يُوحَنَّا 10:20; إِنْجِيلُ مَرْقُسَ 3:22"> لِأَنَّهُمْ قَالُوا: «إِنَّ مَعَهُ رُوحًا نَجِسًا».</span>

## عمل مشيئة الله

<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="إِنْجِيلُ لُوقَا 8:19–21; إِنْجِيلُ مَتَّى 12:46–50"> فَجَاءَتْ حِينَئِذٍ إِخْوَتُهُ وَأُمُّهُ وَوَقَفُوا خَارِجًا وَأَرْسَلُوا إِلَيْهِ يَدْعُونَهُ.</span>
