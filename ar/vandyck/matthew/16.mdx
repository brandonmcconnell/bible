## طلب آية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إِنْجِيلُ لُوقَا 11:16; إِنْجِيلُ مَتَّى 16:6; أَعْمَالُ ٱلرُّسُلِ 4:1"> وَجَاءَ إِلَيْهِ ٱلْفَرِّيسِيُّونَ وَٱلصَّدُّوقِيُّونَ لِيُجَرِّبُوهُ، فَسَأَلُوهُ أَنْ يُرِيَهُمْ آيَةً مِنَ ٱلسَّمَاءِ.</span>
<span className="v" id="v2" data-v="2" data-xref="إِنْجِيلُ لُوقَا 12:54–56"> فَأَجَابَ وَقَالَ لَهُمْ: «إِذَا كَانَ ٱلْمَسَاءُ قُلْتُمْ: صَحْوٌ لِأَنَّ ٱلسَّمَاءَ مُحْمَرَّةٌ.</span>
<span className="v" id="v3" data-v="3" data-xref="إِنْجِيلُ لُوقَا 12:56; أَخْبَارِ ٱلْأَيَّامِ ٱلْأَوَّلُ 12:32; إِنْجِيلُ مَتَّى 11:5"> وَفِي ٱلصَّبَاحِ: ٱلْيَوْمَ شِتَاءٌ لِأَنَّ ٱلسَّمَاءَ مُحْمَرَّةٌ بِعُبُوسَةٍ.يَا مُرَاؤُونَ! تَعْرِفُونَ أَنْ تُمَيِّزُوا وَجْهَ ٱلسَّمَاءِ، وَأَمَّا عَلَامَاتُ ٱلْأَزْمِنَةِ فَلَا تَسْتَطِيعُونَ!</span>
<span className="v" id="v4" data-v="4" data-xref="إِنْجِيلُ لُوقَا 11:29–30; إِنْجِيلُ مَتَّى 12:39–40; يُونَان 1:17"> جِيلٌ شِرِّيرٌ فَاسِقٌ يَلْتَمِسُ آيَةً، وَلَا تُعْطَى لَهُ آيَةٌ إِلَّا آيَةَ يُونَانَ ٱلنَّبِيِّ». ثُمَّ تَرَكَهُمْ وَمَضَى.</span>

## خمير الفريسيين والصدوقيين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="إِنْجِيلُ مَرْقُسَ 8:13–14; إِنْجِيلُ مَتَّى 15:39"> وَلَمَّا جَاءَ تَلَامِيذُهُ إِلَى ٱلْعَبْرِ نَسُوا أَنْ يَأْخُذُوا خُبْزًا.</span>
<span className="v" id="v6" data-v="6" data-xref="إِنْجِيلُ لُوقَا 12:1; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ غَلَاطِيَّةَ 5:9; إِنْجِيلُ مَتَّى 16:12"> وَقَالَ لَهُمْ يَسُوعُ: «ٱنْظُرُوا، وَتَحَرَّزُوا مِنْ خَمِيرِ ٱلْفَرِّيسِيِّينَ وَٱلصَّدُّوقِيِّينَ».</span>
<span className="v" id="v7" data-v="7" data-xref="إِنْجِيلُ مَرْقُسَ 8:16–18; إِنْجِيلُ مَتَّى 15:16–18; أَعْمَالُ ٱلرُّسُلِ 10:14"> فَفَكَّرُوا فِي أَنْفُسِهِمْ قَائِلِينَ: «إِنَّنَا لَمْ نَأْخُذْ خُبْزًا».</span>
<span className="v" id="v8" data-v="8" data-xref="إِنْجِيلُ مَتَّى 6:30; إِنْجِيلُ مَتَّى 14:31; إِنْجِيلُ مَتَّى 8:26"> فَعَلِمَ يَسُوعُ وَقَالَ لَهُمْ: «لِمَاذَا تُفَكِّرُونَ فِي أَنْفُسِكُمْ يا قَلِيلِي ٱلْإِيمَانِ أَنَّكُمْ لَمْ تَأْخُذُوا خُبْزًا؟</span>
<span className="v" id="v9" data-v="9" data-xref="إِنْجِيلُ مَتَّى 14:17–21; إِنْجِيلُ يُوحَنَّا 6:9–13; إِنْجِيلُ لُوقَا 9:13–17"> أَحَتَّى ٱلْآنَ لَا تَفْهَمُونَ؟ وَلَا تَذْكُرُونَ خَمْسَ خُبْزَاتِ ٱلْخَمْسَةِ ٱلْآلَافِ وَكَمْ قُفَّةً أَخَذْتُمْ؟</span>
<span className="v" id="v10" data-v="10" data-xref="إِنْجِيلُ مَتَّى 15:34–38; إِنْجِيلُ مَرْقُسَ 8:17–21; إِنْجِيلُ مَرْقُسَ 8:5–9"> وَلَا سَبْعَ خُبْزَاتِ ٱلْأَرْبَعَةِ ٱلْآلَافِ وَكَمْ سَلًّا أَخَذْتُمْ؟</span>
<span className="v" id="v11" data-v="11" data-xref="إِنْجِيلُ مَتَّى 16:6; إِنْجِيلُ لُوقَا 12:56; إِنْجِيلُ مَرْقُسَ 8:21"> كَيْفَ لَا تَفْهَمُونَ أَنِّي لَيْسَ عَنِ ٱلْخُبْزِ قُلْتُ لَكُمْ أَنْ تَتَحَرَّزُوا مِنْ خَمِيرِ ٱلْفَرِّيسِيِّينَ وَٱلصَّدُّوقِيِّينَ؟».</span>
<span className="v" id="v12" data-v="12" data-xref="إِنْجِيلُ مَتَّى 5:20; أَعْمَالُ ٱلرُّسُلِ 23:8; إِنْجِيلُ مَتَّى 15:4–9"> حِينَئِذٍ فَهِمُوا أَنَّهُ لَمْ يَقُلْ أَنْ يَتَحَرَّزُوا مِنْ خَمِيرِ ٱلْخُبْزِ، بَلْ مِنْ تَعْلِيمِ ٱلْفَرِّيسِيِّينَ وَٱلصَّدُّوقِيِّينَ.</span>
</div>

## اعتراف بطرس بالمسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إِنْجِيلُ لُوقَا 9:18–20; إِنْجِيلُ مَرْقُسَ 8:27–29; دَانِيآل 7:13"> وَلَمَّا جَاءَ يَسُوعُ إِلَى نَوَاحِي قَيْصَرِيَّةِ فِيلُبُّسَ سَأَلَ تَلَامِيذَهُ قَائِلًا: «مَنْ يَقُولُ ٱلنَّاسُ إِنِّي أَنَا ٱبْنُ ٱلْإِنْسَانِ؟».</span>
<span className="v" id="v14" data-v="14" data-xref="إِنْجِيلُ مَرْقُسَ 6:15; إِنْجِيلُ مَتَّى 14:2; مَلَاخِي 4:5"> فَقَالُوا: «قَوْمٌ: يُوحَنَّا ٱلْمَعْمَدَانُ، وَآخَرُونَ: إِيلِيَّا، وَآخَرُونَ: إِرْمِيَا أَوْ وَاحِدٌ مِنَ ٱلْأَنْبِيَاءِ».</span>
<span className="v" id="v15" data-v="15" data-xref="إِنْجِيلُ لُوقَا 9:20; إِنْجِيلُ مَتَّى 13:11; إِنْجِيلُ مَرْقُسَ 8:29"> قَالَ لَهُمْ: «وَأَنْتُمْ، مَنْ تَقُولُونَ إِنِّي أَنَا؟».</span>
<span className="v" id="v16" data-v="16" data-xref="إِنْجِيلُ مَتَّى 14:33; إِنْجِيلُ يُوحَنَّا 11:27; رِسَالَةُ يُوحَنَّا ٱلرَّسُولِ ٱلْأُولَى 4:15"> فَأَجَابَ سِمْعَانُ بُطْرُسُ وَقَالَ: «أَنْتَ هُوَ ٱلْمَسِيحُ ٱبْنُ ٱللهِ ٱلْحَيِّ!».</span>
<span className="v" id="v17" data-v="17" data-xref="إِنْجِيلُ يُوحَنَّا 6:45; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ أَفَسُسَ 1:17–18; إِنْجِيلُ مَتَّى 11:25–27"> فَأجَابَ يَسُوعُ وَقَالَ لَهُ: «طُوبَى لَكَ يا سِمْعَانُ بْنَ يُونَا، إِنَّ لَحْمًا وَدَمًا لَمْ يُعْلِنْ لَكَ، لَكِنَّ أَبِي ٱلَّذِي فِي ٱلسَّمَاوَاتِ.</span>
<span className="v" id="v18" data-v="18" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ أَفَسُسَ 2:19–22; رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلأُولَى إِلَى أَهْلِ كُورِنْثُوسَ 3:9–11; إِشَعْيَاءَ 28:16"> وَأَنَا أَقُولُ لَكَ أَيْضًا: أَنْتَ بُطْرُسُ، وَعَلَى هَذِهِ ٱلصَّخْرَةِ أَبْنِي كَنِيسَتِي، وَأَبْوَابُ ٱلْجَحِيمِ لَنْ تَقْوَى عَلَيْهَا.</span>
<span className="v" id="v19" data-v="19" data-xref="إِنْجِيلُ مَتَّى 18:18; إِشَعْيَاءَ 22:22; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 1:18"> وَأُعْطِيكَ مَفَاتِيحَ مَلَكُوتِ ٱلسَّمَاوَاتِ، فَكُلُّ مَا تَرْبِطُهُ عَلَى ٱلْأَرْضِ يَكُونُ مَرْبُوطًا فِي ٱلسَّمَاوَاتِ. وَكُلُّ مَا تَحُلُّهُ عَلَى ٱلْأَرْضِ يَكُونُ مَحْلُولًا فِي ٱلسَّمَاوَاتِ».</span>
<span className="v" id="v20" data-v="20" data-xref="إِنْجِيلُ مَرْقُسَ 8:30; إِنْجِيلُ لُوقَا 9:21; إِنْجِيلُ يُوحَنَّا 1:41"> حِينَئِذٍ أَوْصَى تَلَامِيذَهُ أَنْ لَا يَقُولُوا لِأَحَدٍ إِنَّهُ يَسُوعُ ٱلْمَسِيحُ.</span>

## يسوع يُنبئ بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="إِنْجِيلُ لُوقَا 24:6–7; إِنْجِيلُ مَتَّى 20:17–19; إِنْجِيلُ مَتَّى 17:22–23"> مِنْ ذَلِكَ ٱلْوَقْتِ ٱبْتَدَأَ يَسُوعُ يُظْهِرُ لِتَلَامِيذِهِ أَنَّهُ يَنْبَغِي أَنْ يَذْهَبَ إِلَى أُورُشَلِيمَ وَيَتَأَلَّمَ كَثِيرًا مِنَ ٱلشُّيُوخِ وَرُؤَسَاءِ ٱلْكَهَنَةِ وَٱلْكَتَبَةِ، وَيُقْتَلَ، وَفِي ٱلْيَوْمِ ٱلثَّالِثِ يَقُومَ.</span>
<span className="v" id="v22" data-v="22" data-xref="إِنْجِيلُ مَرْقُسَ 8:32; إِنْجِيلُ يُوحَنَّا 13:6–8; إِنْجِيلُ مَتَّى 16:16–17"> فَأَخَذَهُ بُطْرُسُ إِلَيْهِ وَٱبْتَدَأَ يَنْتَهِرُهُ قَائِلًا: «حَاشَاكَ يَارَبُّ! لَا يَكُونُ لَكَ هَذَا!».</span>
<span className="v" id="v23" data-v="23" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 8:5–8; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ كُولُوسِّي 3:2; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ فِيلِبِّي 3:19"> فَٱلْتَفَتَ وَقَالَ لِبُطْرُسَ: «ٱذْهَبْ عَنِّي يَاشَيْطَانُ! أَنْتَ مَعْثَرَةٌ لِي، لِأَنَّكَ لَا تَهْتَمُّ بِمَا لِلهِ لَكِنْ بِمَا لِلنَّاسِ».</span>
<span className="v" id="v24" data-v="24" data-xref="إِنْجِيلُ مَتَّى 10:38; إِنْجِيلُ لُوقَا 14:27; رِسَالَةُ بُطْرُسَ ٱلرَّسُولِ ٱلْأُولَى 4:1–2"> حِينَئِذٍ قَالَ يَسُوعُ لِتَلَامِيذِهِ: «إِنْ أَرَادَ أَحَدٌ أَنْ يَأْتِيَ وَرَائِي فَلْيُنْكِرْ نَفْسَهُ وَيَحْمِلْ صَلِيبَهُ وَيَتْبَعْنِي،</span>
<span className="v" id="v25" data-v="25" data-xref="إِنْجِيلُ مَتَّى 10:39; إِنْجِيلُ مَرْقُسَ 8:35; رُؤْيَا يُوحَنَّا ٱللَّاهُوتِيِّ 12:11"> فَإِنَّ مَنْ أَرَادَ أَنْ يُخَلِّصَ نَفْسَهُ يُهْلِكُهَا، وَمَنْ يُهْلِكُ نَفْسَهُ مِنْ أَجْلِي يَجِدُهَا.</span>
<span className="v" id="v26" data-v="26" data-xref="إِنْجِيلُ مَرْقُسَ 8:36–37; أَيُّوبَ 27:8; إِنْجِيلُ لُوقَا 9:25"> لِأَنَّهُ مَاذَا يَنْتَفِعُ ٱلْإِنْسَانُ لَوْ رَبِحَ ٱلْعَالَمَ كُلَّهُ وَخَسِرَ نَفْسَهُ؟ أَوْ مَاذَا يُعْطِي ٱلْإِنْسَانُ فِدَاءً عَنْ نَفْسِهِ؟</span>
<span className="v" id="v27" data-v="27" data-xref="رِسَالَةُ بُولُسَ ٱلرَّسُولِ ٱلثَّانِيةُ إِلَى أَهْلِ كُورِنْثُوسَ 5:10; اَلْمَزَامِيرُ 62:12; رِسَالَةُ بُولُسَ ٱلرَّسُولِ إِلَى أَهْلِ رُومِيَةَ 2:6"> فَإِنَّ ٱبْنَ ٱلْإِنْسَانِ سَوْفَ يَأْتِي فِي مَجْدِ أَبِيهِ مَعَ مَلَائِكَتِهِ، وَحِينَئِذٍ يُجَازِي كُلَّ وَاحِدٍ حَسَبَ عَمَلِهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="إِنْجِيلُ لُوقَا 9:27; إِنْجِيلُ مَرْقُسَ 9:1; إِنْجِيلُ يُوحَنَّا 8:52"> اَلْحَقَّ أَقُولُ لَكُمْ: إِنَّ مِنَ ٱلْقِيَامِ هَهُنَا قَوْمًا لَا يَذُوقُونَ ٱلْمَوْتَ حَتَّى يَرَوْا ٱبْنَ ٱلْإِنْسَانِ آتِيًا فِي مَلَكُوتِهِ».</span>
