## موت إبراهيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أخبار الأيام الأول 1:32–33; التكوين 23:1–2; التكوين 28:1"> وَعَادَ إِبْرَاهِيمُ فَاتَّخَذَ لِنَفْسِهِ زَوْجَةً تُدْعَى قَطُورَةَ،</span>
<span className="v" id="v2" data-v="2" data-xref="أخبار الأيام الأول 1:32–33; العدد 25:17–18; أيوب 2:11"> فَأَنْجَبَتْ لَهُ زِمْرَانَ وَيَقْشَانَ وَمَدَانَ وَمِدْيَانَ وَيِشْبَاقَ وَشُوحاً.</span>
<span className="v" id="v3" data-v="3" data-xref="حزقيال 25:13; إرميا 49:8; حزقيال 27:20"> وَأَنْجَبَ يَقْشَانُ شَبَا وَدَدَانَ. أَمَّا أَبْنَاءُ دَدَانَ فَهُمْ: أَشُّورِيمُ وَلَطُوشِيمُ وَلأُمِّيمُ.</span>
<span className="v" id="v4" data-v="4" data-xref="إشعياء 60:6"> وَأَبْنَاءُ مِدْيَانَ هُمْ: عَيْفَةُ وَعِفْرُ وَحَنُوكُ وَأَبِيدَاعُ، وَأَلْدَعَةُ. وَهَؤُلاءِ جَمِيعاً مِنْ ذُرِّيَّةِ قَطُورَةَ.</span>
<span className="v" id="v5" data-v="5" data-xref="التكوين 24:36; الرسالة إلى العبرانيين 1:2; الرسالة إلى مؤمني غلاطية 3:29"> وَوَرَّثَ إِبْرَاهِيمُ إِسْحاقَ كُلَّ مَالَهُ.</span>
<span className="v" id="v6" data-v="6" data-xref="التكوين 21:14; القضاة 6:3; أعمال الرسل 14:17"> أَمَّا أَبْنَاؤُهُ مِنْ سَرَارِيهِ فَأَعْطَاهُمْ إِبْرَاهِيمُ عَطَايَا، وَصَرَفَهُمْ فِي أَثْنَاءِ حَيَاتِهِ نَحْوَ أَرْضِ الْمَشْرِقِ بَعِيداً عَنْ إِسْحاقَ ابْنِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التكوين 12:4"> وَعَاشَ إِبْرَاهِيمُ مِئَةً وَخَمْساً وَسَبْعِينَ سَنَةً.</span>
<span className="v" id="v8" data-v="8" data-xref="التكوين 35:28–29; التكوين 15:15; أخبار الأيام الأول 29:28"> ثُمَّ مَاتَ بِشَيْبَةٍ صَالِحَةٍ وَانْضَمَّ إِلَى أَسْلافِهِ،</span>
<span className="v" id="v9" data-v="9" data-xref="التكوين 50:13; التكوين 35:29; التكوين 49:29–30"> فَدَفَنَهُ ابْنَاهُ إِسْحاقُ وَإِسْمَاعِيلُ فِي مَغَارَةِ الْمَكْفِيلَةِ، فِي حَقْلِ عِفْرُونَ بْنِ صُوحَرَ الْحِثِّيِّ مُقَابِلَ مَمْرَا،</span>
<span className="v" id="v10" data-v="10" data-xref="التكوين 23:16; التكوين 49:31"> وَهُوَ الْحَقْلُ الَّذِي اشْتَرَاهُ إِبْرَاهِيمُ مِنَ الْحِثِّيِّينَ، وَفِيهِ دُفِنَ إِبْرَاهِيمُ وَزَوْجَتُهُ سَارَةُ.</span>
<span className="v" id="v11" data-v="11" data-xref="التكوين 22:17; التكوين 16:14; التكوين 24:62"> وَبَعْدَ وَفَاةِ إِبْرَاهِيمَ بَارَكَ اللهُ إِسْحاقَ ابْنَهُ، وَأَقَامَ إِسْحاقُ عِنْدَ بِئْرِ لَحَيْ رُئِي.</span>
</div>
</div>

## ذرية إسماعيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المزامير 83:6; التكوين 17:20; التكوين 21:13"> وَهَذَا سِجِلُّ مَوَالِيدِ إِسْمَاعِيلَ بْنِ إِبْرَاهِيمَ الَّذِي أَنْجَبَتْهُ هَاجَرُ الْمِصْرِيَّةُ جَارِيَةُ سَارَةَ لإِبْرَاهِيمَ.</span>
<span className="v" id="v13" data-v="13" data-xref="أخبار الأيام الأول 1:29–31; إشعياء 60:7; نشيد الأنشاد 1:5"> وَهَذِهِ أَسْمَاءُ أَبْنَاءِ إِسْمَاعِيلَ مَدَوَّنَةً حَسَبَ تَرْتِيبِ وِلادَتِهِمْ: نَبَايُوتُ بِكْرُ إِسْمَاعِيلَ، وَقِيدَارُ وَأَدَبْئِيلُ وَمِبْسَامُ،</span>
<span className="v" id="v14" data-v="14" data-xref="إشعياء 21:11; إشعياء 21:16"> وَمِشْمَاعُ وَدُومَةُ وَمَسَّا،</span>
<span className="v" id="v15" data-v="15" data-xref="أخبار الأيام الأول 5:19; أيوب 2:11"> وَحَدَارُ وَتَيْمَا وَيَطُورُ وَنَافِيشُ وَقِدْمَةُ.</span>
<span className="v" id="v16" data-v="16" data-xref="التكوين 17:20; التكوين 17:23"> هَؤُلاءِ هُمْ بَنُو إِسْمَاعِيلَ، وَهَذِهِ هِيَ أَسْمَاؤُهُمْ حَسَبَ دِيَارِهِمْ وَحُصُونِهِمْ، وَقَدْ صَارُوا اثْنَي عَشَرَ رَئِيساً لاثْنَتَيْ عَشْرَةَ قَبِيلَةً.</span>
<span className="v" id="v17" data-v="17" data-xref="التكوين 15:15; التكوين 25:7–8"> وَمَاتَ إِسْمَاعِيلُ وَلَهُ مِنَ الْعُمْرِ مِئَةٌ وَسَبْعٌ وَثَلاثُونَ سَنَةً، وَلَحِقَ بِقَوْمِهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="التكوين 16:12; التكوين 20:1; ملوك الثاني 23:29"> أَمَّا ذُرِّيَّتُهُ فَقَدِ انْتَشَرَتْ مِنْ حَوِيلَةَ إِلَى شُورَ الْمُتَاخِمَةِ لِمِصْرَ فِي اتِّجَاهِ أَشُّورَ، وَكَانَتْ عَلَى عَدَاءٍ مَعَ بَقِيَّةِ إِخْوَتِهَا.</span>
</div>
</div>

## يعقوب وعيسو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه متى 1:2; الإنجيل كما دونه لوقا 3:34; أعمال الرسل 7:8"> وَهَذَا سِجِلُّ مَوَالِيدِ إِسْحاقَ بْنِ إِبْرَاهِيمَ. أَنْجَبَ إِبْرَاهِيمُ إِسْحاقَ.</span>
<span className="v" id="v20" data-v="20" data-xref="التكوين 24:29; التكوين 24:67; التكوين 22:23"> وَكَانَ إِسْحاقُ فِي الأَرْبَعِينَ مِنْ عُمْرِهِ عِنْدَمَا تَزَوَّجَ رِفْقَةَ بِنْتَ بَتُوئِيلَ الأَرَامِيِّ مِنْ سَهْلِ أَرَامَ، وَأُخْتَ لابَانَ الأَرَامِيِّ.</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه لوقا 1:13; إشعياء 58:9; عزرا 8:23"> وَصَلَّى إِسْحاقُ إِلَى الرَّبِّ مِنْ أَجْلِ امْرَأَتِهِ لأَنَّهَا كَانَتْ عَاقِراً، فَاسْتَجَابَ لَهُ الرَّبُّ، فَحَمَلَتْ رِفْقَةُ زَوْجَتُهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="صموئيل الأول 9:9; صموئيل الأول 28:6; صموئيل الأول 30:8"> وَإِذْ تَصَارَعَ الطِّفْلانِ فِي بَطْنِهَا قَالَتْ: «إِنْ كَانَ الأَمْرُ هَكَذَا فَمَالِي وَالْحَبَلُ؟» وَمَضَتْ لِتَسْتَفْهِمَ مِنَ الرَّبِّ</span>
<span className="v" id="v23" data-v="23" data-xref="الرسالة إلى مؤمني روما 9:10–13; التكوين 27:40; التكوين 27:29"> فَقَالَ لَهَا الرَّبُّ: «فِي أَحْشَائِكِ أُمَّتَانِ، يَتَفَرَّعُ مِنْهُمَا شَعْبَانِ. شَعْبٌ يَسْتَقْوِي عَلَى شَعْبٍ، وَكَبِيرٌ يُسْتَعْبَدُ لِصَغِيرٍ».</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24"> وَعِنْدَمَا اكْتَمَلَتْ أَيَّامُهَا لِتَلِدَ إِذَا فِي أَحْشَائِهَا تَوْأَمَانِ.</span>
<span className="v" id="v25" data-v="25" data-xref="التكوين 27:11; التكوين 27:16; التكوين 27:23"> فَخَرَجَ الأَوَّلُ مَكْسُوّاً بِالشَّعْرِ وَكَأَنَّهُ يَرْتَدِي فَرْوَةً حَمْرَاءَ، فَدَعَوْهُ عِيسُو (وَمَعْنَاهُ أَشْعَرُ).</span>
<span className="v" id="v26" data-v="26" data-xref="التكوين 27:36; هوشع 12:3; التكوين 38:28–30"> ثُمَّ خَرَجَ أَخُوهُ وَيَدُهُ قَابِضَةٌ عَلَى عَقِبِ عِيسُو فَدَعَوْهُ يَعْقُوبَ (وَمَعَنَاهُ مُتَعَقِّبٌ). وَكَانَ إِسْحاقُ فِي السِّتِّينَ مِنْ عُمْرِهِ عِنْدَمَا أَنْجَبَتْهُمَا لَهُ رِفْقَةُ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الرسالة إلى العبرانيين 11:9; التكوين 21:20; التكوين 10:9"> وَكَبُرَ الْوَلَدَانِ، فَأَصْبَحَ عِيسُو صَيَّاداً مَاهِراً وَرَجُلَ بَرِّيَّةٍ، بَيْنَمَا كَانَ يَعْقُوبُ رَجُلاً هَادِئاً يُقِيمُ فِي الْخِيَامِ.</span>
<span className="v" id="v28" data-v="28" data-xref="التكوين 27:19; التكوين 27:31; التكوين 27:4"> وَأَحَبَّ إِسْحاقُ عِيسُو لأَنَّهُ كَانَ يَأْكُلُ مِنْ صَيْدِهِ، أَمَّا رِفْقَةُ فَقَدْ أَحَبَّتْ يَعْقُوبَ.</span>
<span className="v" id="v29" data-v="29" data-xref="الأمثال 13:25; إشعياء 40:30–31; صموئيل الأول 14:28"> وَذَاتَ مَرَّةٍ عَادَ عِيسُو مِنَ الْحَقْلِ مُرْهَقاً فَوَجَدَ يَعْقُوبَ قَدْ طَبَخَ طَعَاماً،</span>
<span className="v" id="v30" data-v="30" data-xref="التكوين 36:1; الخروج 15:15; التكوين 36:9"> فَقَالَ عِيسُو لِيَعْقُوبَ: «أَطْعِمْنِي مِنْ هَذَا الطَّبِيخِ الأَحْمَرِ لأَنَّنِي جَائِعٌ جِدّاً». لِهَذَا دُعِيَ عِيسُو بِأَدُومَ.</span>
<span className="v" id="v31" data-v="31"> فَقَالَ يَعْقُوبُ: «بِعْنِي أَوَّلاً امْتِيَازَاتِ بَكُورِيَّتِكَ».</span>
<span className="v" id="v32" data-v="32" data-xref="ملاخي 3:14; أيوب 21:15; أيوب 22:17"> فَقَالَ عِيسُو: «أَنَا لابُدَّ مَائِتٌ، فَأَيُّ نَفْعٍ لِي مِنْ بَكُورِيَّتِي؟»</span>
<span className="v" id="v33" data-v="33" data-xref="الرسالة إلى العبرانيين 12:16; التكوين 27:36; الرسالة إلى العبرانيين 6:16"> فَأَجَابَهُ يَعْقُوبُ: «احْلِفْ لِي أَوَّلًا». فَحَلَفَ لَهُ، وَبَاعَ امْتِيَازَاتِ بَكُورِيَّتِهِ لِيَعْقُوبَ.</span>
<span className="v" id="v34" data-v="34" data-xref="الرسالة إلى مؤمني فيلبي 3:18–19; الرسالة إلى العبرانيين 12:16–17; الرسالة الأولى إلى مؤمني كورنثوس 15:32"> عِنْدَئِذٍ أَعْطَى يَعْقُوبُ عِيسُو خُبْزاً وَطَبِيخَ عَدَسٍ، فَأَكَلَ وَشَرِبَ ثُمَّ قَامَ وَمَضَى فِي سَبِيلِهِ. وَهَكَذَا احْتَقَرَ عِيسُو امْتِيَازَاتِ الْبَكُورِيَّةِ.</span>
</div>
