## استعداد يعقوب للقاء عيسو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المزامير 91:11; المزامير 34:7; الرسالة إلى العبرانيين 1:4"> وَلَمَّا مَضَى يَعْقُوبُ فِي سَبِيلِهِ لاقَتْهُ مَلائِكَةُ اللهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="صموئيل الثاني 2:8; يشوع 5:14; يشوع 21:38"> فَقَالَ يَعْقُوبُ: «هَذَا جُنْدُ اللهِ». فَدَعَا اسْمَ ذَلِكَ الْمكَانِ مَحَنَايِمَ. (وَمَعْنَاهُ: الْمُعَسْكَرَانِ).</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="التكوين 25:30; يشوع 24:4; التثنية 2:5"> وَبَعَثَ يَعْقُوبُ قُدَّامَهُ رُسُلاً إِلَى أَخِيهِ عِيسُو فِي أَرْضِ سَعِيرَ بِلادِ أَدُومَ.</span>
<span className="v" id="v4" data-v="4" data-xref="رسالة بطرس الأولى 3:6; الإنجيل كما دونه لوقا 14:11; التكوين 33:8"> وَأَوْصَاهُمْ قَائِلاً: «هَذَا مَا تَقُولُونَهُ لِسَيِّدِي عِيسُو: هَكَذَا يَقُولُ عَبْدُكَ يَعْقُوبُ: لَقَدْ تَغَرَّبْتُ عِنْدَ لابَانَ وَمَكَثْتُ هُنَاكَ إِلَى الآنَ،</span>
<span className="v" id="v5" data-v="5" data-xref="التكوين 33:8; التكوين 33:15; التكوين 33:11"> وَاقْتَنَيْتُ بَقَراً وَحَمِيراً وَغَنَماً وَعَبِيداً وَإِمَاءً وَأَرْسَلْتُ لأُعْلِمَ سَيِّدِي لَعَلَّنِي أَحْظَى بِرِضَاكَ».</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="التكوين 33:1; التكوين 32:8; التكوين 32:11"> فَرَجَعَ الرُّسُلُ إِلَى يَعْقُوبَ قَائِلِينَ: «لَقَدْ قَدِمْنَا عَلَى أَخِيكَ عِيسُو وَهَا هُوَ مُقْبِلٌ إِلَيْكَ، وَمَعَهُ أَرْبَعُ مِئَةِ رَجُلٍ».</span>
<span className="v" id="v7" data-v="7" data-xref="التكوين 35:3; الرسالة الثانية إلى مؤمني كورنثوس 1:8–10; الخروج 14:10"> فَاعْتَرَى يَعْقُوبَ خَوْفٌ وَكَرَبٌ عَظِيمَانِ وَقَسَّمَ الْقَوْمَ الَّذِينَ مَعَهُ وَالْغَنَمَ وَالْبَقَرَ وَالْجِمَالَ إِلَى جَمَاعَتَيْنِ.</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه متى 10:16; التكوين 33:1–3"> وَقَالَ: «إِنْ صَادَفَ عِيسُو إِحْدَى الْجَمَاعَتَيْنِ وَأَهْلَكَهَا، تَنْجُ الْجَمَاعَةُ الْبَاقِيَةُ».</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="التكوين 31:42; المزامير 34:4–6; التكوين 31:3"> وَصَلَّى يَعْقُوبُ: «يَا إِلَهَ أَبِي إِبْرَاهِيمَ وَإِلَهَ أَبِي إِسْحاقَ، أَيُّهَا الرَّبُّ الَّذِي قَالَ لِي: ارْجِعْ إِلَى أَرْضِكَ وَإِلَى قَوْمِكَ فَأُحْسِنَ إِلَيْكَ.</span>
<span className="v" id="v10" data-v="10" data-xref="المزامير 18:35; التكوين 24:27; صموئيل الثاني 7:18"> أَنَا لَا أَسْتَحِقُّ جَمِيعَ إِحْسَانَاتِكَ وَأَمَانَتَكَ الَّتيِ أَبْدَيْتَهَا نَحْوَ عَبْدِكَ، فَقَدْ عَبَرْتُ الأُرْدُنَّ وَلَيْسَ مَعِي سِوَى عَصَايَ، وَهَا أَنَا أَعُودُ وَقَدْ أَصْبَحْتُ جَيْشَيْنِ.</span>
<span className="v" id="v11" data-v="11" data-xref="المزامير 59:1–2; الأمثال 18:19; الإنجيل كما دونه متى 6:13"> نَجِّنِي مِنْ يَدِ أَخِي عِيسُو لأَنِّي خَائِفٌ أَنْ يَقْدِمَ عَلَيَّ فَيُهْلِكَنِي وَيُهْلِكَ مَعِي الأُمَّهَاتِ وَالْبَنِينَ.</span>
<span className="v" id="v12" data-v="12" data-xref="التكوين 28:13–15; الرسالة إلى العبرانيين 6:17; الرسالة إلى تيطس 1:2"> وَأَنْتَ قُلْتَ: إِنِّي أُحْسِنُ إِلَيْكَ وَأَجْعَلُ ذُرِّيَّتَكَ كَرَمْلِ الْبَحْرِ فَلا تُحْصَى لِكَثْرَتِهَا».</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الأمثال 18:16; التكوين 43:11; الأمثال 19:6"> وَبَاتَ هُنَاكَ تِلْكَ اللَّيْلَةَ، وَانْتَقَى مِمَّا لَدَيْهِ هَدِيَّةً لأَخِيهِ عِيسُو.</span>
<span className="v" id="v14" data-v="14" data-xref="أيوب 1:3; التثنية 8:18; أيوب 42:12"> فَكَانَتْ مِئَتَيْ عَنْزٍ وَعِشْرِينَ تَيْساً وَمِئَتَيْ نَعْجَةٍ وَعِشْرِينَ كَبْشاً،</span>
<span className="v" id="v15" data-v="15"> وَثَلاثِينَ نَاقَةً مُرْضِعَةً مَعَ أَوْلادِهَا، وَأَرْبَعِينَ بَقَرَةً وَعَشَرَةَ ثِيرَانٍ وَعِشْرِينَ أَتَاناً وَعَشَرَةَ حَمِيرٍ،</span>
<span className="v" id="v16" data-v="16" data-xref="التكوين 33:8–9; التكوين 32:20; الأمثال 2:11"> وَعَهِدَ بِها إِلَى أَيْدِي عَبِيدِهِ، كُلِّ قَطِيعٍ عَلَى حِدَةٍ. وَقَالَ لِعَبِيدِهِ: «تَقَدَّمُونِي، وَاجْعَلُوا بَيْنَ كُلِّ قَطِيعٍ وَقَطِيعٍ مَسَافَةً».</span>
<span className="v" id="v17" data-v="17" data-xref="التكوين 33:3"> وَأَوْصَى طَلِيعَتَهُمْ قَائِلاً: «إِذَا لَقِيتَ أَخِي عِيسُو وَسَأَلَكَ: لِمَنْ أَنْتَ؟ وَإِلَى أَيْنَ تَذْهَبُ؟ وَمَنْ هُوَ صَاحِبُ الْقَطِيعِ الَّذِي أَمَامَكَ؟</span>
<span className="v" id="v18" data-v="18" data-xref="التكوين 32:4–5"> أَنَّكَ تُجِيبُ: هِيَ لِعَبْدِكَ يَعْقُوبَ، هَدِيَّةٌ بَعَثَ بِها لِسَيِّدِي عِيسُو. وَهَا هُوَ قَادِمٌ خَلْفَنَا».</span>
<span className="v" id="v19" data-v="19"> وَأَوْصَى أَيْضاً بَقِيَّةَ السَّائِرِينَ وَرَاءَ الْقُطْعَانِ بِمِثْلِ هَذَا الْكَلامِ وَأَضَافَ:</span>
<span className="v" id="v20" data-v="20" data-xref="أيوب 42:8–9; الأمثال 21:14; التكوين 43:11"> «تَقُولُونَ أَيْضاً: هُوَذَا عَبْدُكَ يَعْقُوبُ قَادِمٌ وَرَاءَنَا». وَكَانَ يَعْقُوبُ يَقُولُ فِي نَفْسِهِ: «أَسْتَعْطِفُهُ بِالْهَدَايَا الَّتِي تَتَقَدَّمُنِي، ثُمَّ بَعْدَ ذَلِكَ أُشَاهِدُ وَجْهَهُ لَعَلَّهُ يَرْضَى عَنِّي».</span>
<span className="v" id="v21" data-v="21"> وَهَكَذَا تَقَدَّمَتْهُ هَدَايَاهُ. أَمَّا هُوَ فَقَضَى لَيْلَتَهُ فِي الْمُخَيَّمِ.</span>
</div>
</div>

## يعقوب يصارع مع الله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="التثنية 3:16; التثنية 2:37; الرسالة الأولى إلى تيموثاوس 5:8"> ثُمَّ قَامَ فِي تِلْكَ اللَّيْلَةِ وَصَحِبَ مَعَهُ زَوْجَتَيْهِ وَجَارِيَتَيْهِ وَأَوْلادَهُ الأَحَدَ عَشَرَ، وَعَبَرَ بِهِمْ مَخَاضَةَ يَبُّوقَ،</span>
<span className="v" id="v23" data-v="23"> وَلَمَّا أَجَازَهُمْ وَكُلَّ مَالَهُ عَبْرَ الْوَادِي،</span>
<span className="v" id="v24" data-v="24" data-xref="هوشع 12:3–5; الرسالة إلى مؤمني أفسس 6:12; الإنجيل كما دونه لوقا 13:24"> وَبَقِيَ وَحْدَهُ، صَارَعَهُ إِنْسَانٌ حَتَّى مَطْلَعِ الْفَجْرِ.</span>
<span className="v" id="v25" data-v="25" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 12:7–9; هوشع 12:3–4; الإنجيل كما دونه متى 26:41"> وَعِنْدَمَا رَأَى أَنَّهُ لَمْ يَتَغَلَّبْ عَلَى يَعْقُوبَ، ضَرَبَهُ عَلَى حُقِّ فَخْذِهِ، فَانْخَلَعَ مَفْصِلُ فَخْذِ يَعْقُوبَ فِي مُصَارَعَتِهِ مَعَهُ.</span>
<span className="v" id="v26" data-v="26" data-xref="هوشع 12:4; المزامير 115:12–13; الرسالة الثانية إلى مؤمني كورنثوس 12:8–9"> وَقَالَ لَهُ: «أَطْلِقْنِي، فَقَدْ طَلَعَ الْفَجْرُ». فَأَجَابَهُ يَعْقُوبُ: «لا أُطْلِقُكَ حَتَّى تُبَارِكَنِي».</span>
<span className="v" id="v27" data-v="27"> فَسَأَلَهُ: «مَا اسْمُكَ؟» فَأَجَابَ: «يَعْقُوبُ».</span>
<span className="v" id="v28" data-v="28" data-xref="رؤيا يوحنا 2:17; التكوين 35:10; إشعياء 62:2–4"> فَقَالَ: «لا يُدْعَى اسْمُكَ فِيمَا بَعْدُ يَعْقُوبَ، بَلْ إِسْرَائِيلَ (وَمَعْنَاهُ: يُجَاهِدُ مَعَ اللهِ)، لأَنَّكَ جَاهَدْتَ مَعَ اللهِ وَالنَّاسِ وَقَدَرْتَ».</span>
<span className="v" id="v29" data-v="29" data-xref="إشعياء 9:6; القضاة 13:16–18; أيوب 11:7"> فَسَأَلَهُ يَعْقُوبُ: «أَخْبِرْنِي مَا اسْمُكَ؟» فَقَالَ: «لِمَاذَا تَسْأَلُ عَنِ اسْمِي؟» وَبَارَكَهُ هُنَاكَ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="الخروج 24:10–11; العدد 12:8; إشعياء 6:5"> وَدَعَا يَعْقُوبُ اسْمَ الْمَكَانِ فَنِيئِيلَ (وَمَعْنَاهُ: وَجْهُ اللهِ) إِذْ قَالَ: «لأَنِّي شَاهَدْتُ اللهَ وَجْهاً لِوَجْهٍ وَبَقِيتُ حَيًّا».</span>
<span className="v" id="v31" data-v="31" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 12:7; التكوين 19:15; القضاة 8:8"> وَمَا إِنْ عَبَرَ فَنِيئِيلَ حَتَّى أَشْرَقَتْ عَلَيْهِ الشَّمْسُ فَسَارَ وَهُوَ عَارِجٌ مِنْ فَخْذِهِ</span>
<span className="v" id="v32" data-v="32" data-xref="صموئيل الأول 5:5"> لِذَلِكَ يَمْتَنِعُ بَنُو إِسْرَائِيلَ عَنْ أَكْلِ عِرْقِ النَّسَا الَّذِي عَلَى حُقِّ الْفَخْذِ إِلَى هَذَا الْيَوْمِ، لأَنَّ الرَّجُلَ ضَرَبَ حُقَّ فَخْذِ يَعْقُوبَ عَلَى عِرْقِ النَّسَا.</span>
</div>
