## رجوع يعقوب إلى بيت إيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التكوين 31:13; هوشع 12:4; المزامير 91:15"> ثُمَّ قَالَ اللهُ لِيَعْقُوبَ: «اصْعَدْ إِلَى بَيْتِ إِيلَ وَاسْكُنْ هُنَاكَ، وَشَيِّدْ مَذْبَحاً لِلهِ الَّذِي ظَهَرَ لَكَ عِنْدَمَا كُنْتَ هَارِباً مِنْ أَمَامِ أَخِيكَ عِيسُو».</span>
<span className="v" id="v2" data-v="2" data-xref="الخروج 19:14; الخروج 19:10; حزقيال 18:31"> فَأَمَرَ يَعْقُوبُ أَهْلَ بَيْتِهِ، وَكُلَّ مَنْ كَانَ مَعَهُ: «تَخَلَّصُوا مِنَ الآلِهَةِ الْغَرِيبَةِ الَّتِي بَيْنَكُمْ، وَتَطَهَّرُوا وَأَبْدِلُوا ثِيَابَكُمْ،</span>
<span className="v" id="v3" data-v="3" data-xref="إشعياء 43:2; التكوين 32:7; التكوين 31:3"> ثُمَّ تَعَالَوْا لِنَذْهَبَ إِلَى بَيْتِ إِيلَ لأُشَيِّدَ هُنَاكَ مَذْبَحاً لِلهِ الَّذِي اسْتَجَابَ لِي فِي يَوْمِ ضِيقَتِي، وَرَافَقَنِي فِي الطَّرِيقِ الَّتِي سَلَكْتُهَا».</span>
<span className="v" id="v4" data-v="4" data-xref="القضاة 9:6; التثنية 7:25; الخروج 32:20"> فَسَلَّمُوا يَعْقُوبَ كُلَّ الآلِهَةِ الْغَرِيبَةِ الَّتِي كَانَتْ لَدَيْهِمْ وَالأَقْرَاطَ الَّتِي فِي آذَانِهِمْ، فَطَمَرَ هَا يَعْقُوبُ تَحْتَ الْبُطْمَةِ الَّتِي فِي شَكِيمَ.</span>
<span className="v" id="v5" data-v="5" data-xref="الخروج 23:27; أخبار الأيام الثاني 17:10; يشوع 5:1"> ثُمَّ ارْتَحَلُوا، فَهَيْمَنَ رُعْبُ اللهِ عَلَى الْمُدُنِ الَّتِي حَوْلَهُمْ، فَلَمْ يَسْعَوْا وَرَاءَهُمْ</span>
<span className="v" id="v6" data-v="6" data-xref="التكوين 28:19; التكوين 28:22; التكوين 12:8"> فَوَصَلَ يَعْقُوبُ وَجَمِيعُ الْقَوْمِ الَّذِينَ مَعَهُ إِلَى لُوزَ فِي أَرْضِ الْكَنْعَانِيِّينَ، وَهِيَ نَفْسُهَا بَيْتُ إِيلَ.</span>
<span className="v" id="v7" data-v="7" data-xref="التكوين 35:3; التكوين 35:1; التكوين 28:13"> وَشَيَّدَ مَذْبَحاً هُنَاكَ، وَدَعَا الْمَكَانَ «بَيْتَ إِيلَ» لأَنَّ اللهَ تَجَلَّى لَهُ هُنَاكَ عِنْدَمَا كَانَ هَارِباً مِنْ أَمَامِ أَخِيهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التكوين 24:59; صموئيل الأول 31:13; القضاة 2:5"> وَمَاتَتْ هُنَاكَ دَبُورَةُ مُرْضِعَةُ رِفْقَةَ، فَدُفِنَتْ فِي مُنْخَفَضِ بَيْتِ إِيلَ تَحْتَ شَجَرَةِ الْبَلُّوطِ، وَسَمُّوهَا «أَلُّونَ بَاكُوتَ» (وَمَعْنَاهَا: بَلُّوطَةُ الْبُكَاءِ).</span>
<span className="v" id="v9" data-v="9" data-xref="التكوين 35:1; التكوين 18:1; التكوين 26:2"> وَظَهَرَ اللهُ لِيَعْقُوبَ مَرَّةً أُخْرَى بَعْدَ رُجُوعِهِ مِنْ سَهْلِ أَرَامَ وَبَارَكَهُ،</span>
<span className="v" id="v10" data-v="10" data-xref="التكوين 17:5; التكوين 32:27–28; التكوين 17:15"> وَقَالَ لَهُ: «لَنْ يُدْعَى اسْمُكَ يَعْقُوبَ فِي مَا بَعْدُ، بَلْ إِسْرَائِيلَ» (وَمَعْنَاهُ: يُجَاهِدُ مَعَ اللهِ). وَهَكَذَا سَمَّاهُ إِسْرَائِيلَ.</span>
<span className="v" id="v11" data-v="11" data-xref="التكوين 17:1; التكوين 9:1; التكوين 22:17"> وَقَالَ اللهُ لَهُ: «أَنَا هُوَ اللهُ الْقَدِيرُ. أَثْمِرْ وَاكْثُرْ، فَيَكُونَ مِنْكَ أُمَّةٌ وَطَوَائِفُ أُمَمٍ، وَمِنْ صُلْبِكَ يَخْرُجُ مُلُوكٌ.</span>
<span className="v" id="v12" data-v="12" data-xref="التكوين 12:7; التكوين 28:13; التكوين 26:3–4"> وَالأَرْضُ الَّتِي وَهَبْتُهَا لإِبْرَاهِيمَ وَإِسْحاقَ أُعْطِيهَا لَكَ وَلِذُرِّيَّتِكَ مِنْ بَعْدِكَ أَيْضاً».</span>
<span className="v" id="v13" data-v="13" data-xref="التكوين 17:22; التكوين 18:33; القضاة 6:21"> ثُمَّ فَارَقَهُ اللهُ فِي الْمَكَانِ الَّذِي خَاطَبَهُ فِيهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="التكوين 28:18–19; صموئيل الأول 7:12; الخروج 17:15"> وَأَقَامَ يَعْقُوبُ عَمُوداً مِنْ حَجَرٍ فِي الْمَكَانِ الَّذِي تَكَلَّمَ فِيهِ مَعَهُ، وَسَكَبَ عَلَيْهِ سَكِيبَ قُرْبَانٍ وَصَبَّ عَلَيْهِ زَيْتاً أَيْضاً.</span>
<span className="v" id="v15" data-v="15" data-xref="التكوين 28:19"> وَدَعَا اسْمَ الْمَكَانِ «بَيْتَ إِيلَ» (وَمَعْنَاهُ: بَيْتُ اللهِ) لأَنَّ اللهَ خَاطَبَهُ هُنَاكَ.</span>
</div>
</div>

## موت راحيل وإسحاق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ميخا 5:2; التكوين 48:7; ملوك الثاني 5:19"> ثُمَّ ارْتَحَلُوا مِنْ بَيْتِ إِيلَ. وَإِذْ كَانُوا بَعْدُ عَلَى مَسَافَةٍ مِنْ أَفْرَاتَةَ شَعَرَتْ رَاحِيلُ بِالْمَخَاضِ وَتَعَسَّرَتْ وِلادَتُهَا.</span>
<span className="v" id="v17" data-v="17" data-xref="التكوين 30:24; صموئيل الأول 4:19–21"> وَإِذْ كَانَتْ تُقَاسِي فِي وِلادَتِهَا قَالَتْ لَهَا الْقَابِلَةُ: «لا تَخَافِي، فَإِنَّ هَذَا أَيْضاً ابْنٌ آخَرُ لَكِ».</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه لوقا 12:20; أعمال الرسل 7:59; الإنجيل كما دونه لوقا 23:46"> وَبَيْنَمَا كَانَتْ تَلْفِظُ أَنْفَاسَهَا عِنْدَ مَوْتِهَا دَعَتْهُ «بِنْ أُوْنِي» (وَمَعْنَاهُ: ابْنُ حُزْنِي) غَيْرَ أَنَّ أَبَاهُ دَعَاهُ «بِنْيَامِينَ» (وَمَعْنَاهُ: ابْنُ يَمِينِي).</span>
<span className="v" id="v19" data-v="19" data-xref="ميخا 5:2; التكوين 48:7; راعوث 4:11"> ثُمَّ مَاتَتْ رَاحِيلُ وَدُفِنَتْ فِي الطَّرِيقِ الْمُؤَدِّيَةِ إِلَى أَفْرَاتَةَ، أَيْ بَيْتِ لَحْمٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="صموئيل الأول 10:2; التكوين 35:14; صموئيل الثاني 18:17–18"> وَأَقَامَ يَعْقُوبُ عَمُوداً عَلَى قَبْرِهَا، وَهُوَ الْمَعْرُوفُ بِعَمُودِ قَبْرِ رَاحِيلَ إِلَى الْيَوْمِ.</span>
<span className="v" id="v21" data-v="21" data-xref="ميخا 4:8; الإنجيل كما دونه لوقا 2:8"> وَتَابَعَ إِسْرَائِيلُ رَحِيلَهُ وَنَصَبَ خِيَامَهُ وَرَاءَ «بُرْجِ عِدْرٍ»</span>
<span className="v" id="v22" data-v="22" data-xref="أخبار الأيام الأول 5:1; اللاويين 18:8; صموئيل الثاني 20:3"> وَبَيْنَمَا كَانَ إِسْرَائِيلُ يُقِيمُ فِي تِلْكَ الأَرْضِ مَضَى رَأُوبَيْنُ وَضَاجَعَ بِلْهَةَ سُرِّيَّةَ أَبِيهِ. وَعَرَفَ إِسْرَائِيلُ بِالأَمْرِ. وَهَؤُلاءِ هُمْ أَبْنَاءُ يَعْقُوبَ الاثْنَا عَشَرَ:</span>
<span className="v" id="v23" data-v="23" data-xref="التكوين 30:18–20; التكوين 46:8–15; التكوين 29:32–35"> أَبْنَاءُ لَيْئَةَ: رَأُوبَيْنُ بِكْرُ يَعْقُوبَ، وَشِمْعُونُ وَلاوِي وَيَهُوذَا وَيَسَّاكَرُ وَزَبُولُونُ.</span>
<span className="v" id="v24" data-v="24" data-xref="التكوين 30:22–24; التكوين 46:19–22; التكوين 35:16–18"> وَابْنَا رَاحِيلَ: يُوسُفُ وَبَنْيَامِينُ.</span>
<span className="v" id="v25" data-v="25" data-xref="التكوين 46:23–25; التكوين 37:2; التكوين 30:4–8"> وَابْنَا بِلْهَةَ جَارِيَةِ رَاحِيلَ: دَانٌ وَنَفْتَالِي.</span>
<span className="v" id="v26" data-v="26" data-xref="التكوين 30:9–13; التكوين 46:16–18; التكوين 25:20"> وَابْنَا زِلْفَةَ جَارِيَةِ لَيْئَةَ: جَادٌ وَأَشِيرُ. وَهَؤُلاءِ هُمْ أَوْلادُ يَعْقُوبَ الَّذِينَ وُلِدُوا لَهُ فِي سَهْلِ أَرَامَ.</span>
<span className="v" id="v27" data-v="27" data-xref="التكوين 13:18; التكوين 23:19; التكوين 18:1"> وَقَدِمَ يَعْقُوبُ عَلَى إِسْحاقَ أَبِيهِ إِلَى مَمْرَا فِي قَرْيَةِ أَرْبَعَ الْمَعْرُوفَةِ بِحَبْرُونَ حَيْثُ تَغَرَّبَ إِبْرَاهِيمُ وَإِسْحاقُ.</span>
<span className="v" id="v28" data-v="28" data-xref="التكوين 25:7; التكوين 50:26; التكوين 47:28"> وَعَاشَ إِسْحاقُ مِئَةً وَثَمَانِينَ سَنَةً،</span>
<span className="v" id="v29" data-v="29" data-xref="التكوين 15:15; التكوين 49:33; التكوين 49:31"> ثُمَّ أَسْلَمَ رُوحَهُ وَلَحِقَ بِقَوْمِهِ شَيْخاً طَاعِناً فِي السِّنِّ وَدَفَنَهُ ابْنَاهُ عِيسُو وَيَعْقُوبُ.</span>
