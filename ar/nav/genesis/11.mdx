## برج بابل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أعمال الرسل 2:6; صفنيا 3:9; إشعياء 19:18"> وَكَانَ أَهْلُ الأَرْضِ جَمِيعاً يَتَكَلَّمُونَ أَوَّلاً بِلِسَانٍ وَاحِدٍ وَلُغَةٍ وَاحِدَةٍ.</span>
<span className="v" id="v2" data-v="2" data-xref="التكوين 10:10; دانيال 1:2; التكوين 14:1"> وَإِذِ ارْتَحَلُوا شَرْقاً وَجَدُوا سَهْلاً فِي أَرْضِ شِنْعَارَ فَاسْتَوْطَنُوا هُنَاكَ.</span>
<span className="v" id="v3" data-v="3" data-xref="التكوين 14:10; الخروج 1:14; الخروج 2:3"> فَقَالَ بَعْضُهُمْ لِبَعْضٍ: «لِنَصْنَعْ طُوباً مَشْوِيًّا أَحْسَنَ شَيٍّ». فَاسْتَخْدَمُوا الطُّوبَ بَدِيلاً لِلْحِجَارَةِ بِالطُّوبِ، وِالْحُمْرَ بَدِيلاً للطِّينِ.</span>
<span className="v" id="v4" data-v="4" data-xref="التثنية 1:28; الإنجيل كما دونه يوحنا 5:44; الإنجيل كما دونه لوقا 1:51"> ثُمَّ قَالُوا: «لِنُشَيِّدْ لأَنْفُسِنَا مَدِينَةً وَبُرْجاً يَبْلُغُ رَأْسُهُ السَّمَاءَ، فَنُخَلِّدَ لَنَا اسْماً لِئَلّا نَتَشَتَّتَ عَلَى وَجْهِ الأَرْضِ كُلِّهَا».</span>
<span className="v" id="v5" data-v="5" data-xref="التكوين 18:21; المزامير 33:13–14; الإنجيل كما دونه يوحنا 3:13"> وَنَزَلَ الرَّبُّ لِيَشْهَدَ الْمَدِينَةَ وَالْبُرْجَ اللَّذَيْنِ شَرَعَ بَنُو الْبَشَرِ فِي بِنَائِهِمَا.</span>
<span className="v" id="v6" data-v="6" data-xref="المزامير 2:1–4; التكوين 3:22; التكوين 11:1"> فَقَالَ الرَّبُّ: «إِنْ كَانُوا، كَشَعْبٍ وَاحِدٍ يَنْطِقُونَ بِلُغَةٍ وَاحِدَةٍ، قَدْ عَمِلُوا هَذَا مُنْذُ أَوَّلِ الأَمْرِ، فَلَنْ يَمْتَنِعَ إِذاً عَلَيْهِمْ أَيُّ شَيْءٍ عَزَمُوا عَلَى فِعْلِهِ.</span>
<span className="v" id="v7" data-v="7" data-xref="المزامير 33:10; المزامير 55:9; أيوب 5:12–13"> هَيَّا نَنْزِلْ إِلَيْهِمْ وَنُبَلْبِلْ لِسَانَهُمْ، حَتَّى لَا يَفْهَمَ بَعْضُهُمْ كَلامَ بَعْضٍ».</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه لوقا 1:51; التثنية 32:8; التكوين 10:25"> وَهَكَذَا شَتَّتَهُمُ الرَّبُّ مِنْ هُنَاكَ عَلَى سَطْحِ الأَرْضِ كُلِّهَا، فَكَفُّوا عَنْ بِنَاءِ الْمَدِينَةِ،</span>
<span className="v" id="v9" data-v="9" data-xref="التكوين 10:10; التكوين 10:5; التكوين 10:20"> لِذَلِكَ سُمِّيَتِ الْمَدِينَةُ «بَابِلَ» لأَنَّ الرَّبَّ بَلْبَلَ لِسَانَ أَهْلِ كُلِّ الأَرْضِ، وَبِالتَّالِي شَتَّتَهُمْ مِنْ هُنَاكَ فِي أَرْجَاءِ الأَرْضِ كُلِّهَا.</span>
</div>

## من سام إلى أبرام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="أخبار الأيام الأول 1:17–27; التكوين 10:21–22; الإنجيل كما دونه لوقا 3:34–36"> وَهَذَا سِجِلُّ مَوَالِيدِ سَامٍ. لَمَّا كَانَ سَامٌ ابْنَ مِئَةِ سَنَةٍ وَلَدَ أَرْفَكْشَادَ بَعْدَ الطُّوفَانِ بِسَنَتَيْنِ.</span>
<span className="v" id="v11" data-v="11" data-xref="التكوين 9:7; المزامير 144:12; التكوين 1:28"> وَعَاشَ سَامٌ بَعْدَ ذَلِكَ خَمْسَ مِئَةِ سَنَةٍ، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه لوقا 3:36"> وَعِنْدَمَا بَلَغَ أَرْفَكْشَادُ خَمْساً وَثَلاثِينَ سَنَةً مِنَ الْعُمْرِ وَلَدَ شَالَحَ.</span>
<span className="v" id="v13" data-v="13"> وَعَاشَ بَعْدَ ذَلِكَ أَرْبَعَ مِئَةٍ وَثَلاثَ سَنَوَاتٍ، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v14" data-v="14"> وَكَانَ شَالَحُ فِي الثَّلاثِينَ مِنْ عُمْرِهِ عِنْدَمَا وَلَدَ عَابِرَ.</span>
<span className="v" id="v15" data-v="15"> وَعَاشَ بَعْدَ ذَلِكَ أَرْبَعَ مِئَةٍ وَثَلاثَ سَنَوَاتٍ، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v16" data-v="16" data-xref="التكوين 10:21; أخبار الأيام الأول 1:19; العدد 24:24"> وَكَانَ عُمْرُ عَابِرَ أَرْبَعاً وَثَلاثِينَ سَنَةً عِنْدَمَا وَلَدَ فَالَجَ.</span>
<span className="v" id="v17" data-v="17"> وَعَاشَ عَابِرُ بَعْدَ ذَلِكَ أَرْبَعَ مِئَةٍ وَثَلاثِينَ سَنَةً، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه لوقا 3:35"> وَكَانَ عُمْرُ فَالَجَ ثَلاثِينَ سَنَةً عِنْدَمَا وَلَدَ رَعُوَ.</span>
<span className="v" id="v19" data-v="19"> وَعَاشَ فَالَجُ بَعْدَ ذَلِكَ مِئَتَيْنِ وَتِسْعَ سِنِينَ وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه لوقا 3:35"> وَكَانَ عُمْرُ رَعُوَ اثْنَتَيْنِ وَثَلاثِينَ سَنَةً عِنْدَمَا وَلَدَ سَرُوجَ.</span>
<span className="v" id="v21" data-v="21"> وَعَاشَ رَعُو بَعْدَ ذَلِكَ مِئَتَيْنِ وَسَبْعَ سِنِينَ، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v22" data-v="22" data-xref="يشوع 24:2"> وَكَانَ عُمْرُ سَرُوجَ ثَلاثِينَ سَنَةً عِنْدَمَا وَلَدَ نَاحُورَ.</span>
<span className="v" id="v23" data-v="23"> وَعَاشَ سَرُوجُ بَعْدَ ذَلِكَ مِئَتَيْ سَنَةٍ، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه لوقا 3:34; يشوع 24:2"> وَكَانَ عُمْرُ نَاحُورَ تِسْعاً وَعِشْرِينَ سَنَةً عِنْدَمَا وَلَدَ تَارَحَ.</span>
<span className="v" id="v25" data-v="25"> وَعَاشَ نَاحُورُ بَعْدَ ذَلِكَ مِئَةً وَتِسْعَ عَشْرَةَ سَنَةً، وُلِدَ لَهُ فِيهَا بَنُونَ وَبَنَاتٌ.</span>
<span className="v" id="v26" data-v="26" data-xref="يشوع 24:2; أخبار الأيام الأول 1:26–27; التكوين 22:20–24"> وَعِنْدَمَا بَلَغَ تَارَحُ السَّبْعِينَ مِنْ عُمْرِهِ أَنْجَبَ أَبْرَامَ وَنَاحُورَ وَهَارَانَ.</span>
</div>
</div>

## سلسلة نسب أبرام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التكوين 12:4; التكوين 14:12; التكوين 11:31"> وَهَذَا هُوَ سِجِلُّ مَوَالِيدِ تَارَحَ: وَلَدَ تَارَحُ أَبْرَامَ وَنَاحُورَ وَهَارَانَ. وَوَلَدَ هَارَانُ لُوطاً.</span>
<span className="v" id="v28" data-v="28" data-xref="التكوين 15:7; نحميا 9:7; أعمال الرسل 7:2–4"> وَمَاتَ هَارَانُ قَبْلَ تَارَحَ أَبِيهِ فِي أَرْضِ مَوْلِدِهِ فِي أُورِ الْكَلْدَانِيِّينَ.</span>
<span className="v" id="v29" data-v="29" data-xref="التكوين 17:15; التكوين 22:20; التكوين 24:15"> وَتَزَوَّجَ كُلٌّ مِنْ أَبْرَامَ وَنَاحُورَ. وَكَانَ اسْمُ زَوْجَةِ أَبْرَامَ سَارَايَ، وَاسْمُ زَوْجَةِ نَاحُورَ مِلْكَةَ بِنْتَ هَارَانَ الَّذِي أَنْجَبَ مِلْكَةَ وَيِسْكَةَ.</span>
<span className="v" id="v30" data-v="30" data-xref="التكوين 29:31; التكوين 21:1–2; الإنجيل كما دونه لوقا 1:7"> وَكَانَتْ سَارَايُ عَاقِراً لَيْسَ لَهَا وَلَدٌ.</span>
<span className="v" id="v31" data-v="31" data-xref="نحميا 9:7; أعمال الرسل 7:2–4; التكوين 24:10"> وَأَخَذَ تَارَحُ ابْنَهُ أَبْرَامَ وَحَفِيدَهُ لُوطاً بْنَ هَارَانَ، وَسَارَايَ كَنَّتَهُ زَوْجَةَ ابْنِهِ أَبْرَامَ، وَارْتَحَلَ بِهِمْ مِنْ أُورِ الْكَلْدَانِيِّينَ لِيَذْهَبُوا إِلَى أَرْضِ كَنْعَانَ.</span>
<span className="v" id="v32" data-v="32"> لَكِنَّهُمْ وَصَلُوا إِلَى حَارَانَ وَاسْتَقَرُّوا فِيهَا. وَهُنَاكَ مَاتَ تَارَحُ وَلَهُ مِنَ الْعُمْرِ مِئَتَانِ وَخَمْسُ سِنِينَ.</span>
