## مَثل العشر العذارى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه لوقا 12:35–36; الرسالة إلى مؤمني فيلبي 2:15–16; الإنجيل كما دونه متى 24:42–51"> حِينَئِذٍ يُشَبَّهُ مَلَكُوتُ السَّمَاوَاتِ بِعَشْرِ عَذَارَى أَخَذْنَ مَصَابِيحَهُنَّ وَانْطَلَقْنَ لِمُلاقَاةِ الْعَرِيسِ.</span>
<span className="v" id="v2" data-v="2" data-xref="إرميا 24:2; الرسالة الأولى إلى مؤمني كورنثوس 10:1–5; الإنجيل كما دونه متى 13:19–23"> وَكَانَتْ خَمْسٌ مِنْهُنَّ حَكِيمَاتٍ، وَخَمْسٌ جَاهِلاتٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="رؤيا يوحنا 3:15–16; الرسالة الثانية إلى تيموثاوس 3:5; إشعياء 48:1–2"> فَأَخَذَتِ الْجَاهِلاتُ مَصَابِيحَهُنَّ دُونَ زَيْتٍ.</span>
<span className="v" id="v4" data-v="4" data-xref="الرسالة إلى مؤمني روما 8:9; رسالة يوحنا الأولى 2:27; رسالة يوحنا الأولى 2:20"> وَأَمَّا الْحَكِيمَاتُ، فَأَخَذْنَ مَعَ مَصَابِيحِهِنَّ زَيْتاً وَضَعْنَهُ فِي أَوْعِيَةٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="حبقوق 2:3; الرسالة الأولى إلى مؤمني تسالونيكي 5:6–8; رسالة بطرس الثانية 3:4–9"> وَإِذْ أَبْطَأَ الْعَرِيسُ، نَعَسْنَ جَمِيعاً وَنِمْنَ.</span>
<span className="v" id="v6" data-v="6" data-xref="رسالة بطرس الثانية 3:10; الإنجيل كما دونه متى 24:31; الإنجيل كما دونه متى 24:44"> وَفِي مُنْتَصَفِ اللَّيْلِ، دَوَّى الْهُتَافُ: هَا هُوَ الْعَرِيسُ آتٍ؛ فَانْطَلِقْنَ لِمُلاقَاتِهِ!</span>
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه لوقا 12:35; رؤيا يوحنا 3:19–20; رسالة بطرس الثانية 3:14"> فَنَهَضَتِ الْعَذَارَى جَمِيعاً وَجَهَّزْنَ مَصَابِيحَهُنَّ.</span>
<span className="v" id="v8" data-v="8" data-xref="الأمثال 13:9; الإنجيل كما دونه لوقا 12:35; الرسالة إلى العبرانيين 4:1"> وَقَالَتِ الْجَاهِلاتُ لِلْحَكِيمَاتِ: أَعْطِينَنَا بَعْضَ الزَّيْتِ مِنْ عِنْدِكُنَّ، فَإِنَّ مَصَابِيحَنَا تَنْطَفِئُ!</span>
<span className="v" id="v9" data-v="9" data-xref="المزامير 49:7–9; رؤيا يوحنا 3:17–18; إشعياء 55:6–7"> فَأَجَابَتِ الْحَكِيمَاتُ: رُبَّمَا لَا يَكْفِي لَنَا وَلَكُنَّ. فَاذْهَبْنَ بِالأَحْرَى إِلَى بَائِعِي الزَّيْتِ وَاشْتَرِينَ لَكُنَّ!</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه لوقا 13:25; الإنجيل كما دونه متى 24:44; الإنجيل كما دونه لوقا 12:36–37"> وَبَيْنَمَا الْعَذَارَى الْجَاهِلاتُ ذَاهِبَاتٌ لِلشِّرَاءِ، وَصَلَ الْعَرِيسُ، فَدَخَلَتِ الْعَذَارَى الْمُسْتَعِدَّاتُ مَعَهُ إِلَى قَاعَةِ الْعُرْسِ، وَأُغْلِقَ الْبَابُ.</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه متى 7:21–23; الرسالة إلى العبرانيين 12:16–17"> وَبَعْدَ حِينٍ، رَجَعَتِ الْعَذَارَى الأُخْرَيَاتُ، وَقُلْنَ: يَا سَيِّدُ، يَا سَيِّدُ، افْتَحْ لَنَا!</span>
<span className="v" id="v12" data-v="12" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 8:3; الرسالة الثانية إلى تيموثاوس 2:19; الإنجيل كما دونه يوحنا 10:27"> فَأَجَابَ الْعَرِيسُ: الْحَقَّ أَقُولُ لَكُنَّ: إِنِّي لَا أَعْرِفُكُنَّ!</span>
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه متى 24:42–44; رؤيا يوحنا 16:15; الرسالة الأولى إلى مؤمني تسالونيكي 5:6"> فَاسْهَرُوا إِذَنْ، لأَنَّكُمْ لَا تَعْرِفُونَ الْيَوْمَ وَلا السَّاعَةَ!</span>
</div>

## مَثل الوزنات

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه مرقس 13:34; الإنجيل كما دونه لوقا 19:12–27; رسالة بطرس الأولى 4:9–11"> فَذَلِكَ أَشْبَهُ بِإِنْسَانٍ مُسَافِرٍ، اسْتَدْعَى عَبِيدَهُ وَسَلَّمَهُمْ أَمْوَالَهُ،</span>
<span className="v" id="v15" data-v="15" data-xref="الإنجيل كما دونه متى 18:24; الإنجيل كما دونه لوقا 19:13–14; الإنجيل كما دونه لوقا 12:48"> فَأَعْطَى وَاحِداً مِنْهُمْ خَمْسَ وَزْنَاتٍ (مِنَ الْفِضَّةِ)، وَأَعْطَى آخَرَ وَزْنَتَيْنِ، وَأَعْطَى الثَّالِثَ وَزْنَةً وَاحِدَةً، كُلَّ وَاحِدٍ عَلَى قَدْرِ طَاقَتِهِ، ثُمَّ سَافَرَ.</span>
<span className="v" id="v16" data-v="16" data-xref="رسالة يوحنا الثالثة 1:5–8; الرسالة الأولى إلى تيموثاوس 6:17–18; الرسالة الثانية إلى تيموثاوس 2:6"> وَفِي الْحَالِ مَضَى الَّذِي أَخَذَ الْوَزْنَاتِ الْخَمْسَ وَتَاجَرَ بِها، فَرَبِحَ خَمْسَ وَزْنَاتٍ أُخْرَى.</span>
<span className="v" id="v17" data-v="17" data-xref="الأمثال 3:9–10; الرسالة إلى مؤمني أفسس 5:16; رسالة بطرس الأولى 4:10"> وَعَمِلَ مِثْلَهُ الَّذِي أَخَذَ الوَزْنَتَيْنِ، فَرَبِحَ وَزْنَتَيْنِ أُخْرَيَيْنِ.</span>
<span className="v" id="v18" data-v="18" data-xref="رسالة بطرس الثانية 1:8; حجي 1:2–4; ملاخي 1:10"> وَلَكِنَّ الَّذِي أَخَذَ الْوَزْنَةَ الْوَاحِدَةَ، مَضَى وَحَفَرَ حُفْرَةً فِي الأَرْضِ وَدَفَنَ مَالَ سَيِّدِهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 3:12–15; الرسالة الثانية إلى مؤمني كورنثوس 5:10; الرسالة إلى مؤمني روما 14:7–12"> وَبَعْدَ مُدَّةٍ طَوِيلَةٍ، رَجَعَ سَيِّدُ أُولئِكَ الْعَبِيدِ وَاسْتَدْعَاهُمْ لِيُحَاسِبَهُمْ.</span>
<span className="v" id="v20" data-v="20" data-xref="أعمال الرسل 20:24; الإنجيل كما دونه لوقا 19:16–17; الرسالة إلى مؤمني كولوسي 1:29"> فَجَاءَهُ الَّذِي أَخَذَ الْوَزْنَاتِ الْخَمْسَ، وَقَدَّمَ الْوَزْنَاتِ الْخَمْسَ الأُخْرَى، وَقَالَ: يَا سَيِّدُ، أَنْتَ سَلَّمْتَنِي خَمْسَ وَزْنَاتٍ، فَهَذِهِ خَمْسُ وَزْنَاتٍ غَيْرُهَا رَبِحْتُهَا!</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه متى 25:23; الإنجيل كما دونه لوقا 16:10; الإنجيل كما دونه يوحنا 12:26"> فَقَالَ لَهُ سَيِّدُهُ: حَسَناً فَعَلْتَ أَيُّهَا الْعَبْدُ الصَّالِحُ وَالأَمِينُ! كُنْتَ أَمِيناً عَلَى الْقَلِيلِ، فَسَأُقِيمُكَ عَلَى الْكَثِيرِ. اُدْخُلْ إِلَى فَرَحِ سَيِّدِكَ!</span>
<span className="v" id="v22" data-v="22" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 8:12; الرسالة الثانية إلى مؤمني كورنثوس 8:7–8; الرسالة الثانية إلى مؤمني كورنثوس 8:1–3"> ثُمَّ جَاءَهُ أَيْضاً الَّذِي أَخَذَ الْوَزْنَتَيْنِ وَقَالَ: يَا سَيِّدُ أَنْتَ سَلَّمْتَنِي وَزْنَتَيْنِ، فَهَاتَانِ وَزْنَتَانِ غَيْرُهُمَا رَبِحْتُهُمَا!</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه متى 25:21; الإنجيل كما دونه مرقس 14:8–9; الإنجيل كما دونه مرقس 12:41–44"> فَقَالَ لَهُ سَيِّدُهُ: حَسَناً فَعَلْتَ أَيُّهَا الْعَبْدُ الصَّالِحُ وَالأَمِينُ! كُنْتَ أَمِيناً عَلَى الْقَلِيلِ، فَسَأُقِيمُكَ عَلَى الْكَثِيرِ. اُدْخُلْ إِلَى فَرَحِ سَيِّدِكَ!</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه لوقا 19:20–22; الإنجيل كما دونه لوقا 15:29; ملاخي 3:14–15"> ثُمَّ جَاءَهُ أَيْضاً الَّذِي أَخَذَ الْوَزْنَةَ الْوَاحِدَةَ، وَقَالَ: يَا سَيِّدُ، عَرَفْتُكَ رَجُلاً قَاسِياً، تَحْصُدُ مِنْ حَيْثُ لَمْ تَزْرَعْ، وَتَجْمَعُ مِنْ حَيْثُ لَمْ تَبْذُرْ،</span>
<span className="v" id="v25" data-v="25" data-xref="الرسالة الثانية إلى تيموثاوس 1:6–7; الرسالة إلى مؤمني روما 8:15; رؤيا يوحنا 21:8"> فَخِفْتُ، فَذَهَبْتُ وَدَفَنْتَ وَزْنَتَكَ فِي الأَرْضِ. فَهَذَا هُوَ مَالُكَ!</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه متى 18:32; أيوب 15:5–6"> فَأَجَابَهُ سَيِّدُهُ: أَيُّهَا الْعَبْدُ الشِّرِّيرُ الْكَسُولُ! عَرَفْتَ أَنِّي أَحْصُدُ مِنْ حَيْثُ لَمْ أَزْرَعْ، وَأَجْمَعُ مِنْ حَيْثُ لَمْ أَبْذُرْ،</span>
<span className="v" id="v27" data-v="27" data-xref="التثنية 23:19–20; الرسالة إلى مؤمني روما 3:19; رسالة يهوذا 1:15"> فَكَانَ يَحْسُنُ بِكَ أَنْ تُوْدِعَ مَالِي عِنْدَ الصَّيَارِفَةِ لِكَيْ أَسْتَرِدَّهُ لَدَى عَوْدَتِي مَعَ فَائِدَتِهِ!</span>
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه لوقا 19:24; الإنجيل كما دونه لوقا 10:42"> ثُمَّ قَالَ لِعَبِيدِهِ: خُذُوا مِنْهُ الْوَزْنَةَ، وَأَعْطُوهَا لِصَاحِبِ الْوَزْنَاتِ الْعَشْرِ:</span>
<span className="v" id="v29" data-v="29" data-xref="الإنجيل كما دونه لوقا 8:18; الإنجيل كما دونه مرقس 4:25; الإنجيل كما دونه متى 13:12"> فَإِنَّ كُلَّ مَنْ عِنْدَهُ، يُعْطَى الْمَزِيدَ فَيَفِيضُ؛ وَمَنْ لَيْسَ عِنْدَهُ، فَحَتَّى الَّذِي عِنْدَهُ يُنْتَزَعُ مِنْهُ.</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه متى 8:12; رؤيا يوحنا 3:15–16; الإنجيل كما دونه متى 13:50"> أَمَّا هَذَا الْعَبْدُ الَّذِي لَا نَفْعَ مِنْهُ، فَاطْرَحُوهُ فِي الظُّلْمَةِ الْخَارِجِيَّةِ، هُنَاكَ يَكُونُ البُكَاءُ وَصَرِيرُ الأَسْنَانِ!</span>

## الخراف والجداء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="رؤيا يوحنا 1:7; الرسالة الثانية إلى مؤمني تسالونيكي 1:7–8; الإنجيل كما دونه متى 19:28"> وَعِنْدَمَا يَعُودُ ابْنُ الإِنْسَانِ فِي مَجْدِهِ وَمَعَهُ جَمِيعُ مَلائِكَتِهِ، فَإِنَّهُ يَجْلِسُ عَلَى عَرْشِ مَجْدِهِ،</span>
<span className="v" id="v32" data-v="32" data-xref="ملاخي 3:18; الإنجيل كما دونه متى 13:49; الرسالة الثانية إلى مؤمني كورنثوس 5:10"> وَتَجْتَمِعُ أَمَامَهُ الشُّعُوبُ كُلُّهَا، فَيَفْصِلُ بَعْضَهُمْ عَنْ بَعْضٍ كَمَا يَفْصِلُ الرَّاعِي الْغَنَمَ عَنِ الْمِعَازِ،</span>
<span className="v" id="v33" data-v="33" data-xref="المزامير 100:3; الإنجيل كما دونه يوحنا 10:26–28; الإنجيل كما دونه يوحنا 21:15–17"> فَيُوقِفُ الْغَنَمَ عَنْ يَمِينِهِ، وَالْمِعَازِ عَنْ يَسَارِهِ؛</span>
<span className="v" id="v34" data-v="34" data-xref="رسالة بطرس الأولى 1:19–20; رسالة يعقوب 2:5; رؤيا يوحنا 21:7"> ثُمَّ يَقُولُ الْمَلِكُ لِلَّذِينَ عَنْ يَمِينِهِ: تَعَالَوْا يَا مَنْ بَارَكَهُمْ أَبِي، رِثُوا الْمَلَكُوتَ الَّذِي أُعِدَّ لَكُمْ مُنْذُ إِنْشَاءِ الْعَالَمِ:</span>
<span className="v" id="v35" data-v="35" data-xref="رسالة يوحنا الأولى 3:16–19; رسالة يعقوب 2:15–16; التثنية 15:7–11"> لأَنِّي جُعْتُ فَأَطْعَمْتُمُونِي، عَطِشْتُ فَسَقَيْتُمُونِي، كُنْتُ غَرِيباً فَآوَيْتُمُونِي،</span>
<span className="v" id="v36" data-v="36" data-xref="رسالة يعقوب 1:27; الرسالة الثانية إلى تيموثاوس 1:16–18; رسالة يعقوب 5:14–15"> عُرْيَاناً فَكَسَوْتُمُونِي، مَرِيضاً فَزُرْتُمُونِي، سَجِيناً فَأَتَيْتُمْ إِلَيَّ!</span>
<span className="v" id="v37" data-v="37" data-xref="الإنجيل كما دونه متى 6:3; أخبار الأيام الأول 29:14; رسالة بطرس الأولى 5:5–6"> فَيَرُدُّ الصَّالِحونَ قَائِلِينَ: يَا رَبُّ، مَتَى رَأَيْنَاكَ جَائِعاً فَأَطْعَمْنَاكَ، أَوْ عَطْشَاناً فَسَقَيْنَاكَ؟</span>
