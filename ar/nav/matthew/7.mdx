## الحكم على الآخرين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه لوقا 6:37; الرسالة إلى مؤمني روما 2:1–2; رسالة يعقوب 4:11–12"> لَا تَدِينُوا لِئَلّا تُدَانُوا.</span>
<span className="v" id="v2" data-v="2" data-xref="رسالة يعقوب 2:13; الإنجيل كما دونه مرقس 4:24; الإنجيل كما دونه لوقا 6:38"> فَإِنَّكُمْ بِالدَّيْنُونَةِ الَّتِي بِها تَدِينُونَ تُدَانُونَ؛ وَبِالْكَيْلِ الَّذِي بِهِ تَكِيلُونَ يُكَالُ لَكُمْ.</span>
<span className="v" id="v3" data-v="3" data-xref="الإنجيل كما دونه يوحنا 8:7–9; الرسالة إلى مؤمني غلاطية 6:1; الإنجيل كما دونه لوقا 6:41–42"> لِمَاذَا تُلاحِظُ الْقَشَّةَ فِي عَيْنِ أَخِيكَ، وَلكِنَّكَ لَا تَتَنَبَّهُ إِلَى الْخَشَبَةِ الْكَبِيرَةِ الَّتِي فِي عَيْنِكَ؟</span>
<span className="v" id="v4" data-v="4"> أَوْ كَيْفَ تَقُولُ لأَخِيكَ: دَعْنِي أُخْرِجُ الْقَشَّةَ مِنْ عَيْنِكَ، وَهَا هِيَ الْخَشَبَةُ فِي عَيْنِكَ أَنْتَ!</span>
<span className="v" id="v5" data-v="5" data-xref="الإنجيل كما دونه لوقا 6:42; المزامير 51:9–13; الإنجيل كما دونه متى 23:13–28"> يَا مُنَافِقَ! أَخْرِجْ أَوَّلاً الْخَشَبَةَ مِنْ عَيْنِكَ، وَعِنْدَئِذٍ تُبْصِرُ جَيِّداً لِتُخْرِجَ الْقَشَّةَ مِنْ عَيْنِ أَخِيكَ.</span>
<span className="v" id="v6" data-v="6" data-xref="الأمثال 9:7–8; الأمثال 23:9; الإنجيل كما دونه متى 10:14–15"> لَا تُعْطُوا الْمُقَدَّسَاتِ لِلْكِلابِ، وَلا تَطْرَحُوا جَوَاهِرَكُمْ أَمَامَ الْخَنَازِيرِ، لِكَيْ لَا تَدُوسَهَا بِأَرْجُلِهَا وَتَنْقَلِبَ عَلَيْكُمْ فَتُمَزِّقَكُمْ.</span>
</div>
</div>

## اسألوا، اطلبوا، اقرعوا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه متى 21:22; رسالة يوحنا الأولى 3:22; الإنجيل كما دونه يوحنا 15:7"> اِسْأَلُوا، تُعْطَوْا. اُطْلُبُوا، تَجِدُوا. اِقْرَعُوا، يُفْتَحْ لَكُمْ.</span>
<span className="v" id="v8" data-v="8" data-xref="المزامير 81:10; المزامير 81:16; الإنجيل كما دونه لوقا 23:42–43"> فَكُلُّ مَنْ يَسْأَلْ، يَنَلْ؛ وَمَنْ يَطْلُبْ، يَجِدْ؛ وَمَنْ يَقْرَعْ، يُفْتَحْ لَهُ.</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه لوقا 11:11–13"> وَإلَّا، فَأَيُّ إِنْسَانٍ مِنْكُمْ يَطْلُبُ مِنْهُ ابْنُهُ خُبْزاً، فَيُعْطِيهِ حَجَراً،</span>
<span className="v" id="v10" data-v="10"> أَوْ سَمَكَةً، فَيُعْطِيهِ حَيَّةً؟</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه لوقا 11:11–13; رسالة يعقوب 1:17; الرسالة إلى مؤمني روما 8:32"> فَإِنْ كُنْتُمْ وَأَنْتُمْ أَشْرَارٌ، تَعْرِفُونَ أَنْ تُعْطُوا أَوْلادَكُمْ عَطَايَا جَيِّدَةً، فَكَمْ بِالأَوْلَى جِدّاً يُعْطِي أَبُوكُمُ السَّمَاوِيُّ عَطَايَا جَيِّدَةً لِلَّذِينَ يَطْلُبُونَ مِنْهُ؟</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه لوقا 6:31; الرسالة إلى مؤمني روما 13:8–10; الرسالة إلى مؤمني غلاطية 5:13–14"> إِذَنْ، كُلُّ مَا تُرِيدُونَ أَنْ يُعَامِلَكُمُ النَّاسُ بِهِ، فَعَامِلُوهُمْ أَنْتُمْ بِهِ أَيْضاً: هَذِهِ خُلاصَةُ تَعْلِيمِ الشَّرِيعَةِ وَالأَنْبِيَاءِ.</span>
</div>
</div>

## الباب الضيق والباب الواسع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه لوقا 13:24–25; الأمثال 16:25; الإنجيل كما دونه يوحنا 14:6"> اُدْخُلُوا مِنَ الْبَابِ الضَّيِّقِ! فَإِنَّ الْبَابَ الْمُؤَدِّيَ إِلَى الْهَلاكِ وَاسِعٌ وَطَرِيقَهُ رَحْبٌ؛ وَكَثِيرُونَ هُمُ الَّذِينَ يَدْخُلُونَ مِنْهُ.</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه لوقا 13:23–30; الإنجيل كما دونه متى 22:14; الإنجيل كما دونه متى 16:24–25"> مَا أَضْيَقَ الْبَابَ وَأَصْعَبَ الطَّرِيقَ الْمُؤَدِّيَ إِلَى الْحَيَاةِ! وَقَلِيلُونَ هُمُ الَّذِينَ يَهْتَدُونَ إِلَيْهِ.</span>
</div>
</div>

## الشجرة وثمرها

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="رسالة يوحنا الأولى 4:1; رسالة بطرس الثانية 2:1–3; الإنجيل كما دونه متى 24:11"> اِحْذَرُوا الأَنْبِيَاءَ الدَّجَّالِينَ الَّذِينَ يَأْتُونَ إِلَيْكُمْ لابِسِينَ ثِيَابَ الْحُمْلانِ، وَلَكِنَّهُمْ مِنَ الدَّاخِلِ ذِئَابٌ خَاطِفَةٌ!</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه لوقا 6:43–45; الإنجيل كما دونه متى 12:33; رسالة يعقوب 3:12"> مِنْ ثِمَارِهِمْ تَعْرِفُونَهُمْ. هَلْ يُجْنَى مِنَ الشَّوْكِ عِنَبٌ، أَوْ مِنَ الْعُلَّيْقِ تِينٌ؟</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه متى 12:33–35; الرسالة إلى مؤمني غلاطية 5:22–24; الرسالة إلى مؤمني أفسس 5:9"> هَكَذَا، كُلُّ شَجَرَةٍ جَيِّدَةٍ تُثْمِرُ ثَمَراً جَيِّداً. أَمَّا الشَّجَرَةُ الرَّدِيئَةُ، فَإِنَّهَا تُثْمِرُ ثَمَراً رَدِيئاً.</span>
<span className="v" id="v18" data-v="18" data-xref="الرسالة إلى مؤمني غلاطية 5:17; رسالة يوحنا الأولى 3:9–10"> لَا يُمْكِنُ أَنْ تُثْمِرَ الشَّجَرَةُ الْجَيِّدَةُ ثَمَراً رَدِيئاً، وَلا الشَّجَرَةُ الرَّدِيئَةُ ثَمَراً جَيِّداً.</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه متى 3:10; الرسالة إلى العبرانيين 6:8; الإنجيل كما دونه يوحنا 15:2–6"> وَكُلُّ شَجَرَةٍ لَا تُثْمِرُ ثَمَراً جَيِّداً، تُقْطَعُ وَتُطْرَحُ فِي النَّارِ.</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه متى 7:16; أعمال الرسل 5:38"> إِذَنْ مِنْ ثِمَارِهِمْ تَعْرِفُونَهُمْ.</span>
</div>
</div>

## التلاميذ الحقيقيون والتلاميذ الزائفون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="الرسالة إلى مؤمني روما 2:13; الإنجيل كما دونه لوقا 6:46; رسالة يعقوب 1:22"> لَيْسَ كُلُّ مَنْ يَقُولُ لِي: يَا رَبُّ، يَا رَبُّ! يَدْخُلُ مَلَكُوتَ السَّمَاوَاتِ، بَلْ مَنْ يَعْمَلُ بِإِرَادَةِ أَبِي الَّذِي فِي السَّمَاوَاتِ.</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه لوقا 13:25–27; الرسالة الأولى إلى مؤمني كورنثوس 13:1–2; ملاخي 3:17–18"> فِي ذَلِكَ الْيَوْمِ سَيَقُولُ لِي كَثِيرُونَ: يَا رَبُّ، يَا رَبُّ، أَلَيْسَ بِاسْمِكَ تَنَبَّأْنَا، وَبِاسْمِكَ طَرَدْنَا الشَّيَاطِينَ، وَبِاسْمِكَ عَمِلْنَا مُعْجِزَاتٍ كَثِيرَةً؟</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه متى 25:41; المزامير 6:8; الإنجيل كما دونه لوقا 13:25"> وَلَكِنِّي عِنْدَئِذٍ أُصَرِّحُ لَهُمْ: إِنِّي لَمْ أَعْرِفْكُمْ قَطُّ! ابْتَعِدُوا عَنِّي يَا فَاعِلِي الإِثْمِ!</span>
</div>
</div>

## البناء الحكيم والبناء الجاهل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه لوقا 6:47–49; الإنجيل كما دونه يوحنا 13:17; الرسالة الأولى إلى مؤمني كورنثوس 3:10–11"> فَأَيُّ مَنْ يَسْمَعُ أَقْوَالِي هَذِهِ وَيَعْمَلُ بِها، أُشَبِّهُهُ بِرَجُلٍ حَكِيمٍ بَنَى بَيْتَهُ عَلَى الصَّخْرِ،</span>
<span className="v" id="v25" data-v="25" data-xref="رسالة يعقوب 1:12; المزامير 125:1–2; رسالة بطرس الأولى 1:7"> فَنَزَلَتِ الأَمْطَارُ، وَجَرَتِ السُّيُولُ، وَهَبَّتِ الْعَوَاصِفُ، فَضَرَبَتْ ذَلِكَ الْبَيْتَ، فَلَمْ يَسْقُطْ لأَنَّهُ مُؤَسَّسٌ عَلَى الصَّخْرِ.</span>
<span className="v" id="v26" data-v="26" data-xref="إرميا 8:9; الإنجيل كما دونه لوقا 6:49; رسالة يعقوب 2:20"> وَأَيُّ مَنْ يَسْمَعُ أَقْوَالِي هَذِهِ وَلا يَعْمَلُ بِها، يُشَبَّهُ بِرَجُلٍ غَبِيٍّ بَنَى بَيْتَهُ عَلَى الرَّمْلِ،</span>
<span className="v" id="v27" data-v="27" data-xref="الإنجيل كما دونه متى 13:19–22; رسالة بطرس الثانية 2:20–22; الرسالة الأولى إلى مؤمني كورنثوس 3:13"> فَنَزَلَتْ الأَمْطَارُ، وَجَرَتْ السُّيُولُ، وَهَبَّتِ الْعَوَاصِفُ، فَضَرَبَتْ ذَلِكَ الْبَيْتَ، فَسَقَطَ، وَكَانَ سُقُوطُهُ عَظِيماً».</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه لوقا 4:32; الإنجيل كما دونه مرقس 1:22; الإنجيل كما دونه يوحنا 7:46"> وَلَمَّا أَنْهَى يَسُوعُ هَذَا الْكَلامَ، ذُهِلَتِ الْجُمُوعُ مِنْ تَعْلِيمِهِ،</span>
