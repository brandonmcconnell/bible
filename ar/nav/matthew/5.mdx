## مقدمة للموعظة على الجبل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه مرقس 3:13; الإنجيل كما دونه متى 15:29; الإنجيل كما دونه مرقس 4:1"> وَإِذْ رَأَى جُمُوعَ النَّاسِ، صَعِدَ إِلَى الْجَبَلِ. وَمَا إِنْ جَلَسَ، حَتَّى اقْتَرَبَ إِلَيْهِ تَلامِيذُهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="أعمال الرسل 8:35; الإنجيل كما دونه لوقا 6:20–26; أعمال الرسل 10:34"> فَتَكَلَّمَ وَأَخَذَ يُعَلِّمُهُمْ. التطويبات<br />فَقَالَ:</span>
<span className="v" id="v3" data-v="3" data-xref="إشعياء 66:2; الأمثال 29:23; المزامير 34:18"> «طُوبَى لِلْمَسَاكِينِ بِالرُّوحِ، فَإِنَّ لَهُمْ مَلَكُوتَ السَّمَاوَاتِ.</span>
<span className="v" id="v4" data-v="4" data-xref="رؤيا يوحنا 21:4; الرسالة الثانية إلى مؤمني كورنثوس 7:9–10; الإنجيل كما دونه يوحنا 16:20–22"> طُوبَى لِلْحَزَانَى، فَإِنَّهُمْ يَتَعَزَّوْنَ.</span>
<span className="v" id="v5" data-v="5" data-xref="المزامير 37:11; إشعياء 29:19; رسالة يعقوب 3:13"> طُوبَى لِلْوُدَعَاءِ، فَإِنَّهُمْ سَيَرِثُونَ الأَرْضَ.</span>
<span className="v" id="v6" data-v="6" data-xref="الإنجيل كما دونه يوحنا 4:14; المزامير 107:9; الإنجيل كما دونه يوحنا 6:27"> طُوبَى لِلْجِيَاعِ وَالْعِطَاشِ إِلَى الْبِرِّ، فَإِنَّهُمْ سَيُشْبَعُونَ.</span>
<span className="v" id="v7" data-v="7" data-xref="الرسالة إلى مؤمني أفسس 4:32–1; الإنجيل كما دونه متى 6:14–15; صموئيل الثاني 22:26"> طُوبَى لِلرُّحَمَاءِ، فَإِنَّهُمْ سَيُرْحَمُونَ.</span>
<span className="v" id="v8" data-v="8" data-xref="المزامير 24:4; المزامير 51:10; الرسالة إلى العبرانيين 10:22"> طُوبَى للأَنْقِيَاءِ الْقَلْبِ، فَإِنَّهُمْ سَيَرَوْنَ اللهَ.</span>
<span className="v" id="v9" data-v="9" data-xref="الرسالة إلى العبرانيين 12:14; رسالة يعقوب 3:16–18; الرسالة إلى مؤمني روما 12:18"> طُوبَى لِصَانِعِي السَّلامِ، فَإِنَّهُمْ سَيُدْعَوْنَ أَبْنَاءَ اللهِ.</span>
<span className="v" id="v10" data-v="10" data-xref="رسالة يعقوب 1:12; رسالة بطرس الأولى 3:13–14; الرسالة الثانية إلى مؤمني كورنثوس 4:17"> طُوبَى لِلْمُضْطَهَدِينَ مِنْ أَجْلِ الْحَقِّ، فَإِنَّ لَهُمْ مَلَكُوتَ السَّمَاوَاتِ.</span>
<span className="v" id="v11" data-v="11" data-xref="رسالة بطرس الأولى 4:14; الإنجيل كما دونه لوقا 21:17; الإنجيل كما دونه مرقس 13:13"> طُوبَى لَكُمْ مَتَى أَهَانَكُمُ النَّاسُ وَاضْطَهَدُوكُمْ، وَقَالُوا عَلَيْكُمْ مِنْ أَجْلِي كُلَّ سُوءٍ كَاذِبِينَ.</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه لوقا 6:23; رسالة يعقوب 1:2; الرسالة إلى مؤمني كولوسي 3:24"> اِفْرَحُوا وَتَهَلَّلُوا، فَإِنَّ مُكَافَأَتَكُمْ فِي السَّمَاوَاتِ عَظِيمَةٌ. فَإِنَّهُمْ هكَذَا اضْطَهَدُوا الأَنْبِيَاءَ مِنْ قَبْلِكُمْ!</span>
</div>
</div>

## ملح الأرض ونور العالم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الرسالة إلى مؤمني كولوسي 4:6; الإنجيل كما دونه لوقا 14:34–35; الإنجيل كما دونه مرقس 9:49–50"> أَنْتُمْ مِلْحُ الأَرْضِ. فَإِذَا فَسَدَ الْمِلْحُ، فَمَاذَا يُعِيدُ إِلَيْهِ مُلُوحَتَهُ؟ إِنَّهُ لَا يَعُودُ يَصْلُحُ لِشَيْءٍ إِلّا لأَنْ يُطْرَحَ خَارِجاً لِتَدُوسَهُ النَّاسُ!</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الرسالة إلى مؤمني فيلبي 2:15; الإنجيل كما دونه يوحنا 8:12; الرسالة إلى مؤمني أفسس 5:8–14"> أَنْتُمْ نُورُ الْعَالَمِ. لَا يُمْكِنُ أَنْ تُخْفَى مَدِينَةٌ مَبْنِيَّةٌ عَلَى جَبَلٍ؛</span>
<span className="v" id="v15" data-v="15" data-xref="الإنجيل كما دونه لوقا 11:33; الإنجيل كما دونه مرقس 4:21; الإنجيل كما دونه لوقا 8:16"> وَلا يُضِيءُ النَّاسُ مِصْبَاحاً ثُمَّ يَضَعُونَهُ تَحْتَ مِكْيَالٍ، بَلْ يَضَعُونَهُ فِي مَكَانٍ مُرْتَفِعٍ لِيُضِيءَ لِجَمِيعِ مَنْ فِي الْبَيْتِ.</span>
<span className="v" id="v16" data-v="16" data-xref="رسالة بطرس الأولى 2:12; الرسالة إلى تيطس 2:7; الإنجيل كما دونه يوحنا 15:8"> هكَذَا، فَلْيُضِئْ نُورُكُمْ أَمَامَ النَّاسِ، لِيَرَوْا أَعْمَالَكُمُ الْحَسَنَةَ وَيُمَجِّدُوا أَبَاكُمُ الَّذِي فِي السَّمَاوَاتِ.</span>
</div>
</div>

## تحقيق وإكمال الشريعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="الرسالة إلى مؤمني روما 10:4; الإنجيل كما دونه لوقا 16:17; الرسالة إلى مؤمني غلاطية 4:4–5"> لَا تَظُنُّوا أَنِّي جِئْتُ لأُلْغِيَ الشَّرِيعَةَ أَوِ الأَنْبِيَاءَ. مَا جِئْتُ لأُلْغِيَ، بَلْ لأُكَمِّلَ.</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه لوقا 16:17; إشعياء 40:8; رسالة بطرس الأولى 1:25"> فَالْحَقَّ أَقُولُ لَكُمْ: إِلَى أَنْ تَزُولَ الأَرْضُ وَالسَّمَاءُ، لَنْ يَزُولَ حَرْفٌ وَاحِدٌ أَوْ نُقْطَةٌ وَاحِدَةٌ مِنَ الشَّرِيعَةِ، حَتَّى يَتِمَّ كُلُّ شَيْءٍ.</span>
<span className="v" id="v19" data-v="19" data-xref="التثنية 12:32; رسالة يعقوب 2:10–11; الإنجيل كما دونه متى 28:20"> فَأَيُّ مَنْ خَالَفَ وَاحِدَةً مِنْ هَذِهِ الْوَصَايَا الصُّغْرَى، وَعَلَّمَ النَّاسَ أَنْ يَفْعَلُوا فِعْلَهُ، يُدْعَى الأَصْغَرَ فِي مَلَكُوتِ السَّمَاوَاتِ. وَأَمَّا مَنْ عَمِلَ بِها وَعَلَّمَهَا، فَيُدْعَى عَظِيماً فِي مَلَكُوتِ السَّمَاوَاتِ.</span>
<span className="v" id="v20" data-v="20" data-xref="الرسالة إلى مؤمني روما 10:2–3; الرسالة إلى مؤمني فيلبي 3:9; الرسالة إلى مؤمني روما 9:30–32"> فَإِنِّي أَقُولُ لَكُمْ: إِنْ لَمْ يَزِدْ صَلاحُكُمْ عَلَى صَلاحِ الْكَتَبَةِ والْفَرِّيسِيِّينَ، فَلَنْ تَدْخُلُوا مَلَكُوتَ السَّمَاوَاتِ أَبَداً.</span>
</div>
</div>

## القتل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التثنية 5:17; الخروج 20:13; العدد 35:30–34"> سَمِعْتُمْ أَنَّهُ قِيلَ لِلأَقْدَمِينَ: لَا تَقْتُلْ! وَمَنْ قَتَلَ يَسْتَحِقُّ الْمُحَاكَمَةَ.</span>
<span className="v" id="v22" data-v="22" data-xref="رسالة يوحنا الأولى 3:14–15; رسالة يوحنا الأولى 4:20–21; رسالة يوحنا الأولى 2:9"> أَمَّا أَنَا فَأَقُولُ لَكُمْ: كُلُّ مَنْ يَغْضَبُ عَلَى أَخِيهِ، يَسْتَحِقُّ الْمُحَاكَمَةَ؛ وَمَنْ يَقُولُ لأَخِيهِ: يَا تَافِهُ! يَسْتَحِقُّ الْمُثُولَ أَمَامَ الْمَجْلِسِ الأَعْلَى؛ وَمَنْ يَقُولُ: يَا أَحْمَقُ! يَسْتَحِقُّ نَارَ جَهَنَّمَ!</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه مرقس 11:25; اللاويين 6:2–6; الإنجيل كما دونه متى 5:24"> فَإِذَا جِئْتَ بِتَقْدِمَتِكَ إِلَى الْمَذْبَحِ، وَهُنَاكَ تَذَكَّرْتَ أَنَّ لأَخِيكَ شَيْئاً عَلَيْكَ،</span>
<span className="v" id="v24" data-v="24" data-xref="رسالة يعقوب 5:16; الرسالة إلى مؤمني روما 12:17–18; الإنجيل كما دونه متى 18:15–17"> فَاتْرُكْ تَقْدِمَتَكَ أَمَامَ الْمَذْبَحِ، وَاذْهَبْ أَوَّلاً وَصَالِحْ أَخَاكَ، ثُمَّ ارْجِعْ وَقَدِّمْ تَقْدِمَتَكَ.</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه لوقا 12:58–59; الأمثال 25:8; الرسالة إلى العبرانيين 3:13"> سَارِعْ إِلَى اسْتِرْضَاءِ خَصْمِكَ وَأَنْتَ مَعَهُ فِي الطَّرِيقِ إِلَى الْمَحْكَمَةِ، قَبْلَ أَنْ يُسَلِّمَكَ الْخَصْمُ إِلَى الْقَاضِي، فَيُسَلِّمَكَ الْقَاضِي إِلَى الشُّرَطِيِّ، فَيُلْقِيَكَ فِي السِّجْنِ.</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه متى 18:34; الإنجيل كما دونه لوقا 12:59; الرسالة الثانية إلى مؤمني تسالونيكي 1:9"> وَالْحَقَّ أَقُولُ لَكَ إِنَّكَ لَنْ تَخْرُجَ مِنَ السِّجْنِ حَتَّى تُوْفِيَ الْفَلْسَ الأَخِيرَ!</span>
</div>

## الحلف

<span className="v" id="v35" data-v="35" data-xref="المزامير 48:2; ملاخي 1:14; أخبار الأيام الثاني 6:6"> وَلا بِالأَرْضِ لأَنَّهَا مَوْطِئُ قَدَمَيْهِ، وَلا بِأُورُشَلِيمَ لأَنَّهَا مَدِينَةُ الْمَلِكِ الأَعْظَمِ.</span>
<span className="v" id="v36" data-v="36" data-xref="الإنجيل كما دونه لوقا 12:25; الإنجيل كما دونه متى 6:27; الإنجيل كما دونه متى 23:16–21"> وَلا تَحْلِفْ بِرَأْسِكَ لأَنَّكَ لَا تَقْدِرُ أَنْ تَجْعَلَ شَعْرَةً وَاحِدَةً فِيهَا بَيْضَاءَ أَوْ سَوْدَاءَ.</span>
