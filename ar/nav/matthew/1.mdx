## نسب يسوع المسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التكوين 22:18; الرسالة إلى مؤمني روما 1:3; الرسالة إلى مؤمني غلاطية 3:16"> هَذَا سِجِلُّ نَسَبِ يَسُوعَ الْمَسِيحِ ابْنِ دَاوُدَ ابْنِ إِبْرَاهِيمَ:</span>
<span className="v" id="v2" data-v="2" data-xref="أعمال الرسل 7:8; الرسالة إلى العبرانيين 11:17–18; التكوين 46:8–27"> إِبْرَاهِيمُ أَنْجَبَ إِسْحَاقَ. وَإِسْحاقُ أَنْجَبَ يَعْقُوبَ. وَيَعْقُوبُ أَنْجَبَ يَهُوذَا وَإِخْوَتَهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="التكوين 46:12; التكوين 38:29–30; أخبار الأيام الأول 4:1"> وَيَهُوذَا أَنْجَبَ فَارِصَ وَزَارَحَ مِنْ ثَامَارَ. وَفَارِصُ أَنْجَبَ حَصْرُونَ. وَحَصْرُونُ أَنْجَبَ أَرَامَ.</span>
<span className="v" id="v4" data-v="4" data-xref="أخبار الأيام الأول 2:10–12; العدد 7:12; العدد 2:3"> وَأَرَامُ أَنْجَبَ عَمِّينَادَابَ. وَعَمِّينَادَابُ أَنْجَبَ نَحْشُونَ. وَنَحْشُونُ أَنْجَبَ سَلْمُونَ.</span>
<span className="v" id="v5" data-v="5" data-xref="الرسالة إلى العبرانيين 11:31; راعوث 1:16–17; رسالة يعقوب 2:25"> وَسَلْمُونُ أَنْجَبَ بُوعَزَ مِنْ رَاحَابَ. وَبُوعَزُ أَنْجَبَ عُوبِيدَ مِنْ رَاعُوثَ. وَعُوبِيدُ أَنْجَبَ يَسَّى.</span>
<span className="v" id="v6" data-v="6" data-xref="ملوك الأول 15:5; صموئيل الأول 17:12; صموئيل الثاني 12:24–25"> وَيَسَّى أَنْجَبَ دَاوُدَ الْمَلِكَ. وَدَاوُدُ أَنْجَبَ سُلَيْمَانَ مِنَ الَّتِي كَانَتْ زَوْجَةً لأُورِيَّا.</span>
<span className="v" id="v7" data-v="7" data-xref="أخبار الأيام الثاني 14:1–15; ملوك الأول 14:31; ملوك الأول 11:43–24"> وَسُلَيْمَانُ أَنْجَبَ رَحَبْعَامَ. وَرَحَبْعَامُ أَنْجَبَ أَبِيَّا. وَأَبِيَّا أَنْجَبَ آسَا.</span>
<span className="v" id="v8" data-v="8" data-xref="أخبار الأيام الأول 3:11; أخبار الأيام الثاني 26:1–23; ملوك الثاني 8:16"> وَآسَا أَنْجَبَ يَهُوشَافَاطَ. وَيَهُوشَافَاطُ أَنْجَبَ يُورَامَ. وَيُورَامُ أَنْجَبَ عُزِّيَّا.</span>
<span className="v" id="v9" data-v="9" data-xref="إشعياء 7:1–13; أخبار الأيام الثاني 26:21; أخبار الأيام الثاني 27:1–32"> وَعُزِّيَّا أَنْجَبَ يُوثَامَ. وَيُوثَامُ أَنْجَبَ آحَازَ. وَآحَازُ أَنْجَبَ حِزْقِيَّا.</span>
<span className="v" id="v10" data-v="10" data-xref="أخبار الأيام الثاني 32:33–27; ملوك الأول 13:2; أخبار الأيام الأول 3:13–15"> وَحِزْقِيَّا أَنْجَبَ مَنَسَّى. وَمَنَسَّى أَنْجَبَ آمُونَ. وَآمُونُ أَنْجَبَ يُوشِيَّا.</span>
<span className="v" id="v11" data-v="11" data-xref="إرميا 27:20; إرميا 52:28–30; دانيال 1:2"> وَيُوشِيَّا أَنْجَبَ يَكُنْيَا وَإِخْوَتَهُ فِي أَثْنَاءِ السَّبْيِ إِلَى بَابِلَ.</span>
<span className="v" id="v12" data-v="12" data-xref="أخبار الأيام الأول 3:17; عزرا 3:2; الإنجيل كما دونه لوقا 3:27"> وَبَعْدَ السَّبْيِ إِلَى بَابِلَ، يَكُنْيَا أَنْجَبَ شَأَلْتِئِيلَ. وَشَأَلْتِئِيلُ أَنْجَبَ زَرُبَّابِلَ.</span>
<span className="v" id="v13" data-v="13"> وَزَرُبَّابِلُ أَنْجَبَ أَبِيهُودَ. وَأَبِيهُودُ أَنْجَبَ أَلِيَاقِيمَ. وَأَلِيَاقِيمُ أَنْجَبَ عَازُورَ.</span>
<span className="v" id="v14" data-v="14"> وَعَازُورُ أَنْجَبَ صَادُوقَ. وَصَادُوقُ أَنْجَبَ أَخِيمَ. وَأَخِيمُ أَنْجَبَ أَلِيُودَ.</span>
<span className="v" id="v15" data-v="15"> وَأَلِيُودُ أَنْجَبَ أَلِيعَازَرَ. وَأَلِيعَازَرُ أَنْجَبَ مَتَّانَ. وَمَتَّانُ أَنْجَبَ يَعْقُوبَ.</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه متى 27:17; الإنجيل كما دونه يوحنا 4:25; الإنجيل كما دونه لوقا 3:23"> وَيَعْقُوبُ أَنْجَبَ يُوسُفَ رَجُلَ مَرْيَمَ الَّتِي وُلِدَ مِنْهَا يَسُوعُ الَّذِي يُدْعَى الْمَسِيحَ.</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه متى 1:11–12; ملوك الثاني 24:14; الإنجيل كما دونه متى 11:2"> فَجُمْلَةُ الأَجْيَالِ مِنْ إِبْرَاهِيمَ إِلَى دَاوُدَ أَرْبَعَةَ عَشَرَ جِيلاً؛ وَمِنْ دَاوُدَ إِلَى السَّبْيِ الْبَابِلِيِّ أَرْبَعَةَ عَشَرَ جِيلاً؛ وَمِنَ السَّبْيِ الْبَابِلِيِّ إِلَى الْمَسِيحِ أَرْبَعَةَ عَشَرَ جِيلاً.</span>
</div>
</div>

## ميلاد يسوع المسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الرسالة إلى مؤمني غلاطية 4:4–5; الإنجيل كما دونه لوقا 1:27–38; أيوب 14:4"> أَمَّا يَسُوعُ الْمَسِيحُ فَقَدْ تَمَّتْ وِلادَتُهُ هكَذَا: كَانَتْ أُمُّهُ مَرْيَمُ مَخْطُوبَةً لِيُوسُفَ؛ وَقَبْلَ أَنْ يَجْتَمِعَا مَعاً، وُجِدَتْ حُبْلَى مِنَ الرُّوحِ الْقُدُسِ.</span>
<span className="v" id="v19" data-v="19" data-xref="التثنية 22:21–24; التثنية 24:1–4; الإنجيل كما دونه يوحنا 8:4–5"> وَإِذْ كَانَ يُوسُفُ خَطِيبُهَا رَجُلاً صَالِحاً، وَلَمْ يُرِدْ أَنْ يُشَهِّرَ بِها، قَرَّرَ أَنْ يَتْرُكَهَا سِرّاً.</span>
<span className="v" id="v20" data-v="20" data-xref="الأمثال 3:5–6; الإنجيل كما دونه لوقا 1:19; الإنجيل كما دونه متى 1:18"> وَبَيْنَمَا كَانَ يُفَكِّرُ فِي الأَمْرِ، إِذَا مَلاكٌ مِنَ الرَّبِّ قَدْ ظَهَرَ لَهُ فِي حُلْمٍ يَقُولُ: «يَا يُوسُفُ ابْنَ دَاوُدَ! لَا تَخَفْ أَنْ تَأْتِيَ بِمَرْيَمَ عَرُوسِكَ إِلَى بَيْتِكَ، لأَنَّ الَّذِي هِيَ حُبْلَى بِهِ إِنَّمَا هُوَ مِنَ الرُّوحِ الْقُدُسِ.</span>
<span className="v" id="v21" data-v="21" data-xref="أعمال الرسل 13:23; أعمال الرسل 4:12; الإنجيل كما دونه يوحنا 1:29"> فَسَتَلِدُ ابْناً، وَأَنْتَ تُسَمِّيهِ يَسُوعَ، لأَنَّهُ هُوَ الَّذِي يُخَلِّصُ شَعْبَهُ مِنْ خَطَايَاهُمْ».</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه لوقا 24:44; الإنجيل كما دونه متى 2:23; ملوك الأول 8:15"> حَدَثَ هَذَا كُلُّهُ لِيَتِمَّ مَا قَالَهُ الرَّبُّ بِلِسَانِ النَّبِيِّ الْقَائِلِ:</span>
<span className="v" id="v23" data-v="23" data-xref="إشعياء 7:14; إشعياء 9:6–7; المزامير 46:11"> «هَا إِنَّ الْعَذْرَاءَ تَحْبَلُ، وَتَلِدُ ابْناً، وَيُدْعَى عِمَّانُوئِيلَ!» أَيِ «اللهُ مَعَنَا».</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه يوحنا 15:14; رسالة يعقوب 2:21–26; الإنجيل كما دونه يوحنا 2:5–8"> وَلَمَّا نَهَضَ يُوسُفُ مِنْ نَوْمِهِ، فَعَلَ مَا أَمَرَهُ بِهِ مَلاكُ الرَّبِّ؛ فَأَتَى بِعَرُوسِهِ إِلَى بَيْتِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه لوقا 2:21; الخروج 13:2; الإنجيل كما دونه متى 1:21"> وَلكِنَّهُ لَمْ يَدْخُلْ بِها حَتَّى وَلَدَتِ ابْناً، فَسَمَّاهُ يَسُوعَ.</span>
