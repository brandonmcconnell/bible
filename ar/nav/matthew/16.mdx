## الفريسيون يطلبون آية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه لوقا 11:16; الإنجيل كما دونه متى 16:6; أعمال الرسل 4:1"> وَجَاءَ بَعْضُ الْفَرِّيسِيِّينَ وَالصَّدُّوقِيِّينَ إِلَى يَسُوعَ لِيُوْقِعُوا بِهِ، فَطَلَبُوا إِلَيْهِ أَنْ يُرِيَهُمْ مُعْجِزَةً مِنَ السَّمَاءِ.</span>
<span className="v" id="v2" data-v="2" data-xref="الإنجيل كما دونه لوقا 12:54–56"> فَأَجَابَهُمْ: «إِذَا كَانَتِ السَّمَاءُ حَمْرَاءَ صَافِيَةً عِنْدَ الْغُرُوبِ، تَقُولُونَ: سَيَكُونُ الْجَوُّ صَحْواً!</span>
<span className="v" id="v3" data-v="3" data-xref="الإنجيل كما دونه لوقا 12:56; أخبار الأيام الأول 12:32; الإنجيل كما دونه متى 11:5"> وَإذَا كَانَتِ السَّمَاءُ حَمْرَاءَ مُلَبَّدَةً بِالْغُيُومِ فِي الصَّبَاحِ، تَقُولُونَ: الْيَوْمَ مَطَرٌ! إِنَّكُمْ تَسْتَدِلُّونَ عَلَى حَالَةِ الطَّقْسِ مِنْ مَنْظَرِ السَّمَاءِ. أَمَّا عَلامَاتُ الأَزْمِنَةِ، فَلا تَسْتَطِيعُونَ الِاسْتِدْلالَ عَلَيْهَا!</span>
<span className="v" id="v4" data-v="4" data-xref="الإنجيل كما دونه لوقا 11:29–30; الإنجيل كما دونه متى 12:39–40; يونان 1:17"> جِيلٌ شِرِّيرٌ خَائِنٌ يَطْلُبُ آيَةً، وَلَنْ يُعْطَى آيَةً إِلّا مَا حَدَثَ لِلنَّبِيِّ يُونَانَ». ثُمَّ فَارَقَهُمْ وَمَضَى.</span>
</div>
</div>

## خمير الفريسيين والصدوقيين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="الإنجيل كما دونه مرقس 8:13–14; الإنجيل كما دونه متى 15:39"> وَلَمَّا وَصَلَ تَلامِيذُهُ إِلَى الشَّاطِئِ الآخَرِ، كَانُوا قَدْ نَسُوا أَنْ يَتَزَوَّدُوا خُبْزاً.</span>
<span className="v" id="v6" data-v="6" data-xref="الإنجيل كما دونه لوقا 12:1; الرسالة إلى مؤمني غلاطية 5:9; الإنجيل كما دونه متى 16:12"> وَقَالَ لَهُمْ يَسُوعُ: «انْتَبِهُوا! خُذُوا حِذْرَكُمْ مِنْ خَمِيرِ الْفَرِّيسِيِّينَ وَالصَّدُّوقِيِّينَ!»</span>
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه مرقس 8:16–18; الإنجيل كما دونه متى 15:16–18; أعمال الرسل 10:14"> فَبَدَأُوا يُحَاوِرُونَ بَعْضُهُمْ بَعْضاً، قَائِلِينَ: «هَذَا لأَنَّنَا لَمْ نَأْخُذْ خُبْزاً!»</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه متى 6:30; الإنجيل كما دونه متى 14:31; الإنجيل كما دونه متى 8:26"> وَعَلِمَ يَسُوعُ بِذَلِكَ، فَقَالَ لَهُمْ: «يَا قَلِيلِي الإِيمَانِ، لِمَاذَا تُحَاوِرُونَ بَعْضُكُمْ بَعْضاً لأَنَّكُمْ لَمْ تَأْخُذُوا خُبْزاً؟</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه متى 14:17–21; الإنجيل كما دونه يوحنا 6:9–13; الإنجيل كما دونه لوقا 9:13–17"> أَلا تَفْهَمُونَ بَعْدُ؟ أَمْ نَسِيتُمُ الأَرْغِفَةَ الْخَمْسَةَ الَّتِي أَشْبَعَتِ الْخَمْسَةَ الآلافِ، وَكَمْ قُفَّةً رَفَعْتُمْ مِنْهَا؟</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه متى 15:34–38; الإنجيل كما دونه مرقس 8:17–21; الإنجيل كما دونه مرقس 8:5–9"> أَوَ نَسِيتُمُ الأَرْغِفَةَ السَّبْعَةَ الَّتِي أَشْبَعَتِ الأَرْبَعَةَ الآلافِ، وَكَمْ سَلًّا رَفَعْتُمْ مِنْهَا؟</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه متى 16:6; الإنجيل كما دونه لوقا 12:56; الإنجيل كما دونه مرقس 8:21"> كَيْفَ لَا تَفْهَمُونَ أَنِّي لَمْ أَكُنْ أَعْنِي الْخُبْزَ حِينَ قُلْتُ لَكُمْ: خُذُوا حِذْرَكُمْ مِنْ خَمِيرِ الْفَرِّيسِيِّينَ وَالصَّدُّوقِيِّينَ؟»</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه متى 5:20; أعمال الرسل 23:8; الإنجيل كما دونه متى 15:4–9"> عِنْدَئِذٍ أَدْرَكَ التَّلامِيذُ أَنَّهُ لَمْ يَكُنْ يُحَذِّرُهُمْ مِنْ خَمِيرِ الْخُبْزِ، بَلْ مِنْ تَعْلِيمِ الْفَرِّيسِيِّينَ وَالصَّدُّوقِيِّينَ.</span>
</div>
</div>

## اعتراف بطرس بالمسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه لوقا 9:18–20; الإنجيل كما دونه مرقس 8:27–29; دانيال 7:13"> وَلَمَّا وَصَلَ يَسُوعُ إِلَى نَوَاحِي قَيْصَرِيَّةِ فِيلِبُّسَ، سَأَلَ تَلامِيذَهُ: «مَنْ يَقُولُ النَّاسُ إِنِّي أَنَا، ابْنَ الإِنْسَانِ؟»</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه مرقس 6:15; الإنجيل كما دونه متى 14:2; ملاخي 4:5"> فَأَجَابُوهُ: «يَقُولُ بَعْضُهُمْ إِنَّكَ يُوحَنَّا الْمَعْمَدَانُ، وَغَيْرُهُمْ إِنَّكَ النَّبِيُّ إِيلِيَّا، وآخَرُونَ إِنَّكَ إِرْمِيَا، أَوْ وَاحِدٌ مِنَ الأَنْبِيَاءِ».</span>
<span className="v" id="v15" data-v="15" data-xref="الإنجيل كما دونه لوقا 9:20; الإنجيل كما دونه متى 13:11; الإنجيل كما دونه مرقس 8:29"> فَسَأَلَهُمْ: «وَأَنْتُمْ، مَنْ تَقُولُونَ إِنِّي أَنَا؟»</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه متى 14:33; الإنجيل كما دونه يوحنا 11:27; رسالة يوحنا الأولى 4:15"> فَأَجَابَ سِمْعَانُ بُطْرُسُ قَائِلاً: «أَنْتَ هُوَ الْمَسِيحُ ابْنُ اللهِ الْحَيِّ!»</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه يوحنا 6:45; الرسالة إلى مؤمني أفسس 1:17–18; الإنجيل كما دونه متى 11:25–27"> فَقَالَ لَهُ يَسُوعُ: «طُوبَى لَكَ يَا سِمْعَانَ بْنَ يُونَا. فَمَا أَعْلَنَ لَكَ هَذَا لَحْمٌ وَدَمٌ، بَلْ أَبِي الَّذِي فِي السَّمَاوَاتِ.</span>
<span className="v" id="v18" data-v="18" data-xref="الرسالة إلى مؤمني أفسس 2:19–22; الرسالة الأولى إلى مؤمني كورنثوس 3:9–11; إشعياء 28:16"> وَأَنَا أَيْضاً أَقُولُ لَكَ: أَنْتَ صَخْرٌ. وَعَلَى هَذِهِ الصَّخْرَةِ أَبْنِي كَنِيسَتِي وَقُوَّاتُ الْجَحِيمِ لَنْ تَقْوَى عَلَيْهَا!</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه متى 18:18; إشعياء 22:22; رؤيا يوحنا 1:18"> وَأُعْطِيكَ مَفَاتِيحَ مَلَكُوتِ السَّمَاوَاتِ: فَكُلُّ مَا تَرْبِطُهُ عَلَى الأَرْضِ، يَكُونُ قَدْ رُبِطَ فِي السَّمَاءِ؛ وَمَا تَحُلُّهُ عَلَى الأَرْضِ، يَكُونُ قَدْ حُلَّ فِي السَّمَاءِ!»</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه مرقس 8:30; الإنجيل كما دونه لوقا 9:21; الإنجيل كما دونه يوحنا 1:41"> ثُمَّ حَذَّرَ تَلامِيذَهُ مِنْ أَنْ يَقُولُوا لأَحَدٍ إِنَّهُ هُوَ الْمَسِيحُ.</span>
</div>
</div>

## المسيح يعلن عن موته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه لوقا 24:6–7; الإنجيل كما دونه متى 20:17–19; الإنجيل كما دونه متى 17:22–23"> مِنْ ذَلِكَ الْوَقْتِ، بَدَأَ يَسُوعُ يُعْلِنُ لِتَلامِيذِهِ أَنَّهُ لابُدَّ أَنْ يَمْضِيَ إِلَى أُورُشَلِيمَ، وَيَتَأَلَّمَ عَلَى أَيْدِي الشُّيُوخِ وَرُؤَسَاءِ الْكَهَنَةِ وَالْكَتَبَةِ، وَيُقْتَلَ، وَفِي الْيَوْمِ الثَّالِثِ يُقَامُ.</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه مرقس 8:32; الإنجيل كما دونه يوحنا 13:6–8; الإنجيل كما دونه متى 16:16–17"> فَانْتَحَى بِهِ بُطْرُسُ جَانِباً، وَأَخَذَ يَلُومُهُ، قَائِلاً: «حَاشَا لَكَ يَا رَبُّ أَنْ يَحْدُثَ لَكَ هَذَا!»</span>
<span className="v" id="v23" data-v="23" data-xref="الرسالة إلى مؤمني روما 8:5–8; الرسالة إلى مؤمني كولوسي 3:2; الرسالة إلى مؤمني فيلبي 3:19"> فَالْتَفَتَ يَسُوعُ إِلَى بُطْرُسَ وَقَالَ لَهُ: «اغْرُبْ مِنْ أَمَامِي يَا شَيْطَانُ! أَنْتَ عَقَبَةٌ أَمَامِي، لأَنَّكَ تُفَكِّرُ لَا بِأُمُورِ اللهِ، بَلْ بِأُمُورِ النَّاسِ!»</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه متى 10:38; الإنجيل كما دونه لوقا 14:27; رسالة بطرس الأولى 4:1–2"> ثُمَّ قَالَ يَسُوعُ لِتَلامِيذِهِ: «إِنْ أَرَادَ أَحَدٌ أَنْ يَسِيرَ وَرَائِي، فَلْيُنْكِرْ نَفْسَهُ وَيَحْمِلْ صَلِيبَهُ وَيَتْبَعْنِي.</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه متى 10:39; الإنجيل كما دونه مرقس 8:35; رؤيا يوحنا 12:11"> فَأَيُّ مَنْ أَرَادَ أَنْ يُخَلِّصَ نَفْسَهُ، يَخْسَرُهَا؛ وَلكِنَّ مَنْ يَخْسَرُ نَفْسَهُ لأَجْلِي، فَإِنَّهُ يَجِدُهَا.</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه مرقس 8:36–37; أيوب 27:8; الإنجيل كما دونه لوقا 9:25"> فَمَاذَا يَنْتَفِعُ الإِنْسَانُ لَوْ رَبِحَ الْعَالَمَ كُلَّهُ وَخَسِرَ نَفْسَهُ؟ أَوْ مَاذَا يُقَدِّمُ الإِنْسَانُ فِدَاءً عَنْ نَفْسِهِ؟</span>
<span className="v" id="v27" data-v="27" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 5:10; المزامير 62:12; الرسالة إلى مؤمني روما 2:6"> فَإِنَّ ابْنَ الإِنْسَانِ سَوْفَ يَعُودُ فِي مَجْدِ أَبِيهِ مَعَ مَلائِكَتِهِ، فَيُجَازِي كُلَّ وَاحِدٍ حَسَبَ أَعْمَالِهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه لوقا 9:27; الإنجيل كما دونه مرقس 9:1; الإنجيل كما دونه يوحنا 8:52"> الْحَقَّ أَقُولُ لَكُمْ: إِنَّ بَعْضاً مِنَ الْوَاقِفِينَ هُنَا لَنْ يَذُوقُوا الْمَوْتَ، قَبْلَ أَنْ يَرَوْا ابْنَ الإِنْسَانِ آتِياً فِي مَلَكُوتِهِ».</span>
