## تكريس هرون وأبنائه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وَقَالَ الرَّبُّ لِمُوسَى:</span>
<span className="v" id="v2" data-v="2" data-xref="الخروج 28:2–4; الخروج 39:41; الرسالة إلى العبرانيين 7:27"> «أَحْضِرْ هَرُونَ وَأَبْنَاءَهُ وَثِيَابَهُمْ وَدُهْنَ الْمَسْحَةِ وَثَوْرَ ذَبِيحَةِ الْخَطِيئَةِ وَالْكَبْشَيْنِ وَسَلَّ الْفَطِيرِ،</span>
<span className="v" id="v3" data-v="3" data-xref="أخبار الأيام الثاني 30:13; أعمال الرسل 2:1; نحميا 8:1"> وَاجْمَعْ سَائِرَ بَنِي إِسْرَائِيلَ عِنْدَ مَدْخَلِ خَيْمَةِ الاجْتِمَاعِ».</span>
<span className="v" id="v4" data-v="4" data-xref="الخروج 39:26; الخروج 39:7; الخروج 39:5"> فَنَفَّذَ مُوسَى أَمْرَ الرَّبِّ، فَاجْتَمَعَتِ الْجَمَاعَةُ عِنْدَ مَدْخَلِ خَيْمَةِ الاجْتِمَاعِ،</span>
<span className="v" id="v5" data-v="5" data-xref="الخروج 29:4–37"> فَقَالَ لَهُمْ مُوسَى: «هَذَا مَا أَمَرَ الرَّبُّ بِهِ».</span>
<span className="v" id="v6" data-v="6" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 6:11; الرسالة إلى مؤمني أفسس 5:26; الخروج 29:4"> فَقَدَّمَ مُوسَى هَرُونَ وَأَبْنَاءَهُ وَغَسَّلَهُمْ بِمَاءٍ،</span>
<span className="v" id="v7" data-v="7" data-xref="الخروج 28:4; إشعياء 61:10; الرسالة إلى مؤمني غلاطية 3:27"> وَأَلْبَسَ هَرُونَ الْقَمِيصَ وَنَطَّقَهُ بِالْحِزَامِ، وَكَسَاهُ بِالْجُبَّةِ، وَوَضَعَ عَلَيْهِ الرِّدَاءَ، وَنَطَّقَهُ بِحِزَامِ الرِّدَاءِ وَشَدَّهُ بِهِ،</span>
<span className="v" id="v8" data-v="8" data-xref="عزرا 2:63; الرسالة الأولى إلى مؤمني تسالونيكي 5:8; إشعياء 59:17"> وَجَعَلَ عَلَيْهِ الصُّدْرَةَ، وَثَبَّتَ فِي الصُّدْرَةِ الْأُورِيمَ وَالتُّمِّيمَ،</span>
<span className="v" id="v9" data-v="9" data-xref="الرسالة إلى مؤمني فيلبي 2:9–11; زكريا 6:11–14; الخروج 28:36–38"> وَوَضَعَ مُوسَى عَلَى رَأْسِ هَرُونَ الْعِمَامَةَ، وَعَلَّقَ عَلَيْهَا فَوْقَ الْجَبْهَةِ صَفِيحَةَ الذَّهَبِ، الإِكْلِيلَ الْمُقَدَّسَ، تَمَاماً كَمَا أَمَرَهُ الرَّبُّ.</span>
<span className="v" id="v10" data-v="10" data-xref="الخروج 40:9–11; اللاويين 8:2; الخروج 30:23–29"> ثُمَّ أَخَذَ مُوسَى دُهْنَ الْمَسْحَةِ وَمَسَحَ الْمَسْكِنَ وَكُلَّ مَا فِيهِ وَقَدَّسَهُ لِلرَّبِّ.</span>
<span className="v" id="v11" data-v="11" data-xref="الرسالة إلى تيطس 3:6; الخروج 30:29; الخروج 16:19"> ثُمَّ رَشَّ مِنَ الدُّهْنِ عَلَى الْمَذْبَحِ سَبْعَ مَرَّاتٍ، وَمَسَحَ الْمَذْبَحَ وَجَمِيعَ أَوَانِيهِ وَحَوْضَ الاغْتِسَالِ وَقَاعِدَتَهُ لِتَقْدِيسِهَا،</span>
<span className="v" id="v12" data-v="12" data-xref="الخروج 30:30; المزامير 133:2; الخروج 29:7"> وَصَبَّ مِنْ دُهْنِ الْمَسْحَةِ عَلَى رَأْسِ هَرُونَ وَمَسَحَهُ تَكْرِيساً لَهُ.</span>
<span className="v" id="v13" data-v="13" data-xref="الخروج 29:8–9; رؤيا يوحنا 5:10; رؤيا يوحنا 1:6"> ثُمَّ أَحْضَرَ مُوسَى أَبْنَاءَ هَرُونَ وَأَلْبَسَهُمْ أَقْمِصَةً، وَنَطَّقَهُمْ بِأَحْزِمَةٍ، وَعَصَبَ عَلَى رُؤُوسِهِمِ الْقَلانِسَ كَمَا أَمَرَهُ الرَّبُّ.</span>
</div>
</div>

## ذبائح الخطيئة والمحرقات عن الكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="حزقيال 43:19; المزامير 66:15; رسالة بطرس الأولى 3:18"> ثُمَّ أَتَى بِثَوْرِ الْخَطِيئَةِ، فَوَضَعَ هَرُونُ وَأَبْنَاؤُهُ أَيْدِيَهُمْ عَلَى رَأْسِهِ،</span>
<span className="v" id="v15" data-v="15" data-xref="اللاويين 4:7; الرسالة إلى العبرانيين 9:18–23; حزقيال 43:19–27"> فَذَبَحَهُ مُوسَى، وَأَخَذَ الدَّمَ وَوَضَعَهُ عَلَى قُرُونِ الْمَذْبَحِ الأَرْبَعَةِ، وَعَلَى الْمَذْبَحِ نَفْسِهِ لِيُقَدِّسَهُ. ثُمَّ صَبَّ بَقِيَّةَ الدَّمِ عِنْدَ قَاعِدَتِهِ، وَكَرَّسَهُ تَكْفِيراً عَنْهُ.</span>
<span className="v" id="v16" data-v="16" data-xref="الخروج 29:13; اللاويين 3:3–5; اللاويين 4:8–9"> وَأَخَذَ مُوسَى شَحْمَ أَعْضَاءِ الثَّوْرِ الدَّاخِلِيَّةِ، وَكَذَلِكَ الْمَرَارَةَ وَالْكُلْيَتَيْنِ وَشَحْمَهُمَا وَأَحْرَقَهَا كُلَّهَا عَلَى الْمَذْبَحِ.</span>
<span className="v" id="v17" data-v="17" data-xref="اللاويين 4:11–12; الخروج 29:14; اللاويين 6:30"> وَأَمَّا جِلْدُ الثَّوْرِ وَلَحْمُهُ وَفَرْثُهُ فَأَحْرَقَهَا بِنَارٍ خَارِجَ الْمُخَيَّمِ كَمَا أَمَرَهُ الرَّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="اللاويين 8:2; اللاويين 1:4–13; الخروج 29:15–18"> ثُمَّ قَدَّمَ كَبْشَ الْمُحْرَقَةِ، فَوَضَعَ هَرُونُ وَأَبْنَاؤُهُ أَيْدِيَهُمْ عَلَى رَأْسِهِ،</span>
<span className="v" id="v19" data-v="19"> فَذَبَحَهُ مُوسَى وَرَشَّ الدَّمَ عَلَى جَوَانِبِ الْمَذْبَحِ،</span>
<span className="v" id="v20" data-v="20" data-xref="اللاويين 1:8"> وَقَطَّعَ الْكَبْشَ إِلَى أَجْزَائِهِ، ثُمَّ أَحْرَقَ الرَّأْسَ وَالأَجْزَاءَ وَالشَّحْمَ.</span>
<span className="v" id="v21" data-v="21" data-xref="الرسالة إلى مؤمني أفسس 5:2; الخروج 29:18; اللاويين 1:17"> وَأَمَّا أَمْعَاؤُهُ وَأَكَارِعُهُ فَقَدْ غَسَلَهَا بِمَاءٍ. وَأَحْرَقَ مُوسَى كُلَّ الْكَبْشِ عَلَى الْمَذْبَحِ، فَكَانَ مُحْرَقَةً لِرِضَى الرَّبِّ وَسُرُورِهِ. وَقُودٌ هُوَ لِلرَّبِّ، تَمَاماً كَمَا أَمَرَ الرَّبُّ.</span>
</div>

## كبش التكريس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اللاويين 8:2; اللاويين 8:29; الخروج 29:19–31"> ثُمَّ أَحْضَرَ الْكَبْشَ الثَّانِي، كَبْشَ التَّكْرِيسِ، فَوَضَعَ هَرُونُ وَأَبْنَاؤُهُ أَيْدِيَهُمْ عَلَى رَأْسِهِ،</span>
<span className="v" id="v23" data-v="23" data-xref="الخروج 29:20; اللاويين 14:14; الرسالة إلى مؤمني روما 12:1"> فَذَبَحَهُ مُوسَى، وَأَخَذَ مِنْ دَمِهِ فَوَضَعَهُ عَلَى شَحْمَةِ أُذْنِ هَرُونَ الْيُمْنَى، وَعَلَى إِبْهَامِ يَدِهِ الْيُمْنَى وَإِبْهَامِ رِجْلِهِ الْيُمْنَى.</span>
<span className="v" id="v24" data-v="24" data-xref="الرسالة إلى العبرانيين 9:18–22"> ثُمَّ قَدَّمَ مُوسَى أَبْنَاءَ هَرُونَ وَوَضَعَ مِنَ الدَّمِ عَلَى شَحْمِ آذَانِهِمِ الْيُمْنَى، وَعَلَى أَبَاهِمِ أَيْدِيهِمِ الْيُمْنَى، وَعَلَى أَبَاهِمِ أَرْجُلِهِمِ الْيُمْنَى، وَرَشَّ بَقِيَّةَ الدَّمِ عَلَى جَوَانِبِ الْمَذْبَحِ.</span>
<span className="v" id="v25" data-v="25" data-xref="اللاويين 3:9; اللاويين 3:3–5; إشعياء 53:10"> ثُمَّ أَخَذَ شَحْمَ الأَلْيَةِ وَكُلَّ شَحْمِ الأَعْضَاءِ الدَّاخِلِيَّةِ وَالْمَرَارَةَ وَالْكُلْيَتَيْنِ وَشَحْمَهُمَا وَالسَّاقَ الْيُمْنَى،</span>
<span className="v" id="v26" data-v="26" data-xref="الخروج 29:23; الرسالة الأولى إلى تيموثاوس 2:5; أعمال الرسل 5:12"> وَأَخَذَ مِنْ سَلِّ الْفَطِيرِ الَّذِي فِي حَضْرَةِ الرَّبِّ قُرْصَ فَطِيرٍ وَاحِداً، وَكَعْكَةً وَاحِدَةً مَعْجُونَةً بِزَيْتٍ وَرِقَاقَةً، وَوَضَعَهَا عَلَى الشَّحْمِ وَعَلَى السَّاقِ الْيُمْنَى،</span>
<span className="v" id="v27" data-v="27" data-xref="إرميا 30:21; الرسالة إلى العبرانيين 9:14; الخروج 29:24–37"> وَوَضَعَهَا جَمِيعاً عَلَى كَفَّيْ هَرُونَ وَأَكُفِّ أَبْنَائِهِ لِيُرَجِّحُوهَا أَمَامَ الرَّبِّ،</span>
<span className="v" id="v28" data-v="28" data-xref="الخروج 29:25; المزامير 22:13–14; الرسالة إلى العبرانيين 10:14–22"> ثُمَّ أَخَذَهَا مُوسَى عَنْ أَكُفِّهِمْ وَأَحْرَقَهَا أَمَامَ الرَّبِّ عَلَى الْمَذْبَحِ فَوْقَ الْمُحْرَقَةِ، فَكَانَتْ قُرْبَانَ تَكْرِيسٍ لِرِضَى الرَّبِّ وَسُرُورِهِ. إِنَّهَا مُحْرَقَةٌ لِلرَّبِّ.</span>
<span className="v" id="v29" data-v="29" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 10:31; رسالة بطرس الأولى 4:11; اللاويين 7:30–34"> وَتَنَاوَلَ مُوسَى صَدْرَ كَبْشِ التَّكْرِيسِ وَرَجَّحَهُ أَمَامَ الرَّبِّ، فَكَانَ الصَّدْرُ نَصِيبَ مُوسَى كَمَا أَمَرَهُ الرَّبُّ.</span>
<span className="v" id="v30" data-v="30" data-xref="الخروج 29:21; العدد 3:3; الخروج 30:30"> ثُمَّ أَخَذَ مُوسَى بَعْضَ دُهْنِ الْمَسْحَةِ وَبَعْضَ الدَّمِ الَّذِي عَلَى الْمَذْبَحِ، وَرَشَّهَا عَلَى هَرُونَ وَثِيَابِهِ، وَعَلَى أَبْنَاءِ هَرُونَ وَثِيَابِهِمْ، فَقَدَّسَهُمْ جَمِيعاً مَعَ مَلابِسِهِمْ لِلرَّبِّ.</span>
<span className="v" id="v31" data-v="31" data-xref="الخروج 29:31–32; الرسالة إلى مؤمني غلاطية 2:20; صموئيل الأول 2:13–17"> ثُمَّ قَالَ لِهَرُونَ وَأَبْنَائِهِ: «اطْبُخُوا اللَّحْمَ عِنْدَ مَدْخَلِ خَيْمَةِ الاجْتِمَاعِ، وَكُلُوهُ هُنَاكَ مَعَ خُبْزِ سَلِّ التَّكْرِيسِ، كَمَا أَوْصَيْتُ أَلّا يَأْكُلَهُ سِوَى هَرُونَ وَأَبْنَائِهِ.</span>
<span className="v" id="v32" data-v="32" data-xref="الخروج 29:34; الخروج 12:10; الرسالة إلى العبرانيين 3:13–14"> وَمَا تَبَقَّى مِنَ اللَّحْمِ وَالْخُبْزِ تُحْرِقُونَهُ بِنَارٍ.</span>
<span className="v" id="v33" data-v="33" data-xref="الخروج 29:35; العدد 19:12; حزقيال 43:25–27"> وَلا تُفَارِقُوا مَدْخَلَ خَيْمَةِ الاجْتِمَاعِ سَبْعَةَ أَيَّامٍ، إِلَى يَوْمِ اكْتِمَالِ تَكْرِيسِكُمْ، لأَنَّ الرَّبَّ يُكَرِّسُكُمْ سَبْعَةَ أَيَّامٍ.</span>
<span className="v" id="v34" data-v="34" data-xref="الرسالة إلى العبرانيين 7:27; الرسالة إلى العبرانيين 7:16; الرسالة إلى العبرانيين 10:11–12"> إِنَّ مَا جَرَى الْيَوْمَ هُوَ مَا أَمَرَ بِهِ الرَّبُّ تَكْفِيراً عَنْكُمْ.</span>
<span className="v" id="v35" data-v="35" data-xref="العدد 3:7; التثنية 11:1; ملوك الأول 2:3"> فَامْكُثُوا عِنْدَ مَدْخَلِ خَيْمَةِ الاجْتِمَاعِ سَبْعَةَ أَيَّامٍ، لَيْلاً وَنَهَاراً، عَامِلِينَ شَعَائِرَ الرَّبِّ، فَلا تَمُوتُوا لأَنَّ هَذَا مَا أَوْصَيْتُ بِهِ».</span>
<span className="v" id="v36" data-v="36" data-xref="التثنية 12:32; التثنية 4:2; الخروج 40:16"> فَنَفَّذَ هَروُنُ وَأَبْنَاؤُهُ كُلَّ مَا أَمَرَ بِهِ الرَّبُّ عَلَى لِسَانِ مُوسَى.</span>
