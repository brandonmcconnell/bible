## ذبيحة الإِثم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اللاويين 6:17; اللاويين 5:14–7; اللاويين 6:25"> وَهَذِهِ نُصُوصُ تَعْلِيمَاتِ ذَبِيحَةِ الإِثْمِ. إِنَّهَا قُدْسُ أَقْدَاسٍ.</span>
<span className="v" id="v2" data-v="2" data-xref="اللاويين 1:5; اللاويين 6:25; اللاويين 1:11"> تَذْبَحُونَ ذَبِيحَةَ الإِثْمِ فِي نَفْسِ الْمَوْضِعِ الَّذِي تَذْبَحُونَ فِيهِ ذَبِيحَةَ الْمُحْرَقَةِ، وَيُرَشُّ دَمُهَا عَلَى جَوَانِبِ الْمَذْبَحِ الْمُحِيطَةِ بِهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخروج 29:13; اللاويين 3:15–16; المزامير 51:6"> وَيُقَرِّبُ الْكَاهِنُ مِنْهَا كُلَّ شَحْمِهَا: الأَلْيَةَ وَشَحْمَ الأَعْضَاءِ الدَّاخِلِيَّةِ،</span>
<span className="v" id="v4" data-v="4" data-xref="اللاويين 3:4"> وَالْكُلْيَتَيْنِ وَشَحْمَهُمَا الَّذِي عَلَى الْخَاصِرَتَيْنِ وَيَنْزِعُ الْمَرَارَةَ مَعَ الْكُلْيَتَيْنِ،</span>
<span className="v" id="v5" data-v="5" data-xref="اللاويين 1:13; الرسالة إلى مؤمني غلاطية 5:24; رسالة بطرس الأولى 4:1–2"> وَيُحْرِقُهَا الْكَاهِنُ عَلَى الْمَذْبَحِ وَقُوداً لِلرَّبِّ. إِنَّهَا قُرْبَانُ إِثْمٍ.</span>
<span className="v" id="v6" data-v="6" data-xref="اللاويين 6:29; العدد 18:9–10; اللاويين 2:3"> كُلُّ ذَكَرٍ مِنَ الْكَهَنَةِ يَأْكُلُ مِنْهَا فِي مَكَانٍ مُقَدَّسٍ، لأَنَّهَا قُدْسُ أَقْدَاسٍ.</span>
<span className="v" id="v7" data-v="7" data-xref="اللاويين 6:25–26; اللاويين 14:13; الرسالة الأولى إلى مؤمني كورنثوس 9:13"> وَشَرِيعَةُ ذَبِيحَةِ الإِثْمِ مُمَاثِلَةٌ لِذَبِيحَةِ الْخَطِيئَةِ، إِذْ تَكُونُ مِنْ نَصِيبِ الْكَاهِنِ الَّذِي يُكَفِّرُ بِها.</span>
<span className="v" id="v8" data-v="8" data-xref="اللاويين 4:11; الخروج 29:14; الرسالة إلى مؤمني روما 13:14"> وَالْكَاهِنُ الَّذِي يُقَرِّبُ مُحْرَقَةَ إِنْسَانٍ يَكُونُ جِلْدُ الْمُحْرَقَةِ الْمُقَرَّبَةِ مِنْ نَصِيبِهِ أَيْضاً،</span>
<span className="v" id="v9" data-v="9" data-xref="العدد 18:9; حزقيال 44:29; اللاويين 2:10"> وَكَذَلِكَ كُلُّ تَقْدِمَةِ دَقِيقٍ مَخْبُوزٍ فِي فُرْنٍ أَوْ مِقْلَاةٍ أَوْ عَلَى الصَّاجِ تَكُونُ لِلْكَاهِنِ الَّذِي يُقَرِّبُهَا</span>
<span className="v" id="v10" data-v="10" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 8:14; الخروج 16:18"> وَكُلُّ تَقْدِمَةِ دَقِيقٍ بِزَيْتٍ أَوْ جَافَّةٍ تَكُونُ مِنْ نَصِيبِ أَبْنَاءِ هَرُونَ، تُوَزَّعُ عَلَيْهِمْ بِالتَّسَاوِي.</span>
</div>
</div>

## تقدمة الدقيق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اللاويين 22:18–21; حزقيال 45:15; اللاويين 3:1–17"> وَهَذِهِ نُصُوصُ تَعْلِيمَاتِ ذَبِيحَةِ السَّلامِ الْمُقَرَّبَةِ إِلَى الرَّبِّ:</span>
<span className="v" id="v12" data-v="12" data-xref="اللاويين 2:4; العدد 6:15; رسالة بطرس الأولى 2:5"> إِنْ قَرَّبَهَا أَحَدٌ لأَجْلِ الشُّكْرِ، فَلْيُقَدِّمْ مَعَهَا كَعْكاً غَيْرَ مُخْتَمِرٍ مَعْجُوناً بِزَيْتٍ، وَرِقَاقَ فَطِيرٍ مَدْهُونَةً بِالزَّيْتِ وَدَقِيقاً مَلْتُوتاً بِزَيْتٍ.</span>
<span className="v" id="v13" data-v="13" data-xref="عاموس 4:5; اللاويين 23:17; الإنجيل كما دونه متى 13:33"> فَضْلاً عَنْ أَرْغِفَةِ خُبْزٍ مُخْتَمِرَةٍ، يُقَرِّبُهَا مَعَ ذَبِيحَةِ شُكْرِ سَلامَتِهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="العدد 18:19; الخروج 29:27–28; العدد 18:24–32"> وَعَلَيْهِ أَنْ يُقَدِّمَ وَاحِداً مِنْ كُلِّ قُرْبَانٍ يَرْفَعُهُ وَيُرَجِّحُهُ أَمَامَ الرَّبِّ وَيَكُونُ مِنْ نَصِيبِ الْكَاهِنِ الَّذِي يَرُشُّ دَمَ ذَبِيحَةِ السَّلامِ.</span>
<span className="v" id="v15" data-v="15" data-xref="اللاويين 22:29–30"> أَمَّا لَحْمُ ذَبِيحَةِ شُكْرِ سَلامَتِهِ فَيُؤْكَلُ فِي نَفْسِ يَوْمِ تَقْدِيمِ الذَّبِيحَةِ كَقُرْبَانٍ، لَا يُبْقِي مِنْهُ شَيْئاً إِلَى الصَّبَاحِ.</span>
<span className="v" id="v16" data-v="16" data-xref="اللاويين 19:5–8; التثنية 12:6; المزامير 116:14"> وَإذَا كَانَتْ ذَبِيحَةُ تَقْدِمَتِهِ نَذْراً أَوْ ذَبِيحَةً اخْتِيَارِيَّةً أُخْرَى، فَإِنَّهَا تُؤْكَلُ فِي يَوْمِ تَقْدِيمِهَا، وَفِي الْيَوْمِ التَّالِي يُؤْكَلُ مَا فَضَلَ مِنْهَا،</span>
<span className="v" id="v17" data-v="17" data-xref="الخروج 12:10; هوشع 6:2; الرسالة الأولى إلى مؤمني كورنثوس 15:4"> وَيُحْرَقُ كُلُّ مَا يَتَبَقَّى مِنَ الذَّبِيحَةِ لِلْيَوْمِ الثَّالِثِ.</span>
<span className="v" id="v18" data-v="18" data-xref="اللاويين 19:7–8; اللاويين 11:41; العدد 18:27"> وَكُلُّ مَنْ يَأْكُلُ مِنْ ذَبِيحَةِ سَلامَتِهِ فِي الْيَوْمِ الثَّالِثِ يَكُونُ مُذْنِباً، لأَنَّهَا تُصْبِحُ غَيْرَ مَقْبُولَةٍ وَلا يَحْسَبُهَا اللهُ لِمَنْ قَرَّبَهَا إِذْ تَكُونُ نَجِسَةً.</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه لوقا 11:41; الرسالة إلى تيطس 1:15; أعمال الرسل 10:15–16"> وَأَيُّ لَحْمٍ يَمَسُّ شَيْئاً نَجِساً لَا تَأْكُلُوا مِنْهُ بَلْ أَحْرِقُوهُ بِالنَّارِ. لَا يَأْكُلُ مَنَ اللَّحْمِ إِلا مَنْ كَانَ طَاهِراً.</span>
<span className="v" id="v20" data-v="20" data-xref="اللاويين 15:2–33; التكوين 17:14; اللاويين 22:3–7"> وَكُلُّ نَجِسٍ يَأْكُلُ مِنْ لَحْمِ ذَبِيحَةِ السَّلامِ الْمُكَرَّسَةِ لِلرَّبِّ يُسْتَأْصَلُ مِنْ بَيْنِ شَعْبِهِ</span>
<span className="v" id="v21" data-v="21" data-xref="اللاويين 12:1–59; اللاويين 11:20; حزقيال 4:14"> وَالنَّفْسُ الَّتِي تَمَسُّ شَيْئاً نَجِساً، سَوَاءٌ أَكَانَ نَجَاسَةَ إِنْسَانٍ أَمْ حَيَوَانٍ أَمْ مَكْرُوهاً مَا نَجِساً، ثُمَّ تَتَنَاوَلُ مِنْ لَحْمِ ذَبِيحَةِ السَّلامِ، تُسْتَأْصَلُ تِلْكَ النَّفْسُ مِنْ بَيْنِ شَعْبِهَا».</span>
</div>
</div>

## تحريم أكل الشحم والدم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22"> وَقَالَ الرَّبُّ لِمُوسَى:</span>
<span className="v" id="v23" data-v="23" data-xref="اللاويين 3:16–17; اللاويين 17:6; صموئيل الأول 2:15–17"> «أَوْصِ بَنِي إِسْرَائِيلَ: لَا تَأْكُلُوا كُلَّ شَحْمِ ثَوْرٍ أَوْ كَبْشٍ أَوْ مَاعِزٍ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخروج 22:31; اللاويين 17:15; اللاويين 22:8"> أَمَّا شَحْمُ الْحَيَوَانَاتِ الْمَيْتَةِ أَوِ الْمُفْتَرَسَةِ فَاسْتَخْدِمُوهُ فِي أَغْرَاضِكُمُ الْمُخْتَلِفَةِ، إِنَّمَا إِيَّاكُمْ أَنْ تَأْكُلُوا مِنْهُ.</span>
<span className="v" id="v25" data-v="25"> مَنْ يَأْكُلُ شَحْمَ بَهَائِمِ الْمُحْرَقَاتِ الَّتِي تُقَدِّمُونَهَا لِلرَّبِّ يُبَادُ مِنْ بَيْنِ شَعْبِهِ.</span>
<span className="v" id="v26" data-v="26" data-xref="التكوين 9:4; اللاويين 3:17; أعمال الرسل 15:20"> لَا تَأْكُلُوا فِي جَمِيعِ مَسَاكِنِكُمْ مِنْ دَمِ الطَّيْرِ أَوِ الْبَهَائِمِ.</span>
<span className="v" id="v27" data-v="27" data-xref="الرسالة إلى العبرانيين 10:29; اللاويين 7:20–21; اللاويين 7:25"> مَنْ يَأْكُلُ شَيْئاً مِنَ الدَّمِ يُبَادُ مِنْ بَيْنِ شَعْبِهِ».</span>

## نصيب الكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28"> وَقَالَ الرَّبُّ لِمُوسَى:</span>
<span className="v" id="v29" data-v="29" data-xref="الرسالة إلى مؤمني كولوسي 1:20; رسالة يوحنا الأولى 1:7; اللاويين 3:1–17"> «أَوْصِ بَنِي إِسْرَائِيلَ: مَنْ يُقَدِّمُ لِلرَّبِّ ذَبِيحَةَ سَلامَتِهِ عَلَيْهِ أَنْ يُحْضِرَهَا بِنَفْسِهِ.</span>
<span className="v" id="v30" data-v="30" data-xref="العدد 6:20; اللاويين 3:9; اللاويين 3:14"> هُوَ نَفْسُهُ يَأْتِي بِوَقَائِدِ الرَّبِّ. يَأْتِي بِالشَّحْمِ وَالصَّدْرِ. فَيُرَجِّحُ الصَّدْرَ أَمَامَ الرَّبِّ،</span>
<span className="v" id="v31" data-v="31" data-xref="اللاويين 3:5; اللاويين 3:11; اللاويين 3:16"> أَمَّا الشَّحْمُ فَيُوْقِدُهُ الْكَاهِنُ عَلَى الْمَذْبَحِ، وَيَكُونُ الصَّدْرُ مِنْ نَصِيبِ هَرُونَ وَبَنِيهِ.</span>
<span className="v" id="v32" data-v="32" data-xref="العدد 6:20; اللاويين 7:34; اللاويين 9:21"> وَتُقَدِّمُونَ السَّاقَ الْيُمْنَى مِنْ ذَبَائِحِ سَلامَتِكُمْ إِلَى الْكَاهِنِ نَصِيباً لَهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="اللاويين 26:1–46; اللاويين 6:1–30; اللاويين 7:3"> فَمَنْ يُقَرِّبُ دَمَ قُرْبَانِ السَّلامِ وَالشَّحْمَ مِنْ أَبْنَاءِ هَرُونَ تَكُونُ السَّاقُ الْيُمْنَى نَصِيباً لَهُ،</span>
<span className="v" id="v34" data-v="34" data-xref="العدد 18:18–19; اللاويين 10:14–15; الخروج 29:9"> لأَنَّنِي قَدْ أَخَذْتُ صَدْرَ التَّرْجِيحِ وَسَاقَ ذَبِيحَةِ سَلامِ بَنِي إِسْرَائِيلَ وَأَعْطَيْتُهَا لِهَرُونَ الْكَاهِنِ وَأَبْنَائِهِ، فَرِيضَةً دَائِمَةً، جِيلاً بَعْدَ جِيلٍ.</span>
<span className="v" id="v35" data-v="35" data-xref="رسالة يوحنا الأولى 2:20; رسالة يوحنا الأولى 2:27; العدد 18:7–19"> هَذَا هُوَ نَصِيبُ هَرُونَ وَنَصِيبُ أَبْنَائِهِ مِنْ مُحْرَقَاتِ الرَّبِّ، يَوْمَ تَكْرِيسِهِمْ لِيَكُونُوا كَهَنَةً لِلرَّبِّ.</span>
<span className="v" id="v36" data-v="36" data-xref="اللاويين 8:12; اللاويين 8:30; الخروج 40:13–15"> وَقَدْ أَمَرَ الرَّبُّ أَنْ يُعْطَى لَهُمْ يَوْمَ تَكْرِيسِهِمْ مِنْ بَيْنِ بَنِي إِسْرَائِيلَ، فَرِيضَةً دَائِمَةً جِيلاً بَعْدَ جِيلٍ.</span>
<span className="v" id="v37" data-v="37" data-xref="الخروج 29:1; اللاويين 1:1–6; اللاويين 6:9–18"> تِلْكَ هِي نُصُوصُ تَعْلِيمَاتِ شَرِيعَةِ الْمُحْرَقَةِ وَتَقْدِمَةِ الدَّقِيقِ وَذَبِيحَةِ الْخَطيئَةِ وَذَبِيحَةِ الإِثْمِ، وَذَبِيحَةِ التَّكْرِيسِ وَذَبِيحَةِ السَّلامِ،</span>
<span className="v" id="v38" data-v="38" data-xref="اللاويين 1:1–2"> الَّتِي أَوْصَى الرَّبُّ بِها مُوسَى فِي جَبَلِ سِينَاءَ، عِنْدَمَا أَمَرَ بَنِي إِسْرَائِيلَ بِتَقْدِيمِ قَرَابِينِهِمْ لِلرَّبِّ فِي صَحْرَاءِ سِينَاءَ».</span>
