## عشائر يهوذا الأخرى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التكوين 46:12; التكوين 38:29; العدد 26:20–21"> وَهَذَا سِجِلٌّ بِمَوَالِيدِ يَهُوذَا: فَارَصُ، وَحَصْرُونُ وَكَرْمِي وَحُورُ وَشُوبَالُ.</span>
<span className="v" id="v2" data-v="2" data-xref="القضاة 13:25; أخبار الأيام الأول 2:52–54; يشوع 15:33"> وَأَنْجَبَ رَآيَا بْنُ شُوبَالَ يَحَثَ، وَأَنْجَبَ يَحَثُ أَخُومَايَ وَلاهَدَ. وَاسْتَوْطَنَ نَسْلُهُمَا فِي صَرْعَةَ.</span>
<span className="v" id="v3" data-v="3" data-xref="القضاة 15:11; أخبار الأيام الثاني 11:6"> وَهَذِهِ أَسْمَاءُ أَبْنَاءِ عِيطَمَ: يزْرَعِيلُ وَيَشْمَا وَيَدْبَاشُ، وَاسْمُ أُخْتِهِمْ هَصَّلَلْفُونِي.</span>
<span className="v" id="v4" data-v="4" data-xref="أخبار الأيام الأول 4:18; أخبار الأيام الأول 2:50; أخبار الأيام الأول 4:39"> وَفَنُوئِيلُ الَّذِي أَسَّسَ مَدِينَةَ جَدُورَ، وَعَازَرُ مُؤَسِّسُ مَدِينَةِ حُوشَةَ. وَجَمِيعُ هَؤُلاءِ مِنْ ذُرِّيَّةِ حُورَ بِكْرِ كَالَبَ مِنْ زَوْجَتِهِ أَفْرَاتَةَ. وَهُوَ الَّذِي قَامَ بِبِنَاءِ مَدِينَةِ بَيْتِ لَحْمٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="أخبار الأيام الأول 2:24"> وَكَانَ لأَشْحُورَ مُؤَسِّسِ مَدِينَةِ تَقُوعَ زَوْجَتَانِ هُمَا: حَلاةُ وَنَعْرَةُ.</span>
<span className="v" id="v6" data-v="6"> فَأَنْجَبَتْ لَهُ نَعْرَةُ أَخُزَّامَ وَحَافَرَ وَالتَّيْمَانِيَّ وَالأَخَشْتَارِيَّ. وَجَمِيعُ هَؤُلاءِ أَبْنَاءُ نَعْرَةَ.</span>
<span className="v" id="v7" data-v="7"> أَمَّا أَبْنَاءُ حَلاةَ فَهُمْ: صَرَثُ وَصُوحَرُ وَأَثْنَانُ.</span>
<span className="v" id="v8" data-v="8"> وَأَنْجَبَ قُوصٌ عَانُوبَ وَهَصُوبِيبَةَ، وَتَحَدَّرَتْ مِنْهُ عَشَائِرُ أَخَرْحِيلَ بْنِ هَارُمَ.</span>
<span className="v" id="v9" data-v="9" data-xref="صموئيل الأول 4:21; أخبار الأيام الأول 7:23; التكوين 35:18"> وَكَانَ يَعْبِيصُ أَنْبَلَ إِخْوَتِهِ وَقَدْ سَمَّتْهُ أُمُّهُ يَعْبِيصَ قَائِلَةً: «لأَنَّنِي عَانَيْتُ فِي وِلادَتِهِ».</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه متى 7:7–11; المزامير 116:1–4; الأمثال 10:22"> وَتَضَرَّعَ يَعْبِيصُ لإِلَهِ إِسْرَائِيلَ قَائِلاً: «لَيْتَكَ تُبَارِكُنِي وَتُوَسِّعُ مِنْ حُدُودِ أَرْضِي، وَتَعْضُدُنِي، وَتَقِينِي مِنَ الشَّرِّ فَلا يُشْقِينِي». فَاسْتَجَابَ اللهُ دُعَاءَهُ.</span>
<span className="v" id="v11" data-v="11"> وَأَنْجَبَ كَلُوبُ أَخُو شُوحَةَ مَحِيرَ أَبَا أَشْتُونَ.</span>
<span className="v" id="v12" data-v="12"> وَأَنْجَبَ أَشْتُونُ بَيْتَ رَافَا، وَفَاسِحَ وَتَحِنَّةَ الَّذِي أَسَّسَ مَدِينَةَ نَاحَاشَ، وَجَمِيعُ هَؤُلاءِ أَهْلُ رَيْكَةَ.</span>
<span className="v" id="v13" data-v="13" data-xref="يشوع 15:17; القضاة 1:13; القضاة 3:9–11"> وَابْنَا قَنَازَ هُمَا: عُثْنِيئِيلُ وَسَرَايَا. وَأَنْجَبَ عُثْنِيئِيلُ حَثَاثَ.</span>
<span className="v" id="v14" data-v="14" data-xref="نحميا 11:35; ملوك الثاني 24:14"> وَمَعُونُوثَايُ وَلَدَ عَفْرَةَ. وَأَنْجَبَ سَرَايَا يُوآبَ الَّذِي أَسَّسَ وَادِي الصُّنَّاعِ مَقَرَّ إِقَامَةِ الصُّنَّاعِ.</span>
<span className="v" id="v15" data-v="15" data-xref="العدد 13:6; العدد 14:6–10; يشوع 15:13–20"> وَأَنْجَبَ كَالَبُ بْنُ يَفُنَّةَ عِيرُو وَأَيْلَةَ وَنَاعِمَ، وَوَلَدَ أَيْلَةُ قَنَازَ.</span>
<span className="v" id="v16" data-v="16"> أَمَّا أَبْنَاءُ يَهْلَلْئِيلَ فَهُمْ زِيفُ وَزِيفَةُ وَتِيرِيَّا وَأَسَرْئِيلُ.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبار الأيام الأول 4:19; أخبار الأيام الأول 6:57; يشوع 15:50"> وَأَبْنَاءُ عَزْرَةَ هُمْ: يَثَرُ وَمَرَدُ وَعَافِرُ وَيَالُونُ. وَتَزَوَّج مَرَدُ بِثْيَةَ ابْنَةَ فِرْعَوْنَ فَأَنْجَبَتْ لَهُ مَرْيَمَ وَشَمَّايَ وَيِشْبَحَ مُؤَسِّسَ مَدِينَةِ أَشْتَمُوعَ.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبار الأيام الأول 2:42; أخبار الأيام الأول 4:39; يشوع 15:58"> أَمَّا زَوْجَتُهُ الْيَهُودِيَّةُ فَقَدْ أَنْجَبَتْ لَهُ يَارَدَ الَّذِي أَسَّسَ مَدِينَةَ جَدُورَ، وَحَابِرَ مُؤَسِّسَ مَدِينَةِ سُوكُوَ، وَيَقُوثِيئِيلَ مُؤَسِّسَ مَدِينَةِ زَانُوحَ.</span>
<span className="v" id="v19" data-v="19" data-xref="يشوع 15:44; صموئيل الأول 23:1–13"> وَكَانَتْ زَوْجَةُ هُودِيَةَ شَقِيقَةَ نَحَمَ، وَقَدْ أَسَّسَ أَحَدُ وَلَدَيْهَا مَدِينَةَ قَعِيلَةَ الَّتِي قَطَنَتْهَا قَبِيلَةُ جَرْمٍ، وَأَسَّسَ الآخَرُ مَدِينَةَ أَشْتَمُوعَ الَّتِي اسْتَوْطَنَتْهَا قَبِيلَةُ مَعْكَةَ.</span>
<span className="v" id="v20" data-v="20"> وَأَبْنَاءُ شِيمُونَ: أَمْنُونُ وَرِنَّةُ بْنُ حَانَانَ وَتِيلُونُ. وَابْنَا يِشْعِي: زُوحَيْتُ وَبَنْزُوحَيْتُ.</span>
<span className="v" id="v21" data-v="21" data-xref="التكوين 38:5; أخبار الأيام الأول 2:3; العدد 26:20"> وَأَبْنَاءُ شِيلَةَ بْنِ يَهُوذَا: عِيرُ مُؤَسِّسُ مَدِينَةِ لَيْكَةَ، وَلَعْدَةُ مُؤَسِّسُ مَدِينَةِ مَرِيشَةَ وَرَأْسُ نَسَّاجِي الْكَتَّانِ الَّذِينَ سَكَنُوا فِي بَيْتِ أَشْبَيْعَ.</span>
<span className="v" id="v22" data-v="22"> وَيُوقِيمُ، وَأَهْلُ مَدِينَةِ كَزِيبَا، وَيُوآشُ وَسَارَافُ الَّذِي حَكَمَ فِيِ مُوآبَ قَبْلَ أَنْ يَرْجِعَ إِلَى يَشُوبِي لَحْمٍ. وَهَذِهِ أَخْبَارٌ مَنْقُولَةٌ عَنْ سِجِلّاتٍ قَدِيمَةٍ.</span>
<span className="v" id="v23" data-v="23" data-xref="المزامير 81:6; أخبار الأيام الأول 4:14"> وَكَانَ هَؤُلاءِ خَزَّافِينَ يَعْمَلُونَ فِي خِدْمَةِ الْمَلِكِ، وَأَقَامُوا فِي مَدِينَتَيْ نَتَاعِيمَ وَجَدِيرَةَ.</span>

## شمعون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="التكوين 46:10; الخروج 6:15; العدد 26:12–14"> أَمَّا أَبْنَاءُ شِمْعُونَ فَهُمْ: نَمُوئِيلُ وَيَامِينُ وَيَرِيبُ وَزَارَحُ وَشَاوُلُ.</span>
<span className="v" id="v25" data-v="25"> وَأَنْجَبَ شَاوُلُ شَلُّومَ، وَشَلُّومُ مِبْسَامَ، وَمِبْسَامُ مِشْمَاعَ.</span>
<span className="v" id="v26" data-v="26"> وَأَنْجَبَ مِشْمَاعُ حَمُوئِيلَ، وَحَمُوئِيلُ زَكُّورَ وَالِدَ شِمْعِي.</span>
<span className="v" id="v27" data-v="27" data-xref="العدد 26:14; العدد 26:22; العدد 2:13"> وَكَانَ لِشِمْعِي سِتَّةَ عَشَرَ ابْناً وَسِتُّ بَنَاتٍ. وَأَمَّا إِخْوَتُهُ فَلَمْ يُعْقِبُوا أَبْنَاءً عَدِيدِينَ، وَلَمْ تَتَكَاثَرْ عَشَائِرُ سِبْطِ شِمْعُونَ كَمَا تَكَاثَرَتْ عَشَائِرُ أَبْنَاءِ يَهُوذَا.</span>
<span className="v" id="v28" data-v="28" data-xref="يشوع 19:2–3; يشوع 15:28–29; يشوع 19:9"> وَأَقَامَتْ عَشَائِرُهُمْ فِي بِئْرِ سَبْعٍ وَمُولادَةَ وَحَصَرِ شُوعَالَ،</span>
<span className="v" id="v29" data-v="29" data-xref="يشوع 19:3–4"> وَفِي بِلْهَةَ وَعَاصِمَ وَتُولادَ،</span>
<span className="v" id="v30" data-v="30" data-xref="نحميا 11:28; يشوع 19:4–5; صموئيل الأول 30:1"> وَبَتُوئِيلَ وَحُرْمَةَ وَصِقْلَغَ،</span>
<span className="v" id="v31" data-v="31" data-xref="يشوع 19:5–6"> وَفِي بَيْتِ مَرْكَبُوتَ وَحَصَرِ سُوسِيمَ وَبَيْتِ بِرْئِي وَشَعَرَايِمَ. فَكَانَتْ هَذِهِ مُدُنَهُمُ الَّتِي أَقَامُوا فِيهَا إِلَى أَيَّامِ الْمَلِكِ دَاوُدَ.</span>
<span className="v" id="v32" data-v="32" data-xref="يشوع 19:7"> أَمَّا قُرَاهُمْ فَكَانَتْ: عِيطَمَ وَعَيْنَ وَرِمُّونَ وَتُوكَنَ وَعَاشَانَ، وَهِيَ فِي جُمْلَتِهَا خَمْسُ قُرىً،</span>
<span className="v" id="v33" data-v="33" data-xref="يشوع 19:8"> فَضْلاً عَنِ الضَّوَاحِي الْمُحِيطَةِ بِهَذِهِ الْقُرَى حَتَّى حُدُودِ بَعْلٍ. تِلْكَ هِيَ مُسْتَوْطَنَاتُهُمْ وَسِجِلاتُ أَنْسَابِهِمْ.</span>
<span className="v" id="v34" data-v="34"> وَمِنْ رُؤَسَاءِ عَائِلاتِهِمْ: مَشُوبَابُ وَيَمْلِيكُ وَيُوشَا بْنُ أَمَصْيَا،</span>
<span className="v" id="v35" data-v="35"> وَيُوئِيلُ وَيَاهُو بْنُ يُوشِبْيَا بْنِ سَرَايَا بْنِ عَسِيئيِلَ،</span>
<span className="v" id="v36" data-v="36"> وَأَلِيُوعِينَايُ وَيَعْقُوبَا وَيَشُوحَايَا وَعَسَايَا وَعَدِيئِيلُ وَيَسِيمِيئِيلُ وَبَنايَا،</span>
<span className="v" id="v37" data-v="37"> وَزِيزَا بْنُ شِفْعِي بْنِ أَلُّونَ بْنِ يَدَايَا بْنِ شِمْرِي بْنِ شَمْعِيَا.</span>
<span className="v" id="v38" data-v="38" data-xref="أخبار الأيام الأول 5:24; التكوين 6:4"> وَجَمِيعُ هَؤُلاءِ الْوَارِدَةِ أَسْمَاؤُهُمْ هُمْ رُؤَسَاءُ فِي عَشَائِرِهِمْ، وَرُؤُوسٌ فِي بُيُوتَاتِ آبَائِهِمْ، وَقَدِ انْتَشَرُوا كَثِيراً</span>
<span className="v" id="v39" data-v="39" data-xref="أخبار الأيام الأول 4:4; أخبار الأيام الأول 4:18; يشوع 15:58"> حَتَّى بَلَغُوا فِي بَحْثِهِمْ عَنِ الْمَرَاعِي لِمَاشِيَتِهِمْ مَدْخَلَ جَدُورَ شَرْقَ الْوَادِي،</span>
<span className="v" id="v40" data-v="40" data-xref="القضاة 18:7–10; المزامير 105:23; المزامير 78:51"> وَهُنَاكَ عَثَرُوا عَلَى مَرَاعٍ خَصِيبَةٍ تَمْتَدُّ فيِ أَرَاضٍ شَاسِعَةٍ وَادِعَةٍ آمِنَةٍ، لأَنَّ نَسْلَ حَامَ كَانُوا قَدِ اسْتَوْطَنُوهَا مُنْذُ الْقِدَمِ.</span>
<span className="v" id="v41" data-v="41" data-xref="إشعياء 14:28–32; ملوك الثاني 18:8–12; أخبار الأيام الأول 4:34–38"> فَهَاجَمَ هَؤُلاءِ الرُّؤَسَاءُ، الَّذِينَ وَرَدَتْ أَسْمَاؤُهُمْ فِي أَيَّامِ حَزَقِيَّا مَلِكِ يَهُوذَا، سُكَّانَ الأَرْضِ وَقَلَعُوا خِيَامَهُمْ، وَقَضَوْا أَيْضاً عَلَى الْمَعُونِيِّينَ الَّذِينَ اسْتَوْطَنُوا مَعَ آلِ حَامٍ وَأَفْنَوْهُمْ إلَى هَذَا اليَوْمِ، ثُمَّ احْتَلُّوا أَرْضَهُمْ لِرِعَايَةِ مَوَاشِيهِمْ.</span>
<span className="v" id="v42" data-v="42" data-xref="التكوين 36:8–9; التثنية 1:2"> كَمَا انْطَلَقَ نَحْوُ خَمْسِ مِئَةِ رَجُلٍ مِنْهُمْ إِلَى جَبَلِ سَعِيرَ، وَعَلَى رَأْسِهِمْ فَلَطْيَا وَنَعْرِيَا ورَفَايَا وَعُزِّيئِيلُ أَبْنَاءُ يَشْعِي،</span>
<span className="v" id="v43" data-v="43" data-xref="صموئيل الأول 30:17; صموئيل الأول 15:7–8; صموئيل الثاني 8:12"> وَقَتَلُوا مَنْ بَقِيَ مِنْ عَمَالِيقَ، وَاسْتَوْطَنُوا مَكَانَهُمْ إِلَى هَذَا الْيَوْمِ.</span>
