## لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التكوين 46:11; الخروج 6:16; العدد 26:57"> أَمَّا أَبْنَاءُ لاوِي فَهُمْ: جَرْشُونُ وَقَهَاتُ وَمَرَارِي.</span>
<span className="v" id="v2" data-v="2" data-xref="الخروج 6:18; الخروج 6:21–24; أخبار الأيام الأول 23:12"> وَأَبْنَاءُ قَهَاتَ: عَمْرَامُ وَيِصْهَارُ وَحَبْرُونُ وَعُزِّيئِيلُ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخروج 6:20; اللاويين 10:1; اللاويين 10:12"> وَمِنْ ذُرِّيَّةِ عَمْرَامَ هَرُونُ وَمُوسَى وَمَرْيَمُ. وَأَنْجَبَ هَرُونُ نَادَابَ وَأَبِيهُو وَأَلِيعَازَارَ وَإِيثَامَارَ،</span>
<span className="v" id="v4" data-v="4" data-xref="أخبار الأيام الأول 6:50; الخروج 6:25; عزرا 7:1–5"> وَأَنْجَبَ أَلِيعَازَارُ فِينْحَاسَ، وَفِينْحَاسُ أَبِيشُوعَ،</span>
<span className="v" id="v5" data-v="5"> وَأَبِيشُوعُ بُقِّيَ، وَبُقِّي عُزِّيَ.</span>
<span className="v" id="v6" data-v="6"> وَأَنْجَبَ عُزِّي زَرَحْيَا، وَزَرَحْيَا مَرَايُوثَ،</span>
<span className="v" id="v7" data-v="7"> وَمَرَايُوثُ أَمَرْيَا، وَأَمَرْيَا أَخِيطُوبَ،</span>
<span className="v" id="v8" data-v="8" data-xref="صموئيل الثاني 8:17; صموئيل الثاني 15:27; ملوك الأول 1:44"> وَأَنْجَبَ أَخِيطُوبُ صَادُوقَ، وَصَادُوقُ أَخِيمَعَصَ،</span>
<span className="v" id="v9" data-v="9"> وَأَخِيمَعَصُ عَزَرْيَا، وَعَزَرْيَا يُوحَانَانَ،</span>
<span className="v" id="v10" data-v="10" data-xref="أخبار الأيام الثاني 3:4; أخبار الأيام الثاني 3:1; ملوك الأول 6:1–7"> الَّذِي أَنْجَبَ عَزَرْيَا. وَقَدْ أَصْبَحَ عَزَرْيَا رَئِيسَ الْكَهَنَةِ فِي الْهَيْكَلِ الَّذِي بَنَاهُ سُلَيْمَانُ فِي أُورُشَلِيمَ.</span>
<span className="v" id="v11" data-v="11" data-xref="عزرا 7:3"> وَأَنْجَبَ عَزَرْيَا أَمَرْيَا، وَأَمَرْيَا أَخِيطُوبَ،</span>
<span className="v" id="v12" data-v="12" data-xref="نحميا 11:11; أخبار الأيام الأول 9:11"> وَأَخِيطُوبُ صَادُوقَ، وَصَادُوقُ شَلُّومَ.</span>
<span className="v" id="v13" data-v="13" data-xref="أخبار الأيام الثاني 35:8; ملوك الثاني 22:12–14; ملوك الثاني 22:4"> وَأَنْجَبَ شَلُّومُ حِلْقِيَا، وَحِلْقِيَا عَزَرْيَا،</span>
<span className="v" id="v14" data-v="14" data-xref="ملوك الثاني 25:18; نحميا 11:11; زكريا 6:11"> وَعَزَرْيَا سَرَايَا، وَسَرَايَا يَهُوصَادَاقَ.</span>
<span className="v" id="v15" data-v="15" data-xref="حجي 2:2; ملوك الثاني 25:18; حجي 1:1"> وَذَهَبَ يَهُوصَادَاقُ فِي الأَسْرِ عِنْدَمَا سَمَحَ الرَّبُّ لِنَبُوخَذْنَصَّرَ بِسَبْيِ يَهُوذَا وَأُورُشَلِيمَ.</span>
<span className="v" id="v16" data-v="16" data-xref="الخروج 6:16; أخبار الأيام الأول 6:1"> وَأَبْنَاءُ لاوِي: جَرْشُومُ وَقَهَاتُ وَمَرَارِي.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبار الأيام الأول 23:7; العدد 3:18; العدد 3:21"> أَمَّا اسْمَا ابْنَيْ جَرْشُومَ فَهُمَا لِبْنِي وَشِمْعِي.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبار الأيام الأول 23:12; أخبار الأيام الأول 6:2–3"> وَأَبْنَاءُ قَهَاتَ: عَمْرَامُ وَيِصْهَارُ وَحَبْرُونُ وَعُزِّيئِيلُ.</span>
<span className="v" id="v19" data-v="19" data-xref="أخبار الأيام الأول 23:21; أخبار الأيام الأول 24:26; الخروج 6:19"> وَابْنَا مَرَارِي: مَحْلِي وَمُوشِي. وَهَذِهِ أَسْمَاءُ عَشَائِرِ اللّاوِيِّينَ حَسَبَ تَرْتِيبِ عَائِلاتِهِمْ:</span>
<span className="v" id="v20" data-v="20" data-xref="أخبار الأيام الأول 6:43"> أَنْجَبَ جَرْشُومُ لِبْنِي، وَلِبْنِي يَحَثَ، وَيَحَثُ زِمَّةَ،</span>
<span className="v" id="v21" data-v="21"> وَزِمَّةُ يُوآخَ، وَيُوآخُ عِدُّو، وَعِدُّو زَارَحَ، وَزَارَحُ يَأَثْرَايَ.</span>
<span className="v" id="v22" data-v="22" data-xref="الخروج 6:24; أخبار الأيام الأول 6:18; أخبار الأيام الأول 6:2"> وَأَنْجَبَ قَهَاتُ عَمِّينَادَابَ، وَعَمِّينَادَابُ قُورَحَ، وَقُورَحُ أَسِّيرَ،</span>
<span className="v" id="v23" data-v="23"> وَأَسِّيرُ أَلْقَانَةَ، وَأَلْقَانَةُ أَبِيأَسَافَ، وَأَبِيأَسَافُ أَسِّيرَ،</span>
<span className="v" id="v24" data-v="24"> وَأَسِّيرُ تَحَثَ، وَتَحَثُ أُورِيئِيلَ، وَأُورِيئِيلُ عُزِّيَّا، وَعُزِّيَّا شَاوُلَ.</span>
<span className="v" id="v25" data-v="25" data-xref="الخروج 6:24; أخبار الأيام الأول 6:35–36"> وَشاوُلُ أَلْقَانَةَ؛ وَوَلَدَ أَلْقَانَةُ ابْنَيْنِ هُمَا عَمَاسَايُ وَأَخِيمُوتُ.</span>
<span className="v" id="v26" data-v="26" data-xref="صموئيل الأول 1:1; أخبار الأيام الأول 6:35"> وَأَنْجَبَ أَخِيمُوتُ أَلْقَانَةَ، وَوَلَدَ أَلْقَانَةُ صُوفَايَ، وَصُوفَايُ نَحَثَ.</span>
<span className="v" id="v27" data-v="27" data-xref="صموئيل الأول 1:1; صموئيل الأول 1:19–20"> وَنَحَثُ أَلِيآبَ، وَأَلِيآبُ يَرُوحَامَ، وَيَرُوحَامُ أَلْقَانَةَ (الَّذِي أَنْجَبَ صَمُوئِيلَ).</span>
<span className="v" id="v28" data-v="28" data-xref="أخبار الأيام الأول 6:33; صموئيل الأول 8:2"> وَكَانَ لِصَمُوئِيلَ ابْنَانِ أَكْبَرُهُمَا وَشْنِي وَالثَّانِي أَبِيَّا.</span>
<span className="v" id="v29" data-v="29" data-xref="أخبار الأيام الأول 6:19; العدد 3:33"> وَأَنْجَبَ مَرَارِي مَحْلِي، وَمَحْلِي لِبْنِي، وَلِبْنِي شِمْعِي، وَشِمْعِي عُزَّةَ.</span>
<span className="v" id="v30" data-v="30"> وَعُزَّةُ شِمْعِي، وَشِمْعِي حَجِيَّا، وَحَجِيَّا عَسَايَا.</span>

## موسيقيو الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="صموئيل الثاني 6:17; أخبار الأيام الأول 15:25–1; أخبار الأيام الأول 15:16–22"> وَبَعْدَ أَنِ اسْتَقَرَّ التَّابُوتُ فِي بَيْتِ الرَّبِّ عَيَّنَ دَاوُدُ قَادَةً لِجَوْقَةِ التَّسْبِيحِ.</span>
<span className="v" id="v32" data-v="32" data-xref="المزامير 135:1–3; ملوك الأول 8:6–13; عزرا 6:18"> فَوَاظَبُوا عَلَى الْخِدْمَةِ أَمَامَ مَسْكَنِ خَيْمَةِ الاجْتِمَاعِ إِلَى أَنْ بَنَى سُلَيْمَانُ هَيْكَلَ الرَّبِّ فِي أُورُشَلِيمَ، فَاسْتَمَرُّوا قَائِمِينَ بِالْخِدْمَةِ حَسَبَ تَرْتِيبِهِمْ.</span>
<span className="v" id="v33" data-v="33" data-xref="أخبار الأيام الأول 6:28; أخبار الأيام الأول 15:17; أخبار الأيام الأول 25:1–5"> وَهَذَا سِجِلٌّ بِنَسَبِ قَادَةِ الْمُغَنِّينَ وَأَوْلادِهِمْ مِنْ أَبْنَاءِ الْقَهَاتِيِّينَ: هَيْمَانُ الْمُغَنِّي ابْنُ يُوئِيلَ بْنِ صَمُوئِيلَ،</span>
<span className="v" id="v34" data-v="34"> بْنِ أَلْقَانَةَ بْنِ يَرُوحَامَ بْنِ إِيلِيئِيلَ بْنِ تُوحٍ،</span>
<span className="v" id="v35" data-v="35"> بْنِ صُوفَ بْنِ أَلْقَانَةَ بْنِ مَحَثَ بْنِ عَمَاسَايَ،</span>
<span className="v" id="v36" data-v="36"> بْنِ أَلْقَانَةَ بْنِ يُوئِيلَ بْنِ عَزَرْيَا بْنِ صَفَنْيَا،</span>
<span className="v" id="v37" data-v="37" data-xref="المزامير 45:1; العدد 26:10–11; الخروج 6:21–24"> بْنِ تَحَثَ بْنِ أَسِّيرَ بْنِ أَبِيَاسَافَ بْنِ قُورَحَ،</span>
<span className="v" id="v38" data-v="38" data-xref="العدد 3:19; أخبار الأيام الأول 16:1"> بْنِ يِصْهَارَ بْنِ قَهَاتَ بْنِ لاوِي بْنِ إِسْرَائِيلَ.</span>
<span className="v" id="v39" data-v="39" data-xref="نحميا 7:44; عزرا 2:41; المزامير 75:1"> وَكَانَ آسَافُ مُسَاعِداً لِهَيْمَانَ، وَهُوَ آسَافُ بْنُ بَرَخْيَا بْنِ شِمْعِي،</span>
<span className="v" id="v40" data-v="40"> بْنِ مِيخَائِيلَ بْنِ بَعَسِيَا بْنِ مَلْكِيَا،</span>
<span className="v" id="v41" data-v="41"> بْنِ أَثْنَايَ بْنِ زَارَحَ بْنِ عَدَايَا،</span>
<span className="v" id="v42" data-v="42"> بْنِ أَيْثَانَ بْنِ زِمَّةَ بْنِ شِمْعِي،</span>
<span className="v" id="v43" data-v="43" data-xref="الخروج 6:16; أخبار الأيام الأول 6:16–17; التكوين 46:11"> بْنِ يَحَثَ بْنِ جَرْشُومَ بْنِ لاوِي.</span>
<span className="v" id="v44" data-v="44" data-xref="المزامير 89:1; أخبار الأيام الأول 15:17; أخبار الأيام الأول 25:1"> وَكَانَ أَيْثَانُ مُسَاعِداً ثَانِياً لِهَيْمَانَ، وَهُوَ مِنْ ذُرِّيَّةِ مَرَارِي، وَأَبُوهُ قِيشِي بْنُ عَبْدِي بْنِ مَلُّوخَ،</span>
<span className="v" id="v45" data-v="45"> بْنِ حَشَبْيَا بْنِ أَمَصْيَا بْنِ حِلْقِيَّا،</span>
<span className="v" id="v46" data-v="46"> بْنِ أَمْصِي بْنِ بَانِي، بْنِ شَامِرَ،</span>
<span className="v" id="v47" data-v="47" data-xref="يشوع 21:7; الخروج 6:19; أخبار الأيام الأول 23:21"> بْنِ مَحْلِي بْنِ مُوشِي بْنِ مَرَارِي بْنِ لاوِي.</span>
<span className="v" id="v48" data-v="48" data-xref="أخبار الأيام الأول 23:2; العدد 3:1–4; العدد 18:1–32"> وَقَدْ تَوَلَّى بَقِيَّةُ إِخْوَتِهِمِ اللّاوِيِّينَ، خِدْمَةَ مَسْكَنِ بَيْتِ الرَّبِّ.</span>
<span className="v" id="v49" data-v="49" data-xref="اللاويين 4:20; الخروج 27:1–8; الخروج 30:1–7"> أَمَّا هرُونُ وَذُرِّيَّتُهُ فَقَدْ تَوَلَّوْا خِدْمَةَ تَقْدِيمِ الْمُحْرَقَاتِ عَلَى مَذْبَحِ الْمُحْرَقَةِ وَالْبَخُورِ عَلَى مَذْبَحِ الْبَخُورِ، فَضْلاً عَنْ تَأْدِيَةِ كُلِّ خَدَمَاتِ قُدْسِ الأَقْدَاسِ وَلِلتَّكْفِيرِ عَنْ إِسْرَائِيلَ بِمُوْجِبِ مَا أَمَرَ بِهِ مُوسَى عَبْدُ اللهِ.</span>
<span className="v" id="v50" data-v="50" data-xref="اللاويين 10:16; أخبار الأيام الأول 24:1; العدد 20:26–28"> وَهَذِهِ أَسْمَاءُ أَبْنَاءِ هرُونَ وَنَسْلِهِمْ: أَلِعَازَارُ الَّذِي أَنْجَبَ فِينْحَاسَ، وَفِينْحَاسُ أَبِيشُوعَ،</span>
<span className="v" id="v51" data-v="51"> وأَبِيشُوعُ بُقِّيَ، وَبُقِّي عُزِّي، وعُزِّي زَرَحْيَا،</span>
<span className="v" id="v52" data-v="52"> وَزَرَحْيَا مَرَايُوثَ، وَمَرَايُوثُ أَمَرْيَا، وَأَمَرْيَا أَخِيطُوبَ،</span>
<span className="v" id="v53" data-v="53" data-xref="صموئيل الثاني 8:17; حزقيال 44:15; صموئيل الثاني 15:35–36"> وَأَخِيطُوبُ صَادُوقَ، وَصَادُوقُ أَخِيمَعَصَ.</span>
<span className="v" id="v54" data-v="54" data-xref="التكوين 25:16; يشوع 21:10; يشوع 21:3–8"> وَهَذِهِ هِيَ مَوَاضِعُ مَسَاكِنِ الْقَهَاتِيِّينَ مِنْ ذُرِّيَّةِ هرُونَ وَضِيَاعُهُمْ وَحُدُودُهُمُ الَّتِي وَقَعَتِ الْقُرْعَةُ عَلَيْهَا.</span>
<span className="v" id="v55" data-v="55" data-xref="يشوع 14:13; يشوع 21:11–13; يشوع 15:13"> فَأَعْطَوْهُمْ حَبْرُونَ فِي أَرْضِ يَهُوذَا مَعَ مَرَاعِيهَا الْمُحِيطَةِ بِها.</span>
<span className="v" id="v56" data-v="56" data-xref="يشوع 15:13; يشوع 14:13"> وَأَمَّا حُقُولُ الْمَدِينَةِ وَضِيَاعُهَا فَقدْ أُعْطِيَتْ لِكَالَبَ بْنِ يَفُنَّةَ.</span>
<span className="v" id="v57" data-v="57" data-xref="يشوع 21:13–14; يشوع 15:48; يشوع 10:29"> كَمَا أُعْطِيَتْ لأَبْنَاءِ هرُونَ مُدُنُ الْمَلْجَإِ: حَبْرُونُ وَلِبْنَةُ وَمَرَاعِيهَا وَيَتِّيرُ وَأَشْتَمُوعُ وَمَرَاعِيهَا،</span>
<span className="v" id="v58" data-v="58" data-xref="يشوع 10:3; يشوع 15:49; يشوع 12:13"> وَحِيلَيْنُ وَمَرَاعِيهَا، وَدَبِيرُ وَمَرَاعِيهَا،</span>
<span className="v" id="v59" data-v="59" data-xref="يشوع 21:16; يشوع 15:42; صموئيل الأول 6:12–19"> وَعَاشَانُ وَمَرَاعِيهَا، وَبَيْتُ شَمْسٍ وَمَرَاعِيهَا.</span>
<span className="v" id="v60" data-v="60" data-xref="إرميا 1:1; ملوك الأول 2:26; إرميا 11:23"> وَأَعْطَوْهُمْ أَيْضاً مِنْ أَرْضِ سِبْطِ بِنْيَامِينَ: جَبَعَ وَمَرَاعِيهَا، وَعَلْمَثَ وَمَرَاعِيهَا، وَعَنَاثُوثَ وَمَرَاعِيهَا، فَكَانَتْ جُمْلَةُ مُدُنِهِمْ ثَلاثَ عَشْرَةَ مَدِينَةً وَفْقاً لِعَشَائِرِهِمْ.</span>
<span className="v" id="v61" data-v="61" data-xref="يشوع 21:4–5; أخبار الأيام الأول 6:18; يشوع 21:20–26"> وَأُعْطِيَتْ بِالْقُرْعَةِ عَشْرُ مُدُنٍ لِبَقِيَّةِ عَشِيرَةِ قَهَاتَ مِنْ مُدُنِ نِصْفِ سِبْطِ مَنَسَّى.</span>
<span className="v" id="v62" data-v="62" data-xref="يشوع 21:27–33; أخبار الأيام الأول 6:71–76; الخروج 2:22"> وَوُهِبَتْ لِعَائِلاتِ عَشَائِرِ جَرْشُومَ بِالْقُرْعَةِ ثَلاثَ عَشْرَةَ مَدِينَةً فِي أَرَاضِي أَسْبَاطِ يَسَّاكَرَ وَأَشِيرَ وَنَفْتَالِي وَمَنَسَّى فِي بَاشَانَ.</span>
<span className="v" id="v63" data-v="63" data-xref="يشوع 21:7; يشوع 21:34–40; العدد 3:20"> كَمَا وُهِبَتْ لِعَائِلاتِ عَشَائِرِ مَرَارِي بِالْقُرْعَةِ اثْنَتَا عَشْرَةَ مَدِينَةً مِنْ مُدُنِ أَسْبَاطِ رَأُوبَيْنَ وَجَادٍ وَزَبُولُونَ.</span>
<span className="v" id="v64" data-v="64" data-xref="يشوع 21:41–42; يشوع 21:3; العدد 35:1–8"> وَهَكَذَا أَعْطَى بَنُو إِسْرَائِيلَ اللّاوِيِّينَ مُدُناً يُقِيمُونَ فيهَا مَعَ مَرَاعِيهَا.</span>
<span className="v" id="v65" data-v="65" data-xref="أخبار الأيام الأول 6:57–60"> وَقَدْ أُعْطِيَتْ هَذِهِ الْمُدُنُ الْمَذْكُورَةُ بِأَسْمَائِهَا بِالْقُرْعَةِ مِنْ مُدُنِ أَرَاضِي أَسْبَاطِ يَهُوذَا، وَشِمْعُونَ، وَبَنْيَامِينَ.</span>
<span className="v" id="v66" data-v="66" data-xref="يشوع 21:20–26"> كَمَا كَانَتْ بَعْضُ مُدُنِ الْقَهَاتِيِّينَ ضِمْنَ حُدُودِ أَرَاضِي سِبْطِ أَفْرَايِمَ.</span>
<span className="v" id="v67" data-v="67" data-xref="يشوع 21:21; التكوين 33:18; التكوين 35:4"> وَخَصَّصُوا لَهُمْ أَيْضاً مُدُنَ مَلْجَإٍ: شَكِيمَ وَمَرَاعِيهَا فِي جَبَلِ أَفْرَايِمَ، وَجَازَرَ وَمَرَاعِيهَا،</span>
<span className="v" id="v68" data-v="68" data-xref="يشوع 21:22; يشوع 16:5; صموئيل الأول 13:18"> وَيَقْمَعَامَ وَمَرَاعِيهَا، وَبَيْتَ حُورُونَ وَمَرَاعِيهَا،</span>
<span className="v" id="v69" data-v="69" data-xref="يشوع 10:12; يشوع 21:24–25"> وَأَيَّلُونَ وَمَرَاعِيهَا، وَجَتَّ رِمُّونَ وَمَرَاعِيهَا.</span>
<span className="v" id="v70" data-v="70" data-xref="يشوع 21:25; يشوع 17:11"> وَأَعْطَوْا لِعَشِيرَةِ أَبْنَاءِ قَهَاتَ الْبَاقِينَ مِنْ مُدُنِ نِصْفِ سِبْطِ مَنَسَّى مَدِينَتَيْ عَانِيرَ وَمَرَاعِيهَا وَبِلْعَامَ وَمَرَاعِيهَا.</span>
<span className="v" id="v71" data-v="71" data-xref="يشوع 21:27; يشوع 9:10; يشوع 20:8"> وَأَعْطَوْا لِعَشِيرَةِ الْجَرْشُومِيِّينَ مِنْ مُدُنِ نِصْفِ سِبْطِ مَنَسَّى الْمُسْتَوْطِنِينَ فيِ بَاشَانَ، جُولانَ وَمَرَاعِيهَا وَعَشْتَارُوتَ وَمَرَاعِيهَا.</span>
<span className="v" id="v72" data-v="72" data-xref="يشوع 21:28–29; يشوع 21:32; يشوع 19:37"> وَمِنْ مُدُنِ أَرَاضِي سِبْطِ يَسَّاكَرَ، قَادَشَ وَمَرَاعِيهَا وَدَبَرَةَ وَمَرَاعِيهَا،</span>
<span className="v" id="v73" data-v="73"> وَرَامُوتَ وَمَرَاعِيهَا، وَعَانِيمَ وَمَرَاعِيهَا.</span>
<span className="v" id="v74" data-v="74" data-xref="يشوع 19:26; يشوع 21:30"> وَمِنْ مُدُنِ أَرَاضِيِ سِبْطِ أَشِيرَ، مَشْآلَ وَمَرَاعِيهَا وَعَبْدُونَ وَمَرَاعِيهَا،</span>
<span className="v" id="v75" data-v="75" data-xref="يشوع 21:31"> وَحُقُوقَ وَمَرَاعِيهَا، وَرَحُوبَ وَمَرَاعِيهَا.</span>
<span className="v" id="v76" data-v="76" data-xref="يشوع 21:32; يشوع 19:37; القضاة 4:6"> وَمِنْ مُدُنِ أَرَاِضي سِبْطِ نَفْتَالِي، قَادَشَ فِي الْجَلِيلِ وَمَرَاعِيهَا، وَحَمُّونَ وَمَرَاعِيهَا وَقَرْيَتَايِمَ وَمَرَاعِيهَا.</span>
<span className="v" id="v77" data-v="77" data-xref="يشوع 21:34–35"> وَأَعْطَوْا بَقِيَّةَ أَبْنَاءِ مَرَارِي مِنْ مُدُنِ أَرَاضِي سِبْطِ زَبُولُونَ رِمُّونُو وَمَرَاعِيهَا، وَتَابُورَ وَمَرَاعِيهَا.</span>
<span className="v" id="v78" data-v="78" data-xref="يشوع 20:8; يشوع 21:36–37; التثنية 4:41–43"> كَمَا وَهَبُوهُمْ مِنْ مُدُنِ أَرَاضِي رَأُوبَيْنَ فِي شَرْقِيِّ نَهْرِ الأُرْدُنِّ مُقَابِلَ أَرِيحَا بَاصِرَ وَمَرَاعِيهَا فِي الصَّحْرَاءِ، وَيَهْصَةَ وَمَرَاعِيهَا،</span>
<span className="v" id="v79" data-v="79"> وَقَدِيمُوتَ وَمَرَاعِيهَا، وَمَيْفَعَةَ وَمَرَاعِيهَا.</span>
<span className="v" id="v80" data-v="80" data-xref="التكوين 32:2; ملوك الثاني 9:1; يشوع 21:38–39"> وَمِنْ مُدُنِ أَرَاضِي سِبْطِ جَادٍ فِي جِلْعَادَ رَامُوتَ وَمَرَاعِيهَا، وَمَحَنَايِمَ وَمَرَاعِيهَا،</span>
<span className="v" id="v81" data-v="81" data-xref="يشوع 21:39; العدد 21:32; العدد 32:3"> وَحَشْبُونَ وَمَرَاعِيهَا، وَيَعْزِيرَ وَمَرَاعِيهَا.</span>
