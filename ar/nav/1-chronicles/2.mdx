## من يهوذا إلى أبناء حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="التكوين 38:2–10; التكوين 46:12; العدد 26:19"> أَمَّا أَبْنَاءُ يَهُوذَا فَهُمْ: عِيرُ وَأُونَانُ وَشَيْلَةُ. وَقَدْ أَنْجَبَتْ بِنْتُ شُوعَ الْكَنْعَانِيَّةُ هَؤُلاءِ الثَّلاثَةَ. وَأَمَاتَ الرَّبُّ عِيرَ، بِكْرَ يَهُوذَا، لأَنَّهُ كَانَ شِرِّيراً فِي عَيْنَيْهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="الإنجيل كما دونه متى 1:3; التكوين 38:13–30; نحميا 11:4"> وَأَنْجَبَ يَهُوذَا مِنْ كَنَّتِهِ ثَامَارَ: فَارَصَ وَزَارَحَ، فَكَانَتْ جُمْلَةُ أَوْلادِهِ خَمْسَةً.</span>
<span className="v" id="v5" data-v="5" data-xref="التكوين 46:12; راعوث 4:18; العدد 26:21"> وَأَنْجَبَ فَارَصُ: حَصْرُونَ وَحَامُولَ.</span>
<span className="v" id="v6" data-v="6" data-xref="ملوك الأول 4:31; يشوع 7:17–18; يشوع 7:1"> كَمَا أَنْجَبَ زَارَحُ: زِمْرِي وَأَيْثَانَ وَهَيْمَانَ وَكَلْكُولَ وَدَارَعَ. فَكَانُوا خَمْسَةً فِي جُمْلَتِهِمْ.</span>
<span className="v" id="v7" data-v="7" data-xref="يشوع 6:18; التثنية 7:26; يشوع 7:25"> وَعَخَانُ بْنُ كَرْمِي هُوَ الَّذِي سَبَّبَ كَارِثَةً لإِسْرَائِيلَ حِينَ خَانَ فَسَرَقَ مِمَّا هُوَ مُحَرَّمٌ.</span>
<span className="v" id="v8" data-v="8"> وَأَنْجَبَ أَيْثَانُ عَزَرْيَا.</span>
<span className="v" id="v9" data-v="9" data-xref="أخبار الأيام الأول 2:42; الإنجيل كما دونه متى 1:3; راعوث 4:19"> أَمَّا أَبْنَاءُ حَصْرُونَ فَهُمْ: يَرْحَمْئِيلُ، وَرَامُ وَكَلُوبَايُ.</span>
</div>
</div>

## من رام بن حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه متى 1:4; العدد 1:7; العدد 2:3"> وَأَنْجَبَ رَامُ عَمِّينَادَابَ، وَعَمِّينَادَابُ نَحْشُونَ، رَئِيسَ بَنِي يَهُوذَا.</span>
<span className="v" id="v11" data-v="11" data-xref="راعوث 4:21; الإنجيل كما دونه متى 1:4–5; الإنجيل كما دونه لوقا 3:32"> وَأَنْجَبَ نَحْشُونُ سَلْمُوَ الَّذِي أَنْجَبَ بُوعَزَ.</span>
<span className="v" id="v12" data-v="12" data-xref="الرسالة إلى مؤمني روما 15:12; أخبار الأيام الأول 10:14; إشعياء 11:10"> وَأَنْجَبَ بُوعَزُ عُوبِيدَ وَالِدَ يَسَّى.</span>
<span className="v" id="v13" data-v="13" data-xref="صموئيل الأول 17:13; صموئيل الأول 17:28; أخبار الأيام الأول 20:7"> وَأَنْجَبَ يَسَّى بِكْرَهُ أَلِيآبَ، ثُمَّ أَبِينَادَابَ، فَشِمْعَى،</span>
<span className="v" id="v14" data-v="14"> ثُمَّ نَثْنِئِيلَ فَرَدَّايَ،</span>
<span className="v" id="v15" data-v="15" data-xref="صموئيل الأول 16:10–11; صموئيل الأول 17:12–14"> فَأُوصَمَ وَأَخِيراً دَاوُدَ.</span>
<span className="v" id="v16" data-v="16" data-xref="صموئيل الأول 26:6; صموئيل الثاني 3:39; صموئيل الثاني 2:18–23"> كَمَا أَنْجَبَ يَسَّى ابْنَتَيْنِ هُمَا صُرُويَّةُ وَأَبِيجَايِلُ. وَأَبْنَاءُ صُرُويَّةَ ثَلاثَةٌ هُمْ: أَبِيشَايُ وَيُوآبُ وَعَسَائِيلُ.</span>
<span className="v" id="v17" data-v="17" data-xref="صموئيل الثاني 17:25; ملوك الأول 2:32; صموئيل الثاني 20:4–12"> أَمَّا أَبِيجَايِلُ فَقَدْ أَنْجَبَتْ: عَمَاسَا مِنْ يَثْرَ الإِسْمَاعِيلِيِّ.</span>
</div>
</div>

## كالب بن حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18"> وَكَانَ كَالَبُ بْنُ حَصْرُونَ مُتَزَوِّجاً مِنْ عَزُوبَةَ وَيَرِيعُوثَ. فَأَنْجَبَتْ لَهُ عَزُوبَةُ يَاشَرَ وَشُوبَابَ وَأَرْدُونَ.</span>
<span className="v" id="v19" data-v="19" data-xref="أخبار الأيام الأول 2:50; أخبار الأيام الأول 2:24; أخبار الأيام الأول 4:4"> وَعِنْدَمَا مَاتَتْ عَزُوبَةُ تَزَوَّجَ كَالَبُ مِنْ أَفْرَاتَ فَأَنْجَبَتْ لَهُ حُورَ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخروج 31:2; الخروج 36:1–2; الخروج 37:1"> وَأَنْجَبَ حُورُ أُورِيَ وَأَنْجَبَ أُورِي بَصَلْئِيلَ.</span>
<span className="v" id="v21" data-v="21" data-xref="العدد 27:1; التثنية 3:15; العدد 26:29"> وَتَزَوَّجَ حَصْرُونُ وَهُوَ فِي السِّتِّينَ مِنْ عُمْرِهِ ابْنَةَ مَاكِيرَ أَبِي جِلْعَادَ وَأَنْجَبَ مِنْهَا سَجُوبَ.</span>
<span className="v" id="v22" data-v="22" data-xref="يشوع 13:30; التثنية 3:14; العدد 32:41"> وَأَنْجَبَ سَجُوبُ يَائِيرَ الَّذِي امْتَلَكَ ثَلاثاً وَعِشْرِينَ مَدِينَةً فِي أَرْضِ جِلْعَادَ.</span>
<span className="v" id="v23" data-v="23" data-xref="يشوع 13:30; التثنية 3:14; العدد 32:41–42"> غَيْرَ أَنَّ مَمْلَكَةَ جَشُورَ وَمَمْلَكَةَ أَرَامَ اسْتَوْلَتَا عَلَى حَوُّوثَ يَائِيرَ مَعَ قَنَاةَ وَقُرَاهَا، فَكَانَتْ فِي جُمْلَتِهَا سِتِّينَ مَدِينَةً. وَكَانَ كُلُّ أَهْلِهَا مُنْحَدِرِينَ مِنْ ذُرِّيَّةِ مَاكِيرَ أَبِي جِلْعَادَ.</span>
<span className="v" id="v24" data-v="24" data-xref="أخبار الأيام الأول 4:5; أخبار الأيام الأول 2:9; صموئيل الأول 30:14"> وَبَعْدَ وَفَاةِ حَصْرُونَ فِي كَالَبِ أَفْرَاتَةَ، تَزَوَّجَ ابْنُهُ كَالَبُ أَبِيَّاهَ أَرْمَلَةَ أَبِيهِ، فَأَنْجَبَتْ لَهُ أَشْحُورَ مُؤَسِّسَ مَدِينَةِ تَقُوعَ.</span>
</div>
</div>

## يرحمئيل بن حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25"> أَمَّا أَبْنَاءُ يَرْحَمْئِيلَ بِكْرِ حَصْرُونَ فَهُمْ: الْبِكْرُ رَامٌ، ثُمَّ بُونَةُ وَأَوْرَنُ وَأُوصَمُ وَأَخِيَّا.</span>
<span className="v" id="v26" data-v="26"> وَكَانَ لِيَرْحَمْئِيلَ زَوْجَةٌ أُخْرَى تُدْعَى عَطَارَةَ هِيَ أُمُّ أُونَامَ.</span>
<span className="v" id="v27" data-v="27"> وَأَبْنَاءُ رَامٍ بِكْرِ يَرْحَمْئِيلَ هُمْ: مَعَصُ وَيَمِينُ وَعَاقَرُ.</span>
<span className="v" id="v28" data-v="28"> وَابْنَا أُونَامَ: شَمَّايُ وَيَادَاعُ. وَابْنَا شَمَّايَ: نَادَابُ وَأَبِيشُورُ.</span>
<span className="v" id="v29" data-v="29"> وَاسْمُ زَوْجَةِ أَبِيشُورَ أَبِيجَايِلُ، وَقَدْ أَنْجَبَتْ لَهُ أَحْبَانَ وَمُولِيدَ.</span>
<span className="v" id="v30" data-v="30"> أَمَّا ابْنَا نَادَابَ فَهُمَا: سَلَدُ وَأَفَّايِمُ. وَمَاتَ سَلَدُ مِنْ غَيْرِ عَقِبٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="أخبار الأيام الأول 2:34–35"> وَأَنْجَبَ أَفَّايِمُ يَشْعِي. وَيَشْعِي وَلَدَ شِيشَانَ الَّذِي أَنْجَبَ أَحْلايَ.</span>
<span className="v" id="v32" data-v="32"> وَأَنْجَبَ يَادَاعُ أَخُو شَمَّايَ: يَثَرَ وَيُونَاثَانَ. وَمَاتَ يَثَرُ مِنْ غَيْرِ عَقِبٍ.</span>
<span className="v" id="v33" data-v="33"> وَأَنْجَبَ يُونَاثَانُ ابْنَيْنِ هُمَا: فَالَتُ وَزَازَا. وَجَمِيعُ هَؤُلاءِ مِنْ ذُرِّيِّةِ يِرْحَمْئِيلَ.</span>
<span className="v" id="v34" data-v="34"> وَلَمْ يُعْقِبْ شِيشَانُ أَبْنَاءً بَلْ بَنَاتٍ، وَكَانَ لِشِيشَانَ خَادِمٌ مِصْرِيٌّ اسْمُهُ يَرْحَعُ.</span>
<span className="v" id="v35" data-v="35"> فَزَوَّجَ شِيشَانُ ابْنَتَهُ لِيَرْحَعَ، فَأَنْجَبَتْ لَهُ عَتَّايَ.</span>
<span className="v" id="v36" data-v="36" data-xref="أخبار الأيام الأول 11:41"> وَأَنْجَبَ عَتَّايُ نَاثَانَ، وَنَاثَانُ وَلَدَ زَابَادَ.</span>
<span className="v" id="v37" data-v="37"> وَأَنْجَبَ زَابَادُ أَفْلالَ، وَأَفْلالُ وَلَدَ عُوبِيدَ.</span>
<span className="v" id="v38" data-v="38"> وَأَنْجَبَ عُوبِيدُ يَاهُوَ الَّذِي وَلَدَ عَزَرْيَا.</span>
<span className="v" id="v39" data-v="39"> وَأَنْجَبَ عَزَرْيَا حَالَصَ، وَحَالَصُ إِلْعَاسَةَ.</span>
<span className="v" id="v40" data-v="40"> وَأَنْجَبَ إِلْعَاسَةُ سِسَمَايَ وَسِسَمَايُ شَلُّومَ.</span>
<span className="v" id="v41" data-v="41"> وَأَنْجَبَ شَلُّومُ يَقَمْيَةَ، وَيَقَمْيَةُ أَلِيشَمَعَ.</span>
</div>
</div>

## عشائر كالب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="أخبار الأيام الأول 2:23–24; الرسالة إلى العبرانيين 12:23; الرسالة إلى مؤمني روما 8:29"> أَمَّا بِكْرُ كَالَبَ أَخِي يَرْحَمْئِيلَ فَهُو مِيشَاعُ أَبُو زِيفَ الَّذِي أَنْجَبَ مَرِيشَةَ وَالِدَ حَبْرُونَ.</span>
<span className="v" id="v43" data-v="43"> أَمَّا أَبْنَاءُ حَبْرُونَ فَهُمْ: قُورَحُ وَتَفُّوحُ وَرَاقَمُ وَشَامَعُ.</span>
<span className="v" id="v44" data-v="44"> وَأَنْجَبَ شَامَعُ رَاقَمَ أَبَا يَرُقْعَامَ. وَأَنْجَبَ رَاقَمُ شَمَّايَ.</span>
<span className="v" id="v45" data-v="45" data-xref="يشوع 15:58"> وَأَنْجَبَ شَمَّايُ مَعُونَ الَّذِي بَنَى بَيْتَ صُورَ.</span>
<span className="v" id="v46" data-v="46" data-xref="أخبار الأيام الأول 2:18–19; أخبار الأيام الأول 2:48"> وَأَنْجَبَتْ عَيفَةُ مَحْظِيَّةُ كَالَبَ حَارَانَ وَمُوصَا وَجَازِيزَ. وَأَنْجَبَ حَارَانُ ابْناً سَمَّاهُ جَازِيزَ.</span>
<span className="v" id="v47" data-v="47"> وَأَبْنَاءُ يَهْدَايَ: رَجَمُ وَيُوثَامُ وَجِيشَانُ وَفَلَطُ وَعِيفَةُ وَشَاعَفُ.</span>
<span className="v" id="v48" data-v="48" data-xref="التكوين 25:5–6; أخبار الأيام الأول 2:46"> وَأَنْجَبَتْ مَعْكَةُ مَحْظِيَّةٌ أُخْرَى لِكَالَبَ، شَبَرَ وَتَرْحَنَةَ.</span>
<span className="v" id="v49" data-v="49" data-xref="يشوع 15:31; أخبار الأيام الأول 2:42; إشعياء 10:31"> ثُمَّ أَنْجَبَتْ شَاعَفَ بَانِي مَدِينَةِ مَدْمَنَةَ، وَشَوَا بَانِي مَدِينَتَيْ مَكْبِينَا وَجَبَعَا. وَكَانَ لِكَالَبَ بِنْتٌ اسْمُهَا عَكْسَةُ.</span>
<span className="v" id="v50" data-v="50" data-xref="صموئيل الأول 7:1; يشوع 15:60; أخبار الأيام الأول 2:53"> وَهَؤُلاءِ بَعْضُ ذُرِّيَّةِ كَالَبَ: حُورُ بِكْرُهُ مِنْ زَوْجَتِهِ أَفْرَاتَ وَقَدْ أَنْجَبَتْ شُوبَالَ مُؤَسِّسَ قَرْيَةِ يَعَارِيمَ،</span>
<span className="v" id="v51" data-v="51" data-xref="أخبار الأيام الأول 4:4; الإنجيل كما دونه يوحنا 7:42; الإنجيل كما دونه متى 2:6"> وَسَلْمَا مُؤَسِّسَ بَيْتِ لَحْمٍ، وَحَارِيفَ مُؤَسِّسَ بَيْتِ جَادِيرَ.</span>
<span className="v" id="v52" data-v="52" data-xref="أخبار الأيام الأول 4:2"> أَمَّا ذُرِّيَّةُ شُوبَالَ مُؤَسِّسِ قَرْيَةِ يَعَارِيمَ فَهُمْ قَبِيلَةُ هَرُوَاهُ وَنِصْفُ قَبِيلَةِ هَمَّنُوحُوتَ.</span>
<span className="v" id="v53" data-v="53" data-xref="صموئيل الثاني 23:38; القضاة 13:25; القضاة 13:2"> وَعَشَائِرُ قَرْيَةِ يَعَارِيمَ هُمْ: الْيَثْرِيُّونَ وَالْفُوتِيُّونَ وَالشَّمَاتِيُّونَ وَالْمَشْرَاعِيُّونَ. وَتَفَرَّعَ مِنْ هَؤُلاءِ الصَّرْعِيُّونَ وَالأَشْتَأُولِيُّونَ.</span>
