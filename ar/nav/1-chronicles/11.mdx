## داود يملك على إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صموئيل الثاني 5:1–16; أخبار الأيام الأول 12:23–40; القضاة 9:2"> وَتَجَمَّعَ كُلُّ رُؤَسَاءِ الإِسْرَائِيلِيِّينَ حَوْلَ دَاوُدَ فِي حَبْرُونَ وَقَالُوا: «نَحْنُ مِنْ لَحْمِكَ وَدَمِكَ،</span>
<span className="v" id="v2" data-v="2" data-xref="صموئيل الثاني 5:2; الإنجيل كما دونه يوحنا 10:4; المزامير 78:71"> وَقَدْ كُنْتَ قَائِدَنَا، تَخُوضُ الْمَعَارِكَ فِي طَلِيعَتِنَا مُنْذُ الأَيَّامِ السَّابِقَةِ حِينَ كَانَ شَاوُلُ مَلِكاً عَلَيْنَا، وَقَدْ قَالَ لَكَ الرَّبُّ إِلَهُكَ: أَنْتَ تَرْعَى شَعْبِي وَتَتَوَلَّى حُكْمَهُ».</span>
<span className="v" id="v3" data-v="3" data-xref="صموئيل الأول 16:12–13; صموئيل الأول 16:1; صموئيل الثاني 5:3"> وَعِنْدَمَا اجْتَمَعَ جَمِيعُ شُيُوخِ إِسْرِائِيلَ إِلَى الْمَلِكِ فِي حَبْرُونَ أَبْرَمَ مَعَهُمْ عَهْداً أَمَامَ الرَّبِّ، فَمَسَحُوهُ مَلِكاً عَلَيْهِمْ تَتْمِيماً لِكَلامِ الرَّبِّ الَّذِي نَطَقَ بِهِ عَلَى لِسَانِ صَمُوئِيلَ.</span>
</div>
</div>

## داود يستولي على أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="القضاة 1:21; التكوين 15:21; يشوع 15:8"> وَتَوَجَّهَ دَاوُدُ عَلَى رَأْسِ الإِسْرَائِيلِيِّينَ إِلَى أُورُشَلِيمَ، أَيْ يَبُوسَ الآهِلَةِ بِسُكَّانِهَا الْيَبُوسِيِّينَ.</span>
<span className="v" id="v5" data-v="5" data-xref="أخبار الأيام الثاني 5:2; ملوك الأول 8:1; المزامير 87:2"> فَقَالَ الْيَبُوسِيُّونَ لِدَاوُدَ: «لا يُمْكِنُكَ أَنْ تَدْخُلَ إِلَى هُنَا». فَاسْتَوْلَى دَاوُدُ عَلَى قَلْعَةِ صِهْيَوْنَ الَّتِي دُعِيَتْ فِي مَا بَعْدُ مَدِينَةَ دَاوُدَ.</span>
<span className="v" id="v6" data-v="6" data-xref="صموئيل الثاني 8:16; صموئيل الثاني 20:23; صموئيل الثاني 3:27"> ثُمَّ قَالَ دَاوُدُ لِرِجَالِهِ: «إِنَّ مَنْ يَقْتَحِمُ الْيَبُوسِيِّينَ يُصْبِحُ قَائِداً لِلْجَيْشِ». فَهَاجَمَهُمْ يُوآبُ ابْنُ صُرُوِيَّةَ أَوَّلاً، وَأَصْبَحَ هُوَ الْقَائِدَ.</span>
<span className="v" id="v7" data-v="7" data-xref="أخبار الأيام الأول 11:5; صموئيل الثاني 5:7; المزامير 2:6"> وَمَكَثَ دَاوُدُ فِي الْحِصْنِ فَدُعِيَ لِذَلِكَ مَدِينَةَ دَاوُدَ.</span>
<span className="v" id="v8" data-v="8" data-xref="القضاة 9:6; ملوك الأول 11:27; ملوك الثاني 12:20"> وَبَنَى الْمَدِينَةَ مِنْ حَوْلِهَا ابْتِدَاءً مِنَ الْقَلْعَةِ إِلَى السُّورِ الْمُحِيطِ بِها. ثُمَّ قَامَ يُوآبُ بِتَجْدِيدِ سَائِرِ الْمَدِينَةِ.</span>
<span className="v" id="v9" data-v="9" data-xref="صموئيل الثاني 3:1; إشعياء 41:10; أخبار الأيام الأول 9:20"> وَكَانَ دَاوُدُ يَزْدَادُ عُلُوَّ شَأْنٍ، لأَنَّ الرَّبَّ الْقَدِيرَ كَانَ مَعَهُ.</span>
</div>
</div>

## محاربو داود الجبابرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="أخبار الأيام الأول 11:3; صموئيل الأول 16:1; أخبار الأيام الأول 12:38"> وَهَذِهِ أَسْمَاءُ الرِّجَالِ الَّذِينَ آزَرُوهُ مَعَ بَقِيَّةِ إِسْرَائِيلَ لِيَجْعَلُوهُ مَلِكاً، بِمُقْتَضَى وَعْدِ الرَّبِّ الْمُتَعَلِّقِ بِإِسْرَائِيلَ،</span>
<span className="v" id="v11" data-v="11" data-xref="صموئيل الثاني 23:8; أخبار الأيام الأول 27:2; أخبار الأيام الأول 12:18"> وَهَؤُلاءِ هُمْ أَبْطَالُ دَاوُدَ: يَشُبْعَامُ بْنُ حَكْمُونِي، رَئِيسُ الأَبْطَالِ الثَّلاثَةِ، هَاجَمَ بِرُمْحِهِ ثَلاثَ مِئَةٍ وَقَتَلَهُمْ دُفْعَةً وَاحِدَةً.</span>
<span className="v" id="v12" data-v="12" data-xref="أخبار الأيام الأول 27:4; صموئيل الثاني 23:17–19; صموئيل الثاني 23:23"> ثُمَّ أَلِعَازَارُ بْنُ دُودُو الأَخُوخِيُّ، وَهُوَ وَاحِدٌ مِنَ الأَبْطَالِ الثَّلاثَةِ.</span>
<span className="v" id="v13" data-v="13" data-xref="صموئيل الثاني 23:11–12; صموئيل الأول 17:1"> كَانَ مَعَ دَاوُدَ فِي فَسَّ دَمِّيمَ حِينَ احْتَشَدَ الْفِلِسْطِينِيُّونَ لِلْحَرْبِ فِي حَقْلِ شَعِيرٍ، فَهَرَبَ الْجَيْشُ أَمَامَ الْفِلِسْطِينِيِّينَ.</span>
<span className="v" id="v14" data-v="14" data-xref="المزامير 144:10; ملوك الثاني 5:1; صموئيل الأول 14:23"> غَيْرَ أَنَّهُ ثَبَتَ مَعَ رِجَالِهِ فِي وَسَطِ الْحَقْلِ وَأَنْقَذَهُ وَقَضَى عَلَى الْفِلِسْطِينِيِّينَ، فَآتَاهُمُ الرَّبُّ نُصْرَةً عَظِيمَةً.</span>
<span className="v" id="v15" data-v="15" data-xref="أخبار الأيام الأول 14:9; صموئيل الثاني 5:18; إشعياء 17:5"> وَانْحَدَرَ ثَلاثَةٌ مِنَ الثَّلاثِينَ قَائِداً إِلَى الْمِنْطَقَةِ الصَّخْرِيَّةِ حَيْثُ كَانَ دَاوُدُ يُقِيمُ فِي مَغَارَةِ عَدُلَّامَ، بَيْنَمَا جَيْشُ الْفِلِسْطِينِيِّينَ مُعَسْكِرٌ فِي وَادِي الرَّفَائِيِّينَ.</span>
<span className="v" id="v16" data-v="16" data-xref="صموئيل الأول 10:5; صموئيل الأول 13:4; صموئيل الأول 13:23"> وَكَانَ دَاوُدُ آنَئِذٍ مُتَمَنِّعاً فِي الْحِصْنِ، وَحَامِيَةُ الْفِلِسْطِينِيِّينَ قَدِ احْتَلَّتْ بَيْتَ لَحْمٍ.</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه يوحنا 4:14; إشعياء 12:3; صموئيل الثاني 23:15–16"> فَتَأَوَّهَ دَاوُدُ وَقَالَ: «مَنْ يَسْقِينِي مَاءً مِنْ بِئْرِ بَيْتِ لَحْمٍ الْقَائِمَةِ عِنْدَ بَوَّابَةِ الْمَدِينَةِ؟»</span>
<span className="v" id="v18" data-v="18" data-xref="أعمال الرسل 21:13; صموئيل الأول 19:5; أعمال الرسل 20:24"> فَاقْتَحَمَ الثَّلاثَةُ مُعَسْكَرَ الْفِلِسْطِينِيِّينَ وَجَاءُوا بِمَاءٍ مِنْ بِئْرِ بَيْتِ لَحْمٍ الْقَائِمَةِ عِنْدَ بَوَّابَةِ الْمَدِينَةِ وَحَمَلُوهُ إِلَى دَاوُدَ، فَأَبَى أَنْ يَشْرَبَ مِنْهُ وَسَكَبَهُ لِلرَّبِّ.</span>
<span className="v" id="v19" data-v="19" data-xref="صموئيل الثاني 23:17; المزامير 72:14; القضاة 9:17"> وَقَالَ: «مَعَاذَ اللهِ أَنْ أَفْعَلَ ذَلِكَ! أَأَشْرَبُ دَمَ هَؤُلاءِ الرِّجَالِ الَّذِينَ جَازَفُوا بِحَيَاتِهِمْ، إِذْ خَاطَرُوا بِأَنْفُسِهِمْ لِيَأْتُوا بِهِ إِلَيَّ؟» وَأَبَى أَنْ يَشْرَبَ مِنْهُ. هَذَا مَا أَقْدَمَ عَلَيْهِ الأَبْطَالُ الثَّلاثَةُ.</span>
<span className="v" id="v20" data-v="20" data-xref="صموئيل الأول 26:6; أخبار الأيام الأول 2:16; صموئيل الثاني 20:6"> وَكَانَ أَبِيشَايُ أَخُو يُوآبَ رَئِيسَ الثَّلاثِينَ أَيْضاً، وَقَدْ هَاجَمَ بِرُمْحِهِ ثَلاثَ مِئَةٍ فَقَتَلَهُمْ، وَاشْتَهَرَ اسْمُهُ إِلَى جَانِبِ الْقُوَّادِ الثَّلاثَةِ.</span>
<span className="v" id="v21" data-v="21" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 15:41; الإنجيل كما دونه متى 13:8"> وَمَعَ أَنَّهُ لَمْ يَبْلُغْ مَرْتَبَةَ الْقُوَّادِ الثَّلاثَةِ الأَوَّلِينَ، إِلّا أَنَّهُ كَانَ قَائِداً لِلثَّلاثِينَ رَئِيساً.</span>
<span className="v" id="v22" data-v="22" data-xref="صموئيل الثاني 23:20–23; صموئيل الثاني 8:18; يشوع 15:21"> وَهُنَاكَ أَيْضاً بَنَايَاهُو بْنُ يَهُويَادَاعَ، مُحَارِبٌ جَبَّارٌ كَثِيرُ الْبُطُولاتِ، مِنْ قَبْصِيئِيلَ، وَهُوَ الَّذِي قَتَلَ بَطَلَيْ مُوآبَ، وَقَضَى عَلَى أَسَدٍ فِي وَسَطِ جُبٍّ فِي يَوْمٍ مُثْلِجٍ،</span>
<span className="v" id="v23" data-v="23" data-xref="صموئيل الأول 17:7; صموئيل الأول 17:4; التثنية 3:11"> كَمَا قَتَلَ عِمْلاقاً مِصْرِيًّا طُولُهُ خَمْسُ أَذْرُعٍ (نَحْوَ مِتْرَيْنِ وَنِصْفٍ)، كَانَ مُتَسَلِّحاً بِرُمْحٍ كَنَوْلِ النَّسَّاجِينَ، فَتَقَدَّمَ مِنْهُ بِعَصاً وَخَطَفَ الرُّمْحَ مِنْ يَدِهِ وَقَتَلَهُ بِهِ.</span>
<span className="v" id="v24" data-v="24"> هَذَا مَا أَقْدَمَ عَلَيْهِ بَنَايَاهُو بْنُ يَهُويَادَاعَ فَذَاعَتْ شُهْرَتُهُ إِلَى جَانِبِ الأَبْطَالِ الثَّلاثَةِ،</span>
<span className="v" id="v25" data-v="25" data-xref="صموئيل الثاني 20:23"> وَعَلا شَأْنُهُ بَيْنَ الثَّلاثِينَ قَائِداً، وَإِنْ لَمْ يَبْلُغْ مَرْتَبَةَ الأَبْطَالِ الثَّلاثَةِ. فَجَعَلَهُ دَاوُدُ مِنْ أُمَنَاءِ سِرِّهِ.</span>
<span className="v" id="v26" data-v="26" data-xref="أخبار الأيام الأول 27:7; صموئيل الثاني 21:19; صموئيل الثاني 2:18–23"> أَمَّا أَبْطَالُ الْجَيْشِ فَهُمْ: عَسَائِيلُ أَخُو يُوآبَ، وَأَلْحَانَانُ بْنُ دُودُوَ مِنْ بَيْتِ لَحْمٍ.</span>
<span className="v" id="v27" data-v="27" data-xref="صموئيل الثاني 23:25–26"> وَشَمُّوتُ الْهَرُورِيُّ، وَحَالَصُ الْفَلُونِيُّ،</span>
<span className="v" id="v28" data-v="28" data-xref="أخبار الأيام الأول 27:12; أخبار الأيام الأول 27:9; صموئيل الثاني 23:27"> وَعِيرَا بْنُ عِقِّيشَ التَّقُوعِيُّ، وَأَبِيعَزَرُ الْعَنَاثُوثِيُّ،</span>
<span className="v" id="v29" data-v="29" data-xref="صموئيل الثاني 23:27–28; أخبار الأيام الأول 27:11"> وَسِبْكَايُ الْحُوشَاتِيُّ، وَعِيلايُ الأَخُوخِيُّ،</span>
<span className="v" id="v30" data-v="30" data-xref="أخبار الأيام الأول 27:13; أخبار الأيام الأول 27:15; صموئيل الثاني 23:29"> وَمَهْرَايُ النَّطُوفَاتِيُّ، وَخَالِدُ بْنُ بَعْنَةَ النَّطُوفَاتِيُّ،</span>
<span className="v" id="v31" data-v="31" data-xref="صموئيل الثاني 23:29"> وَإِتَّايُ بْنُ رِيبَايَ مِنْ جِبْعَةِ بِنِي بِنْيَامِينَ، وَبَنَايَا الْفَرْعَتُونِيُّ،</span>
<span className="v" id="v32" data-v="32" data-xref="صموئيل الثاني 23:30–31"> وَحُورَايُ مِنْ أَوْدِيَةِ جَاعَشَ، وَأَبِيئِيلُ الْعَرَبَاتِيُّ،</span>
<span className="v" id="v33" data-v="33"> وَعَزْمُوتُ الْبَحْرُومِيُّ، وَإِلْيَحْبَا الشَّعَلْبُونِيُّ،</span>
<span className="v" id="v34" data-v="34" data-xref="صموئيل الثاني 23:32–33"> وَأَبْنَاءُ هَاشِمَ الْجَزُونِيِّ، وَيُونَاثَانُ بْنُ شَاجَايَ الْهَرَارِيُّ،</span>
<span className="v" id="v35" data-v="35" data-xref="صموئيل الثاني 23:33–34"> وَأَخِيآمُ بْنُ سَاكَارَ الْهَرَارِيُّ، وَأَلِيفَالُ بْنُ أُورَ،</span>
<span className="v" id="v36" data-v="36"> وَحَافِرُ الْمَكِيرَاتِيُّ، وَأَخِيَّا الْفَلُونِيُّ،</span>
<span className="v" id="v37" data-v="37" data-xref="صموئيل الثاني 23:35"> وَحَصْرُو الْكَرْمَلِيُّ، وَنَعْرَايُ بْنُ أَزْبَايَ،</span>
<span className="v" id="v38" data-v="38" data-xref="صموئيل الثاني 23:36"> وَيُوئِيلُ أَخُو نَاثَانَ، وَمِبْحَارُ بْنُ هَجْرِي،</span>
<span className="v" id="v39" data-v="39"> وَصَالِقُ الْعَمُّونِيُّ، وَنَحْرَايُ الْبَئِيرُوتِيُّ حَامِلُ سِلاحِ يُوآبَ ابْنِ صُرُوِيَّةَ،</span>
<span className="v" id="v40" data-v="40" data-xref="صموئيل الثاني 20:26"> وَعِيرَا الْيِثْرِيُّ، وَجَارِبُ الْيِثْرِيُّ،</span>

<span className="v" id="v42" data-v="42"> وَعَدِينَا بْنُ شِيزَا الرَّأُوبَيْنِيُّ زَعِيمُ الرَّأُوبَيْنِيِّينَ وَمَعَهُ ثَلاثُونَ جُنْدِيًّا،</span>
