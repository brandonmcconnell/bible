## أسرة داود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صموئيل الثاني 3:2–5; يشوع 15:56; صموئيل الثاني 13:1"> وَهَذَا سِجِلٌّ بِمَوَالِيدِ دَاوُدَ الَّذِينَ أَنْجَبَهُمْ فِي حَبْرُونَ: بِكْرُهُ أَمْنُونُ مِنْ أَخِينُوعَمَ الْيَزْرَعِيلِيَّةِ، ثُمَّ دَانِيئِيلُ مِنْ أَبِيجَايِلَ الْكَرْمَلِيَّةِ،</span>
<span className="v" id="v2" data-v="2" data-xref="صموئيل الثاني 19:4–10; صموئيل الثاني 3:4; ملوك الأول 2:24–25"> وَالثَّالِثُ أَبْشَالُومُ بْنُ مَعْكَةَ بِنْتِ تَلْمَايَ مَلِكِ جَشُورَ، وَالرَّابِعُ أَدُونِيَّا بْنُ حَجِّيثَ،</span>
<span className="v" id="v3" data-v="3" data-xref="صموئيل الثاني 3:5"> وَالْخَامِسُ شَفَطْيَا مِنْ أَبِيطَالَ، وَالسَّادِسُ يَثَرْعَامُ مِنْ عَجْلَةَ زَوْجَتِهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="صموئيل الثاني 2:11; صموئيل الثاني 5:4–5; ملوك الأول 2:11"> فَكَانَتْ جُمْلَةُ الْمَوْلُودِينَ لَهُ فِي حَبْرُونَ سِتَّةَ أَبْنَاءٍ، وَقَدْ مَلَكَ هُنَاكَ سَبْعَ سِنِينَ وَسِتَّةَ أَشْهُرٍ، ثُمَّ مَلَكَ فِي أُورُشَلِيمَ ثَلاثاً وَثَلاثِينَ سَنَةً.</span>
<span className="v" id="v5" data-v="5" data-xref="صموئيل الثاني 11:3; صموئيل الثاني 12:24–25; صموئيل الثاني 12:1–15"> أَمَّا الَّذِينَ أَنْجَبَهُمْ فِي أُورُشَلِيمَ فَهُمْ: شِمْعَى وَشُوبَابُ وَنَاثَانُ وَسُلَيْمَانُ، وَهَؤُلاءِ الأَرْبَعَةُ وَلَدَتْهُمْ بَثْشَبَعُ بِنْتُ عَمِّيئِيلَ.</span>
<span className="v" id="v6" data-v="6" data-xref="صموئيل الثاني 5:15; أخبار الأيام الأول 14:5; صموئيل الأول 14:5"> وَكَانَ لَهُ تِسْعَةُ أَبْنَاءٍ آخَرُونَ هُمْ يِبْحَارُ وَأَلِشَامَعُ وَأَلِيفَالَطُ،</span>
<span className="v" id="v7" data-v="7" data-xref="صموئيل الثاني 5:15–16"> وَنُوجَهُ وَنَافَجُ وَيَافِيعُ،</span>
<span className="v" id="v8" data-v="8" data-xref="أخبار الأيام الأول 14:7; صموئيل الثاني 5:14–16"> وَأَلِيشَمَعُ وَأَلِيَادَاعُ وَأَلِيفَلَطُ.</span>
<span className="v" id="v9" data-v="9" data-xref="صموئيل الثاني 13:1–20; صموئيل الثاني 5:13"> وَجَمِيعُهُمْ أَبْنَاءُ دَاوُدَ مَاعَدَا أَبْنَاءَ الْمَحْظِيَّاتِ. وَكَانَتْ لَهُمْ أُخْتٌ تُدْعَى ثَامَارَ.</span>
</div>

## ملوك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ملوك الأول 11:43; ملوك الأول 14:31–1; ملوك الأول 15:24"> وَهَذِهِ أَسْمَاءُ أَبْنَاءِ سُلَيْمَانَ وَأَحْفَادِهِ عَلَى التَّعَاقُبِ الَّذِينَ تَوَالَوْا عَلَى الْمُلْكِ: رَحَبْعَامُ، أَبِيَّا، آسَا، يَهُوشَافَاطُ،</span>
<span className="v" id="v11" data-v="11" data-xref="ملوك الثاني 8:24; أخبار الأيام الثاني 21:17; ملوك الثاني 11:21"> يُورَامُ، أَخَزْيَا، يُوآشُ،</span>
<span className="v" id="v12" data-v="12" data-xref="أخبار الأيام الثاني 25:1; ملوك الثاني 14:21; ملوك الثاني 14:1"> أَمَصْيَا، عَزَرْيَا، يُوثَامُ،</span>
<span className="v" id="v13" data-v="13" data-xref="ملوك الثاني 18:1; ملوك الثاني 16:1; أخبار الأيام الثاني 29:1"> آحَازُ، حَزَقِيَّا، مَنَسَّى،</span>
<span className="v" id="v14" data-v="14" data-xref="ملوك الثاني 23:30; أخبار الأيام الثاني 34:1; ملوك الثاني 21:19"> آمُونُ وَيُوشِيَّا.</span>
<span className="v" id="v15" data-v="15" data-xref="ملوك الثاني 23:34"> أَمَّا أَبْنَاءُ يُوشِيَّا فَهُمْ: الْبِكْرُ يُوحَانَانُ، ثُمَّ يَهُويَاقِيمُ، وَصِدْقِيَّا، وَأَخِيراً شَلُّومُ.</span>
<span className="v" id="v16" data-v="16" data-xref="ملوك الثاني 24:6; الإنجيل كما دونه متى 1:11; ملوك الثاني 24:17"> وَابْنَا يَهُويَاقِيمُ يَكُنْيَا وَصِدْقِيَّا.</span>

## النسل الملكي بعد السبي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="عزرا 3:2; عزرا 5:2; الإنجيل كما دونه متى 1:12"> وَأَنْجَبَ يَكُنْيَا: أَسِيرَ وَشَأَلْتِيئِيلَ (وَمِنْ أَحْفَادِ يَهُويَاقِيمَ):</span>
<span className="v" id="v18" data-v="18"> مَلْكِيرَامُ وَفَدَايَا وَشِنْأَصَّرُ وَيَقَمْيَا وَهُوشَامَاعُ وَنَدَبْيَا.</span>
<span className="v" id="v19" data-v="19" data-xref="عزرا 2:2; عزرا 3:2; حجي 2:2"> وَأَنْجَبَ فَدَايَا: زَرُبَّابَلَ وَشِمْعِي. أَمَّا أَبْنَاءُ زَرُبَّابَلَ فَهُمْ مَشُلّامُ، وَحَنَنْيَا وَأُخْتُهُمْ شَلُومِيَةُ،</span>
<span className="v" id="v20" data-v="20"> وَحَشُوبَةُ وَأُوهَلُ، وَبَرَخْيَا وَحَسَدْيَا، وَيُوشَبُ حَسَدَ، وَهُمْ خَمْسَةٌ في جُمْلَتِهِمْ.</span>
<span className="v" id="v21" data-v="21" data-xref="نحميا 10:22"> وَابْنَا حَنَنْيَا: فَلَطْيَا، وَيِشْعِيَا، وَمِنْ أَحْفَادِهِ: أَبْنَاءُ رَفَايَا وَأَرْنَانَ وَعُوبَدْيَا وَشَكَنْيَا.</span>
<span className="v" id="v22" data-v="22" data-xref="عزرا 8:2"> وَأَنْجَبَ شَكُنْيَا شَمْعِيَا؛ وَأَبْنَاءُ شَمْعِيَا الْخَمْسَةُ هُمْ: حَطُّوشُ وَيَجْآلُ وَبَارِيحُ وَنَعَرْيَا وَشَافَاطُ.</span>
<span className="v" id="v23" data-v="23"> وَكَانَ لِنَعَرْيَا ثَلاثَةُ أَبْنَاءٍ هُمْ: أَلْيُوعِينِيُّ، وَحَزَقِيَّا، وَعَزْرِيقَامُ.</span>
<span className="v" id="v24" data-v="24"> أَمَّا أَبْنَاءُ الْيُوعِينِيِّ فَهُمْ هُودَايَاهُو وَأَلِيَاشِيبُ وَفَلايَا وَعَقُّوبُ وَيُوحَانَانُ وَدَلايَا وَعَنَانِي. وَهُمْ سَبْعَةٌ.</span>
