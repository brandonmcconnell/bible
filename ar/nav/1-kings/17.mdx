## إيليا يتنبأ بجفاف عظيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه لوقا 4:25–26; رسالة يعقوب 5:17; الإنجيل كما دونه لوقا 1:17"> وَقَالَ إِيلِيَّا التَّشْبِيُّ مِنْ أَهْلِ جِلْعَادَ لأَخْآبَ: «حَيٌّ هُوَ الرَّبُّ إِلَهُ إِسْرَائِيلَ الَّذِي أَخْدُمُهُ، إِنَّهُ لَنْ يَهْطِلَ نَدىً وَلا مَطَرٌ فِي هَذِهِ السِّنِينَ، إِلّا حِينَ أُعْلِنُ ذَلِكَ».</span>

## الغربان تعول إيليا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="إرميا 18:1; إرميا 7:1; هوشع 1:1–2"> وَأَمَرَ الرَّبُّ إِيلِيَّا:</span>
<span className="v" id="v3" data-v="3" data-xref="رؤيا يوحنا 12:6; إرميا 36:26; المزامير 83:3"> «امْضِ مِنْ هُنَا وَاتَّجِهْ نَحْوَ الْمَشْرِقِ، وَاخْتَبِئْ عِنْدَ نَهْرِ كَرِيثَ الْمُقَابِلِ لِنَهْرِ الأُرْدُنِّ،</span>
<span className="v" id="v4" data-v="4" data-xref="ملوك الأول 17:9; المزامير 147:9; أيوب 38:41"> فَتَشْرَبَ مِنْ مِيَاهِهِ وَتَقْتَاتَ مِمَّا تُحْضِرُهُ لَكَ الْغِرْبَانُ الَّتِي أَمَرْتُهَا أَنْ تَعُولَكَ هُنَاكَ».</span>
<span className="v" id="v5" data-v="5" data-xref="ملوك الأول 19:9; الإنجيل كما دونه متى 16:24; الإنجيل كما دونه يوحنا 15:14"> فَانْطَلَقَ وَنَفَّذَ أَمْرَ الرَّبِّ، وَأَقَامَ عِنْدَ نَهْرِ كَرِيثَ مُقَابِلَ نَهْرِ الأُرْدُنِّ،</span>
<span className="v" id="v6" data-v="6" data-xref="المزامير 78:23–24; الإنجيل كما دونه متى 6:31–33; إرميا 37:21"> فَكَانَتِ الْغِرْبَانُ تُحْضِرُ إِلَيْهِ الْخُبْزَ وَاللَّحْمَ صَبَاحاً وَمَسَاءً، وَكَانَ يَشْرَبُ مِنْ مَاءِ النَّهْرِ.</span>
<span className="v" id="v7" data-v="7" data-xref="إشعياء 54:10; إشعياء 40:30–31"> وَمَا لَبِثَ أَنْ جَفَّ النَّهْرُ بَعْدَ زَمَنٍ، لأَنَّهُ لَمْ يَهْطِلْ مَطَرٌ عَلَى الأَرْضِ.</span>
</div>
</div>

## إيليا وأرملة صرفة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="التكوين 22:14; ملوك الأول 17:2; الرسالة إلى العبرانيين 13:6"> فَخَاطَبَ الرَّبُّ إِيلِيَّا:</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه لوقا 4:26; عوبديا 1:20; الرسالة الثانية إلى مؤمني كورنثوس 4:7"> «قُمْ وَتَوَجَّهْ إِلَى صِرْفَةَ التَّابِعَةِ لِصِيدُونَ، وَامْكُثْ هُنَاكَ، فَقَدْ أَمَرْتُ هُنَاكَ أَرْمَلَةً أَنْ تَتَكَفَّلَ بِإِعَالَتِكَ».</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه يوحنا 4:7; التكوين 24:17; التكوين 21:15"> فَذَهَبَ إِلَى صِرْفَةَ. وَعِنْدَمَا وَصَلَ إِلَى بَوَّابَةِ الْمَدِينَةِ شَاهَدَ امْرَأَةً تَجْمَعُ حَطَباً، فَقَالَ لَهَا: «هَاتِي لِي بَعْضَ الْمَاءِ فِي إِنَاءٍ لأَشْرَبَ».</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه متى 25:35–40; ملوك الأول 18:4; الرسالة إلى العبرانيين 13:2"> وَفِيمَا هِيَ ذَاهِبَةٌ لِتُحْضِرَهُ نَادَاهَا ثَانِيَةً وَقَالَ: «هَاتِي لِي كِسْرَةَ خُبْزٍ مَعَكِ».</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه متى 15:33–34; ملوك الأول 17:1; التكوين 21:16"> فَأَجَابَتْهُ: «حَيٌّ هُوَ الرَّبُّ إِلَهُكَ إِنَّهُ لَيْسَ لَدَيَّ كَعْكَةٌ، إِنَّمَا حَفْنَةُ دَقِيقٍ فِي الْجَرَّةِ، وَقَلِيلٌ مِنَ الزَّيْتِ فِي قَارُورَةٍ. وَهَا أَنَا أَجْمَعُ بَعْضَ عِيدَانِ الْحَطَبِ لِآخُذَهَا وَأُعِدَّ لِي ولاِبْنِي طَعَاماً نَأْكُلُهُ ثُمَّ نَمُوتُ».</span>
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه متى 6:33; ملاخي 3:10; التكوين 22:1–2"> فَقَالَ لَهَا إِيلِيَّا: «لا تَخَافِي. امْضِي وَاصْنَعِي كَمَا قُلْتِ، وَلَكِنْ أَعِدِّي لِي مِنْهُ كَعْكَةً صَغِيرَةً أَوَّلاً وَأَحْضِرِيهَا لِي، ثُمَّ اعْمَلِي لَكِ ولاِبْنِكِ أَخِيراً،</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه متى 15:36–38; الإنجيل كما دونه متى 14:17–20; ملوك الثاني 4:42–44"> لأَنَّ هَذَا مَا يَقُولُهُ الرَّبُّ إِلَهُ إِسْرَائِيلَ: إِنَّ جَرَّةَ الدَّقِيقِ لَنْ تَفْرُغَ، وَقَارُورَةَ الزَّيْتِ لَنْ تَنْقُصَ، إِلَى الْيَوْمِ الَّذِي يُرْسِلُ فِيهِ الرَّبُّ مَطَراً عَلَى وَجْهِ الأَرْضِ».</span>
<span className="v" id="v15" data-v="15" data-xref="الرسالة إلى العبرانيين 11:17; الإنجيل كما دونه مرقس 12:43; الإنجيل كما دونه متى 15:28"> فَرَاحَتْ إِلَى مَنْزِلِهَا وَنَفَّذَتْ كَلامَ إِيلِيَّا، فَتَوَافَرَ لَهَا طَعَامٌ لِتَأْكُلَ هِيَ وَابْنُهَا وَإِيلِيَّا لِمُدَّةٍ طَوِيلَةٍ.</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه متى 9:28–30; الإنجيل كما دونه يوحنا 4:50–51; الإنجيل كما دونه لوقا 1:37"> جَرَّةُ الدَّقِيقِ لَمْ تَفْرُغْ، وَقَارُورَةُ الزَّيْتِ لَمْ تَنْقُصْ، تَمَاماً كَمَا قَالَ الرَّبُّ عَلَى لِسَانِ إِيلِيَّا.</span>
<span className="v" id="v17" data-v="17" data-xref="رسالة بطرس الأولى 1:7; دانيال 5:23; الإنجيل كما دونه يوحنا 11:14"> وَحَدَثَ بَعْدَ زَمَنٍ أَنَّ ابْنَ الْمَرْأَةِ صَاحِبَةِ الْبَيْتِ اشْتَدَّ عَلَيْهِ الْمَرَضُ، وَمَاتَ،</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه لوقا 5:8; الإنجيل كما دونه يوحنا 2:4; ملوك الثاني 3:13"> فَقَالَتْ لإِيِليَّا: «أَيُّ ذَنْبٍ جَنَيْتُهُ بِحَقِّكَ يَا رَجُلَ اللهِ؟ هَلْ جِئْتَ إِليَّ لِتُذَكِّرَنِي بِإِثْمِي وَتُمِيتَ ابْنِي؟»</span>
<span className="v" id="v19" data-v="19" data-xref="ملوك الثاني 4:32; ملوك الثاني 4:10; أعمال الرسل 9:37"> فَقَالَ لَهَا: «أَعْطِينِي ابْنَكِ». وَأَخَذَهُ مِنْهَا وَصَعِدَ بِهِ إِلَى الْعُلِّيَّةِ الَّتِي كَانَ مُقِيماً فِيهَا وَأَضْجَعَهُ عَلَى سَرِيرِهِ،</span>
<span className="v" id="v20" data-v="20" data-xref="رسالة يعقوب 5:15–18; الإنجيل كما دونه متى 21:22; يشوع 7:8–9"> وَاسْتَغَاثَ بِالرَّبِّ مُتَضَرِّعاً: «أَيُّهَا الرَّبُّ إِلَهِي، أَإِلَى الأَرْمَلَةِ الَّتِي أَنَا نَازِلٌ عِنْدَهَا تُسِيءُ أَيْضاً وَتُمِيتُ ابْنَهَا؟»</span>
<span className="v" id="v21" data-v="21" data-xref="أعمال الرسل 20:10; أعمال الرسل 9:40; ملوك الثاني 4:33–35"> ثُمَّ تَمَدَّدَ إِيلِيَّا عَلَى جُثَّةِ الْوَلَدِ ثَلاثَ مَرَّاتٍ وَابْتَهَلَ إِلَى الرَّبِّ: «يَا رَبُّ إِلَهِي، أَرْجِعْ نَفْسَ هَذَا الْوَلَدِ إِلَيْهِ».</span>
<span className="v" id="v22" data-v="22" data-xref="التثنية 32:39; رؤيا يوحنا 11:11; الرسالة إلى العبرانيين 11:35"> فَاسْتَجَابَ الرَّبُّ دُعَاءَ إِيلِيَّا، وَرَجَعَتْ نَفْسُ الْوَلَدِ إِلَيْهِ فَعَاشَ.</span>
<span className="v" id="v23" data-v="23" data-xref="ملوك الثاني 4:36–37; أعمال الرسل 9:41; الإنجيل كما دونه لوقا 7:15"> فَأَخَذَ إِيلِيَّا الْوَلَدَ وَنَزَلَ بِهِ مِنَ الْعُلِّيَّةِ إِلَى الْبَيْتِ، وَسَلَّمَهُ إِلَى أُمِّهِ، وَقَالَ لَهَا: «انْظُرِي، إِنَّ ابْنَكِ حَيٌّ»</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه يوحنا 16:30; الإنجيل كما دونه يوحنا 2:11; الإنجيل كما دونه يوحنا 3:2"> فَقَالَتِ الْمَرْأَةُ لإِيلِيَّا: «الآنَ عَلِمْتُ أَنَّكَ رَجُلُ اللهِ، وَأَنَّ اللهَ يَنْطِقُ عَلَى لِسَانِكَ بِالْحَقِّ».</span>
