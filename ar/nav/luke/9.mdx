## يسوع يرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه يوحنا 14:12; الإنجيل كما دونه لوقا 10:19; الإنجيل كما دونه متى 16:19"> ثُمَّ جَمَعَ يَسُوعُ الاِثْنَيْ عَشَرَ، وَمَنَحَهُمْ قُدْرَةً وَسُلْطَةً عَلَى جَمِيعِ الشَّيَاطِينِ وَعَلَى الأَمْرَاضِ لِشِفَائِهَا،</span>
<span className="v" id="v2" data-v="2" data-xref="الإنجيل كما دونه متى 10:7–8; الإنجيل كما دونه لوقا 10:9; الإنجيل كما دونه متى 3:2"> وَأَرْسَلَهُمْ لِيُبَشِّرُوا بِمَلَكُوتِ اللهِ وَيَشْفُوا.</span>
<span className="v" id="v3" data-v="3" data-xref="الإنجيل كما دونه لوقا 22:35; المزامير 37:3; الإنجيل كما دونه لوقا 10:4–12"> وَقَالَ لَهُمْ: «لا تَحْمِلُوا لِلطَّرِيقِ شَيْئاً: لَا عَصاً، وَلا زَاداً، وَلا خُبْزاً، وَلا مَالاً، وَلا يَحْمِلِ الْوَاحِدُ ثَوْبَيْنِ.</span>
<span className="v" id="v4" data-v="4" data-xref="الإنجيل كما دونه متى 10:11; الإنجيل كما دونه مرقس 6:10; الإنجيل كما دونه لوقا 10:5–8"> وَأَيَّ بَيْتٍ دَخَلْتُمْ فَهُنَاكَ أَقِيمُوا وَمِنْ هُنَاكَ ارْحَلُوا.</span>
<span className="v" id="v5" data-v="5" data-xref="أعمال الرسل 13:51; أعمال الرسل 18:6; الإنجيل كما دونه مرقس 6:11"> وَإِنْ كَانَ أَحَدٌ لَا يَقْبَلُكُمْ فِي مَدِينَةٍ مَا، فَاخْرُجُوا مِنْ هُنَاكَ، وَانْفُضُوا الْغُبَارَ عَنْ أَقْدَامِكُمْ، شَهَادَةً عَلَيْهِمْ».</span>
<span className="v" id="v6" data-v="6" data-xref="أعمال الرسل 4:30; الإنجيل كما دونه مرقس 16:20; أعمال الرسل 5:15"> فَانْطَلَقُوا يَجْتَازُونَ فِي الْقُرَى وَهُمْ يُبَشِّرُونَ وَيَشْفُونَ فِي كُلِّ مَكَانٍ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه مرقس 6:14–28; الإنجيل كما دونه متى 14:1–12; الإنجيل كما دونه لوقا 3:1"> وَسَمِعَ هِيرُودُسُ حَاكِمُ الرُّبْعِ بِكُلِّ مَا كَانَ يَجْرِي، فَتحَيَّرَ، لأَنَّ بَعْضاً كَانُوا يَقُولُونَ: «إِنَّ يُوحَنَّا قَامَ مِنْ بَيْنِ الأَمْوَاتِ!»</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه يوحنا 1:21; الإنجيل كما دونه لوقا 9:19; الإنجيل كما دونه مرقس 6:15"> وَبَعْضاً يَقُولُونَ: «إِنَّ إِيلِيَّا ظَهَرَ!» وَآخَرِينَ: «إِنَّ وَاحِداً مِنَ الأَنْبِيَاءِ الْقُدَامَى قَامَ!»</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه لوقا 23:8; الإنجيل كما دونه لوقا 13:31–32"> فَقَالَ هِيرُودُسُ: «يُوحَنَّا، أَنَا قَطَعْتُ رَأْسَهُ، وَلكِنْ مَنْ هُوَ هَذَا الَّذِي أَسْمَعُ عَنْهُ مِثْلَ هذِهِ الأُمُورِ؟» وَكَانَ يَرْغَبُ فِي أَنْ يَرَاهُ.</span>
</div>
</div>

## يسوع يطعم الخمسة الآلاف

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه متى 11:21; الرسالة إلى العبرانيين 13:17; الإنجيل كما دونه متى 14:13–21"> وَبَعْدَمَا رَجَعَ الرُّسُلُ، أَخْبَرُوهُ بِجَمِيعِ مَا فَعَلُوا، فَأَخَذَهُمْ وَذَهَبَ بِهِمْ عَلَى انْفِرَادٍ إِلَى مَدِينَةٍ اسْمُهَا بَيْتُ صَيْدَا.</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه لوقا 9:2; الإنجيل كما دونه لوقا 5:31; الرسالة إلى العبرانيين 4:16"> وَلكِنَّ الْجُمُوعَ عَلِمُوا بِذَلِكَ فَلَحِقُوا بِهِ، فَاسْتَقْبَلَهُمْ وَحَدَّثَهُمْ عَنْ مَلَكُوتِ اللهِ، وَشَفَى مِنْهُمْ مَنْ كَانُوا مُحْتَاجِينَ إِلَى الشِّفَاءِ.</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه يوحنا 6:1; الإنجيل كما دونه يوحنا 6:5–15; هوشع 13:5"> وَلَمَّا كَادَ النَّهَارُ يَنْقَضِي، تَقَدَّمَ إِلَيْهِ الاثْنَا عَشَرَ وَقَالُوا لَهُ: «اصْرِفِ الْجَمْعَ لِيَذْهَبُوا إِلَى الْقُرَى الْمُجَاوِرَةِ، وَإِلَى الْمَزَارِعِ، فَيَبِيتُوا هُنَاكَ وَيَجِدُوا طَعَاماً، لأَنَّنَا هُنَا فِي مَكَانٍ مُقْفِرٍ!»</span>
<span className="v" id="v13" data-v="13" data-xref="الأمثال 11:24–25; الإنجيل كما دونه يوحنا 6:5–9; ملوك الثاني 4:42–43"> فَقَالَ لَهُمْ: «أَعْطُوهُمْ أَنْتُمْ لِيَأْكُلُوا!» أَجَابُوا: «لَيْسَ عِنْدَنَا أَكْثَرُ مِنْ خَمْسَةِ أَرْغِفَةٍ وَسَمَكَتَيْنِ إِلّا إِذَا ذَهَبْنَا وَاشْتَرَيْنَا طَعَاماً لِهَذَا الشَّعْبِ كُلِّهِ».</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه مرقس 8:6; الرسالة الأولى إلى مؤمني كورنثوس 14:40; الإنجيل كما دونه مرقس 6:39–40"> فَقَدْ كَانُوا نَحْوَ خَمْسَةِ آلافِ رَجُلٍ. ثُمَّ قَالَ لِتَلامِيذِهِ: «أَجْلِسُوهُمْ فِي جَمَاعَاتٍ تَتَأَلَّفُ كُلٌّ مِنْهَا مِنْ خَمْسِينَ».</span>
<span className="v" id="v15" data-v="15"> فَفَعَلُوا، وَأَجْلَسُوا الْجَمِيعَ.</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه متى 14:19; الرسالة الأولى إلى تيموثاوس 4:4–5; الإنجيل كما دونه متى 15:36"> فَأَخَذَ الأَرْغِفَةَ الْخَمْسَةَ وَالسَّمَكَتَيْنِ، وَرَفَعَ عَيْنَيْهِ نَحْوَ السَّمَاءِ، ثُمَّ بَارَكَهَا وَكَسَّرَهَا وَأَعْطَى التَّلامِيذَ لِيُقَدِّمُوا إِلَى الْجَمْعِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ملوك الثاني 4:44; الإنجيل كما دونه متى 15:37–38; الإنجيل كما دونه يوحنا 6:11–13"> فَأَكَلَ الْجَمِيعُ وَشَبِعُوا. ثُمَّ رُفِعَ مِنَ الْكِسَرِ الْفَاضِلَةِ عَنْهُمُ اثْنَتَا عَشْرَةَ قُفَّةً.</span>
</div>
</div>

## بطرس يشهد بحقيقة المسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه متى 16:13–16; الإنجيل كما دونه لوقا 11:1; الإنجيل كما دونه متى 26:36"> وَفِيمَا كَانَ يُصَلِّي عَلَى انْفِرَادٍ وَالتَّلامِيذُ مَعَهُ، سأَلَهُمْ: «مَنْ يَقُولُ الْجُمُوعُ إِنِّي أَنَا؟»</span>
<span className="v" id="v19" data-v="19" data-xref="ملاخي 4:5; الإنجيل كما دونه لوقا 9:7–8; الإنجيل كما دونه متى 14:2"> فَأَجَابُوهُ: «يَقُولُ بَعْضُهُمْ إِنَّكَ يُوحَنَّا الْمَعْمَدَانُ، وَآخَرُونَ إِنَّكَ إِيِليَّا، وَآخَرُونَ إِنَّكَ وَاحِدٌ مِنَ الأَنْبِيَاءِ الْقُدَامَى وَقَدْ قَامَ!»</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه متى 16:15–17; الإنجيل كما دونه يوحنا 4:29; الإنجيل كما دونه يوحنا 7:41"> فَسَأَلَهُمْ: «وَأَنْتُمْ، مَنْ تَقُولُونَ إِنِّي أَنَا؟» فَأَجَابَهُ بُطْرُسُ: «أَنْتَ مَسِيحُ اللهِ!».</span>
</div>

## يسوع ينبئ بموته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه متى 16:20; الإنجيل كما دونه متى 17:9; الإنجيل كما دونه مرقس 8:30–31"> وَلكِنَّهُ حَذَّرَهُمْ، مُوْصِياً أَلّا يُخْبِرُوا أَحَداً بِذَلِكَ.</span>
<span className="v" id="v22" data-v="22" data-xref="إشعياء 53:1–12; الإنجيل كما دونه لوقا 24:7; الإنجيل كما دونه لوقا 18:31–34"> وَقَالَ: «لابُدَّ أَنْ يَتَأَلَّمَ ابْنُ الإِنْسَانِ كَثِيراً وَيَرْفُضَهُ الشُّيُوخُ وَرُؤَسَاءُ الْكَهَنَةِ وَالْكَتَبَةُ، وَيُقْتَلَ، وَفِي الْيَوْمِ الثَّالِثِ يُقَامَ».</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه متى 10:38–39; الإنجيل كما دونه لوقا 14:26–27; الرسالة إلى مؤمني روما 8:13"> ثُمَّ قَالَ لِلْجَمِيعِ: «إِنْ أَرَادَ أَحَدٌ أَنْ يَسِيرَ وَرَائِي، فَلْيُنْكِرْ نَفْسَهُ وَيَحْمِلْ صَلِيبهُ كُلَّ يَوْمِ وَيَتْبَعْنِي.</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه لوقا 17:33; الإنجيل كما دونه يوحنا 12:25; أعمال الرسل 20:23–24"> فَأَيُّ مَنْ أَرَادَ أَنْ يُخَلِّصَ نَفْسَهُ، يَخْسَرُهَا؛ وَلكِنَّ مَنْ يَخْسَرُ نَفْسَهُ مَنْ أَجْلِي، فَهُوَ يُخَلِّصُهَا.</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه مرقس 8:36; الإنجيل كما دونه متى 16:26; المزامير 49:6–8"> فَمَاذَا يَنْتَفِعُ الإِنْسَانُ لَوْ رَبِحَ الْعَالَمَ كُلَّهُ وَخَسِرَ نَفْسَهُ أَوْ أَهْلَكَهَا؟</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه متى 10:32–33; الرسالة الثانية إلى تيموثاوس 2:12; الإنجيل كما دونه لوقا 12:8–9"> فَإِنَّ مَنْ يَسْتَحِي بِي وَبِكَلامِي، فَبِهِ يَسْتَحِي ابْنُ الإِنْسَانِ لَدَى عَوْدَتِهِ فِي مَجْدِهِ وَمَجْدِ الآبِ وَالْمَلائِكَةِ الْمُقَدَّسِينَ.</span>
<span className="v" id="v27" data-v="27" data-xref="الإنجيل كما دونه متى 16:28; الإنجيل كما دونه مرقس 9:1; الإنجيل كما دونه يوحنا 21:22–23"> وَلكِنِّي أَقُولُ لَكُمْ بِحَقٍّ إِنَّ بَيْنَ الْوَاقِفِينَ هُنَا بَعْضاً لَنْ يَذُوقُوا الْمَوْتَ حَتَّى يَرَوْا مَلَكوتَ اللهِ».</span>
</div>

## التجلي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه متى 17:1–13; الإنجيل كما دونه متى 26:37–39; الإنجيل كما دونه مرقس 14:33–36"> وَحَدَثَ بَعْدَ هَذَا الْكَلامِ بِثَمَانِيَةِ أَيَّامٍ تَقْرِيباً أَنْ أَخَذَ يَسُوعُ بُطْرُسَ وَيُوحَنَّا وَيَعْقُوبَ، وَصَعِدَ إِلَى جَبَلٍ لِيُصَلِّيَ.</span>
<span className="v" id="v29" data-v="29" data-xref="الإنجيل كما دونه متى 17:2; الإنجيل كما دونه مرقس 9:2–3; الخروج 34:29–35"> وَبَيْنَمَا هُوَ يُصَلِّي، تَجَلَّتْ هَيْئَةُ وَجْهِهِ وَصَارَتْ ثِيَابُهُ بَيْضَاءَ لَمَّاعَةً.</span>
<span className="v" id="v30" data-v="30" data-xref="الرسالة إلى العبرانيين 3:3–6; الرسالة الثانية إلى مؤمني كورنثوس 3:7–11; الإنجيل كما دونه لوقا 1:17"> وَإذَا رَجُلانِ يَتَحَدَّثَانِ مَعَهُ، هُمَا مُوسَى وَإِيلِيَّا،</span>
<span className="v" id="v31" data-v="31" data-xref="الرسالة إلى مؤمني فيلبي 3:21; رسالة بطرس الأولى 1:11–12; رسالة بطرس الأولى 5:10"> وَقَدْ ظَهَرَا بِمَجْدٍ وَتَكَلَّمَا عَنْ رَحِيلِهِ الَّذِي كَانَ عَلَى وَشْكِ إِتْمَامِهِ فِي أُورُشَلِيمَ.</span>
<span className="v" id="v32" data-v="32" data-xref="دانيال 8:18; الإنجيل كما دونه يوحنا 1:14; دانيال 10:9"> وَمَعَ أَنَّ بُطْرُسَ وَرَفِيقَيْهِ قَدْ غَالَبَهُمُ النَّوْمُ، فَإِنَّهُمْ حِينَ اسْتَيْقَظُوا تَمَاماً، شَاهَدُوا مَجْدَهُ وَالرَّجُلَيْنِ الْوَاقِفَيْنِ مَعَهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه مرقس 9:5–6; الإنجيل كما دونه مرقس 10:38; المزامير 4:6–7"> وَفِيمَا كَانَا يُفَارِقَانِهِ، قَالَ بُطْرُسُ لِيَسُوعَ: «يَا مُعَلِّمُ، مَا أَحْسَنَ أَنْ نَبْقَى هُنَا! فَلْنَنْصُبْ ثَلاثَ خِيَامٍ: وَاحِدَةً لَكَ، وَوَاحِدَةً لِمُوسَى، وَوَاحِدَةً لإِيلِيَّا» وَهُوَ لَا يَدْرِي مَا يَقُولُ.</span>
<span className="v" id="v34" data-v="34" data-xref="الخروج 40:34–38; الإنجيل كما دونه متى 17:5–7; القضاة 13:22"> وَلكِنَّهُ فِيمَا كَانَ يَقُولُ ذلِكَ، جَاءَتْ سَحَابَةٌ فَخَيَّمَتْ عَلَيْهِمْ، فَخَافَ التَّلامِيذُ عِنْدَمَا طَوَّقَتْهُمُ السَّحَابَةُ،</span>
<span className="v" id="v35" data-v="35" data-xref="الإنجيل كما دونه لوقا 3:22; أعمال الرسل 3:22–23; الإنجيل كما دونه متى 3:17"> وَانْطَلَقَ صَوْتٌ مِنَ السَّحَابَةِ يَقُولُ: «هَذَا هُوَ ابْنِي الَّذِي اخْتَرْتُهُ. لَهُ اسْمَعُوا!»</span>
<span className="v" id="v36" data-v="36" data-xref="الإنجيل كما دونه متى 17:9; الإنجيل كما دونه مرقس 9:9–10; الإنجيل كما دونه مرقس 9:6"> وَفِيمَا انْطَلَقَ الصَّوْتُ، وُجِدَ يَسُوعُ وَحْدَهُ. وَقَدْ كَتَمُوا الْخَبَرَ فَلَمْ يُخْبِرُوا أَحَداً فِي تِلْكَ الأَيَّامِ بِأَيِّ شَيْءٍ مِمَّا رَأَوْهُ.</span>

## شفاء صبي به روح شرير

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="الإنجيل كما دونه متى 17:14–20; الإنجيل كما دونه مرقس 9:14–29"> وَفِي الْيَوْمِ التَّالِي، لَمَّا نَزَلُوا مِنَ الْجَبَلِ، لاقَاهُ جَمْعٌ عَظِيمٌ.</span>
<span className="v" id="v38" data-v="38" data-xref="زكريا 12:10; الإنجيل كما دونه يوحنا 4:47; الإنجيل كما دونه لوقا 7:12"> وَإذَا فِي الْجَمْعِ رَجُلٌ نَادَى قَائِلاً: «يَا مُعَلِّمُ، أَتَوَسَّلُ إِلَيْكَ أَنْ تَنْظُرَ إِلَى ابْنِي، فَإِنَّهُ وَلَدِي الْوَحِيدُ.</span>
<span className="v" id="v39" data-v="39" data-xref="الإنجيل كما دونه مرقس 9:20; الإنجيل كما دونه لوقا 8:29; الإنجيل كما دونه لوقا 4:35"> وَهَا إِنَّ رُوحاً يَتَمَلَّكُهُ، فَيَصْرُخُ فَجْأَةً، وَيَصْرَعُهُ الرُّوحُ فَيُزْبِدُ، وَبِالْجَهْدِ يُفَارِقُهُ بَعْدَ أَنْ يُرَضِّضَهُ.</span>
<span className="v" id="v40" data-v="40" data-xref="الإنجيل كما دونه لوقا 9:1; أعمال الرسل 19:13–16; الإنجيل كما دونه لوقا 10:17–19"> وَقَدِ الْتَمَسْتُ مِنْ تَلامِيذِكَ أَنْ يَطْرُدُوهُ، فَلَمْ يَقْدِرُوا».</span>
<span className="v" id="v41" data-v="41" data-xref="التثنية 32:5; الرسالة إلى مؤمني روما 2:4; الإنجيل كما دونه يوحنا 20:27"> فَأَجَابَ يَسُوعُ قَائِلاً: «أَيُّهَا الْجِيلُ غَيْرُ الْمُؤْمِنِ وَالْمُنْحَرِفُ! إِلَى مَتَى أَبْقَى مَعَكُمْ وَأَحْتَمِلُكُمْ؟» (وَقَالَ لِلرَّجُلِ): «أَحْضِرِ ابْنَكَ إِلَى هُنَا!»</span>
<span className="v" id="v42" data-v="42" data-xref="الإنجيل كما دونه مرقس 9:26–27; الإنجيل كما دونه لوقا 7:15; رؤيا يوحنا 12:12"> وَفِيمَا الْوَلَدُ آتٍ، صَرَعَهُ الشَّيْطَانُ وَخَبَطَهُ بِعُنْفٍ. فَزَجَرَ يَسُوعُ الرُّوحَ النَّجِسَ، وَشَفَى الْوَلَدَ وَسَلَّمَهُ إِلَى أَبِيهِ.</span>
<span className="v" id="v43" data-v="43" data-xref="الإنجيل كما دونه لوقا 9:43–45; الإنجيل كما دونه لوقا 5:26; الإنجيل كما دونه متى 17:22"> فَذُهِلَ الْجَمِيعُ مِنْ عَظَمَةِ اللهِ. يسوع يتنبأ ثانية بموته<br />وَبَيْنَمَا كَانَ الْجَمِيعُ يَتَعَجَّبُونَ مِنْ كُلِّ مَا عَمِلَهُ يَسُوعُ، قَالَ لِتَلامِيذِهِ:</span>
<span className="v" id="v44" data-v="44" data-xref="الإنجيل كما دونه لوقا 9:22; أعمال الرسل 2:23; الإنجيل كما دونه لوقا 2:19"> «لِتَدْخُلْ هذِهِ الْكَلِمَاتُ آذَانَكُمْ: إِنَّ ابْنَ الإِنْسَانِ عَلَى وَشْكِ أَنْ يُسَلَّمَ إِلَى أَيْدِي النَّاسِ!»</span>
<span className="v" id="v45" data-v="45" data-xref="الإنجيل كما دونه لوقا 18:34; الإنجيل كما دونه لوقا 2:50; الإنجيل كما دونه يوحنا 12:16"> إِلّا أَنَّهُمْ لَمْ يَفْهَمُوا هَذَا الْقَوْلَ، وَقَدْ أُغْلِقَ عَلَيْهِمْ فَلَمْ يُدْرِكُوهُ، وَخَافُوا أَنْ يَسْأَلُوهُ عَنْهُ.</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="الإنجيل كما دونه متى 18:1–5; الإنجيل كما دونه مرقس 9:33–37; الإنجيل كما دونه متى 20:20–22"> وَحَدَثَ بَيْنَهُمْ جِدَالٌ حَوْلَ مَنْ هُوَ الأَعْظَمُ فِيهِمْ.</span>
<span className="v" id="v47" data-v="47" data-xref="المزامير 139:23; الإنجيل كما دونه يوحنا 16:30; الإنجيل كما دونه لوقا 5:22"> فَإِذْ عَلِمَ يَسُوعُ نِيَّاتِ قُلُوبِهِمْ، أَخَذَ وَلَداً صَغِيراً وَأَوْقَفَهُ بِجَانِبِهِ،</span>
<span className="v" id="v48" data-v="48" data-xref="رسالة بطرس الأولى 5:6; الإنجيل كما دونه لوقا 14:11; الإنجيل كما دونه مرقس 9:37"> وَقَالَ لَهُمْ: «أَيُّ مَنْ قَبِلَ بِاسْمِي هَذَا الْوَلَدَ الصَّغِيرَ، فَقَدْ قَبِلَنِي؛ وَمَنْ قَبِلَنِي، يَقْبَلُ الَّذِي أَرْسَلَنِي. فَإِنَّ مَنْ كَانَ الأَصْغَرَ بَيْنَكُمْ جَمِيعاً، فَهُوَ الْعَظِيمُ».</span>
</div>

<span className="v" id="v49" data-v="49" data-xref="الإنجيل كما دونه مرقس 9:38–40; العدد 11:27–29; الإنجيل كما دونه مرقس 10:13–14"> وَتَكَلَّمَ يُوحَنَّا فَقَالَ: «يَا سَيِّدُ، رَأَيْنَا وَاحِداً يَطْرُدُ الشَّيَاطِينَ بِاسْمِكَ، فَمَنَعْنَاهُ لأَنَّهُ لَا يَتْبَعُكَ مَعَنَا».</span>

## معارضة السامريين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v51" data-v="51" data-xref="إشعياء 50:5–9; الإنجيل كما دونه مرقس 16:19; الإنجيل كما دونه لوقا 13:22"> وَلَمَّا تَمَّتِ الأَيَّامُ لارْتِفَاعِهِ، صَمَّمَ بِعَزْمٍ عَلَى الْمُضِيِّ إِلَى أُورُشَلِيمَ.</span>
<span className="v" id="v52" data-v="52" data-xref="الإنجيل كما دونه متى 10:5; الإنجيل كما دونه لوقا 17:16; الإنجيل كما دونه لوقا 10:1"> فَأَرْسَلَ قُدَّامَهُ بَعْضَ الرُّسُلِ. فَذَهَبُوا وَدَخَلُوا قَرْيَةً لِلسَّامِرِيِّينَ، لِيُعِدُّوا لَهُ (مَنْزِلاً فِيهَا).</span>
<span className="v" id="v53" data-v="53" data-xref="الإنجيل كما دونه يوحنا 4:9; الإنجيل كما دونه يوحنا 4:40–42; الإنجيل كما دونه لوقا 9:48"> وَلكِنَّهُمْ رَفَضُوا اسْتِقْبَالَهُ لأَنَّهُ كَانَ مُتَّجِهاً صَوْبَ أُورُشَلِيمَ.</span>
<span className="v" id="v54" data-v="54" data-xref="ملوك الثاني 1:10–14; رسالة يعقوب 3:14–18; رسالة يعقوب 1:19–20"> فَلَمَّا رَأَى ذلِكَ تِلْمِيذَاهُ يَعْقُوبُ وَيُوحَنَّا، قَالا: «يَا رَبُّ، أَتُرِيدُ أَنْ نَأْمُرَ بِأَنْ تَنْزِلَ النَّارُ مِنَ السَّمَاءِ وَتَلْتَهِمَهُمْ؟»</span>
<span className="v" id="v55" data-v="55" data-xref="رؤيا يوحنا 3:19; رسالة بطرس الأولى 3:9; إرميا 17:9"> فَالْتَفَتَ إِلَيْهِمَا وَوَبَّخَهُمَا قَائِلاً: «لا تَعْلَمَانِ مَنْ أَيِّ رُوحٍ أَنْتُمَا،</span>
<span className="v" id="v56" data-v="56" data-xref="الإنجيل كما دونه متى 20:28; الرسالة إلى مؤمني روما 12:21; الإنجيل كما دونه يوحنا 3:17"> لأَنَّ ابْنَ الإِنْسَانِ أَتَى لَا لِيُهْلِكَ نُفُوسَ الَّنَاسِ، بَلْ لِيُخَلِّصَهَا.» ثُمَّ ذَهَبُوا إِلَى قَرْيَةٍ أُخْرَى.</span>

## ثمن تبعية يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v57" data-v="57" data-xref="الإنجيل كما دونه متى 8:19–22; الإنجيل كما دونه يوحنا 13:37; الإنجيل كما دونه لوقا 9:57–60"> وَبَيْنَمَا كَانُوا سَائِرِينَ فِي الطَّرِيقِ، قَالَ لَهُ أَحَدُ النَّاسِ: «يَا سَيِّدُ، سَأَتْبَعُكَ أَيْنَمَا تَذْهَبُ!»</span>
<span className="v" id="v58" data-v="58" data-xref="الإنجيل كما دونه متى 8:20; الإنجيل كما دونه لوقا 14:26–33; رسالة يعقوب 2:5"> فَقَالَ لَهُ يَسُوعُ: «لِلثَّعَالِبِ أَوْجَارٌ، وَلِطُيُورِ السَّمَاءِ أَوْكَارٌ؛ وَأَمَّا ابْنُ الإِنْسَانِ فَلَيْسَ لَهُ مَكَانٌ يُسْنِدُ إِلَيْهِ رَأْسَهُ».</span>
