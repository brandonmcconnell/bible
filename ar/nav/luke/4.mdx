## تجربة يسوع في البرية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه متى 4:1–11; أعمال الرسل 10:38; الإنجيل كما دونه مرقس 1:12–13"> أَمَّا يَسُوعُ، فَعَادَ مِنَ الأُرْدُنِّ مُمْتَلِئاً مِنَ الرُّوحِ الْقُدُسِ. فَاقْتَادَهُ الرُّوحُ فِي الْبَرِّيَّةِ</span>
<span className="v" id="v2" data-v="2" data-xref="الخروج 34:28; الرسالة إلى العبرانيين 4:15; التثنية 9:9"> أَرْبَعِينَ يَوْماً، وَإِبْلِيسُ يُجَرِّبُهُ، وَلَمْ يَأْكُلْ شَيْئاً طَوَالَ تِلْكَ الأَيَّامِ. فَلَمَّا تَمَّتْ، جَاعَ.</span>
<span className="v" id="v3" data-v="3" data-xref="الإنجيل كما دونه متى 4:3; الإنجيل كما دونه لوقا 3:22"> فَقَالَ لَهُ إِبْلِيسُ: «إِنْ كُنْتَ ابْنَ اللهِ، فَقُلْ لِهَذَا الْحَجَرِ أَنْ يَتَحَوَّلَ إِلَى خُبْزٍ».</span>
<span className="v" id="v4" data-v="4" data-xref="التثنية 8:3; الإنجيل كما دونه متى 4:4; الإنجيل كما دونه متى 6:31"> فَرَدَّ عَلَيْهِ يَسُوعُ قَاِئلاً: «قَدْ كُتِبَ: لَيْسَ بِالْخُبْزِ وَحْدَهُ يَحْيَا الإِنْسَانُ، (بَلْ بِكُلِّ كَلِمَةٍ تَخْرُجُ مِنَ فَمِ اللهِ)!»</span>
<span className="v" id="v5" data-v="5" data-xref="رسالة يوحنا الأولى 2:15–16; الرسالة الأولى إلى مؤمني كورنثوس 15:52; الإنجيل كما دونه متى 4:8"> ثُمَّ أَصْعَدَهُ إِبْلِيسُ إِلَى جَبَلٍ عَالٍ، وَأَرَاهُ مَمَالِكَ الْعَالَمِ كُلَّهَا فِي لَحْظَةٍ مِنَ الزَّمَنِ،</span>
<span className="v" id="v6" data-v="6" data-xref="الإنجيل كما دونه يوحنا 12:31; رسالة يوحنا الأولى 5:19; الإنجيل كما دونه يوحنا 14:30"> وَقَالَ لَهُ: «أُعْطِيكَ السُّلْطَةَ عَلَى هذِهِ الْمَمَالِكِ كُلِّهَا وَمَا فِيهَا مِنْ عَظَمَةٍ، فَإِنَّهَا قَدْ سُلِّمَتْ إِلَيَّ وَأَنَا أُعْطِيهَا لِمَنْ أَشَاءُ.</span>
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه لوقا 8:28; المزامير 72:11; إشعياء 45:14"> فَإِنْ سَجَدْتَ أَمَامِي، تَصِيرُ كُلُّهَا لَكَ!»</span>
<span className="v" id="v8" data-v="8" data-xref="التثنية 6:13; التثنية 10:20; الإنجيل كما دونه متى 4:10"> فَرَدَّ عَلَيْهِ يَسُوعُ قَائِلاً: «قَدْ كُتِبَ: لِلرَّبِّ إِلهِكَ تَسْجُدُ، وَإِيَّاهُ وَحْدَهُ تَعْبُدُ!»</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه متى 4:5–6; الإنجيل كما دونه لوقا 4:3; الإنجيل كما دونه متى 8:29"> ثُمَّ اقْتَادَهُ إِبْلِيسُ إِلَى أُورُشَلِيمَ، وَأَوْقَفَهُ عَلَى حَافَةِ سَطْحِ الْهَيْكَلِ، وَقَالَ لَهُ: «إِنْ كُنْتَ ابْنَ اللهِ، فَاطْرَحْ نَفْسَكَ مِنْ هُنَا إِلَى الأَسْفَلِ</span>
<span className="v" id="v10" data-v="10" data-xref="المزامير 91:11–12; الرسالة إلى العبرانيين 1:14; الرسالة الثانية إلى مؤمني كورنثوس 11:14"> فَإِنَّهُ قَدْ كُتِبَ: يُوصِي مَلائِكَتَهُ بِكَ لِكَيْ يَحْفَظُوكَ،</span>
<span className="v" id="v11" data-v="11" data-xref="المزامير 91:12"> فَعَلَى أَيْدِيهِمْ يَحْمِلُونَكَ، لِئَلّا تَصْدِمَ قَدَمَكَ بِحَجَرٍ».</span>
<span className="v" id="v12" data-v="12" data-xref="التثنية 6:16; الرسالة إلى العبرانيين 3:8–9; الرسالة الأولى إلى مؤمني كورنثوس 10:9"> فَرَدَّ عَلَيْهِ يَسُوعُ قَائِلاً: «قَدْ قِيلَ: لَا تُجَرِّبِ الرَّبَّ إِلهَكَ!»</span>
<span className="v" id="v13" data-v="13" data-xref="الرسالة إلى العبرانيين 4:15; رسالة يعقوب 4:7; الإنجيل كما دونه متى 4:11"> وَبَعْدَمَا أَكْمَلَ إِبْلِيسُ كُلَّ تَجْرِبَةٍ، انْصَرَفَ عَنْ يَسُوعَ إِلَى حِينٍ.</span>
</div>

## الناصرة ترفض يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه متى 4:23–25; أعمال الرسل 10:37; الإنجيل كما دونه متى 4:12"> وَعَادَ يَسُوعُ إِلَى مِنْطَقَةِ الْجَلِيلِ بِقُدْرَةِ الرُّوحِ؛ وَذَاعَ صِيتُهُ فِي الْقُرَى الْمُجَاوِرَةِ كُلِّهَا.</span>
<span className="v" id="v15" data-v="15" data-xref="الإنجيل كما دونه متى 4:23; الإنجيل كما دونه متى 9:8; الإنجيل كما دونه متى 9:35"> وَكَانَ يُعَلِّمُ فِي مَجَامِعِ الْيَهُودِ، وَالْجَمِيعُ يُمَجِّدُونَهُ.</span>
<span className="v" id="v16" data-v="16" data-xref="أعمال الرسل 17:2; أعمال الرسل 13:14–16; الإنجيل كما دونه متى 2:23"> وَجَاءَ إِلَى النَّاصِرَةِ حَيْثُ كَانَ قَدْ نَشَأَ، وَدَخَلَ الْمَجْمَعَ، كَعَادَتِهِ، يَوْمَ السَّبْتِ، وَوَقَفَ لِيَقْرَأَ.</span>
<span className="v" id="v17" data-v="17" data-xref="إشعياء 61:1–3; أعمال الرسل 13:15; أعمال الرسل 7:42"> فَقُدِّمَ إِلَيْهِ كِتَابُ النَّبِيِّ إِشَعْيَاءَ، فَلَمَّا فَتَحَهُ وَجَدَ الْمَكَانَ الَّذِي كُتِبَ فِيهِ:</span>
<span className="v" id="v18" data-v="18" data-xref="إشعياء 61:1; الإنجيل كما دونه متى 11:5; المزامير 146:7–8"> «رُوحُ الرَّبِّ عَلَيَّ، لأَنَّهُ مَسَحَنِي لأُبَشِّرَ الْفُقَرَاءَ؛ أَرْسَلَنِي لأُنَادِيَ لِلْمَأْسُورِينَ بِالإِطْلاقِ وَلِلْعُمْيَانِ بِالْبَصَرِ، لأُطْلِقَ الْمَسْحُوقِينَ أَحْرَاراً،</span>
<span className="v" id="v19" data-v="19" data-xref="إشعياء 61:2; إشعياء 63:4; الرسالة الثانية إلى مؤمني كورنثوس 6:1"> وَأُبَشِّرَ بِسَنَةِ الْقَبُولِ عِنْدَ الرَّبِّ».</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه لوقا 4:17; الإنجيل كما دونه يوحنا 8:2; الإنجيل كما دونه لوقا 19:48"> ثُمَّ طَوَى الْكِتَابَ وَسَلَّمَهُ إِلَى الْخَادِمِ، وَجَلَسَ. وَكَانَتْ عُيُونُ جَمِيعِ الْحَاضِرِينَ فِي الْمَجْمَعِ شَاخِصَةً إِلَيْهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه يوحنا 4:25–26; أعمال الرسل 3:18; أعمال الرسل 2:16–18"> فَأَخَذَ يُخَاطِبُهُمْ قَائِلاً: «الْيَوْمَ تَمَّ مَا قَدْ سَمِعْتُمْ مِنْ آيَاتٍ.»</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه متى 13:54–56; المزامير 45:2; أعمال الرسل 6:10"> وَشَهِدَ لَهُ جَمِيعُ الْحَاضِرِينَ، مُتَعَجِّبِينَ مِنْ كَلامِ النِّعْمَةِ الْخَارِجِ مِنْ فَمِهِ، وَتَسَاءَلُوا: «أَلَيْسَ هَذَا ابْنَ يُوسُفَ؟»</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه يوحنا 4:46–53; الإنجيل كما دونه متى 11:23–24; الإنجيل كما دونه مرقس 2:1–12"> فَقَالَ لَهُمْ: «لا شَكَّ أَنَّكُمْ تَقُولُونَ لِي هَذَا الْمَثَلَ: أَيُّهَا الطَّبِيبُ اشْفِ نَفْسَكَ! فَاصْنَعْ هُنَا فِي بَلْدَتِكَ مَا سَمِعْنَا أَنَّهُ جَرَى فِي كَفْرَنَاحُومَ.»</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه متى 13:57; الإنجيل كما دونه يوحنا 4:44; الإنجيل كما دونه مرقس 6:4–5"> ثُمَّ أَضَافَ: «الْحَقَّ أَقُولُ لَكُمْ: مَا مِنْ نَبِيٍّ يُقْبَلُ فِي بَلْدَتِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="رسالة يعقوب 5:17; ملوك الأول 17:1; الإنجيل كما دونه متى 20:15"> وَبِالْحَقِيقَةِ أَقُولُ لَكُمْ: كَانَ فِي إِسْرَائِيلَ أَرَامِلُ كَثِيرَاتٌ فِي زَمَانِ إِيلِيَّا، حِينَ أُغْلِقَتِ السَّمَاءُ ثَلاثَ سِنِينَ وَسِتَّةَ أَشْهُرٍ، حَتَّى حَدَثَتْ مَجَاعَةٌ عَظِيمَةٌ فِي الأَرْضِ كُلِّهَا؛</span>
<span className="v" id="v26" data-v="26" data-xref="ملوك الأول 17:9–24; الإنجيل كما دونه متى 11:21; عوبديا 1:20"> وَلَكِنَّ إِيلِيَّا لَمْ يُرْسَلْ إِلَى أَيَّةِ وَاحِدَةٍ مِنْهُنَّ بَلْ إِلَى امْرَأَةٍ أَرْمَلَةٍ فِي صِرْفَةِ صَيْدَا.</span>
<span className="v" id="v27" data-v="27" data-xref="ملوك الثاني 5:1–27; الإنجيل كما دونه يوحنا 17:12; الإنجيل كما دونه متى 12:4"> وَكَانَ فِي إِسْرَائِيلَ، فِي زَمَانِ النَّبِيِّ أَلِيشَعَ، كَثِيرُونَ مُصَابُونَ بِالْبَرَصِ؛ وَلكِنْ لَمْ يُطَهَّرْ أَيُّ وَاحِدٍ مِنْهُمْ، سِوَى نُعْمَانَ السُّورِيِّ!»</span>
<span className="v" id="v28" data-v="28" data-xref="أعمال الرسل 5:33; أخبار الأيام الثاني 24:20–21; إرميا 38:6"> فَامْتَلأَ جَمِيعُ مَنْ فِي الْمَجْمَعِ غَضَباً لَمَّا سَمِعُوا هذِهِ الأُمُورَ،</span>
<span className="v" id="v29" data-v="29" data-xref="الإنجيل كما دونه يوحنا 8:59; أعمال الرسل 21:28–32; العدد 15:35"> وَقَامُوا يَدْفَعُونَهُ إِلَى خَارِجِ الْمَدِينَةِ وَسَاقُوهُ إِلَى حَافَةِ الْجَبَلِ الَّذِي بُنِيَتْ علَيْهِ مَدِينَتُهُمْ لِيَطْرَحُوهُ إِلَى الأَسْفَلِ.</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه يوحنا 10:39; الإنجيل كما دونه يوحنا 8:59; الإنجيل كما دونه يوحنا 18:6–7"> إِلّا أَنَّهُ اجْتَازَ مِنْ وَسْطِهِمْ، وَانْصَرَفَ.</span>
</div>

## يسوع يخرج روحاً نجساً

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الإنجيل كما دونه مرقس 1:21–28; الإنجيل كما دونه متى 4:13; أعمال الرسل 13:50–2"> وَنَزَلَ إِلَى كَفْرَنَاحُومَ، وَهِيَ مَدِينَةٌ بِمِنْطَقَةِ الْجَلِيلِ، وَأَخَذَ يُعَلِّمُ الشَّعْبَ أَيَّامَ السَّبْتِ.</span>
<span className="v" id="v32" data-v="32" data-xref="الإنجيل كما دونه لوقا 4:36; الإنجيل كما دونه متى 7:28–29; الإنجيل كما دونه يوحنا 6:63"> فَذُهِلُوا مِنْ تَعْلِيمِهِ، لأَنَّ كَلِمَتَهُ كَانَتْ ذَاتَ سُلْطَةٍ.</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه مرقس 1:23"> وَكَانَ فِي الْمَجْمَعِ رَجُلٌ يَسْكُنُهُ رُوحُ شَيْطَانٍ نَجِسٍ. فَصَرَخَ بِصَوْتٍ عَالٍ:</span>
<span className="v" id="v34" data-v="34" data-xref="الإنجيل كما دونه مرقس 1:24; رسالة يعقوب 2:19; الإنجيل كما دونه متى 8:29"> «آهِ! مَا شَأْنُكَ بِنَا يَا يَسُوعُ النَّاصِرِيُّ؟ أَجِئْتَ لِتُهْلِكَنَا؟ أَنَا أَعْرِفُ مَنْ أَنْتَ: أَنْتَ قُدُّوسُ اللهِ».</span>
<span className="v" id="v35" data-v="35" data-xref="الإنجيل كما دونه لوقا 4:41; الإنجيل كما دونه لوقا 4:39; الإنجيل كما دونه متى 8:26"> فَزَجَرَهُ يَسُوعُ قَائِلاً: «اخْرَسْ، وَاخْرُجْ مِنْهُ». وَإِذْ طَرَحَهُ الشَّيْطَانُ فِي الْوَسَطِ، خَرَجَ مِنْهُ وَلَمْ يُصِبْهُ بِأَذىً.</span>
<span className="v" id="v36" data-v="36" data-xref="الإنجيل كما دونه لوقا 4:32; رسالة بطرس الأولى 3:22; الإنجيل كما دونه متى 12:22–23"> فَاسْتَوْلَتِ الدَّهْشَةُ عَلَى الْجَمِيعِ، وَأَخَذُوا يَتَسَاءَلُونَ فِي مَا بَيْنَهُمْ: «أَيُّ كَلِمَةٍ هِيَ هذِهِ؟ فَإِنَّهُ بِسُلْطَانٍ وَقُدْرَةٍ يَأْمُرُ الأَرْوَاحَ النَّجِسَةَ فَتَخْرُجُ!»</span>
<span className="v" id="v37" data-v="37" data-xref="الإنجيل كما دونه مرقس 1:45; الإنجيل كما دونه لوقا 4:14; إشعياء 52:13"> وَذَاعَ صِيتُهُ فِي كُلِّ مَكَانٍ مِنَ الْمِنْطَقَةِ الْمُجَاوِرَةِ.</span>

## يسوع يشفي كثيرين

<div className="bt" dir="rtl">
