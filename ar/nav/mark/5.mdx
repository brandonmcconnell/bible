## شفاء رجل به شيطان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه لوقا 8:26–39; الإنجيل كما دونه متى 8:28–34; الإنجيل كما دونه مرقس 4:35"> ثُمَّ وَصَلُوا إِلَى الضَّفَّةِ الْمُقَابِلَةِ مِنَ الْبُحَيْرَةِ، إِلَى بَلْدَةِ الْجِرَاسِيِّينَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الإنجيل كما دونه مرقس 1:23; إشعياء 65:4; الإنجيل كما دونه مرقس 5:8"> وَحَالَمَا نَزَلَ مِنَ الْقَارِبِ، لاقَاهُ مِنْ بَيْنِ الْقُبُورِ إِنْسَانٌ يَسْكُنُهُ رُوحٌ نَجِسٌ،</span>
<span className="v" id="v3" data-v="3" data-xref="إشعياء 65:4; الإنجيل كما دونه لوقا 8:29; دانيال 4:32–33"> كَانَ يُقِيمُ فِي الْقُبُورِ. وَلَمْ يَكُنْ أَحَدٌ يَقْدِرُ أَنْ يُقَيِّدَهُ وَلَوْ بِالسَّلاسِلِ.</span>
<span className="v" id="v4" data-v="4" data-xref="رسالة يعقوب 3:7–8"> فَإِنَّهُ كَثِيراً مَا رُبِطَ بِالْقُيُودِ وَالسَّلاسِلِ، فَكَانَ يُقَطِّعُ السَّلاسِلَ وَيُحَطِّمُ الْقُيُودَ، وَلَمْ يَقْدِرْ أَحَدٌ أَنْ يُخْضِعَهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="أيوب 2:7–8; ملوك الأول 18:28; الإنجيل كما دونه يوحنا 8:44"> وَكَانَ فِي الْقُبُورِ وَفِي الْجِبَالِ دَائِماً، لَيْلاً وَنَهَاراً، يَصِيحُ وَيُجَرِّحُ جِسْمَهُ بِالْحِجَارَةِ.</span>
<span className="v" id="v6" data-v="6" data-xref="رسالة يعقوب 2:19; المزامير 66:3; الإنجيل كما دونه لوقا 4:41"> وَلَكِنَّهُ لَمَّا رَأَى يَسُوعَ مِنْ بَعِيدٍ، رَكَضَ وَسَجَدَ لَهُ،</span>
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه متى 8:29; أعمال الرسل 16:17; الإنجيل كما دونه متى 26:63"> وَصَرَخَ بِأَعْلَى صَوْتِهِ: «مَا شَأْنُكَ بِي يَا يَسُوعُ ابْنَ اللهِ الْعَلِيِّ؟ أَسْتَحْلِفُكَ بِاللهِ أَلّا تُعَذِّبَنِي!»</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه مرقس 1:25; الإنجيل كما دونه مرقس 9:25–26; أعمال الرسل 16:18"> فَإِنَّ يَسُوعَ كَانَ قَدْ قَالَ لَهُ: «أَيُّهَا الرُّوحُ النَّجِسُ، اخْرُجْ مِنَ الإِنْسَانِ!»</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه لوقا 8:30; الإنجيل كما دونه مرقس 5:15; الإنجيل كما دونه متى 26:53"> وَسَأَلَهُ يَسُوعُ: «مَا اسْمُكَ؟» فَأَجَابَ: «اسْمِي لَجِيُونُ لأَنَّنَا جَيْشٌ كَبِيرٌ!»</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه مرقس 3:22; الإنجيل كما دونه مرقس 5:13"> وَتَوَسَّلَ إِلَيْهِ بِإِلْحَاحٍ أَلّا يَطْرُدَ الأَرْوَاحَ النَّجِسَةَ إِلَى خَارِجِ تِلْكَ الْمِنْطَقَةِ.</span>
<span className="v" id="v11" data-v="11" data-xref="اللاويين 11:7–8; إشعياء 65:4; إشعياء 66:3"> وَكَانَ هُنَاكَ قَطِيعٌ كَبِيرٌ مِنَ الْخَنَازِيرِ يَرْعَى عِنْدَ الْجَبَلِ،</span>
<span className="v" id="v12" data-v="12" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 2:11; أيوب 1:10–12; الإنجيل كما دونه لوقا 22:31–32"> فَتَوَسَّلَتِ الأَرْوَاحُ النَّجِسَةُ إِلَى يَسُوعَ قَائِلَةً: «أَرْسِلْنَا إِلَى الْخَنَازِيرِ لِنَدْخُلَ فِيهَا!»</span>
<span className="v" id="v13" data-v="13" data-xref="ملوك الأول 22:22; رسالة بطرس الأولى 3:22; أيوب 2:6"> فَأَذِنَ لَهَا بِذَلِكَ. فَخَرَجَتِ الأَرْوَاحُ النَّجِسَةُ وَدَخَلَتْ فِي الْخَنَازِيرِ، فَانْدَفَعَ قَطِيعُ الْخَنَازِيرِ مِنْ عَلَى حَافَةِ الْجَبَلِ إِلَى الْبُحَيْرَةِ، فَغَرِقَ فِيهَا. وَكَانَ عَدَدُهُ نَحْوَ أَلْفَيْنِ.</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه لوقا 8:34; الإنجيل كما دونه متى 8:33"> أَمَّا رُعَاةُ الْخَنَازِيرِ فَهَرَبُوا وَأَذَاعُوا الْخَبَرَ فِي الْمَدِينَةِ وَفِي الْمَزَارِعِ. فَخَرَجَ النَّاسُ لِيَرَوْا مَا قَدْ جَرَى،</span>
<span className="v" id="v15" data-v="15" data-xref="الرسالة إلى مؤمني كولوسي 1:13; الرسالة الثانية إلى تيموثاوس 1:7; صموئيل الأول 6:20–21"> وَجَاءُوا إِلَى يَسُوعَ، فَرَأَوْا الَّذِي كَانَ مَسْكُوناً بِالشَّيَاطِينِ جَالِساً وَلابِساً وَصَحِيحَ الْعَقْلِ، فَاسْتَوْلَى عَلَيْهِمِ الْخَوْفُ.</span>
<span className="v" id="v16" data-v="16"> فَحَدَّثَهُمُ الَّذِينَ رَأَوْا مَا جَرَى بِمَا حَدَثَ لِلْمَجْنُونِ وَلِلْخَنَازِيرِ</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه متى 8:34; الإنجيل كما دونه مرقس 1:24; الإنجيل كما دونه لوقا 8:37"> فَأَخَذُوا يَرْجُونَ مِنْ يَسُوعَ أَنْ يَرْحَلَ عَنْ دِيَارِهِمْ.</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه لوقا 8:38–39; الرسالة إلى مؤمني فيلبي 1:23–24; الإنجيل كما دونه مرقس 5:17"> وَفِيمَا كَانَ يَرْكَبُ الْقَارِبَ، تَوَسَّلَ إِلَيْهِ الإِنْسَانُ الَّذِي كَانَتِ الشَّيَاطِينُ تَسْكُنُهُ أَنْ يُرَافِقَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="المزامير 66:16; الإنجيل كما دونه يوحنا 4:29; إشعياء 38:9–20"> فَلَمْ يَسْمَحْ لَهُ، بَلْ قَالَ لَهُ: «اذْهَبْ إِلَى بَيْتِكَ، وَإِلَى أَهْلِكَ، وَأَخْبِرْهُمْ بِمَا عَمِلَهُ الرَّبُّ بِكَ وَرَحِمَكَ».</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه مرقس 7:31; الإنجيل كما دونه متى 4:25"> فَانْطَلَقَ وَأَخَذَ يُنَادِي فِي الْمُدُنِ الْعَشْرِ بِمَا عَمِلَهُ يَسُوعُ بِهِ. فَتَعَجَّبَ الْجَمِيعُ.</span>
</div>

## طفلة ماتت وامرأة مريضة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه متى 9:1; الإنجيل كما دونه مرقس 4:1; الإنجيل كما دونه لوقا 8:40"> وَلَمَّا عَادَ يَسُوعُ وَعَبَرَ فِي الْقَارِبِ إِلَى الضَّفَّةِ الْمُقَابِلَةِ مِنَ الْبُحَيْرَةِ، اجْتَمَعَ إِلَيْهِ وَهُوَ عِنْدَ الشَّاطِئِ جَمْعٌ كَبِيرٌ.</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه لوقا 13:14; أعمال الرسل 18:8; الإنجيل كما دونه مرقس 5:35–36"> وَإذَا وَاحِدٌ مِنْ رُؤَسَاءِ الْمَجْمَعِ، وَاسْمُهُ يَايِرُسُ، قَدْ جَاءَ إِلَيْهِ. وَمَا إِنْ رَآهُ، حَتَّى ارْتَمَى عِنْدَ قَدَمَيْهِ،</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه لوقا 13:13; الإنجيل كما دونه لوقا 4:40; الإنجيل كما دونه مرقس 16:18"> وَتَوَسَّلَ إِلَيْهِ بِإِلْحَاحٍ، قَائِلاً: «ابْنَتِي الصَّغِيرَةُ مُشْرِفَةٌ عَلَى الْمَوْتِ. فَتَعَالَ وَالْمِسْهَا بِيَدِكَ لِتُشْفَى فَتَحْيَا!»</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه مرقس 5:31; الإنجيل كما دونه مرقس 3:20; الإنجيل كما دونه لوقا 7:6"> فَذَهَبَ مَعَهُ، يَتْبَعُهُ جَمْعٌ كَبِيرٌ وَهُمْ يَزْحَمُونَهُ.</span>
<span className="v" id="v25" data-v="25" data-xref="اللاويين 15:25–27; الإنجيل كما دونه لوقا 8:43–44; الإنجيل كما دونه متى 9:20–22"> وَكَانَتْ هُنَاكَ امْرَأَةٌ مُصَابَةٌ بِنَزِيفٍ دَمَوِيٍّ مُنْذُ اثْنَتَيْ عَشْرَةَ سَنَةً،</span>
<span className="v" id="v26" data-v="26" data-xref="إرميا 30:12–13; إرميا 8:22; المزامير 108:12"> وَقَدْ عَانَتِ الْكَثِيرَ مِنَ الأَلَمِ عَلَى أَيْدِي أَطِبَّاءَ كَثِيرِينَ، وَأَنْفَقَتْ فِي سَبِيلِ عِلاجِهَا كُلَّ مَا تَمْلِكُ، فَلَمْ تَجْنِ أَيَّةَ فَائِدَةٍ، بَلْ بِالأَحْرَى ازْدَادَتْ حَالَتُهَا سُوءاً.</span>
<span className="v" id="v27" data-v="27" data-xref="أعمال الرسل 19:12; الإنجيل كما دونه متى 14:36; أعمال الرسل 5:15"> فَإِذْ كَانَتْ قَدْ سَمِعَتْ عَنْ يَسُوعَ، جَاءَتْ فِي زَحْمَةِ الْجَمْعِ مِنْ خَلْفِهِ وَلَمَسَتْ رِدَاءَهُ،</span>
<span className="v" id="v28" data-v="28"> لأَنَّهَا قَالَتْ: «يَكْفِي أَنْ أَلْمِسَ ثِيَابَهُ لأُشْفَى».</span>
<span className="v" id="v29" data-v="29" data-xref="الإنجيل كما دونه مرقس 5:34; المزامير 30:2; الإنجيل كما دونه مرقس 3:10"> وَفِي الْحَالِ انْقَطَعَ نَزِيفُ دَمِهَا وَأَحَسَّتْ فِي جِسْمِهَا أَنَّهَا شُفِيَتْ مِنْ عِلَّتِهَا.</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه لوقا 6:19; الإنجيل كما دونه لوقا 8:46; الإنجيل كما دونه لوقا 5:17"> وَحَالَمَا شَعَرَ يَسُوعُ فِي نَفْسِهِ بِالْقُوَّةِ الَّتِي خَرَجَتْ مِنْهُ، أَدَارَ نَظَرَهُ فِي الْجَمْعِ وَسَأَلَ: «مَنْ لَمَسَ ثِيَابِي؟»</span>
<span className="v" id="v31" data-v="31" data-xref="الإنجيل كما دونه لوقا 8:45; الإنجيل كما دونه لوقا 9:12"> فَقَالَ لَهُ تَلامِيذُهُ: «أَنْتَ تَرَى الْجَمْعَ يَزْحَمُونَكَ، وَتَسْأَلُ: مَنْ لَمَسَنِي؟»</span>
<span className="v" id="v32" data-v="32"> وَلَكِنَّهُ ظَلَّ يَتَطَلَّعُ حَوْلَهُ لِيَرَى الَّتِي فَعَلَتْ ذَلِكَ.</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه لوقا 8:47; الإنجيل كما دونه لوقا 1:29; الإنجيل كما دونه لوقا 1:12"> فَمَا كَانَ مِنَ الْمَرْأَةِ، وَقَدْ عَلِمَتْ بِمَا حَدَثَ لَهَا، إِلّا أَنْ جَاءَتْ وَهِيَ خَائِفَةٌ تَرْتَجِفُ، وَارْتَمَتْ أَمَامَهُ وَأَخْبَرَتْهُ بِالْحَقِيقَةِ كُلِّهَا.</span>
<span className="v" id="v34" data-v="34" data-xref="الإنجيل كما دونه لوقا 8:48; الإنجيل كما دونه متى 9:22; الإنجيل كما دونه لوقا 7:50"> فَقَالَ لَهَا: «يَا ابْنَةُ، إِيمَانُكِ قَدْ شَفَاكِ. فَاذْهَبِي بِسَلامٍ وَتَعَافَيْ مِنْ عِلَّتِكِ!»</span>
<span className="v" id="v35" data-v="35" data-xref="الإنجيل كما دونه لوقا 8:49; الإنجيل كما دونه يوحنا 11:39; الإنجيل كما دونه مرقس 5:22"> وَبَيْنَمَا يَسُوعُ يَتَكَلَّمُ، جَاءَ بَعْضُهُمْ مِنْ بَيْتِ رَئِيسِ الْمَجْمَعِ قَائِلِينَ: «ابْنَتُكَ قَدْ مَاتَتْ. فَلِمَاذَا تُكَلِّفُ الْمُعَلِّمَ بَعْدُ؟»</span>
<span className="v" id="v36" data-v="36" data-xref="الإنجيل كما دونه لوقا 8:50; الإنجيل كما دونه متى 9:28–29; الإنجيل كما دونه يوحنا 11:40"> وَلَكِنَّ يَسُوعَ، مَا إِنْ سَمِعَ بِذَلِكَ الْخَبَرِ، حَتَّى قَالَ لِرَئِيسِ الْمَجْمَعِ: «لا تَخَفْ؛ آمِنْ فَقَطْ!»</span>
<span className="v" id="v37" data-v="37" data-xref="أعمال الرسل 9:40; الإنجيل كما دونه مرقس 14:33; الرسالة الثانية إلى مؤمني كورنثوس 13:1"> وَلَمْ يَدَعْ أَحَداً يُرَافِقُهُ إِلّا بُطْرُسَ وَيَعْقُوبَ وَيُوحَنَّا أَخَا يَعْقُوبَ.</span>
<span className="v" id="v38" data-v="38" data-xref="الإنجيل كما دونه متى 9:23–24; الإنجيل كما دونه لوقا 8:52–53; الإنجيل كما دونه مرقس 5:22"> وَوَصَلَ إِلَى بَيْتِ رَئِيسِ الْمَجْمَعِ، فَرَأَى الضَّجِيجَ وَالنَّاسَ يَبْكُونَ وَيُوَلْوِلُونَ كَثِيراً.</span>
<span className="v" id="v39" data-v="39" data-xref="أعمال الرسل 20:10; الرسالة الأولى إلى مؤمني تسالونيكي 5:10; الرسالة الأولى إلى مؤمني تسالونيكي 4:13–14"> فَلَمَّا دَخَلَ، قَالَ لَهُمْ: «لِمَاذَا تَضِجُّونَ وَتَبْكُونَ؟ لَمْ تَمُتِ الصَّبِيَّةُ، بَلْ هِيَ نَائِمَةٌ».</span>
<span className="v" id="v40" data-v="40" data-xref="نحميا 2:19; ملوك الثاني 4:33; الإنجيل كما دونه متى 9:24–25"> فَضَحِكُوا مِنْهُ. أَمَّا هُوَ، فَأَخْرَجَهُمْ جَمِيعاً، ثُمَّ أَخَذَ مَعَهُ أَبَا الصَّبِيَّةِ وَأُمَّهَا وَالَّذِينَ كَانُوا يُرَافِقُونَهُ، وَدَخَلَ حَيْثُ كَانَتِ الصَّبِيَّةُ.</span>
<span className="v" id="v41" data-v="41" data-xref="الإنجيل كما دونه مرقس 1:31; الإنجيل كما دونه لوقا 7:14–15; الإنجيل كما دونه لوقا 8:54–55"> وَإِذْ أَمْسَكَ بِيَدِهَا قَالَ: «طَلِيثَا قُومِي!» أَيْ: «يَا صَبِيَّةُ، لَكِ أَقُولُ: قُومِي».</span>
<span className="v" id="v42" data-v="42" data-xref="الإنجيل كما دونه مرقس 4:41; أعمال الرسل 3:10–13; الإنجيل كما دونه مرقس 7:37"> فَنَهَضَتِ الصَّبِيَّةُ حَالاً وَأَخَذَتْ تَمْشِي، إِذْ كَانَ عُمْرُهَا اثْنَتَيْ عَشْرَةَ سَنَةً. فَدُهِشَ الْجَمِيعُ دَهْشَةً عَظِيمَةً.</span>
<span className="v" id="v43" data-v="43" data-xref="الإنجيل كما دونه متى 8:4; الإنجيل كما دونه متى 9:30; الإنجيل كما دونه مرقس 1:43"> فَأَمَرَهُمْ بِشِدَّةٍ أَنْ لَا يَعْلَمَ أَحَدٌ بِذَلِكَ، وَطَلَبَ أَنْ تُعْطَى طَعَاماً لِتَأْكُلَ.</span>
