## يوحنا المعمدان يعد الطريق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="رسالة يوحنا الأولى 5:11–12; الإنجيل كما دونه متى 4:3; الإنجيل كما دونه متى 3:17"> هَذِهِ بِدَايَةُ إِنْجِيلِ يَسُوعَ الْمَسِيحِ ابْنِ اللهِ:</span>
<span className="v" id="v2" data-v="2" data-xref="ملاخي 3:1; الإنجيل كما دونه متى 11:10; الإنجيل كما دونه لوقا 1:76"> كَمَا كُتِبَ فِي كِتَابِ إِشَعْيَاءَ: «هَا أَنَا أُرْسِلُ قُدَّامَكَ مَلاكِي الَّذِي يُعِدُّ لَكَ الطَّرِيقَ؛</span>
<span className="v" id="v3" data-v="3" data-xref="إشعياء 40:3–5; الإنجيل كما دونه متى 3:3; الإنجيل كما دونه لوقا 3:4–6"> صَوْتُ مُنَادٍ فِي الْبَرِّيَّةِ: أَعِدُّوا طَرِيقَ الرَّبِّ، وَاجْعَلُوا سُبُلَهُ مُسْتَقِيمَةً!»</span>
<span className="v" id="v4" data-v="4" data-xref="أعمال الرسل 19:3–4; الإنجيل كما دونه متى 3:6; أعمال الرسل 13:24–25"> فَقَدْ ظَهَرَ يُوحَنَّا الْمَعْمَدَانُ فِي الْبَرِّيَّةِ يُنَادِي بِمَعْمُودِيَّةِ التَّوْبَةِ لِمَغْفِرَةِ الْخَطَايَا.</span>
<span className="v" id="v5" data-v="5" data-xref="اللاويين 26:40–42; أعمال الرسل 19:18; الإنجيل كما دونه متى 3:5–6"> وَخَرَجَ إِلَيْهِ أَهْلُ مِنْطَقَةِ الْيَهُودِيَّةِ وَأَهْلُ أُورُشَلِيمَ جَمِيعاً، فَكَانُوا يَتَعَمَّدُونَ عَلَى يَدِهِ فِي نَهْرِ الأُرْدُنِّ مُعْتَرِفِينَ بِخَطَايَاهُمْ.</span>
<span className="v" id="v6" data-v="6" data-xref="ملوك الثاني 1:8; اللاويين 11:22; الإنجيل كما دونه متى 3:4"> وَكَانَ يُوحَنَّا يَلْبَسُ ثَوْباً مِنْ وَبَرِ الْجِمَالِ، وَيَلُفُّ وَسَطَهُ بِحِزَامٍ مِنْ جِلْدٍ، وَيَأْكُلُ الْجَرَادَ وَالْعَسَلَ الْبَرِّيَّ.</span>
<span className="v" id="v7" data-v="7" data-xref="أعمال الرسل 13:25; الإنجيل كما دونه لوقا 3:16; الإنجيل كما دونه يوحنا 1:27"> وَكَانَ يَعِظُ قَائِلاً: «سَيَأْتِي بَعْدِي مَنْ هُوَ أَقْدَرُ مِنِّي، مَنْ لَا أَسْتَحِقُّ أَنْ أَنْحَنِيَ لأَحُلَّ رِبَاطَ حِذَائِهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="أعمال الرسل 1:5; الرسالة الأولى إلى مؤمني كورنثوس 12:13; أعمال الرسل 19:4–6"> أَنَا عَمَّدْتُكُمْ بِالْمَاءِ؛ أَمَّا هُوَ فَسَوْفَ يُعَمِّدُكُمْ بِالرُّوحِ الْقُدُسِ».</span>
</div>

## معمودية يسوع وتجربة الشيطان له

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه متى 3:13–17; الإنجيل كما دونه متى 2:23; الإنجيل كما دونه لوقا 3:21–22"> فِي تِلْكَ الأَيَّامِ جَاءَ يَسُوعُ مِنَ النَّاصِرَةِ بِمِنْطَقَةِ الْجَلِيلِ، وَتَعَمَّدَ فِي نَهْرِ الأُرْدُنِّ عَلَى يَدِ يُوحَنَّا.</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه يوحنا 1:31–34; الإنجيل كما دونه لوقا 3:22; إشعياء 42:1"> وَبَمُجَرَّدِ أَنْ صَعِدَ مِنَ الْمَاءِ، رَأَى السَّمَاوَاتِ قَدِ انْفَتَحَتْ، وَالرُّوحَ القُدُسَ هَابِطاً عَلَيْهِ كَأَنَّهُ حَمَامَةٌ،</span>
<span className="v" id="v11" data-v="11" data-xref="إشعياء 42:1; الإنجيل كما دونه مرقس 9:7; الإنجيل كما دونه متى 3:17"> وَإذَا صَوْتٌ مِنَ السَّمَاوَاتِ يَقُولُ: «أَنْتَ ابْنِي الْحَبِيبُ، بِكَ سُرِرْتُ كُلَّ سُرُورٍ!»</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه متى 4:1–11; الإنجيل كما دونه لوقا 4:1–13"> وَفِي الْحَالِ اقْتَادَ الرُّوحُ يَسُوعَ إِلَى الْبَرِّيَّةِ،</span>
<span className="v" id="v13" data-v="13" data-xref="الرسالة إلى العبرانيين 2:17–18; الرسالة إلى العبرانيين 4:15; الخروج 34:28"> فَقَضَى فِيهَا أَرْبَعِينَ يَوْماً وَالشَّيْطَانُ يُجَرِّبُهُ. وَكَانَ بَيْنَ الْوُحُوشِ وَمَلائِكَةٌ تَخْدِمُهُ.</span>
</div>
</div>

</div>

## دعوة التلاميذ الأولين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه متى 4:18–22; الإنجيل كما دونه يوحنا 1:40–42; الإنجيل كما دونه يوحنا 6:8"> وَفِيمَا كَانَ يَسُوعُ يَمْشِي عَلَى شَاطِئِ بُحَيْرَةِ الْجَلِيلِ، رَأَى سِمْعَانَ وَأَخَاهُ أَنْدَرَاوُسَ يُلْقِيَانِ الشَّبَكَةَ فِي الْبُحَيْرَةِ، فَقَدْ كَانَا صَيَّادَيْنِ.</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه لوقا 5:10; الإنجيل كما دونه متى 4:19–20; أعمال الرسل 2:38–41"> فَقَالَ لَهُمَا يَسُوعُ: «هَيَّا اتْبَعَانِي، فَأَجْعَلَكُمَا صَيَّادَيْنِ لِلنَّاسِ!»</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه لوقا 18:28–30; الإنجيل كما دونه لوقا 5:11; الإنجيل كما دونه مرقس 10:28–31"> فَتَرَكَا شِبَاكَهُمَا وَتَبِعَاهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه مرقس 5:37; أعمال الرسل 1:13; الإنجيل كما دونه مرقس 10:35"> ثُمَّ سَارَ مِنْ هُنَاكَ قَلِيلاً، فَرَأَى يَعْقُوبَ بْنَ زَبَدِي وَيُوحَنَّا أَخَاهُ فِي الْقَارِبِ يُصْلِحَانِ الشِّبَاكَ،</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه متى 10:37; الإنجيل كما دونه لوقا 14:26; الإنجيل كما دونه مرقس 10:29"> فَدَعَاهُمَا فِي الْحَالِ لِيَتْبَعَاهُ، فَتَرَكَا أَبَاهُمَا زَبَدِي فِي الْقَارِبِ مَعَ الأُجَرَاءِ، وَتَبِعَاهُ.</span>
</div>
</div>

## يسوع يطرد روحاً نجساً

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه مرقس 6:2; الإنجيل كما دونه مرقس 1:39; الإنجيل كما دونه متى 4:23"> ثُمَّ ذَهَبُوا إِلَى كَفْرَنَاحُومَ. فَدَخَلَ حَالاً، فِي يَوْمِ السَّبْتِ، إِلَى الْمَجْمَعِ وَأَخَذَ يُعَلِّمُ.</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه متى 7:28–29; الإنجيل كما دونه متى 13:54; الإنجيل كما دونه لوقا 4:32"> فَذُهِلَ الْحَاضِرُونَ مِنْ تَعْلِيمِهِ، لأَنَّهُ كَانَ يُعَلِّمُهُمْ كَصَاحِبِ سُلْطَانٍ وَلَيْسَ كَالْكَتَبَةِ.</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه مرقس 9:25; الإنجيل كما دونه مرقس 7:25; الإنجيل كما دونه مرقس 1:34"> وَكَانَ فِي مَجْمَعِهِمْ رَجُلٌ يَسْكُنُهُ رُوحٌ نَجِسٌ، فَصَرَخَ</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه متى 8:29; رسالة يعقوب 2:19; أعمال الرسل 3:14"> وَقَالَ: «مَا شَأْنُكَ بِنَا يَا يَسُوعُ النَّاصِرِيُّ؟ أَجِئْتَ لِتُهْلِكَنَا؟ أَنَا أَعْرِفُ مَنْ أَنْتَ. أَنْتَ قُدُّوسُ اللهِ!»</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه مرقس 1:34; الإنجيل كما دونه لوقا 4:41; أعمال الرسل 16:17"> فَزَجَرَهُ يَسُوعُ قَائِلاً: «اخْرَسْ وَاخْرُجْ مِنْهُ!»</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه مرقس 9:26; الإنجيل كما دونه مرقس 9:20; الإنجيل كما دونه لوقا 11:22"> فَطَرَحَ الرُّوحُ النَّجِسُ الرَّجُلَ، وَصَرَخَ صَرْخَةً عَالِيَةً، وَخَرَجَ مِنْهُ.</span>
<span className="v" id="v27" data-v="27" data-xref="الإنجيل كما دونه متى 15:31; الإنجيل كما دونه مرقس 7:37; الإنجيل كما دونه لوقا 9:1"> فَدُهِشَ الْجَمِيعُ حَتَّى أَخَذُوا يَتَسَاءَلُونَ فِيمَا بَيْنَهُمْ: «مَا هَذَا؟ إِنَّهُ تَعْلِيمٌ جَدِيدٌ، يُلْقَى بسُلْطَانٍ، فَحَتَّى الأَرْوَاحُ النَّجِسَةُ يَأْمُرُهَا فَتُطِيعُهُ!»</span>
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه متى 9:31; الإنجيل كما دونه لوقا 4:37; الإنجيل كما دونه مرقس 1:45"> وَفِي الْحَالِ انْتَشَرَ خَبَرُ يَسُوعَ فِي كُلِّ مَكَانٍ مِنَ الْمِنْطَقَةِ الْمُجَاوِرَةِ لِلْجَلِيلِ.</span>

## يسوع يشفي كثيرين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="الإنجيل كما دونه لوقا 4:38–39; الإنجيل كما دونه متى 8:14–15; الإنجيل كما دونه مرقس 1:21"> وَحَالَمَا غَادَرُوا الْمَجْمَعَ، دَخَلُوا بَيْتَ سِمْعَانَ وَأَنْدَرَاوُسَ، وَمَعَهُمْ يَعْقُوبُ وَيُوحَنَّا.</span>
<span className="v" id="v30" data-v="30" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 9:5; الإنجيل كما دونه مرقس 5:23; الإنجيل كما دونه يوحنا 11:3"> وَكَانَتْ حَمَاةُ سِمْعَانَ طَرِيحَةَ الْفِرَاشِ، تُعَانِي مِنَ الْحُمَّى. فَفِي الْحَالِ كَلَّمُوا يَسُوعَ بِشَأْنِهَا.</span>
<span className="v" id="v31" data-v="31" data-xref="أعمال الرسل 9:41; المزامير 116:12; الإنجيل كما دونه مرقس 5:41"> فَاقْتَرَبَ إِلَيْهَا، وَأَمْسَكَ بِيَدِهَا وَأَنْهَضَهَا. فَذَهَبَتْ عَنْهَا الْحُمَّى حَالاً، وَقَامَتْ تَخْدِمُهُمْ.</span>
<span className="v" id="v32" data-v="32" data-xref="الإنجيل كما دونه متى 8:16; الإنجيل كما دونه مرقس 3:2; الإنجيل كما دونه لوقا 4:40"> وَعِنْدَ حُلُولِ الْمَسَاءِ، لَمَّا غَرَبَتِ الشَّمْسُ، أَحْضَرَ النَّاسُ إِلَيْهِ جَمِيعَ مَنْ كَانُوا مَرْضَى وَمَسْكُونِينَ بِالشَّيَاطِينِ،</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه مرقس 1:5; أعمال الرسل 13:44"> حَتَّى احْتَشَدَ أَهْلُ الْمَدِينَةِ كُلُّهُمْ عِنْدَ البَابِ.</span>
<span className="v" id="v34" data-v="34" data-xref="الإنجيل كما دونه لوقا 4:41; الإنجيل كما دونه مرقس 3:12; الإنجيل كما دونه متى 4:23"> فَشَفَى كَثِيرِينَ كَانُوا يُعَانُونَ مِنْ أَمْرَاضٍ مُخْتَلِفَةٍ، وَطَرَدَ شَيَاطِينَ كَثِيرَةً، وَلكِنَّهُ لَمْ يَسْمَحْ لِلشَّيَاطِينِ بِأَنْ يَتَكَلَّمُوا، لأَنَّهُمْ عَرَفُوا مَنْ هُوَ.</span>
</div>

## يسوع يصلي في مكان منعزل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="الإنجيل كما دونه لوقا 6:12; الإنجيل كما دونه لوقا 5:16; الإنجيل كما دونه لوقا 22:39–46"> وَفِي الْيَوْمِ التَّالِي، نَهَضَ بَاكِراً قَبْلَ الْفَجْرِ، وَخَرَجَ إِلَى مَكَانٍ مُنْعَزِلٍ وَأَخَذَ يُصَلِّي هُنَاكَ.</span>
<span className="v" id="v36" data-v="36"> فَذَهَبَ سِمْعَانُ وَمَنْ مَعَهُ يَبْحَثُونَ عَنْهُ. فَلَمَّا وَجَدُوهُ قَالُوا لَهُ:</span>

<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="الإنجيل كما دونه لوقا 5:12–14; الإنجيل كما دونه متى 8:2–4; اللاويين 13:1–14"> وَجَاءَهُ رَجُلٌ مُصَابٌ بِالْبَرَصِ يَتَوَسَّلُ إِلَيْهِ. فَارْتَمَى عَلَى رُكْبَتَيْهِ أَمَامَهُ وَقَالَ: «إِنْ أَرَدْتَ، فَأَنْتَ تَقْدِرُ أَنْ تُطَهِّرَنِي!»</span>
