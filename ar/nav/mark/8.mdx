## يسوع يُطعم أربعة آلاف

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه متى 15:32–39; الإنجيل كما دونه مرقس 6:34–44"> فِي تِلْكَ الأَيَّامِ، إِذِ احْتَشَدَ أَيْضاً جَمْعٌ كَبِيرٌ، وَلَمْ يَكُنْ عِنْدَهُمْ مَا يَأْكُلُونَ، دَعَا يَسُوعُ تَلامِيذَهُ إِلَيْهِ وَقَالَ لَهُمْ:</span>
<span className="v" id="v2" data-v="2" data-xref="الإنجيل كما دونه متى 9:36; ميخا 7:19; المزامير 103:13"> «إِنِّي أُشْفِقُ عَلَى الْجَمْعِ لأَنَّهُمْ مَازَالُوا مَعِي مُنْذُ ثَلاثَةِ أَيَّامٍ، وَلَيْسَ عِنْدَهُمْ مَا يَأْكُلُونَ.</span>
<span className="v" id="v3" data-v="3" data-xref="إشعياء 40:31; القضاة 8:4–6; صموئيل الأول 30:10–12"> وَإِنْ صَرَفْتُهُمْ إِلَى بُيُوتِهِمْ صَائِمِينَ، تَخُورُ قُوَاهُمْ فِي الطَّرِيقِ، لأَنَّ بَعْضاً مِنْهُمْ جَاءُوا مِنْ أَمَاكِنَ بَعِيدَةٍ».</span>
<span className="v" id="v4" data-v="4" data-xref="ملوك الثاني 4:42–44; ملوك الثاني 7:2; الإنجيل كما دونه مرقس 6:52"> فَقَالَ لَهُ التَّلامِيذُ: «مِنْ أَيْنَ يَسْتَطِيعُ أَحَدٌ أَنْ يُشْبِعَ هؤُلاءِ خُبْزاً هُنَا فِي هَذَا الْمَكَانِ الْمُقْفِرِ؟»</span>
<span className="v" id="v5" data-v="5" data-xref="الإنجيل كما دونه متى 14:15–17; الإنجيل كما دونه متى 15:34; الإنجيل كما دونه لوقا 9:13"> فَسَأَلَهُمْ: «كَمْ رَغِيفاً عِنْدَكُمْ؟» أَجَابُوا: «سَبْعَةٌ!»</span>
<span className="v" id="v6" data-v="6" data-xref="الرسالة إلى مؤمني كولوسي 3:17; الإنجيل كما دونه متى 15:35–36; الإنجيل كما دونه يوحنا 6:10–11"> فَأَمَرَ الْجَمْعَ أَنْ يَجْلِسُوا عَلَى الأَرْضِ. ثُمَّ أَخَذَ الأَرْغِفَةَ السَّبْعَةَ، وَشَكَرَ، وَأَعْطَى تَلامِيذَهُ لِيُقَدِّمُوا لِلْجَمْعِ، فَفَعَلُوا</span>
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه متى 14:19; الإنجيل كما دونه يوحنا 21:5; الإنجيل كما دونه لوقا 24:41–42"> وَكَانَ مَعَهُمْ أَيْضاً بَعْضُ سَمَكَاتٍ صِغَارٍ، فَبَارَكَهَا وَأَمَرَ بِتَقْدِيمِهَا أَيْضاً إِلَى الْجَمْعِ.</span>
<span className="v" id="v8" data-v="8" data-xref="ملوك الثاني 4:42–44; المزامير 145:16; الإنجيل كما دونه متى 16:10"> فَأَكَلَ الْجَمِيعُ حَتَّى شَبِعُوا. ثُمَّ رَفَعَ التَّلامِيذُ سَبْعَةَ سِلالٍ مَلأُوهَا بِمَا فَضَلَ مِنَ الْكِسَرِ.</span>
<span className="v" id="v9" data-v="9"> وَكَانَ الآكِلُونَ نَحْوَ أَرْبَعَةِ آلافٍ. ثُمَّ صَرَفَهُمْ،</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه متى 15:39"> وَفِي الْحَالِ رَكِبَ الْقَارِبَ مَعَ تَلامِيذِهِ، وَجَاءَ إِلَى نَوَاحِي دَلْمَانُوثَةَ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 10:9; الإنجيل كما دونه متى 12:38; التثنية 6:16"> فَأَقْبَلَ الْفَرِّيسِيُّونَ وَأَخَذُوا يُجَادِلُونَهُ، طَالِبِينَ مِنْهُ مُعْجِزَةً مِنَ السَّمَاءِ لِيَمْتَحِنُوهُ.</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه متى 12:39–40; الإنجيل كما دونه مرقس 7:34; الإنجيل كما دونه متى 16:4"> فَتَنَهَّدَ مُتَضَايِقاً، وَقَالَ: «لِمَاذَا يَطْلُبُ هَذَا الْجِيلُ آيَةً؟ الْحَقَّ أَقُولُ لَكُمْ: لَنْ يُعْطَى هَذَا الْجِيلُ آيَةً!»</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه يوحنا 12:36; أعمال الرسل 13:45–46; المزامير 81:12"> ثُمَّ تَرَكَهُمْ وَعَادَ فَرَكِبَ الْقَارِبَ وَعَبَرَ إِلَى الضَّفَّةِ الْمُقَابِلَةِ.</span>
</div>
</div>

## خمير الفريسيين وهيرودس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه متى 16:5"> وَكَانُوا قَدْ نَسُوا أَنْ يَتَزَوَّدُوا خُبْزاً، وَلَمْ يَكُنْ مَعَهُمْ فِي الْقَارِبِ إِلّا رَغِيفٌ وَاحِدٌ.</span>
<span className="v" id="v15" data-v="15" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 5:6–8; الإنجيل كما دونه متى 16:6; الإنجيل كما دونه متى 16:11–12"> وَأَوْصَاهُمْ قَائِلاً: «انْتَبِهُوا! خُذُوا حِذْرَكُمْ مِنْ خَمِيرِ الْفَرِّيسِيِّينَ وَخَمِيرِ هِيرُودُسَ».</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه لوقا 9:46; الإنجيل كما دونه لوقا 20:5; الإنجيل كما دونه متى 16:7–8"> فَأَخَذُوا يُجَادِلُونَ بَعْضُهُمْ بَعْضاً، قَائِلِينَ: «ذلِكَ لأَنَّهُ لَيْسَ عِنْدَنَا خُبْزٌ».</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه مرقس 6:52; إشعياء 63:17; الإنجيل كما دونه مرقس 2:8"> فَعَلِمَ يَسُوعُ بِذلِكَ، وَقَالَ لَهُمْ: «لِمَاذَا يُجَادِلُ بَعْضُكُمْ بَعْضاً لأَنَّهُ لَيْسَ عِنْدَكُمْ خُبْزٌ؟ أَلا تُدْرِكُونَ بَعْدُ وَلا تَفْهَمُونَ؟ أَمَا زَالَتْ قُلُوبُكُمْ مُتَقَسِّيَةً؟</span>
<span className="v" id="v18" data-v="18" data-xref="أعمال الرسل 28:26–27; الإنجيل كما دونه متى 13:14–15; إرميا 5:21"> لَكُمْ عُيُونٌ، أَلا تُبْصِرُونَ؟ لَكُمْ آذَانٌ، أَلا تَسْمَعُونَ؟ أَوَلَسْتُمْ تَذْكُرُونَ؟</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه لوقا 9:12–17; الإنجيل كما دونه متى 14:17–21; الإنجيل كما دونه يوحنا 6:5–13"> عِنْدَمَا كَسَّرْتُ الأَرْغِفَةَ الْخَمْسَةَ لِلْخَمْسَةِ الآلافِ، كَمْ قُفَّةً مَلأَى بِالْكِسَرِ رَفَعْتُمْ؟» قَالُوا لَهُ: «اثْنَتَيْ عَشْرَةَ».</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه متى 15:34–38; الإنجيل كما دونه مرقس 8:1–9"> «وَعِنْدَمَا كَسَّرْتُ الأَرْغِفَةَ السَّبْعَةَ لِلأَرْبَعَةِ الآلافِ، كَمْ سَلًّا مَلِيئاً بِالْكِسَرِ رَفَعْتُمْ؟» قَالُوا: «سَبْعَةً!»</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه مرقس 6:52; الإنجيل كما دونه متى 16:11–12; الإنجيل كما دونه مرقس 8:12"> فَقَالَ لَهُمْ: «وَكَيْفَ لَا تَفْهَمُونَ بَعْدُ؟»</span>
</div>
</div>

## شفاء أعمى في بيت صيدا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه متى 11:21; الإنجيل كما دونه مرقس 6:45; الإنجيل كما دونه لوقا 10:13"> وَجَاءُوا إِلَى بَلْدَةِ بَيْتِ صَيْدَا، فَأَحْضَرَ بَعْضُهُمْ إِلَيْهِ أَعْمَى وَتَوَسَّلُوا إِلَيْهِ أَنْ يَضَعَ يَدَهُ عَلَيْهِ.</span>
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه مرقس 7:33; الإنجيل كما دونه يوحنا 9:6–7; الإنجيل كما دونه مرقس 5:23"> فَأَمْسَكَ بِيَدِ الأَعْمَى وَاقْتَادَهُ إِلَى خَارِجِ الْقَرْيَةِ، وَبَعْدَمَا تَفَلَ عَلَى عَيْنَيْهِ، وَضَعَ يَدَيْهِ عَلَيْهِ وَسَأَلَهُ: «هَلْ تَرَى شَيْئاً؟»</span>
<span className="v" id="v24" data-v="24" data-xref="إشعياء 32:3; إشعياء 29:18; الرسالة الأولى إلى مؤمني كورنثوس 13:9–12"> فَتَطَلَّعَ، وَقَالَ: «أَرَى أُنَاساً، كَأَنَّهُمْ أَشْجَارٌ، يَمْشُونَ».</span>
<span className="v" id="v25" data-v="25" data-xref="رسالة بطرس الأولى 2:9; رسالة بطرس الثانية 3:18; الأمثال 4:18"> فَوَضَعَ يَدَيْهِ ثَانِيَةً عَلَى عَيْنَيْهِ، فَتَطَلَّعَ بِانْتِبَاهٍ، وَعَادَ صَحِيحاً يَرَى كُلَّ شَيْءٍ وَاضِحاً.</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه متى 8:4; الإنجيل كما دونه مرقس 8:23; الإنجيل كما دونه مرقس 5:43"> فَأَرْسَلَهُ إِلَى بَيْتِهِ قَائِلاً: «لا تَدْخُلْ وَلا إِلَى الْقَرْيَةِ!»</span>
</div>
</div>

## بطرس يشهد بحقيقة المسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الإنجيل كما دونه متى 16:13–20; الإنجيل كما دونه لوقا 9:18–20"> ثُمَّ تَوَجَّهَ يَسُوعُ وَتَلامِيذُهُ إِلَى قُرَى قَيْصَرِيَّةِ فِيلِبُّسَ. وَفِي الطَّرِيقِ، سَأَلَ تَلامِيذَهُ: «مَنْ يَقُولُ النَّاسُ إِنِّي أَنَا؟»</span>
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه يوحنا 1:21; الإنجيل كما دونه متى 14:2; ملاخي 4:5"> فَأَجَابُوهُ: «(يَقُولُ بَعْضُهُمْ) إِنَّكَ يُوحَنَّا الْمَعْمَدَانُ، وَغَيْرُهُمْ إِنَّكَ إِيلِيَّا، وَآخَرُونَ إِنَّكَ وَاحِدٌ مِنَ الأَنْبِيَاءِ».</span>
<span className="v" id="v29" data-v="29" data-xref="رسالة يوحنا الأولى 4:15; الإنجيل كما دونه متى 16:15–16; رسالة يوحنا الأولى 5:1"> فَسَأَلَهُمْ: «وَأَنْتُمْ، مَنْ تَقُولُونَ إِنِّي أَنَا؟» فَأَجَابَهُ بُطْرُسُ: «أَنْتَ الْمَسِيحُ!»</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه لوقا 9:21; الإنجيل كما دونه متى 16:20; الإنجيل كما دونه مرقس 9:9"> فَحَذَّرَهُمْ مِنْ أَنْ يُخْبِرُوا أَحَداً بِأَمْرِهِ.</span>
</div>
</div>

## يسوع يُعلِنُ عن موته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الإنجيل كما دونه مرقس 10:33–34; الإنجيل كما دونه يوحنا 2:19; الإنجيل كما دونه لوقا 24:26"> وَأَخَذَ يُعَلِّمُهُمْ أَنَّ ابْنَ الإِنْسَانِ لابُدَّ أَنْ يَتَأَلَّمَ كَثِيراً، وَيَرْفُضَهُ الشُّيُوخُ وَرُؤَسَاءُ الْكَهَنَةِ وَالْكَتَبَةُ، وَيُقْتَلَ، وَبَعْدَ ثَلاثَةِ أَيَّامٍ يَقُومُ.</span>
<span className="v" id="v32" data-v="32" data-xref="الإنجيل كما دونه متى 16:22; الإنجيل كما دونه يوحنا 16:25; الإنجيل كما دونه يوحنا 16:29"> وَقَدْ تَحَدَّثَ عَنْ هَذَا الأَمْرِ صَرَاحَةً. فَانْتَحَى بِهِ بُطْرُسُ جَانِباً وَأَخَذَ يُوَبِّخُهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه متى 4:10; الرسالة إلى مؤمني فيلبي 3:19; رسالة يوحنا الأولى 2:15"> وَلَكِنَّهُ الْتَفَتَ وَنَظَرَ إِلَى تَلامِيذِهِ وَزَجَرَ بُطْرُسَ قَائِلاً: «اغْرُبْ مِنْ أَمَامِي يَا شَيْطَانُ، لأَنَّكَ تُفَكِّرُ لَا بِأُمُورِ اللهِ بَلْ بِأُمُورِ النَّاسِ!»</span>
</div>
</div>

## طريق الصليب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="الإنجيل كما دونه متى 10:38; الإنجيل كما دونه لوقا 14:26–27; الإنجيل كما دونه يوحنا 10:27"> ثُمَّ دَعَا الْجَمْعَ مَعَ تَلامِيذِهِ، وَقَالَ لَهُمْ: «إِنْ أَرَادَ أَحَدٌ أَنْ يَسِيرَ وَرَائِي، فَلْيُنْكِرْ نَفْسَهُ، وَيَحْمِلْ صَلِيبَهُ، وَيَتْبَعْنِي.</span>
<span className="v" id="v35" data-v="35" data-xref="الرسالة الثانية إلى تيموثاوس 2:11–13; الرسالة الأولى إلى مؤمني كورنثوس 9:23; الإنجيل كما دونه لوقا 17:33"> فَأَيُّ مَنْ أَرَادَ أَنْ يُخَلِّصَ نَفْسَهُ، يَخْسَرُهَا. وَلكِنَّ مَنْ يَخْسَرُ نَفْسَهُ مِنْ أَجْلِي وَمِنْ أَجْلِ الإِنْجِيلِ، فَهُوَ يُخَلِّصُهَا.</span>
