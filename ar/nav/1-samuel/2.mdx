## صلاة حنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه لوقا 1:46–56; المزامير 13:5; إشعياء 12:2–3"> وَصَلَّتْ حَنَّةُ قَائِلَةً: «ابْتَهَجَ قَلْبِي بِالرَّبِّ وَسَمَتْ عِزَّتِي بِهِ. أَفْتَخِرُ عَلَى أَعْدَائِي لأَنِّي فَرِحْتُ بِخَلاصِكَ.</span>
<span className="v" id="v2" data-v="2" data-xref="صموئيل الثاني 22:32; إشعياء 44:6; المزامير 86:8"> إِذْ لَيْسَ قُدُّوسٌ نَظِيرَ الرَّبِّ، وَلا يُوْجَدُ مَنْ يُمَاثِلُكَ، وَلَيْسَ صَخْرَةٌ كَإِلَهِنَا.</span>
<span className="v" id="v3" data-v="3" data-xref="الأمثال 8:13; ملوك الأول 8:39; الأمثال 24:12"> كُفُّوا عِنِ الْكِبْرِيَاءِ، وَكُمُّوا أَفْوَاهَكُمْ عَنِ الْغُرُورِ لأَنَّ الرَّبَّ إِلَهٌ عَلِيمٌ وَبِهِ تُوْزَنُ الأَعْمَالُ.</span>
<span className="v" id="v4" data-v="4" data-xref="المزامير 76:3; المزامير 37:15; المزامير 46:9"> لَقَدْ تَحَطَّمَتْ أَقْوَاسُ الْجَبَابِرَةِ وَتَنَطَّقَ الضُّعَفَاءُ بِالْقُوَّةِ.</span>
<span className="v" id="v5" data-v="5" data-xref="إرميا 15:9; الإنجيل كما دونه لوقا 1:53; المزامير 113:9"> الَّذِينَ كَانُوا شَبَاعَى آجَرُوا أَنْفُسَهُمْ لِقَاءَ الطَّعَامِ، وَالَّذِينَ كَانُوا جِيَاعاً مَلأَهُمُ الشِّبَعُ. أَنْجَبَتِ الْعَاقِرُ سَبْعَةً، أَمَّا كَثِيرَةُ الأَبْنَاءِ فَقَدْ ذَبُلَتْ.</span>
<span className="v" id="v6" data-v="6" data-xref="التثنية 32:39; إشعياء 26:19; رؤيا يوحنا 1:18"> الرَّبُّ يُمِيتُ وَيُحْيِي، يَطْرَحُ إِلَى الْهَاوِيَةِ وَيُصْعِدُ مِنْهَا.</span>
<span className="v" id="v7" data-v="7" data-xref="المزامير 75:7; أيوب 5:11; التثنية 8:17–18"> الرَّبُّ يُفْقِرُ وَيُغْنِي، يُذِلُّ وَيُعِزُّ.</span>
<span className="v" id="v8" data-v="8" data-xref="المزامير 113:7–8; دانيال 2:48; رسالة يعقوب 2:5"> يُنْهِضُ الْمِسْكِينَ مِنَ التُّرَابِ، وَيَرْفَعُ الْبَائِسَ مِنْ كَوْمَةِ الرَّمَادِ، لِيُجْلِسَهُ مَعَ النُّبَلاءِ، وَيُمَلِّكَهُ عَرْشَ الْمَجْدِ، لأَنَّ لِلرَّبِّ أَسَاسَاتِ الأَرْضِ الَّتِي أَرْسَى عَلَيْهَا الْمَسْكُونَةَ.</span>
<span className="v" id="v9" data-v="9" data-xref="المزامير 121:3; المزامير 91:11–12; المزامير 33:16–17"> هُوَ يَحْفَظُ أَقْدَامَ أَتْقِيَائِهِ، أَمَّا الأَشْرَارُ فَيَنْطَوُونَ فِي الظَّلامِ، لأَنَّهُ لَيْسَ بِالْقُوَّةِ يَتَغَلَّبُ الإِنْسَانُ.</span>
<span className="v" id="v10" data-v="10" data-xref="المزامير 2:9; المزامير 89:24; المزامير 18:13–14"> مُخَاصِمُو الرَّبِّ يَتَحَطَّمُونَ، وَمِنَ السَّمَاءِ يَقْذِفُ رُعُودَهُ عَلَيْهِمْ؛ يَدِينُ الرَّبُّ أَقَاصِيَ الأَرْضِ، وَيَمْنَحُ عِزَّةً لِمَنْ يَخْتَارُهُ مَلِكاً وَيُمَجِّدُ مَسِيحَهُ».</span>
<span className="v" id="v11" data-v="11" data-xref="صموئيل الأول 3:1; صموئيل الأول 2:18; صموئيل الأول 1:28"> ثُمَّ رَجَعَ أَلْقَانَةُ إِلَى بَيْتِهِ فِي الرَّامَةِ، وَظَلَّ الصَّبِيُّ يَخْدُمُ الرَّبَّ لَدَى عَالِي الْكَاهِنِ.</span>
</div>
</div>

## تصرفات ابني عالي السّيئة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الرسالة إلى مؤمني روما 1:28–30; إرميا 2:8; القضاة 2:10"> أَمَّا ابْنَا عَالِي فَكَانَا مُتَوَرِّطَيْنِ فِي الشَّرِّ لَا يَعْرِفَانِ الرَّبَّ</span>
<span className="v" id="v13" data-v="13" data-xref="اللاويين 7:29–34"> وَلا حَقَّ الْكَهَنَةِ الْمَتَوَجِّبَ عَلَى الشَّعْبِ. فَكَانَ كُلَّمَا قَدَّمَ رَجُلٌ ذَبِيحَةً يَأْتِي غُلامُ الْكَاهِنِ عِنْدَ طَبْخِ اللَّحْمِ حَامِلاً بِيَدِهِ خُطَّافاً ذَا ثلاثِ شُعَبٍ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخروج 29:27–28; إشعياء 56:11; صموئيل الأول 2:29"> فَيَغْرِزُهُ فِي اللَّحْمِ الَّذِي في الْمَرْحَضَةِ أَوِ الْمِرْجَلِ أَوِ الْمِقْلَى أَوِ الْقِدْرِ، وَيَأْخُذُ الْكَاهِنُ كُلَّ مَا يَعْلَقُ بِشُعَبِ الْخُطَّافِ. هَكَذَا كَانَا يُعَامِلانِ جَمِيعَ الإِسْرَائِيلِيِّينَ الْقَادِمِينَ إِلَى شِيلُوهَ.</span>
<span className="v" id="v15" data-v="15" data-xref="اللاويين 3:16; اللاويين 3:3–5; الرسالة إلى مؤمني روما 16:18"> كَذَلِكَ كَانَ خَادِمُ الْكَاهِنِ يَأْتِي إِلَى ذَابِحِ الْقُرْبَانِ وَيَقُولُ لَهُ قَبْلَ إِحْرَاقِ الشَّحْمِ: «أَعْطِ لَحْماً لِلْكَاهِنِ حَتَّى يُشْوَى، فَإِنَّهُ لَا يَقْبَلُ مِنْكَ لَحْماً مَطْبُوخاً بَلْ نِيئاً».</span>
<span className="v" id="v16" data-v="16" data-xref="اللاويين 7:23–25; اللاويين 3:16; رسالة بطرس الأولى 5:2–3"> فَيُجِيبُهُ الرَّجُلُ: «لِيُحْرِقُوا أَوَّلاً شَحْمَ الذَّبِيحَةِ، ثُمَّ خُذْ مَا تَشْتَهِيهِ نَفْسُكَ». فَيَقُولُ الْخَادِمُ: «لا، بَلْ أَعْطِنِي الآنَ اللَّحْمَ وَإلَّا آخُذُهُ بِالرَّغْمِ عَنْكَ».</span>
<span className="v" id="v17" data-v="17" data-xref="ملاخي 2:7–9; التكوين 13:13; ملاخي 2:13"> فَعَظُمَتْ خَطِيئَةُ أَبْنَاءِ عَالِي أَمَامَ الرَّبِّ، لأَنَّ الشَّعْبَ اسْتَهَانَ بِذَبِيحَةِ الرَّبِّ مِنْ جَرَّاءِ تَصَرُّفَاتِهِمَا.</span>
</div>
</div>

## صموئيل يخدم أمام الرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="صموئيل الأول 3:1; الخروج 28:4; صموئيل الأول 2:11"> وَكَانَ صَمُوئِيلُ آنَئِذٍ يَخْدُمُ فِي مَحْضَرِ الرَّبِّ وَهُوَ مَا بَرِحَ صَبِيًّا، يَرْتَدِي أَفُوداً مِنْ كَتَّانٍ.</span>
<span className="v" id="v19" data-v="19" data-xref="صموئيل الأول 1:3; صموئيل الأول 1:21; الخروج 23:14"> وَكَانَتْ أُمُّهُ تَصْنَعُ لَهُ جُبَّةً صَغِيرَةً، تُحْضِرُهَا مَعَهَا كُلَّ سَنَةٍ عِنْدَ مَجِيءِ رَجُلِهَا لِتَقْرِيبِ الذَّبِيحَةِ السَّنَوِيَّةِ،</span>
<span className="v" id="v20" data-v="20" data-xref="صموئيل الأول 1:27–28; الإنجيل كما دونه لوقا 2:34; راعوث 4:11"> فَيُبَارِكُ عَالِي أَلْقَانَةَ وَزَوْجَتَهُ قَائِلاً: «لِيَرْزُقْكَ الرَّبُّ ذُرِّيَّةً مِنْ هَذِهِ الْمَرْأَةِ عِوَضاً عَنِ الصَّبِيِّ الَّذِي وَهَبْتُمَاهُ لِلرَّبِّ». ثُمَّ يَرْجِعَانِ إِلَى حَيْثُ يُقِيمَانِ.</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه لوقا 2:40; صموئيل الأول 3:19; التكوين 21:1"> وَعِنْدَمَا افْتَقَدَ الرَّبُّ حَنَّةَ، حَمَلَتْ وَأَنْجَبَتْ ثَلاثَةَ أَبْنَاءٍ وَبِنْتَيْنِ. أَمَّا صَمُوئِيلُ فَقَدْ تَرَعْرَعَ فِي خِدْمَةِ الرَّبِّ.</span>
<span className="v" id="v22" data-v="22" data-xref="الخروج 38:8; صموئيل الأول 2:13–17; إرميا 7:9–10"> وَطَعَنَ عَالِي فِي السِّنِّ. وَبَلَغَهُ مَا ارْتَكَبَهُ بَنوهُ مَنْ مَسَاوِئَ بِحَقِّ جَمِيِع الإِسْرَائِيلِيِّينَ، وَأَنَّهُمْ كَانَوا يُضَاجِعَونِ النِّسَاءَ الْمُجْتَمِعَاتِ عِنْدَ مَدْخَلِ خَيْمَةِ الاجْتِمَاعِ.</span>
<span className="v" id="v23" data-v="23" data-xref="إرميا 8:12; الرسالة إلى مؤمني فيلبي 3:19; أعمال الرسل 9:4"> فَقَال لَهُمْ: «لِمَاذَا تَرْتَكِبُونَ هَذِهِ الْفَوَاحِشَ، فَقَدْ بَلَغَتْنِي أَخْبَارُ مَسَاوِئِكُمْ مِنْ جَمِيعِ هَذَا الشَّعْبِ؟</span>
<span className="v" id="v24" data-v="24" data-xref="ملاخي 2:8; الرسالة الأولى إلى تيموثاوس 3:7; أعمال الرسل 6:3"> لا، يا بَنِيَّ، فَالأَخْبَارُ الَّتِي بَلَغَتْنِي مُشِينَةٌ، إِذْ إِنَّكُمْ تَجْعَلُونَ الشَّعْبَ يَتَعَدَّى عَلَى شَرِيعَةِ الرَّبِّ.</span>
<span className="v" id="v25" data-v="25" data-xref="صموئيل الأول 3:14; يشوع 11:20; الرسالة إلى العبرانيين 10:26"> فَإِنْ أَخْطَأَ إِنْسَانٌ نَحْوَ إِنْسَانٍ، فَاللهُ يَدِينُهُ، وَلَكِنْ إِنْ أَخْطَأَ إِلَى الرَّبِّ فَمَنْ يُصَلِّي مِنْ أَجْلِهِ؟» لَكِنَّهُمْ لَمْ يُعِيرَوا تَوْبِيخَ أَبِيهِمْ أَيَّ اهْتِمَامٍ لأَنَّ الرَّبَّ شَاءَ أَنْ يُمِيتَهُمْ.</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه لوقا 2:52; الإنجيل كما دونه لوقا 2:40; أعمال الرسل 2:47"> أَمَّا الصَّبِيُّ صَمُوئِيلُ فَاسْتَمَرَّ يَنْمُو فِي الصَّلاحِ وَيَحْظَى بِرِضَى اللهِ وَالنَّاسِ.</span>
</div>
</div>

## نبوءة بهلاك أُسرة عالي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التثنية 33:1; الخروج 4:27; الخروج 4:14–16"> وَذَاتَ يَوْمٍ جَاءَ نَبِيٌّ إِلَى عَالِي بِرِسَالَةٍ مِنَ اللهِ، قَالَ: «أَلَمْ أَتَجَلَّ لِبَيْتِ أَبِيكَ وَهُمْ مَا بَرِحُوا فِي مِصْرَ فِي دِيَارِ فِرْعَوْنَ،</span>
<span className="v" id="v28" data-v="28" data-xref="الخروج 28:1; اللاويين 8:7–8; العدد 18:19"> وَانْتَخَبْتُ أَبَاكُمْ هرُونَ مِنْ بَيْنِ جَمِيعِ أَسْبَاطِ إِسْرَائِيلَ لِيَكُونَ لِي كَاهِناً يُصْعِدُ عَلَى مَذْبَحِي قَرَابِينَ وَيُوْقِدُ بَخُوراً، وَيَرْتَدِي أَمَامِي أَفُوداً، وَوَهَبْتُ لِبَيْتِ أَبِيكَ جَمِيعَ وَقائِدِ بَنِي إِسْرَائِيلَ.</span>
<span className="v" id="v29" data-v="29" data-xref="الإنجيل كما دونه متى 10:37; التثنية 32:15; صموئيل الأول 2:13–17"> فَلِمَاذَا تَحْتَقِرُونَ ذَبِيحَتِي وَتَقْدِمَتِي الَّتِي أَمَرْتُ بِها لِلْمَسْكَنِ، وَتُفَضِّلُ ابْنَيْكَ عَنِّي لِتُكَدِّسُوا الشَّحْمَ عَلَى أَبْدَانِكُمْ، مِمَّا تَخَيَّرْتُمُوهُ مِنْ قَرَابِينِ شَعْبِي؟</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه يوحنا 12:26; إرميا 18:9–10; الأمثال 3:9–10"> لِذَلِكَ يَقُولُ الرَّبُّ إِلَهُ إِسْرَائِيلَ: لَقَدْ وَعَدْتُ أَنْ يَظَلَّ بَيْتُكَ وَبَيْتُ أَبِيكَ يَخْدُمُونَ فِي مَحْضَرِي إِلَى الأَبَدِ. أَمَّا الآنَ، يَقُولُ الرَّبُّ: فَحَاشَا لِي أَنْ أَفْعَلَ ذَلِكَ، لأَنَّنِي أُكْرِمُ الَّذِينَ يُكْرِمُونَنِي، أَمَّا الَّذِينَ يَحْتَقِرُونَنِي فَيَصْغُرُونَ.</span>
<span className="v" id="v31" data-v="31" data-xref="صموئيل الأول 22:17–20; ملوك الأول 2:26–27; المزامير 37:17"> هَا هِي أَيَّامٌ مُقْبِلَةٌ يَخْطِفُ فِيهَا الْمَوْتُ رِجَالَكُمْ فَلا يَبْقَى شَيْخٌ فِي بَيْتِكَ.</span>
<span className="v" id="v32" data-v="32" data-xref="زكريا 8:4; صموئيل الأول 4:11; المزامير 78:59–64"> وَتَشْهَدُ ضِيقاً فِي مَسْكَنِي، بَيْنَمَا يَنْعَمُ الإِسْرَائِيلِيُّونَ بِالرَّفَاهِيَةِ وَيَخْلُو بَيْتُكَ مِنَ الشُّيُوخِ كُلَّ الأَيَّامِ.</span>
<span className="v" id="v33" data-v="33" data-xref="ملوك الأول 2:26–27; ملوك الأول 1:7; صموئيل الأول 22:21–23"> وَيَكُونُ مَنْ أَسْتَحْيِيهِ مِنْ ذُرِّيَّتِكَ لِخِدْمَةِ مَذْبَحِي سَبَباً فِي إعْشَاءِ عَيْنَيْكَ بِالْدُّمُوعِ وَإِذَابَةِ قَلْبِكَ بِالْحُزْنِ، وَبَقِيَّةُ ذُرِّيَّتِكَ يَمُوتُونَ شُبَّاناً.</span>
<span className="v" id="v34" data-v="34" data-xref="صموئيل الأول 4:11; ملوك الأول 13:3; صموئيل الأول 4:17"> وَتَصْدِيقاً لِقَوْلِي أُعْطِيكَ عَلامَةً تُصِيبُ ابْنَيْكَ حُفْنِي وَفِينْحَاسَ: إِنَّهُمَا فِي يَوْمٍ وَاحِدٍ يَمُوتَانِ كِلاهُمَا.</span>
<span className="v" id="v35" data-v="35" data-xref="صموئيل الثاني 7:27; صموئيل الثاني 7:11; ملوك الأول 11:38"> فَأَخْتَارُ لِنَفْسِي كَاهِناً مُخْلِصاً يَعْمَلُ بِمُقْتَضَى مَا بِقَلْبِي وَنَفْسِي فَأُقِيمُ لَهُ بَيْتاً أَمِيناً، وَيَصِيرُ كَاهِناً لِلْمَلِكِ الَّذِي أَخْتَارُهُ.</span>
