## شرائع يوم السبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخروج 34:32; الخروج 31:1–11; الرسالة إلى مؤمني روما 2:13"> وَعَقَدَ مُوسَى اجْتِمَاعاً لِكُلِّ شَعْبِ إِسْرَائِيلَ وَقَالَ لَهُمْ: «هَذِهِ هِيَ الأُمُورُ الَّتِي أَوْصَى الرَّبُّ أَنْ تُطِيعُوهَا:</span>
<span className="v" id="v2" data-v="2" data-xref="اللاويين 23:3; الإنجيل كما دونه لوقا 13:14–15; الخروج 20:9–10"> سِتَّةَ أَيَّامٍ تَنْصَرِفُونَ فِيهَا إِلَى أَعْمَالِكُمْ. أَمَّا السَّابِعُ فَيَكُونُ يَوْمَ رَاحَةٍ لَكُمْ مُقَدَّساً لِعِبَادَةِ الرَّبِّ. كُلُّ مَنْ يَقُومُ فِيهِ بِأَيِّ عَمَلٍ يُقْتَلُ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخروج 16:23; العدد 15:32–36; الخروج 12:16"> لَا تُوْقِدُوا نَاراً فِي بُيُوتِكُمْ فِي يَوْمِ السَّبْتِ».</span>
</div>
</div>

## التبرع بمواد بناء المقدس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الخروج 25:1–2"> ثُمَّ قَالَ مُوسَى لِكُلِّ شَعْبِ إِسْرَائِيلَ: «هَذَا هُوَ أَمْرُ الرَّبِّ:</span>
<span className="v" id="v5" data-v="5" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 9:7; المزامير 110:3; الخروج 25:2–7"> هَاتُوا مِمَّا لَدَيْكُمْ تَقْدِمَةً لِلرَّبِّ. كُلُّ مَنْ هُوَ سَخِيُّ النَّفْسِ فَلْيَتَبَرَّعْ بِتَقْدِمَةٍ لِلرَّبِّ: ذَهَباً وَفِضَّةً وَنُحَاساً،</span>
<span className="v" id="v6" data-v="6" data-xref="الخروج 26:7–14; الخروج 26:31; الخروج 28:15"> وَأَقْمِشَةً زَرْقَاءَ وَبَنَفْسَجِيَّةً وَحَمْرَاءَ، وَمَنْسُوجَاتٍ كَتَّانِيَّةً، وَمِنْ شَعْرِ الْمِعْزَى،</span>
<span className="v" id="v7" data-v="7"> وَجُلُودَ كِبَاشٍ، وَجُلُودَ دُلْفِينٍ، وَخَشَبَ السَّنْطِ،</span>
<span className="v" id="v8" data-v="8" data-xref="الخروج 27:20; الخروج 25:1–40; الخروج 30:23"> وَزَيْتاً لِلإِنَارَةِ، وَأَطْيَاباً لِدُهْنِ الْمَسْحَةِ وَلِلْبَخُورِ الْعَطِرِ،</span>
<span className="v" id="v9" data-v="9" data-xref="الخروج 39:6–14; الخروج 28:9; الخروج 25:5"> وَحِجَارَةَ جَزْعٍ، وَحِجَارَةً كَرِيمَةً لِتَرْصِيعِ الرِّدَاءِ وَالصُّدْرَةِ.</span>
<span className="v" id="v10" data-v="10" data-xref="الخروج 31:1–6; الخروج 36:1–4"> وَلْيَتَقَدَّمْ كُلُّ صَانِعٍ مَاهِرٍ بَيْنَكُمْ لِتَنْفِيذِ مَا أَمَرَ بِهِ الرَّبُّ:</span>
<span className="v" id="v11" data-v="11" data-xref="الخروج 26:1–37; الخروج 36:8–34; الخروج 31:7–9"> المَسْكَنَ وَخَيْمَتَهُ وَسُقُوفَهَا وَمَشَابِكَهَا وَأَلْوَاحَهَا وَعَوَارِضَهَا وَأَعْمِدَتَهَا وَقَوَاعِدَ الأَعْمِدَةِ،</span>
<span className="v" id="v12" data-v="12" data-xref="الخروج 25:10–22; الخروج 26:31–33; الخروج 37:1–9"> وَالتَّابُوتَ وَعَصَوَيْهِ، وَغِطَاءَ الْكَفَّارَةِ، وَالسِّتَارَ الَّذِي يَحْجُبُ الْتَّابُوتَ،</span>
<span className="v" id="v13" data-v="13" data-xref="الخروج 25:23–30; اللاويين 24:5–6; الخروج 37:10–16"> وَالْمَائِدَةَ وَعَصَوَيْهَا، وَجَمِيعَ أَوَانِيهَا، وَخُبْزَ التَّقْدِمَةِ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخروج 25:31–39; الإنجيل كما دونه متى 5:14–15; الخروج 37:17–24"> وَمَنَارَةَ الإِضَاءَةِ وَأَوَانِيهَا وَسُرُجَهَا وَزَيْتَ ضَوْئِهَا،</span>
<span className="v" id="v15" data-v="15" data-xref="الخروج 30:1–10; الخروج 30:22–38; الخروج 37:25–28"> وَمَذْبَحَ الْبَخُورِ وَعَصَوَيْهِ، وَدُهْنَ الْمَسْحَةِ، وَالْبَخُورَ الْعَطِرَ، وَسِتَارَ بَابِ مَدْخَلِ الْمَسْكِنِ،</span>
<span className="v" id="v16" data-v="16" data-xref="الخروج 27:1–8; الخروج 30:18–21; الخروج 38:1–8"> وَمَذْبَحَ الْمُحْرَقَةِ، وَشَبَكَتَهُ النُّحَاسِيَّةَ وَعَصَوَيْهِ وَكُلَّ أَوَانِيهِ، وَحَوْضَ الاغْتِسَالِ وَقَاعِدَتَهُ،</span>
<span className="v" id="v17" data-v="17" data-xref="صموئيل الثاني 7:2; الخروج 38:9–20; الخروج 27:9–19"> وَسَتَائِرَ سَاحَةِ الْمَسْكِنِ وَأَعْمِدَتَهَا، وَقَوَاعِدَ الأَعْمِدَةِ، وَسِتَارَ السَّاحَةِ،</span>
<span className="v" id="v18" data-v="18" data-xref="الخروج 27:19"> وَأَوْتَادَ الْخَيْمَةِ وَالسَّاحَةِ وَحِبَالَهُمَا،</span>
<span className="v" id="v19" data-v="19" data-xref="الخروج 31:10; العدد 4:5–15; الخروج 28:1–43"> وَثِيَابَ هَرُونَ الْكَاهِنِ، وَثِيَابَ بَنِيهِ الْمُقَدَّسَةَ، وَالثِّيَابَ الْمَنْسُوجَةَ لِلْخِدْمَةِ فِي الْمَقْدِسِ».</span>
<span className="v" id="v20" data-v="20"> فَانْصَرَفَ كُلُّ شَعْبِ إِسْرَائِيلَ مِنْ قُدَّامِ مُوسَى.</span>
<span className="v" id="v21" data-v="21" data-xref="الخروج 25:2; الخروج 35:26; أخبار الأيام الأول 29:14"> ثُمَّ أَقْبَلَ كُلُّ مَنْ حَثَّهُ قَلْبُهُ، وَكُلُّ مَنْ سَخَتْ نَفْسُهُ، حَامِلِينَ تَقْدِمَةَ الرَّبِّ، لإِقَامَةِ خَيْمَةِ الاجْتِمَاعِ وَأَوَانِي خِدْمَتِهَا وَلِتَجْهِيزِ الثِّيَابِ الْمُقَدَّسَةِ.</span>
<span className="v" id="v22" data-v="22" data-xref="نحميا 7:70–72; الإنجيل كما دونه متى 2:11; عزرا 2:68–69"> وَتَوَافَدَ الرِّجَالُ مَعَ النِّسَاءِ مِنْ ذَوِي النُّفُوسِ السَّخِيَّةِ، مُتَبَرِّعِينَ بِأَسَاوِرَ وَأَقْرَاطٍ وَخَوَاتِمَ وَقَلائِدَ وَغَيْرِ ذَلِكَ مِنَ الأَمْتِعَةِ الذَّهَبِيَّةِ، فَكَانَتْ كُلُّ تَقْدِمَاتِ الْمُتَبَرِّعِينَ مِنَ الذَّهَبِ لِلرَّبِّ.</span>
<span className="v" id="v23" data-v="23" data-xref="أخبار الأيام الأول 29:8; الخروج 25:2–7; الخروج 35:6–10"> وَكَذَلِكَ كُلُّ مَنْ وُجِدَ لَدَيْهِ أَقْمِشَةٌ زَرْقَاءُ وَبَنَفْسَجِيَّةٌ وَحَمْرَاءُ، وَمَنْسُوجَاتٌ كَتَّانِيَّةٌ، وَمِنْ شَعْرِ الْمِعْزَى، وَجُلُودُ كِبَاشٍ، وَجُلُودُ دُلْفِينٍ، تَبَرَّعَ بِها.</span>
<span className="v" id="v24" data-v="24" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 8:12"> وَمِنْهُمْ أَيْضاً مَنْ تَبَرَّعَ بِتَقْدِمَاتِ فِضَّةٍ وَنُحَاسٍ، جَاءَ بِها لِلرَّبِّ وَكُلُّ مَنْ وُجِدَ لَدَيْهِ خَشَبُ سَنْطٍ صَالِحٌ لِلاسْتِخْدَامِ فِي عَمَلٍ مَا، تَبَرَّعَ بِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="الرسالة إلى مؤمني روما 16:12; الإنجيل كما دونه لوقا 8:2–3; الأمثال 14:1"> وَغَزَلَتْ جَمِيعُ النِّسَاءِ الْمَاهِرَاتِ بِأَيْدِيهِنَّ مَنْسُوجَاتٍ مِنْ خُيُوطٍ زَرْقَاءَ وَبَنَفْسَجِيَّةٍ وَحَمْرَاءَ، وَمَنْسُوجَاتٍ كَتَّانِيَّةً،</span>
<span className="v" id="v26" data-v="26" data-xref="الخروج 35:21; الخروج 36:8; الخروج 35:29"> كَمَا غَزَلَتِ النِّسَاءُ اللَّوَاتِي حَثَّتْهُنَّ قُلُوبُهُنَّ، مَنْسُوجَاتٍ مِنْ شَعْرِ الْمِعْزَى.</span>
<span className="v" id="v27" data-v="27" data-xref="عزرا 2:68; أخبار الأيام الأول 29:6; الخروج 35:9"> وَتَبَرَّعَ وُجَهَاءُ الشَّعْبِ بِحِجَارَةِ الْجَزْعِ وَحِجَارَةٍ كَرِيمَةٍ لِتَرْصِيعِ الرِّدَاءِ وَالصُّدْرَةِ،</span>
<span className="v" id="v28" data-v="28" data-xref="الخروج 30:23–38; الخروج 35:8"> وَبِالأَطْيَابِ وَزَيْتِ الإِنَارَةِ وَلِدُهْنِ الْمَسْحَةِ وَلِلْبَخُورِ الْعَطِرِ.</span>
<span className="v" id="v29" data-v="29" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 9:7; الخروج 35:21–22; الخروج 35:4"> فَتَبَرَّعَ جَمِيعُ الرِّجَالِ وَالنِّسَاءِ مِنْ بَنِي إِسْرَائِيلَ، مِمَّنْ سَخَتْ قُلُوبُهُمْ، مُقَدِّمِينَ شَيْئاً لِلرَّبِّ لِتَنْفِيذِ كُلِّ الْعَمَلِ الَّذِي كَلَّفَ الرَّبُّ مُوسَى بِالإِشْرَافِ عَلَى إِنْجَازِهِ.</span>
</div>

## بصلئيل وأهوليآب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="الخروج 31:1–6; رسالة يعقوب 1:17; الرسالة الأولى إلى مؤمني كورنثوس 12:11"> وَقَالَ مُوسَى لِبَنِي إِسْرَائِيلَ: «هَا إِنَّ الرَّبَّ قَدْ دَعَا بَصَلْئِيلَ بْنَ أُورِي حَفِيدَ حُورٍ، مِنْ سِبْطِ يَهُوذَا بِاسْمِهِ،</span>
<span className="v" id="v31" data-v="31" data-xref="رسالة يعقوب 1:17; إشعياء 11:2–5; إشعياء 61:1–3"> وَمَلأَهُ مِنْ رُوحِ اللهِ وَوَهَبَهُ الْحِكْمَةَ وَالْمَهَارَةَ وَالْمَعْرِفَةَ فِي كُلِّ أَنْوَاعِ الْحِرَفِ،</span>
<span className="v" id="v32" data-v="32"> لاِبْتِكَارِ فُنُونِ التَّصْمِيمَاتِ الْمَصْنُوعَةِ مِنَ الذَّهَبِ وَالْفِضَّةِ وَالنُّحَاسِ،</span>
<span className="v" id="v33" data-v="33"> وَنَقْشِ الْحِجَارَةِ وَتَرْصِيعِهَا، وَنِجَارَةِ الْخَشَبِ، وَلِيَكُونَ مُحْتَرِفاً لِكُلِّ صِنَاعَةٍ.</span>
<span className="v" id="v34" data-v="34" data-xref="الخروج 31:6; أخبار الأيام الثاني 2:14; عزرا 7:27"> وَقَدْ مَنَحَهُ الرَّبُّ هُوَ وَأَهُولِيآبَ بْنَ أَخِيسَامَاكَ مِنْ دَانٍ، الْقُدْرَةَ عَلَى تَعْلِيمِ الآخَرِينَ.</span>
<span className="v" id="v35" data-v="35" data-xref="الخروج 35:31; ملوك الأول 7:14; إشعياء 28:26"> وَجَعَلَهُمَا حَاذِقَيْنِ مُتَفَوِّقَيْنِ فِي صِنَاعَةِ نَقْشِ الْجَوَاهِرِ وَالنِّجَارَةِ وَالْحِيَاكَةِ، وَتَطْرِيزِ الْمَنْسُوجَاتِ الزَّرْقَاءِ وَالْبَنَفْسَجِيَّةِ وَالْحَمْرَاءِ وَالْمَنْسُوجَاتِ الْكَتَّانِيَّةِ، وَسَائِرِ حِرَفِ النَّسِيجِ، وَفِي كُلِّ حِرْفَةِ صَانِعٍ وَمُبْتَكِرِ تَصْمِيمٍ».</span>
