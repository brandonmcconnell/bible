## نصيب يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العدد 26:55–56; يشوع 14:2; العدد 33:36–37"> وَهَذِهِ هِيَ قُرْعَةُ سِبْطِ يَهُوذَا بِحَسَبِ عَشَائِرِهِمْ: امْتَدَّتْ حُدُودُهُمُ الْجَنُوبِيَّةُ إِلَى آخِرِ أَطْرَافِ صَحْرَاءِ صِينَ الْمُتَاخِمَةِ لِحُدُودِ أَدُومَ.</span>
<span className="v" id="v2" data-v="2" data-xref="التكوين 14:3; حزقيال 47:8; إشعياء 11:15"> كَمَا بَدَأَتْ حُدُودُهُمُ الْجَنُوبِيَّةُ مِنَ الْخَلِيجِ فِي أَقْصَى الطَّرَفِ الْجَنُوبِيِّ لِلْبَحْرِ الْمَيِّتِ.</span>
<span className="v" id="v3" data-v="3" data-xref="العدد 34:4; القضاة 1:36; العدد 20:1"> مَارَّةً بِعَقَبَةِ عَقْرِبِّيمَ جَنُوباً، وَعَابِرَةً صَحْرَاءَ صِينَ، حَتَّى تَبْلُغَ جَنُوبِيَّ قَادَشَ بَرْنِيعَ، وَتَتَّجِهُ إِلَى حَصْرُونَ، وَمِنْهَا صُعُوداً إِلَى أَدَّارَ، ثُمَّ تَلْتَفُّ نَحْوَ قَرْقَعَ،</span>
<span className="v" id="v4" data-v="4" data-xref="العدد 34:5; يشوع 13:3; الخروج 23:31"> وَمِنْهَا تَعْبُرُ إِلَى عَصْمُونَ حَتَّى تَصِلَ إِلَى وَادِي مِصْرَ، حَيْثُ تَنْتَهِي عِنْدَ الْبَحْرِ. هَذِهِ هِيَ حُدُودُهُمُ الْجَنُوبِيَّةُ.</span>
<span className="v" id="v5" data-v="5" data-xref="العدد 34:12; يشوع 18:15–19; العدد 34:10"> أَمَّا الْحُدُودُ الشَّرْقِيَّةُ فَهِيَ الْبَحْرُ الْمَيِّتُ إِلَى طَرَفِ نَهْرِ الأُرْدُنِّ. وتَبْدَأُ الْحُدُودُ الشِّمَالِيَّةُ مِنْ لِسَانِ الْبَحْرِ مِنْ أَقْصَى الأُرْدُنِّ،</span>
<span className="v" id="v6" data-v="6" data-xref="يشوع 18:17; يشوع 18:19–21"> وَتَتَّجِهُ إِلَى بَيْتِ حُجْلَةَ عُبُوراً مِنْ شِمَالِيِّ بَيْتِ عَرَبَةَ، وَصُعُوداً إِلَى حَجَرِ بُوهَنَ بْنِ رَأُوبَيْنَ.</span>
<span className="v" id="v7" data-v="7" data-xref="ملوك الأول 1:9; صموئيل الثاني 17:17; يشوع 7:26"> وَتُتَابِعُ امْتِدَادَهَا إِلَى دَبِيرَ مِنْ وَادِي عَخُورَ مُتَّجِهَةً شِمَالاً إِلَى الْجِلْجَالِ الَّتِي مُقَابِلَ عَقَبَةِ أَدُمِّيمَ جَنُوبِيَّ الْوَادِي، وَتَسْتَمِرُّ عَلَى طُولِ مِيَاهِ عَيْنِ شَمْسٍ حَتَّى تَنْتَهِيَ بِعَيْنِ رُوجَلَ.</span>
<span className="v" id="v8" data-v="8" data-xref="يشوع 18:16; يشوع 15:63; يشوع 18:28"> ثُمَّ تَصْعَدُ بِاتِّجَاهِ وَادِي ابْنِ هِنُّومَ عَلَى مُحَاذَاةِ الْمُنْحَدَرِ الْجَنُوبِيِّ لأُورُشَلِيمَ مَدِينَةِ الْيَبُوسِيِّينَ، وَتُتَابِعُ صُعُودَهَا إِلَى قِمَّةِ الْجَبَلِ قُبَالَةَ وَادِي هِنُّومَ غَرْباً، الْوَاقِعِ فِي طَرَفِ وَادِي الرَّفَائِيِّينَ شِمَالاً.</span>
<span className="v" id="v9" data-v="9" data-xref="أخبار الأيام الأول 13:6; يشوع 18:15; القضاة 18:12"> ثُمَّ تَمْتَدُّ هَذِهِ الْحُدُودُ مِنْ قِمَّةِ الْجَبَلِ إِلَى مَنْبَعِ مِيَاهِ نَفْتُوحَ حَتَّى تَصِلَ إِلَى مُدُنِ جَبَلِ عِفْرُونَ فَتَبْلُغَ بَعَلَةَ الَّتِي هِيَ قَرْيَةُ يَعَارِيمَ.</span>
<span className="v" id="v10" data-v="10" data-xref="القضاة 14:1; القضاة 14:5; التكوين 38:12–13"> وَتَتَّجِهُ مِنْ بَعَلَةَ غَرْباً إِلَى جَبَلِ سِعِيرَ عُبُوراً إِلَى جَانِبِ جَبَلِ يَعَارِيمَ شِمَالاً الَّتِي هِيَ كَسَالُونَ، ثُمَّ تَنْحَدِرُ نَحْوَ بَيْتِ شَمْسٍ مُرُوراً بِتِمْنَةَ،</span>
<span className="v" id="v11" data-v="11" data-xref="يشوع 15:45; ملوك الثاني 1:2–3; صموئيل الأول 5:10"> وَمِنْهَا تَخْرُجُ إِلَى جَانِبِ عِقْرُونَ نَحْوَ الشِّمَالِ وَتَمْتَدُّ إِلَى شَكْرُونَ فَجَبَلِ الْبَعَلَةِ حَتَّى تَبْلُغَ يَبْنِئِيلَ وَتَنْتَهِيَ عِنْدَ الْبَحْرِ.</span>
<span className="v" id="v12" data-v="12" data-xref="العدد 34:6–7; يشوع 15:47; حزقيال 47:20"> أَمَّا الْحُدُودُ الْغَرْبِيَّةُ فَهِيَ شَوَاطِئُ الْبَحْرِ الْمُتَوَسِّطِ. هَذِهِ هِيَ حُدُودُ أَرْضِ سِبْطِ يَهُوذَا مِنْ جَمِيعِ جِهَاتِهَا بِحَسَبِ عَشَائِرِهِمْ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="العدد 13:30; التثنية 1:34–36; العدد 14:23–24"> وَوَهَبَ يَشُوعُ بِمُقْتَضَى أَمْرِ الرَّبِّ كَالَبَ بْنَ يَفُنَّةَ مِلْكاً قَرْيَةَ أَرْبَعَ أَبِي عَنَاقَ وَهِيَ حَبْرُونُ الْوَاقِعَةُ فِي وَسَطِ أَبْنَاءِ يَهُوذَا.</span>
<span className="v" id="v14" data-v="14" data-xref="القضاة 1:20; القضاة 1:10; العدد 13:22–23"> فَطَرَدَ كَالَبُ مِنْهَا الْعَنَاقِيِّينَ الثَّلاثَةَ: شِيشَايَ وَأَخِيمَانَ وَتَلْمَايَ مِنْ ذُرِّيَّةِ عَنَاقَ.</span>
<span className="v" id="v15" data-v="15" data-xref="يشوع 10:38; يشوع 10:3; القضاة 1:11–13"> وَتَقَدَّمَ مِنْ هُنَاكَ لِمُحَارَبَةِ أَهْلِ دَبِيرَ. وَكَانَتْ دَبِيرُ تُدْعَى قَبْلاً قَرْيَةَ سِفْرٍ.</span>
<span className="v" id="v16" data-v="16" data-xref="القضاة 1:12–13; القضاة 1:6"> فَقَالَ كَالَبُ: «مَنْ يُهَاجِمُ قَرْيَةَ سِفْرٍ وَيَسْتَوْلِي عَلَيْهَا، أُزَوِّجُهُ ابْنَتِي عَكْسَةَ»</span>
<span className="v" id="v17" data-v="17" data-xref="القضاة 3:9; القضاة 1:13; يشوع 14:6"> فَاسْتَوْلَى عَلَيْهَا عُثْنِيئِيلُ بْنُ قَنَازَ أَخُو كَالَبَ، فَزَوَّجَهُ ابْنَتَهُ عَكْسَةَ.</span>
<span className="v" id="v18" data-v="18" data-xref="التكوين 24:64; صموئيل الأول 25:23"> وَعِنْدَمَا أَقْبَلَتْ عَلَيْهِ حَثَّتْهُ عَلَى طَلَبِ حَقْلٍ مِنْ أَبِيهَا، فَتَرَجَّلَتْ عَنِ الْحِمَارِ، فَسَأَلَهَا كَالَبُ: «مَالَكِ؟»</span>
<span className="v" id="v19" data-v="19" data-xref="القضاة 1:14–15; صموئيل الأول 25:27; التثنية 33:7"> فَأَجَابَتْهُ: «اصْنَعْ مَعِي مَعْرُوفاً، فَأَنْتَ قَدْ وَهَبْتَنِي أَرْضاً قَاحِلَةً، فَأَعْطِنِي أَيْضاً يَنَابِيعَ مَاءٍ». فَأَعْطَاهَا السَّوَاقِيَ الْعُلْيَا وَالسَّوَاقِيَ السُّفْلَى.</span>
<span className="v" id="v20" data-v="20" data-xref="التكوين 49:8–12; التثنية 33:7"> وَهَذِهِ هِيَ قُرْعَةُ سِبْطِ يَهُوذَا بِحَسَبِ عَشَائِرِهِمْ:</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التكوين 35:21; نحميا 11:25"> كَانَتِ الْمُدُنُ الْقَصِيَّةُ التَّابِعَةُ لِسِبْطِ يَهُوذَا جَنُوباً بِاتِّجَاهِ تُخُومِ أَدُومَ هِيَ: قَبْصِئِيلُ وَعِيدَرُ وَيَاجُورُ،</span>
<span className="v" id="v22" data-v="22"> وَقَيْنَةُ وَدِيمُونَةُ وَعَدْعَدَةُ،</span>
<span className="v" id="v23" data-v="23" data-xref="يشوع 12:22; العدد 33:37; التثنية 1:19"> وَقَادَشُ وَحَاصُورُ وَيِثْنَانُ،</span>
<span className="v" id="v24" data-v="24" data-xref="صموئيل الأول 23:14; صموئيل الأول 23:24; صموئيل الأول 23:19"> وَزِيفُ وَطَالَمُ وَبَعْلُوتُ،</span>
<span className="v" id="v25" data-v="25"> وَحَاصُورُ وَحَدَتَّةُ وَقَرْيُوتُ وَحَصْرُونُ الَّتِي هِيَ حَاصُورُ.</span>
<span className="v" id="v26" data-v="26" data-xref="أخبار الأيام الأول 4:28"> وَأَمَامُ وَشَمَاعُ وَمُولادَةُ،</span>
<span className="v" id="v27" data-v="27" data-xref="نحميا 11:26"> وَحَصَرُ جَدَّةَ وَحَشْمُونُ وَبَيْتُ فَالَطَ،</span>
<span className="v" id="v28" data-v="28" data-xref="التكوين 26:33; أخبار الأيام الأول 4:28; التكوين 21:31–33"> وَحَصَرُ شُوعَالَ وَبِئْرُ سَبْعٍ وَبِزْيُوتِيَةُ،</span>
<span className="v" id="v29" data-v="29" data-xref="يشوع 15:9–11; يشوع 19:3; أخبار الأيام الأول 4:29"> وَبَعَلَةُ وَعِيِّيمُ وَعَاصَمُ،</span>
<span className="v" id="v30" data-v="30" data-xref="يشوع 19:4; العدد 14:45; التثنية 1:44"> وَأَلْتُولَدُ وَكِسِيلُ وَحُرْمَةُ،</span>
<span className="v" id="v31" data-v="31" data-xref="صموئيل الأول 27:6; صموئيل الأول 30:1; أخبار الأيام الأول 12:1"> وَصِقْلَغُ وَمَدْمَنَّةُ وَسَنْسَنَّةُ،</span>
<span className="v" id="v32" data-v="32" data-xref="العدد 34:11; نحميا 11:29"> وَلَبَاوُتُ وَشِلْحِيمُ وَعَيْنُ وَرِمُّونُ. فَكَانَتْ فِي جُمْلَتِهَا تِسْعاً وَعِشْرِينَ مَدِينَةً مَعَ ضِيَاعِهَا.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="القضاة 13:25; القضاة 16:31; العدد 13:23"> أَمَّا مُدُنُ السَّهْلِ الْغَرْبِيِّ فَكَانَتْ: أَشْتَأُولَ وَصَرْعَةَ وَأَشْنَةَ،</span>
<span className="v" id="v34" data-v="34" data-xref="يشوع 12:17; يشوع 15:53"> وَزَانُوحَ وَعَيْنَ جَنِّيمَ وَتَفُّوحَ وَعَيْنَامَ</span>
<span className="v" id="v35" data-v="35" data-xref="صموئيل الأول 17:1; صموئيل الأول 22:1; يشوع 10:3"> وَيَرْمُوتَ وَعَدُلَّامَ وَسُوكُوهَ وَعَزِيقَةَ،</span>
<span className="v" id="v36" data-v="36" data-xref="صموئيل الأول 17:52"> وَشَعَرَايِمَ وَعَدِيتَايِمَ وَالْجُدَيْرَةَ وَجُدَيْرُوتَايِمَ. وَهِيَ فِي جُمْلَتِهَا أَرْبَعَ عَشْرَةَ مَدِينَةً مَعَ ضِيَاعِهَا.</span>
</div>
<div className="bp">

<span className="v" id="v38" data-v="38" data-xref="ملوك الثاني 14:7; القضاة 20:1; صموئيل الأول 10:17"> وَدِلْعَانَ وَالْمِصْفَاةَ وَيَقْتِيئِيلَ،</span>
<span className="v" id="v39" data-v="39" data-xref="يشوع 10:3; ملوك الثاني 22:1; يشوع 10:31–32"> وَلَخِيشَ وَبَصْقَةَ وَعَجْلُونَ،</span>
<span className="v" id="v40" data-v="40"> وَكَبُّونَ وَلَحْمَامَ وَكِتْلِيشَ.</span>
<span className="v" id="v41" data-v="41" data-xref="يشوع 10:21; يشوع 12:16; يشوع 10:10"> وَجُدَيْرُوتَ بَيْتَ دَاجُونَ وَنَعَمَةَ وَمَقِّيدَةَ. وَهِيَ فِي جُمْلَتِهَا سِتَّ عَشْرَةَ مَدِينَةً مَعَ ضِيَاعِهَا.</span>
<span className="v" id="v42" data-v="42" data-xref="يشوع 10:29; ملوك الثاني 8:22; يشوع 19:7"> وَلِبْنَةَ وعَاتَرَ وَعَاشَانَ،</span>
<span className="v" id="v43" data-v="43"> وَيَفْتَاحَ وَأَشْنَةَ وَنَصِيبَ،</span>
<span className="v" id="v44" data-v="44" data-xref="التكوين 38:5; ميخا 1:14–15; صموئيل الأول 23:1–14"> وَقَعِيلَةَ وَأَكْزِيبَ وَمَرِيشَةَ. وَهِي فِي جُمْلَتِهَا تِسْعُ مُدُنٍ مَعَ ضِيَاعِهَا.</span>
</div>
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="يشوع 13:3; عاموس 1:8; صموئيل الأول 6:17"> وَكَذَلِكَ عَقْرُونَ وَقُرَاهَا وَضِيَاعَهَا</span>
<span className="v" id="v46" data-v="46" data-xref="أخبار الأيام الثاني 26:6; عاموس 1:8; صموئيل الأول 5:1"> كَمَا اشْتَمَلَتْ حُدُودُ سِبْطِ يَهُوذَا مِنْ عَقْرُونَ غَرْباً، عَلَى كُلِّ الْمِنْطَقَةِ الْمُجَاوِرَةِ لأَشْدُودَ وَضِيَاعِهَا.</span>
<span className="v" id="v47" data-v="47" data-xref="يشوع 15:4; الخروج 23:31; عاموس 1:6–7"> فَكَانَتْ لَهُمْ أَشْدُودُ وَقُرَاهَا وَضِيَاعُهَا، وَغَزَّةُ وَقُرَاهَا وَضِيَاعُهَا، حَتَّى وَادِي مِصْرَ وَشَاطِئِ الْبَحْرِ الْمُتَوَسِّطِ.</span>
</div>
