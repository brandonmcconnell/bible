## يسوع يحول الماء لخمر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه يوحنا 4:46; الأمثال 31:10–12; الإنجيل كما دونه يوحنا 21:2"> وَفِي الْيَوْمِ الثَّالِثِ كَانَ عُرْسٌ فِي قَانَا بِمِنْطَقَةِ الْجَلِيلِ، وَكَانَتْ هُنَاكَ أُمُّ يَسُوعَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الإنجيل كما دونه يوحنا 1:40–49; الإنجيل كما دونه متى 10:40–42; الرسالة الأولى إلى مؤمني كورنثوس 7:39"> وَدُعِيَ إِلَى الْعُرْسِ أَيْضاً يَسُوعُ وَتَلامِيذُهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="المزامير 104:15; الرسالة إلى مؤمني فيلبي 4:6; إشعياء 24:11"> فَلَمَّا نَفَدَتِ الْخَمْرُ، قَالَتْ أُمُّ يَسُوعَ لَهُ: «لَمْ يَبْقَ عِنْدَهُمْ خَمْرٌ!»</span>
<span className="v" id="v4" data-v="4" data-xref="الإنجيل كما دونه يوحنا 7:6; الإنجيل كما دونه يوحنا 13:1; الإنجيل كما دونه يوحنا 7:30"> فَأَجَابَهَا: «مَا شَأْنُكِ بِي يَا امْرَأَةُ؟ سَاعَتِي لَمْ تَأْتِ بَعْدُ!»</span>
<span className="v" id="v5" data-v="5" data-xref="الإنجيل كما دونه يوحنا 15:14; التكوين 6:22; الرسالة إلى العبرانيين 11:8"> فَقَالَتْ أُمُّهُ لِلْخَدَمِ: «افْعَلُوا كُلَّ مَا يَأْمُرُكُمْ بِهِ».</span>
<span className="v" id="v6" data-v="6" data-xref="الإنجيل كما دونه يوحنا 3:25; الرسالة إلى العبرانيين 9:10; الرسالة إلى مؤمني أفسس 5:26"> وَكَانَتْ هُنَاكَ سِتَّةُ أَجْرَانٍ حَجَرِيَّةٍ، يَسْتَعْمِلُ الْيَهُودُ مَاءَهَا لِلتَّطَهُّرِ، يَسَعُ الْوَاحِدُ مِنْهَا مَا بَيْنَ مِكْيَالَيْنِ أَوْ ثَلاثَةٍ (أَيْ مَا بَيْنَ ثَمَانِينَ إِلَى مِئَةٍ وَعِشْرِينَ لِتْراً).</span>
<span className="v" id="v7" data-v="7" data-xref="ملوك الثاني 4:2–6; الإنجيل كما دونه يوحنا 2:5; الإنجيل كما دونه يوحنا 2:3"> فَقَالَ يَسُوعُ لِلْخَدَمِ: «امْلأُوا الأَجْرَانَ مَاءً». فَمَلأُوهَا حَتَّى كَادَتْ تَفِيضُ.</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه يوحنا 2:9; الأمثال 3:5–6; الرسالة إلى مؤمني روما 13:7"> ثُمَّ قَالَ لَهُمْ: «وَالآنَ اغْرِفُوا مِنْهَا وَقَدِّمُوا إِلَى رَئِيسِ الْوَلِيمَةِ!» فَفَعَلُوا.</span>
<span className="v" id="v9" data-v="9" data-xref="الإنجيل كما دونه يوحنا 4:46; الإنجيل كما دونه يوحنا 7:17; المزامير 119:100"> وَلَمَّا ذَاقَ رَئِيسُ الْوَلِيمَةِ الْمَاءَ الَّذِي كَانَ قَدْ تَحَوَّلَ إِلَى خَمْرٍ، وَلَمْ يَكُنْ يَعْرِفُ مَصْدَرَهُ، أَمَّا الْخَدَمُ الَّذِينَ قَدَّمُوهُ فَكَانُوا يَعْرِفُونَ، اسْتَدْعَى الْعَرِيسَ،</span>
<span className="v" id="v10" data-v="10" data-xref="رؤيا يوحنا 7:16–17; الأمثال 9:1–6; الإنجيل كما دونه لوقا 16:25"> وَقَالَ لَهُ: «النَّاسُ جَمِيعاً يُقَدِّمُونَ الْخَمْرَ الْجَيِّدَةَ أَوَّلاً، وَبَعْدَ أَنْ يَسْكَرَ الضُّيُوفُ يُقَدِّمُونَ لَهُمْ مَا كَانَ دُونَهَا جَوْدَةً. أَمَّا أَنْتَ فَقَدْ أَبْقَيْتَ الْخَمْرَ الْجَيِّدَةَ حَتَّى الآنَ!»</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه يوحنا 20:30–31; الإنجيل كما دونه يوحنا 1:14; الإنجيل كما دونه يوحنا 3:2"> هَذِهِ الْمُعْجِزَةُ هِيَ الآيَةُ الأُولَى الَّتِي أَجْرَاهَا يَسُوعُ فِي قَانَا بِالْجَلِيلِ، وَأَظْهَرَ مَجْدَهُ، فَآمَنَ بِهِ تَلامِيذُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه متى 12:46; الإنجيل كما دونه متى 13:55–56; الإنجيل كما دونه مرقس 6:3"> وَبَعْدَ هَذَا، نَزَلَ يَسُوعُ وَأُمُّهُ وَإخْوَتُهُ وَتَلامِيذُهُ إِلَى مَدِينَةِ كَفْرَنَاحُومَ، حَيْثُ أَقَامُوا بِضْعَةَ أَيَّامٍ.</span>
</div>
</div>

## يسوع يطهر الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه لوقا 2:41; الإنجيل كما دونه يوحنا 11:55; الإنجيل كما دونه يوحنا 2:23"> وَإِذِ اقْتَرَبَ عِيدُ الْفِصْحِ الْيَهُودِيُّ، صَعِدَ يَسُوعُ إِلَى أُورُشَلِيمَ،</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه لوقا 19:45–46; الإنجيل كما دونه مرقس 11:15; الإنجيل كما دونه مرقس 11:17"> فَوَجَدَ فِي الْهَيْكَلِ بَاعَةَ الْبَقَرِ وَالْغَنَمِ وَالْحَمَامِ، وَالصَّيَارِفَةَ جَالِسِينَ إِلَى مَوَائِدِهِمْ،</span>
<span className="v" id="v15" data-v="15" data-xref="الإنجيل كما دونه يوحنا 18:6; زكريا 4:6; الرسالة الثانية إلى مؤمني كورنثوس 10:4"> فَجَدَلَ سَوْطاً مِنْ حِبَالٍ، وَطَرَدَهُمْ جَمِيعاً مِنَ الْهَيْكَلِ، مَعَ الْغَنَمِ وَالْبَقَرِ، وَبَعْثَرَ نُقُودَ الصَّيَارِفَةِ وَقَلَبَ مَنَاضِدَهُمْ،</span>
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه متى 21:13; إرميا 7:11; الإنجيل كما دونه مرقس 11:17"> وَقَالَ لِبَائِعِي الْحَمَامِ: «أَخْرِجُوا هذِهِ مِنْ هُنَا. لَا تَجْعَلُوا بَيْتَ أَبِي بَيْتاً لِلتِّجَارَةِ!»</span>
<span className="v" id="v17" data-v="17" data-xref="المزامير 69:9; المزامير 119:139"> فَتَذَكَّرَ تَلامِيذُهُ أَنَّهُ جَاءَ فِي الْكِتَابِ: «الْغَيْرَةُ عَلَى بَيْتِكَ تَأْكُلُنِي».</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه متى 21:23; الإنجيل كما دونه مرقس 8:11; الإنجيل كما دونه مرقس 11:27–28"> فَتَصَدَّى الْيَهُودُ لِيَسُوعَ وَقَالُوا لَهُ: «هَاتِ آيَةً تُثْبِتُ سُلْطَتَكَ لِفِعْلِ مَا فَعَلْتَ!»</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه مرقس 14:58; الإنجيل كما دونه مرقس 15:29; الإنجيل كما دونه متى 27:40"> أَجَابَهُمْ يَسُوعُ: «اهْدِمُوا هَذَا الْهَيْكَلَ، وَفِي ثَلاثَةِ أَيَّامٍ أُقِيمُهُ».</span>
<span className="v" id="v20" data-v="20"> فَقَالَ لَهُ الْيَهُودُ: «اقْتَضَى بِنَاءُ هَذَا الْهَيْكَلِ سِتَّةً وَأَرْبَعِينَ عَاماً، فَهَلْ تُقِيمُهُ أَنْتَ فِي ثَلاثَةِ أَيَّامٍ؟»</span>
<span className="v" id="v21" data-v="21" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 6:19; الرسالة الأولى إلى مؤمني كورنثوس 3:16; الرسالة إلى مؤمني أفسس 2:20–22"> وَلكِنَّهُ كَانَ يُشِيرُ إِلَى هَيْكَلِ جَسَدِهِ.</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه يوحنا 12:16; الإنجيل كما دونه يوحنا 16:4; الإنجيل كما دونه لوقا 24:7–8"> فَلَمَّا قَامَ مِنْ بَيْنِ الأَمْوَاتِ فِيمَا بَعْدُ تَذَكَّرَ تَلامِيذُهُ قَوْلَهُ هَذَا، فَآمَنُوا بِالْكِتَابِ وَبِالْكَلامِ الَّذِي قَالَهُ يَسُوعُ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه يوحنا 7:31; الإنجيل كما دونه لوقا 8:13; الإنجيل كما دونه يوحنا 6:14"> وَبَيْنَمَا كَانَ فِي أُورُشَلِيمَ فِي عِيدِ الْفِصْحِ، آمَنَ بِاسْمِهِ كَثِيرُونَ إِذْ شَهِدُوا الآيَاتِ الَّتِي أَجْرَاهَا.</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه يوحنا 6:64; الرسالة إلى العبرانيين 4:13; أعمال الرسل 1:24"> وَلَكِنَّهُ هُوَ لَمْ يَأْتَمِنْهُمْ عَلَى نَفْسِهِ، لأَنَّهُ كَانَ يَعْرِفُ الْجَمِيعَ</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه يوحنا 13:11; الإنجيل كما دونه متى 9:4; الإنجيل كما دونه يوحنا 6:64"> وَلَمْ يَكُنْ بِحَاجَةٍ إِلَى مَنْ يَشْهَدُ لَهُ عَنِ الإِنْسَانِ، لأَنَّهُ يَعْرِفُ دَخِيلَةَ الإِنْسَانِ.</span>
</div>
