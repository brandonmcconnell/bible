## شفاء مشلول عند البركة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التثنية 16:16; الخروج 23:14–17; الخروج 34:23"> وَبَعْدَ ذلِكَ صَعِدَ يَسُوعُ إِلَى أُورُشَلِيمَ فِي أَحَدِ الأَعْيَادِ الْيَهُودِيَّةِ،</span>
<span className="v" id="v2" data-v="2" data-xref="نحميا 12:39; نحميا 3:1; نحميا 3:32"> وَكَانَ بِالْقُرْبِ مِنْ بَابِ الْغَنَمِ فِي أُورُشَلِيمَ بِرْكَةٌ اسْمُهَا بِالْعِبْرِيَّةِ بَيْتُ حِسْدَا، حَوْلَهَا خَمْسُ قَاعَاتٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="الإنجيل كما دونه مرقس 3:1–4; الإنجيل كما دونه لوقا 7:22; الإنجيل كما دونه متى 15:30"> يَرْقُدُ فِيهَا جَمْعٌ كَبِيرٌ مِنَ الْمَرْضَى مِنْ عُمْيَانٍ وَعُرْجٍ وَمَشْلُولِينَ، يَنْتَظِرُونَ أَنْ تَتَحَرَّكَ مِيَاهُ الْبِرْكَةِ،</span>
<span className="v" id="v4" data-v="4"> لأَنَّ مَلاكاً كَانَ يَأْتِي مِنْ حِينٍ لِآخَرَ إِلَى الْبِرْكَةِ وَيُحَرِّكُ مَاءَهَا، فَكَانَ الَّذِي يَنْزِلُ أَوَّلاً يُشْفَى، مَهْمَا كَانَ مَرَضُهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="أعمال الرسل 3:2; الإنجيل كما دونه لوقا 8:43; الإنجيل كما دونه مرقس 9:21"> وَكَانَ عِنْدَ الْبِرْكَةِ مَرِيضٌ مُنْذُ ثَمَانٍ وَثَلاثِينَ سَنَةً،</span>
<span className="v" id="v6" data-v="6" data-xref="الإنجيل كما دونه لوقا 18:41; الرسالة إلى العبرانيين 4:15; الرسالة إلى العبرانيين 4:13"> رَآهُ يَسُوعُ رَاقِداً هُنَاكَ فَعَرَفَ أَنَّ مُدَّةً طَوِيلَةً انْقَضَتْ وَهُوَ عَلَى تِلْكَ الْحَالِ، فَسَأَلَهُ: «أَتُرِيدُ أَنْ تُشْفَى؟»</span>
<span className="v" id="v7" data-v="7" data-xref="المزامير 72:12; التثنية 32:36; الرسالة الأولى إلى مؤمني كورنثوس 9:24"> فَأَجَابَهُ الْمَرِيضُ: «يَا سَيِّدُ، لَيْسَ لِي إِنْسَانٌ يُلْقِينِي فِي الْبِرْكَةِ مَتَى تَحَرَّكَ الْمَاءُ. وَكَمْ مِنْ مَرَّةٍ حَاوَلْتُ النُّزُولَ، فَكَانَ غَيْرِي يَنْزِلُ قَبْلِي دَائِماً».</span>
<span className="v" id="v8" data-v="8" data-xref="الإنجيل كما دونه لوقا 5:24; الإنجيل كما دونه متى 9:6; الإنجيل كما دونه مرقس 2:11"> فَقَالَ لَهُ يَسُوعُ: «قُمِ احْمِلْ فِرَاشَكَ وَامْشِ».</span>
<span className="v" id="v9" data-v="9" data-xref="أعمال الرسل 3:7–8; الإنجيل كما دونه يوحنا 9:14; الإنجيل كما دونه مرقس 3:2–4"> وَفِي الْحَالِ شُفِيَ الرَّجُلُ وَحَمَلَ فِرَاشَهُ وَمَشَى. وَكَانَ ذَلِكَ يَوْمَ سَبْتٍ.</span>
<span className="v" id="v10" data-v="10" data-xref="إرميا 17:21; الإنجيل كما دونه لوقا 6:2; الإنجيل كما دونه يوحنا 9:16"> فَقَالَ الْيَهُودُ لِلرَّجُلِ الَّذِي شُفِيَ: «الْيَوْمَ سَبْتٌ. لَا يَحِلُّ لَكَ أَنْ تَحْمِلَ فِرَاشَكَ!»</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه مرقس 2:9–11; الإنجيل كما دونه يوحنا 9:16"> فَأَجَابَهُمْ: «الَّذِي أَعَادَ إِلَيَّ الصِّحَّةَ هُوَ قَالَ لِي: احْمِلْ فِرَاشَكَ وَامْشِ».</span>
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه متى 21:23; القضاة 6:29; الرسالة إلى مؤمني روما 10:2"> فَسَأَلُوهُ: «وَمَنْ هُوَ الَّذِي قَالَ لَكَ: احْمِلْ فِرَاشَكَ وَامْشِ؟»</span>
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه لوقا 4:30; الإنجيل كما دونه لوقا 24:31; الإنجيل كما دونه يوحنا 8:59"> وَلكِنَّ الْمَرِيضَ الَّذِي شُفِيَ لَمْ يَكُنْ يَعْرِفُ مَنْ هُوَ، لأَنَّ يَسُوعَ كَانَ قَدِ ابْتَعَدَ، إِذْ كَانَ فِي الْمَكَانِ جَمْعٌ.</span>
<span className="v" id="v14" data-v="14" data-xref="الإنجيل كما دونه يوحنا 8:11; اللاويين 26:23–24; المزامير 118:18"> وَبَعْدَ ذَلِكَ وَجَدَهُ يَسُوعُ فِي الْهَيْكَلِ، فَقَالَ لَهُ: «هَا أَنْتَ قَدْ عُدْتَ صَحِيحاً فَلا تَرْجِعْ إِلَى الْخَطِيئَةِ لِئَلّا يُصِيبَكَ مَا هُوَ أَسْوَأُ!»</span>
<span className="v" id="v15" data-v="15" data-xref="الإنجيل كما دونه يوحنا 9:11–12; الإنجيل كما دونه يوحنا 4:29; الإنجيل كما دونه يوحنا 9:25"> فَلَمَّا عَرَفَ الرَّجُلُ أَنَّ يَسُوعَ هُوَ الَّذِي شَفَاهُ، أَسْرَعَ يُخْبِرُ الْيَهُودَ بِذلِكَ.</span>
</div>

## سلطان الابن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه يوحنا 10:39; الإنجيل كما دونه يوحنا 7:19–20; الإنجيل كما دونه مرقس 3:6"> فَأَخَذَ الْيَهُودُ يُضَايِقُونَ يَسُوعَ لأَنَّهُ كَانَ يَعْمَلُ هذِهِ الأَعْمَالَ يَوْمَ السَّبْتِ.</span>
<span className="v" id="v17" data-v="17" data-xref="الإنجيل كما دونه يوحنا 9:4; الإنجيل كما دونه يوحنا 14:10; الرسالة إلى مؤمني كولوسي 1:16"> وَلكِنَّ يَسُوعَ قَالَ لَهُمْ: «مَازَالَ أَبِي يَعْمَلُ إِلَى الآنَ. وَأَنَا أَيْضاً أَعْمَلُ!»</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه يوحنا 10:30; الإنجيل كما دونه يوحنا 10:33; الإنجيل كما دونه يوحنا 19:7"> لِهَذَا ازْدَادَ سَعْيُ الْيَهُودِ إِلَى قَتْلِهِ، لَيْسَ فَقَطْ لأَنَّهُ خَالَفَ سُنَّةَ السَّبْتِ، بَلْ أَيْضاً لأَنَّهُ قَالَ إِنَّ اللهَ أَبُوهُ، مُسَاوِياً نَفْسَهُ بِاللهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الإنجيل كما دونه يوحنا 12:49; الإنجيل كما دونه يوحنا 8:28; الإنجيل كما دونه يوحنا 14:10"> فَقَالَ لَهُمْ يَسُوعُ: «الْحَقَّ الْحَقَّ أَقُولُ لَكُمْ إِنَّ الاِبْنَ لَا يَقْدِرُ أَنْ يَفْعَلَ شَيْئاً مِنْ تِلْقَاءِ نَفْسِهِ، بَلْ يَفْعَلُ مَا يَرَى الآبَ يَفْعَلُهُ. فَكُلُّ مَا يَعْمَلُهُ الآبُ، يَعْمَلُهُ الاِبْنُ كَذلِكَ،</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه يوحنا 3:35; الإنجيل كما دونه متى 3:17; الإنجيل كما دونه يوحنا 17:26"> لأَنَّ الآبَ يُحِبُّ الاِبْنَ، وَيُرِيهِ جَمِيعَ مَا يَفْعَلُهُ، وَسَيُرِيهِ أَيْضاً أَعْمَالاً أَعْظَمَ مِنْ هَذَا الْعَمَلِ، فَتُدْهَشُونَ.</span>
<span className="v" id="v21" data-v="21" data-xref="الرسالة إلى مؤمني روما 8:11; التثنية 32:39; الإنجيل كما دونه يوحنا 11:25"> فَكَمَا يُقِيمُ الآبُ الْمَوْتَى وَيُحْيِيهِمْ، كَذلِكَ يُحْيِي الاِبْنُ مَنْ يَشَاءُ.</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه يوحنا 5:27; أعمال الرسل 17:31; الرسالة الثانية إلى مؤمني كورنثوس 5:10"> وَالآبُ لَا يُحَاكِمُ أَحَداً، بَلْ أَعْطَى الاِبْنَ سُلْطَةَ الْقَضَاءِ كُلَّهَا،</span>
<span className="v" id="v23" data-v="23" data-xref="رسالة يوحنا الأولى 2:23; الإنجيل كما دونه لوقا 10:16; الإنجيل كما دونه متى 11:27"> لِيُكْرِمَ الْجَمِيعُ الاِبْنَ كَمَا يُكْرِمُونَ الآبَ. وَمَنْ لَا يُكْرِمِ الاِبْنَ لَا يُكْرِمِ الآبَ الَّذِي أَرْسَلَهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="الإنجيل كما دونه يوحنا 3:36; الإنجيل كما دونه يوحنا 6:40; رسالة يوحنا الأولى 5:11–13"> الْحَقَّ الْحَقَّ أَقُولُ لَكُمْ إِنَّ مَنْ يَسْمَعْ كَلامِي وَيُؤْمِنْ بِالَّذِي أَرْسَلَنِي تَكُنْ لَهُ الْحَيَاةُ الأَبَدِيَّةُ، وَلا يُحَاكَمْ فِي الْيَوْمِ الأَخِيرِ، لأَنَّهُ قَدِ انْتَقَلَ مِنَ الْمَوْتِ إِلَى الْحَيَاةِ.</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه يوحنا 5:28; الرسالة إلى مؤمني أفسس 5:14; الرسالة إلى مؤمني أفسس 2:5"> الْحَقَّ الْحَقَّ أَقُولُ لَكُمْ: إِنَّ السَّاعَةَ الَّتِي يَسْمَعُ فِيهَا الأَمْوَاتُ صَوْتَ ابْنِ اللهِ سَتَأْتِي بَلْ هِيَ الآنَ وَالَّذِينَ يَسْمَعُونَهُ يَحْيَوْنَ.</span>
<span className="v" id="v26" data-v="26" data-xref="الإنجيل كما دونه يوحنا 11:26; الإنجيل كما دونه يوحنا 1:4; الخروج 3:14"> لأَنَّهُ كَمَا أَنَّ لِلآبِ حَيَاةً فِي ذَاتِهِ، فَقَدْ أَعْطَى الاِبْنَ أَيْضاً أَنْ تَكُونَ لَهُ حَيَاةٌ فِي ذَاتِهِ،</span>
<span className="v" id="v27" data-v="27" data-xref="أعمال الرسل 10:42; الإنجيل كما دونه يوحنا 5:22; أعمال الرسل 17:31"> وَأَعْطَاهُ سُلْطَةً أَنْ يَدِينَ، لأَنَّهُ ابْنُ الإِنْسَانِ.</span>
<span className="v" id="v28" data-v="28" data-xref="رؤيا يوحنا 20:12; الرسالة الأولى إلى مؤمني تسالونيكي 4:14–17; إشعياء 26:19"> لَا تَتَعَجَّبُوا مِنْ هَذَا: فَسَوْفَ تَأْتِي سَاعَةٌ يَسْمَعُ فِيهَا جَمِيعُ مَنْ فِي الْقُبُورِ صَوْتَهُ،</span>
<span className="v" id="v29" data-v="29" data-xref="دانيال 12:2–3; الرسالة إلى مؤمني روما 2:6–10; أعمال الرسل 24:15"> فَيَخْرُجُونَ مِنْهَا: فَالَّذِينَ عَمِلُوا الصَّالِحَاتِ يَخْرُجُونَ فِي الْقِيَامَةِ الْمُؤَدِّيَةِ إِلَى الْحَيَاةِ، وَأَمَّا الَّذِينَ عَمِلُوا السَّيِّئَاتِ فَفِي الْقِيَامَةِ الْمُؤَدِّيَةِ إِلَى الدَّيْنُونَةِ.</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه يوحنا 6:38; الإنجيل كما دونه يوحنا 5:19; الإنجيل كما دونه يوحنا 4:34"> وَأَنَا لَا يُمْكِنُ أَنْ أَفْعَلَ شَيْئاً مِنْ تِلْقَاءِ ذَاتِي، بَلْ أَحْكُمُ حَسْبَمَا أَسْمَعُ، وَحُكْمِي عَادِلٌ، لأَنِّي لَا أَسْعَى لِتَحْقِيقِ إِرَادَتِي بَلْ إِرَادَةِ الَّذِي أَرْسَلَنِي.</span>
</div>

## الآب يشهد للابن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الأمثال 27:2; الإنجيل كما دونه يوحنا 8:54; رؤيا يوحنا 3:14"> لَوْ كُنْتُ أَشْهَدُ لِنَفْسِي، لَكَانَتْ شَهَادَتِي غَيْرَ صَادِقَةٍ،</span>
<span className="v" id="v32" data-v="32" data-xref="الإنجيل كما دونه يوحنا 1:33; الإنجيل كما دونه يوحنا 12:28–30; الإنجيل كما دونه يوحنا 5:36–37"> وَلكِنْ غَيْرِي يَشْهَدُ لِي، وَأَنَا أَعْلَمُ أَنَّ شَهَادَتَهُ لِي هِيَ حَقٌّ.</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه يوحنا 1:29–34; الإنجيل كما دونه يوحنا 3:26–36; الإنجيل كما دونه يوحنا 1:15–27"> وَقَدْ بَعَثْتُمْ رُسُلاً إِلَى يُوحَنَّا فَشَهِدَ لِلْحَقِّ.</span>
<span className="v" id="v34" data-v="34" data-xref="الإنجيل كما دونه لوقا 19:41–42; الإنجيل كما دونه يوحنا 20:31; رسالة يوحنا الأولى 5:9"> وَأَنَا أَقُولُ هَذَا لَا لأَنِّي أَعْتَمِدُ عَلَى شَهَادَةِ إِنْسَانٍ، بَلْ مِنْ أَجْلِ خَلاصِكُمْ،</span>
<span className="v" id="v35" data-v="35" data-xref="رسالة بطرس الثانية 1:19; الإنجيل كما دونه يوحنا 1:7–8; الإنجيل كما دونه متى 11:11"> فَقَدْ كَانَ يُوحَنَّا مِصْبَاحاً مُتَوَهِّجاً مُضِيئاً، وَشِئْتُمْ أَنْ تَسْتَمْتِعُوا بِنُورِهِ فَتْرَةً مِنَ الزَّمَنِ.</span>
<span className="v" id="v36" data-v="36" data-xref="الإنجيل كما دونه يوحنا 10:25; الإنجيل كما دونه يوحنا 10:37–38; الإنجيل كما دونه يوحنا 15:24"> وَلكِنَّ لِي شَهَادَةً أَعْظَمَ مِنْ شَهَادَةِ يُوحَنَّا، وَهِيَ شَهَادَةُ الأَعْمَالِ الَّتِي كَلَّفَنِي الآبُ أَنْ أُنْجِزَهَا وَالَّتِي أَعْمَلُهَا، فَهِيَ تَشْهَدُ لِي مُبَيِّنَةً أَنَّ الآبَ أَرْسَلَنِي،</span>
<span className="v" id="v37" data-v="37" data-xref="الإنجيل كما دونه يوحنا 8:18; الإنجيل كما دونه يوحنا 1:18; الإنجيل كما دونه متى 3:17"> وَالآبُ الَّذِي أَرْسَلَنِي هُوَ نَفْسُهُ أَيْضاً يَشْهَدُ لِي. وَأَنْتُمْ لَمْ تَسْمَعُوا صَوْتَهُ قَطُّ، وَلا رَأَيْتُمْ هَيْئَتَهُ،</span>
