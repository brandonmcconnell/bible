## سكب العطر على يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الإنجيل كما دونه مرقس 14:3–8; الإنجيل كما دونه يوحنا 11:1; الإنجيل كما دونه لوقا 24:50"> وَقَبْلَ الْفِصْحِ بِسِتَّةِ أَيَّامٍ جَاءَ يَسُوعُ إِلَى بَيْتَ عَنْيَا، بَلْدَةِ لِعَازَرَ الَّذِي أَقَامَهُ مِنْ بَيْنِ الأَمْوَاتِ.</span>
<span className="v" id="v2" data-v="2" data-xref="الإنجيل كما دونه لوقا 22:27; الإنجيل كما دونه لوقا 10:38–42; رؤيا يوحنا 3:20"> فَأُقِيمَتْ لَهُ وَلِيمَةُ عَشَاءٍ، وَأَخَذَتْ مَرْثَا تَخْدِمُ، وَكَانَ لِعَازَرُ أَحَدَ الْمُتَّكِئِينَ مَعَهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="الإنجيل كما دونه مرقس 14:3–9; الإنجيل كما دونه يوحنا 11:2; الإنجيل كما دونه لوقا 10:38–39"> فَأَخَذَتْ مَرْيَمُ مَناً (أَيْ ثُلْثَ لِتْرٍ) مِنْ عِطْرِ النَّارِدِينِ الْخَالِصِ الْغَالِي الثَّمَنِ، وَدَهَنَتْ بِهِ قَدَمَيْ يَسُوعَ، ثُمَّ مَسَحَتْهُمَا بِشَعْرِهَا، فَمَلأَتِ الرَّائِحَةُ الطَّيِّبَةُ أَرْجَاءَ الْبَيْتِ كُلِّهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="الإنجيل كما دونه متى 10:4; صموئيل الأول 17:28–29; الإنجيل كما دونه يوحنا 13:2"> فَقَالَ أَحَدُ التَّلامِيذِ، وَهُوَ يَهُوذَا الإِسْخَريُوطِيُّ، الَّذِي كَانَ سَيَخُونُ يَسُوعَ:</span>
<span className="v" id="v5" data-v="5" data-xref="الإنجيل كما دونه لوقا 18:22; الإنجيل كما دونه لوقا 12:33; الإنجيل كما دونه مرقس 14:4–5"> «لِمَاذَا لَمْ يُبَعْ هَذَا الْعِطْرُ بِثَلاثِ مِئَةِ دِينَارٍ تُوَزَّعُ عَلَى الْفُقَرَاءِ؟»</span>
<span className="v" id="v6" data-v="6" data-xref="الإنجيل كما دونه يوحنا 13:29; المزامير 50:16–20; رسالة يعقوب 2:6"> وَلَمْ يَقُلْ هَذَا لأَنَّهُ كَانَ يَعْطِفُ عَلَى الْفُقَرَاءِ، بَلْ لأَنَّهُ كَانَ لِصّاً، فَقَدْ كَانَ أَمِيناً لِلصُّنْدُوقِ وَكَانَ يَخْتَلِسُ مِمَّا يُوْدَعُ فِيهِ.</span>
<span className="v" id="v7" data-v="7" data-xref="الإنجيل كما دونه متى 26:12; الإنجيل كما دونه مرقس 14:6; الإنجيل كما دونه متى 27:57–60"> فَأَجَابَهُ يَسُوعُ: «دَعْهَا! فَقَدِ احْتَفَظَتْ بِهَذَا الْعِطْرِ لِيَوْمِ دَفْنِي،</span>
<span className="v" id="v8" data-v="8" data-xref="التثنية 15:11; الإنجيل كما دونه مرقس 14:7; الإنجيل كما دونه متى 26:11"> لأَنَّ الْفُقَرَاءَ عِنْدَكُمْ فِي كُلِّ حِينٍ؛ أَمَّا أَنَا فَلَنْ أَكُونَ عِنْدَكُمْ فِي كُلِّ حِينٍ».</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="أعمال الرسل 4:14; الإنجيل كما دونه يوحنا 11:43–45; أعمال الرسل 3:10–11"> وَعَلِمَ كَثِيرُونَ مِنَ الْيَهُودِ أَنَّ يَسُوعَ فِي بَيْتَ عَنْيَا، فَجَاءُوا لَا لِيَرَوْا يَسُوعَ فَقَطْ، بَلْ لِيَرَوْا أَيْضاً لِعَازَرَ الَّذِي أَقَامَهُ مِنْ بَيْنِ الأَمْوَاتِ.</span>
<span className="v" id="v10" data-v="10" data-xref="الإنجيل كما دونه لوقا 16:31; الإنجيل كما دونه متى 2:16; الإنجيل كما دونه يوحنا 11:47–53"> فَقَرَّرَ رُؤَسَاءُ الْكَهَنَةِ أَنْ يَقْتُلُوا لِعَازَرَ أَيْضاً،</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه يوحنا 11:45; الإنجيل كما دونه يوحنا 12:18; أعمال الرسل 13:45"> لأَنَّ كَثِيرِينَ مِنَ الْيَهُودِ كَانُوا يَهْجُرُونَهُمْ بِسَبَبِهِ وَيُؤْمِنُونَ بِيَسُوعَ.</span>
</div>
</div>

## الدخول الانتصاري

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الإنجيل كما دونه مرقس 11:7–10; الإنجيل كما دونه متى 21:4–9; الإنجيل كما دونه لوقا 19:35–38"> وَفِي الْيَوْمِ التَّالِي، عَرَفَ الْجُمْهُورُ الْكَبِيرُ الَّذِي جَاءَ إِلَى العِيدِ أَنَّ يَسُوعَ قَادِمٌ إِلَى أُورُشَلِيمَ.</span>
<span className="v" id="v13" data-v="13" data-xref="المزامير 118:25–26; رؤيا يوحنا 15:3; الإنجيل كما دونه متى 21:9–11"> فَحَمَلُوا سَعَفَ النَّخْلِ وَخَرَجُوا لاِسْتِقْبَالِهِ هَاتِفِينَ: «أُوصَنَّا! تَبَارَكَ الآتِي بِاسْمِ الرَّبِّ! إِنَّهُ مَلِكُ إِسْرَائِيلَ!»</span>
<span className="v" id="v14" data-v="14" data-xref="زكريا 9:9; الإنجيل كما دونه مرقس 11:1–7; الإنجيل كما دونه متى 21:1–7"> وَوَجَدَ يَسُوعُ جَحْشاً فَرَكِبَ عَلَيْهِ، كَمَا قَدْ كُتِبَ:</span>
<span className="v" id="v15" data-v="15" data-xref="زكريا 9:9; إشعياء 62:11; صفنيا 3:16–17"> «لا تَخَافِي يَا بِنْتَ صِهْيَوْنَ، فَإِنَّ مَلِكَكِ قَادِمٌ إِلَيْكِ رَاكِباً عَلَى جَحْشِ أَتَانٍ».</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الإنجيل كما دونه يوحنا 2:22; الإنجيل كما دونه يوحنا 14:26; الإنجيل كما دونه يوحنا 7:39"> وَلَمْ يُدْرِكْ تَلامِيذُ يَسُوعَ أَوَّلَ الأَمْرِ أَنَّ هَذَا إِتْمَامٌ لِلنُّبُوءَةِ. وَلكِنْ بَعْدَمَا تَمَجَّدَ يَسُوعُ تَذَكَّرُوا أَنَّ الْكِتَابَ قَالَ هَذَا عَنْهُ، وَأَنَّهُمْ فَعَلُوا هَذَا مِنْ أَجْلِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="رؤيا يوحنا 1:2; رسالة يوحنا الأولى 5:9–12; الإنجيل كما دونه يوحنا 19:35"> وَالْجَمْعُ الَّذِينَ كَانُوا مَعَهُ حِينَ دَعَا لِعَازَرَ مِنَ الْقَبْرِ وَأَقَامَهُ مِنْ بَيْنِ الأَمْوَاتِ، كَانُوا يَشْهَدُونَ لَهُ بِذَلِكَ.</span>
<span className="v" id="v18" data-v="18" data-xref="الإنجيل كما دونه يوحنا 12:11"> وَلِذلِكَ خَرَجَتِ الْجُمُوعُ لاِسْتِقْبَالِهِ، لأَنَّهُمْ سَمِعُوا أَنَّهُ أَجْرَى تِلْكَ الآيَةَ.</span>
<span className="v" id="v19" data-v="19" data-xref="أعمال الرسل 17:6; المزامير 22:27; رسالة يوحنا الأولى 2:2"> فَقَالَ الْفَرِّيسِيُّونَ بَعْضُهُمْ لِبَعْضٍ: «أَرَأَيْتُمْ كَيْفَ أَنَّكُمْ لَمْ تَسْتَفِيدُوا شَيْئاً؟ هَا قَدِ انْطَلَقَ الْعَالَمُ كُلُّهُ وَرَاءَهُ!»</span>
</div>
</div>

## يسوع ينبئ بموته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه يوحنا 7:35; إشعياء 11:10; أعمال الرسل 17:4"> وَكَانَ بَيْنَ الَّذِينَ قَصَدُوا أُورُشَلِيمَ لِلْعِبَادَةِ فِي أَثْنَاءِ الْعِيدِ بَعْضُ الْيُونَانِيِّينَ،</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه متى 2:2; الإنجيل كما دونه لوقا 19:2–4; الإنجيل كما دونه يوحنا 6:40"> فَذَهَبُوا إِلَى فِيلِبُّسَ، وَهُوَ مِنْ بَيْتَ صَيْدَا فِي مِنْطَقَةِ الْجَلِيلِ، وَقَالُوا لَهُ: «يَا سَيِّدُ، نُرِيدُ أَنْ نَرَى يَسُوعَ».</span>
<span className="v" id="v22" data-v="22" data-xref="الإنجيل كما دونه مرقس 10:13–14; الإنجيل كما دونه لوقا 9:49–50; الإنجيل كما دونه يوحنا 1:40–41"> فَجَاءَ فِيلِبُّسُ وَأَخْبَرَ أَنْدَرَاوُسَ، ثُمَّ ذَهَبَا مَعاً وَأَخْبَرَا يَسُوعَ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الإنجيل كما دونه يوحنا 13:31–32; الإنجيل كما دونه يوحنا 17:1–5; إشعياء 60:9"> فَقَالَ يَسُوعُ لَهُمَا: «قَدِ اقْتَرَبَتْ سَاعَةُ تَمْجِيدِ ابْنِ الإِنْسَانِ.</span>
<span className="v" id="v24" data-v="24" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 15:36–38; إشعياء 53:10–12; الرسالة إلى العبرانيين 2:9–10"> الْحَقَّ الْحَقَّ أَقُولُ لَكُمْ: إِنَّ حَبَّةَ الْحِنْطَةِ تَبْقَى وَحِيدَةً إِنْ لَمْ تَقَعْ فِي الأَرْضِ وَتَمُتْ. أَمَّا إِذَا مَاتَتْ، فَإِنَّهَا تُنْتِجُ حَبّاً كَثِيراً.</span>
<span className="v" id="v25" data-v="25" data-xref="الإنجيل كما دونه متى 10:39; الإنجيل كما دونه لوقا 9:23–24; الإنجيل كما دونه مرقس 8:35"> مَنْ يَتَمَسَّكْ بِحَيَاتِهِ، يَخْسَرْهَا. وَمَنْ نَبَذَهَا فِي هَذَا الْعَالَمِ يُوَفِّرُهَا لِلْحَيَاةِ الأَبَدِيَّةِ.</span>
<span className="v" id="v26" data-v="26" data-xref="الرسالة إلى مؤمني كولوسي 3:24; الرسالة إلى مؤمني غلاطية 1:10; صموئيل الأول 2:30"> مَنْ أَرَادَ أَنْ يَخْدِمَنِي فَلْيَتْبَعْنِي. وَحَيْثُ أَكُونُ أَنَا يَكُونُ خَادِمِي أَيْضاً. وَكُلُّ مَنْ يَخْدِمْنِي يُكْرِمْهُ أَبِي.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الإنجيل كما دونه يوحنا 12:23; الرسالة إلى العبرانيين 5:7; الإنجيل كما دونه متى 26:42"> نَفْسِي الآنَ مُضْطَرِبَةٌ، فَمَاذَا أَقُولُ؟ أَيُّهَا الآبُ أَنْقِذْنِي مِنَ السَّاعَةِ الْقَادِمَةِ عَلَيَّ؟ لا! فَمِنْ أَجْلِ هذِهِ السَّاعَةِ أَتَيْتُ.</span>
<span className="v" id="v28" data-v="28" data-xref="الإنجيل كما دونه يوحنا 13:31–32; الإنجيل كما دونه متى 26:42; رؤيا يوحنا 5:9–14"> أَيُّهَا الآبُ، مَجِّدِ اسْمَكَ!»<br />فَإِذَا صَوْتٌ مِنَ السَّمَاءِ يُجِيبُ: «قَدْ مَجَّدْتُهُ وَسَأُمَجِّدُهُ أَيْضاً».</span>
<span className="v" id="v29" data-v="29" data-xref="أيوب 37:2–5; رؤيا يوحنا 6:1; رؤيا يوحنا 14:2"> فَقَالَ بَعْضُ الْحَاضِرِينَ مِمَّنْ سَمِعُوا الصَّوْتَ: «هَذَا صَوْتُ رَعْدٍ!» وَلكِنَّ غَيْرَهُمْ قَالُوا: «حَدَّثَهُ مَلاكٌ».</span>
<span className="v" id="v30" data-v="30" data-xref="الإنجيل كما دونه يوحنا 11:42; الإنجيل كما دونه يوحنا 11:15; الإنجيل كما دونه يوحنا 5:34"> فَأَجَابَ يَسُوعُ: «لَمْ يَكُنْ هَذَا الصَّوْتُ لأَجْلِي بَلْ لأَجْلِكُمْ.</span>
<span className="v" id="v31" data-v="31" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 4:4; رسالة يوحنا الأولى 5:19; رسالة يوحنا الأولى 3:8"> الآنَ وَقْتُ الْحُكْمِ عَلَى هَذَا الْعَالَمِ! الآنَ يُطْرَدُ سَيِّدُ هَذَا الْعَالَمِ خَارِجاً!</span>
<span className="v" id="v32" data-v="32" data-xref="الإنجيل كما دونه يوحنا 3:14; الإنجيل كما دونه يوحنا 8:28; رسالة بطرس الأولى 2:24"> وَحِينَ أَرْتَفِعُ عَنِ الأَرْضِ أَجْذِبُ إِلَيَّ الْجَمِيعَ».</span>
<span className="v" id="v33" data-v="33" data-xref="الإنجيل كما دونه يوحنا 18:32; الإنجيل كما دونه يوحنا 21:19"> قَالَ هَذَا مُشِيراً إِلَى الْمِيتَةِ الَّتِي سَيَمُوتُهَا.</span>
<span className="v" id="v34" data-v="34" data-xref="حزقيال 37:24–25; إشعياء 9:7; دانيال 7:14"> فَقَالَ بَعْضُ الْحَاضِرِينَ: «عَلَّمَتْنَا الشَّرِيعَةُ أَنَّ الْمَسِيحَ يَبْقَى حَيًّا إِلَى الأَبَدِ، فَكَيْفَ تَقُولُ إِنَّ ابْنَ الإِنْسَانِ لابُدَّ أَنْ يُعَلَّقَ؟ مَنْ هُوَ ابْنُ الإِنْسَانِ هَذَا؟»</span>
<span className="v" id="v35" data-v="35" data-xref="الإنجيل كما دونه يوحنا 12:46; الإنجيل كما دونه يوحنا 16:16; الرسالة إلى مؤمني أفسس 5:8"> فَقَالَ لَهُمْ يَسُوعُ: «النُّورُ بَاقٍ مَعَكُمْ وَقْتاً قَصِيراً. فَوَاصِلُوا سَيْرَكُمْ مَادَامَ النُّورُ يُشْرِقُ عَلَيْكُمْ، لِئَلّا يُطْبِقَ عَلَيْكُمُ الظَّلامُ، فَإِنَّ الَّذِي يَمْشِي فِي الظَّلامِ لَا يَعْلَمُ أَيْنَ يَذْهَبُ.</span>
<span className="v" id="v36" data-v="36" data-xref="الرسالة إلى مؤمني أفسس 5:8; الإنجيل كما دونه يوحنا 8:59; أعمال الرسل 13:47–48"> آمِنُوا بِالنُّورِ مَادَامَ النُّورُ مَعَكُمْ، فَتَصِيرُوا أَبْنَاءَ النُّورِ».<br />وَبَعْدَمَا قَالَ يَسُوعُ هَذَا، ذَهَبَ وَأَخْفَى نَفْسَهُ عَنْهُمْ.</span>
