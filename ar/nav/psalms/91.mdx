<span className="v" id="v1" data-v="1" data-xref="المزامير 32:7; المزامير 27:5; المزامير 36:7"> الْمُحْتَمِي بِقُدْسِ أَقْدَاسِ الْعَلِيِّ، فِي ظِلِّ الْقَدِيرِ يَبِيتُ،</span>
<span className="v" id="v2" data-v="2" data-xref="المزامير 18:2; إشعياء 26:3–4; الأمثال 18:10"> أَقُولُ لِلرَّبِّ: أَنْتَ مَلْجَإِي وَحِصْنِي، إِلَهِي الَّذِي بِهِ وَثِقْتُ</span>
<span className="v" id="v3" data-v="3" data-xref="المزامير 124:7; المزامير 141:9; أيوب 5:10–22"> لأَنَّهُ يُنْقِذُكَ حَقّاً مِنْ فَخِّ الصَّيَّادِ وَمِنَ الْوَبَاءِ الْمُهْلِكِ.</span>
<span className="v" id="v4" data-v="4" data-xref="المزامير 61:4; المزامير 57:1; المزامير 17:8"> بِرِيشِهِ النَّاعِمِ يُظَلِّلُكَ، وَتَحْتَ أَجْنِحَتِهِ تَحْتَمِي، فَتَكُونُ لَكَ وُعُودُهُ الأَمِينَةُ تُرْساً وَمِتْرَاساً،</span>
<span className="v" id="v5" data-v="5" data-xref="إشعياء 43:2; أيوب 5:19–27; الأمثال 3:23–25"> فَلَا تَخَافُ مِنْ هَوْلِ اللَّيْلِ وَلَا مِنْ سَهْمٍ يَطِيرُ فِي النَّهَارِ.</span>
<span className="v" id="v6" data-v="6" data-xref="المزامير 121:5–6; العدد 16:48; الإنجيل كما دونه متى 24:6–7"> وَلَا مِنْ وَبَاءٍ يَسْرِي فِي الظَّلامِ، وَلَا مِنْ هَلاكٍ يُفْسِدُ فِي الظَّهِيرَةِ.</span>
<span className="v" id="v7" data-v="7" data-xref="الخروج 12:12–13; التكوين 7:23; المزامير 32:6"> يَتَسَاقَطُ عَنْ جَانِبِكَ أَلْفُ إِنْسَانٍ، وَعَنْ يَمِينِكَ عَشَرَةُ آلافٍ، وَأَنْتَ لَا يَمَسُّكَ سُوءٌ.</span>
<span className="v" id="v8" data-v="8" data-xref="المزامير 37:34; الأمثال 3:25–26; المزامير 92:11"> إِنَّمَا تُشَاهِدُ بِعَيْنَيْكَ مُعَاقَبَةَ الأَشْرَارِ.</span>
<span className="v" id="v9" data-v="9" data-xref="المزامير 71:3; المزامير 90:1; المزامير 91:1–2"> لأَنَّكَ قُلْتَ: الرَّبُّ مَلْجَإِي، وَاتَّخَذْتَ الْعَلِيَّ مَلاذاً،</span>
<span className="v" id="v10" data-v="10" data-xref="الأمثال 12:21; المزامير 121:7; التثنية 7:15"> فَلَنْ يُصِيبَكَ شَرٌّ وَلَنْ تَقْتَرِبَ بَلِيَّةٌ مِنْ مَسْكَنِكَ</span>
<span className="v" id="v11" data-v="11" data-xref="الإنجيل كما دونه لوقا 4:10–11; المزامير 34:7; الرسالة إلى العبرانيين 1:14"> فَإِنَّهُ يُوصِي مَلائِكَتَهُ بِكَ لِكَيْ يَحْفَظُوكَ فِي جَمِيعِ طُرُقِكَ.</span>
<span className="v" id="v12" data-v="12" data-xref="المزامير 37:24; الأمثال 3:23; الإنجيل كما دونه لوقا 4:11"> عَلَى أَيْدِيهِمْ يَحْمِلُونَكَ لِئَلّا تَصْدِمَ بِحَجَرٍ قَدَمَكَ.</span>
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه لوقا 10:19; دانيال 6:22; الإنجيل كما دونه مرقس 16:18"> تَطَأُ عَلَى الأَسَدِ وَالأَفْعَى، تَدُوسُ الشِّبْلَ وَالثُّعْبَانَ.</span>
<span className="v" id="v14" data-v="14" data-xref="المزامير 9:10; الإنجيل كما دونه يوحنا 14:23; رسالة يعقوب 1:12"> قَالَ الرَّبُّ: أُنَجِّيهِ لأَنَّهُ تَعَلَّقَ بِي. أُرَفِّعُهُ لأَنَّهُ عَرَفَ اسْمِي.</span>
<span className="v" id="v15" data-v="15" data-xref="إرميا 33:3; إشعياء 65:24; المزامير 50:15"> يَدْعُونِي فَأَسْتَجِيبُ لَهُ، أُرَافِقُهُ فِي الضِّيقِ، أُنْقِذُهُ وَأُكْرِمُهُ</span>
<span className="v" id="v16" data-v="16" data-xref="الأمثال 3:2; الأمثال 3:16; المزامير 21:4"> أُطِيلُ عُمْرَهُ، وَأُرِيهِ خَلاصِي.</span>
