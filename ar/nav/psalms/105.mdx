<span className="v" id="v1" data-v="1" data-xref="إشعياء 12:4; المزامير 145:11–12; الرسالة إلى مؤمني روما 10:13"> قَدِّمُوا الشُّكْرَ لِلرَّبِّ. ادْعُوا بِاسْمِهِ. عَرِّفُوا بِأَعْمَالِهِ بَيْنَ الشُّعُوبِ.</span>
<span className="v" id="v2" data-v="2" data-xref="المزامير 119:27; المزامير 98:5; إشعياء 12:5–6"> غَنُّوا لَهُ، اشْدُوا لَهُ. حَدِّثُوا بِكُلِّ عَجَائِبِهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="مراثي إرميا 3:25; الرسالة إلى مؤمني غلاطية 6:14; الرسالة الأولى إلى مؤمني كورنثوس 1:29"> تَبَاهَوْا بِاسْمِهِ الْقُدُّوسِ، لِتَفْرَحْ قُلُوبُ طَالِبِي الرَّبِّ.</span>
<span className="v" id="v4" data-v="4" data-xref="المزامير 27:8; صفنيا 2:2–3; أخبار الأيام الثاني 6:41"> اطْلُبُوا الرَّبَّ وَقُوَّتَهُ، الْتَمِسُوا وَجْهَهُ دَائِماً.</span>
<span className="v" id="v5" data-v="5" data-xref="المزامير 77:11; المزامير 119:13; التثنية 32:7"> اذْكُرُوا عَجَائِبَهُ الَّتِي صَنَعَهَا، مُعْجِزَاتِهِ وَأَحْكَامَهُ الَّتِي نَطَقَ بِها.</span>
<span className="v" id="v6" data-v="6" data-xref="إشعياء 44:1–2; الإنجيل كما دونه يوحنا 15:16; رسالة بطرس الأولى 2:9"> يَا ذُرِّيَّةَ إِبْرَاهِيمَ عَبْدِهِ، يَا بَنِي يَعْقُوبَ الَّذِينَ اخْتَارَهُمْ.</span>
<span className="v" id="v7" data-v="7" data-xref="إشعياء 26:9; التثنية 26:17–18; الخروج 20:2"> هُوَ الرَّبُّ إِلَهُنَا، أَحْكَامُهُ تَمْلأُ الأَرْضَ كُلَّهَا.</span>
<span className="v" id="v8" data-v="8" data-xref="التثنية 7:9; المزامير 106:45; المزامير 111:5"> لَمْ يَنْسَ عَهْدَهُ قَطُّ وَلَا وَعْدَهُ الَّذِي قَطَعَهُ إِلَى أَلْفِ جِيلٍ،</span>
<span className="v" id="v9" data-v="9" data-xref="التكوين 17:2; التكوين 22:16–18; التكوين 26:3"> الْعَهْدَ الَّذِي أَبْرَمَهُ مَعَ إِبْرَاهِيمَ، وَالْقَسَمَ الَّذِي أَقْسَمَ بِهِ لإِسْحَاقَ.</span>
<span className="v" id="v10" data-v="10" data-xref="التكوين 17:7–8; صموئيل الثاني 23:5; التكوين 28:13–15"> ثُمَّ ثَبَّتَهُ لِيَعْقُوبَ فَرِيضَةً، وَلإِسْرَائِيلَ مِيثَاقاً أَبَدِيًّا،</span>
<span className="v" id="v11" data-v="11" data-xref="التكوين 13:15; التكوين 15:18; المزامير 78:55"> قَائِلاً: «لَكَ أُعْطِي أَرْضَ كَنْعَانَ نَصِيبَ مِيرَاثٍ لَكُمْ».</span>
<span className="v" id="v12" data-v="12" data-xref="التكوين 34:30; التثنية 7:7; الرسالة إلى العبرانيين 11:9"> إِذْ كَانُوا قِلَّةً بَعْدُ، نَفَراً ضَئِيلاً مُتَغَرِّبِينَ فِي الأَرْضِ.</span>
<span className="v" id="v13" data-v="13"> مُتَنَقِّلِينَ مِنْ أُمَّةٍ إِلَى أُمَّةٍ، وَمِنْ مَمْلَكَةٍ إِلَى أُخْرَى.</span>
<span className="v" id="v14" data-v="14" data-xref="التكوين 35:5; الخروج 7:16–17; التكوين 26:14–33"> فَلَمْ يَدَعْ أَيَّ إِنْسَانٍ يَظْلِمُهُمْ، بَلْ وَبَّخَ مُلُوكاً مِنْ أَجْلِهِمْ.</span>
<span className="v" id="v15" data-v="15" data-xref="زكريا 2:8; التكوين 26:11; التكوين 20:7"> قَائِلاً: «لَا تَمَسُّوا مُسَحَائِي، وَلَا تُؤْذُوا أَنْبِيَائِي».</span>
<span className="v" id="v16" data-v="16" data-xref="حزقيال 4:16; التكوين 41:54; إشعياء 3:1"> ثُمَّ أَفْشَى مَجَاعَةً فِي الأَرْضِ، وَقَطَعَ الْخُبْزَ قِوَامَ طَعَامِهِمْ.</span>
<span className="v" id="v17" data-v="17" data-xref="أعمال الرسل 7:9; التكوين 37:36; التكوين 45:4–5"> لَكِنَّهُ أَرْسَلَ أَمَامَهُمْ يُوسُفَ، فَبِيعَ عَبْداً.</span>
<span className="v" id="v18" data-v="18" data-xref="أعمال الرسل 16:24; التكوين 39:20; التكوين 40:15"> آذَوْا بِالْقُيُودِ قَدَمَيْهِ، وَبِالْحَدِيدِ طَوَّقُوا عُنُقَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="أعمال الرسل 7:10; الأمثال 21:1; التكوين 40:20–21"> إِلَى أَنْ تَحَقَّقَ تَفْسِيرُهُ لِلأَحْلامِ فِي أَوَانِهِ، وَبَرْهَنَتْ كَلِمَةُ الرَّبِّ صِدْقَهُ.</span>
<span className="v" id="v20" data-v="20" data-xref="التكوين 41:14"> فَأَرْسَلَ الْمَلِكُ وَأَطْلَقَهُ، حَاكِمُ الشَّعْبِ حَرَّرَهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="التكوين 41:40–44; التكوين 45:8; التكوين 41:55"> أَقَامَهُ سَيِّداً عَلَى قَصْرِهِ، وَمُتَسَلِّطاً عَلَى جَمِيعِ أَمْلاكِهِ.</span>
<span className="v" id="v22" data-v="22" data-xref="إشعياء 19:11; التكوين 41:33; التكوين 41:38"> يَتَصَرَّفُ بِوُلاتِهِ حَسَبَ مَسَرَّتِهِ، وَيُلَقِّنُ شُيُوخَهُ الْحِكْمَةَ.</span>
<span className="v" id="v23" data-v="23" data-xref="أعمال الرسل 7:11–15; أعمال الرسل 13:17; المزامير 78:51"> ثُمَّ جَاءَ إِسْرَائِيلُ إِلَى أَرْضِ مِصْرَ. تَغَرَّبَ يَعْقُوبُ فِي بَلَدِ حَامٍ.</span>
<span className="v" id="v24" data-v="24" data-xref="التكوين 46:3; الخروج 1:7–9; أعمال الرسل 7:17"> فَكَثَّرَ اللهُ شَعْبَهُ، وَجَعَلَهُ أَقْوَى مِنْ أَعْدَائِهِ،</span>
<span className="v" id="v25" data-v="25" data-xref="أعمال الرسل 7:19; التكوين 15:13; الخروج 9:16"> الَّذِينَ حَوَّلَ قُلُوبَهُمْ لِيُبْغِضُوا شَعْبَهُ، وَيَكِيدُوا لِعَبِيدِهِ.</span>
<span className="v" id="v26" data-v="26" data-xref="الخروج 3:10; الخروج 29:5–46; الخروج 28:12"> عِنْدَئِذٍ أَرْسَلَ مُوسَى عَبْدَهُ وَهَارُونَ مُخْتَارَهُ.</span>
<span className="v" id="v27" data-v="27" data-xref="المزامير 78:43–51; نحميا 9:10–11; المزامير 105:23"> فَأَجْرَيَا بَيْنَهُمْ آيَاتِهِ، وَصَنَعَا عَجَائِبَ فِي مِصْرَ.</span>
<span className="v" id="v28" data-v="28" data-xref="المزامير 99:7; يوئيل 2:31; يوئيل 2:2"> بَعَثَ ظَلاماً، تَغَشَّتْ بِهِ الأَرْضُ، وَلَكِنَّ الْمِصْرِيِّينَ عَانَدُوا كَلِمَتَهُ.</span>
<span className="v" id="v29" data-v="29" data-xref="الخروج 7:20–21; المزامير 78:44; إشعياء 50:2"> حَوَّلَ مِيَاهَهُمْ إِلَى دَمٍ وَأَمَاتَ أَسْمَاكَهُمْ.</span>
<span className="v" id="v30" data-v="30" data-xref="رؤيا يوحنا 16:13–14; الخروج 8:3–14; المزامير 78:45"> فَاضَتْ أَرْضُهُمْ ضَفَادِعَ حَتَّى بَلَغَتْ مَخَادِعَ مُلُوكِهِمْ.</span>
<span className="v" id="v31" data-v="31" data-xref="الخروج 8:16–18; الخروج 8:21–24; إشعياء 7:18"> أَمَرَ فَأَقْبَلَ الذُّبَابُ وَالْبَعُوضُ فَانْتَشَرَ فِي كُلِّ أَرْضِهِمْ</span>
<span className="v" id="v32" data-v="32" data-xref="المزامير 78:47–48; الخروج 9:18–28; رؤيا يوحنا 8:7"> أَمْطَرَ عَلَيْهِمْ بَرَداً وَأَلْهَبَ أَرْضَهُمْ بِالْبُرُوقِ.</span>
<span className="v" id="v33" data-v="33" data-xref="المزامير 78:47; رؤيا يوحنا 9:4"> ضَرَبَ كُرُومَهُمْ وَتِينَهُمْ، وَهَشَّمَ كُلَّ أَشْجَارِهِمْ</span>
<span className="v" id="v34" data-v="34" data-xref="الخروج 10:12–15; المزامير 78:46; يوئيل 1:4–7"> أَمَرَ، فَتَوَافَدَ الْجَرَادُ الطَّيَّارُ وَالزَّحَّافُ بِأَعْدَادٍ لَا تُحْصَى،</span>
<span className="v" id="v35" data-v="35"> فَالْتَهَمَ كُلَّ عُشْبِ أَرْضِهِمْ، وَأَكَلَ ثِمَارَ حُقُولِهِمْ.</span>
<span className="v" id="v36" data-v="36" data-xref="التكوين 49:3; الخروج 12:29–30; المزامير 78:51"> ثُمَّ قَتَلَ الرَّبُّ جَمِيعَ أَبْكَارِ أَرْضِهِمْ، أَوَائِلَ ثِمَارِ خُصُوبَتِهِمْ جَمِيعاً.</span>
<span className="v" id="v37" data-v="37" data-xref="الخروج 12:35–36; الخروج 3:22; أعمال الرسل 13:17"> وَأَخْرَجَ شَعْبَهُ مُحَمَّلِينَ بِفِضَّةٍ وَذَهَبٍ، وَلَمْ يَتَعَثَّرْ فِي الْمَسِيرِ مِنْ عَشَائِرِهِمْ وَاحِدٌ.</span>
<span className="v" id="v38" data-v="38" data-xref="الخروج 12:33; الخروج 15:16; التكوين 35:5"> فَرِحَ أَهْلُ مِصْرَ بِخُرُوجِهِمْ لأَنَّ رُعْبَهُمْ غَلَبَ عَلَيْهِمْ.</span>
<span className="v" id="v39" data-v="39" data-xref="نحميا 9:12; المزامير 78:14; إشعياء 4:5"> نَشَرَ سَحَابَةً فَوْقَ شَعْبِهِ، غِطَاءً لَهُمْ، وَأَرْسَلَ نَاراً تُضِيءُ لَهُمْ لَيْلاً.</span>
<span className="v" id="v40" data-v="40" data-xref="المزامير 78:18; العدد 11:31–33; يشوع 5:12"> طَلَبُوا طَعَاماً فَبَعَثَ لَهُمْ طُيُورَ السَّلْوَى وَمِنْ خُبْزِ السَّمَاءِ أَشْبَعَهُمْ.</span>
<span className="v" id="v41" data-v="41" data-xref="العدد 20:11; الرسالة الأولى إلى مؤمني كورنثوس 10:4; الخروج 17:6"> فَلَقَ الصَّخْرَةَ وَفَجَّرَ مِنْهَا الْمِيَاهَ، فَجَرَتْ فِي الصَّحْرَاءِ كَأَنَّهَا نَهْرٌ.</span>
<span className="v" id="v42" data-v="42" data-xref="الخروج 2:24; الإنجيل كما دونه لوقا 1:72–73; التكوين 12:7"> لأَنَّهُ ذَكَرَ كَلِمَتَهُ الْمُقَدَّسَةَ الَّتِي وَعَدَ بِها إِبْرَاهِيمَ عَبْدَهُ.</span>
<span className="v" id="v43" data-v="43" data-xref="إشعياء 35:10; أعمال الرسل 7:36; إرميا 31:11–12"> وَهَكَذَا أَخْرَجَ شَعْبَهُ مِنْ مِصْرَ بِابْتِهَاجٍ وَمُخْتَارِيهِ بِتَرَانِيمِ الظَّفَرِ.</span>
<span className="v" id="v44" data-v="44" data-xref="التثنية 6:10–11; يشوع 24:13; المزامير 78:55"> وَوَهَبَهُمْ أَرَاضِيَ الأُمَمِ، فَامْتَلَكُوا غَلَّاتٍ تَعِبَتْ فِيهَا شُعُوبٌ أُخْرَى.</span>
<span className="v" id="v45" data-v="45" data-xref="التثنية 4:40; الرسالة إلى تيطس 2:14; الرسالة إلى مؤمني أفسس 2:8–10"> لِيُمَارِسُوا فَرَائِضَهُ وَيُطِيعُوا شَرَائِعَهُ. هَلِّلُويَا.</span>
