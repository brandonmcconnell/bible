<div className="hs">مَزْمُورٌ تَسْبِيحَةٌ لِيَوْمِ السَّبْتِ</div>
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المزامير 147:1; المزامير 107:1; الرسالة إلى مؤمني أفسس 5:19"> مَا أَحْسَنَ تَقْدِيمَ الشُّكْرِ لَكَ يَا رَبُّ وَالتَّرْنِيمَ لاسْمِكَ أَيُّهَا العَلِيُّ!</span>
<span className="v" id="v2" data-v="2" data-xref="مراثي إرميا 3:22–23; المزامير 89:1–2; الإنجيل كما دونه يوحنا 1:17"> مَا أَحْسَنَ أَنْ يُلْهَجَ بِرَحْمَتِكَ فِي الصَّبَاحِ، وَبِأَمَانَتِكَ فِي اللَّيَالِي،</span>
<span className="v" id="v3" data-v="3" data-xref="المزامير 33:2; أخبار الأيام الأول 13:8; نحميا 12:27"> عَلَى أَنْغَامِ الآلاتِ الْمُوسِيقِيَّةِ الْوَتَرِيَّةِ، وَعَلَى الرَّبَابِ وَأَلْحَانِ الْعُودِ الْعَذْبَةِ!</span>
<span className="v" id="v4" data-v="4" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 2:14; المزامير 8:6; الإنجيل كما دونه لوقا 1:47"> سَأُشِيدُ بِكُلِّ مَا عَمِلَتْهُ يَدَاكَ لأَنَّكَ يَا رَبُّ فَرَّحْتَنِي بِصَنِيعِكَ.</span>
<span className="v" id="v5" data-v="5" data-xref="الرسالة إلى مؤمني روما 11:33–34; إشعياء 55:8–9; المزامير 40:5"> يَا رَبُّ مَا أَعْظَمَ أَعْمَالَكَ! أَفْكَارُكَ عَمِيقَةٌ جِدّاً،</span>
<span className="v" id="v6" data-v="6" data-xref="المزامير 73:22; الرسالة الأولى إلى مؤمني كورنثوس 2:14; المزامير 94:8"> لَا يَعْرِفُهَا الْغَبِيُّ وَلَا يَفْهَمُهَا الْجَاهِلُ.</span>
<span className="v" id="v7" data-v="7" data-xref="المزامير 73:18–20; المزامير 37:38; المزامير 37:35–36"> إِذَا زَهَا الأَشْرَارُ كَالْعُشْبِ، وَأَزْهَرَ جَمِيعُ فَاعِلِي الإِثْمِ فَإِنَّهُمْ كَالْعُشْبِ يُبَادُونَ إِلَى الأَبَدِ.</span>
<span className="v" id="v8" data-v="8" data-xref="المزامير 83:18; أعمال الرسل 12:22–24; أعمال الرسل 12:1"> أَمَّا أَنْتَ يَا رَبُّ فَمُتَعَالٍ إِلَى الأَبَدِ.</span>
<span className="v" id="v9" data-v="9" data-xref="المزامير 37:20; المزامير 89:10; الرسالة الثانية إلى مؤمني تسالونيكي 1:7–9"> فَيَا رَبُّ، هَا هُمْ أَعْدَاؤُكَ يَهْلِكُونَ إِلَى الدَّهْرِ، إذْ يَتَبَدَّدُ جَمِيعُ فَاعِلِي الإِثْمِ.</span>
<span className="v" id="v10" data-v="10" data-xref="المزامير 23:5; المزامير 89:17; المزامير 45:7"> أَمَّا أَنَا فَتَرْفَعُ شَأْنِي كَمَا يَرْتَفِعُ قَرْنُ الثَّوْرِ الْوَحْشِيِّ، وَأَنْتَعِشُ كَمَنْ تَدَهَّنَ بِزَيْتٍ جَدِيدٍ</span>
<span className="v" id="v11" data-v="11" data-xref="المزامير 91:8; المزامير 54:7; المزامير 59:10"> وَتَنْظُرُ عَيْنَايَ عِقَابَ أَعْدَائِي الْمُتَرَبِّصِينَ لِي، وَتَسْمَعُ أُذُنَايَ بِمَصِيرِ فَاعِلِي الشَّرِّ الثَّائِرِينَ عَلَيَّ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إرميا 17:8; المزامير 1:3; المزامير 52:8"> الصِّدِّيقُ يَزْهُو كَالنَّخْلَةِ وَيَنْمُو كَالأَرْزِ فِي لُبْنَانَ.</span>
<span className="v" id="v13" data-v="13" data-xref="رسالة بطرس الثانية 3:18; إشعياء 61:3; الرسالة إلى مؤمني أفسس 3:17"> لأَنَّ الْمَغْرُوسِينَ فِي بَيْتِ الرَّبِّ يَزْدَهِرُونَ فِي دِيَارِ بَيْتِ إِلَهِنَا</span>
<span className="v" id="v14" data-v="14" data-xref="إشعياء 46:4; إرميا 17:8; المزامير 71:18"> يُثْمِرُونَ أَيْضاً فِي الشَّيْخُوخَةِ، وَيَظَلُّونَ مَوْفُورِي الْعَافِيَةِ وَالنُّضْرَةِ</span>
<span className="v" id="v15" data-v="15" data-xref="المزامير 18:2; الرسالة إلى مؤمني روما 9:14; المزامير 62:6"> لِيَشْهَدُوا أَنَّ الرَّبَّ مُسْتَقِيمٌ. إِنَّهُ صَخْرَتِي وَلَيْسَ فِيهِ سُوءٌ.</span>
