<span className="v" id="v1" data-v="1" data-xref="الأمثال 16:9; الأمثال 20:24; عزرا 6:22"> قَلْبُ الْمَلِكِ فِي يَدِ الرَّبِّ كَجَدَاوِلِ مِيَاهٍ يُمِيلُهُ حَيْثُمَا شَاءَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الأمثال 16:2; الأمثال 16:25; الإنجيل كما دونه لوقا 16:15"> جَمِيعُ تَصَرُّفَاتِ الإِنْسَانِ تَبْدُو نَقِيَّةً فِي عَيْنَيْ نَفْسِهِ، وَلَكِنَّ الرَّبَّ مُطَّلِعٌ عَلَى حَوَافِزِ الْقُلُوبِ.</span>
<span className="v" id="v3" data-v="3" data-xref="صموئيل الأول 15:22; هوشع 6:6; ميخا 6:6–8"> إِجْرَاءُ الْعَدْلِ وَالْحَقِّ أَكْثَرُ قُبُولاً عِنْدَ الرَّبِّ مِنَ الذَّبِيحَةِ.</span>
<span className="v" id="v4" data-v="4" data-xref="الأمثال 6:17; رسالة بطرس الأولى 5:5; إشعياء 2:17"> تَشَامُخُ الْعَيْنَيْنِ مِنْ غَطْرَسَةِ الْقَلْبِ، وَسِرَاجُ الأَشْرَارِ خَطِيئَةٌ.</span>
<span className="v" id="v5" data-v="5" data-xref="الأمثال 10:4; الأمثال 13:4; الرسالة إلى مؤمني أفسس 4:28"> خُطَطُ الْمُجْتَهِدِ تُفْضِي حَتْماً إِلَى الْخِصْبِ، وَالْعَجُولُ مَصِيرُهُ الْعَوَزُ.</span>
<span className="v" id="v6" data-v="6" data-xref="الأمثال 13:11; الأمثال 10:2; الأمثال 20:21"> ادِّخَارُ الْكُنُوزِ بِلِسَانٍ مُنَافِقٍ، دُخَانٌ مَتَلاشٍ وَفَخٌّ مُمِيتٌ.</span>
<span className="v" id="v7" data-v="7" data-xref="الأمثال 21:21; الرسالة إلى مؤمني أفسس 5:6; الأمثال 1:18–19"> جَوْرُ الأَشْرَارِ يَجْرُفُهُمْ لِرَفْضِهِمْ إِجْرَاءَ الْعَدْلِ.</span>
<span className="v" id="v8" data-v="8" data-xref="الأمثال 2:15; رسالة يوحنا الأولى 2:29; المزامير 14:2–3"> طَرِيقُ الْمُذْنِبِ مُعَوَّجَةٌ، أَمَّا تَصَرُّفُ الزَّكِيِّ فَقَوِيمٌ.</span>
<span className="v" id="v9" data-v="9" data-xref="الأمثال 19:13; الأمثال 21:19; الأمثال 27:15–16"> الإِقَامَةُ فِي رُكْنِ سَطْحٍ خَيْرٌ مِنْ مُشَاطَرَةِ بَيْتٍ مَعَ زَوْجَةٍ نَكِدَةٍ.</span>
<span className="v" id="v10" data-v="10" data-xref="ميخا 3:2–3; الأمثال 3:29; المزامير 36:4"> نَفْسُ الْمُنَافِقِ تَشْتَهِي الشَّرَّ، وَقَرِيبُهُ لَا يَحْظَى بِرِضَاهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="الأمثال 19:25; الأمثال 18:15; الأمثال 9:9"> إِذَا عُوقِبَ الْمُسْتَهْزِئُ صَارَ الْجَاهِلُ حَكِيماً، وَإِنْ أُرْشِدَ الْحَكِيمُ اكْتَسَبَ مَعْرِفَةً.</span>
<span className="v" id="v12" data-v="12" data-xref="الأمثال 14:11; المزامير 37:35–36; رسالة بطرس الثانية 3:6–7"> يَتَأَمَّلُ الصِّدِّيقُ فِي بَيْتِ الشِّرِّيرِ، (فَيَرَاهُ) يُلْقَى بِهِ إِلَى الْبَلايَا.</span>

## كنوز الحكيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الأمثال 28:27; الإنجيل كما دونه متى 6:14; الأمثال 1:28"> مَنْ أَصَمَّ أُذُنَهُ عَنْ صُرَاخِ الْمِسْكِينِ، يَصْرُخُ هُوَ أَيْضاً وَلا مِنْ مُجِيبٍ.</span>
<span className="v" id="v14" data-v="14" data-xref="الأمثال 18:16; الأمثال 19:6; الأمثال 17:8"> الْهَدِيَّةُ فِي الْخَفَاءِ تُخْمِدُ الْغَضَبَ، وَالرِّشْوَةُ فِي الْحِضْنِ تُسَكِّنُ السَّخَطَ.</span>
<span className="v" id="v15" data-v="15" data-xref="الأمثال 10:29; الإنجيل كما دونه متى 7:23; الإنجيل كما دونه متى 13:41–42"> الْحُكْمُ بِالْعَدْلِ فَرَحٌ لِلصِّدِّيقِ، وَرُعْبٌ لِفَاعِلِي الإِثْمِ.</span>
<span className="v" id="v16" data-v="16" data-xref="الرسالة إلى العبرانيين 10:38; المزامير 125:5; المزامير 49:14"> الرَّجُلُ الشَّارِدُ عَنْ طَرِيقِ الْمَعْرِفَةِ يَسْكُنُ بَيْنَ جَمَاعَةِ الْمَوْتَى.</span>
<span className="v" id="v17" data-v="17" data-xref="الأمثال 23:21; الرسالة الأولى إلى تيموثاوس 5:6; الرسالة الثانية إلى تيموثاوس 3:4"> عَاشِقُ اللَّذَّةِ فَقِيرٌ، وَالْمُوْلَعُ بِالْخَمْرِ وَالطِّيبِ لَا يَغْتَنِي.</span>
<span className="v" id="v18" data-v="18" data-xref="الأمثال 11:8; إشعياء 43:3–4; إشعياء 53:4–5"> الشِّرِّيرُ فِدَاءٌ عَنِ الصِّدِّيقِ، وَالْغَادِرُ عَنِ الْمُسْتَقِيمِينَ.</span>
<span className="v" id="v19" data-v="19" data-xref="الأمثال 21:9; المزامير 120:5–6; المزامير 55:6–7"> الإِقَامَةُ فِي أَرْضٍ مُقْفِرَةٍ خَيْرٌ مِنَ السُّكْنَى مَعَ امْرَأَةٍ مُشَاكِسَةٍ شَرِسَةٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="الإنجيل كما دونه متى 25:3–4; المزامير 112:3; الجامعة 7:11"> فِي بَيْتِ الْحَكِيمِ كُنُوزٌ وَزَيْتٌ مُدَّخَرَةٌ، أَمَّا الإِنْسَانُ الْجَاهِلُ فَيُتْلِفُ مَا لَدَيْهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="الإنجيل كما دونه متى 5:6; الرسالة الأولى إلى مؤمني كورنثوس 15:58; الأمثال 22:4"> مَنِ اتَّبَعَ الْعَدْلَ وَالرَّحْمَةَ يَلْقَى الْحَيَاةَ وَالْحَقَّ وَالْمَجْدَ.</span>
<span className="v" id="v22" data-v="22" data-xref="الجامعة 7:19; الأمثال 24:5; الجامعة 9:13–18"> الْحَكِيمُ يَتَسَلَّقُ سُورَ مَدِينَةِ الْجَبَابِرَةِ وَيُدَمِّرُ مَعْقِلَ اعْتِمَادِهِمْ</span>
<span className="v" id="v23" data-v="23" data-xref="الأمثال 13:3; الأمثال 18:21; رسالة يعقوب 1:26"> مَنْ يَصُونُ فَمَهُ وَلِسَانَهُ عَنِ اللَّغْوِ يَحْفَظُ نَفْسَهُ مِنَ الْمَتَاعِبِ.</span>
<span className="v" id="v24" data-v="24" data-xref="المزامير 1:1; الأمثال 18:12; الأمثال 1:22"> الْمُتَشَامِخُ الْمُنْتَفِخُ يُدْعَى الْمُسْتَهْزِئَ لأَنَّهُ يَتَصَرَّفُ بِغُرُورِ الْكِبْرِيَاءِ.</span>
<span className="v" id="v25" data-v="25" data-xref="الأمثال 13:4; الأمثال 12:24; الإنجيل كما دونه متى 25:26"> أَوْهَامُ الْكَسُولِ تَقْتُلُهُ لأَنَّ يَدَيْهِ تَأْبَيَانِ الْعَمَلَ.</span>
<span className="v" id="v26" data-v="26" data-xref="المزامير 37:26; أعمال الرسل 20:33–35; الرسالة إلى مؤمني أفسس 4:28"> يَظَلُّ طَوَالَ النَّهَارِ مُتَشَهِّياً مُتَمَنِّياً، أَمَّا الصِّدِّيقُ فَيَسْخُو وَلا يَضِنُّ</span>
<span className="v" id="v27" data-v="27" data-xref="إشعياء 66:3; الأمثال 15:8; إرميا 6:20"> ذَبِيحَةُ الشِّرِّيرِ رِجْسٌ عِنْدَ الرَّبِّ، فَكَمْ بِالْحَرِيِّ إِنْ قَرَّبَهَا بِنِيَّةٍ أَثِيمَةٍ؟</span>
<span className="v" id="v28" data-v="28" data-xref="الرسالة الثانية إلى مؤمني كورنثوس 4:13; الأمثال 19:5; الأمثال 19:9"> شَاهِدُ الزُّورِ يَهْلِكُ، أَمَّا أَقْوَالُ الرَّجُلِ الْحَرِيصِ عَلَى الاسْتِمَاعِ فَتَدُومُ.</span>
<span className="v" id="v29" data-v="29" data-xref="الرسالة الأولى إلى مؤمني تسالونيكي 3:11; الأمثال 11:5; حجي 2:18–19"> الرَّجُلُ الشِّرِّيرُ يُغْلِظُ وَجْهَهُ، أَمَّا الْمُسْتَقِيمُ فَيَعْمَلُ عَلَى تَقْوِيمِ طُرُقِهِ.</span>
<span className="v" id="v30" data-v="30" data-xref="الأمثال 19:21; أعمال الرسل 5:39; إرميا 9:23"> لَيْسَ مِنْ حِكْمَةٍ، وَلا مِنْ مَشُورَةٍ، وَلا مِنْ فِطْنَةٍ بِقَادِرَةٍ عَلَى مُقَاوَمَةِ اللهِ.</span>
<span className="v" id="v31" data-v="31" data-xref="المزامير 20:7; إشعياء 31:1; المزامير 33:17–18"> مَعَ أَنَّ الْفَرَسَ مُعَدٌّ لِيَوْمِ الْقِتَالِ، فَإِنَّ النَّصْرَ هُوَ مِنْ عِنْدِ اللهِ.</span>
