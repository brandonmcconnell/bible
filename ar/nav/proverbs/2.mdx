## ثواب الحكمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الأمثال 3:1; الأمثال 1:3; الأمثال 4:1"> يَا ابْنِي إِنْ قَبِلْتَ كَلامِي، وَادَّخَرْتَ وَصَايَايَ فِي قَلْبِكَ،</span>
<span className="v" id="v2" data-v="2" data-xref="الأمثال 23:12; الأمثال 18:1; المزامير 119:111–112"> وَأَرْهَفْتَ أُذُنَكَ إِلَى الْحِكْمَةِ، وَأَمَلْتَ قَلْبَكَ نَحْوَ الْفَهْمِ،</span>
<span className="v" id="v3" data-v="3" data-xref="رسالة يعقوب 1:5; الرسالة إلى مؤمني أفسس 1:17–18; الأمثال 3:6"> وَإِنْ نَشَدْتَ الْفِطْنَةَ، وَهَتَفْتَ دَاعِياً الْفَهْمَ.</span>
<span className="v" id="v4" data-v="4" data-xref="الإنجيل كما دونه متى 13:44; الأمثال 16:16; الإنجيل كما دونه متى 19:29"> إِنِ الْتَمَسْتَهُ كَمَا تُلْتَمَسُ الْفِضَّةُ، وَبَحَثْتَ عَنْهُ كَمَا يُبْحَثُ عَنِ الْكُنُوزِ الدَّفِينَةِ،</span>
<span className="v" id="v5" data-v="5" data-xref="الأمثال 1:7; إرميا 24:7; إرميا 32:40–41"> عِنْدَئِذٍ تُدْرِكُ مَخَافَةَ الرَّبِّ وَتَعْثُرُ عَلَى مَعْرِفَةِ اللهِ،</span>
<span className="v" id="v6" data-v="6" data-xref="الرسالة إلى مؤمني أفسس 1:17–18; رسالة يعقوب 1:5; أيوب 32:8"> لأَنَّ الرَّبَّ يَهَبُ الْحِكْمَةَ، وَمِنْ فَمِهِ تَتَدَفَّقُ الْمَعْرِفَةُ وَالْفَهْمُ.</span>
<span className="v" id="v7" data-v="7" data-xref="رسالة يعقوب 3:15–17; الأمثال 30:5; المزامير 84:11"> يَذْخَرُ لِلْمُسْتَقِيمِينَ فِطْنَةً، وَهُوَ تُرْسٌ لِلسَّالِكِينَ بِالْكَمَالِ.</span>
<span className="v" id="v8" data-v="8" data-xref="المزامير 23:3; صموئيل الأول 2:9; الإنجيل كما دونه يوحنا 10:28–29"> يَرْعَى سُبُلَ الْعَدْلِ، وَيُحَافِظُ عَلَى طَرِيقِ أَتْقِيَائِهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="إرميا 6:16; المزامير 143:8–10; الإنجيل كما دونه متى 7:13–14"> حِينَئِذٍ تُدْرِكُ الْعَدْلَ وَالْحَقَّ وَالاسْتِقَامَةَ، وَكُلَّ سَبِيلٍ صَالِحٍ.</span>
</div>
</div>

## فوائد الحكمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إرميا 15:16; الرسالة إلى مؤمني كولوسي 3:16; أيوب 23:12"> إِنِ اسْتَقَرَّتِ الْحِكْمَةُ فِي قَلْبِكَ وَاسْتَلَذَّتْ نَفْسُكَ الْمَعْرِفَةَ،</span>
<span className="v" id="v11" data-v="11" data-xref="الأمثال 4:6; الرسالة إلى مؤمني أفسس 5:15; المزامير 25:21"> يَرْعَاكَ التَّعَقُّلُ، وَيَحْرُسُكَ الْفَهْمُ.</span>
<span className="v" id="v12" data-v="12" data-xref="الرسالة الأولى إلى مؤمني كورنثوس 15:33; الأمثال 8:13; الأمثال 13:20"> إِنْقَاذاً لَكَ مِنْ طَرِيقِ الشَّرِّ وَمِنَ النَّاطِقِينَ بِالأَكَاذِيبِ.</span>
<span className="v" id="v13" data-v="13" data-xref="الإنجيل كما دونه يوحنا 3:19–20; الأمثال 4:19; الأمثال 21:16"> مِنَ الَّذِينَ يَبْتَعِدُونَ عَنْ سَوَاءِ السَّبِيلِ وَيَسْلُكُونَ فِي طُرُقِ الظُّلْمَةِ،</span>
<span className="v" id="v14" data-v="14" data-xref="الرسالة إلى مؤمني روما 1:32; الأمثال 10:23; إرميا 11:15"> الَّذِينَ يَفْرَحُونَ بِارْتِكَابِ الْمَسَاوِئِ، وَيَبْتَهِجُونَ بِنِفَاقِ الشَّرِّ،</span>
<span className="v" id="v15" data-v="15" data-xref="المزامير 125:5; الأمثال 21:8; الرسالة إلى مؤمني فيلبي 2:15"> مِنْ ذَوِي الْمَسَالِكِ الْمُلْتَوِيَةِ وَالسُّبُلِ الْمُعْوَجَّةِ.</span>
<span className="v" id="v16" data-v="16" data-xref="الجامعة 7:26; الأمثال 22:14; الأمثال 23:27"> وَإِنْقَاذاً لَكَ مِنَ الْمَرْأَةِ الْغَرِيبَةِ الْمُخَاتِلَةِ الَّتِي تَتَمَلَّقُكَ بِكَلامِهَا،</span>
<span className="v" id="v17" data-v="17" data-xref="حزقيال 16:59–60; حزقيال 16:8; إرميا 3:4"> الَّتِي نَبَذَتْ شَرِيكَ صِبَاهَا وَتَنَاسَتْ عَهْدَ إِلَهِهَا.</span>
<span className="v" id="v18" data-v="18" data-xref="الأمثال 9:18; الرسالة الأولى إلى مؤمني كورنثوس 6:9–11; الأمثال 5:4–14"> لأَنَّ بَيْتَهَا يَغُوصُ عَمِيقاً إِلَى الْمَوْتِ، وَسُبُلَهَا تُفْضِي إِلَى عَالَمِ الأَرْوَاحِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الجامعة 7:26; المزامير 81:12; هوشع 4:14"> كُلُّ مَنْ يَدْخُلُ إِلَيْهَا لَا يَرْجِعُ وَلا يَبْلُغُ سُبُلَ الْحَيَاةِ.</span>
<span className="v" id="v20" data-v="20" data-xref="الرسالة إلى العبرانيين 6:12; رسالة يوحنا الثالثة 1:11; المزامير 119:63"> لِهَذَا سِرْ فِي طَرِيقِ الأَخْيَارِ، وَاحْفَظْ سَبِيلَ الأَبْرَارِ،</span>
<span className="v" id="v21" data-v="21" data-xref="المزامير 84:11; المزامير 37:29; المزامير 37:9"> لأَنَّ الْمُسْتَقِيمِينَ يَسْكُنُونَ الأَرْضَ، وَالْكَامِلِينَ يَمْكُثُونَ دَائِماً فِيهَا.</span>
<span className="v" id="v22" data-v="22" data-xref="المزامير 52:5; التثنية 7:22; إشعياء 3:10–11"> أَمَّا الأَشْرَارُ فَيَنْقَرِضُونَ مِنَ الأَرْضِ، وَالْغَادِرُونَ يُسْتَأْصَلُونَ مِنْهَا.</span>
