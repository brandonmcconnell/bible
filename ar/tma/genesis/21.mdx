<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مقتطفات من كتاب الزبور 12:6; التَّكوين 18:10; التَّكوين 18:14"> وأنعم الله على سارة كما وعد،</span>
<span className="v" id="v2" data-v="2" data-xref="غلاطية 4:22; العبرانيين 11:11; التَّكوين 18:10"> فحملت وأنجبت ابنًا لإبراهيم (عليه السّلام) في شيخوخته، في الوقت الّذي حدّده وحي الله.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 17:19; روما 9:7; التَّكوين 21:6"> وسمّى النبي إبراهيم ابنه من سارة إسحق، ومعناه "ضحك".</span>
<span className="v" id="v4" data-v="4" data-xref="سيرة الحواريّين 7:8; Deuteronomy 12:32; يوحنّا 7:22–23"> وختن النبي إبراهيم ابنه إسحق في اليوم الثامن من عمره تطبيقًا لِما أمره الله به.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 17:17; روما 4:19; التَّكوين 17:1"> وحين رُزق إبراهيمُ بإسحق (عليهما السلام) كان عمره مئة سنة.</span>
<span className="v" id="v6" data-v="6" data-xref="مقتطفات من كتاب الزبور 126:2; إشَعْياء 54:1; التَّكوين 17:17"> وقالت سارة: "لقد أضحكني الله، وكلّ مَن يسمع بهذا الخبر يضحك معي".</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 18:11–12; إشَعْياء 66:8; تسالونكي الثانية 1:10"> وقالت: "مَن كان يتوقّع أني سأَرضع لإبراهيم بنين وهو شيخ؟ ها قد أنجبت له ابنًا!"</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="صَمُوئيلَ الأوَّلُ 1:22; Hosea 1:8; التَّكوين 40:20"> بمناسبة فطام إسحق، أقام النبي إبراهيم وليمة عظيمة إكرامًا له.</span>
<span className="v" id="v9" data-v="9" data-xref="غلاطية 4:29; التَّكوين 16:15; غلاطية 4:22"> ورأت سارة أن إسماعيل ابن هاجر المصرية يسخر من ابنها إسحق،</span>
<span className="v" id="v10" data-v="10" data-xref="يوحنّا 8:35; يوحنّا الأولى 2:19; غلاطية 4:22–31"> فقالت للنبي إبراهيم: "أَطْرِدْ هذه الجارية وابنها! لا يرث ابن هذه الجارية مع ابني إسحَق".</span>
<span className="v" id="v11" data-v="11" data-xref="العبرانيين 12:11; التَّكوين 17:18; متّى 10:37"> وانزعج النبي إبراهيم من هذا الكلام كثيرًا، لأنّ إسماعيل (عليه السّلام) ابنه أيضًا.</span>
<span className="v" id="v12" data-v="12" data-xref="العبرانيين 11:18; روما 9:7–8; التَّكوين 17:19"> فأوحى الله إليه: "لا تنزعج بشأن ابنك وجاريتك. واعمل بما تقوله لك سارة، فمِن إسحق ينحدر الورثة وبهم أحقق وعدي لك.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 16:10; التَّكوين 17:20; التَّكوين 21:18"> ومن إسماعيل أجعل أمّة عظيمة لأنّه ابنك أيضًا".</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الجامِعَة 9:10; مقتطفات من كتاب الزبور 119:60; التَّكوين 16:7"> وفي الصباح الباكر أخذ إبراهيم (عليه السّلام) خبزًا وقربة ماء وأعطاهما لهاجر ووضعهما على كتفها وأطلقها هي وابنها، فمضت تهيم على وجهِها في صحراء بئر السبع.</span>
<span className="v" id="v15" data-v="15" data-xref="2 Kings 3:9; مقتطفات من كتاب الزبور 63:1; إرْمِيا 14:3"> وعندما فرغ الماء من القربة، تركت ابنها (عليه السّلام) تحت شجرة صغيرة</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 49:15; التَّكوين 29:11; صَمُوئيلَ الأوَّلُ 24:16"> ومضت تراقبه وهي جالسة على بعد رميتي قوس (حوالي مئة متر) لأنّها قالت في نفسها: "لا أستطيع أن أتحمّل رؤية ابني وهو يموت أمامي". وأجهشت بالبكاء.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 16:11; مقتطفات من كتاب الزبور 50:15; الخُرُوج 3:7"> وسمع الله إسماعيل (عليه السّلام) يستغيث، فهتف ملاكٌ من السماء لهاجر وأوحى إليها: "لا عليكِ يا هاجر! لا تخافي، لأنّ الله سمع صوت الولد وهو ينادي من حيث ألقي!</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 21:13; التَّكوين 16:10; التَّكوين 17:20"> والآن انهضي وخذي ابنك وشدّي بيده، فالله سيجعل من نسله أمّة عظيمة".</span>
<span className="v" id="v19" data-v="19" data-xref="Numbers 22:31; 2 Kings 6:17–20; إشَعْياء 35:5–6"> وفتح اللهُ بصيرة هاجر فرأت بئر ماء، فهرولت إليها وملأتِ القربةَ ماءً وسقتِ ابنها (عليه السّلام).</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 28:15; التَّكوين 39:21; التَّكوين 16:12"> وكان الله مع إسماعيل (عليه السّلام) خير معين. ومع مرور الزمن كبُر الصّبيُّ وأقام في صحراء فاران، وأصبح صيادًا ماهرًا في الرماية بالقوس والسهام. وزوَّجته أُمُّهُ بنتا من أرض مصر.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="Joshua 3:7; التَّكوين 26:26; التَّكوين 26:28"> وفي تلك الأيام أقبل أبيمالكُ وفيكول قائد جيشه على النبي إبراهيم. فطلب أبيمالك من إبراهيم (عليه السّلام): "أرى أنّ الله يُعاونك في كلّ ما تفعله،</span>
<span className="v" id="v23" data-v="23" data-xref="Joshua 2:12; التَّكوين 20:14; العبرانيين 6:16"> فأقسمْ بالله هنا أنّك لا تغدر بي ولا بأولادي ولا بنسلي، بل تبادلني معروفًا بمعروف، أنا وسكّان الأرض الّتي تغرَّبتَ فيها".</span>
<span className="v" id="v24" data-v="24" data-xref="العبرانيين 6:16; التَّكوين 14:13; روما 12:18"> فأجابه النبي إبراهيم: "أقسم لك".</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 26:15–22; التَّكوين 29:8; الأمثال 25:9"> واشتكى إبراهيم (عليه السّلام) لأبيمالك من خدمه الّذين سلبوا منه بئرًا.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 13:7; 2 Kings 5:20–24"> فأجابه أبيمالك: "لا أعرف تحديدًا من الّذي ارتكب هذا، أنت لم تخبرني ولم أسمع بهذه الحادثة إلاّ الآن".</span>
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 14:22–23; إشَعْياء 32:8; روما 1:31"> وأخذ النبي إبراهيم أغنامًا وأبقارًا ليبرم المعاهدة مع أبيمالك.</span>
<span className="v" id="v28" data-v="28"> زيادة على ذلك، أفرد (عليه السّلام) سبع نعاج،</span>
<span className="v" id="v29" data-v="29" data-xref="صَمُوئيلَ الأوَّلُ 15:14; التَّكوين 33:8; الخُرُوج 12:26"> فقال له أبيمالكُ: "ما هذه النّعاج السبع الّتي أفردتَها جانبًا؟"</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 31:52; Joshua 24:27; Joshua 22:27–28"> فأجاب النبي إبراهيم: "أرجوك أن تأخذ مني هذه النعاج السبع اعترافًا منك أنّي أنا من حفر هذه البئر".</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 26:33; التَّكوين 21:14; 1 Kings 4:25"> وسُمي ذلك الموضع بئرَ السبع (ومعناه بئر القسم)، لأنّ المعاهدة تمّت فيه.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 13:17; التَّكوين 21:27; Judges 13:1"> وبعد أن تعاهدا في بئر السبع عاد أبيمالكُ وفيكول قائد جيشه إلى بلاد فِليسطيا في ساحل فلسطين</span>
<span className="v" id="v33" data-v="33" data-xref="تيموتاوي الأولى 1:17; إشَعْياء 40:28; مقتطفات من كتاب الزبور 90:2"> وغرس إبراهيم (عليه السّلام) شجرة طرفاء في بئر السبع وتعبّد الله الصمد.</span>
<span className="v" id="v34" data-v="34" data-xref="1 Chronicles 29:15; صخر الأولى 2:11; العبرانيين 11:9"> وتغرَّب (عليه السّلام) في بلاد فِليسطيا فترة طويلة.</span>
</div>
