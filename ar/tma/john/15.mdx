<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 2:21; إشَعْياء 27:2–3; كورنتوس الأولى 3:9"> وأثناءَ الطَّريقِ استَكمَلَ سَيِّدُنا عيسى (سلامُهُ علينا) حَديثَهُ قائلاً: "أنا الكَرمُ الحَقُّ، واللهُ أبي الصَّمَدُ هو صاحِبُهُ الّذي يَقطَعُ عَقيمَ الأغصانِ، ويُبقي المُثمِرَ مِنها فيُشَذِّبُهُ ويُنَقِّيهِ حتّى يُثمِرَ أكثرَ وأفضَلَ.</span>
<span className="v" id="v3" data-v="3" data-xref="يوحنّا 17:17; أفاسوس 5:26; صخر الأولى 1:22"> أنتُم أنقياءُ ومُشَذَّبونَ بسبَبِ تَعاليمي الّتي أُخاطِبُكم بها.</span>
<span className="v" id="v4" data-v="4" data-xref="يوحنّا الأولى 2:6; غلاطية 2:20; يوحنّا 8:31"> فاُثبُتوا فيَّ كَما أثبُتُ أنا فيكُم. ومِثلَما لا يُثمِرُ الغُصنُ إذا كانَ مُنفَصِلاً عن جِذعِ الشَّجَرةِ، فكذلِكَ أنتُم، لا تُثمِرونَ إن لم تَثبُتوا فيَّ.</span>
<span className="v" id="v5" data-v="5" data-xref="فيليبّي 4:13; كولوسي 1:10; يوحنّا 15:16"> فأنا الكَرمةُ وأنتُم أغصاني، ومَن ثَبَتَ فيَّ وأنا فيهِ أَثمَرَ خَيرًا كثيرًا. أمّا بمَعزلٍ عنّي فأنتُم عَقيمونَ.</span>
<span className="v" id="v6" data-v="6" data-xref="متّى 7:19; العبرانيين 6:7–8; متّى 13:41"> وكُلُّ مَن يَنفَصِلُ عنِّي يُرمى بِهِ بَعيدًا كالعُودِ الجافِّ الّذي يَنتَظِرُ حَرقَهُ.</span>
<span className="v" id="v7" data-v="7" data-xref="مقتطفات من كتاب الزبور 37:4; يوحنّا الأولى 5:14; يوحنّا 14:13"> وإن ثَبَتُّم مُخلِصينَ لي وَثَبَتَ كَلامي فيكُم، فإنّ لكُم ما تَطلُبونَ مِنَ الله.</span>
<span className="v" id="v8" data-v="8" data-xref="يوحنّا 8:31; متّى 5:16; فيليبّي 1:11"> وإن داوَمتُم على الصّالحِ مِنَ الأعمالِ، أبَنتُم بذلِكَ أنّكُم أتباعي حَقًّا، وعَظَّمتُم شأنَ اللهِ أبي الصَّمَدِ".</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="يوحنّا 17:23–24; يوحنّا 17:26; يوحنّا الأولى 2:28"> ثُمَّ أضافَ (سلامُهُ علينا) قائلاً: "أحبَبتُكُم كَما أحبَّني اللهُ الأبُ الرَّحمنُ، فاثبُتوا على مَحَبّتي ولا تَدَعوا أيّ حاجِزٍ يَفصِلُكُم عَنها.</span>
<span className="v" id="v10" data-v="10" data-xref="يوحنّا 14:15; يوحنّا الأولى 5:3; يوحنّا الأولى 3:21–24"> أنا أُفيضُ عَليكُم مِن مَحَبَّتي، كَما يُفيضُ اللهُ أبي الصَّمَدُ عليَّ مِن مَحَبّتِهِ، وأنا أعمَلُ بوصاياهُ وأَستَمِرُّ في مَحَبَّتِهِ، فاعمَلوا أنتُم بوَصاياي لكي تَستَمِرُّوا في مَحَبّتي.</span>
<span className="v" id="v11" data-v="11" data-xref="يوحنّا 17:13; روما 15:13; يوحنّا 16:24"> وما قَولي هذا إلاّ لتَفرَحوا مِثلي ويَفيضَ فَرَحُكُم.</span>
<span className="v" id="v12" data-v="12" data-xref="يوحنّا الأولى 4:21; صخر الأولى 4:8; يوحنّا 13:34"> هذِهِ هي وَصيَّتي لكُم، فتَمَسَّكوا بها واحفَظوها: أحِبُّوا بَعضُكُم بَعضًا كَما أحبَبتُكُم أنا،</span>
<span className="v" id="v13" data-v="13" data-xref="يوحنّا 10:11; أفاسوس 5:2; روما 5:6–8"> ولَيسَ مِن مَحَبّةٍ يَهَبُها المُحِبُّ أعظَمَ مِن تَضحِيتِهِ بنَفسِهِ في سَبيلِ مَن يُحِبُّ.</span>
<span className="v" id="v14" data-v="14" data-xref="متّى 12:50; يوحنّا الأولى 5:3; يعقوب 2:23"> إن كُنتُم عَمِلتُم بتَعاليمي فأنتُم أحبابي،</span>
<span className="v" id="v15" data-v="15" data-xref="مقتطفات من كتاب الزبور 25:14; كورنتوس الأولى 2:9–12; متّى 13:11"> ولن أدعُوكم بَعدَ الآنَ عَبيدًا، لأنّ العَبدَ مُنفصِلٌ عن سَيِّدِهِ جاهلٌ بأعمالِهِ، ولكنّني أدعُوكُم أحبابي وقد كَشَفتُ لكُم كُلَّ ما أبانَهُ لي اللهُ أبي الصَّمَدُ مِن حَقائقَ.</span>
<span className="v" id="v16" data-v="16" data-xref="يوحنّا 15:19; إرْمِيا 1:5–7; أفاسوس 2:10"> ولا يَذهبَنَّ بكُم الظَّنُّ إلى أنَّكُم أنتُم اختَرتُموني بل أنا مَنِ اختارَكُم لتَكونوا مِمَّن يُثمِرُ أعمالاً يَدومُ تأثيرُها إلى الأبَدِ، فيَمنَحُكُم اللهُ الأبُ الرَّحمنُ بذلِكَ كُلَّ ما تسألُونَهُ باسمي.</span>
<span className="v" id="v17" data-v="17" data-xref="يوحنّا 15:12; يوحنّا الأولى 3:14–17; صخر الأولى 2:17"> تِلكَ هي وَصيَّتي أُكَرِّرُها عليكُم: أحِبُّوا بَعضُكم بَعضًا.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="يوحنّا الأولى 3:13; يوحنّا 7:7; مرقس 13:13"> وتابَعَ (سلامُهُ علينا) حَديثَهُ مَعَ أتباعِهِ قائلاً: "لئن كَرِهَكُم أهلُ الدُّنيا، فاعلَمُوا أنّهُم قد كَرِهوني قَبلَكُم!</span>
<span className="v" id="v19" data-v="19" data-xref="يوحنّا الأولى 4:4–5; يوحنّا 17:14–16; أفاسوس 1:4–11"> ولو كُنتُم مِن أهلِ الدُّنيا، لأحَبُّوكُم كَما أحبُّوا عَبيدَها، ولقد انتقَيتُكُم وفَصَلتُكُم عَنهُم فأنتُم لستُم مِثلَهم، لذلِكَ يُبغضونَكُم.</span>
<span className="v" id="v20" data-v="20" data-xref="تيموتاوي الثانية 3:12; يوحنّا 13:16; متّى 10:24"> واذكُروا ما كُنتُ قد حَدَّثتُكُم بِهِ: "لَيسَ العَبدُ أعظَمَ مِن سَيِّدِهِ". فإنِ اضطَهَدَني أهلُ الدُّنيا، فسيَضطَهِدُونَكُم، وإن عَمِلوا برِسالتي، فسيَعمَلونَ بتَعاليمِكُم.</span>
<span className="v" id="v21" data-v="21" data-xref="متّى 10:22; يوحنّا 16:3; متّى 10:18"> وما تَشريدُكُم وتَعذيبُكُم إلاّ لأنّكُم أتباعي، لأنّهُم يَجهَلونَ اللهَ الّذي أرسَلَني.</span>
<span className="v" id="v22" data-v="22" data-xref="يوحنّا 9:41; يعقوب 4:17; سيرة الحواريّين 17:30"> وما كانَ عليهِم مِن وِزرٍ لو لم أُبعثْ فيهِم، ولكن لا عُذرَ لهُم بَعدَ الآنَ وقد بُعِثتُ فيهِم وكَلّمتُهُم.</span>
<span className="v" id="v23" data-v="23" data-xref="يوحنّا 8:40–42; يوحنّا الثانية 1:9; يوحنّا الأولى 2:23"> مَن يَكرَهُني يَكرَهِ اللهَ أبي الصَّمَدَ،</span>
<span className="v" id="v24" data-v="24" data-xref="يوحنّا 5:36; يوحنّا 14:9; يوحنّا 7:31"> ولو لم أَعمَلْ بَينَهُم مُعجِزاتٍ لم يَعمَلهَا أَحَدٌ غَيري، لمَا كانَ عليهِم ذَنبٌ، ولكنّهُم أنكَروا المُعجِزاتِ الّتي رأوها ورَفَضوا أن يُؤمِنوا بي جُحُودًا واستِكبارًا، فكَرِهُوني وكَرِهُوا اللهَ أبي الصَّمَدَ.</span>
<span className="v" id="v25" data-v="25" data-xref="مقتطفات من كتاب الزبور 109:3; مقتطفات من كتاب الزبور 69:4; مقتطفات من كتاب الزبور 35:19"> وكانَ ذلِكَ كُلُّهُ تَصديقًا لِما جاءَ في الزَّبُورِ: "كَرِهُوني بلا سَبَبٍ".</span>
<span className="v" id="v26" data-v="26" data-xref="يوحنّا 14:26; يوحنّا 14:16–17; يوحنّا 16:13–15"> وعِندَما أُرسِلُ إليكُم المُعينَ، رُوحَ اللهِ الّتي تَخرُجُ مِن عِندِ اللهِ الأبِ الرَّحمنِ، فتُذَكِّرُكم بي.</span>
<span className="v" id="v27" data-v="27" data-xref="يوحنّا الأولى 4:14; يوحنّا 21:24; سيرة الحواريّين 10:39–42"> فاشهَدوا لي أمامَ النّاسِ وادعُوهُم إليّ، لأنّكُم كُنتُم رُفقائي مُنذُ البدايةِ".</span>
</div>
