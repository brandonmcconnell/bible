<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="متّى 23:5; يوحنّا 12:43; العبرانيين 6:10"> ثُمّ حَدَّثَ (سلامُهُ علينا) أتباعَهُ قائلاً: "إيّاكُم والنِّفاقَ في عِباداتِكُم واحذَروا أن تَعبَدوا اللهَ مُراءَاةً للنّاسِ ليَمتَدحوكُم، إذ بهَذا لن يَكونَ لكُم ثَوابٌ عِندَ اللهِ أبيكُم الرَّحمنِ في عُلاه.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="متّى 6:5; متّى 6:16; كورنتوس الثانية 9:6–15"> فاجتَنِبوا أن تَمنَحوا صَدقاتِكُم جَهرًا، بل امنَحوها المُحتاجينَ سِرًّا ولا تَكونوا كالمُنافقينَ الّذينَ يُشهِرونَ صَدَقاتِهِم في بُيوتِ العِبادةِ وفي الشّارعِ رياءَ النّاسِ فيَكيلونَ لهُم المَديحَ. الحَقَّ أقولُ لكُم، ما لهؤلاء مِن جَزاءٍ إلاّ ما قد وَصَلَهُم مِن النّاسِ.</span>
<span className="v" id="v3" data-v="3" data-xref="يوحنّا 7:4; متّى 9:30; مرقس 1:44"> امنَحوا صَدَقاتِكُم سِرًّا فلا تَعرِفُ شِمالُكُم ما قَدَّمَت يَمينُكُم،</span>
<span className="v" id="v4" data-v="4" data-xref="متّى 6:18; متّى 6:6; إرْمِيا 17:10"> واللهُ أبوكُم الرَّحمنُ يَعلَمُ ما أسرَرتُم ويَمنَحُكُم الجَزاءَ الأوفى".</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="لوقا 18:10–11; متّى 6:2; لوقا 18:13"> وأخبَرَهُم (سلامُهُ علينا) عن الصَّلاةِ فقالَ: "ولا تُصَلّوا صَلاةَ المُنافِقينَ الّذينَ يُقيمونَ صَلاتَهُم في دُورِ العِبادةِ وبينَ العامّةِ في الطُّرُقاتِ حتّى يَراهُم الجَميعُ. والحَقَّ أقولُ لكُم: لقد نالَ هَؤلاءِ أجرَهُم مِن النّاسِ ولَيسَ لهُم عِندَ اللهِ ذَرّةُ جَزاءٍ أو إحسان.</span>
<span className="v" id="v6" data-v="6" data-xref="2 Kings 4:33; متّى 14:23; إشَعْياء 65:24"> أمّا أنتُم يا أحبابي فأقيموا صَلاتَكُم في خُشوعٍ للهِ أبيكُم الرَّحمنِ في غُرَفِكُم مُغلِقينَ الأبوابَ لتَكونوا في مَنأى عن أَعيُنِ النّاسِ وعن ألسِنَتِهِم، فتَنالونَ ثَوابَ أبيكُم الرَّحمنِ عالمِ السِّرِّ والنَّجوى.</span>
<span className="v" id="v7" data-v="7" data-xref="الجامِعَة 5:2–3; الجامِعَة 5:7; 1 Kings 18:26–29"> وعِندَ صَلاتِكُم، لا تَتَوجَّهوا إليهِ بعَقيمِ الكَلامِ المُكَرَّرِ كَما يَفعَلُ الوَثَنيّونَ الّذينَ يَحسَبونَ أنّهُ بكَثرةِ كَلامِهِم تُستَجابُ دَعواتُهُم.</span>
<span className="v" id="v8" data-v="8" data-xref="متّى 6:32; لوقا 12:30; فيليبّي 4:6"> نعم، لا تَكونُنَّ مِثلَهُم، واعلموا أنّ اللهَ أباكُم الرَّحمنَ مُطلِّعٌ على حالِكُم قَبلَ أن تَسألوا.</span>
</div>
<div className="bp poetry">
<span className="v" id="v9" data-v="9" data-xref="لوقا 11:1–4; الرؤيا 4:11; روما 8:15"> وإذا صَلّيتُم فصَلّوا على الوَجهِ التّالي قائلينَ:<br /><span className="pl pl-1">"اللهُمَّ يا أبانا الرَّحمن في عُلاك،</span><br /><span className="pl pl-2">تَباركَ اسمُكَ وتَقَدَّسَ،</span></span>
<span className="v" id="v10" data-v="10" data-xref="العبرانيين 10:36; متّى 26:42; متّى 12:50"> <span className="pl pl-1">لتَقُم مَملكتُكَ المَوعودةُ في الأرضِ،</span><br /><span className="pl pl-2">ليَسعَ مَن في الأرضِ لمَرضاتِكَ كسَعْي مَن في السَّماءِ.</span></span>
<span className="v" id="v11" data-v="11" data-xref="الأمثال 30:8; لوقا 11:3; متّى 4:4"> <span className="pl pl-1">وارزُقنا قُوتَ يومِنا،</span></span>
<span className="v" id="v12" data-v="12" data-xref="أفاسوس 4:32; مرقس 11:25; أفاسوس 1:7"> <span className="pl pl-1">واغفِر لنا سَيئاتِنا</span><br /><span className="pl pl-2">كَما نُسامحُ مَن أخطأ في حَقِّنا،</span></span>
<span className="v" id="v13" data-v="13" data-xref="كورنتوس الأولى 10:13; الرؤيا 7:14–17; متّى 26:41"> <span className="pl pl-1">وجَنّبنا المِحنَ والبَلاء، ونَجِّنا مِن كَيدِ الشَّيطانِ الأثيمِ،</span><br /><span className="pl pl-2">فَلَكَ وَحدَكَ المُلكُ والقُوّةُ والجَلالُ إلى الأبَدِ آمين".</span></span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="أفاسوس 4:32; مرقس 11:25; كولوسي 3:13"> وتابَعَ (سلامُهُ علينا) قائلاً: "إن أنتُم صَفَحتُم عَمّا ارتَكَبَهُ النّاسُ في حَقِّكُم، فإنّ اللهَ أباكُم الرَّحمنَ يَغفِرُ لكُم آثامَكُم أيضًا،</span>
<span className="v" id="v15" data-v="15" data-xref="متّى 18:35"> وإلاّ فلن يَغفِرَ أبوكُم الرَّحمنُ خَطاياكُم".</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 58:3–5; متّى 9:14–15; دانيال 9:3"> ثُمّ أضافَ (سلامُهُ علينا) يُرشِدُهُم عن الصّومِ قائلا: "واحذَروا أحبابي عِندَ صَومِكُم أن تَكونوا عابِسي الوُجوهِ كَما يَفعَلُ المُنافِقونَ الّذينَ يَصطَنِعونَ التَّعبَ ليَمتَدِحَهُم النّاسُ. والحَقَّ أقولُ لكُم، لقد نالَ هؤلاءِ النّاسُ أجرَهُم.<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="6‏.16 يوم الكفّارة هو اليوم المعيّن في التوراة للصيام من كلّ سنة، ولكن رغم ذلك فإنّ العديد من اليهود المتديّنين كانوا يصومون يومين في الأسبوع، كما يصومون في المناسبات الخاصّة للتعبير عن ندمهم وتوبتهم وخشوعهم وطاعتهم لله. وفي صيامهم يمتنعون عن الطعام والمتع الأخرى مثل دهن رؤوسهم وأجسامهم بالزيت لإبراز جفاف بشرتهم حتّى يكون صيامهم باديًا للعيان.">a</a></sup></span>
<span className="v" id="v17" data-v="17" data-xref="راعُوث 3:3; الجامِعَة 9:8; 2 Samuel 12:20"> أمّا أنتُم فعِندَ صَومِكُم اغسِلوا وجوهَكُم وتَعَطَّروا،</span>
<span className="v" id="v18" data-v="18" data-xref="متّى 6:6; متّى 6:4; كورنتوس الثانية 10:18"> كي لا يَبدوَ أثرُ الصّيامِ على وُجوهِكُم، فلا يَعلَمَ بذلكَ غَيرُ اللهِ أبيكُم الرَّحمنِ الّذي يَعلَمُ ما تُخفي الصُّدورُ فيَمنَحُكُم على إخلاصِكُم الأجرَ والثَّوابَ".</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="العبرانيين 13:5; يوحنّا الأولى 2:15–16; لوقا 12:21"> وتابَعَ (سلامُهُ علينا) يُذَكِّرُهُم قائلا: "احذَروا كُنوزَ الدُّنيا فهي آيِلةٌ للتَّلَفِ، مَوعودةٌ بالفَسادِ، مُعَرَّضةٌ لسَرِقةِ اللُّصوص،</span>
<span className="v" id="v20" data-v="20" data-xref="لوقا 12:33; متّى 19:21; يعقوب 2:5"> ولكن ادَّخِروا لأنفُسِكُم كَنزًا عِندَ رَبِّكُم لا يَفنى، بَعيدًا عن التَّلَفِ، آمنًا مِن الفَسادِ، لا تَطالُهُ يَدُ اللُّصوصِ!</span>
<span className="v" id="v21" data-v="21" data-xref="كولوسي 3:1–3; لوقا 12:34; كورنتوس الثانية 4:18"> لأنّ رَغبةَ قُلوبِكُم مُوَجّهةٌ حَيثُما تَدَّخِرونَ كُنوزَكُم".</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="لوقا 11:34–36; كورنتوس الثانية 11:3; كولوسي 3:22"> "وانتَبِهوا لعُيونِكُم فمِن خِلالِها يَنفُذُ شُعاعُ النُّورِ إلى باطِنِكُم، فإذا نَظَرتُم إلى الآخَرينَ وقد قنَعتْ نُفوسُكم بما لَدَيكُم، أضاءَ حياتَكُم نورُ الرِّضا.</span>
<span className="v" id="v23" data-v="23" data-xref="يوحنّا الأولى 2:11; متّى 6:22; إشَعْياء 5:20–21"> وإذا نَظَرتُم إليهِم نِظرةَ حَسَدٍ، سَكَنَت الظُّلمةُ باطِنَكُم! وعِندئذٍ فأيُّ ظَلامٍ شَديدٍ على قُلوبِكُم تَتَخَبّطونَ فيهِ!</span>
<span className="v" id="v24" data-v="24" data-xref="يوحنّا الأولى 2:15–16; لوقا 16:13; يعقوب 4:4"> واعلَموا ألاّ أحَدَ يَقدِرُ أن يَخدُمَ سَيّدَين، وإلاّ كانَ في قَلبِهِ لأحَدِهِما حُبٌّ وللآخَرِ كُرهٌ، وسيَكونُ مُخلِصًا لهذا، مُهمِلاً لذاك. واعلَموا أن حُبَّ اللهِ لا يَكونُ في قَلبٍ يَسكُنُهُ حُبُّ المالِ".</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="فيليبّي 4:6; صخر الأولى 5:7; روما 8:32"> "فلا تَشغَلوا أنفُسَكُم بالخَوفِ على قُوتِ يَومِكُم، ولا على ما يَستُرُ أجسادَكُم، إنّ الحَياةَ أعظَمُ شأنًا مِن طَعامٍ تأكلونَهُ، والجَسَدَ أهمُّ مِن مُجَرَّدِ كِساءٍ تَلبِسونَهُ.</span>
<span className="v" id="v26" data-v="26" data-xref="متّى 10:29–31; مقتطفات من كتاب الزبور 147:9; لوقا 12:24–32"> أفلا تَرونَ الطُّيورَ في السَّماءِ تَغدو جياعًا وتَعودُ شِباعًا وما زَرَعَت ولا حَصَدَت ولا خَزَنَت، وما ذلكَ إلاّ لأنّ اللهَ أباكُم الرَّحمنَ رازِقُها! وأنتُم عِندَ اللهِ أهمُّ مِنها وأكرمُ.</span>
<span className="v" id="v27" data-v="27" data-xref="لوقا 12:25–26; الجامِعَة 3:14; مقتطفات من كتاب الزبور 39:5–6"> وأقولُ لكُم هل لأحَدِكُم، مَهما استَبَدَّ بِهِ القَلَقُ، قُدرةٌ على أن يَمُدَّ في عُمرِهِ ساعةً؟</span>
<span className="v" id="v28" data-v="28" data-xref="متّى 6:25; فيليبّي 4:6; لوقا 12:27"> فكَيفَ تَقلَقونَ على لِباسِكُم؟ ألا فانظُروا إلى الزَّنابقِ بكِسائِها الّذي لم تَتعَب في حِياكتِهِ،</span>
<span className="v" id="v29" data-v="29" data-xref="2 Chronicles 9:20–22; 2 Chronicles 9:4–6; صخر الأولى 3:2–5"> وتأمّلوا هذا الكِساءَ الّذي لم يَلبَسهُ أحَدٌ ولا حتّى النَّبيُّ سُليمانُ رَغمَ مَقامِهِ الجَليلِ.</span>
<span className="v" id="v30" data-v="30" data-xref="متّى 14:31; لوقا 12:28; صخر الأولى 1:24"> فإن كانَ اللهُ يَكسو العُشبَ هذا الكِساءَ، ولَيسَ مَصيرُ العُشبِ الّذي يَنمو في الحَقلِ اليَومَ ويَنتَشِرُ بَهيًّا إلاّ أن يَكونَ غدًا وَقودًا، فكَيفَ بكُم؟ ألَيسَ اللهُ بقادرٍ على أن يَكسوَكُم يا ضُعفاءَ الإيمان؟!</span>
<span className="v" id="v31" data-v="31" data-xref="صخر الأولى 5:7; لوقا 12:29; متّى 4:4"> فلا يَنشَغِلَنّ بالُ أحَدِكُم بأمرِ طَعامِهِ أو شَرابِه أو كِسائِهِ،</span>
<span className="v" id="v32" data-v="32" data-xref="متّى 6:8; لوقا 12:30; أفاسوس 4:17"> إنّ جَميعَ هذِهِ الأُمورِ مِن مَشاغِلِ الوَثَنيّينَ. إنّ اللهَ أباكُم الرَّحمنُ يَرعاكُم كَما يَرعَى الأبُ الحَنونُ أبناءَهُ، فهو أدرى بحاجاتِكُم.</span>
<span className="v" id="v33" data-v="33" data-xref="لوقا 12:31; متّى 5:6; يوحنّا 6:27"> فلْيَكُنْ سَعيُكُم في سَبيلِ مَملكتِهِ الّتي وَعَدَ بها ورِضوانِهِ، وسيَعطيكُم اللهُ ما طَلَبتُم ويَجودُ عليكُم أيضًا برِزقٍ.</span>
<span className="v" id="v34" data-v="34" data-xref="متّى 6:25; يوحنّا 14:27; العبرانيين 13:5–6"> فانظُروا في أُمورِ يَومِكُم واترُكوا الغَدَ إلى مَن لهُ الأمرُ، فلِكُلِّ يومٍ ما يَكفيهِ مِن المَتاعبِ".</span>
</div>
