<span className="v" id="v1" data-v="1" data-xref="متّى 20:1–16; متّى 12:43–45; لوقا 14:16"> ثُمّ عادَ سَيِّدُنا عيسى (سلامُهُ علينا) يَضرِبُ لهُم الأمثالَ ثانيةً</span>
<span className="v" id="v2" data-v="2" data-xref="الرؤيا 19:7–9; متّى 13:24; أفاسوس 5:24–32"> فقالَ: "إنّ مَثَلَ الدَّعوةِ إلى المَملكةِ الرَّبّانيّةِ، كمَثَلِ مَلِكٍ أقامَ وَليمةً في عُرسِ ابنِهِ ودَعا إليها جَمعًا مِن النّاسِ، فاستَجابوا لهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="الرؤيا 22:17; يوحنّا 5:40; لوقا 14:15–17"> وعِندَما أصبَحَت الوليمةُ جاهِزةً أرسَلَ عَبيدَهُ لاستِدعاءِ المَدعوّينَ ولكنّهُم رَفَضوا القُدومَ،</span>
<span className="v" id="v4" data-v="4" data-xref="سيرة الحواريّين 13:46; متّى 21:36; يوحنّا 6:50–57"> فأرسَلَ إليهِم عَبيدًا آخَرينَ قائلاً: "أخبِروا المَدعوّينَ بأنّني قد جَهَّزتُ كُلَّ شَيء، فقد ذَبَحتُ ثيراني وعُجولي السَّمينةَ، وأصبَحَ كُلُّ شَيءٍ في انتِظارِكُم فأقبِلوا!"</span>
<span className="v" id="v5" data-v="5" data-xref="العبرانيين 2:3; لوقا 14:18–20; يوحنّا الأولى 2:15–16"> ولكنّهُم لم يَكتَرِثوا إذ ذَهَبَ أحَدُهُم إلى حَقلِهِ، وانشَغِلَ آخرُ بتِجارتِهِ،</span>
<span className="v" id="v6" data-v="6" data-xref="يوحنّا 15:19–20; يوحنّا 16:2–3; سيرة الحواريّين 7:51–57"> وقَبَضَ بَعضُهُم على عَبيدِهِ وأساؤوا مُعامَلتَهُم وقَتَلوهُم.</span>
<span className="v" id="v7" data-v="7" data-xref="لوقا 19:27; صخر الأولى 4:17–18; لوقا 21:24"> فاستَبَدَّ الغَضَبُ بالمَلِكِ، وأرسَلَ جُندَهُ فأهلَكوهُم وأحرَقوا مَدينَتَهُم.</span>
<span className="v" id="v8" data-v="8" data-xref="تسالونكي الثانية 1:5; سيرة الحواريّين 13:46; الرؤيا 3:4"> ثُمّ قالَ لمَواليهِ: "الوَليمةُ جاهِزةٌ، إلاّ أنّ مَن دَعوتُهُم لا يَستَحِقّونَها،</span>
<span className="v" id="v9" data-v="9" data-xref="لوقا 14:21–24; الأمثال 1:20–23; مرقس 16:15–16"> فاذهَبوا وطُوفوا في الطُّرُقات داعينَ كُلَّ مَن تُصادِفونَهُ إليها".</span>
<span className="v" id="v10" data-v="10" data-xref="كورنتوس الأولى 6:9–11; الرؤيا 19:6–9; كورنتوس الثانية 12:21"> وهكذا مَضى العَبيدُ يَبحَثونَ في الشَّوارعِ والطُّرُقات لاِحضارِ كُلِّ مارٍّ وعابِرٍ، صالِحًا كانَ أم طالِحًا، فامتَلأت قاعةُ العُرسِ بالضُّيوفِ.</span>
<span className="v" id="v11" data-v="11" data-xref="الرؤيا 3:4–5; الرؤيا 19:8; العبرانيين 4:12–13"> ثُمّ دَخَلَ المَلِكُ وطافَ بَينَ المَدعوّينَ فرأى رَجُلاً لا يَرتَدي ثيابًا تَليقُ بهذِهِ المُناسَبةِ،</span>
<span className="v" id="v12" data-v="12" data-xref="صَمُوئيلَ الأوَّلُ 2:9; أيُّوب 5:16; متّى 20:13"> فقالَ لهُ: "أيُّها الضَّيفُ، ما غَرَّكَ بي فلم تُكَرِّمني بارتِداءِ ثيابٍ تَليقُ بالعُرسِ؟"</span>
<span className="v" id="v13" data-v="13" data-xref="متّى 8:12; متّى 25:30; متّى 24:51"> لكنّ لِسانَ الرَّجُلِ انعَقَدَ ولم يَتَفَوَّه بكَلِمةٍ، فأمَرَ المَلِكُ عِندئذٍ عَبيدَهُ قائلاً: "قَيِّدوهُ وارمُوا بِهِ خارجًا في الظُّلُمات حَيثُ يَبكي ويَصِرُّ أسنانَهُ نَدَمًا وحَسرةً!"</span>
<span className="v" id="v14" data-v="14" data-xref="صخر الثانية 1:10; لوقا 13:23–24; متّى 7:13–14"> والعِبرةُ مِن هذا المَثَلِ أيُّها الحاضِرونَ كَثرةُ المَدعوّينَ أمامَ قِلّةِ مَن اصطَفاهُم اللهُ".</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="لوقا 20:20–26; مرقس 12:13–17; مقتطفات من كتاب الزبور 2:2"> ومَضى بَعضُ المُتَشَدِّدينَ في التّآمرِ على سَيِّدِنا عيسى (سلامُهُ علينا) مِن أجلِ الإيقاعِ بِهِ. فأرادوا أن يُوقِعوهُ في كَلامٍ يُدينُهُ،</span>
<span className="v" id="v16" data-v="16" data-xref="مرقس 3:6; مرقس 8:15; مرقس 12:14"> فأرسَلوا مُناصريهِم مَعَ بَعضٍ مِن مُؤيّدي عائلةِ الأميرِ أنْتيباسَ بِن هيرودُسَ وسألوا عيسى (سلامُهُ علينا) قائلينَ: "أيّها المُعَلِّمُ، نَعرِفُ أنّكَ صادِقٌ في كَلامِكَ وتَعليمِكَ، لا تَخافُ في الحقِّ لومةَ لائمٍ، ونَعلَمُ أنّكَ تَهدي النّاسَ بالحقِّ إلى الصِّراطِ المُستَقيمِ، ولا تُحابي أحَدًا لأنّكَ لا تُعيرُ المَناصِبَ اهتِمامًا.</span>
<span className="v" id="v17" data-v="17" data-xref="لوقا 2:1; روما 13:6–7; متّى 17:25"> فأخبِرنا برأيكَ في الجِزيةِ الّتي تُدفَعُ إلى القَيصرِ، أحَلالٌ دَفعُها أم حَرامٌ؟"</span>
<span className="v" id="v18" data-v="18" data-xref="يوحنّا 8:6; لوقا 5:22; يوحنّا 2:25"> فأدرَكَ عيسى (سلامُهُ علينا) ما يَرمُونَ إليهِ، والتَفَتَ إليهِم وأجابَهُم: "أيُّها المُنافِقونَ، لِمَ تُحاولونَ الإيقاعَ بي؟<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="22‏.18 كان سيدنا عيسى (سلامُهُ علينا) على علم بمكرهم، فإذا وافقهم على دفع الجزية للرومان المحتلين، صار في مواجهة مع المتشددين ومع شعبه، وإذا أجابهم بعدم جواز دفع الجزية، صار في مواجهة مع القيصر ومؤيديه ومناصريه.">a</a></sup></span>
<span className="v" id="v19" data-v="19" data-xref="متّى 20:2; الرؤيا 6:6; متّى 18:28"> ائتُوني بالعُملةِ الّتي تَدفعونَها جِزيةً إلى الرّومان!" فجيءَ إليهِ بدينارٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="لوقا 20:24"> فقالَ لهُم: "أيُّ صورةٍ ترَونَها على هذا الدِّينار، واسمَ مَن تَقرؤونَ عليهِ؟!"</span>
<span className="v" id="v21" data-v="21" data-xref="روما 13:7; الأمثال 24:21; سيرة الحواريّين 4:19"> فأجابوهُ: "إنّها صُورةُ القَيصرِ". فقالَ: "إذن أعطُوا ما لقَيصرَ لقَيصرَ، وما للهِ للهِ".</span>
<span className="v" id="v22" data-v="22" data-xref="مرقس 12:12; متّى 22:46; لوقا 20:25–26"> وعِندَما سَمِعوا إجابتَهُ هذِهِ أصابَهُم الذُّهولُ، فتَرَكوهُ وانصَرَفوا.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="سيرة الحواريّين 4:1; مرقس 12:18–27; سيرة الحواريّين 23:6–8"> ثُمّ جاءَهُ في اليَومِ نَفسِهِ، جَماعةٌ مِن الصَّدّوقيّينَ، وهُم قَومٌ يُنكِرونَ القيامةَ والحَياة بَعدَ المَوتِ، فسألوهُ:</span>
<span className="v" id="v24" data-v="24" data-xref="Deuteronomy 25:5–10; لوقا 20:28; راعُوث 1:11"> "أيُّها المُعَلِّمُ، أوصانا سَيِّدُنا موسى في التَّوراةِ بما يَلي: "إذا ماتَ رَجُلٌ عن امرأةٍ دونَ أن يُخَلِّفَ ولدًا يَرِثُهُ، فعلى أخيهِ الاقتِرانُ بأرمَلتِهِ لإنجابِ أولادٍ فيَنحَصِرَ إرثُ أخيهِ فيهِم".</span>
<span className="v" id="v25" data-v="25" data-xref="لوقا 20:29–33; مرقس 12:19–23; العبرانيين 9:27"> وإنّا لَسائلوكَ عن سَبعةِ إخوةٍ، تُوفّيَ أحَدُهُم عن امرأةٍ ولم يَكُن لهُ وَريثٌ مِنها، فتَرَكَ زَوجَتَهُ لأخيهِ،</span>
<span className="v" id="v26" data-v="26"> وكذلِكَ الأخُ الثّاني ثُمّ الثّالثُ، وهكذا دَوالَيكَ حتّى تَزَوَّجَها السّابعُ،</span>
<span className="v" id="v27" data-v="27"> ثُمّ تُوُفِيَّت تِلكَ الأرمَلةُ.</span>
<span className="v" id="v28" data-v="28"> فإنْ كانَت هُناكَ قيامةٌ، ثُمّ أتوا وإيّاها في ذلِكَ اليومِ، فلأيّهِم تَكون زَوجةً وقد عَقَدَ عليها القِرانَ الإخوةُ السَّبعةُ في الدُّنيا الواحِدُ تلوَ الآخر؟!"<sup className="fn"><a href="#fn-b" id="fnref-b" data-fn="22‏.28 أراد الصدّوقيون أن يبينوا من خلال هذا السؤال أن الإيمان بقيامة الموتى ليس من المنطق (كما زعموا)، وأرادوا أيضا إحراج السيد المسيح.">b</a></sup></span>
<span className="v" id="v29" data-v="29" data-xref="يوحنّا 20:9; روما 15:4; لوقا 1:37"> فأجابَهُم (سلامُهُ علينا): "ألا إنّكُم على ضَلالٍ مُبينٍ، فأنتُم لا تُدرِكونَ ما جاءَ في الكُتُبِ السَّماويّةِ المُقَدَّسة، كما لا تُدرِكونَ قُدرةَ اللهِ!</span>
<span className="v" id="v30" data-v="30" data-xref="لوقا 20:34–36; مرقس 12:24–25; متّى 24:38"> فلا زَواجَ يَومَ القيامة، بل سيَكونُ النّاسُ مِن هذِهِ النّاحيةِ مِثلَ المَلائكةِ في السَّماء.</span>
<span className="v" id="v31" data-v="31" data-xref="متّى 12:3; متّى 21:16; متّى 21:42"> أمّا عن حَقيقةِ قيامةِ المَوتى، أفلم تَقرأُوا ما قالَهُ اللهُ في التَّوراةِ:</span>
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 3:6; مرقس 12:26–27; لوقا 20:37–38"> "يا موسى، إنّي أنا اللهُ، رَبُّ إبراهيمَ ورَبُّ إسحقَ ورَبُّ يَعقوبَ"؟ وهذا يَعني أنّ عِبادَ اللهِ أحياءٌ عِندَ رَبِّهِم، ولَيسوا بأمواتٍ".</span>
<span className="v" id="v33" data-v="33" data-xref="مرقس 12:17; مرقس 6:2; يوحنّا 7:46"> فتَعَجَّبَ النّاسُ مِن كَلامِهِ وقوّةِ حُجّتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="مرقس 12:28–31; متّى 25:3–5; متّى 12:14"> وعِندَما عَلِمَ المُتَشَدِّدونَ بخَبَرِ إبطالِ سَيِّدِنا عيسى (سلامُهُ علينا) دَعوى الصَّدّوقيّينَ، اجتَمَعوا مَعًا</span>
<span className="v" id="v35" data-v="35" data-xref="لوقا 7:30; لوقا 10:25–37; لوقا 11:52"> يُريدونَ أن يُحرِجوهُ (سلامُهُ علينا) بسؤالٍ، فقالَ أحَدُهُم وهو فَقيهٌ بالتَّوراةِ:</span>
<span className="v" id="v36" data-v="36" data-xref="مرقس 12:28–33; متّى 5:19–20; لوقا 11:42"> "أيُّها المُعَلِّمُ، ما هي أهمُّ وَصيّةٍ جاءَت في التَّوراة؟"</span>
<span className="v" id="v37" data-v="37" data-xref="مرقس 12:29–30; لوقا 10:27; Deuteronomy 6:5"> فأجابَهُم قائلاً: "أحِبَّ اللهَ رَبَّكَ بكُلِّ قَلبِكَ وبكُلِّ نَفسِكَ وبكُلّ عَقلِكَ"،</span>
<span className="v" id="v38" data-v="38"> فتِلكَ هي الوَصيّةُ الأعظَمُ والأهَمّ،</span>
<span className="v" id="v39" data-v="39" data-xref="مرقس 12:31; لوقا 10:27–37; Leviticus 19:18"> والوَصيّةُ الثّانية مِثْلُها وهي: "أحِبَّ جارَكَ كَما تُحِبُّ نَفسَكَ".</span>
<span className="v" id="v40" data-v="40" data-xref="روما 13:9; متّى 7:12; يوحنّا الأولى 4:19–21"> وعلى هاتينِ الوَصيّتينِ يَقومُ كُلُّ ما في التَّوراةِ وكُتُبِ الأنبياءِ مِن وَصايا".</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="لوقا 20:41–44; مرقس 12:35–37; متّى 22:34"> وعِندَما كانَ المُتَشَدِّدونَ مُجتَمِعينَ عِندَهُ، سألَهُم سَيِّدُنا عيسى (سلامُهُ علينا) قائلاً:</span>
<span className="v" id="v42" data-v="42" data-xref="إرْمِيا 23:5–6; يوحنّا 6:68–69; متّى 16:13–17"> "ماذا تَقولونَ في نَسَبِ المَسيحِ المُنتَظَرِ؟" فأجابوهُ: "إنّهُ مِن سُلالةِ النَّبيِّ داودَ وسيَرِثُ عَرشَهُ".</span>
<span className="v" id="v43" data-v="43" data-xref="2 Samuel 23:2; الرؤيا 4:2; مرقس 12:36"> فسألَهُم ثانيةً: "فكَيفَ دَعاهُ النَّبيُّ داودُ بِوحيِ رُوحِ اللهِ مَولاهُ؟</span>
<span className="v" id="v44" data-v="44" data-xref="مقتطفات من كتاب الزبور 110:1; العبرانيين 1:13; سيرة الحواريّين 2:34–35"> فقد جاءَ على لِسانِ النَّبيِّ داودَ في الزَّبورِ: "قالَ اللهُ تَعالى لمَولاي: اجلِس عن يَميني، حتّى أقهَرَ أَعدَاءَكَ وأجعَلَهُم تَحتَ قَدمَيْكَ صاغِرينَ".</span>
<span className="v" id="v45" data-v="45" data-xref="روما 1:3–4; الرؤيا 22:16; يوحنّا 8:58"> فما دامَ النَّبيُّ داودُ نَفسُهُ يَدعو المَسيحَ مَولاهُ، أفلَيسَ هذا دَليلاً على أنّ المَسيحَ المُنتَظَرَ أعظَمُ مِن النَّبيِّ داودَ؟"</span>
<span className="v" id="v46" data-v="46" data-xref="مرقس 12:34; لوقا 20:40; لوقا 14:6"> فسَكَتوا جَميعًا، ولم يَجرؤْ أحَدٌ مُنذُ ذلِكَ اليومِ على إحراجِهِ بسؤالٍ.</span>
