<span className="v" id="v1" data-v="1" data-xref="مرقس 15:1; يوحنّا 18:28; Micah 2:1"> وعِندَ طُلوعِ الفَجرِ، تَشاوَرَ جَميعُ رُؤساءِ الأحبارِ وشُيوخُ الشَّعبِ على عيسى (سلامُهُ علينا) ليُعدِمَهُ الرّومانُ.</span>
<span className="v" id="v2" data-v="2" data-xref="متّى 20:19; سيرة الحواريّين 3:13; تيموتاوي الأولى 6:13"> فقَيَّدوهُ وساقُوهُ مِن أجلِ تَسليمِهِ إلى بيلاطُسَ الحاكمِ الرُّومانيّ.<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="27‏.2 لم يكن لليهود تحت الاحتلال الحق في الحكم بالإعدام على المساجين. فقد كان ذلك امتيازا للحاكم الروماني. لذلك أخذ المجلس الأعلى سيدنا عيسى إلى بيلاطس للحكم عليه بالإعدام.">a</a></sup></span>
<span className="v" id="v3" data-v="3" data-xref="كورنتوس الثانية 7:10; لوقا 22:47–48; أيُّوب 20:15–29"> وبَعدَ أن تَمَّت إدانةُ سَيِّدَنا عيسى (سلامُهُ علينا)، رأى يَهوذا الخائنُ ذلِكَ فنَدِمَ نَدمًا شَديدًا، وأعادَ ثَمَنَ خيانتِهِ (الثلاثين قِطعةً من الفضّة) إلى رُؤساءِ الأحبارِ والشُّيوخِ</span>
<span className="v" id="v4" data-v="4" data-xref="صَمُوئيلَ الأوَّلُ 15:30; تيتوس 1:16; سيرة الحواريّين 18:15–17"> قائلاً: "لقد أخطَأتُ إذ خُنتُ إنسانًا بَريئًا!" ولكنّهُم لم يأبَهوا لقَولِهِ وأجابوهُ قائلينَ: "لا يَهُمُّنا الأمرُ، فالشَّأنُ شَأنُكَ". ورَفَضوا أخذَ المالِ.</span>
<span className="v" id="v5" data-v="5" data-xref="سيرة الحواريّين 1:18–19; 2 Samuel 17:23; 1 Kings 16:18"> وانصَرَفَ عَنهُم يَهوذا بَعدَ أن رَمى ذلِكَ المالَ في الحَرَمِ الشَّريفِ، ثُمّ تَوَجَّهَ إلى مَكانٍ يَشنُقُ فيهِ نَفسَهُ.</span>
<span className="v" id="v6" data-v="6" data-xref="لوقا 6:7–9; Deuteronomy 23:18; إشَعْياء 61:8"> والتَقَطَ رُؤساءُ الأحبارِ ما رَمى بِهِ مِن قِطَعِ الفِضّةِ قائلينَ: "هذا المالُ حَرامٌ لأنّهُ ثَمَنُ دَمِ إنسانٍ، ولا يَجوزُ وَضعُهُ مَعَ مالِ بَيتِ اللهِ".</span>
<span className="v" id="v7" data-v="7"> وأخَذوا يَتَشاوَرونَ فيما بَينَهُم حَول كَيفيّةِ التَّصَرُّفِ فيهِ، فقَرَّروا شِراءَ حَقلٍ يُدعَى حَقلَ الفَخّاريّ، ليَكونَ مَدفَنًا للأغرابِ.<sup className="fn"><a href="#fn-b" id="fnref-b" data-fn="27‏.7 كان يعتبر دفن الغرباء الذين لا أهل لهم عمل خيْر.">b</a></sup></span>
<span className="v" id="v8" data-v="8" data-xref="سيرة الحواريّين 1:19; متّى 28:15; Joshua 4:9"> وسُمِّيَ هذا الحَقلُ مُنذُ ذلِكَ الحينِ إلى اليَومِ حَقلِ الدَّمِ.</span>
<span className="v" id="v9" data-v="9" data-xref="Zechariah 11:12–13; متّى 1:22; Leviticus 27:2–7"> وبهذا تَمَّ قولُ النَّبيِّ إرميا: "وأخَذوا الثّلاثينَ قِطعةً مِن الفِضّةِ، المَبلَغَ الّذي اتَفَّقَ عليهِ بَنو اسرائيلَ ليَكونَ ثَمَنَهُ،</span>
<span className="v" id="v10" data-v="10" data-xref="Zechariah 11:13"> واشتَروا بها حَقلَ الفَخّاريّ. هكذا قَضَى اللهُ".</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="لوقا 23:2–3; مرقس 15:2–5; متّى 26:25"> وأُحضِرَ سَيِّدُنا عيسى (سلامُهُ علينا) إلى الحاكِمِ الرّومانيِّ بيلاطُسَ فسألَهُ: "هل أنتَ مَلِكُ اليَهودِ؟" فأجابَهُ (سلامُهُ علينا): "هو ما تَقولُ".</span>
<span className="v" id="v12" data-v="12" data-xref="متّى 26:63; إشَعْياء 53:7; متّى 27:14"> وكانَ رُؤساءُ الأحبارِ والشُّيوخُ يَكيلونَ التُّهَمَ لهُ، وهو (سلامُهُ علينا) صامِتٌ.</span>
<span className="v" id="v13" data-v="13" data-xref="متّى 26:62; يوحنّا 18:35; سيرة الحواريّين 22:24"> فالتَفَتَ إليهِ بيلاطُسُ قائلاً: "ألم تَسمَعْ ما يُوَجِّهونَ إليكَ مِن تُهَم؟ ألا تُجيبُهُم، ألا تَرُدُّ عليهِم؟!"</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 8:18; مقتطفات من كتاب الزبور 71:7; كورنتوس الأولى 4:9"> وظَلَّ عيسى (سلامُهُ علينا) على صَمتِهِ، مِمّا أثارَ دَهشةَ الحاكمِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="يوحنّا 18:38–40; لوقا 23:16; يوحنّا 19:16"> وكانَ مِن عادةِ الحاكِمِ بمُناسَبةِ العِيدِ، أن يُطلِقَ سَراحَ أحَدِ المَساجين مِمّن يَطلُبُ النّاسُ إطلاقَ سَراحِهِ.</span>
<span className="v" id="v16" data-v="16" data-xref="مرقس 15:7; سيرة الحواريّين 3:14; لوقا 23:25"> وكانَ هُناكَ سَجينٌ ذو سَوابِقَ خَطيرةٍ اسمُهُ باراباسُ.</span>
<span className="v" id="v17" data-v="17" data-xref="يوحنّا 19:15; 1 Kings 18:21; مرقس 15:9–12"> فلمّا اجتَمَعَ النّاسُ ليَختاروا مَن سيُطلِقُ الحاكِمُ سَراحَهُ، كانَ بيلاطُس على يَقينٍ بأنّ رُؤساءَ الأحبارِ إنّما سَلَّموا عيسى (سلامُهُ علينا) حَسَدًا للشَّعبيّةِ الّتي حَظِيَ بها بَينَ النّاسِ، فخاطَبَ الجُمهورَ قائلاً: " مَن مِنَ الاثنينِ تُريدونَ إطلاقَ سَراحِهِ؟ أباراباسُ أم عيسى الذّي يُدعَى المَسيحَ؟"</span>
<span className="v" id="v19" data-v="19" data-xref="Zechariah 9:9; الأمثال 29:1; متّى 27:24"> وعِندَما كانَ بيلاطسُ جالِسًا على مَنَصَّةِ القَضاءِ يَنتَظِرُ قَرارَ النّاسِ، أرسَلَت إليهِ زَوجَتُهُ تُحَذِّرُهُ قائلة: "إيّاكَ أن تُؤذيَ هذا الرَّجُلَ الصّالحَ، فقد أفَقتُ مِن نومي مُتَألِّمةً مِن أجلِهِ بَعدَ رؤيا جاءَتني لَيلةَ أمسِ".</span>
<span className="v" id="v20" data-v="20" data-xref="سيرة الحواريّين 3:14–15; يوحنّا 19:15–16; يوحنّا 18:40"> إلاّ أنّ رُؤساءَ الأحبارِ والشُّيوخَ حَرَّضوا النّاسَ حتّى يُطالِبوا بإطلاقِ سَراحِ باراباسَ وإعدامِ عيسى (سلامُهُ علينا).</span>
<span className="v" id="v21" data-v="21"> فسألَهُم الحاكِمُ ثانيةً: "أيُّهُما تُريدونَ أن أُطلِقَ سَراحَهُ؟" فأجابوهُ: "باراباسُ!"</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 53:2–3; إشَعْياء 49:7; سيرة الحواريّين 13:38"> فقالَ لهُم بيلاطسُ: "وماذا بشأنِ عيسى المَدعوِّ المَسيحَ المُنتَظَرِ؟!" فأجابوهُ بصَوتٍ واحدٍ: "اُصلُبْه! اُصلُبه!"</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 37:18–19; سيرة الحواريّين 7:57; سيرة الحواريّين 23:12–15"> فرَدَّ عليهِم قائلاً: "وما هو الذَّنبُ الّذي ارتَكَبَهُ؟!" فقالوا صارِخينَ: "اُصلُبْهُ! اُصلُبْهُ!"</span>
<span className="v" id="v24" data-v="24" data-xref="متّى 27:4; مقتطفات من كتاب الزبور 26:6; كورنتوس الثانية 5:21"> وعِندَما رأى بيلاطسُ أنّهُ لا فائدةَ مِن إلحاحِهِ، وأنّ اضطِرابَ الجُموعِ يَشتَدُّ، وأنّهُم مُصَمِّمونَ على صَلبِ سَيِّدِنا عيسى، غَسَلَ يدَيهِ أمامَهُم قائلاً: "إنّي بَريءٌ مِن دَمِ هذا الرَّجُلِ، وإنّكُم تَتَحَمَّلونَ وِزرَ قَرارِكُم!"</span>
<span className="v" id="v25" data-v="25" data-xref="Joshua 2:19; سيرة الحواريّين 5:28; سيرة الحواريّين 7:52"> فأجابوا جَميعًا: "نحنُ وأولادُنا نَتَحَمَّلُ وِزرَ دَمِهِ!"</span>
<span className="v" id="v26" data-v="26" data-xref="إشَعْياء 53:5; يوحنّا 19:1; مرقس 15:15"> فما كانَ مِن بيلاطسَ إلاّ أن أصدَرَ أمرًا بإطلاقِ سَراحِ باراباسَ، وأمَرَ بجَلدِ سَيِّدِنا عيسى وصَلبِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="يوحنّا 18:28; يوحنّا 18:33; سيرة الحواريّين 10:1"> فأخَذَ جُندُ الحاكِمِ عيسى (سلامُهُ علينا) إلى قَصرِ الوِلايةِ، واجتَمَعَت عليهِ الكَتيبةُ كُلُّها،</span>
<span className="v" id="v28" data-v="28" data-xref="مرقس 15:17; يوحنّا 19:2–5; لوقا 23:11"> ثُمّ نَزَعوا عَنهُ ثِيابَهُ وألبَسوهُ رِداءً قِرمِزيّ اللّونِ استِهزاءً،<sup className="fn"><a href="#fn-c" id="fnref-c" data-fn="27‏.28 كان الرداء القرمزي لأحد الجنود الرومان على الأرجح، وهو يختلف عن الرداء الأرجواني الذي ألبسه هيرودس لسيدنا عيسى استهزاء به.">c</a></sup></span>
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 53:3; متّى 20:19; مقتطفات من كتاب الزبور 69:19–20"> وضَفروا لهُ تاجًا مِن الشَّوكِ ووَضَعوهُ على رأسِهِ، ووَضَعوا عَصًا في يدِهِ اليُمنى، ورَكَعوا على رُكَبِهِم أمامَهُ وهم يَقولونَ ساخِرينَ: "عاشَ مَلِكُ اليَهودِ!"</span>
<span className="v" id="v30" data-v="30" data-xref="إشَعْياء 50:6; متّى 26:67; إشَعْياء 53:7"> ثُمّ بَصَقوا عليهِ وأخَذوا مِنهُ العَصا، وانهالوا بها ضَربًا على رأسِهِ.</span>
<span className="v" id="v31" data-v="31" data-xref="إشَعْياء 53:7; 1 Kings 21:13; 1 Kings 21:10"> ثُمّ نَزَعوا الرِّداءَ عَنهُ، وألبَسوهُ ثِيابَهُ، واقتادوهُ للصَّلبِ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="مرقس 15:21; متّى 16:24; لوقا 23:26"> وعِندَ خُروجِهِم مِن المَدينةِ، قابَلوا رَجُلاً اسمُهُ سِمْعانُ مِن مَدينةِ قورينا في ليبيا، فأجبَروهُ على حَملِ صَليبِ سَيِّدِنا عيسى (سلامُهُ علينا).</span>
<span className="v" id="v33" data-v="33" data-xref="يوحنّا 19:17–19; مرقس 15:22–32; مرقس 15:34"> وعِندَما وَصَلوا إلى مَكانٍ يُدعَى الجُلْجُثَةَ، أي مَكانُ الجُمجُمةِ،</span>
<span className="v" id="v34" data-v="34" data-xref="مقتطفات من كتاب الزبور 69:21; مرقس 15:23; متّى 27:48"> قَدَّموا لعيسى خَمرةً مَمزوجةً بمادّةٍ مُرّةِ المَذاق، فلمّا تَذَوَّقَها (سلامُهُ علينا)، لَفَظَها.<sup className="fn"><a href="#fn-d" id="fnref-d" data-fn="27‏.34 قدّم الجنود هذا الشراب المُرّ لكي يضاعفوا تعذيبه إذ كان يعطش كثيرًا.">d</a></sup></span>
<span className="v" id="v35" data-v="35" data-xref="مقتطفات من كتاب الزبور 22:18; مرقس 15:24–32; لوقا 23:34"> ثُمّ صَلَبوهُ واقتَرَعوا على ثِيابِهِ فيما بَينَهُم، واقتَسموها.</span>
<span className="v" id="v36" data-v="36" data-xref="متّى 27:54; مرقس 15:44; مرقس 15:39"> وجَلَسوا حَولَهُ لِحِراسَتِهِ،</span>
<span className="v" id="v37" data-v="37" data-xref="لوقا 23:38; يوحنّا 19:19–22; مرقس 15:26"> ووَضَعوا فَوقَ رأسِهِ لَوحةً تَحمِلُ التُّهمةَ الّتي وُجِّهَت إليهِ وكانَ مَكتوبًا فيها: "هذا عيسى مَلِكُ اليَهودِ".</span>
<span className="v" id="v38" data-v="38" data-xref="يوحنّا 19:18; لوقا 23:39–43; إشَعْياء 53:12"> وقد صُلِبَ مَعَهُ رَجُلانِ مُتَمَرِّدانِ أحَدُهُما عن يَمينِهِ، والآخرُ عن شِمالِهِ.<sup className="fn"><a href="#fn-e" id="fnref-e" data-fn="27‏.38 أطلق الرومان على الثوار اسم قطاع الطرق لنزع الشرعية عن تحرّكاتهم الثورية. فقطاع الطرق في اليونانية تُرجمت بـ&quot;اللص&quot;، إلا أن الدارسين يجمعون اليوم على أن أولئك المشار إليهم في الحقيقة كانوا ثوارا.">e</a></sup></span>
<span className="v" id="v39" data-v="39" data-xref="مقتطفات من كتاب الزبور 109:25; Lamentations 1:12; مقتطفات من كتاب الزبور 69:7–12"> وكانَ المارّةُ يَكيلونَ لهُ (سلامُهُ علينا) الشَّتائمَ مُحَرِّكينَ رُؤوسَهُم تَنَدُّرًا بِهِ</span>
<span className="v" id="v40" data-v="40" data-xref="متّى 26:61; متّى 4:6; متّى 4:3"> قائلينَ: "يا هادِمَ بيتِ اللهِ وبانيهِ في ثَلاثةِ أيّامٍ! هَيّا أَنقِذْ نَفسَكَ وتَخَلَّصْ مِن الصَّلبِ إن كُنتَ حقًّا الابنَ الرّوحيّ للهِ!"</span>
<span className="v" id="v41" data-v="41" data-xref="مقتطفات من كتاب الزبور 35:26; إشَعْياء 28:22; مقتطفات من كتاب الزبور 22:12–13"> وكانَ رُؤساءُ الأحبارِ والفُقَهاءُ والشُّيوخُ يَهزأُونَ مِنهُ أيضًا بالطّريقةِ نَفسِها ويَقولونَ:</span>
<span className="v" id="v42" data-v="42" data-xref="يوحنّا 1:49; متّى 27:37; سيرة الحواريّين 4:14"> "أنقَذَ غَيرَهُ ولَيسَ بقادرٍ على إنقاذِ نَفسِهِ، يَدّعِي أنّهُ مَلِكُ بَني يَعقوبَ! فليُخَلِّصْ نَفسَهُ مِن الصَّلبِ وعِندئذٍ نؤمنُ بِهِ.</span>
<span className="v" id="v43" data-v="43" data-xref="مقتطفات من كتاب الزبور 22:8; يوحنّا 10:36; إشَعْياء 37:10"> لقد كانَ يَتَوَكِّلُ على اللهِ ويَقولُ: "أنا الابنُ الرُّوحيُّ لهُ تَعالى". فإن كانَ اللهُ راضيًا عَنهُ حقًّا فليُنقِذْهُ".</span>
<span className="v" id="v44" data-v="44" data-xref="متّى 27:38; مرقس 15:32; لوقا 23:39–43"> إضافةً إلى ذلِكَ كُلِّهِ، لم يَنجُ مِن إهانةِ المُتَمَرِّديْنِ اللّذَينِ كانا على جانبَيهِ إذ كانا يَكيلانِ لهُ الشّتائمَ بالأسلوبِ نَفسِهِ.</span>
</div>
