<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 65:17–19; صخر الثانية 3:13; إشَعْياء 66:22"> وقد تَجَلَّت أمامي سَماءٌ جَديدةٌ وأرضٌ جَديدةٌ، بَعدَ اختِفاءِ السَّماءِ الأُولى والأرضِ الأُولى، وغابَ البَحرُ عنِ الأرضِ الجَديدةِ.</span>
<span className="v" id="v2" data-v="2" data-xref="العبرانيين 11:10; الرؤيا 3:12; الرؤيا 21:10"> ورأيتُ المَدينةَ المُقَدَّسةَ، وهِي المَكانُ القُدسِيُّ الجَديدُ، تَنزِلُ مِن السَّماءِ مِن عِندِ اللهِ كعَروسٍ تَبَرَّجَت وتَهَيّأت لِلِقاءِ زَوجِها.</span>
<span className="v" id="v3" data-v="3" data-xref="Ezekiel 37:27; كورنتوس الثانية 6:16; يوحنّا 14:23"> وسَمِعتُ صَوتًا عاليًا لهاتِفٍ انبَعَثَ مِن قُربِ عَرشِ اللهِ المَجيدِ قائِلاً: "الآنَ يَتَجَلّى اللهُ بَينَ العالَمينَ ويَحِلُّ بَينَهُم، ومِن الآنَ تُصبِحُ شُعوبُ العالَمِ كُلِّها عِبادَهُ الصّالِحينَ، إنّ اللهَ رَفيقُهُم الأعلى.</span>
<span className="v" id="v4" data-v="4" data-xref="إشَعْياء 25:8; الرؤيا 7:17; كورنتوس الأولى 15:26"> ومِن عُيونِهِم يُكَفكِفُ تَعالى كُلّ دَمعةٍ. فلا مَوتَ يَتَسَلَّطُ عليهِم بَعدَ الآنَ، ولا بُكاءَ ولا ألمَ يُصيبُهُم ولا هُم يَحزَنونَ، فكُلُّ ما مَضى، قدِ انتَهى وانقَضى".</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="كورنتوس الثانية 5:17; إشَعْياء 42:9; إشَعْياء 43:19"> وأوحى إليّ رَبُّ العَرشِ: "إنّي خالِقٌ الآنَ كُلَّ شَيءٍ مِن جَديدٍ". ثُمّ أوحى إليّ ثانيةً: "أُكتُبْ ما يَلي: إنّ هذا كَلامُ صِدقٍ وحقٍّ مُبينٍ".</span>
<span className="v" id="v6" data-v="6" data-xref="يوحنّا 7:37–38; يوحنّا 4:10; الرؤيا 7:17"> وقالَ أيضًا: "لقد قُضِي الأمرُ. أنا الألِفُ والياءُ، البِدايةُ والنِّهايةُ. أروي كُلَّ ظَمآنٍ مِن يَنبوعِ ماءِ الخُلدِ بِلا حِسابٍ.</span>
<span className="v" id="v7" data-v="7" data-xref="كورنتوس الثانية 6:18; العبرانيين 8:10; الرؤيا 2:11"> إنّ هذا جَزاءُ الفائِزينَ، وأنا أكونُ رَبَّهُم، وهُم يَكونونَ أهلَ بَيتي الأحِبّاءَ.</span>
<span className="v" id="v8" data-v="8" data-xref="كورنتوس الأولى 6:9–10; الرؤيا 2:11; أفاسوس 5:5–6"> أمّا الجُبَناءُ والكُفّارُ والفاسِدونَ، والمُجرِمونَ والفُجّارُ والسَّحَرَةُ وعَبَدَةُ الأصنامِ والكذّابونَ، فمَصيرُهُم الوُقوعُ في البُحيرةِ المُتَّقِدةِ نارًا وحِجارةَ كِبريتِ مُشتَعِلة. وهذِهِ البُحيرةُ هي الهَلاكُ المُنتَظَر، ومَعناهُ الهَلاكُ في النّارِ".</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الرؤيا 21:2; الرؤيا 19:7; الرؤيا 16:1–17"> وأقبَلَ عليّ أحَدُ المَلائِكةِ السَّبعةِ أولئكَ الّذينَ سَكَبوا كؤوسَهُم السَّبعَ المَليئةَ بِالبَلايا السَّبعِ الأخيرةِ، وقالَ لي: "أقبِلْ لأُرِيك جَماعةَ المؤمنينَ، الّذينَ تَهيَّؤوا لِمَولانا عيسى الذِّبحِ العَظيمِ، كَما تَتَهَيّأُ العَروسُ لزَوجِها الكَريمِ".</span>
<span className="v" id="v10" data-v="10" data-xref="الرؤيا 21:2; Ezekiel 11:24; Ezekiel 40:1–49"> وأخَذَني المَلاكُ بِقُوّةِ رُوحِ اللهِ إلى جَبَلٍ شاهِقٍ، وأراني المَكانَ القُدُسيّ الجَديد، إنّهُ المَدينةُ المُقَدَّسةَ وهي تَنزِلُ مِن السَّماءِ مِن عِندِ اللهِ،</span>
<span className="v" id="v11" data-v="11" data-xref="الرؤيا 4:6; الرؤيا 22:5; الرؤيا 22:1"> تَتَألّقُ بِما عليها مِن جَلالٍ، وتَتَلألأُ نورًا وجَمالاً كَحَجَرٍ كَريمٍ وكَاليَشبِ والبِلّورِ النَّقيِّ.</span>
<span className="v" id="v12" data-v="12" data-xref="Ezekiel 48:31–34; الرؤيا 7:4–8; الرؤيا 21:17–21"> ويُحيطُ بها سُورٌ ضَخمٌ عالٍ، وفيهِ اثنتا عَشَرة بَوّابةٍ يَحميها اثنا عَشَر مَلاك. وقد كُتِبَ على كُلِّ بابٍ اسمٌ مِن أسماءِ عَشائِرِ بَني يَعقوبَ الاثنَي عَشَر،</span>
<span className="v" id="v13" data-v="13" data-xref="Ezekiel 48:31–34"> ثلاثُ بَوّاباتٍ مِنها مِن الشَّرقِ، وثَلاثٌ مِن الشّمالِ، وثَلاثٌ مِن الجَنوبِ، وثَلاثٌ مِن الغَربِ.</span>
<span className="v" id="v14" data-v="14" data-xref="العبرانيين 11:10; أفاسوس 2:20; إشَعْياء 54:11"> ويَقومُ سُورُ المَدينةِ على اثنَتَي عَشَرة رَكيزةٍ، كُتِبَ عليها أسماءُ الحَواريّينَ الاثني عَشَر الّذينَ أرسَلَهُم سَيِّدُنا عيسى الذِّبحُ العَظيمُ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="Zechariah 2:1; الرؤيا 11:1–2; Ezekiel 41:1–5"> وكانَ المَلاكُ الّذي كَلَّمَني يَحمِلُ عَصا قِياسٍ مِن ذَهَبٍ لِيَقيسَ بِها المَدينةَ وبَوّاباتِها وسُورَها.</span>
<span className="v" id="v16" data-v="16" data-xref="Ezekiel 48:35; Ezekiel 48:8–20"> وكانَتِ المَدينةُ مُكَعَّبةً في شَكلِها، لا فَرقَ بَينَ عَرضِها وطُولِها وعُلوّها. فقاسَها المَلاكُ بِالعَصا، فكانَت مَساحتُها ألفًا وخَمسَ مِئةِ ميلٍ وفيها تَتَساوى الأبعادُ.<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="21‏.16 يمثّل المحراب الأقدس في بيت الله الذي بناه النبي سليمان الكعبة المثلى والكاملة آذاك، شأن الخيمة المقدسة التي سبقته في زمن النبي موسى. وفي هذا المقطع تذكّر المدينة الموصوفة على أنها مكعّب القرّاء بهذه المواقع التاريخية التي تجلّى فيها الله.">a</a></sup></span>
<span className="v" id="v17" data-v="17" data-xref="الرؤيا 7:4; Deuteronomy 3:11; الرؤيا 13:18"> ثُمّ قاسَ عَرضَ السُّورِ فرأى أنّهُ مِئةٌ وأربَعةٌ وأربَعونَ ذِراعًا،<sup className="fn"><a href="#fn-b" id="fnref-b" data-fn="21‏.17 يساوي عرض السور المذكور هنا تقريبا 65 مِتْرًا.">b</a></sup>وكانَ المَلاكُ يَقيسُ بِالذِّراعِ الّتي يَقيسُ بِها البَشَرُ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الرؤيا 21:11; الرؤيا 21:21; الرؤيا 21:19"> وكانَ السُّورُ مِن اليَشبِ، والمَدينةُ مِن الذَّهَبِ النَّقيِّ كأنّهُ البِلّورُ الشّفّافُ.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 54:11–12; الخُرُوج 28:17–21; أيُّوب 28:16–19"> وقد زُيِّنَت رَكائِزُ السُّورِ بِكُلِّ أنواعِ الحَجَرِ الكَريمِ: فالرَّكيزةُ الأُولى يَشبٌ، والثّانِيةُ ياقوتٌ أزرقُ، والثّالِثةُ عَقيقٌ أبيَضُ، والرّابِعةُ زُمُرُّدٌ،</span>
<span className="v" id="v20" data-v="20" data-xref="الرؤيا 4:3"> والخامِسةُ عَقيقٌ قاتِمٌ، والسّادِسةُ عَقيقٌ أحمَرُ، والسّابِعةُ زَبَرجَدٌ، والثّامِنةُ زُمُرُّدٌ سِلقيٌّ، والتّاسِعةُ ياقوتٌ أصفَرُ، والعاشِرةُ عَقيقٌ أخضَرُ، والحادِيةَ عشرة فَيروزٌ، والثّانِيّةَ عشرةَ جَمَشْتٌ.</span>
<span className="v" id="v21" data-v="21" data-xref="الرؤيا 21:18; الرؤيا 17:4; الرؤيا 22:2"> وكانَت البَوّاباتُ الاثنَتا عَشَرة تُشبِهُ اثنَتَي عَشَرة لُؤلُؤة، وكُلُّ بابٍ صُنِعَ مِن لُؤلُؤةٍ واحِدةٍ تَفوقُ الوَصفَ. أمّا ساحةُ المَدينةِ فكانَت مِن ذَهَبٍ خالِصٍ كالزُّجاجِ الشّفّافِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يوحنّا 4:23; 1 Kings 8:27; 2 Chronicles 2:6"> ولم أجِد في المَدينةِ بَيتًا لِلعِبادةِ لأنّ اللهَ المَولى القَديرَ، وسَيِّدَنا عيسى الذِّبحَ العَظيمَ، هُما بِمَثابةِ بَيتٍ وقِبلةٍ للمؤمنينَ.</span>
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 60:19–20; الرؤيا 22:5; الرؤيا 21:11"> ولا حاجةَ في المَدينةِ إلى نُورِ الشَّمسِ أوِ القَمَرِ، فبَهاءُ اللهِ نورٌ لها، والذِّبحُ العَظيمُ مِصباحُها المُنيرُ.</span>
<span className="v" id="v24" data-v="24" data-xref="الرؤيا 21:26; الرؤيا 22:2; Zechariah 8:22–23"> وبنُورِ المَدينةِ تَهتَدي الأُمَمُ، ويَجيءُ مُلوكُ الأرضِ بِكُنُوزِهِم إليها فَرِحينَ.</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 60:11; Zechariah 14:7; الرؤيا 22:5"> وتَظَلُّ بَوّاباتُ المَدينةِ مَفتوحةً على مِصراعَيها طُول النَّهارِ ولن تُوصَد، ولن يَقتَرِب اللَّيلُ مِنها بَعدَ الآنَ.</span>
<span className="v" id="v26" data-v="26" data-xref="الرؤيا 21:24"> ويأتي إليها النّاسُ مِن كُلِّ الأُمَمِ مُحَمَّلينَ بِكُنوزِهِم ومُمتَلَكاتِهِم، فيُقَدِّمونَها إكرامًا للهِ.</span>
<span className="v" id="v27" data-v="27" data-xref="الرؤيا 22:14–15; إشَعْياء 52:1; الرؤيا 20:15"> ولن يَدخُل النّجِسونَ المَدينةَ، ولا عَبَدَةُ الأصنامِ، ولا الكاذِبونَ، إنّما يَدخُلُها الّذينَ دُوِّنَت أسماؤهُم في سِجِلِّ الخالِدينَ، سِجِلِّ سَيِّدِنا عيسى الذِّبحِ العَظيمِ.</span>
</div>
