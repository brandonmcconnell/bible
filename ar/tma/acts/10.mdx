<span className="v" id="v1" data-v="1" data-xref="سيرة الحواريّين 27:1; متّى 8:5–13; متّى 27:27"> وفي مَدينةِ قَيصريّة كانَ رَجُلٌ رُومانيٌّ اسمُهُ كورْنيليوسُ، وهو ضابِطٌ على الكَتيبةِ الإيطاليّة،<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="10‏.1 كانت مدينة القيصرية مركز قيادة جيش الاحتلال الروماني.">a</a></sup></span>
<span className="v" id="v2" data-v="2" data-xref="متّى 7:7–8; سيرة الحواريّين 10:4; سيرة الحواريّين 10:22"> وكانَ هو وأهلُ بيتِهِ وَرِعينَ يَخافونَ اللهَ، وهو كَثيرُ التَّصَدُّقِ على ذَوِي الحاجةِ مِن بَني يَعقوبَ وكَثيرُ الصَّلاةِ والعِبادَةِ للهِ.<sup className="fn"><a href="#fn-b" id="fnref-b" data-fn="10‏.2 لم يكن كورنيليوس يهوديًا، ولذلك لم يكن مختونًا، وكان الختان أحد الشروط لكي يصبح الرجل يهوديًا. أمّا كورنيليوس فكان تقيًا وآمن بما جاء في التوراة وغيرها من كتب الأنبياء. وكان هناك العديد من الرجال والنساء الذين آمنوا مثله بتعاليم اليهودية، ولكنّهم لم يستطيعوا قبول جميع التقاليد التي كانت تُطلب منهم ليصبحوا يهودًا.">b</a></sup></span>
<span className="v" id="v3" data-v="3" data-xref="دانيال 9:20–21; سيرة الحواريّين 3:1; سيرة الحواريّين 5:19"> وذاتَ يومٍ خِلالَ وَقتِ العَصرِ، تَجَلَّى لهُ مَلاكٌ في رُؤيا<sup className="fn"><a href="#fn-c" id="fnref-c" data-fn="10‏.3 إنّ الرؤيا تشبه الحلم، ومصدرها اللّه، وهي تحصل عندما يكون الشخص يقظًا أو نائمًا على حدٍّ سواء.">c</a></sup>وقالَ لهُ: "كُورنيليوسُ".</span>
<span className="v" id="v4" data-v="4" data-xref="فيليبّي 4:6; العبرانيين 6:10; سيرة الحواريّين 10:31"> فحَدَّقَ إليهِ هذا الأخيرُ خائفًا وأجابَهُ: "ما الأمرُ يا سَيِّدي؟" فقالَ لهُ المَلاكُ: "قد تَقَبَّلَ اللهُ مِنكَ صَلواتِكَ وصَدَقاتِكَ، وهو رَاضٍ عَنكَ.</span>
<span className="v" id="v5" data-v="5" data-xref="سيرة الحواريّين 16:9; سيرة الحواريّين 9:38; سيرة الحواريّين 15:7"> فَقُمْ وأرسِلْ بَعضَ الرِّجالِ إلى مَدينةِ يافا، واستَدعِ سَمعانَ، المُلَقَّبَ ببُطرُسَ الصَّخر،</span>
<span className="v" id="v6" data-v="6" data-xref="سيرة الحواريّين 9:43; سيرة الحواريّين 11:13–14; روما 10:14–17"> فقد نَزَلَ ضَيفًا عِندَ سَمعانَ الدَّبّاغ في دارِهِ على شاطِئ البَحرِ".</span>
<span className="v" id="v7" data-v="7" data-xref="فِليمون 1:16; صَمُوئيلَ الأوَّلُ 14:6–7; متّى 8:9–10"> وعِندَما انصَرَفَ المَلاكُ، دَعا كُورنيليوسُ اثنينِ مِن خَدَمِهِ وجُنديًّا مِن مُساعِدِيهِ كانَ على جَانبٍ كَبيرٍ مِنَ التَّقوى،</span>
<span className="v" id="v8" data-v="8" data-xref="سيرة الحواريّين 9:36; مقتطفات من كتاب الزبور 119:59–60; سيرة الحواريّين 10:33"> وحَدَّثَهُم بِكُلِّ ما رَأى، ثُمَّ أرسَلَهُم إلى يافا.<sup className="fn"><a href="#fn-d" id="fnref-d" data-fn="10‏.8 تبعد مدينة يافا عن مدينة قيصرية بحوالي 48 كيلومتر باتّجاه الجنوب.">d</a></sup></span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="Zephaniah 1:5; مقتطفات من كتاب الزبور 55:17; صَمُوئيلَ الأوَّلُ 9:25"> وفي اليَومِ التّالي وبَينَما كانوا على مَقرُبةٍ مِنَ المَدينةِ، صَعِدَ بُطرسُ إلى السَّطحِ ليُصَلِّي خِلالَ وَقتِ الظُّهرِ،<sup className="fn"><a href="#fn-e" id="fnref-e" data-fn="10‏.9 في ذلك الوقت، كانت صلوات اليهود تقام في الصباح وفي العصر، وكانت هناك صلاة إضافية تقام في الظهر أو في المغرب.">e</a></sup></span>
<span className="v" id="v10" data-v="10" data-xref="سيرة الحواريّين 22:17; الرؤيا 4:2–3; كورنتوس الثانية 12:2–4"> فأخَذَ مِنهُ الجُوعُ كُلَّ مَأخذٍ، وبَينَما كانوا يُجَهِّزونَ الطَّعامَ، رُفِعَ عَنهُ الحِجابُ.<sup className="fn"><a href="#fn-f" id="fnref-f" data-fn="10‏.10 رفع الحجاب هو انكشاف عالم الغيب للناظر ولا يكون ذلك إلا للرسل والأنبياء والأولياء.">f</a></sup></span>
<span className="v" id="v11" data-v="11" data-xref="أفاسوس 3:6; يوحنّا 1:51; روما 16:25–26"> فرَأى السَّماءَ وقدِ انشَقَّت، ومُلاءَةً كَبيرةً مَعقودَةً مِن أطرافِها الأربَعةِ مُدَلاّةً إلى الأرضِ،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 7:8–9; يوحنّا 7:37; كورنتوس الأولى 6:9–11"> مَملوءَةً بجَميعِ أنواعِ الحَيواناتِ مِن دَوابّ وزَواحِف وطُيور،</span>
<span className="v" id="v13" data-v="13" data-xref="يوحنّا 4:31–34; إرْمِيا 35:2–5; سيرة الحواريّين 10:10"> وسَمِعَ هاتِفًا يَقولُ: "قُم يا بُطرُسُ اِذبَحْ وكُلْ".</span>
<span className="v" id="v14" data-v="14" data-xref="Leviticus 20:25; Ezekiel 4:14; دانيال 1:8"> فأجابَ بُطرسُ: "لا يا سَيِّدي! إنّ مِن بَينَ تِلكَ الحَيواناتِ ما هو مُحَرَّمٌ أكلُهُ! وأنا لم آكُل في حَياتي حَرامًا بل كانَ طَعامي دائِمًا حَلالاً".</span>
<span className="v" id="v15" data-v="15" data-xref="كورنتوس الأولى 10:25; متّى 15:11; تيتوس 1:15"> فأعادَ الهاتِفُ عليهِ القَولَ ثانيَةً: "ما طَهَّرَهُ اللهُ، لاَ تُنَجِّسْهُ أنتَ".</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 41:32; يوحنّا 21:17; كورنتوس الثانية 13:1"> وتَكَرَّرَ ذلِكَ ثَلاثَ مَرّاتٍ، ثُمَّ ارتَفَعَتِ المُلاءةُ إلى السَّماءِ، وأفاقَ بُطرسُ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="يوحنّا 13:12; سيرة الحواريّين 10:7–19; سيرة الحواريّين 5:24"> وبَينَما كانَ غارِقًا في حَيرتِهِ يَتَساءَلُ عَن مَعنى ما رَأى، كانَ الرِّجالُ الّذينَ جَاؤوا مِن عِندَ كورْنيليوسَ عِندَ البابِ يَسألونَ عن دارِ سَمعانَ الدّبَّاغ.</span>
<span className="v" id="v18" data-v="18" data-xref="سيرة الحواريّين 10:5–6; سيرة الحواريّين 11:11"> وعِندَما وَصَلوا أخَذوا يُنادونَ سائِلينَ: "هل حَلَّ بَينَكُم ضَيفٌ اسمُهُ بُطرُسُ؟"</span>
<span className="v" id="v19" data-v="19" data-xref="سيرة الحواريّين 8:29; سيرة الحواريّين 11:12; سيرة الحواريّين 21:4"> وبَينَما كانَ بُطرسُ غارِقًا في أفكارِهِ يُحاوِلُ تَفسيرَ مَعنى ما جَاءَتهُ بِهِ الرُّؤيا، جاءَهُ وَحيٌ مِن رُوحِ اللهِ يُخبِرُهُ: "جاءَ ثَلاثةُ رِجَالٍ في طَلَبِكَ،</span>
<span className="v" id="v20" data-v="20" data-xref="سيرة الحواريّين 15:7–9; Zechariah 2:9–11; إشَعْياء 48:16"> فقُمْ وقابِلْهُم واِذهبْ مَعَهُم بِلا تَرَدُّدٍ فأنا أرسَلتُهُم إليكَ".</span>
<span className="v" id="v21" data-v="21" data-xref="يوحنّا 1:38–39; مرقس 10:51; سيرة الحواريّين 10:29"> فاستَجابَ بُطرُسُ ونَزَلَ إلى الرِّجالِ وقالَ لهُم: "أنا مَن جِئتُم تَطلُبونَ، فما سَبَبُ مَجيئِكُم؟"</span>
<span className="v" id="v22" data-v="22" data-xref="سيرة الحواريّين 11:14; سيرة الحواريّين 10:2; سيرة الحواريّين 22:12"> فأجَابَهُ الرِّجالُ: "قَدِمنا بِطَلَبٍ مِن كُورنيليوسَ الضّابطِ الرُّومانيّ. وهو الرَّجُلُ الصّالحُ التَّقيُّ، الّذي يَحتَرِمُهُ بَنو يَعقوبَ، وقد جاءَهُ مَلاكٌ طَلَبَ مِنهُ استِدعاءَكَ لسَماعِ ما عِندَكَ مِن كَلامٍ".<sup className="fn"><a href="#fn-g" id="fnref-g" data-fn="10‏.22 لم يستطع بطرس قبول دعوة كورنيليوس للمكوث عنده بسهولة لأنّ اليهود كانوا يعتبرون أنّ دخول بيت إنسانٍ غير يهودي ينجّسهم. وقد عرف بطرس أنّ الذين آمنوا بالسيد المسيح من اليهود كانوا لا يزالون متشدّدين فيما يخصّ التقاليد اليهودية، وأنّهم كانوا سينتقدونه لدخوله بيت أجنبي.">g</a></sup></span>
<span className="v" id="v23" data-v="23" data-xref="سيرة الحواريّين 10:45; سيرة الحواريّين 11:12; صخر الأولى 4:9"> فاستَجابَ بُطرسُ الصَّخرُ لِطَلَبِهِم واستَضافَهُم في تِلكَ اللَّيلةِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="سيرة الحواريّين 8:40; Zechariah 8:20–23; Zechariah 3:10"> وفي اليَومِ التّالي رَحَلَ مَعَهُم، ومَعَهُ بَعضُ المُؤمِنينَ مِن يافا.</span>
<span className="v" id="v25" data-v="25" data-xref="الرؤيا 19:10; الرؤيا 22:8–9; دانيال 2:46"> وعِندَ وصُولِهِم إلى مَدينةِ قَيصرِيّةٍ بَعدَ يومٍ مِن السَّفَرِ، كانَ كُورنيليُوسُ في انتِظارِهِم وقد دَعا بَعضًا مِن أقاربِهِ وأصدقائِهِ المُقَرَّبينَ لكي يَستَمِعوا إلى بُطرُسَ. ولمّا وَصَلَ بُطرسُ ومَن مَعَهُ إلى الدّارِ، خَرَجَ كُورنيليوسُ لاستِقبالِهِ، ورَمى بِنَفسِهِ عِندَ قدَمَيهِ، ساجِدًا أمامَهُ،</span>
<span className="v" id="v26" data-v="26" data-xref="الرؤيا 19:10; الرؤيا 22:8–9; سيرة الحواريّين 14:14–15"> فأنهَضَهُ بسُرعةٍ قائِلاً: "قُم حالاً! إنّما أنا مُجَرَّدُ إنسانٍ مِثلُكَ!"</span>
<span className="v" id="v27" data-v="27" data-xref="كورنتوس الثانية 2:12; يوحنّا 4:35; سيرة الحواريّين 10:24"> ودَخَلَ وإيّاهُ وتَبادَلا الحَديثَ، فوَجَدَ بُطرسُ حَشدًا مِنَ النّاسِ،<sup className="fn"><a href="#fn-h" id="fnref-h" data-fn="10‏.27 لم يمنع التقليد اليهودي دخول بيوت غير اليهود فحسب، بل منع أيضًا تناول الطعام والشراب معهم. يعني هذا أنّ اليهود المتشدّدين كانوا يمنعون اليهود من تناول الطعام الذي جهّزه غير اليهود، ولذلك كان الناس من بقيّة الأمم يشعرون أنّ اليهود غير اجتماعيين. ولكنّ الله أوحى إلى بطرس بضرورة تغيير هذه العادات بين أتباع السيد المسيح.">h</a></sup></span>
<span className="v" id="v28" data-v="28" data-xref="يوحنّا 4:9; أفاسوس 3:6–7; غلاطية 2:12–14"> فخَاطَبَهُم قائِلاً: "أنتُم تَعلَمونَ جَميعًا تَقاليدَنا الّتي تَمنَعُ اليَهوديَّ مِنَ التَّعامُلِ مَعَ غَيرِ اليَهوديِّ ومِن زِيارتِهِ، إلاّ أنّ اللهَ تَعالى بَيَّنَ لي أنّ الإنسانَ مِن أيِّ شَعبٍ كانَ لَيسَ نَجِسًا ولا مَرفوضًا،</span>
<span className="v" id="v29" data-v="29" data-xref="مقتطفات من كتاب الزبور 119:60; صخر الأولى 3:15; سيرة الحواريّين 10:19–20"> لِذلِكَ جِئتُكُم دونَ اعتِراضٍ عِندَما أرسَلتُم في طَلَبي. والآنَ أسألُكم لِمَ أرسَلتُم في طَلَبي؟"</span>
<span className="v" id="v30" data-v="30" data-xref="سيرة الحواريّين 10:3; دانيال 9:20–21; Nehemiah 9:1–3"> فأجابَهُ كُورنيليوسُ بقَولِهِ: "بَينَما كُنتُ، مُنذُ أربَعةِ أيّامٍ، أُصَلّي عَصرًا في داري، ظَهَرَ أمامي فَجأةً رَجُلٌ عليهِ ثيابٌ بَرّاقَةٌ وأخبَرَني:</span>
<span className="v" id="v31" data-v="31" data-xref="سيرة الحواريّين 10:4; دانيال 9:23; لوقا 1:13"> "أيا كُورنيليُوسُ، قد تَقَبَّلَ اللهُ صَلاتَكَ وصَدَقاتِكَ ورَضِيَ عَنكَ،</span>
<span className="v" id="v32" data-v="32"> فأرسِلْ برِجالٍ إلى مَدينةِ يافا لاستِدعاءِ رَجُلٍ اسمُهُ بُطرسُ، إنّهُ يَنزِلُ ضَيفًا في دارِ سَمعانَ الدَّبّاغ على شاطئ البَحرِ".</span>
<span className="v" id="v33" data-v="33" data-xref="يعقوب 1:19; غلاطية 4:14; 2 Chronicles 30:12"> وهكذا فَعَلتُ، فأرسَلتُ أستَدعِيكَ فَورًا، ولقَد فَعَلتَ حَسَنًا بأنِ استَجَبتَ، وها نَحنُ الآنَ مُجتَمِعونَ في حَضرةِ اللهِ لِنَستمِعَ إلى كُلِّ ما أمَرَكَ اللهُ بِهِ".</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="روما 2:11; 2 Chronicles 19:7; Deuteronomy 10:17"> وجَلَسَ بُطرُسُ الصَّخرُ ليُحَدِّثَهُم فقالَ: "لقد تَيَّقَنتُ الآن أنّ النّاسَ جَميعًا سَواسيَةٌ عِندَ اللهِ،</span>
<span className="v" id="v35" data-v="35" data-xref="يوحنّا الأولى 2:29; سيرة الحواريّين 10:2; روما 10:12–13"> وأقرَبُهُم إليهِ أتقاهُم وأصلَحُهُم للنَّاسِ، مَهما كانَ جِنسُهُ.</span>
<span className="v" id="v36" data-v="36" data-xref="متّى 28:18; سيرة الحواريّين 2:36; الرؤيا 17:14"> لقد سَمِعتُم بِلا شَكٍّ عنِ البِشارَةِ الّتي بَعَثَها اللهُ إلى بَني يَعقوبَ، الّتي تُعلِنُ بأنّ التَّقَرُّبَ إلى اللهِ تَعالى لا يَكونُ إلاّ مِن خِلالِ عيسى المَسيحِ سَيِّدِ جَميعِ البَشَرِ.</span>

<span className="v" id="v38" data-v="38" data-xref="لوقا 4:18; سيرة الحواريّين 2:22; إشَعْياء 61:1–3"> وعَلِمتُم أنّ اللهَ تَعالى اختارَ، في الوَقتِ عَينِهِ، عيسى النّاصِريَّ فوَهَبَهُ قوّتَهُ ورُوحَهُ القُدّوسَ، وكانَ مَعَهُ أينَما حَلَّ، لقد كانَ (سلامُهُ علينا) يَجوبُ كُلَّ الأماكنِ لِيَعمَلَ الخَيرَ ولِيُبرئَ كُلَّ مَن كانَ تَحتَ سَطوةِ الشَّيطانِ.</span>
