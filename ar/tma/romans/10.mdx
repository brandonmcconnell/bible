<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="كورنتوس الأولى 9:20–22; روما 9:1–3; يوحنّا 5:34"> إخوتي، إنّي في شَوقٍ عَظيمٍ، وأتَوَسَّلُ إلى اللهِ أن يَنجوَ شَعبُ بَني يَعقوبَ،</span>
<span className="v" id="v2" data-v="2" data-xref="سيرة الحواريّين 21:20; روما 9:31–32; الأمثال 19:2"> وأشهَدُ أنّهُم مُتَحَمِّسونَ للهِ، ولكنّهُم لا يَفهَمونَ.</span>
<span className="v" id="v3" data-v="3" data-xref="فيليبّي 3:9; لوقا 16:15; كورنتوس الثانية 5:21"> ولأنّهُم لا يَفهَمونَ كَيفَ يَرضى اللهُ عن النّاسِ بفَضلِ السَّيّدِ المَسِيحِ، فقد حاولوا أن يُحافِظوا على طَريقةٍ تَخُصُّهُم فقط، فرَفَضوا طَريقَ اللهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="غلاطية 3:24; روما 8:3–4; متّى 5:17–18"> وإنّ الغايةَ القُصوى للشَّريعةِ هي الإيمانُ بالمَسيحِ المُنتَظَرِ، وكُلُّ مَن يَتَمَسَّكُ بِهِ يَنالُ مَرضاةَ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="Leviticus 18:5; Ezekiel 20:11; Nehemiah 9:29"> ولقد قيلَ إنّ كَلامَ النَّبيِّ موسى يَحمِلُ دَليلاً على أنّ اللهَ لا يَرضى عن النّاسِ إلاّ بانتِمائِهِم للشَّرعِ اليَهوديِّ حينَ قالَ: "مَن يَتَمَسَّكَ بكُلِّ هذِهِ الوصايا يَحيا حَياةً رَضيّةً"<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="10‏.5 التوراة، سفر اللاويين 18: 5.">a</a></sup></span>
<span className="v" id="v6" data-v="6" data-xref="Deuteronomy 30:11–14; الأمثال 30:4; أفاسوس 4:8–10"> ولكنّنا نَحن نَعلَمُ أنّ اللهَ يَرضى عنّا بالإيمانِ، وهذا الرِّضا لَيسَ مُستَحيلاً، وهو ما جاءَ في التّوراةِ:<sup className="fn"><a href="#fn-b" id="fnref-b" data-fn="10‏.6 يلمّح بولس في الآيتين 6 و7 إلى التوراة، سفر التثنية 30: 12‏-13، ويبيّن أنّهما تتطابقان مع ما جاء في رسالة السيّد المسيح. وربّما لمّح بولس أيضا إلى سفر التثنية 8: 17 و9: 4 وفي سياقها يكذّب الله على لسان النبي موسى (عليه السّلام) الفكرة التي تعتبر أن صلاح بني يعقوب هو السبب في تفوُّقهم العسكري وازدهارهم.">b</a></sup>"لَستَ بحاجةٍ للصُّعودِ إلى السَّماءِ"، وكأنّكَ تَحتاجُ أن تَعودَ بالسَّيّدِ المَسيحِ مِنها لتَنال مَرضاةَ اللهِ.</span>
<span className="v" id="v7" data-v="7" data-xref="العبرانيين 13:20; الرؤيا 1:18; روما 4:25"> ولا أنتَ بحاجةٍ "للنُّزولِ إلى أعماقِ الأرضِ" وكأَنّك تَحتاجُ أن تَجعَلَ السَّيّدَ المَسيحَ يَقومُ حَيًّا مِن بَينِ الأمواتِ لتَنالَ مَرضاةَ اللهِ أيضًا.</span>
<span className="v" id="v8" data-v="8" data-xref="Deuteronomy 30:14; روما 1:16–17; تيموتاوي الأولى 4:6"> إنّما النَّجاةُ في مُتناولِكَ، ألا وهي التَّوَكُّلُ على سَيِّدِنا المَسيحِ، وهذا مَضمونُ دَعوتِنا بَينَ النّاسِ، كَما وَرَدَ في التَّوراةِ: "ما أقرَبَ بَلاغَ اللهِ مِنكَ، ومِلء قَلبِكَ تَشهَدُ بِهِ شَفتيكَ".<sup className="fn"><a href="#fn-c" id="fnref-c" data-fn="10‏.8 يقتبس بولس هنا من التوراة، سفر التثنية 30: 14.">c</a></sup></span>
<span className="v" id="v9" data-v="9" data-xref="لوقا 12:8; سيرة الحواريّين 16:31; فيليبّي 2:11"> كذلِكَ إن كُنتَ شاهِدًا بلِسانِكَ أنّ عيسى سَيِّدُ العالَمِينَ، وإن كُنتَ مُتَيَقِّنًا مِن كُلِّ قَلبِكَ أنّ اللهَ أحياهُ مِنَ المَوتِ، فإنّكَ مِن النّاجينَ.</span>
<span className="v" id="v10" data-v="10" data-xref="يوحنّا الأولى 4:15; روما 10:9; غلاطية 2:16"> فإن وَثِقَ شَخصٌ مِن قَلبِهِ بالسَّيّدِ المَسيحِ، وشَهِدَ بلِسانِهِ أنّهُ مَولاهُ، رَضي اللهُ عَنهُ وأنجاهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 28:16; روما 9:33; صخر الأولى 2:6"> كَما جاءَ في كِتابِ النَّبيِّ أشعيا: "ولا يَخيبُ كُلُّ مَن وَثِقَ بِهِ".<sup className="fn"><a href="#fn-d" id="fnref-d" data-fn="10‏.11 كتاب النبي أشعيا 28: 16.">d</a></sup></span>
<span className="v" id="v12" data-v="12" data-xref="فيليبّي 4:19; الرؤيا 17:14; غلاطية 3:28"> فلا فَرقَ بَينَ يَهوديٍّ وغَيرِ يَهوديٍّ، لأنّ مَولانا سَيِّدُ كُلِّ النّاسِ، يَفيضُ بنَعمتِهِ على كُلِّ مَن يَستَغيثُ بِهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="سيرة الحواريّين 2:21; Joel 2:32"> وهو ما وَرَدَ في كِتابِ النَّبيِّ يوئيل: "كُلُّ مَن يَستَغيثُ بمَولانا يُصبِحُ مِنَ النّاجينَ".<sup className="fn"><a href="#fn-e" id="fnref-e" data-fn="10‏.13 كتاب النبي يوئيل 2: 32.">e</a></sup></span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="مرقس 16:15–16; سيرة الحواريّين 26:17–18; تيتوس 1:3"> أيُعقَلُ أن يَطلُبَ النّاسُ عَونَ السَّيّدِ المَسيحِ، وهُم لا يؤمنونَ بِهِ؟ وكَيفَ يؤمنونَ بِهِ ولم يَسمَعوا عَنهُ؟ وكَيفَ يَسمَعونَ ولم يُخبِرهُم أحَدٌ؟</span>
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 52:7; Nahum 1:15; أفاسوس 2:17"> وهل يَحمِلُ البَلاغَ غَيرُ مَن أرسَلَهُم اللهُ مِنَ الحَوارِيِّين؟ نعم، إنّ اللهَ أرسَلَ إلى بَني يَعقوبَ هؤلاءِ المُخلِصينَ، كَما جاءَ في كِتابِ النَّبيِّ أَشعيا: "ما أحلى خُطُواتِ المُرسَلينَ الّذينَ يُبَشِّرونَ بالخَيرِ".<sup className="fn"><a href="#fn-f" id="fnref-f" data-fn="10‏.15 كتاب النبي أشعيا 52: 7.">f</a></sup></span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 53:1; روما 3:3; يوحنّا 12:38–40"> ولكنّ رِسالةَ سَيِّدِنا المَسيحِ لم يَقبَل بها إلاّ قَليلٌ مِن بَني يَعقوبَ، إذ يَقولُ النَّبيُّ أَشعيا: "يا مَولاي، ما أقَلَّ المؤمنينَ بِرِسالَتِنا!"<sup className="fn"><a href="#fn-g" id="fnref-g" data-fn="10‏.16 كتاب النبي أشعيا 53: 1.">g</a></sup></span>
<span className="v" id="v17" data-v="17" data-xref="غلاطية 3:2; غلاطية 3:5; تسالونكي الأولى 2:13"> فلا يَقدِرُ أحَدٌ على الإيمانِ إلاّ إذا سَمِعَ الرّسالةَ، أي رِسالةَ السَّيِّدِ المَسيحِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="مقتطفات من كتاب الزبور 19:4; روما 1:8; متّى 24:14"> ولكنّني أقولُ: ألَم يَسمَع بَنو يَعقوبَ هذِهِ الرِّسالةَ؟ أجل، إنّهُم على عِلمٍ بِها، فقد جاءَ في المَزاميرِ: "سَمِعَتِ الأرضُ كُلُّها صَوتَ المُرسَلينَ، وبَلَغَ كَلامُهم الأرجاءَ كُلّها".<sup className="fn"><a href="#fn-h" id="fnref-h" data-fn="10‏.18 كتاب الزبور، مزمور 19: 4.">h</a></sup></span>
<span className="v" id="v19" data-v="19" data-xref="Deuteronomy 32:21; روما 11:11; روما 11:14"> وأعودُ فأقولُ: "ألَم يَكُن بَنو يَعقوبَ يُدرِكونَ أنّ هذِهِ الرِّسالةَ مُوَجّهةٌ إلى كُلِّ الأُمَمِ؟" نعم، كانَ يَجِبُ أن يُدرِكوا ذلِكَ. فقد قالَ اللهُ في زَمَنِ النَّبيِّ موسَى: "لأُبارِكَنَّ أُمَّةً مِن غَيرِ بَني يَعقوبَ وأَجعَلُكُم تَغارونَ مِنهُم، وأَفيضُ بِرَحمَتي على أُمَّةٍ بِلا عِلمٍ فَتَغتاظونَ مِنهُم".<sup className="fn"><a href="#fn-i" id="fnref-i" data-fn="10‏.19 يقتبس بولس هنا من التوراة، سفر التثنية 32: 21. كان بنو يعقوب يعتبرون الأقوام الآخرين جهلة دينيا، غير أن النبي موسى (عليه السّلام) تنبأ أن من بين هؤلاء الأقوام من سيفهم الرسالة قبل أن يفهمها بنو يعقوب.">i</a></sup></span>
<span className="v" id="v20" data-v="20" data-xref="إشَعْياء 65:1–2; روما 9:30; إشَعْياء 55:4–5"> وكانَ النّبيُّ أَشعيا جَريئًا حينَ أعلَنَ قَولَ اللهِ: "ها قد وَجَدَني الّذينَ لم يَبحَثوا عنّي، وتَجَلَّيتُ للّذينَ لم يَطلُبوني".<sup className="fn"><a href="#fn-j" id="fnref-j" data-fn="10‏.20 كتاب النبي أشعيا 65: 1.">j</a></sup></span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 65:2–5; صخر الأولى 2:8; إرْمِيا 35:15"> ثُمّ اسمَعوا قَولَهُ تَعالى عن بَني يَعقوبَ: "بَسَطتُ يَديّ بالرَّحمةِ يَومًا بَعدَ يومٍ لشَعبٍ مُتَمَرِّدٍ عَنيدٍ".<sup className="fn"><a href="#fn-k" id="fnref-k" data-fn="10‏.21 أشعيا 65: 2.">k</a></sup></span>
</div>
