<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مرقس 12:41–44; يوحنّا 8:20; متّى 27:6"> وبَينَما كانَ سَيِّدُنا عيسى (سلامُهُ علينا) في الحَرَمِ الشَّريفِ، أخَذَ يُراقِبُ الأغنياءَ الّذينَ جاؤوا بالصَّدَقاتِ إلى صُندوقِ التَّبَرُّعاتِ،</span>
<span className="v" id="v2" data-v="2" data-xref="مرقس 12:42"> فأتَتْ أرمَلةٌ فَقيرةٌ وجادَت بفِلسَينِ.</span>
<span className="v" id="v3" data-v="3" data-xref="كورنتوس الثانية 8:12; سيرة الحواريّين 10:34; كورنتوس الثانية 8:2–3"> فقالَ (سلامُهُ علينا) مُخاطِبًا الحاضِرينَ: "لقد تَصَدَّقَت هذِهِ الأرمَلةُ الفَقيرةُ بأضعافِ ما تَصَدَّقَ بِهِ هؤلاءِ جَميعًا،</span>
<span className="v" id="v4" data-v="4" data-xref="سيرة الحواريّين 2:44–45; لوقا 15:12; سيرة الحواريّين 4:34"> فقد أعطى هؤلاءِ جُزءًا يَسيرًا مِن فَيضِ مالِهِم، أمّا هي فقد أَعطَت كُلَّ ما لَدَيها على رَغمِ فَقرِها وهو ما كانَ يَجِبُ أن تُنفِقَهُ على نَفسِها لتَعيشَ!"</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="يوحنّا 2:20; متّى 24:1–51; مرقس 13:1–37"> وأخَذَ بَعضُ أتباعِهِ (سلامُهُ علينا) بَعدَ ذلِكَ يَتَحَدِّثونَ عن حَرَمِ بَيتِ اللهِ، وعن رَوعةِ ما فيهِ مِن أحجارٍ جَميلةٍ تُزَيِّنُهُ ومِن تُحَفٍ مَنذورةٍ للهِ.</span>
<span className="v" id="v6" data-v="6" data-xref="متّى 24:2; مرقس 13:2; Lamentations 5:18"> فالتَفَتَ إليهِم عيسى (سلامُهُ علينا) قائلاً: "إنّ كُلَّ ما تَرونَهُ إلى زَوالٍ، فسيأتي زَمَنٌ يُهَدَّمُ فيهِ الحَرَمُ بما فيهِ، ولن يَبقَى فيهِ حَجَرٌ على حَجَرٍ".</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="سيرة الحواريّين 1:6–7; مرقس 13:14; مرقس 13:3–4"> فسألوهُ: "أيُّها المُعَلِّمُ، مَتى يَكونُ زَمَنُ ذلِكَ الخَرابِ، وما هي العَلاماتُ الّتي تَسبِقُهُ؟"</span>
<span className="v" id="v8" data-v="8" data-xref="يوحنّا الأولى 4:1; مرقس 13:5–6; أفاسوس 5:6"> فأجابَهُم قائلاً: "احذَروا أن يُضِلَّكُم أحَدٌ عن السَّبيلِ! فسيأتي الكَثيرونَ مِمَّن يَنتَحِلونَ لَقَبي، فيَدَّعي أحَدُهُم قائلاً: "أنا حَبيبُ اللهِ!" و"قد اقتَرَبَت السّاعةُ!" فلا تُصَدِّقوا هؤلاءِ مُطلَقًا.</span>
<span className="v" id="v9" data-v="9" data-xref="لوقا 21:28; متّى 24:6–8; مرقس 13:7–8"> وإنّكُم ستَسمَعونَ عن حُروبٍ وفِتَنٍ تَقومُ في الأرضِ، فلا تَخافوا لأنّهُ لا بُدَّ مِن ظُهورِ هذِهِ العَلاماتِ، ولئن ظَهَرَت فإنّ السّاعةَ لن تَقومَ مِن فَورِها.</span>
<span className="v" id="v10" data-v="10" data-xref="سيرة الحواريّين 2:19–20; مرقس 13:8; 2 Chronicles 15:5–6"> ألا وإنّ حُروبًا بَينَ الدُّوَلِ والمَمالكِ ستَقومُ،</span>
<span className="v" id="v11" data-v="11" data-xref="متّى 24:29–30; لوقا 21:25–27"> وستَحُلُّ الزَّلازِلُ والمَجاعاتُ والأوبئةُ في أنحاءٍ كَثيرةٍ مِن الأرضِ، وستُبدي السَّماءُ آياتِها الرَّهيبةَ الكُبرى!</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="صخر الأولى 4:12–14; سيرة الحواريّين 5:40; سيرة الحواريّين 22:30"> ولكنْ قَبلَ هذِهِ الأُمورِ كُلِّها، سيَحُلُّ عليكُم ظُلمٌ واضطِهادٌ، وسيَسوقُكُم النّاسُ إلى المَحاكمِ الدِّينيّةِ والسُّجونِ، وسيَتَّهِمونَكُم أمامَ المُلوكِ والوُلاةِ بأنّكُم أنصاري!</span>
<span className="v" id="v13" data-v="13" data-xref="فيليبّي 1:12; فيليبّي 1:28; تسالونكي الثانية 1:5"> وستَكونُ تِلكَ فُرصتَكُم لإعلانِ شَهادَتِكُم فيَّ على المَلأِ،</span>
<span className="v" id="v14" data-v="14" data-xref="لوقا 12:11–12; مرقس 13:11; متّى 10:19–20"> ولكنْ لا تُرهِقوا أنفُسَكُم بإعدادِ الدِّفاعِ عن أنفُسِكُم،</span>
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 1:9; أفاسوس 6:19; لوقا 12:12"> لأنّي سأمنَحُكُم لُغةَ البَيانِ والحِكمةِ الّتي يَقِفُ خُصومُكُم عاجِزينَ أمامَها".</span>
<span className="v" id="v16" data-v="16" data-xref="لوقا 12:53; الرؤيا 6:9; Micah 7:5–6"> وسيَخونُكُم آباؤكُم وإخوانُكُم وأقرباؤكُم وأصدقاؤكُم، بل إنّ بَعضَكُم سيُقتَلونَ أيضًا.</span>
<span className="v" id="v17" data-v="17" data-xref="يوحنّا 15:19; لوقا 6:22; يوحنّا 15:21"> وستَكونونَ مَوضِعَ بُغضِ جَميعِ النّاسِ لإخلاصِكُم لي.</span>
<span className="v" id="v18" data-v="18" data-xref="متّى 10:30; صَمُوئيلَ الأوَّلُ 14:45; لوقا 12:7"> ولكن لا خَوفَ عليكُم،</span>
<span className="v" id="v19" data-v="19" data-xref="العبرانيين 10:36; متّى 10:22; يعقوب 1:3"> لأنّكُم بثباتِكُم عِندَ تِلكَ المُلِمّاتِ ستَكسِبونَ أنفُسَكُم.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="لوقا 19:43; دانيال 9:27; مرقس 13:14"> ثُمّ تابَعَ (سلامُهُ علينا) يَقولُ: "وعِندَما تَرونَ مَدينةَ القُدسِ وقد أوشَكَت أن تُحيطَ بها الجُيوشُ، فتأكّدوا أنّ مَوعِدَ خَرابِها قد حانَ.</span>
<span className="v" id="v21" data-v="21" data-xref="إرْمِيا 6:1; التَّكوين 19:17; لوقا 17:31–33"> حينئذٍ ليَهرُبْ مَن كانَ في مِنطقةِ يَهوذا إلى الجِبالِ، وليَهجُر القُدسَ مَن كانَ فيها. وأمّا مَن كانَ خارِجَها فلا يَدخُلْها،</span>
<span className="v" id="v22" data-v="22" data-xref="Hosea 9:7; إشَعْياء 65:12–16; إشَعْياء 63:4"> ففي تِلكَ الفَترةِ ستَنالُ القُدسُ عِقابَها مِن اللهِ. ويَتَحَقّقُ الوَعدُ الحقُّ كَما جاءَ في الكُتُبِ المُقَدَّسةِ.</span>
<span className="v" id="v23" data-v="23" data-xref="مرقس 13:17; متّى 24:19; لوقا 23:29"> ووَيلٌ حينئذٍ للحَواملِ والمُرضِعاتِ مِمّا سيَحُلُّ في هذِهِ الأرضِ مِن اضطِراباتٍ وغَضَبٍ مِن اللهِ على هذا الشَّعبِ</span>
<span className="v" id="v24" data-v="24" data-xref="روما 11:25; الرؤيا 11:2; إشَعْياء 63:18"> الّذي سيُقتَلُ بَعضٌ مِنهُ بحَدِّ السَّيفِ، وسيُساقُ الباقونَ سَبايا إلى كُلِّ الأُمَمِ، وسيَحتَلُّ الأغرابُ المَدينةَ المُقَدَّسةَ إلى أجلٍ مُسَمًّى.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 13:10; Joel 2:30–31; الرؤيا 6:12–14"> وتابَعَ (سلامُهُ علينا) قائلاً: "وستَبدو بَعدَ ذلِكَ في السَّماءِ عَلاماتٌ عَجيبةٌ في الشَّمسِ والقَمَرِ والنُّجومِ، وسيُصيبُ أُمَمَ الأرضِ اضطِرابٌ ورُعبٌ بسَبَبِ هَيَجانِ البِحارِ وثَورةِ أمواجِها،</span>
<span className="v" id="v26" data-v="26" data-xref="متّى 24:29; Deuteronomy 28:65–67; صخر الثانية 3:10–12"> وسيُغمَى على النّاسِ لِهَولِ ما يَرَونَهُ مِن دَمارٍ زاحِفٍ على العالَمِ، إذ تَضطَرِبُ حَرَكةُ الأجرامِ السَّماويّةِ فتَفقِدَ تَوازنَها،<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="21‏.26 كان الاعتقاد سائدا في القديم أنّ النجوم هي قوى روحانية، بينما كان البعض يجعلون النجوم نظير الشعوب، لذا فإنّ ما ذكره سيّدنا المسيح قد يعني بأنّ الله سيزلزل القوى الأرضية (الشعوب) أو القوى الروحانية التي تعارض الله (الجن والشياطين).">a</a></sup></span>
<span className="v" id="v27" data-v="27" data-xref="متّى 24:30; الرؤيا 1:7; مرقس 13:26"> عِندئذٍ سيَرَى النّاسُ سَيِّدَ البَشَرِ قادِمًا في ظُلَلٍ مِنَ الغَمامِ بكُلِّ عِزّةٍ وهَيبةٍ،</span>
<span className="v" id="v28" data-v="28" data-xref="روما 8:23; إشَعْياء 60:1–2; أفاسوس 4:30"> فحينَ تَرَونَ حُدوثَ ذلِكَ كُلِّهِ، قِفوا وارفَعوا رُؤوسَكُم بافتِخارٍ لأنّ نَجاتَكُم صارَت قَريبةً!"</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="متّى 24:32–35; مرقس 13:28–30"> وضَرَبَ لهُم (سلامُهُ علينا) مَثَلاً قائلاً: "عِندَما تَنظُرونَ إلى التِّينةِ أو غَيرِها مِن الأشجارِ،</span>
<span className="v" id="v30" data-v="30" data-xref="لوقا 12:57"> وتَرَونَ اخضِرارَ أوراقِها، تَعلَمونَ أنّ الصَّيفَ قد اقتَرَبَ.</span>
<span className="v" id="v31" data-v="31" data-xref="يعقوب 5:9; العبرانيين 10:37; صخر الأولى 4:7"> وكذلِكَ عِندَما تَرونَ هذِهِ الأحداثَ، فتَيَقَّنوا أنّ مَوعدَ قيامِ المَملَكةِ الرَّبّانيّةِ باتَ قَريبًا.</span>
<span className="v" id="v32" data-v="32" data-xref="مرقس 13:30; متّى 24:34; متّى 23:36"> والحقّ أقولُ لكُم، إنّ كُلَّ هذِهِ الأشياءِ ستَحدُثُ قَبلَ أن تَنقَضيَ هذِهِ الجَماعة.</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 40:8; مرقس 13:31; متّى 24:35"> ألا وإنّ السَّماواتِ والأرضَ زائلاتٌ، وكَلامي باقٍ لن يَزولَ!</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="مرقس 4:19; لوقا 12:45–46; لوقا 12:40"> فانتَبِهوا لأنفُسِكُم واجتَنِبوا السُّكرَ والانغِماسَ في المَلَذّات والغَرقَ في هُمومِ الحَياةِ، لِئلاّ يأتيَ عليكُم ذلِكَ اليَومُ الرَّهيبُ بَغتةً وأنتُم في غَفلةٍ عَنهُ.</span>
<span className="v" id="v35" data-v="35" data-xref="الرؤيا 16:15; مقتطفات من كتاب الزبور 11:6; إرْمِيا 48:43–44"> إنّه سيأتي لكُلِّ إنسانٍ حَيٍّ على وَجهِ الأرضِ.</span>
<span className="v" id="v36" data-v="36" data-xref="يوحنّا الأولى 2:28; متّى 25:13; صخر الأولى 5:8"> فكونوا على حَذَرٍ وواظِبوا على الدُّعاءِ طالِبينَ القوّةَ على الصُّمودِ أمامَ هذِهِ الفِتَنِ لِتَقِفوا أمامَ سَيِّدِ البَشَرِ عِندَ مَجيئِهِ".</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="لوقا 22:39; مرقس 11:19; متّى 21:17"> وكانَ مِن عادةِ سَيِّدِنا عيسى (سلامُهُ علينا) أن يُعَلِّمَ النّاسَ نَهارًا في الحَرَمِ الشَّريفِ، ثُمّ يَخرُجُ مِن القُدسِ للمَبيتِ لَيلاً في جَبَلِ الزَّيتونِ.</span>
<span className="v" id="v38" data-v="38" data-xref="يوحنّا 8:1–2"> وكانَت جُموعُ النّاسِ تُبَكِّرُ صَباحًا إلى الحَرَمِ للإصغاءِ إليهِ.</span>
</div>
