<span className="v" id="v1" data-v="1" data-xref="متّى 8:5–13; لوقا 7:1–10; متّى 7:28–29"> وبَعدَ أن أتَمَّ سَيِّدُنا عيسى (سلامُهُ علينا) ما أرادَ قولَهُ لِمَن اجتَمِعوا حَولَهُ، مَضى إلى قَريةِ كَفْرَناحومَ.</span>
<span className="v" id="v2" data-v="2" data-xref="2 Kings 5:2–3; التَّكوين 24:27; يوحنّا 4:46–47"> وكانَ فيها ضابِطٌ في جَيشِ الاحتِلالِ الرّومانيِّ، لهُ عَبدٌ ذو مَكانةٍ كَبيرةٍ في نَفسِهِ، وقد أصابَ العَبدَ مَرضٌ كادَ يَقضي عليهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="متّى 8:5; يوحنّا 4:47; لوقا 9:38"> وعَلِمَ الضّابِطُ بِخبرِ عيسى (سلامُهُ علينا)، فطَلَبَ مِن بَعضِ وُجَهاءِ اليَهودِ التَّوَسُّطَ لَدَيهِ (سلامُهُ علينا) ليَشفي عَبدَهُ المَريضَ.</span>
<span className="v" id="v4" data-v="4" data-xref="الرؤيا 3:4; متّى 10:11; لوقا 7:6–7"> واستَجابَ الوُجَهاءُ لطَلَبِ الضّابِطِ، فتَوَجَّهوا إلى سَيِّدِنا عيسى (سلامُهُ علينا)، وعِندَ وُصولِهِم ألحّوا عليهِ قائلينَ: "إنَّ هذا الضّابِطَ مُحِبٌّ لأُمّتِنا، حَقيقٌ بأن تُجيبَ دَعوتَهُ، فقد بَنى لنا بَيتًا للعِبادةِ".</span>
<span className="v" id="v6" data-v="6" data-xref="مرقس 5:24; التَّكوين 32:10; يعقوب 4:6"> فاستَجابَ لهُم سَيِّدُنا عيسى (سلامُهُ علينا) وسارَ مَعَهُم، وعِندَ اقتِرابِهِ مِن بَيتِ الضّابِطِ، بَعَثَ إليهِ بَعضَ الأصدِقاءِ ليَقولوا لهُ عن لِسانِهِ: "مَولاي، لا تُكَلِّفْ نَفسَكَ عَناءَ الدُّخولِ إلى بَيتي الوَضيعِ، لأنّني لَستُ أهلاً أن تَستَظِلَّ بِسَقفي،<sup className="fn"><a href="#fn-a" id="fnref-a" data-fn="7‏.6 لقد عرف الضابط أنّ دخول بيت أجنبي عملٌ نجس بالنسبة إلى اليهود.">a</a></sup></span>
<span className="v" id="v7" data-v="7" data-xref="مقتطفات من كتاب الزبور 107:20; مقتطفات من كتاب الزبور 33:9; لوقا 4:36"> بل إنّي لا أستَحِقُّ شَرفَ الذَّهابِ إليكَ ومُقابلَتِكَ، إذ يَكفي أن تأمرَ، وإن مِن بَعيدٍ، فيُشفَى عَبدي.</span>
<span className="v" id="v8" data-v="8" data-xref="كولوسي 3:22; سيرة الحواريّين 23:17; سيرة الحواريّين 22:25–26"> إنّني أعلَمُ أنَّ باستَطاعتِكَ أن تَفعَلَ ذلِكَ، فأنا يا سَيِّدي مَرؤوسٌ ورَئيسٌ، يأمُرُني رُؤسائي فأُطيعُ، وآمُرُ جُنودي فيَطيعونَ دونَ تَرَدُّدٍ. وعِندَما آمُرُهُم لا أحتاجُ إلى أن أُتابِع أوامري لأنّهُم يُنَفِذّونَها كُلِّها".</span>
<span className="v" id="v9" data-v="9" data-xref="متّى 15:28; روما 3:1–3; متّى 8:10"> وأُعجِبَ سَيِّدُنا عيسى (سلامُهُ علينا) بكَلامِ الضّابِطِ والتَفَتَ إلى الجُمهورِ الّذي كانَ يَتبَعُهُ قائلاً: "إنّي لم أجِدْ مَن يؤمنُ بي كَما آمَنَ بي هذا الرَّجُلُ، ولا حتّى في بَني يَعقوبَ!"</span>
<span className="v" id="v10" data-v="10" data-xref="يوحنّا 4:50–53; متّى 8:13; متّى 15:28"> وعادَ أصدِقاءُ الضّابِطِ إلى بَيتِ صَديقِهِم، فوَجَدوا العَبدَ قد زالَت عَنهُ عِلَّتُهُ وأصبَحَ مُعافىً.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="سيرة الحواريّين 10:38"> ثُم تَوَجَّهَ سَيِّدُنا عيسى (سلامُهُ علينا) مَعَ أتباعِهِ إلى بَلدةٍ اسمُها نَعيمُ، يُرافِقُهُ حَشدٌ غَفيرٌ مِن النّاسِ.</span>
<span className="v" id="v12" data-v="12" data-xref="لوقا 8:42; لوقا 8:52; 1 Kings 17:9"> وعِندَ اقتِرابِهِ مِن بَوّابةِ البَلدةِ اعتَرَضَتْهُم جَنازةُ شابٍّ وَحيدِ أُمِّهِ الأرمَلةِ الّتي كانَ يُحيطُ بها جَمعٌ كَبيرٌ مِن أهلِ البَلدةِ.</span>
<span className="v" id="v13" data-v="13" data-xref="مقتطفات من كتاب الزبور 103:13; إشَعْياء 63:9; مقتطفات من كتاب الزبور 86:15"> وعِندَما رآها سَيِّدُنا عيسى اقتَرَبَ مِنها مُشفِقًا مواسيًا بقَولِهِ: "لا تَبكي!"</span>
<span className="v" id="v14" data-v="14" data-xref="يوحنّا 5:25; يوحنّا 11:43–44; سيرة الحواريّين 9:40–41"> ثُمَّ دَنا مِن النَّعشِ ولَمَسَهُ فوَقَفَ مَن كانوا يَحمِلونَهُ. وخاطَبَ المَيّتَ قائلاً: "قُم أيُّها الشّابُّ!"</span>
<span className="v" id="v15" data-v="15" data-xref="2 Kings 13:21; 2 Kings 4:32–37; 1 Kings 17:23–24"> فبُعِثَ الشّابُّ مِن فَورِهِ حيًّا، وجَلَسَ بَينَ النّاسِ يَتَحَدِّثُ مَعَهُم. ثُمَّ سَلَّمَهُ (سلامُهُ علينا) لأُمِّهِ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="متّى 9:8; لوقا 7:39; لوقا 1:68"> ورانَ في تِلكَ اللَّحظةِ الخُشوعُ على قُلوبِ الجَميعِ، وأخَذوا يُسَبِّحونَ اللهَ قائلينَ: "لقد أرسَلَ اللهُ لنا نَبيًّا عَظيمًا، وأنعَمَ على شَعبِهِ وأعانَهُ".</span>
<span className="v" id="v17" data-v="17" data-xref="متّى 4:24; مرقس 6:14; متّى 9:26"> وهكذا انتَشَرَ خَبَرُ إحياءِ الشّابِّ المَيّتِ في مِنطقةِ يَهوذا وفي كُلِّ أنحاءِ فِلَسطين.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="يوحنّا 3:26; متّى 11:2–19"> وسَمِعَ النَّبيُّ يَحيى (عليه السّلام) وهو في السِّجنِ، مِن أتباعِهِ كُلَّ أخبارِ سَيِّدِنا عيسى (سلامُهُ علينا). فاستَدعَى اثنَينِ مِنهُم</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 9:6–7; يوحنّا 4:25; التَّكوين 22:18"> وأوفَدَهُما إليهِ ليَسألاهُ عن حَقيقتِهِ قائلَينَ: "أأنتَ المُنقِذُ المُنتَظَرُ الّذي وَعَدَنا اللهُ بِهِ؟ أم علينا انتِظارُ ذلِكَ المُنقِذِ مِن بَعدِكَ؟!"</span>
<span className="v" id="v20" data-v="20" data-xref="لوقا 7:19"> ولمّا وَصَلَ مُوفَدَا يَحيى (عليه السّلام)، كانَ سَيِّدُنا عيسى (سلامُهُ علينا) يَشفي جُموعًا مِمِّن أصابَتهُم عِلَلٌ مُختَلِفةٌ ومِمِّن سَكَنَتْهُم الشَّياطينُ ويَهِبُ البَصَرَ لِعُميانٍ كَثيرينَ. فسألاهُ كَما أوصاهُما يَحيى،</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 35:5–6; إشَعْياء 61:1–3; لوقا 4:18"> فرَدَّ عليهِما قائلاً: "عُودَا أدراجَكُما إلى يَحيى وأخبِرَاهُ بما شَهَدتُما وسَمِعتُما: إنّ العُميَ يُبصِرونَ، والمُقعَدينَ يَمشونَ، والبُرْصَ يَبرَؤونَ، والصُمَّ يَسمَعونَ، والمَوتى يَحيَونَ، والمَساكينَ يُبَشَّرونَ،<sup className="fn"><a href="#fn-b" id="fnref-b" data-fn="7‏.22 اقتبس السيّد المسيح (سلامُهُ علينا) ذلك القول من نبوّات النبي أشعيا دلالةً على أنّه هو المنقذ المنتظر الذي بشّر به النبي أشعيا.">b</a></sup></span>
<span className="v" id="v23" data-v="23" data-xref="متّى 11:6; صخر الأولى 2:7–8; روما 9:32–33"> فهَنيئًا لمَن لا يَفقِدُ إيمانَهُ بي".</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="أفاسوس 4:14; يوحنّا 1:23; لوقا 1:80"> واستَمَعَ المُوفَدانِ إلى كَلامِ عيسى (سلامُهُ علينا) ثُمّ انصَرَفا، والتَفَتَ (سلامُهُ علينا) إلى حَشدِ النّاسِ مِن حَولِهِ مُحَدِّثًا إيّاهُم عن يَحيى (عليه السّلام) بقَولِهِ: "عِندَما خَرَجتُم إلى الباديةِ، ماذا رأيتُم؟ أَرَجُلاً هَزيلاً كقَصَبةٍ تَدورُ حَيثُما تَوَجَّهَت بها الرِّيحُ؟</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 59:17; Esther 8:15; Esther 1:3"> أم عاينتُم رَجُلاً يَلبَسُ ثَوبًا فاخِرًا؟ كَلاّ، إنّ الّذينَ يَرتَدونَ المَلابِسَ الفاخِرةَ في تَرَفٍ ونَعيمٍ وهم في قُصورِ المُلوكِ مُرَفّهينَ.</span>
<span className="v" id="v26" data-v="26" data-xref="لوقا 1:76; يوحنّا 5:35; لوقا 20:6"> فلِماذا خَرَجتُم، إذًا؟ هل قَصَدتُم رؤيةَ نَبيٍّ؟ أقولُ لكُم: حقًّا إنّ يَحيى نَبيٌّ بل هو أعظَمُ مِن نَبيّ!</span>
<span className="v" id="v27" data-v="27" data-xref="Malachi 3:1; لوقا 1:76; مرقس 1:2"> لقد جاءَ ذِكرُهُ في كِتابِ النَّبيِّ مَلاكي إذ قالَ تَعالى عَنهُ: "إنّي أبعَثُ أمامَ مُختاري رَسولاً ليُهَيّئ لهُ سَبيلاً".<sup className="fn"><a href="#fn-c" id="fnref-c" data-fn="7‏.27 وَعَدَ الله تعالى على لسان النبي ملاكي أنّه سيرسل النبي يحيى ليسهّل لسيّدنا المسيح السبُلَ لإبلاغ دعوته.">c</a></sup></span>
<span className="v" id="v28" data-v="28" data-xref="متّى 13:16–17; متّى 11:11; كولوسي 1:25–27"> وإنّي مُخبِركُم بأنّهُ لم يُولَد إلى حينِ قيامِ مَملكةِ اللهِ على الأرضِ إنسانٌ أعظَمُ مِن يَحيى، والآنَ وقد حانَ وَقتُ هذِهِ المَملكةِ، فإنّ أقَلَّ واحدٍ مَكانةً في المَملكةِ سيَكونُ أعلى شأنًا مِن النَّبيّ يَحيى!</span>
<span className="v" id="v29" data-v="29" data-xref="لوقا 3:12; لوقا 7:35; سيرة الحواريّين 18:25"> والّذينَ سَمِعوا كَلامَ يَحيى مِن عامّةِ النّاسِ ومِن جُباةِ الضَّرائبِ المَكروهينَ، سَبَّحوا اللهَ لأنّهُ اللَّطيفُ الأمينُ على عَهدِهِ فهو الّذي أعطاهُم فُرصةً ليَتوبوا إلى اللهِ عِندَما طَهَّرَهُم.</span>
<span className="v" id="v30" data-v="30" data-xref="لوقا 13:34; كورنتوس الثانية 6:1; متّى 22:35"> أمّا الفُقَهاءُ والمُتَشَدِّدونَ فكانوا لِهدايةِ اللهِ رافِضينَ فلم يَقبَلوا أن يُطَهِّرَهُم بالماءِ".</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="متّى 11:16–19; Lamentations 2:13; مرقس 4:30"> وتابَعَ سَيِّدُنا عيسى (سلامُهُ علينا) حَديثَهُ قائلاً: "بمَن أُشَبِّهُ هؤلاءِ الرّافِضينَ؟</span>
<span className="v" id="v32" data-v="32" data-xref="متّى 11:16–19; إرْمِيا 5:3–5; الأمثال 17:16"> إنّ مَثَلَهُم كمَثَلِ أولادٍ في السّاحةِ يَلعَبونَ يَقولُ بَعضُهُم لبَعضٍ: "ما بِكُم تَرفُضونَ كُلَّ شَيء؟ إن لَعِبنا لُعبةَ العُرسِ، لا تَرقُصونَ، أو لَعِبنا لُعبةَ الجَنازةِ، لا تَبكونَ!"<sup className="fn"><a href="#fn-d" id="fnref-d" data-fn="7‏.32 كان السيّد المسيح يقول إنّ بعض الأشخاص من اليهود ورجال الدين كانوا يتنازعون كالأطفال، فبعضهم كانوا يرغبون في لعب الألعاب المرحة، بينما يريد آخرون لعب الألعاب الحزينة. ويقول السيد المسيح إنّ بعض رجال الدين لم يكونوا ليوافقوا على نموذج النبي يحيى لأنّه كان صارمًا، ولم يكونوا يحبّون طريقة السيّد المسيح، لأنّه، في رأيهم، لم يكن حازمًا بما فيه الكفاية.">d</a></sup></span>
<span className="v" id="v33" data-v="33" data-xref="لوقا 1:15; مرقس 1:6; يوحنّا 8:52"> كَذا فَعَلتُم عِندَما رَفَضتُم يَحيى بِن زَكَريّا وأنكَرتُم عليهِ زُهدَهُ في الخُبزِ والخَمرِ وادّعَيتُم عليهِ بقَولِكُم: "مَسَّهُ شَيطانٌ"</span>
<span className="v" id="v34" data-v="34" data-xref="لوقا 15:2; لوقا 7:36; يوحنّا 12:2"> وكذلِكَ فَعَلتُم عِندَما أتاكُم سَيِّدُ البَشَرِ الّذي يأكُلُ ويَشرَبُ كبَقيَّةِ الخَلقِ وادّعَيتُم عليهِ بقَولِكُم: "إن هو إلاّ رَجُلٌ يأكُلُ ويَشرَبُ بِشَرَهٍ، ويُصادِقُ جُباةَ الضَّرائبِ والضّالّينَ مِن النّاسِ!"</span>
<span className="v" id="v35" data-v="35" data-xref="الأمثال 8:32–36; متّى 11:19; Hosea 14:9"> ولكن اعلَموا أنّ الحُكماءَ هُم الّذينَ يَستَجيبونَ لِحِكمةِ اللهِ عِندَما تَظهَرُ لهُم".</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="مرقس 14:3–9; متّى 26:5–6; لوقا 14:1"> وكانَ أن دَعا أحَدُ المُتَشَدِّدينَ سَيِّدَنا عيسى (سلامُهُ علينا) إلى بَيتِهِ، فأجابَ دَعوتَهُ. وبَينَما كانَ يَتَناولُ الطّعامَ مَعَ جَمعٍ مِن النّاسِ،</span>
<span className="v" id="v37" data-v="37" data-xref="مرقس 14:3–9; متّى 26:6–13; متّى 21:31"> أتَتْهُ امرأةٌ سَيّئةُ السُّمعةِ قد عَلِمَت بوجودِهِ في بَيتِ ذلِكَ المُتَشَدّدِ، وكانَت تَحمِلُ قارورةً مِن المَرمَرِ بها عِطرٌ ثَمينٌ،</span>
<span className="v" id="v38" data-v="38" data-xref="مقتطفات من كتاب الزبور 126:5–6; مقتطفات من كتاب الزبور 51:17; التَّكوين 18:4"> فوَقَفَت وَراءَهُ باكيةً، ورَكَعَت غاسِلةً بِدُموعِها قدَمَيْهِ،<sup className="fn"><a href="#fn-e" id="fnref-e" data-fn="7‏.38 كانت غرفة العشاء مفتوحة على الشارع (حسب العادة آنذاك) وكان المدعوّون يجلسون إلى موائد حولها أرائك مصفوفة يتّكئون عليها. وكان عامّة الناس يحضرون تلك المجالس لمشاهدتها فقط.">e</a></sup>ثُمّ مَسَحَتْهُما بشَعرِها، وقَبَّلَتْهُما، وسَكَبَت عِطرَ القارورةِ عليهِما.</span>
