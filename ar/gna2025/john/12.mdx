## امرأة تسكب الطيب على يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 14:3–8; يُوحَنّا 11:1; لُوقا 24:50"> وقَبلَ الفِصحِ بسِتَّةِ أيّامٍ، جاءَ يَسوعُ إلى بَيت عنيا ونزَلَ عِندَ لِعازَرَ الّذي أقامَهُ مِنْ بَينِ الأمواتِ.</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 22:27; لُوقا 10:38–42; رُؤيا يُوحَنّا 3:20"> فهَيَّأوا لَه عَشاءً، وأخَذَت مرتا تَخدُمُ، وكانَ لِعازَرُ أحدَ الجالِسينَ معَهُ لِلطَّعامِ.</span>
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 14:3–9; يُوحَنّا 11:2; لُوقا 10:38–39"> فتَناوَلَت مَريَمُ قارورَةَ طِيبٍ غالي الثَّمنِ مِنَ النَّارِدينِ النَّقيِّ، وسكَبَتْها على قدَمَي يَسوعَ ومَسَحتْهُما بِشَعرِها. فاَمتَلأ البَيتُ بِرائِحَةِ الطِّيبِ،</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 10:4; صَمُوئيلَ الأوَّلُ 17:28–29; يُوحَنّا 13:2"> فقَالَ يَهوذا الإسخَريوطيُّ، أحدُ تلاميذِهِ، وهوَ الّذي سيُسلِّمُهُ:</span>
<span className="v" id="v5" data-v="5" data-xref="لُوقا 18:22; لُوقا 12:33; مَرْقُس 14:4–5"> ”أما كانَ خَيرًا أنْ يُباعَ هذا الطِّيبُ بِثلاثِ مئةِ دينارٍ لِتُوَزَّعَ على الفُقراءِ؟“</span>
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 13:29; المَزامِير 50:16–20; يَعقُوب 2:6"> قالَ هذا لا لِعَطفِهِ على الفُقراءِ، بَلْ لأنَّهُ كانَ لِصًّا وكانَ أمينَ الصُّندوقِ، فيَختلِسُ ما يُودَعُ فيهِ.</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 26:12; مَرْقُس 14:6; ﻣﺘﻰ 27:57–60"> فقَالَ يَسوعُ: ”أُترُكوها! هذا الطِّيبُ حَفِظَتْهُ لِيَومِ دَفني.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 15:11; مَرْقُس 14:7; ﻣﺘﻰ 26:11"> فالفُقَراءُ عِندَكُم في كُلِّ حينٍ، وأمَّا أنا فلا أكونُ في كُلِّ حينٍ عِندَكم“.</span>
</div>
</div>

## محاولة قتل لعازر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 4:14; يُوحَنّا 11:43–45; اعمال الرسل 3:10–11"> وعَلِمَ جُمهورٌ كبيرٌ مِنَ اليَهودِ أنَّ يَسوعَ في بَيت عنيا فجاؤُوا، لا مِنْ أجلِهِ وحدَهُ، بَلْ ليُشاهِدوا لِعازَرَ الّذي أقامَهُ مِنْ بَينِ الأمواتِ.</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 16:31; ﻣﺘﻰ 2:16; يُوحَنّا 11:47–53"> فتَشاوَرَ رُؤساءُ الكَهنَةِ ليَقتُلوا لِعازَرَ أيضًا،</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 11:45; يُوحَنّا 12:18; اعمال الرسل 13:45"> لأنَّ كثيرًا مِنَ اليَهودِ كانوا يَترُكونَهُم بِسبَبِه ويُؤمنونَ بيَسوعَ.</span>
</div>
</div>

## يسوع يدخل أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="مَرْقُس 11:7–10; ﻣﺘﻰ 21:4–9; لُوقا 19:35–38"> وفي الغَدِ، سَمِعَتِ الجُموعُ الّتي جاءَت إلى العيدِ أنَّ يَسوعَ قادِمٌ إلى أُورُشليمَ.</span>
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 118:25–26; رُؤيا يُوحَنّا 15:3; ﻣﺘﻰ 21:9–11"> فحَمَلوا أغصانَ النَّخلِ وخَرَجوا لاَستِقبالِهِ وهُم يَهتِفونَ: المَجدُ للهِ! تبارَكَ الآتي باَسمِ الرّبِّ! تبارَكَ مَلِكُ إِسرائيلَ!“</span>
<span className="v" id="v14" data-v="14" data-xref="زَكَريّا 9:9; مَرْقُس 11:1–7; ﻣﺘﻰ 21:1–7"> ووجَدَ يَسوعُ جَحشًا فرَكِبَ علَيهِ، كما جاءَ في الكِتابِ:</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="زَكَريّا 9:9; إشَعْياء 62:11; صَفَنْيا 3:16–17"> ”لا تخافي يا بِنتَ صهيونَ:<br />ها هوَ مَلِكُكِ قادِمٌ إليكِ،<br />راكِبًا على جَحشٍ اَبنِ أتانٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 2:22; يُوحَنّا 14:26; يُوحَنّا 7:39"> وما فَهِمَ التَّلاميذُ في ذلِكَ الوقتِ مَعنى هذا كُلِّهِ. ولكنَّهُم تَذكَّروا، بَعدَما تَمجَّدَ يَسوعُ، أنَّ هذِهِ الآيةَ ورَدَت لتُخبِرَ عَنهُ، وأنَّ الجُموعَ عَمِلوا هذا مِنْ أجلِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="رُؤيا يُوحَنّا 1:2; رسالَةُ يُوحَنّا الأُولَى 5:9–12; يُوحَنّا 19:35"> وكانَ الجَمعُ الّذينَ رافَقوا يَسوعَ عِندَما دَعا لِعازَرَ مِنَ القَبرِ وأقامَهُ مِنْ بَينِ الأمواتِ، يَشهَدونَ لَه بذلِكَ.</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 12:11"> وخَرجَتِ الجَماهيرُ لاَستقبالِهِ لأنَّها سَمِعَت أنَّهُ صَنَعَ تِلكَ الآيةَ.</span>
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 17:6; المَزامِير 22:27; رسالَةُ يُوحَنّا الأُولَى 2:2"> فقَالَ الفَرِّيسيُّونَ بَعضُهُم لِبَعضٍ: ”أرَأيتُم كيفَ أنَّكُم لا تَنفَعونَ شَيئًا. ها هوَ العالَمُ كُلُّهُ يَتبَعُهُ!“</span>
</div>
</div>

## يسوع يُنبئ بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 7:35; إشَعْياء 11:10; اعمال الرسل 17:4"> وكانَ بَعضُ اليونانيّينَ يُرافِقونَ الّذينَ صَعِدوا إلى أُورُشليمَ لِلعبادَةِ في أيّامِ العيدِ.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 2:2; لُوقا 19:2–4; يُوحَنّا 6:40"> فجاؤُوا إلى فيلُبُّسَ، وكانَ مِنْ بَيتِ صيدا في الجليلِ. وقالوا لَه: ”يا سيِّدُ، نُريدُ أنْ نرى يَسوعَ“.</span>
<span className="v" id="v22" data-v="22" data-xref="مَرْقُس 10:13–14; لُوقا 9:49–50; يُوحَنّا 1:40–41"> فذَهَبَ فيلبُّسُ وأخبَرَ أندراوُسَ، وذهَبَ فيلبُّسُ وأندراوُسُ وأخبَرا يَسوعَ.</span>
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 13:31–32; يُوحَنّا 17:1–5; إشَعْياء 60:9"> فأجابَهُما يَسوعُ: ”جاءَتِ السَّاعةُ الّتي فيها يتَمَجَّدُ اَبنُ الإنسانِ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:36–38; إشَعْياء 53:10–12; العِبرانِيّين 2:9–10"> الحقَّ الحقَّ أقولُ لكُم: إنْ كانَتِ الحَبَّةُ مِنَ الحِنطَةِ لا تقَعُ في الأرضِ وتَموتُ، تَبقى وَحدَها. وإذا ماتَت أخرَجَت حَبًّا كثيرًا.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 10:39; لُوقا 9:23–24; مَرْقُس 8:35"> مَنْ أحبَّ نَفسَهُ خَسِرَها، ومَنْ أنكَرَ نَفسَهُ في هذا العالَمِ حَفِظَها للحَياةِ الأبديَّةِ.</span>
<span className="v" id="v26" data-v="26" data-xref="كُولُوسِي 3:24; غَلاطِيَّة 1:10; صَمُوئيلَ الأوَّلُ 2:30"> مَنْ أرادَ أنْ يَخدُمَني، فليَتْبَعْني، وحَيثُ أكونُ أنا يكونُ خادِمي. ومَنْ خَدَمَني أكرَمَهُ الآبُ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 12:23; العِبرانِيّين 5:7; ﻣﺘﻰ 26:42"> الآنَ نَفسي مُضطَرِبَةٌ، فماذا أقولُ؟ هَلْ أقولُ: يا أبي، نَجِّني مِنْ هذِهِ السَّاعةِ؟ ولكنِّي لِهذا جِئتُ.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 13:31–32; ﻣﺘﻰ 26:42; رُؤيا يُوحَنّا 5:9–14"> يا أبي، مَجِّدِ اَسمَكَ!“ فقَالَ صوتٌ مِنَ السَّماءِ: ”مَجَّدتُهُ وسأمَجِّدُهُ!“</span>
<span className="v" id="v29" data-v="29" data-xref="أيُّوب 37:2–5; رُؤيا يُوحَنّا 6:1; رُؤيا يُوحَنّا 14:2"> فسَمِعَهُ الحاضِرونَ، فقَالوا: ”هذا دَوِيُّ رَعدٍ!“ وقالَ بَعضُهُم: ”كَلَّمَهُ مَلاكٌ!“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 11:42; يُوحَنّا 11:15; يُوحَنّا 5:34"> فقَالَ يَسوعُ: ”ما كانَ هذا الصَّوتُ لأجلي، بل لأجلِكُم.</span>
<span className="v" id="v31" data-v="31" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:4; رسالَةُ يُوحَنّا الأُولَى 5:19; رسالَةُ يُوحَنّا الأُولَى 3:8"> اليومَ دَينونَةُ هذا العالَمِ. واليومَ يُطرَدُ سيِّدُ هذا العالَمِ.</span>
<span className="v" id="v32" data-v="32" data-xref="يُوحَنّا 3:14; يُوحَنّا 8:28; رسالَةُ بُطرُسَ الأُولَى 2:24"> وأنا متى اَرتَفَعتُ مِنْ هذِهِ الأرضِ، جَذَبتُ إليَّ النَّاسَ أجمعينَ“.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 18:32; يُوحَنّا 21:19"> قالَ هذا مُشيرًا إلى الميتَةِ الّتي سيَمُوتُها.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="حِزْقِيال 37:24–25; إشَعْياء 9:7; دانيال 7:14"> فأجابَهُ الجَمعُ: ”علَّمَتْنا الشَّريعةُ أنَّ المَسيحَ يَبقى إلى الأبَدِ. فكيفَ تَقولُ: لا بُدَّ لاَبنِ الإنسانِ أنْ يرتَفِعَ؟ فمَنْ هوَ اَبنُ الإنسانِ هذا؟“.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="يُوحَنّا 12:46; يُوحَنّا 16:16; أفَسُس 5:8"> فقَالَ لهُم يَسوعُ: ”سيَبقى النُّورُ مَعكُم وقتًا قليلًا، فاَمشُوا ما دامَ لكُمُ النُّورُ، لِئَلَّا يُباغِتَكُمُ الظَّلامُ. والّذي يَمشي في الظَّلامِ لا يَعرِفُ إلى أينَ يَتَّجِهُ.</span>
<span className="v" id="v36" data-v="36" data-xref="أفَسُس 5:8; يُوحَنّا 8:59; اعمال الرسل 13:47–48"> آمِنوا بالنُّورِ، ما دامَ لكُمُ النُّورُ، فتكونوا أبناءَ النُّورِ“. اليهود يرفضون الإيمان بيسوع<br />واَبتعَدَ يَسوعُ عَنْ أنظارِهِم، بَعدَما قالَ لهُم هذا الكلامَ.</span>
<span className="v" id="v37" data-v="37" data-xref="لُوقا 16:31; يُوحَنّا 1:11; يُوحَنّا 15:24"> ومع أنَّهُ عَمِلَ لهُم كُلَّ هذِهِ الآياتِ، فما آمَنوا بِه،</span>
<span className="v" id="v38" data-v="38" data-xref="إشَعْياء 53:1; ﺭﻭﻣﻴﺔ 10:16; ﺭﻭﻣﻴﺔ 10:20"> لِيَتِمَّ ما قالَ النَّبيُّ إشَعيا: ”يا رَبُّ، مَنِ الّذي آمَنَ بِكلامِنا؟ ولِمَنْ ظَهَرَت يَدُ الرَّبِّ؟“</span>
<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 10:38; يُوحَنّا 5:44; رسالَةُ بُطرُسَ الثّانيَة 2:14"> وما قَدِروا أنْ يُؤمِنوا بِه، لأنَّ إشَعيا قالَ أيضًا:</span>
<span className="v" id="v40" data-v="40" data-xref="إشَعْياء 6:10; ﻣﺘﻰ 13:13–15; لُوقا 8:10"> ”أعمى اللهُ عُيونَهُم وقَسَّى قُلوبَهُم، لِئَلَّا يُبصِروا بِعيونِهِم ويَفهَموا بِقلوبِهِم ويَتوبوا فأشفيَهُم“.</span>
<span className="v" id="v41" data-v="41" data-xref="إشَعْياء 6:1–5; إشَعْياء 6:9–10; يُوحَنّا 1:14"> قالَ إشَعيا هذا الكلامَ لأنَّهُ رأى مَجدَهُ وتَحدَّثَ عَنهُ.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 9:22; يُوحَنّا 7:13; إشَعْياء 51:7"> غيرَ أنَّ كثيرًا مِنْ رُؤساءِ اليَهودِ أنفُسِهِم آمَنوا بيَسوعَ، ولكنَّهُم ما أعلَنُوا إيمانَهُم مُسايَرةً لِلفَرِّيسيِّينَ، لِئَلَّا يُطرَدوا مِنَ المَجمَعِ.</span>
<span className="v" id="v43" data-v="43" data-xref="يُوحَنّا 5:44; ﻣﺘﻰ 23:5–7; يُوحَنّا 5:41"> كانوا يُحبُّونَ رِضى النَّاسِ أكثَرَ مِنْ رِضى اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 10:40; يُوحَنّا 13:20; يُوحَنّا 5:24"> فرَفَعَ يَسوعُ صوتَهُ، قالَ: ”مَنْ آمنَ بي لا يُؤمِنُ بي أنا، بل يُؤمِنُ بالّذي أرسَلَني.</span>
<span className="v" id="v45" data-v="45" data-xref="يُوحَنّا 14:9–10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:6; كُولُوسِي 1:15"> ومَنْ رَآني رأى الّذي أرْسَلَني.</span>
