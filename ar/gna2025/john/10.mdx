## يسوع الراعي الصالح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 7:15; يُوحَنّا 10:8–10; ﺭﻭﻣﻴﺔ 16:18"> ”الحقَّ الحقَّ أقولُ لكُم: مَنْ لا يَدخُلُ حَظيرَةَ الخِرافِ مِنَ البابِ، بل يَصعَدُ علَيها مِنْ مكانٍ آخَرَ، فهوَ سارِقٌ ولِصٌّ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 10:14; اعمال الرسل 20:28; رسالَةُ بُطرُسَ الأُولَى 5:4"> أمَّا مَنْ يَدخُلُ مِنَ البابِ فهوَ راعي الخِراف.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 34:11–16; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:19; يُوحَنّا 10:4"> لَه يَفتَحُ البوّابُ، وإلى صوتِهِ تُصغي الخِرافُ. يَدعو كُلَّ واحدٍ مِنْ خِرافِهِ باَسمِهِ ويُخرِجُهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 1:30; يُوحَنّا 10:16; يُوحَنّا 12:26"> وعِندَما يُخرِجُها يَمشي قُدَّامَها، والخِرافُ تَتبَعُهُ لأنَّها تَعرِفُ صَوتَهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:5–6; كُولُوسِي 2:6–10; مَرْقُس 4:24"> أمَّا الغريبُ فتَهرُبُ مِنهُ ولا تَتبَعُهُ، لأنَّها لا تَعرِفُ صَوتَ الغُرَباءِ“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 56:11; المَزامِير 82:5; الأمثال 28:5"> قالَ يَسوعُ هذا المثَلَ، فما فَهِموا مَعنى كلامِهِ،</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 14:6; أفَسُس 2:18; يُوحَنّا 10:1"> فقَالَ لهُم: ”الحقَّ الحقَّ أقولُ لكُم: أنا بابُ الخِرافِ.</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 34:2; إرْمِيا 23:1; يُوحَنّا 10:1"> جميعُ مَنْ جاؤُوا قَبلي سارِقونَ وَلُصوصٌ، فَما أصغَتْ إلَيهِمِ الخِرافُ.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 14:6; أفَسُس 2:18; يُوحَنّا 10:7"> أنا هوَ البابُ، فمَنْ دخَلَ مِنِّي يَخلُصُ: يدخُلُ ويَخرُجُ ويَجِدُ مَرعًى.</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 19:10; يُوحَنّا 6:51; يُوحَنّا 12:47"> لا يَجيءُ السّارِقُ إلَّا ليَسرِقَ ويَقتُلَ ويَهدِمَ. أمَّا أنا فجِئْتُ لِتكونَ لهُمُ الحياةُ، بل مِلءُ الحياةِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 10:14–15; المَزامِير 23:1; يُوحَنّا 15:13"> أنا الرّاعي الصَّالِحُ، والرّاعي الصَّالِحُ يُضحِّي بِحياتِهِ في سبيلِ الخِرافِ.</span>
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 20:29; زَكَريّا 11:16–17; حِزْقِيال 34:2–6"> وما الأجيرُ مِثلُ الرّاعي، لأنَّ الخِرافَ لا تَخصُّهُ. فإذا رأى الذِّئبَ هاجِمًا، ترَكَ الخِرافَ وهرَبَ، فيَخطَفُ الذِّئبُ الخِرافَ ويُبدِّدُها.</span>
<span className="v" id="v13" data-v="13" data-xref="فِيلِبِّي 2:20; يُوحَنّا 12:6; اعمال الرسل 18:17"> وهوَ يَهرُبُ لأنَّهُ أجيرٌ لا تَهُمُّهُ الخِرافُ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 10:27; يُوحَنّا 10:11; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:19"> أنا الرّاعي الصَّالِحُ، أعرِفُ خِرافي وخِرافي تَعرِفُني،</span>
<span className="v" id="v15" data-v="15" data-xref="غَلاطِيَّة 1:4; يُوحَنّا 10:11; إشَعْياء 53:10"> مِثلَما يَعرِفُني الآبُ وأعرِفُ أنا الآبَ، وأُضَحِّي بحَياتي في سَبيلِ خِرافي.</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 56:8; يُوحَنّا 11:52; رسالَةُ بُطرُسَ الأُولَى 2:25"> ولي خِرافٌ أُخرى مِنْ غَيرِ هذِهِ الحَظيرةِ، فيَجِبُ علَيَّ أنْ أقودَها هيَ أيضًا. ستَسمَعُ صوتي، فتكونُ الرَّعِيَّةُ واحدةً والرّاعي واحدًا.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 10:18; يُوحَنّا 10:11; العِبرانِيّين 2:9"> والآبُ يُحبُّني لأنِّي أُضَحِّي بِحياتي حتّى أستَرِدَّها.</span>
<span className="v" id="v18" data-v="18" data-xref="العِبرانِيّين 2:9; تِيطُس 2:14; يُوحَنّا 2:19–21"> ما مِنْ أحَدٍ يَنتَزِعُ حياتي مِنِّي، بل أنا أُضَحِّي بِها راضِيًا. فَلِيَ القُدرَةُ أنْ أُضَحِّيَ بِها، وليَ القُدرَةُ أنْ أستَرِدَّها. هذِهِ الوَصيَّةُ تَلقَّيتُها مِنْ أبي“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 9:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:3; يُوحَنّا 7:40–43"> ووقَعَ خِلافٌ آخَرُ بَينَ اليَهودِ على هذا الكلامِ،</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 7:20; مَرْقُس 3:21; يُوحَنّا 8:47–48"> فقَالَ كثيرٌ مِنهُم: ”هذا الرَّجُلُ فيهِ شَيطانٌ، فهوَ يَهذي. لماذا تُصغونَ إلَيهِ؟“</span>
<span className="v" id="v21" data-v="21" data-xref="الخُرُوج 4:11; إشَعْياء 35:5–6; المَزامِير 94:9"> وقالَ آخرونَ: ”ما هذا كلامُ رَجُلٍ فيهِ شَيطانٌ. أيَقدِرُ الشَّيطانُ أنْ يَفتَحَ عُيونَ العُميانِ؟“.</span>
</div>
</div>

## اليهود يرفضون يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22"> وجاءَ عيدُ التَّجديدِ في أُورُشليمَ، وذلِكَ في الشِّتاءِ.</span>
<span className="v" id="v23" data-v="23" data-xref="اعمال الرسل 5:12; اعمال الرسل 3:11"> وكانَ يَسوعُ يتَمشَّى في الهَيكَلِ في رِواقِ سُليمانَ،</span>
<span className="v" id="v24" data-v="24" data-xref="لُوقا 22:67–70; يُوحَنّا 1:19; يُوحَنّا 8:25"> فتَجَمَّعَ اليَهودُ حولَهُ وقالوا لَه: ”إلى متى تُبقينا حائِرينَ؟ قُلْ لنا بِصَراحةٍ: هل أنتَ المَسيحُ؟“</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 10:38; يُوحَنّا 8:58; يُوحَنّا 14:11"> فأجابَهُم يَسوعُ: ”قُلتُهُ لكُم، ولكنَّكُم لا تُصَدِّقونَ. الأعمالُ الّتي أعمَلُها باَسمِ أبي تَشهَدُ لي.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 8:47; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:3–4; يُوحَنّا 12:37–40"> وكيفَ تُصدِّقونَ وما أنتُم مِنْ خِرافي.</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 10:16; يُوحَنّا 10:14; رُؤيا يُوحَنّا 3:20"> خِرافي تَسمَعُ صوتي، وأنا أعرِفُها، وهيَ تَتبَعُني.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 6:39–40; يُوحَنّا 6:37; يُوحَنّا 3:15–16"> أُعطيها الحياةَ الأبدِيَّةَ، فلا تَهلِكُ أبدًا ولا يَخطَفُها أحدٌ مِنِّي.</span>
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 6:37; يُوحَنّا 17:2; يُوحَنّا 17:9"> الآبُ الّذي وهَبَها لي هوَ أعظَمُ مِنْ كُلِّ موجودٍ، وما مِنْ أحدٍ يَقدِرُ أن يَخطَفَ مِنْ يدِ الآبِ شَيئًا،</span>
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 1:1–2; يُوحَنّا 14:9; يُوحَنّا 17:21–23"> أنا والآبُ واحِدٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 5:18; يُوحَنّا 8:59; صَمُوئيلَ الأوَّلُ 30:6"> وجاءَ اليَهودُ بِحجارَةٍ ليَرجُموه.</span>
<span className="v" id="v32" data-v="32" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:12; يُوحَنّا 5:36; المَزامِير 35:12"> فقَالَ لهُم يسوعُ: ”أرَيتُكُم كثيرًا مِنَ الأعمالِ الصَّالِحَةِ مِنْ عِندِ الآبِ، فلأيِّ عمَلٍ مِنها تَرجُموني؟“</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 5:18; اللّاويِّين 24:16; يُوحَنّا 10:30"> أجابَهُ اليَهودُ: ”لا نَرجُمُكَ لأيِّ عمَلٍ صالِحٍ عَمِلتَ، بل لِتَجديفِكَ. فما أنتَ إلَّا إنسانٌ، لكِنَّكَ جَعلتَ نَفسَكَ إلهًا“.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="المَزامِير 82:6–7; الخُرُوج 7:1; المَزامِير 82:1"> فقَالَ لهُم يَسوعُ: ”أما جاءَ في شَريعتِكُم أنَّ اللهَ قالَ: أنتُم آلهةٌ؟</span>
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 24:35; ﺭﻭﻣﻴﺔ 13:1; لُوقا 16:17"> فإذا كانَ الّذينَ تكلَّموا بِوَحيٍ مِنَ اللهِ يدعوهُمُ اللهُ آلهةً، على حدِّ قَولِ الشَّريعةِ الّتي لا يَنقُضُها أحَدٌ،</span>
<span className="v" id="v36" data-v="36" data-xref="يُوحَنّا 5:17–18; يُوحَنّا 3:17; إرْمِيا 1:5"> فكيفَ تَقولونَ لي، أنا الّذي قَدَّسَهُ الآبُ وأرسَلَهُ إلى العالَمِ: أنتَ تجدّفُ، لأنِّي قُلتُ: أنا اَبنُ اللهِ؟</span>
<span className="v" id="v37" data-v="37" data-xref="يُوحَنّا 15:24; يُوحَنّا 10:25; يُوحَنّا 12:37–40"> إذا كُنتُ لا أعمَلُ أعمالَ أبي، فلا تُصَدِّقوني.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 14:20; يُوحَنّا 10:25; يُوحَنّا 10:30"> وإذا كُنتُ أعمَلُها، فصَدِّقوا هذِهِ الأعمالَ إنْ كُنتُم لا تُصَدِّقوني، حتّى تَعرِفوا وتُؤمِنوا أنَّ الآبَ فِيَّ وأنا في الآبِ“.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 7:30; يُوحَنّا 8:59; يُوحَنّا 7:44"> فحاوَلوا أنْ يُمسِكوهُ، فأفلَتَ مِنْ أيديهِم.</span>
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 1:28; يُوحَنّا 11:54; يُوحَنّا 7:1"> ورجَعَ إلى عَبرِ الأُردُنِّ حَيثُ كانَ يوحنَّا يُعَمِّدُ مِنْ قَبلُ، فأقامَ هُناكَ.</span>
<span className="v" id="v41" data-v="41" data-xref="يُوحَنّا 1:33–34; يُوحَنّا 1:29–30; مَرْقُس 1:37"> فجاءَ إلَيهِ كثيرٌ مِنَ النَّاسِ وقالوا: ”ما عَمِلَ يوحنَّا آيةً واحِدةً، ولكِنْ ما قالَهُ في هذا الرَّجُلِ كانَ كُلُّهُ صَحيحًا“.</span>
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 4:41; يُوحَنّا 2:23; يُوحَنّا 7:31"> فآمَنَ بِيَسوعَ كثيرٌ مِنَ النَّاسِ هُناكَ.</span>
</div>
