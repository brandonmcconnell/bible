## يسوع يطعم خمسة آلاف رجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 6:31–44; يُوحَنّا 6:23; لُوقا 5:1"> ثُمَّ عبَرَ يَسوعُ بَحرَ الجليلِ وهوَ بُحيرةُ طَبَرِيَّةَ،</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 4:24–25; ﻣﺘﻰ 12:15; ﻣﺘﻰ 15:30–31"> فتَبِعَهُ جُمهورٌ كبيرٌ، لأنَّهُم رأَوا آياتِهِ في شِفاءِ المَرضى.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 6:15; لُوقا 9:28; لُوقا 6:12–13"> فصَعِدَ يَسوعُ إلى الجبَلِ وجلَسَ معَ تلاميذِهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 2:13; يُوحَنّا 11:55; يُوحَنّا 5:1"> وكانَ اَقتَرَبَ عِيدُ الفِصْحِ عِندَ اليَهودِ.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 15:33; لُوقا 9:12–13; ﻣﺘﻰ 14:14–15"> فرفَعَ يَسوعُ عينَيهِ فرأى الجُموعَ مُقبِلَةً إلَيهِ، فقَالَ لِفيلُبُّسَ: ”مِنْ أينَ نَشتَري الخُبزَ لنُطعِمَهُم؟“</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 22:1; التَّثْنِيَة 13:3; أخبارِ الأيّامِ الثّانِي 32:31"> قالَ هذا ليُجَرِّبَ فِيلُبُّسَ، لأنَّ يَسوعَ كانَ يَعرفُ ما سيَعمَلُ.</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 11:21–22; مَرْقُس 6:37; يُوحَنّا 12:5"> فأجابَهُ فِيلبُّسُ: ”لَو اَشتَرَينا خُبزًا بِمئَتي دينارٍ، لَما كَفى أن يَحصُلَ الواحدُ مِنهُم على كِسرةٍ صَغيرةٍ“.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 1:40–44; ﻣﺘﻰ 4:18"> وقالَ لَه أحدُ تلاميذِهِ، وهوَ أندَراوسُ أخو سِمعانَ بُطرُسَ:</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الثّانِي 4:42–44; ﻣﺘﻰ 14:17; مَرْقُس 6:38"> ”هُنا صَبِيٌّ معَهُ خَمسَةُ أرغِفَةٍ مِنْ شَعيرٍ وسَمكتانِ، ولكنْ ما نَفعُها لِمثلِ هذا الجَمعِ؟“</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 9:14–16; ﻣﺘﻰ 15:35–36; مَرْقُس 6:39–41"> فقَالَ يَسوعُ: ”أقعِدوا النَّاسَ“. وكانَ هُناكَ عُشبٌ كثيرٌ فقَعَدُوا. وكانَ عدَدُ الرِّجالِ نحوَ خَمسَةِ آلافٍ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 6:23; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:18; ﻣﺘﻰ 15:36"> فأخَذَ يَسوعُ الأرغِفَةَ وشكَرَ، ثُمَّ وزَّعَ على الحاضِرينَ بِمقدارِ ما أرادوا. وهكذا فعَلَ بالسَّمكَتَينِ.</span>
<span className="v" id="v12" data-v="12" data-xref="لُوقا 1:53; نَحَمْيا 9:25; نَحَمْيا 8:10"> فلمَّا شَبِعوا، قالَ لِتلاميذِهِ: ”إِجمَعوا ما فَضلَ مِنَ الكِسَرِ لِئَلَّا يَضيعَ منها شيءٌ“.</span>
<span className="v" id="v13" data-v="13" data-xref="الأمثال 11:24–25; المُلُوكِ الثّانِي 4:2–7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:8–9"> فجَمَعوها ومَلأُوا اَثنَتَي عشْرَةَ قُفَّةً مِنَ الكِسَرِ الّتي فَضَلَت عَنِ الآكلينَ مِنْ أرغِفَةِ الشَّعيرِ الخَمسَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="التَّثْنِيَة 18:15–18; ﻣﺘﻰ 21:11; اعمال الرسل 7:37"> فلمَّا رأى النَّاسُ هذِهِ الآيةَ الّتي صنَعَها يَسوعُ قالوا: ”بالحقيقَةِ، هذا هوَ النَّبِيُّ الآتي إلى العالَمِ!“</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 18:36; يُوحَنّا 6:15–21; يُوحَنّا 7:3–4"> وعرَفَ يَسوعُ أنَّهُم يَستَعِدُّونَ لاَختِطافِهِ وجَعْلِه مَلِكًا، فاَبتَعدَ عنهُم ورَجَعَ وحدَهُ إلى الجبَلِ.</span>
</div>
</div>

## يسوع يمشي على الماء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16"> ولمَّا جاءَ المَساءُ، نزَلَ تلاميذُهُ إلى بَحرِ الجليلِ.</span>
<span className="v" id="v17" data-v="17" data-xref="مَرْقُس 6:45; يُوحَنّا 6:24–25; يُوحَنّا 2:12"> فركِبوا قارِبًا وعَبَروا بِه إلى كَفْرَناحومَ. وأظلَمَ اللَّيلُ قَبلَ أنْ يَلحَقَهُم يَسوعُ.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 107:25; المَزامِير 135:7; ﻣﺘﻰ 14:24"> وهبَّت عاصِفَةٌ شَديدَةٌ، فَهاجَ البحرُ.</span>
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 6:47–49; ﻣﺘﻰ 14:25–26; أيُّوب 9:8"> وبَعدَما قَطَعوا ثلاثَةَ أميالٍ أو أربعةً، رأوا يَسوعَ يَدنو مِنَ القارِبِ ماشيًا على البحرِ فخافوا.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 14:27–31; إشَعْياء 43:1–2; إشَعْياء 44:8"> فقَالَ لهُم: ”أنا هوَ لا تَخافوا!“</span>
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 6:51; ﻣﺘﻰ 14:32–33; رُؤيا يُوحَنّا 3:20"> وأرادوا أنْ يُصعِدوهُ إلى القارِبِ، فوصَلَ القارِبُ في الحالِ إلى الأرضِ الّتي كانوا يَقصِدونَها.</span>
</div>
</div>

## الناس يطلبون يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="مَرْقُس 6:45; ﻣﺘﻰ 14:22; يُوحَنّا 6:2"> وفي الغَدِ، تذكَّرَ الجَمعُ الّذي بَقِيَ على الشّاطِئِ الآخَرِ أنَّ قارِبًا واحدًا كانَ هُناكَ، وأنَّ يَسوعَ ما صَعِدَ إلَيهِ معَ تلاميذِهِ، بل ذهَبَ التَّلاميذُ وحدَهُم.</span>
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 6:1; يُوحَنّا 6:11–12"> إلَّا أنَّ بَعضَ القواربِ جاءَت مِنْ طَبرِيَّةَ إلى الشَّاطئِ القَريبِ مِنَ المَوضِعِ الّذي أكلوا فيه الخُبزَ بَعدَ أنْ شكَرَ الرَّبَّ.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 6:17; لُوقا 8:40; مَرْقُس 6:53"> فلمَّا رأى الجَمعُ أنَّ يَسوعَ ما كانَ هُناكَ ولا تلاميذُهُ ركِبوا القوارِبَ وذَهَبوا إلى كَفْرَناحومَ يَبحَثونَ عَنهُ.</span>
</div>
</div>

## يسوع خبز الحياة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 1:38–39; ﻣﺘﻰ 23:7"> فلمَّا وجَدُوهُ على الشَّاطئِ الآخَرِ قالوا لَه: ”متى وصَلْتَ إلى هُنا، يا مُعَلِّمُ؟“</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="يَعقُوب 4:3–4; فِيلِبِّي 3:19; فِيلِبِّي 2:21"> فأجابَهُم يَسوعُ: ”الحقَّ الحقَّ أقولُ لكُم: أنتُم تَطلُبوني لا لأنَّكُم رأيتُمُ الآياتِ، بل لأنَّكُم أكلتُمُ الخُبزَ وشَبِعتُم.</span>
<span className="v" id="v27" data-v="27" data-xref="كُولُوسِي 3:2; إشَعْياء 55:2; يُوحَنّا 6:54"> لا تَعمَلوا لِلقوتِ الفاني، بل اَعمَلوا لِلقوتِ الباقي لِلحياةِ الأبدِيَّةِ. هذا القوتُ يَهبُهُ لكُمُ اَبنُ الإنسانِ، لأنَّ اللهَ الآبَ خَتَمَهُ بِخَتْمِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="اعمال الرسل 16:30; اعمال الرسل 9:6; ﻣﺘﻰ 19:16"> قالوا لَه: ”كيفَ نَعمَلُ ما يُريدُه اللهُ؟“</span>
<span className="v" id="v29" data-v="29" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:23; اعمال الرسل 16:31; يُوحَنّا 3:36"> فأجابَهُم: ”أنْ تُؤمِنوا بِمَنْ أرسَلَه: هذا ما يُريدُه اللهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 12:37; يُوحَنّا 10:38; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:22"> فقَالوا لَه: ”أرِنا آيةً حتّى نُؤمِنَ بِكَ! ماذا تَقدِرُ أنتَ أنْ تَعمَلَ؟</span>
<span className="v" id="v31" data-v="31" data-xref="نَحَمْيا 9:15; المَزامِير 105:40; المَزامِير 78:24–25"> آباؤُنا أكَلُوا المَنَّ في البرِّيَّةِ، كما جاءَ في الكِتابِ: ”أعطاهُم خُبزًا مِنَ السَّماءِ ليأكُلوا“.</span>
<span className="v" id="v32" data-v="32" data-xref="يُوحَنّا 15:1; يُوحَنّا 6:58; الخُرُوج 16:4"> فأجابَهُم يَسوعُ: ”الحقَّ الحقَّ أقولُ لكُم: ما أعطاكُم موسى الخُبزَ مِنَ السَّماءِ. أبي وحدَهُ يُعطيكُمُ الخُبزَ الحَقيقيَّ مِنَ السَّماءِ.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 8:42; يُوحَنّا 17:8; يُوحَنّا 3:13"> لأنَّ خُبزَ اللهِ هوَ الّذي يَنزِلُ مِنَ السَّماءِ ويُعطي العالَمَ الحياةَ“.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="يُوحَنّا 4:15; يُوحَنّا 6:26; المَزامِير 4:6"> قالوا لَه: ”يا سيِّدُ، أعطِنا كُلَّ حينٍ مِنْ هذا الخُبزِ“.</span>
<span className="v" id="v35" data-v="35" data-xref="يُوحَنّا 6:48–58; رُؤيا يُوحَنّا 22:17; يُوحَنّا 6:41"> فقَالَ لهُم يَسوعُ:<br />”أنا هوَ خُبزُ الحياةِ.<br />مَنْ جاءَ إِليَّ لا يَجوعُ،<br />ومَنْ آمنَ بي لا يعطَشُ أبدًا.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:8–9; لُوقا 16:31; يُوحَنّا 12:37"> لكنِّي قُلتُ لكُم: تَرَوني ولا تُؤمِنونَ.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 11:28; يُوحَنّا 6:39; يُوحَنّا 17:24"> مَنْ وهَبَهُ الآبُ لي يَجيءُ إليَّ،<br />ومَنْ جاءَ إِليَّ لا أُبعِدُهُ عَنِّي.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 5:30; يُوحَنّا 4:34; يُوحَنّا 3:31"> فما نزَلْتُ مِنَ السَّماءِ لأعمَلَ ما أُريدُهُ أنا،<br />بل ما يُريدُهُ الّذي أرسَلَني.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 17:12; يُوحَنّا 6:40; يُوحَنّا 18:9"> وما يُريدُهُ الّذي أرسَلَني<br />هوَ أنْ لا أخسَرَ أحَدًا مِمَّنْ وهبَهُم لي،<br />بل أُقيمُهُم كُلَّهُم في اليومِ الآخِرِ.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 6:54; يُوحَنّا 5:24; ﺭﻭﻣﻴﺔ 6:23"> وهذا ما يُريدُهُ أبي:<br />إنَّ كُلَّ مَنْ رأى الابنَ وآمنَ بِه نالَ الحياةَ الأبديَّةَ،<br />وأنا أُقيمُهُ في اليومِ الآخِرِ“.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="يُوحَنّا 6:51–52; يُوحَنّا 6:48; يُوحَنّا 6:33"> فتَذَمَّرَ اليَهودُ على يَسوعَ لأنَّهُ قالَ: ”أنا الخُبزُ الّذي نزَلَ مِنَ السَّماءِ“،</span>
<span className="v" id="v42" data-v="42" data-xref="لُوقا 4:22; مَرْقُس 6:3; ﻣﺘﻰ 13:55–56"> وقالوا: ”أما هوَ يَسوعُ اَبنُ يوسُفَ؟ نَحنُ نَعرِفُ أباهُ وأُمَّهُ، فكيفَ يَقولُ الآنَ إنَّهُ نزَلَ مِنَ السَّماءِ؟“</span>
<span className="v" id="v43" data-v="43" data-xref="العِبرانِيّين 4:13; يُوحَنّا 6:64; مَرْقُس 9:33"> فأجابَهُم يَسوعُ:<br />”لا تتَذَمَّروا فيما بَينَكُم.</span>
</div>
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="يُوحَنّا 6:65; يُوحَنّا 6:39–40; ﻣﺘﻰ 11:25–27"> ما مِنْ أحدٍ يَجيءُ إليَّ<br />إلَّا إذا اَجتَذَبَهُ الآبُ الّذي أرسَلَني،<br />وأنا أُقيمُهُ في اليومِ الآخِرِ.</span>
</div>
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="إشَعْياء 54:13; إرْمِيا 31:33–34; العِبرانِيّين 10:16"> ويكونونَ كُلُّهُم تلاميذَ اللهِ،<br />كما كتَبَ الأنبياءُ.<br />فمن سَمِعَ الآبَ وتَعَلَّمَ مِنهُ جاءَ إِليَّ،</span>

<div className="bp">

</div>
