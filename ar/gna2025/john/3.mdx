## يسوع ونيقوديموس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 19:39; يُوحَنّا 7:47–50; يُوحَنّا 3:10"> وكانَ رَجُلٌ فَرِّيسِيٌّ مِنْ رُؤساءِ اليَهودِ اَسمُهُ نيقوديموسُ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 5:36; اعمال الرسل 10:38; اعمال الرسل 2:22"> فجاءَ إلى يَسوعَ ليلًا وقالَ لَهُ: ”يا مُعَلِّمُ، نَحنُ نَعرِفُ أنَّ اللهَ أرسَلَكَ مُعَلِّمًا، فلا أحَدٌ يَقدِرُ أنْ يَصنَعَ ما تَصنَعُهُ مِنَ الآياتِ إلَّا إذا كانَ اللهُ معَهُ“.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:17; يُوحَنّا 3:5–6; رسالَةُ بُطرُسَ الأُولَى 1:23–25"> فأجابَهُ يَسوعُ: ”الحقَّ الحقَّ أقولُ لكَ: ما مِنْ أحَدٍ يُمكنُهُ أنْ يَرى مَلكوتَ اللهِ إلَّا إذا وُلِدَ ثانِيةً“.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:18; يُوحَنّا 3:3"> فقَالَ نيقوديموسُ: ”كيفَ يولَدُ الإنسانُ وهوَ كبيرٌ في السِّنِّ؟ أيقدِرُ أنْ يَدخُلَ بَطْنَ أُمِّهِ ثانِيةً ثُمَّ يُولَدُ؟“</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 36:25–27; أفَسُس 5:26; اعمال الرسل 2:38"> أجابَهُ يَسوعُ: ”الحقَّ الحقَّ أقولُ لكَ: ما مِنْ أحَدٍ يُمكِنُهُ أنْ يَدخُلَ مَلكوتَ اللهِ إلَّا إذا وُلِدَ مِنَ الماءِ والرُّوحِ،</span>
<span className="v" id="v6" data-v="6" data-xref="غَلاطِيَّة 5:16–21; حِزْقِيال 36:26–27; ﺭﻭﻣﻴﺔ 8:4–9"> لأنَّ مَولُودَ الجسَدِ يكونُ جَسدًا ومَولُودَ الرُّوحِ يكونُ رُوحًا.</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 3:3; أفَسُس 4:22–24; ﺭﻭﻣﻴﺔ 12:1–2"> لا تَتعجَّبْ مِنْ قولي لكَ: يَجبُ علَيكُم أنْ تُولَدوا ثانِيةً.</span>
<span className="v" id="v8" data-v="8" data-xref="الجامِعَة 11:4–5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:11; اعمال الرسل 2:2"> فالرّيحُ تَهبُّ حيثُ تَشاءُ، فتَسمَعُ صوتَها ولا تَعرِفُ مِنْ أينَ تَجيءُ وإلى أينَ تَذهَبُ: هكذا كُلُّ مَنْ يُولَدُ مِنَ الرُّوحِ“.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 6:60; لُوقا 1:34; إشَعْياء 42:16"> فقَالَ نيقوديموسُ: ”كيفَ يكونُ هذا؟“</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 22:29; إرْمِيا 8:8–9; حِزْقِيال 11:19"> أجابَهُ يَسوعُ: ”أنتَ مُعَلِّمٌ في إِسرائيلَ ولا تَعرِفُ؟</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 8:38; رسالَةُ يُوحَنّا الأُولَى 5:6–12; يُوحَنّا 12:49"> الحقَّ الحقَّ أقولُ لكَ: نَحنُ نتكَلَّمُ بِما نَعرِفُ، ونَشهَدُ بِما رأينا ولكنَّكُم لا تَقبَلونَ شَهادَتَنا.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:1–2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:7–9; العِبرانِيّين 5:11"> فإذا كُنتُم لا تُصدِّقونَ ما أُخبِرُكُم عَن أمورِ الدُّنيا، فكيفَ تُصدِّقونَ إذا أخبَرتُكُم عَنْ أمورِ السَّماءِ؟</span>
<span className="v" id="v13" data-v="13" data-xref="الأمثال 30:4; يُوحَنّا 6:38; اعمال الرسل 2:34"> ما صَعِدَ أحَدٌ إلى السَّماءِ إلَّا اَبنُ الإنسانِ الّذي نزَلَ مِنَ السَّماءِ.</span>
<span className="v" id="v14" data-v="14" data-xref="العَدَد 21:7–9; يُوحَنّا 8:28; يُوحَنّا 12:32–34"> وكما رفَعَ موسى الحَيَّةَ في البرِّيَّةِ، فكذلِكَ يَجبُ أنْ يُرفَعَ اَبنُ الإنسانِ.</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 3:16; يُوحَنّا 6:40; يُوحَنّا 12:44–46"> لينالَ كُلُّ مَنْ يُؤمِنُ بِه الحياةَ الأبدِيَّةَ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 5:8; رسالَةُ يُوحَنّا الأُولَى 4:9–10; يُوحَنّا 3:15"> هكذا أحبَّ اللهُ العالَمَ حتّى وهَبَ اَبنَهُ الأوحَدَ، فَلا يَهلِكَ كُلُّ مَنْ يُؤمِنُ بِه، بل تكونُ لَهُ الحياةُ الأبدِيَّةُ.</span>
<span className="v" id="v17" data-v="17" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:14; لُوقا 19:10; يُوحَنّا 12:47–48"> واللهُ أرسَلَ اَبنَهُ إلى العالَمِ لا ليدينَ العالَمَ، بل ليُخلِّصَ بِه العالَمَ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="مَرْقُس 16:16; رسالَةُ يُوحَنّا الأُولَى 5:10; ﺭﻭﻣﻴﺔ 8:1"> فمَنْ يُؤمِنُ بالاَبنِ لا يُدانُ. ومَنْ لا يُؤمِنُ بِه دِينَ، لأنَّهُ ما آمَنَ باَبنِ اللهِ الأوحدِ.</span>
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 8:12; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:12; يُوحَنّا 9:39–41"> وهذِهِ الدَّينونَةُ هيَ أنَّ النُّورَ جاءَ إلى العالَمِ، فأحَبَّ النَّاسُ الظَّلامَ بدلًا مِنَ النُّورِ لأنَّهُم يَعمَلونَ الشَّرَّ.</span>
<span className="v" id="v20" data-v="20" data-xref="أفَسُس 5:11–13; يُوحَنّا 7:7; أيُّوب 24:13–17"> فمَنْ يَعمَلُ الشَّرَّ يكرَهُ النُّورَ، فلا يَخرُجُ إلى النُّورِ لِئَلَّا تَنفَضِحَ أعمالُهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="رسالَةُ يُوحَنّا الأُولَى 1:6; المَزامِير 139:23–24; رسالَةُ يُوحَنّا الثّالثَة 1:11"> وأمَّا مَنْ يَعمَلُ لِلحقِّ، فيخرُجُ إلى النُّورِ، حتّى يرى النَّاسُ أنَّ أعمالَهُ كانَت حسَبَ مشيئةِ اللهِ“.</span>
</div>

## يسوع ويوحنا المعمدان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 4:1–3; يُوحَنّا 3:26; يُوحَنّا 2:13"> ثُمَّ جاءَ يَسوعُ وتلاميذُهُ إلى بلادِ اليهودِيَّةِ، فأقامَ فيها مَعهُم وأخَذَ يُعَمِّدُ.</span>
<span className="v" id="v23" data-v="23" data-xref="مَرْقُس 1:4–5; حِزْقِيال 43:2; ﻣﺘﻰ 3:5–6"> وكانَ يوحنَّا يُعَمِّدُ أيضًا في عينِ نونٍ، بِالقُربِ مِنْ ساليمَ، لِكَثرةِ الماءِ فيها. وكانَ النَّاسُ يَجيئونَ ويتَعمَّدونَ،</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 4:12; ﻣﺘﻰ 14:3; مَرْقُس 6:17"> وذلِكَ قَبلَ أنْ يُلقى يوحنَّا في السِّجنِ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 2:6; ﻣﺘﻰ 3:11; العِبرانِيّين 6:2"> وقامَ جِدالٌ بَينَ تلاميذِ يوحنَّا وواحدٍ مِنَ اليَهودِ في مسألَةِ الطَّهارةِ.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 1:7; يُوحَنّا 1:15; يُوحَنّا 12:19"> فجاؤُوا إلى يوحنَّا وقالوا لَه: ”يا مُعَلِّمُ، ها هوَ الرَّجُلُ الّذي كان معَكَ في عَبرِ الأُردُنِّ وشَهِدْتَ لَه، يُعَمِّدُ هُنا، وجميعُ النَّاسِ يَجيئونَ إلَيهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="يَعقُوب 1:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:10"> فأجابَهُم يوحنَّا: ”لا يأخُذُ أحدٌ شيئًا إلَّا إذا أعطَتْهُ إيّاهُ السماءُ.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 1:20; يُوحَنّا 1:23; ﻣﺘﻰ 3:11–12"> أنتُم أنفُسُكُم تَشهَدونَ بأَنِّي قُلتُ: ما أنا المَسيحُ، بل رَسولٌ قُدَّامَهُ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 9:15; هُوشَع 2:19; إشَعْياء 54:5"> مَنْ لَه العَروسُ، فهوَ العَريسُ. وأمَّا صَديقُ العَريسِ، فيَقِفُ بجانِبِه يُصغي فَرِحًا لِهُتافِ العَريسِ. ومِثلُ هذا الفرَحِ فَرَحي، وهوَ الآنَ كامِلٌ.</span>
<span className="v" id="v30" data-v="30" data-xref="كُولُوسِي 1:18; المَزامِير 72:17–19; العِبرانِيّين 3:2–6"> لَه هوَ أنْ يزيدَ، ولي أنا أنْ أنقُصَ.</span>
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 8:23; ﻣﺘﻰ 28:18; ﺭﻭﻣﻴﺔ 9:5"> مَنْ جاءَ مِنْ فوقُ، فهوَ فوقَ النَّاسِ جميعًا. ومَنْ كانَ مِنَ الأرضِ، فهوَ أرضِيٌّ وبِكلامِ أهلِ الأرضِ يَتكَلَّمُ. مَنْ جاءَ مِنَ السَّماءِ، فهوَ فوقَ النَّاسِ جميعًا،</span>
<span className="v" id="v32" data-v="32" data-xref="يُوحَنّا 3:11; يُوحَنّا 15:15; يُوحَنّا 1:11"> يَشهَدُ بِما رأى وسَمِعَ ولا أحَدَ يَقبَلُ شَهادَتَهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="العِبرانِيّين 6:17; أفَسُس 1:13; تِيطُس 1:1–2"> مَنْ قَبِلَ شَهادَتَهُ شَهِدَ أنَّ اللهَ صادِقٌ.</span>
<span className="v" id="v34" data-v="34" data-xref="اعمال الرسل 10:38; ﻣﺘﻰ 12:18; يُوحَنّا 15:26"> فمَنْ أرسَلَهُ اللهُ يتكَلَّمُ بِكلامِ اللهِ، لأنَّ اللهَ يَهَبُ الرُّوحَ بِغيرِ حِسابٍ.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 28:18; يُوحَنّا 5:22; يُوحَنّا 5:20"> الآبُ يُحبُّ الاَبنَ فجَعلَ كُلَّ شيءٍ في يدِهِ.</span>
<span className="v" id="v36" data-v="36" data-xref="يُوحَنّا 5:24; يُوحَنّا 1:12; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:9"> مَنْ يُؤمِنُ بالابنِ، فَلهُ الحياةُ الأبديَّةُ. ومَنْ لا يُؤمنُ بِالابنِ، فلا يرى الحياةَ بل يَحِلُّ علَيهِ غَضَبُ اللهِ“.</span>
</div>
