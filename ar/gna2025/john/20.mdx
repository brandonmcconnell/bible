## قيامة يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 16:9; ﻣﺘﻰ 27:60; يُوحَنّا 19:25"> ويومَ الأحَدِ جاءَت مَريَمُ المَجدَلِيَّةُ إلى القَبرِ باكرًا، وكانَ ظلامٌ بَعدُ فرَأتِ الحجَرَ مَرفوعًا عَن القبرِ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 13:23; يُوحَنّا 20:13; يُوحَنّا 21:24"> فأقبَلت مُسرِعَةً إلى سِمعانَ بُطرُسَ والتِّلميذِ الآخرِ الّذي أحَبَّهُ يَسوعُ، وقالَت لهُما: ”أخذوا الرَّبَّ مِنَ القَبرِ، ولا نَعرِفُ أينَ وَضَعوهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="لُوقا 24:12"> فخرَجَ بُطرُسُ والتِّلميذُ الآخَرُ إلى القَبرِ،</span>
<span className="v" id="v4" data-v="4" data-xref="صموئيلَ الثّانِي 18:23; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:24"> يُسرِعانِ السَّيرَ معًا. ولكِنَّ التِّلميذَ الآخرَ سبَقَ بُطرُسَ، فوَصَلَ قَبلَهُ إلى القَبرِ.</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 19:40; يُوحَنّا 20:11; يُوحَنّا 11:44"> واَنحَنى مِنْ دُونِ أنْ يَدخُلَ، فرَأى الأكفانَ على الأرضِ.</span>
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 18:17; يُوحَنّا 21:7; ﻣﺘﻰ 16:15–16"> ولَحِقَهُ سِمعانُ بُطرُسُ، فدَخَلَ القَبرَ. ورأى الأكفانَ على الأرضِ،</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 11:44"> والمِنديلَ الّذي كانَ على رأسِ يَسوعَ مَلفوفًا في مكانٍ على حِدَةٍ، لا مُلقًى معَ الأكفانِ.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 20:25; يُوحَنّا 20:4; يُوحَنّا 1:50"> ودخَلَ التِّلميذُ الآخَرُ الّذي سَبَقَ بُطرُسَ إلى القَبرِ، فرأى وآمنَ،</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 16:10; إشَعْياء 25:8; إشَعْياء 53:10–12"> لأنَّهُما كانا بَعدُ لا يَفهَمانِ ما جاءَ في الكِتابِ وهوَ أنَّ يَسوعَ يَجبُ أنْ يقومَ مِنْ بَينِ الأمواتِ.</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 7:53; يُوحَنّا 16:32"> ثُمَّ رجَعَ التِّلميذانِ إلى مَنزِلِهِما.</span>
</div>
</div>

## يسوع يظهر لمريم المجدلية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 20:5"> أمَّا مَريَمُ المَجدليَّةُ، فوَقَفَت عِندَ القَبرِ تَبكي. واَنحَنَت نحوَ القَبرِ وهيَ تَبكي،</span>
<span className="v" id="v12" data-v="12" data-xref="لُوقا 24:3–7; رُؤيا يُوحَنّا 3:4; مَرْقُس 16:5–6"> فرأَت مَلاكَينِ في ثيابٍ بَيضاءَ جالِسَينِ حَيثُ كانَ جسَدُ يَسوعَ، أحدُهُما عِندَ الرأسِ والآخَرُ عِندَ القَدَمَينِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 20:15; يُوحَنّا 2:4; صَمُوئيلَ الأوَّلُ 1:8"> فقَالَ لها الملاكانِ: ”لماذا تَبكِينَ، يا اَمرأةُ؟“ أجابَت: ”أخَذوا رَبِّي ولا أعرِفُ أينَ وضَعوهُ!“</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 24:16; يُوحَنّا 21:4; ﻣﺘﻰ 28:9"> قالَت هذا واَلتَفتَت وراءَها فرَأت يَسوعَ واقِفًا، وما عرَفَت أنَّهُ يَسوعُ.</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 20:13; يُوحَنّا 18:7; يُوحَنّا 1:38"> فقَالَ لها يَسوعُ: ”لماذا تَبكينَ، يا اَمرأةُ؟ ومَنْ تَطلُبينَ؟“ فظَنَّت أنَّهُ البُستانيُّ، فقَالَت لَه: ”إذا كُنتَ أنتَ أخَذْتَهُ يا سيِّدي، فَقُلْ لي أينَ وضَعتَهُ حتّى آخُذَهُ“.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 10:3; يُوحَنّا 1:38; اعمال الرسل 10:3"> فقَالَ لها يَسوعُ: ”يا مَريَمُ“. فعَرَفَتهُ وقالَت لَه بالعِبريَّةِ: ”ربُّوني!“ (أي يا مُعَلِّمُ).</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 16:28; أفَسُس 4:8–10; غَلاطِيَّة 4:6–7"> فقَالَ لها يَسوعُ: ”لا تُمسِكيني، لأنِّي ما صَعِدتُ بَعدُ إلى الآبِ، بلِ اَذهَبي إلى إخوَتي وقولي لهُم: أنا صاعِدٌ إلى أبي وأبيكُم، إلهي وإلهِكم“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 24:10; ﻣﺘﻰ 28:10; يُوحَنّا 20:1"> فرَجَعَت مَريَمُ المَجدلِيَّةُ وأخبَرَتِ التَّلاميذَ بأنَّها رأتِ الرَّبَّ وأنَّهُ قالَ لها هذا الكلامَ.</span>
</div>
</div>

## يسوع يظهر لتلاميذه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 20:26; يُوحَنّا 14:27; يُوحَنّا 20:21"> وفي مَساءِ ذلِكَ الأحدِ، كانَ التَّلاميذُ مُجتمِعينَ والأبوابُ مُقفَلةٌ خَوفًا مِنَ اليَهودِ. فجاءَ يَسوعُ ووقَفَ بَينَهُم وقالَ: ”سلامٌ علَيكُم“.</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 16:22; يُوحَنّا 16:20; يُوحَنّا 20:27"> وأراهُم يَدَيهِ وجَنبَهُ، ففَرِحَ التَّلاميذُ عِندَما شاهَدوا الرَّبَّ.</span>
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 17:18–19; يُوحَنّا 13:20; يُوحَنّا 3:17"> فقَالَ لهُم يَسوعُ ثانيَةً: ”سلامٌ علَيكُم! كما أرسَلَني الآبُ أُرسِلُكُم أنا“.</span>
<span className="v" id="v22" data-v="22" data-xref="أيُّوب 33:4; التَّكوين 2:7; اعمال الرسل 8:15"> قالَ هذا ونَفَخَ في وجوهِهِم وقالَ لهُم: ”خُذوا الرُّوحَ القُدُسَ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 18:18; ﻣﺘﻰ 16:19; مَرْقُس 2:5–10"> مَنْ غَفَرْتُم لَه خطاياهُ تُغفَرُ لَه، ومَنْ مَنَعْتُم عَنهُ الغُفرانَ يُمنَعُ عَنهُ“.</span>
</div>
</div>

## يسوع وتوما

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 11:16; يُوحَنّا 6:66–67; العِبرانِيّين 10:25"> وكانَ توما، أحدُ التَّلاميذِ الاثني عشَرَ المُلقَّبُ بالتَّوأمِ، غائِبًا عِندَما جاءَ يَسوعُ.</span>
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 6:30; مَرْقُس 16:11; العِبرانِيّين 3:12"> فقَالَ لَه التَّلاميذُ: ”رأينا الرَّبَّ!“ فأجابَهُم: ”لا أُؤمِنُ إلَّا إذا رَأيتُ أثَرَ المَساميرِ في يَدَيهِ، ووَضَعْتُ إصبَعي في مكانِ المساميرِ ويَدي في جَنبِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 20:19; إشَعْياء 54:10; يُوحَنّا 20:21"> وبَعدَ ثمانيةِ أيّامٍ اَجتَمَعَ التَّلاميذُ في البَيتِ مرَّةً أُخرى، وتوما مَعهُم، فجاءَ يَسوعُ والأبوابُ مُقفَلةٌ، ووقَفَ بَينَهُم وقالَ: ”سلامٌ علَيكُم“.</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 20:25; رسالَةُ يُوحَنّا الأُولَى 1:1–2; يُوحَنّا 20:20"> ثُمَّ قالَ لِتوما: ”هاتِ إصبَعَكَ إلى هُنا واَنظُرْ يَدَيَّ، وهاتِ يدَكَ وضَعْها في جَنبي. ولا تَشُكَّ بَعدَ الآنَ، بل آمِنْ!“</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 1:1; إشَعْياء 9:6; إشَعْياء 7:14"> فأجابَ توما: ”رَبِّي وإلهي!“</span>
<span className="v" id="v29" data-v="29" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:7; العِبرانِيّين 11:1"> فقَالَ لَه يَسوعُ: ”آمَنْتَ يا توما، لأنَّكَ رأيتَني. هَنيئًا لِمَنْ آمَنَ وما رأى“.</span>
</div>
</div>

## غاية هذا الكتاب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 21:25; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:15–17; ﺭﻭﻣﻴﺔ 15:4"> وصنَعَ يَسوعُ أمامَ تلاميذِهِ آياتٍ أُخرى غَيرَ مُدوَّنَةٍ في هذا الكِتابِ.</span>
<span className="v" id="v31" data-v="31" data-xref="اعمال الرسل 10:43; يُوحَنّا 3:15–16; رسالَةُ يُوحَنّا الأُولَى 5:1"> أمَّا الآياتُ المُدوَّنَةُ هُنا، فهيَ لتُؤمِنوا بأنَّ يَسوعَ هوَ المَسيحُ اَبنُ اللهِ. فإذا آمنتُم نِلتُم باَسْمِهِ الحياةَ.</span>
</div>
