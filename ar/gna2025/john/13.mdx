## يسوع يغسل أرجل التَّلاميذ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 12:23; رسالَةُ يُوحَنّا الأُولَى 4:19; يُوحَنّا 13:34"> وكانَ يَسوعُ يَعرِفُ، قَبلَ عيدِ الفِصحِ، أنَّ ساعتَهُ جاءَت ليَنتَقِلَ مِنْ هذا العالَمِ إلى الآبِ، وهوَ الّذي أحبَّ أخِصَّاءَهُ الّذينَ هُم في العالَمِ، أحبَّهُم مُنتَهى الحُبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="يَعقُوب 1:13–17; اعمال الرسل 5:3; يُوحَنّا 13:26–27"> وجلَسَ لِلعَشاءِ معَ تلاميذِهِ. وكانَ إبليسُ وسوَسَ إلى يَهوذا بنِ سِمعانَ الأسخَريوطيِّ أنْ يُسلَّمَ يَسوعَ.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 3:35; ﻣﺘﻰ 11:27; العِبرانِيّين 1:2"> وكانَ يَسوعُ يَعرِفُ أنَّ الآبَ جعَلَ في يدَيهِ كُلَّ شيءٍ، وأنَّهُ جاءَ مِنْ عِندِ اللهِ وإلى اللهِ يَعودُ.</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 22:27; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:9; لُوقا 12:37"> فقامَ عَنِ العَشاءِ وخلَعَ ثوبَهُ وأخَذَ مِنشَفَةً واَتَّزرَ بِها،</span>
<span className="v" id="v5" data-v="5" data-xref="لُوقا 7:44; المَزامِير 51:2; التَّكوين 18:4"> ثُمَّ صَبَّ ماءً في مَغسِلَةٍ وبَدأ يَغسِلُ أرجُلَ التَّلاميذِ ويَمسَحُها بالمِنشَفَةِ الّتي اَتَّزرَ بِها.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 3:11–14; يُوحَنّا 1:27; لُوقا 5:8"> فلمَّا دَنا مِنْ سِمعانَ بُطرُسَ، قالَ لَه سِمعانُ: ”يا ربُّ، أأنتَ تَغسِلُ رِجليَّ؟“</span>
<span className="v" id="v7" data-v="7" data-xref="دانيال 12:8; يُوحَنّا 12:16; يُوحَنّا 14:26"> فأجابَه يَسوعُ: ”أنتَ الآنَ لا تَفهَمُ ما أنا أعمَلُ، ولكنَّكَ ستَفهَمُهُ فيما بَعدُ“.</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 36:25; اعمال الرسل 22:16; تِيطُس 3:5"> فقَالَ لَه بُطرُسُ: ”لن تَغسِلَ رِجليَّ أبدًا“. أجابَه يَسوعُ: ”إنْ كُنتُ لا أغسِلُكَ، فلا نَصيبَ لكَ مَعي“.</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 51:7; المَزامِير 51:2; رسالَةُ بُطرُسَ الأُولَى 3:21"> فقَالَ لَه سِمعانُ بُطرُسُ: ”إذًا يا ربُّ، لا تَغسِلْ رِجليَّ وحدَهُما، بلِ اَغسِلْ مَعَهُما يدَيَّ ورَأسي“.</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 15:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:17"> فقَالَ لَه يَسوعُ: ”مَنِ اَغتسَلَ كانَ طاهِرًا كُلُّهُ، فلا يَحتاجُ إلَّا إلى غَسلِ رِجلَيهِ. أنتُم طاهِرونَ، ولكنْ ما كُلُّكُم طاهِرونَ“.</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 13:26; يُوحَنّا 13:2; يُوحَنّا 13:21"> قالَ يَسوعُ ”ما كُلُّكُم طاهِرونَ“، لأنَّهُ كانَ يَعرِفُ مَنْ سيُسْلِمُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 13:7; يُوحَنّا 13:4; حِزْقِيال 24:24"> فلمَّا غسَلَ أرجُلَهُم ولبِسَ ثوبَهُ وعادَ إلى المائِدَةِ قالَ لهُم: ”أتَفهَمونَ ما عمِلتُهُ لكُم؟</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 6:46; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:3; يُوحَنّا 11:28"> أنتُم تَدعونَني مُعَلِّمًا وربًّا، وحسَنًا تَفعَلونَ لأنِّي هكذا أنا.</span>
<span className="v" id="v14" data-v="14" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:5; اعمال الرسل 20:35; ﺭﻭﻣﻴﺔ 12:16"> وإذا كُنتُ أنا الرّبُّ والمُعَلِّمُ غَسَلتُ أرجُلَكُم، فيَجِبُ علَيكُم أنتُم أيضًا أنْ يَغسِلَ بَعضُكُم أرجُلَ بَعضٍ.</span>
<span className="v" id="v15" data-v="15" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:6; ﻣﺘﻰ 11:29; رسالَةُ بُطرُسَ الأُولَى 3:17–18"> وأنا أعطيتُكُم ما تَقتَدُونَ بِه، فتَعمَلوا ما عَمِلتُهُ لكُم.</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 6:40; يُوحَنّا 15:20; ﻣﺘﻰ 10:24–25"> الحقَّ الحقَّ أقولُ لكُم: ما كانَ خادِمٌ أعظَمَ مِنْ سيِّدِهِ، ولا كانَ رَسولٌ أعظَمَ مِنَ الّذي أرسَلَهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="لُوقا 11:28; يَعقُوب 4:17; يَعقُوب 1:25"> والآنَ عَرَفْتُم هذِهِ الحَقيقَةَ، فهَنيئًا لكُم إذا عَمِلْتُم بِها.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 41:9; ﻣﺘﻰ 26:23; يُوحَنّا 17:12"> لا أقولُ هذا فيكُم كُلِّكُم، فأنا أعرِفُ الّذينَ اَخْتَرْتُهُم. ولكِنْ ما جاءَ في الكُتُبِ المُقَدَّسَةِ لا بُدَّ لَه أنْ يَتِمَّ، وهوَ: أنَّ الّذي أكَلَ خُبزي تَمَرَّدَ علَيَّ.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 43:10; يُوحَنّا 14:29; يُوحَنّا 16:4"> أُخبرُكُم بِهذا الآنَ قَبلَما يَحدُثُ، حتّى متى حَدَثَ تُؤمِنونَ بأنِّي أنا هوَ.</span>
<span className="v" id="v20" data-v="20" data-xref="لُوقا 10:16; لُوقا 9:48; ﻣﺘﻰ 25:40"> الحقَّ الحقَّ أقولُ لكُم: مَنْ قَبِلَ الّذينَ أُرسِلُهُم قَبِلَني. ومَنْ قَبِلَني قَبِلَ الّذي أرسَلَني“.</span>
</div>
</div>

## يسوع يُنبئ بخيانة يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 13:18; ﻣﺘﻰ 26:21; مَرْقُس 14:18"> وعِندَ هذا الكلامِ، اَضطرَبَت نَفْسُ يَسوعَ وقالَ عَلانيةً: ”الحقَّ الحقَّ أقولُ لكُم: واحدٌ مِنكُم سيُسْلِّمُني!“</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="لُوقا 22:23; ﻣﺘﻰ 26:22; مَرْقُس 14:19"> فنَظَرَ التَّلاميذُ بَعضُهُم إلى بَعضٍ حائِرينَ لا يَعرِفونَ مَنْ يَعني بِقولِهِ.</span>
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 19:26; يُوحَنّا 21:7; يُوحَنّا 21:20"> وكانَ أحدُ التَّلاميذِ، وهوَ الّذي يُحبِّهُ يَسوعُ، جالِسًا بِجانِبِهِ.</span>
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 12:17; لُوقا 1:22; اعمال الرسل 13:16"> فأومَأَ إلَيهِ سِمعانُ بُطرُسُ وقالَ لَه: ”سَلْهُ مَنْ يَعني بِقولِهِ“.</span>
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 21:20; التَّكوين 44:4–12; أسْتِير 7:5"> فمالَ التِّلميذُ على صَدرِ يَسوعَ وسألَهُ: ”مَنْ هوَ يا ربُّ؟“</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 26:23; يُوحَنّا 13:30; مَرْقُس 14:19–20"> فأجابَ يَسوعُ: ”هوَ الّذي أُناوِلُه اللُّقمَةَ الّتي أغْمِسُها!“ وغمَسَ يَسوعُ لُقمَةً ورَفَعَها وناوَلَ يَهوذا بنَ سِمعانَ الأسخَريوطيَّ.</span>
<span className="v" id="v27" data-v="27" data-xref="لُوقا 22:3; يُوحَنّا 13:2; الجامِعَة 9:3"> فلمَّا تناوَلَها دخَلَ الشَّيطانُ فيهِ. فقَالَ لَه يَسوعُ: ”إعمَلْ ما أنتَ تَعمَلُهُ ولا تُبطِئْ“.</span>
<span className="v" id="v28" data-v="28"> فما فَهِمَ أحَدٌ مِنَ الجالِسينَ إلى المائِدَةِ لِماذا قالَ لَه هذا الكلامَ.</span>
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 12:5–6; يُوحَنّا 13:1; أفَسُس 4:28"> وكانَ يَهوذا أمينَ الصُّندُوقِ، فظَنَّ بَعضُهُم أنَّ يَسوعَ أوصاهُ أنْ يَشتَرِيَ ما يَحتاجونَ إلَيهِ في العيدِ، أو أنْ يُعطِيَ الفُقراءَ شيئًا.</span>
<span className="v" id="v30" data-v="30" data-xref="إشَعْياء 59:7; أيُّوب 24:13–15; الأمثال 4:16"> وتناوَلَ يَهوذا اللُّقمَةَ وخرَجَ في الحالِ. وكانَ ليلًا.</span>
</div>
</div>

## الوصية الجديدة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 14:13; رسالَةُ بُطرُسَ الأُولَى 4:11; يُوحَنّا 7:39"> فلمَّا خرَجَ قالَ يَسوعُ: ”الآنَ تَمَجَّدَ اَبنُ الإنسانِ وتَمَجَّدَ اللهُ فيهِ.</span>
<span className="v" id="v32" data-v="32" data-xref="رُؤيا يُوحَنّا 22:1; إشَعْياء 53:10–12; يُوحَنّا 17:4–6"> وإذا كانَ اللهُ تَمَجَّدَ فيهِ، فإنَّ اللهَ سَيُمَجِّدُهُ في ذاتِهِ، وبَعدَ قليلٍ سَيُمَجِّدُهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 7:33–34; يُوحَنّا 8:21–24; يُوحَنّا 16:16–22"> يا أبنائي، سأبقى مَعكُم وقَتًا قَليلًا. ستَطلُبوني، ولكِنْ ما قُلتُهُ لِليَهودِ أقولُهُ لكُمُ الآنَ: حَيثُ أنا ذاهِبٌ لا تَقدِرونَ أنتُم أنْ تَجيئوا.</span>
<span className="v" id="v34" data-v="34" data-xref="أفَسُس 5:2; اللّاويِّين 19:18; يُوحَنّا 15:17"> أُعطيكُم وَصيَّةً جَديدةً: أحِبُّوا بَعضُكُم بَعضًا. ومِثلَما أنا أحبَبْتُكُم أحِبُّوا أنتُم بَعضُكُم بَعضًا</span>
<span className="v" id="v35" data-v="35" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:20–21; رسالَةُ يُوحَنّا الأُولَى 3:10–14; رسالَةُ يُوحَنّا الأُولَى 2:10"> فإذا أحبَبْتُم بَعضُكُم بَعضًا، يَعرِفُ النَّاسُ جميعًا أنَّكُم تلاميذي“.</span>
</div>

## يسوع يُنبئ بإنكار بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="رسالَةُ بُطرُسَ الثّانيَة 1:14; يُوحَنّا 13:33; يُوحَنّا 21:21–22"> فقَالَ لَه سِمعانُ بُطرُسُ: ”إلى أينَ أنتَ ذاهِبٌ يا ربُّ؟“ أجابَهُ يَسوعُ: ”حيثُ أنا ذاهبٌ لا تَقدِرُ الآنَ أنْ تَتبَعَني، ولكنَّكَ ستَتبَعُني يومًا“.</span>
<span className="v" id="v37" data-v="37" data-xref="اعمال الرسل 21:13; اعمال الرسل 20:24; مَرْقُس 14:27–31"> فقَالَ لَه بُطرُسُ: ”لِماذا لا أقدِرُ أنْ أتبَعَكَ الآنَ، يا ربُّ؟ أنا مُستَعِدٌّ أنْ أموتَ في سَبيلِكَ!“</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 22:34; ﻣﺘﻰ 26:34; مَرْقُس 14:30"> أجابَهُ يَسوعُ: ”أمُستَعِدٌّ أنتَ أنْ تَموتَ في سَبيلي؟ الحقَّ الحقَّ أقولُ لكَ: لا يَصيحُ الدّيكُ إلَّا وأنكَرتَني ثلاثَ مرَّاتٍ“.</span>
</div>
