## كلمة اللهِ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 1:1; يُوحَنّا 17:5; رُؤيا يُوحَنّا 19:13"> في البَدْءِ كانَ الكَلِمَةُ،<br />والكَلِمَةُ كانَ عِندَ اللهِ،<br />وكانَ الكَلِمَةُ اللهَ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2"> هوَ في البَدْءِ كانَ عِندَ اللهِ.</span>

<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="كُولُوسِي 1:16–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:6; رُؤيا يُوحَنّا 4:11"> بِه كانَ كُلُّ شيءٍ،<br />وبِغَيرِهِ ما كانَ شيءٌ مِمَّا كانَ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 8:12; يُوحَنّا 12:46; يُوحَنّا 5:26"> فيهِ كانَتِ الحياةُ،<br />وحياتُهُ كانَت نُورَ النَّاسِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 3:19–20; يُوحَنّا 12:36–40; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14"> والنُّورُ يُشرِقُ في الظُّلْمَةِ،<br />والظُّلْمَةُ لا تَقوى علَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="مَلاخِي 3:1; يُوحَنّا 3:28; إشَعْياء 40:3–5"> ظهَرَ رَسولٌ مِنَ اللهِ اَسمُهُ يوحنَّا.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 19:4; يُوحَنّا 1:36; يُوحَنّا 5:33–35"> جاءَ يَشهَدُ لِلنُّورِ حتّى يُؤمنَ النَّاسُ على يدهِ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 1:20; يُوحَنّا 3:28; اعمال الرسل 19:4"> ما كانَ هوَ النُّورَ،<br />بل شاهدًا لِلنُّورِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 12:46; يُوحَنّا 1:4; رسالَةُ يُوحَنّا الأُولَى 2:8"> الكَلِمَةُ هوَ النُّورُ الحَقُّ،<br />جاءَ إلى العالَمِ لِيُنيرَ كُلَّ إنسانٍ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 17:25; رسالَةُ يُوحَنّا الأُولَى 3:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:8"> وكانَ في العالَمِ،<br />وبِه كانَ العالَمُ،<br />وما عَرَفَهُ العالَمُ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="لُوقا 19:14; إشَعْياء 53:2–3; يُوحَنّا 3:32"> إلى بَيتِهِ جاءَ،<br />فما قَبِلَه أهلُ بَيتِه.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="غَلاطِيَّة 3:26; ﺭﻭﻣﻴﺔ 8:14; رسالَةُ يُوحَنّا الأُولَى 3:1"> أمَّا الّذينَ قَبِلوهُ، المُؤمِنونَ باَسمِهِ،<br />فأعطاهُم سُلطانًا أنْ يَصيروا أبناءَ اللهِ،</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:9; يَعقُوب 1:18; رسالَةُ بُطرُسَ الأُولَى 1:23"> وهُمُ الّذينَ وُلِدوا لا مِنْ دَمٍ<br />ولا مِنْ رَغبَةِ جسَدٍ<br />ولا مِنْ رَغبَةِ رَجُلٍ،<br />بل مِنَ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:9; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 3:16; إشَعْياء 40:5"> والكَلِمَةُ صارَ بشَرًا وعاشَ بَينَنا،<br />فرأَينا مَجدَهُ<br />مَجدًا يَفيضُ بِالنِّعمَةِ والحَقِّ،<br />نالَهُ مِنَ الآبِ،<br />كاَبْنٍ لَهُ أوحَدَ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 3:11; كُولُوسِي 1:17; لُوقا 3:16"> شَهِدَ لَهُ يوحنَّا فنادى:<br />”هذا هوَ الّذي قُلتُ فيهِ:<br />يَجيءُ بَعدي ويكونُ أعظَمَ مِنِّي،<br />لأنَّهُ كانَ قَبْلي“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="كُولُوسِي 1:19; ﺭﻭﻣﻴﺔ 5:2; أفَسُس 2:5–10"> مِنْ فيضِ نِعَمِهِ نِلْنا جميعًا<br />نِعمةً على نِعمةٍ،</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 6:14; يُوحَنّا 14:6; ﺭﻭﻣﻴﺔ 3:19–26"> لأنَّ اللهَ بِموسى أعطانا الشَّريعةَ،<br />وأمَّا بِيَسوعَ المسيحِ فوَهَبَنا النِّعمَةَ والحقَّ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="كُولُوسِي 1:15; يُوحَنّا 6:46; ﻣﺘﻰ 11:27"> ما مِنْ أحدٍ رأى اللهَ.<br />الإلهُ الأوحَدُ الّذي في حِضنِ الآبِ<br />هوَ الّذي أخبَرَ عَنهُ.</span>
</div>
</div>

## شهادة يوحنا المعمدان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="لُوقا 3:15–18; يُوحَنّا 10:24; اعمال الرسل 19:4"> هذِهِ شَهادةُ يوحنَّا، حينَ أرسَلَ إلَيهِ اليَهودُ مِنْ أُورُشليمَ كَهنَةً ولاويّينَ ليسألوهُ: ”مَنْ أنتَ؟“</span>
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 1:7–8; ﻣﺘﻰ 3:11–12; يُوحَنّا 3:28–36"> فاَعتَرَفَ وما أنكَرَ، إِعتَرَفَ قالَ: ”ما أنا المَسيحُ“.</span>
<span className="v" id="v21" data-v="21" data-xref="مَلاخِي 4:5; التَّثْنِيَة 18:15–18; ﻣﺘﻰ 11:14"> فقَالوا: ”مَنْ أنتَ، إذًا؟ هل أنتَ إيليَّا؟“ قالَ: ”ولا إيليَّا“. قالوا: ”هل أنتَ النَّبيُّ؟“ أجابَ: ”لا“.</span>
<span className="v" id="v22" data-v="22" data-xref="صموئيلَ الثّانِي 24:13"> فقَالوا لَه: ”مَنْ أنتَ، فنَحمِلَ الجَوابَ إلى الّذينَ أرسَلونا. ماذا تقولُ عَنْ نَفسِكَ؟“</span>
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 40:3–5; مَرْقُس 1:3; ﻣﺘﻰ 3:3"> قالَ: ”أنا، كما قالَ النَّبيُّ إشَعيا:<br />”صوتُ صارخٍ في البرِّيَّةِ:<br />قَوِّموا طريقَ الرَّبِّ“.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="لُوقا 11:53; ﻣﺘﻰ 23:13–15; لُوقا 11:39–44"> وكانَ بَينَهُم فَرِّيسيُّونَ،</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 18:18; التَّثْنِيَة 18:15; يُوحَنّا 1:20–22"> فقَالوا ليوحنَّا: ”كيفَ تُعَمِّدُ وما أنتَ المَسيحُ ولا إيليَّا ولا النَّبيُّ؟“</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 1:10–11; لُوقا 3:16; اعمال الرسل 1:5"> أجابَهُم: ”أنا أعمِّدُ بالماءِ، وبَينَكم مَنْ لا تَعرِفونَهُ.</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 1:30; مَرْقُس 1:7; ﻣﺘﻰ 3:11"> هوَ الّذي يَجيءُ بَعدي، ويكونُ أعظَمَ مِنِّي، وما أنا أهلٌ لأنْ أحُلَّ رِباطَ حِذائِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 10:40; يُوحَنّا 3:23; القُضاة 7:24"> جرى هذا كُلُّهُ في بَيت عَنيا، عَبرَ نهرِ الأُردُنِّ، حيثُ كانَ يوحنَّا يُعمِّدُ.</span>
</div>
</div>

## يسوع حمل اللهِ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 53:7; رسالَةُ يُوحَنّا الأُولَى 2:2; رسالَةُ يُوحَنّا الأُولَى 3:5"> وفي الغَدِ رأى يوحنَّا يَسوعَ مُقبِلًا إلَيهِ، فقَالَ: ”ها هوَ حمَلُ اللهِ الّذي يرفَعُ خَطيئةَ العالَمِ.</span>
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 1:27; يُوحَنّا 1:15; لُوقا 3:16"> هذا هوَ الّذي قُلتُ فيهِ: يَجيءُ بَعدي رَجُلٌ صارَ أعظَمَ مِنِّي، لأنَّهُ كانَ قَبلي.</span>
<span className="v" id="v31" data-v="31" data-xref="مَلاخِي 3:1; لُوقا 1:17; مَلاخِي 4:2–5"> وما كُنتُ أعرِفُهُ، فَجِئتُ أُعَمِّدُ بالماءِ حتّى يَظهرَ لإسرائيلَ“.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 1:10; لُوقا 3:22; ﻣﺘﻰ 3:16"> وشَهِدَ يوحنَّا، قالَ: ”رأيتُ الرُّوحَ يَنزِلُ مِنَ السَّماءِ مِثْلَ حمامةٍ ويَستَقِرُّ علَيهِ.</span>
<span className="v" id="v33" data-v="33" data-xref="اعمال الرسل 1:5; ﻣﺘﻰ 3:11; لُوقا 3:16"> وما كُنتُ أعرِفُهُ، لكنَّ الّذي أرسَلَني لأُعَمِّدَ بِالماءِ قالَ لي: ”الّذي ترى الرُّوحَ ينزِلُ ويَستَقِرُّ علَيهِ هوَ الّذي سَيُعَمِّدُ بالرُّوحِ القُدُسِ.</span>
<span className="v" id="v34" data-v="34" data-xref="المَزامِير 2:7; يُوحَنّا 1:49; المَزامِير 89:26–27"> وأنا رأيتُ وشَهِدتُ أنَّهُ هوَ اَبنُ اللهِ“.</span>
</div>
</div>

## تلاميذ يسوع الأولون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="يُوحَنّا 3:25–26; مَلاخِي 3:16"> وكانَ يوحنَّا في الغدِ واقِفًا هُناكَ، ومعَهُ اَثنانِ مِنْ تلاميذِهِ.</span>
<span className="v" id="v36" data-v="36" data-xref="يُوحَنّا 1:29; رسالَةُ بُطرُسَ الأُولَى 1:19–20; إشَعْياء 65:1–2"> فنظَرَ إلى يَسوعَ وهوَ مارٌّ وقالَ: ”ها هوَ حمَلُ اللهِ“.</span>
<span className="v" id="v37" data-v="37" data-xref="ﺭﻭﻣﻴﺔ 10:17; يُوحَنّا 1:43; زَكَريّا 8:21"> فسَمِعَ التِّلميذانِ كلامَهُ، فتَبِعا يَسوعَ.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 18:7; يُوحَنّا 3:2; يُوحَنّا 1:49"> واَلتَفتَ يَسوعُ فرآهُما يَتبَعانِهِ، فقَالَ لهُما: ”ماذا تُريدانِ؟“ قالا: ”رابي (أي يا مُعَلِّمُ) أينَ تُقِيمُ؟“</span>
<span className="v" id="v39" data-v="39" data-xref="لُوقا 24:29; ﻣﺘﻰ 11:28–30; يُوحَنّا 6:37"> قالَ: ”تَعالا واَنظُرا“. فذَهَبا ونَظرا أينَ يُقيمُ، فأقاما معَهُ ذلِكَ اليوم، وكانَتِ السَّاعةُ نحوَ الرّابعةِ بَعدَ الظُّهرِ.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="مَرْقُس 1:16–20; ﻣﺘﻰ 4:18–22; يُوحَنّا 1:40–42"> وكانَ أندَراوسُ أخو سِمعانَ بُطرُسَ أحدَ التِّلميذَينِ اللَّذَينِ سَمِعا كلامَ يوحنَّا فتَبِعا يَسوعَ.</span>
<span className="v" id="v41" data-v="41" data-xref="يُوحَنّا 4:25; يُوحَنّا 1:45; المَزامِير 89:20"> ولَقِيَ أندراوُسُ أخاهُ سِمعانَ، فقَالَ لَهُ: ”وجَدْنا المَسيّا“، أيِ المَسيحَ.</span>
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 21:15–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:22; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:5"> وجاءَ بِه إلى يَسوعَ، فنظَرَ إلَيهِ يَسوعُ وقالَ: ”أنتَ سِمعانُ بنُ يوحنَّا، وسأدْعوكَ ”كِيفا“ (أي صَخرًا)“.</span>
</div>
</div>

## يسوع يدعو فيلبس ونثنائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 9:9; يُوحَنّا 14:8; إشَعْياء 65:1"> وأرادَ يَسوعُ في الغدِ أنْ يذهَبَ إلى الجَليلِ. فلَقِيَ فِيلُبُّسَ، فقَالَ لَه: ”إِتبَعْني“.</span>
<span className="v" id="v44" data-v="44" data-xref="يُوحَنّا 12:21; ﻣﺘﻰ 11:21; مَرْقُس 8:22"> وكانَ فِيلبُّسُ مِنْ بَيت صيدا، مدينةِ أندَراوُسَ وبُطرُسَ.</span>
<span className="v" id="v45" data-v="45" data-xref="التَّثْنِيَة 18:18–22; لُوقا 24:27; ميخا 5:2"> ولَقِيَ فِيلبُّسُ نَثَنائِيلَ، فقَالَ لَه: ”وجَدْنا الّذي ذكرَهُ موسى في الشَّريعةِ، والأنبياءُ في الكُتُبِ، وهوَ يَسوعُ اَبنُ يوسفَ مِنَ النّاصِرَةِ“.</span>
<span className="v" id="v46" data-v="46" data-xref="يُوحَنّا 7:52; يُوحَنّا 7:41–42; يُوحَنّا 4:29"> فقَالَ لَه نَثَنائيلُ: ”أمنَ النّاصرةِ يَخرُجُ شيءٌ صالِحٌ؟“ فأجابَهُ فِيلبُّسُ: ”تَعالَ واَنظُرْ“.</span>
<span className="v" id="v47" data-v="47" data-xref="رُؤيا يُوحَنّا 14:5; المَزامِير 32:2; المَزامِير 73:1"> ورأى يَسوعُ نَثَنائيلَ مُقْبلًا إلَيهِ فقَالَ: ”هذا إِسرائيليٌّ صَميمٌ لا شَكَّ فيهِ!“</span>
<span className="v" id="v48" data-v="48" data-xref="إشَعْياء 65:24; يُوحَنّا 2:25; المَزامِير 139:1–2"> فقَالَ لَه نَثَنائيلُ: ”كيفَ عَرَفتَني؟“ فأجابَه يَسوعُ: ”رأيتُكَ تَحتَ التِّينةِ قَبلَما دَعاكَ فِيلبُّسُ“.</span>
