## الرُّوح القدس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:8; ﻣﺘﻰ 13:21; ﻣﺘﻰ 26:31–33"> ”قُلتُ لكُم هذا الكلامَ لِئَلَّا يَضعُفَ إيمانُكُم.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 9:22; اعمال الرسل 26:9–11; ﺭﻭﻣﻴﺔ 10:2–3"> سيَطرُدونَكُم مِنَ المَجامِعِ، بل تَجيءُ ساعةٌ يَظُنُّ فيها مَنْ يَقتُلُكم أنَّهُ يُؤدّي فَريضةً للهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 15:21; يُوحَنّا 17:25; يُوحَنّا 8:55"> وهُم يَعمَلونَ ذلِكَ لأنَّهُم لا يَعرِفونَ أبي ولا يَعرِفوني.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 14:29; مَرْقُس 13:23; يُوحَنّا 17:12–13"> أقولُ لكُم هذا، حتّى إذا جاءَتِ السَّاعةُ تَتذَكَّرونَ أنِّي قُلتُهُ لكُم. ما قُلتُ لكُم مِنَ البَدَاءَةِ لأنِّي كُنتُ مَعكُم.</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 13:36; يُوحَنّا 7:33; يُوحَنّا 16:28"> أمَّا الآنَ فأنا ذاهِبٌ إلى الّذي أرسَلَني، ولا أحَدَ مِنكُم يَسألُني: إلى أينَ أنتَ ذاهِبٌ؟</span>
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 14:1; يُوحَنّا 16:20–22; لُوقا 24:17"> والآنَ قُلتُ لكُم، فَمَلأَ الحُزنُ قُلوبَكُم.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 14:26; يُوحَنّا 14:16–17; يُوحَنّا 15:26"> صَدِّقوني، مِنَ الخَيرِ لكُم أنْ أذهَبَ، فإنْ كُنتُ لا أذهَبُ لا يَجيئُكُمُ المُعزِّي. أمَّا إذا ذَهَبتُ فأُرسِلُهُ إلَيكُم.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 2:37; يَهُوذا 1:15; زَكَريّا 12:10"> ومتى جاءَ وَبَّخَ العالَمَ على الخَطيئَةِ والبِرِّ والدَّينونَةِ:</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 15:22–25; يُوحَنّا 8:23–24; يُوحَنّا 12:47–48"> أمَّا على الخَطيئَةِ فلأنَّهُم لا يُؤْمِنونَ بي،</span>
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:21; رسالَةُ بُطرُسَ الأُولَى 3:18; ﺭﻭﻣﻴﺔ 10:3–4"> وأمَّا على البِرِّ فلأنِّي ذاهِبٌ إلى الآبِ ولَنْ تَرَوني،</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 12:31; رسالَةُ يُوحَنّا الأُولَى 3:8; العِبرانِيّين 2:14"> وأمَّا على الدَّينونَةِ فلأنَّ سيِّدَ هذا العالَمِ أُدينَ وحُكِمَ علَيهِ.</span>
<span className="v" id="v12" data-v="12" data-xref="مَرْقُس 4:33; يُوحَنّا 15:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:1–2"> عِندي كلامٌ كثيرٌ أقولُهُ لكُم بَعدُ، ولكنَّكُم لا تَقدِرونَ الآنَ أنْ تَحتَمِلوهُ.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 14:26; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:10–13; يُوحَنّا 14:17"> فمَتى جاءَ رُوحُ الحقِّ أَرشَدَكُم إلى الحَقِّ كُلِّهِ، لأنَّهُ لا يتكلَّمُ بِشيءٍ مِنْ عِندِهِ، بل يتكَلَّمُ بِما يَسمَعُ ويُخْبِرُكُم بِما سيَحدُثُ.</span>
<span className="v" id="v14" data-v="14" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:13–14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:14–18; يُوحَنّا 15:26"> سيُمَجِّدُني لأنَّهُ يأخُذُ كلامي ويَقولُهُ لكُم.</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 17:10; ﻣﺘﻰ 11:27; يُوحَنّا 3:35"> وكُلُّ ما لِلآبِ هوَ لي، لذلِكَ قُلتُ لكُم: يأخُذُ كلامي ويَقولُهُ لكُم.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 7:33; اعمال الرسل 1:3; يُوحَنّا 14:18–24"> بَعدَ قَليلٍ لا تَرَونَني، ثُمَّ بَعدَ قَليلٍ تَرَونَني“.</span>
</div>
</div>

## الحزن والفرح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 16:16; مَرْقُس 9:32; يُوحَنّا 16:5"> فقَالَ التَّلاميذُ بَعضُهُم لِبَعضٍ: ”ما هذا الّذي يقولُه لنا: بَعدَ قَليلٍ لا تَرَونَني، ثُمَّ بَعدَ قَليلٍ تَرَونَني. وأنا ذاهِبٌ إلى الآبِ؟“</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 24:25; ﻣﺘﻰ 16:9–11; العِبرانِيّين 5:12"> وتساءَلوا: ”ما مَعنى هذا القَليلِ؟ نَحنُ لا نَفهمُ ما يَقولُ“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 16:30; يُوحَنّا 16:16; يُوحَنّا 14:19"> وعرَفَ يَسوعُ أنَّهُم يُريدونَ أنْ يَسألوهُ، فقَالَ لهُم: ”تَتَساءَلونَ عَنْ مَعنى قولي: بَعدَ قَليلٍ لا تَرَونَني، ثُمَّ بَعدَ قَليلٍ تَرَونَني.</span>
<span className="v" id="v20" data-v="20" data-xref="يَعقُوب 1:2; ﻣﺘﻰ 5:4; المَزامِير 126:5–6"> الحقَّ الحقَّ أقولُ لكُم: ستَبكونَ وتَندُبونَ، وأمَّا العالَمُ فسَيَفرَحُ. ستَحزَنونَ، ولكِنَّ حُزنَكُم يَصيرُ فرَحًا.</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 113:9; إشَعْياء 13:8; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:3"> فالمَرأةُ تَحزَنُ وهيَ تَلِدُ، لأنَّ ساعتَها جاءَت. فإذا وَلَدَت تَنسى أوجاعَها، لِفَرَحِها بِولادَةِ إنسانٍ في العالَمِ.</span>
<span className="v" id="v22" data-v="22" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:8; اعمال الرسل 13:52; يُوحَنّا 16:6"> وكذلِكَ أنتُم تَحزَنونَ الآنَ، ولكنِّي سأعودُ فأراكُم، فتَفرَحُ قُلوبُكُم فرَحًا لا يَنتَزِعُهُ مِنكُم أحَدٌ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 15:7; يُوحَنّا 14:13–14; ﻣﺘﻰ 21:22"> في ذلِكَ اليومِ لا تَطلُبونَ مِنِّي شَيئًا. الحقَّ الحقَّ أقولُ لكُم: كُلُّ ما تَطلُبونَهُ مِنَ الآبِ باَسمي تَنالونَهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 7:7–8; يَعقُوب 4:2–3; يُوحَنّا 15:11"> وما طَلَبتُم شَيئًا باَسمي حتّى الآنَ. أُطلُبوا تَنالوا، فيكتَمِلَ فرَحُكُم.</span>
</div>
</div>

## الانتصار على العالم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 16:12; مَرْقُس 4:13; ﻣﺘﻰ 13:34–35"> ”قُلتُ لكم هذا كُلَّهُ بالأمثالِ. وتَجيءُ ساعةٌ لا أُحَدِّثُكُم فيها بالأمثالِ، بل أحَدِّثُكُم عَنِ الآبِ بِكلامٍ صَريحٍ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﺭﻭﻣﻴﺔ 8:34; يُوحَنّا 17:24; يُوحَنّا 17:19"> وفي ذلِكَ اليومِ أنتُم تَطلُبونَ مِنَ الآبِ باَسمي، ولا أقولُ لكم أنا أطلُبُ مِنهُ لأجلِكم.</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 14:21; يُوحَنّا 14:23; يُوحَنّا 17:23"> فالآبُ نَفسُهُ يُحِبُّكُم، لأنَّكُم أحبَبتُموني وآمَنتُم بأنِّي خَرَجتُ مِنْ عِندِ اللهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 13:1; يُوحَنّا 13:3; يُوحَنّا 8:14"> نعم، خَرَجتُ مِنْ عِندِ الآبِ وجِئتُ إلى العالَمِ وأتْرُكُ العالَمَ وأذهَبُ إلى الآبِ“.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 16:25"> فقَالَ لَه تلاميذُهُ: ”أنتَ الآنَ تَتكلَّمُ كلامًا صَريحًا، لا بالأمثالِ.</span>
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 21:17; يُوحَنّا 17:8; يُوحَنّا 16:17"> ونَرى الآنَ أنَّكَ تَعرِفُ كُلَّ شيءٍ، وأنَّكَ لا تَحتاجُ إلى أنْ يَسألَكَ أحَدٌ عَنْ شيءٍ. فلذلِكَ نُؤمنُ بِأنَّكَ جِئتَ مِنْ عِندِ اللهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="لُوقا 9:44–45; يُوحَنّا 13:38"> أجابَهُم يَسوعُ: ”الآنَ تُؤمِنونَ.</span>
<span className="v" id="v32" data-v="32" data-xref="زَكَريّا 13:7; ﻣﺘﻰ 26:31; يُوحَنّا 8:29"> تَجيءُ ساعَةٌ، بل جاءَتِ الآنَ، تَتَفرَّقونَ فيها، فيذهَبُ كُلُّ واحدٍ في سبيلِهِ وتَترُكوني وحدي. ولكِنْ لا أكونُ وَحدي، لأنَّ الآبَ مَعي.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 14:27; رسالَةُ يُوحَنّا الأُولَى 5:4; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:16"> قُلتُ لكُم هذا كُلَّهُ ليكونَ لكُم سَلامٌ بي. ستُعانونَ الشِّدَّةَ في هذا العالَمِ، فتَشَجَّعوا. أنا غَلَبتُ العالَمَ“.</span>
</div>
