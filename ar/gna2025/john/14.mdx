## الطَّريق والحق والحياة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 26:3; يُوحَنّا 16:22–23; إشَعْياء 12:2–3"> ”لا تَضطَرِبْ قلوبُكُم. أنتُم تُؤمِنونَ باللهِ فآمِنوا بي أيضًا.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:1; يُوحَنّا 13:36; العِبرانِيّين 13:14"> في بَيتِ أبي مَنازِلُ كثيرةٌ، وإلَّا لما قُلتُ لكُم: أنا ذاهِبٌ لأهيِّئ لكُم مكانًا.</span>
<span className="v" id="v3" data-v="3" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:16–17; يُوحَنّا 12:26; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:12"> ومتى ذَهَبتُ وهَيَّأتُ لكُم مكانًا، أرجِعُ وآخُذُكُم إليَّ لِتكونوا حَيثُ أكونُ.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 10:9; يُوحَنّا 16:28; يُوحَنّا 13:3"> أنتُم تَعرِفونَ الطَّريقَ إلى حيثُ أنا ذاهِبٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 11:16; مَرْقُس 9:19; يُوحَنّا 20:25–28"> فقَالَ لَه تُوما: ”يا ربُّ، نَحنُ لا نَعرِفُ إلى أينَ أنتَ ذاهِبٌ، فكيفَ نَعرِفُ الطَّريقَ؟“</span>
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 4:12; يُوحَنّا 10:9; أفَسُس 2:18"> أجابَهُ يَسوعُ: ”أنا هوَ الطَّريقُ والحقُّ والحياةُ، لا يَجيءُ أحَدٌ إلى الآبِ إلَّا بي.</span>
<span className="v" id="v7" data-v="7" data-xref="كُولُوسِي 1:15–17; يُوحَنّا 8:19; لُوقا 10:22"> لَو كُنتُم عَرَفْتُموني لَعَرَفْتُم أبي أيضًا. ومِنَ الآنَ أنتُم تَعرِفونَهُ، ورأَيتُموهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 33:18–23; رُؤيا يُوحَنّا 22:3–5; يُوحَنّا 1:43–46"> فقَالَ لَه فيلبُّسُ: ”يا ربُّ، أرِنا الآبَ وكَفانا“.</span>
<span className="v" id="v9" data-v="9" data-xref="كُولُوسِي 1:15; يُوحَنّا 12:45; يُوحَنّا 14:7"> فقَالَ لَه يَسوعُ: ”أنا مَعكُم كُلَّ هذا الوَقتِ، وما عَرَفتَني بَعدُ يا فيلبُّسُ؟ مَنْ رآني رأى الآبَ، فكيفَ تَقولُ: أرِنا الآبَ؟</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 5:19; يُوحَنّا 10:38; يُوحَنّا 10:30"> ألا تُؤمِنُ بأنِّي في الآبِ وأنَّ الآبَ فيَّ؟ الكلامُ الّذي أقولُهُ لا أقولُهُ مِنْ عِندي، والأعمالُ الّتي أعمَلُها يَعمَلُها الآبُ الّذي هوَ فيَّ.</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 10:38; يُوحَنّا 10:25; يُوحَنّا 5:36"> صدِّقوني إذا قُلتُ: أنا في الآبِ والآبُ فيَّ، أو صدِّقوني مِنْ أجلِ أعمالي.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 21:21; مَرْقُس 16:17; لُوقا 10:17–19"> الحقَّ الحقَّ أقولُ لكُم: مَنْ آمَنَ بي يَعمَلُ الأعمالَ الّتي أعمَلُها، بل أعظَمَ مِنها، لأنِّي ذاهِبٌ إلى الآبِ،</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 15:7; مَرْقُس 11:24; ﻣﺘﻰ 7:7"> فكُلُّ ما تَطلُبونَهُ باَسمي أعمَلُهُ، حتّى يَتَمَجَّدَ الآبُ في الابنِ.</span>
<span className="v" id="v14" data-v="14"> إذا طَلبتُم مِنِّي شيئًا باَسمي أعمَلُهُ.</span>
</div>
</div>

## الوعد بالروح القدس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:3–5; رسالَةُ يُوحَنّا الأُولَى 5:2–3; يُوحَنّا 14:21–24"> ”إذا كُنتُم تُحبّوني عَمِلْتُم بوصايايَ.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 14:26; يُوحَنّا 15:26; يُوحَنّا 16:7–15"> وسأطلُبُ مِنَ الآبِ أنْ يُعطيَكُم مُعَزِّيًا آخَرَ يَبقى مَعكُم إلى الأبَدِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 36:27; يُوحَنّا 16:13; رسالَةُ يُوحَنّا الأُولَى 2:27"> هوَ رُوحُ الحقِّ الّذي لا يَقدِرُ العالَمُ أنْ يَقبَلَهُ، لأنَّهُ لا يَراهُ ولا يَعرِفُهُ. أمَّا أنتُم فتَعرِفونَهُ، لأنَّهُ يُقيمُ مَعكُم ويكونُ فيكُم.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 14:3; المَزامِير 23:4; ﻣﺘﻰ 18:20"> لن أترُكَكُم يتامى، بل أرجِعُ إلَيكُم.</span>
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 16:16; يُوحَنّا 7:33; العِبرانِيّين 7:25"> بَعدَ قليلٍ لن يَراني العالَمُ، أمَّا أنتُم فتَرَونَني. ولأنِّي أحيا، فأنتُم سَتَحيونَ.</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 15:5–7; غَلاطِيَّة 2:20; يُوحَنّا 17:21–23"> وفي ذلِكَ اليومِ تَعرِفونَ أنِّي في أبي، وأنَّكُم أنتُم فيَّ مِثلَما أنا فيكُم.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 14:15; رسالَةُ يُوحَنّا الأُولَى 5:3; رسالَةُ يُوحَنّا الأُولَى 2:5"> مَنْ قَبِلَ وصاياي وعَمِلَ بِها أحَبَّني. ومَنْ أحَبَّني أحَبَّهُ أبي، وأنا أُحِبُّهُ وأُظهِرُ لَه ذاتي“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اعمال الرسل 1:13; لُوقا 6:16; ﻣﺘﻰ 10:3"> فقَالَ لَه يَهوذا، وهوَ غَيرُ يَهوذا الأسخَريوطيِّ: ”يا ربُّ، كيفَ تُظهِرُ ذاتَكَ لنا ولا تُظهِرُها لِلعالَمِ؟“</span>
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 14:21; يُوحَنّا 14:15; رُؤيا يُوحَنّا 3:20–21"> أجابَهُ يَسوعُ: ”مَنْ أحبَّني سَمِعَ كلامي فأحَبَّهُ أبي، ونَجِيءُ إلَيهِ ونُقيمُ عِندَهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 7:16; يُوحَنّا 14:10; يُوحَنّا 8:42"> ومَنْ لا يُحبُّني لا يَسمَعُ كلامي. وما كلامي مِن عِندي، بل مِنْ عِندِ الآبِ الّذي أرسَلَني.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 17:6–8; يُوحَنّا 13:19; يُوحَنّا 16:12"> قُلتُ لكُم هذا كُلَّهُ وأنا مَعكُم.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 15:26; يُوحَنّا 14:16; رسالَةُ يُوحَنّا الأُولَى 2:27"> ولكِنَّ المُعزِّي، وهوَ الرُّوحُ القُدُسُ الّذي يُرسِلُهُ الآبُ باَسْمي، سيُعلِّمُكُم كُلَّ شيءٍ ويَجعَلُكُم تتَذَكَّرونَ كُلَّ ما قُلتُهُ لكُم.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="فِيلِبِّي 4:7; يُوحَنّا 16:33; إشَعْياء 41:10"> سلامًا أترُكُ لكُم، وسَلامي أُعطيكُم، لا كما يُعطيهِ العالَمُ أُعطيكُم أنا. فلا تَضطَرِبْ قُلوبُكُم ولا تَفزَعْ.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 10:29–30; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:24–28; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:3"> قُلتُ لكُم: أنا ذاهِبٌ وسأرجِعُ إلَيكُم، فإنْ كُنتُم تُحبُّوني فَرِحْتُم بأنِّي ذاهِبٌ إلى الآبِ، لأنَّ الآبَ أعظَمُ مِنِّي.</span>
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 13:19; يُوحَنّا 16:4–31; ﻣﺘﻰ 24:24–25"> أخبَرْتُكُم بِهذا قَبلَ أنْ يَحدُثَ، حتّى متى حَدَثَ تُؤمِنونَ.</span>
<span className="v" id="v30" data-v="30" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:19; رسالَةُ يُوحَنّا الأُولَى 4:4; رُؤيا يُوحَنّا 12:9"> لَنْ أُخاطِبَكُم بَعدُ طويلًا، لأنَّ سَيِّدَ هذا العالَمِ سيَجيءُ. لا سُلطانَ لَه علَيَّ،</span>
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 10:18; فِيلِبِّي 2:8; يُوحَنّا 12:49"> ولكِنْ يَجبُ أنْ يَعرِفَ العالَمُ أنِّي أُحِبُّ الآبَ وأنِّي أعمَلُ بما أوصاني الآبُ. قوموا نَذهَبُ مِنْ هُنا.</span>
</div>
