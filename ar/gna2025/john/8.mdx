## المرأة الزانية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 21:1; مَرْقُس 11:1; لُوقا 19:37"> أمَّا يَسوعُ فخرَجَ إلى جبَلِ الزَّيتونِ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 26:55; لُوقا 4:20; لُوقا 21:37"> وعِندَ الفَجرِ رَجَعَ إلى الهَيكَلِ، فأقبَلَ إلَيهِ الشَّعبُ كُلُّهُ. فجَلَسَ وأخَذَ يُعَلِّمُهُم.</span>
<span className="v" id="v3" data-v="3"> وجاءَهُ مُعَلِّمو الشَّريعةِ والفَرِّيسيُّونَ باَمرأةٍ أمسَكَها بَعضُ النَّاسِ وهيَ تَزني، فَأوقفوها في وَسْطِ الحاضرينَ،</span>
<span className="v" id="v4" data-v="4"> وقالوا لَه: ”يا مُعَلِّمُ، أمسَكوا هذِهِ المَرأةَ في الزِّنى.</span>
<span className="v" id="v5" data-v="5" data-xref="اللّاويِّين 20:10; التَّثْنِيَة 22:21–24; حِزْقِيال 16:38–40"> وموسى أوصى في شَريعتِهِ بِرَجْمِ أمثالِها، فماذا تَقولُ أنتَ؟“</span>
<span className="v" id="v6" data-v="6" data-xref="إرْمِيا 17:13; لُوقا 10:25; الجامِعَة 3:7"> وكانوا في ذلِكَ يُحاوِلونَ إحراجَهُ ليَتَّهِموهُ.<br />فاَنحَنى يَسوعُ يكتُبُ بإصبَعِهِ في الأرضِ.</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 7:1–5; ﺭﻭﻣﻴﺔ 2:21–25; ﺭﻭﻣﻴﺔ 2:1–3"> فلمَّا ألحُّوا علَيهِ في السُّؤالِ، رفَعَ رأسَهُ وقالَ لهُم: ”مَنْ كانَ مِنكُم بِلا خَطيئَةٍ، فَليَرْمِها بأوّلِ حجَرٍ“.</span>
<span className="v" id="v8" data-v="8"> واَنحنى ثانيَةً يكتُبُ في الأرضِ.</span>
<span className="v" id="v9" data-v="9" data-xref="الجامِعَة 7:22; ﺭﻭﻣﻴﺔ 2:15; رسالَةُ يُوحَنّا الأُولَى 3:20"> فلمَّا سَمِعوا هذا الكلامَ، أخذَت ضَمائِرُهُم تُبكِّتُهُم، فخَرجوا واحدًا بَعدَ واحدٍ، وكِبارُهُم قَبلَ صِغارِهِم، وبَقِيَ يَسوعُ وحدَهُ والمرأةُ في مكانِها.</span>
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 41:11–12"> فجَلَسَ يَسوعُ وقالَ لها: ”أينَ هُم، يا اَمرأةُ؟ أما حكَمَ علَيكِ أحدٌ مِنهُم؟“</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 3:17; يُوحَنّا 5:14; إشَعْياء 1:16–18"> فأجابَت: ”لا، يا سيِّدي!“ فقَالَ لها يَسوعُ: ”وأنا لا أحكُمُ علَيكِ. إذهَبي ولا تُخطِئي بَعدَ الآنَ“&#93;.</span>
</div>

## يسوع نور العالم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 12:46; يُوحَنّا 9:5; يُوحَنّا 12:35"> وعادَ يَسوعُ إلى مُخاطَبَتِهِم، فقَالَ لهُم: ”أنا نُورُ العالَمِ. مَنْ يَتبَعْني لا يمشي في الظَّلامِ، بل يكونُ لَه نُورُ الحياةِ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 5:31–47"> فقَالَ لَه الفَرِّيسيُّونَ: ”أنتَ تَشهدُ لنَفسِكَ، فشَهادَتُكَ باطِلَةٌ“.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 16:28; يُوحَنّا 13:3; يُوحَنّا 8:42"> فأجابَهُم يَسوعُ: ”نعم، أنا أشهَدُ لِنَفْسي، ولكنَّ شَهادَتي صَحيحَةٌ، لأنِّي أعرِفُ مِنْ أينَ جِئتُ وإلى أينَ أذهَبُ. أمَّا أنتُم، فلا تَعرِفونَ مِنْ أينَ جِئتُ ولا إلى أينَ أذهَبُ.</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 16:7; يُوحَنّا 7:24; يُوحَنّا 12:47"> أنتُمْ تَحكُمونَ بِمقاييسِ البشَرِ، وأنا لا أحكُمُ على أحدٍ.</span>
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 17:31; يُوحَنّا 8:29; يُوحَنّا 16:32"> وإذا حكمتُ، فحُكمي صَحيحٌ لأنِّي لا أحكُمُ وَحدي، بل أنا والآبُ الّذي أرسَلَني.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّثْنِيَة 19:15; التَّثْنِيَة 17:6; العِبرانِيّين 10:28"> وفي شريعتِكُم أنَّ شهادَةَ شاهدَينِ صَحيحةٌ:</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 8:38; يُوحَنّا 8:58; يُوحَنّا 8:25"> فأنا أشهَدُ لِنَفسي، والآبُ الّذي أرسَلَني يَشهَدُ لي“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 16:3; يُوحَنّا 8:54–55; يُوحَنّا 14:6–9"> فقَالوا لَه: ”أينَ أبوكَ؟“ فأجابَهُم: ”أنتُم لا تَعرِفوني ولا تَعرِفونَ أبي. ولَو عَرَفتُموني لَعَرَفتُم أبي“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 7:30; يُوحَنّا 7:8; مَرْقُس 12:41"> قالَ يَسوعُ هذا الكلامَ عِندَ صُندوقِ التَّبرُّعاتِ وهوَ يُعَلِّمُ في الهَيكَلِ، فما أمسكَهُ أحدٌ لأنَّ ساعتَهُ ما جاءَت بَعدُ.</span>
</div>
</div>

## يسوع ينذر اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 7:34; يُوحَنّا 8:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:17–18"> ثُمَّ قالَ لهُم يَسوعُ: ”أنا ذاهِبٌ وستَطلُبوني، وفي خَطيئَتِكُم تَموتونَ. وحَيثُ أنا ذاهِبٌ لا تَقدِرونَ أنتُم أنْ تَجيئوا“.</span>
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 8:52; المَزامِير 22:6; يُوحَنّا 8:48"> فقَالَ اليَهودُ: ”لَعلَّهُ سيَقتُلُ نَفسَهُ، لأنَّهُ قالَ: حَيثُ أنا ذاهِبٌ لا تَقدِرونَ أنتُم أنْ تَجيئوا“.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 3:31; يُوحَنّا 17:14; يُوحَنّا 17:16"> وقالَ لهُم: ”أنتُم مِنْ أسفَلُ، أمَّا أنا فَمِنْ فَوقُ. أنتُم مِنْ هذا العالَمِ، وما أنا مِنْ هذا العالَمِ.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 3:36; مَرْقُس 16:16; يُوحَنّا 3:18"> لذلِكَ قُلْتُ لكُم: ستَموتونَ في خطاياكُم. نَعَم، سَتَموتونَ في خَطاياكُم، إذا كُنتُم لا تُؤمِنونَ أنِّي أنا هوَ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 5:17–29; لُوقا 22:67; يُوحَنّا 1:22"> فقَالوا لَه: ”مَنْ أنتَ؟“ فقَالَ يَسوعُ: ”أخبرتُكُم مِنَ البَدءِ.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 15:15; يُوحَنّا 12:47–50; يُوحَنّا 8:40"> عِندي أشياءُ كثيرةٌ أقولُها فيكُم، وأشياءُ كثيرةٌ أحكُمُ بِها علَيكُم. لكِنَّ الّذي أرَسَلني صادِقٌ، وما سَمِعْتُهُ أقولُهُ لِلعالَمِ“.</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 6:9; يُوحَنّا 8:43; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:3–4"> فما فَهِموا أنَّهُ يُحدِّثُهُم عَنِ الآبِ،</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 5:19; يُوحَنّا 5:30; يُوحَنّا 6:38"> فقَالَ لهُم: ”متى رفَعتُمُ اَبنَ الإنسانِ عَرَفْتُم أنِّي أنا هوَ، وأنِّي لا أعمَلُ شيئًا مِنْ عِندي ولا أقولُ إلَّا ما عَلَّمَني الآبُ.</span>
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 6:38; يُوحَنّا 16:32; يُوحَنّا 4:34"> والآبُ الّذي أرسَلَني هوَ مَعي وما تَركني وَحدي، لأنِّي في كُلِّ حينٍ أعمَلُ ما يُرضيهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 7:31; يُوحَنّا 2:23; يُوحَنّا 11:45"> ولمَّا قالَ هذا الكلامَ، آمنَ بِه كثيرٌ مِنَ النَّاسِ.</span>
</div>
</div>

## الحق يحرّركم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="رسالَةُ يُوحَنّا الثّانيَة 1:9; يُوحَنّا 15:4–9; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:14"> فقَالَ يَسوعُ لِلّذينَ آمنوا بِه مِنَ اليَهودِ: ”إذا ثَبَتُّم في كلامي، صِرتُم في الحَقيقةِ تلاميذي:</span>
<span className="v" id="v32" data-v="32" data-xref="ﺭﻭﻣﻴﺔ 8:2; يُوحَنّا 8:36; يُوحَنّا 17:17"> تَعرِفونَ الحقَّ، والحقُّ يُحرِّرُكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="ﻣﺘﻰ 3:9; يُوحَنّا 8:39; التَّكوين 15:13"> قالوا لَه: ”نَحنُ ذُرِّيَّةُ إبراهيمَ، وما كُنَّا يومًا عَبيدًا لأحدٍ! فكيفَ تَقولُ لنا: ستَصيرونَ أحرارًا؟“</span>
<span className="v" id="v34" data-v="34" data-xref="ﺭﻭﻣﻴﺔ 6:16; رسالَةُ بُطرُسَ الثّانيَة 2:19; الأمثال 5:22"> فأجابَهُم يَسوعُ: ”الحقَّ الحقَّ أقولُ لكُم: مَنْ يَخطأْ كانَ عَبدًا لِلخَطيئةِ،</span>
<span className="v" id="v35" data-v="35" data-xref="غَلاطِيَّة 4:4–7; لُوقا 15:31; ﺭﻭﻣﻴﺔ 8:15–17"> والعَبدُ لا يُقيمُ في البَيتِ إلى الأبدِ، بَلِ الابنُ يُقيمُ إلى الأبدِ.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:17; غَلاطِيَّة 5:1; ﺭﻭﻣﻴﺔ 8:2"> فإذا حَرَّركُمُ الابنُ، صِرتُم بِالحقيقَةِ أحرارًا.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14; يُوحَنّا 8:45–47; يُوحَنّا 7:1"> أنا أعرِفُ أنَّكُم ذُرِّيَّةُ إبراهيمَ. ولكنَّكُم تُريدونَ قَتْلي، لأنَّ كلامي لا مَحلَّ لَه فيكُم.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 5:19; يُوحَنّا 8:44; يُوحَنّا 8:41"> أنا أتكَلَّمُ بِما رأيتُ عِندَ أبي، وأنتُم تَعمَلونَ بِما سمِعْتُم مِنْ أبيكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="غَلاطِيَّة 3:7; يُوحَنّا 8:37; ﺭﻭﻣﻴﺔ 9:7"> فقَالوا لَه: ”أبونا هوَ إبراهيمُ“. فأجابَهُم يَسوعُ: ”لَو كُنتُم أبناءَ إبراهيمَ، لَعَمِلْتُم أعمالَ إبراهيمَ.</span>
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 8:26; غَلاطِيَّة 4:29; غَلاطِيَّة 4:16"> ولكِنَّكُمُ الآنَ تُريدونَ قَتْلي، أنا الّذي كَلَّمَكُم بالحقِّ كما سَمِعتُهُ مِنَ اللهِ، وهذا العَمَلُ ما عَمِلَهُ إبراهيمُ.</span>
<span className="v" id="v41" data-v="41" data-xref="إشَعْياء 64:8; يُوحَنّا 8:38; إشَعْياء 63:16"> أنتُم تَعمَلونَ أعمالَ أبيكُم“. قالوا لَه: ”ما نَحنُ أولادُ زِنًى، ولنا أبٌ واحدٌ هوَ اللهُ!“</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 12:49; يُوحَنّا 14:10; يُوحَنّا 17:8"> فقَالَ لهُم يَسوعُ: ”لَو كانَ اللهُ أباكُم لأحبَبتُموني. لأنِّي خَرَجتُ وجِئتُ مِنْ عِندِ اللهِ، وما جِئْتُ مِنْ تِلقاءِ ذاتي، بل هوَ الّذي أرسَلَني.</span>
<span className="v" id="v43" data-v="43" data-xref="يُوحَنّا 12:39–40; ﺭﻭﻣﻴﺔ 8:7–8; يُوحَنّا 7:17"> لِماذا لا تَفهَمونَ ما أقولُهُ لكُم؟ لأنَّكُم لا تُطيقونَ أنْ تَستَمِعوا إلى كلامي.</span>
<span className="v" id="v44" data-v="44" data-xref="اعمال الرسل 13:10; رسالَةُ يُوحَنّا الأُولَى 3:12; رسالَةُ يُوحَنّا الأُولَى 3:8–10"> فأنتُم أولادُ أبيكُم إبليسَ، وتُريدونَ أنْ تَتَّبِعوا رَغَباتِ أبيكُم، هذا الّذي كانَ مِنَ البَدءِ قاتلًا. ما ثَبَتَ على الحقِّ، لأنْ لا حَقَّ فيهِ. وهوَ يكذِبُ، والكَذِبُ في طَبعِهِ، لأنَّهُ كَذَّابٌ وأبو الكَذِبِ.</span>
<span className="v" id="v45" data-v="45" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:3–4; غَلاطِيَّة 4:16; يُوحَنّا 18:37"> أمَّا أنا، فلا تُصدِّقوني لأنِّي أقولُ الحقَّ.</span>
<span className="v" id="v46" data-v="46" data-xref="العِبرانِيّين 4:15; يُوحَنّا 16:8; رسالَةُ بُطرُسَ الأُولَى 2:22"> فمَنْ مِنْكُم يَقدرُ أنْ يُثبِتَ عَلَيَّ خَطيئةً؟ وإذا كُنتُ أقولُ الحقَّ، فلِماذا لا تُصدِّقوني؟</span>
<span className="v" id="v47" data-v="47" data-xref="يُوحَنّا 10:26–27; رسالَةُ يُوحَنّا الثّانيَة 1:9; يُوحَنّا 18:37"> مَنْ كانَ مِنَ اللهِ سَمِعَ كلامَ اللهِ. وما أنتُم مِنَ اللهِ، لذلِكَ لا تَسمَعونَ“.</span>
</div>
