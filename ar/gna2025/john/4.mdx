## يسوع والمرأة السامرية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 3:26; يُوحَنّا 3:22; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:5"> وعرَفَ الرَّبُّ يَسوعُ أنَّ الفَرِّيسيِّينَ سَمِعوا أنَّهُ تَلمَذَ وعَمَّدَ أكثَرَ مِمَّا تَلمَذَ يوحنَّا وعَمَّدَ،</span>
<span className="v" id="v2" data-v="2" data-xref="اعمال الرسل 10:48; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:13–17"> معَ أنَّ يَسوعَ نَفسَهُ ما كانَ يُعَمِّدُ بل تلاميذُهُ،</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 10:23; يُوحَنّا 10:40; يُوحَنّا 2:11"> فتَرَكَ اليهودِيَّةَ ورجَعَ إلى الجَليلِ.</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 17:11; لُوقا 2:49; لُوقا 9:51–52"> وكانَ لا بُدَّ لَه مِنَ المُرورِ بالسّامِرَةِ،</span>
<span className="v" id="v5" data-v="5" data-xref="يَشُوع 24:32; التَّكوين 33:19; التَّكوين 48:22"> فوصَلَ إلى مدينةٍ سامرِيَّةٍ اَسمُها سُوخارُ، بالقُربِ مِنَ الأرضِ الّتي وهَبَها يَعقوبُ لاَبنِهِ يوسُفَ،</span>
<span className="v" id="v6" data-v="6" data-xref="العِبرانِيّين 4:15; العِبرانِيّين 2:17; ﻣﺘﻰ 8:24"> وفيها بِئرُ يَعقوبَ. وكانَ يَسوعُ تَعِبَ مِنَ السَّفَرِ، فقَعَدَ على حافَّةِ البئرِ. وكانَ الوَقتُ نحوَ الظُّهرِ.</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 10:42; المُلُوكِ الأوَّلُ 17:10; التَّكوين 24:43"> فجاءَتِ اَمرأةٌ سامريَّةٌ تَستَقي مِنْ ماءِ البِئرِ، فقَالَ لها يَسوعُ: ”أعطيني لأشرَبَ“.</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 9:13; يُوحَنّا 4:39; يُوحَنّا 4:5"> وكانَ تلاميذُهُ في المدينةِ يَشتَرُونَ طَعامًا.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 10:28; ﻣﺘﻰ 10:5; المُلُوكِ الثّانِي 17:24–41"> فأجابَتِ المرأةُ: ”أنتَ يَهوديٌّ وأنا سامرِيَّةٌ، فكيفَ تَطلُبُ مِنِّي أنْ أسقِيَكَ؟“ قالَت هذا لأنَّ اليَهودَ لا يُخالِطونَ السّامِريِّينَ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 4:14; رُؤيا يُوحَنّا 7:17; يُوحَنّا 7:37–39"> فقَالَ لها يَسوعُ: ”لو كُنتِ تَعرِفينَ عَطِيَّةَ اللهِ، ومَنْ هوَ الّذي يَقولُ لكِ أعطيني لأشرَبَ، لَطَلَبتِ أنتِ مِنهُ فأعطاكِ ماءَ الحياةِ“.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14; يُوحَنّا 3:4"> قالَت لَه المَرأةُ: ”لا دَلوَ معَكَ، يا سيِّدي، والبِئرُ عَميقَةٌ، فمِنْ أينَ لكَ ماءُ الحياةِ؟</span>
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 4:6; يُوحَنّا 8:53; ﻣﺘﻰ 12:42"> أبونا يَعقوبُ أعطانا هذِهِ البِئرَ، وشَرِبَ مِنها هوَ وأولادُهُ ومواشيهِ، فهَلْ أنتَ أعظَمُ مِنْ يَعقوبَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 6:27; إشَعْياء 65:13–14; لُوقا 16:24"> فأجابَها يَسوعُ: ”كُلُّ مَنْ يَشرَبُ مِنْ هذا الماءِ يَعطَشُ ثانيةً،</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 7:38–39; يُوحَنّا 6:35; إشَعْياء 49:10"> أمَّا مَنْ يَشرَبُ مِنَ الماءِ الّذي أُعطيهِ أنا، فلَنْ يَعطَشَ أبدًا. فالماءُ الّذي أُعطيهِ يَصيرُ فيهِ نَبعًا يَفيضُ بِالحياةِ الأبديَّةِ“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 6:34; ﺭﻭﻣﻴﺔ 8:5; المَزامِير 4:6"> قالَت لَه المرأةُ: ”أَعطِني مِنْ هذا الماءِ يا سيِّدي، فلا أعطَشَ ولا أعودَ إلى هُنا لأستقِيَ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="العِبرانِيّين 4:13; رُؤيا يُوحَنّا 2:23; يُوحَنّا 4:18"> قالَ لَها: ”إِذهَبي واَدْعِي زوجَكِ، واَرجِعِي إلى هُنا“.</span>
<span className="v" id="v17" data-v="17"> فأجابَتِ المَرأةُ: ”لا زَوجَ لي“. فقَالَ لها يَسوعُ: ”أصَبْتِ في قولِكِ: لا زَوجَ لي،</span>
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 3:20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:10–11; ﺭﻭﻣﻴﺔ 7:3"> لأنَّهُ كانَ لكِ خَمسةُ أزواجٍ، والّذي لكِ الآنَ ما هوَ زَوجُكِ. وفي هذا صَدَقْتِ“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="لُوقا 7:39; يُوحَنّا 6:14; يُوحَنّا 9:17"> قالَتِ المَرأةُ: ”أرى أنَّكَ نَبِيٌّ، يا سيِّدي!</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 12:6–7; التَّثْنِيَة 27:12; التَّثْنِيَة 11:29"> آباؤُنا عَبَدوا اللهَ في هذا الجبَلِ، وأنتُمُ اليَهودُ تَقولونَ إنَّ أُورُشليمَ هيَ المكانُ الّذي يَجِبُ أنْ نَعبُدَ اللهَ فيهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:8; مَلاخِي 1:11; يُوحَنّا 4:23"> قالَ لها يَسوعُ: ”صدِّقيني يا اَمرَأةُ، يَحينُ وقتٌ يَعبُدُ النَّاسُ فيهِ الآبَ، لا في هذا الجبَلِ ولا في أُورُشليمَ.</span>
<span className="v" id="v22" data-v="22" data-xref="اعمال الرسل 17:23; إشَعْياء 2:3; ﺭﻭﻣﻴﺔ 9:4–5"> وأنتُم السّامِريّينَ تَعبُدونَ مَنْ تَجهَلونَهُ، ونَحنُ اليَهودَ نَعبُدُ مَنْ نَعرِفُ، لأنَّ الخلاصَ يَجيءُ مِنَ اليَهودِ.</span>
<span className="v" id="v23" data-v="23" data-xref="فِيلِبِّي 3:3; يُوحَنّا 5:25; ﻣﺘﻰ 15:7–9"> ولكِنْ ستَجيءُ ساعَةٌ، بل جاءَتِ الآنَ، يَعبُدُ فيها العابِدونَ الصادِقونَ الآبَ بالرُّوحِ والحَقِّ. هَؤلاءِ هُمُ العابِدونَ الّذينَ يُريدُهُمُ الآبُ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:17; ﻣﺘﻰ 15:8–9; فِيلِبِّي 3:3"> اللهُ رُوحٌ، وبالرُّوحِ والحَقِّ يَجِبُ على العابِدينَ أنْ يَعبُدوهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 1:49; دانيال 9:24–26; يُوحَنّا 1:41–42"> قالَت لَه المَرأةُ: ”أعرِفُ أنَّ المَسيَّا، (أي المسيحَ) سيَجيءُ. ومتى جاءَ أخبَرَنا بِكُلِّ شيءٍ“.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 8:24; مَرْقُس 14:61–62; ﺭﻭﻣﻴﺔ 10:20–21"> قالَ لها يَسوعُ: ”أنا هوَ، أنا الّذي يُكلِّمُكِ“.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="لُوقا 7:39; يُوحَنّا 4:8–9"> وعِندَ ذلِكَ رجَعَ تلاميذُهُ. فتَعَجَّبوا حينَ وجَدوهُ يُحادِثُ اَمرأةً. ولكِنْ لا أحدَ مِنهُم قالَ: ”ماذا تُريدُ مِنها؟“ أو ”لماذا تُحادِثُها؟“</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 4:7; لُوقا 24:33; مَرْقُس 16:8–10"> وتَركَتِ المَرأةُ جَرَّتَها ورَجَعَت إلى المدينةِ. فقَالَت للنّاسِ هُناكَ:</span>
<span className="v" id="v29" data-v="29" data-xref="رُؤيا يُوحَنّا 22:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:24–25; يُوحَنّا 4:25"> ”تَعالَوا اَنظُروا رَجُلًا ذكَرَ لي كُلَّ ما عَمِلتُ. فهَلْ يكونُ هوَ المَسيحُ؟“</span>
<span className="v" id="v30" data-v="30" data-xref="ﺭﻭﻣﻴﺔ 5:20; اعمال الرسل 8:5–8; لُوقا 17:16–18"> فخرَجوا مِنَ المدينةِ وجاؤُوا إلى يَسوعَ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 1:38; التَّكوين 24:33; اعمال الرسل 16:30–34"> وكانَ التَّلاميذُ في أثناءِ ذلِكَ يَقولونَ ليَسوعَ: ”كُلْ، يا مُعَلِّمُ“.</span>
<span className="v" id="v32" data-v="32" data-xref="أيُّوب 23:12; المَزامِير 119:103; يُوحَنّا 4:34"> فقَالَ لهُم: ”لي طعامٌ آكُلُه لا تَعرِفونَهُ أنتُم“.</span>
<span className="v" id="v33" data-v="33" data-xref="ﻣﺘﻰ 16:6–11; لُوقا 9:45"> فأخَذَ التَّلاميذُ يَتساءَلونَ: ”هَل جاءَهُ أحدٌ بِما يُؤكَلُ؟“</span>
<span className="v" id="v34" data-v="34" data-xref="يُوحَنّا 6:38; يُوحَنّا 5:30; أيُّوب 23:12"> وقالَ لهُم يَسوعُ: ”طعامي أنْ أعمَلَ بِمَشيئَةِ الّذي أرْسَلني وأُتمِّمَ عَمَلهُ.</span>
<span className="v" id="v35" data-v="35" data-xref="لُوقا 10:2–3; ﻣﺘﻰ 9:37–38; يُوحَنّا 4:30"> أما تَقولونَ: بَعدَ أربعةِ أشهُرٍ يَجيءُ الحَصادُ؟ وأنا أقولُ لكُم: تَطَلَّعوا واَنظُروا إلى الحُقولِ كيفَ اَبيضَّت ونَضجَت لِلحَصادِ.</span>
<span className="v" id="v36" data-v="36" data-xref="يَعقُوب 5:19–20; ﺭﻭﻣﻴﺔ 1:13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:5–9"> وها هوَ الحاصِدُ يأخُذُ أُجرتَهُ، فيَجمَعُ ثَمَرًا لِلحياةِ الأبديَّةِ. فيفرَحُ الزَّارعُ معَ الحاصِدِ،</span>
<span className="v" id="v37" data-v="37" data-xref="ميخا 6:15; أيُّوب 31:8; لُوقا 19:21"> ويَصْدُقُ القولُ: واحِدٌ يَزرَعُ وآخَرُ يَحصُدُ.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 1:7; أخبارِ الأيّامِ الثّانِي 36:15; اعمال الرسل 10:42–43"> وأنا أرسَلتُكُم لِتَحصُدوا حَقلًا ما تَعِبتُم فيهِ. غَيرُكُم تَعِبَ وأنتُم تَجنُونَ ثَمرَةَ أتعابِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 4:29; يُوحَنّا 4:5; يُوحَنّا 10:41–42"> فآمَنَ بِه كثيرٌ مِنَ السّامِريِّينَ في تِلكَ المدينةِ، لأنَّ المَرأةَ شَهِدَت فقَالَت: ”ذكَرَ لي كُلَّ ما عَمِلتُ“.</span>
<span className="v" id="v40" data-v="40" data-xref="رُؤيا يُوحَنّا 3:20; اعمال الرسل 16:15; لُوقا 24:29"> فلمَّا جاءَ إلَيهِ السّامريُّونَ رَجَوا مِنهُ أنْ يُقيمَ عِندَهُم، فأقامَ يومَينِ.</span>
<span className="v" id="v41" data-v="41" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:4–5; اعمال الرسل 1:8; اعمال الرسل 8:12"> وزادَ كثيرًا عددُ المُؤمنينَ بِه عِندَما سَمِعوا كلامَهُ،</span>
<span className="v" id="v42" data-v="42" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:14; يُوحَنّا 1:29; رسالَةُ يُوحَنّا الأُولَى 5:20"> وقالوا لِلمَرأةِ: ”نَحنُ نُؤمِنُ الآنَ، لا لِكلامِكِ، بل لأنَّنا سَمِعناهُ بأنفُسِنا وعَرَفنا أنَّهُ بالحقيقةِ هوَ مُخَلِّصُ العالَمِ“.</span>
</div>

## يسوع يشفي طفلًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 4:13; ﺭﻭﻣﻴﺔ 15:8; مَرْقُس 7:27–28"> وبَعدَ يومَينِ ذهَبَ يَسوعُ إلى الجليلِ،</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 13:57; لُوقا 4:24; مَرْقُس 6:4"> مَع أنَّهُ هوَ الّذي قالَ: ”لا كرامةَ لِنَبيٍّ في وطَنِهِ“.</span>
<span className="v" id="v45" data-v="45" data-xref="يُوحَنّا 2:23; يُوحَنّا 2:13–16; التَّثْنِيَة 16:16"> فلمَّا وصَلَ إلى الجَليلِ، رَحَّبَ بِه أهلُها، لأنَّهُم كانوا في أُورُشليمَ في عيدِ الفِصحِ، فشاهَدوا كُلَّ ما عَمِلَ مُدَّةَ العيدِ.</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="يُوحَنّا 2:1–11; لُوقا 7:2; هُوشَع 5:15"> وجاءَ أيضًا إلى قانا الجليلِ، حيثُ جَعلَ الماءَ خَمرًا. وكانَ في كَفْرناحومَ رَجُلٌ مِنْ حاشيةِ المَلِكِ، لهُ اَبنٌ مَريضٌ.</span>
<span className="v" id="v47" data-v="47" data-xref="يُوحَنّا 4:3; يُوحَنّا 11:21; يُوحَنّا 11:32"> فلمَّا سَمِعَ أنَّ يَسوعَ وصَلَ مِنَ اليَهوديَّةِ إلى الجَليلِ، جاءَ إلَيهِ يَلتَمِسُ مِنهُ أنْ يَنزِلَ إلى كَفْرناحومَ لِيشفيَ اَبنَهُ الّذي أشرَفَ على الموتِ.</span>
<span className="v" id="v48" data-v="48" data-xref="يُوحَنّا 20:29; العِبرانِيّين 2:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:22"> فقَالَ لَه يَسوعُ: ”أنتُم لا تُؤمنونَ إلَّا إذا رأيتُمُ الآياتِ والعَجائِبَ“.</span>
<span className="v" id="v49" data-v="49" data-xref="مَرْقُس 5:23; المَزامِير 88:10–12; المَزامِير 40:17"> فقَالَ لَه الرَّجُلُ: ”إِنزِلْ يا سيِّدي، قَبلَ أنْ يَموتَ ولَدي“.</span>
