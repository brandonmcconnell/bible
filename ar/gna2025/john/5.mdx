## يسوع يشفي كسيحًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّثْنِيَة 16:16; الخُرُوج 23:14–17; الخُرُوج 34:23"> وجاءَ عِيدٌ لليَهودِ، فصَعِدَ يَسوعُ إلى أُورُشليمَ.</span>
<span className="v" id="v2" data-v="2" data-xref="نَحَمْيا 12:39; نَحَمْيا 3:1; نَحَمْيا 3:32"> وفي أُورُشليمَ، قُربَ بابِ الغنَمِ، بِركَةٌ لها خَمسةُ أروِقَةٍ، يُسمّونَها بالعِبريَّةِ بَيتَ زاتا.</span>
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 3:1–4; لُوقا 7:22; ﻣﺘﻰ 15:30"> وكانَ في الأروِقَةِ جَماعةٌ مِنَ المَرضى، بَينَ عُميانٍ وعُرجانٍ ومَفلوجينَ، &#91;يَنتَظِرونَ تَحريكَ الماءِ،</span>
<span className="v" id="v4" data-v="4"> لأنَّ ملاكَ الرّبِّ كانَ يَنزِلُ أحيانًا في البِركَةِ ويُحرِّكُ الماءَ. فكانَ الّذي يَسبِقُ إلى النُّزولِ بَعدَ تَحريكِ الماءِ يُشفى مِنْ أيِّ مَرضٍ أصابَهُ&#93;.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 3:2; لُوقا 8:43; مَرْقُس 9:21"> وكانَ هُناكَ رَجُلٌ مَريضٌ مِنْ ثمانٍ وثلاثينَ سنَةً.</span>
<span className="v" id="v6" data-v="6" data-xref="لُوقا 18:41; العِبرانِيّين 4:15; العِبرانِيّين 4:13"> فلمَّا رآهُ يَسوعُ مُستَلقيًا، عرَفَ أنَّ لَه مُدَّةً طَويلةً على هذِهِ الحالِ، فقَالَ لَه: ”أتُريدُ أنْ تُشفى؟“</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 72:12; التَّثْنِيَة 32:36; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:24"> فأجابَه المريضُ: ”ما لي أحَدٌ، يا سيِّدي، يُنزِلُني في البِركَةِ عِندَما يتحَرَّكُ الماءُ. وكُلَّما حاوَلتُ الوُصولَ إلَيها سبَقَني غَيري“.</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 5:24; ﻣﺘﻰ 9:6; مَرْقُس 2:11"> فقَالَ لَه يَسوعُ: ”قُمْ واَحمِلْ فِراشَكَ واَمشِ“.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 3:7–8; يُوحَنّا 9:14; مَرْقُس 3:2–4"> فتَعافى الرَّجُلُ في الحالِ، وحمَلَ فِراشَهُ ومَشى.<br />وكانَ ذلِكَ يومَ السَّبتِ.</span>
<span className="v" id="v10" data-v="10" data-xref="إرْمِيا 17:21; لُوقا 6:2; يُوحَنّا 9:16"> فقَالَ اليَهودُ للّذي تَعافَى: ”هذا يومُ السَّبتِ، فلا يَحلُّ لكَ أنْ تَحمِلَ فِراشَكَ“.</span>
<span className="v" id="v11" data-v="11" data-xref="مَرْقُس 2:9–11; يُوحَنّا 9:16"> فأجابَهُم: ”الّذي شَفاني قالَ لي: إحْمِل فِراشَكَ واَمشِ“.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 21:23; القُضاة 6:29; ﺭﻭﻣﻴﺔ 10:2"> فسألوهُ: ”مَنْ هوَ الّذي قالَ لكَ إحمِلْ فِراشَكَ واَمشِ؟“</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 4:30; لُوقا 24:31; يُوحَنّا 8:59"> وكانَ الرَّجُلُ لا يَعرِفُ مَنْ هوَ، لأنَّ يَسوعَ اَبتعَدَ عَن الجمعِ المُحتَشِدِ هُناكَ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 8:11; اللّاويِّين 26:23–24; المَزامِير 118:18"> ولَقِيَهُ يَسوعُ بَعدَ ذلِكَ في الهَيكَلِ، فقَالَ لَه: ”أنتَ الآنَ تَعافَيتَ، فلا تَعُدْ إلى الخَطيئَةِ، لِئَلَّا تُصابَ بِأَسوأَ“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 9:11–12; يُوحَنّا 4:29; يُوحَنّا 9:25"> فذهَبَ الرَّجُلُ إلى اليَهودِ وأخبرَهُم بأنَّ يَسوعَ هوَ الّذي شَفاهُ.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 10:39; يُوحَنّا 7:19–20; مَرْقُس 3:6"> فأخَذَ اليَهودُ يَضطَهِدونَ يَسوعَ، لأنَّهُ كانَ يَفعلُ ذلِكَ يومَ السَّبتِ.</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 9:4; يُوحَنّا 14:10; كُولُوسِي 1:16"> فقَالَ لهُم يَسوعُ: ”أبي يَعمَلُ في كُلِّ حينٍ، وأنا أعمَلُ مِثلَهُ“.</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 10:30; يُوحَنّا 10:33; يُوحَنّا 19:7"> فاَزدادَ سَعيُ اليَهودِ إلى قَتْلِهِ، لأنَّهُ معَ مُخالَفَتِهِ الشَّريعةَ في السَّبتِ، قالَ إنَّ اللهَ أبوهُ، فَساوى نَفسَهُ بِاللهِ.</span>
</div>
</div>

## سلطة الإبن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 12:49; يُوحَنّا 8:28; يُوحَنّا 14:10"> فقَالَ لهُم يَسوعُ:<br />”الحقَّ الحقَّ أقولُ لكُم:<br />لا يَقدِرُ الابنُ أنْ يَعمَلَ شَيئًا مِنْ عِندِهِ،<br />بل يَعمَلُ ما رأى الآبَ يَعمَلُهُ.<br />فما يَعمَلُهُ الآبُ يَعمَلُ مِثلَهُ الابنُ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 3:35; ﻣﺘﻰ 3:17; يُوحَنّا 17:26"> فالآبُ يُحبُّ الابنَ ويُريهِ كُلَّ ما يَعمَلُ،<br />وسَيُريهِ ما هوَ أعظمُ، فتَتَعجَّبونَ</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="ﺭﻭﻣﻴﺔ 8:11; التَّثْنِيَة 32:39; يُوحَنّا 11:25"> فكما يُقيمُ الآبُ الموتى ويُحْييهم،<br />كذلِكَ الابنُ يُحيِي مَنْ يَشاءُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 5:27; اعمال الرسل 17:31; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:10"> والآبُ لا يَدينُ بِنَفسِهِ أحدًا<br />لأنَّهُ جعَلَ الدَّينونَةَ كُلَّها لِلابنِ،</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:23; لُوقا 10:16; ﻣﺘﻰ 11:27"> حتّى يُمجِّدَ جميعُ النَّاسِ الابنَ، كما يُمَجِّدونَ الآبَ.<br />مَنْ لا يُمَجِّدُ الابنَ، لا يُمَجِّدُ الآبَ الّذي أرسَلَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 3:36; يُوحَنّا 6:40; رسالَةُ يُوحَنّا الأُولَى 5:11–13"> الحقَّ الحقَّ أقولُ لكُم:<br />مَنْ يَسمَع لي ويُؤمِن بِمَنْ أرسَلَني<br />فلَهُ الحياةُ الأبديَّةُ، ولا يَحضرُ الدَّينونَةَ،<br />لأنَّهُ اَنتَقَلَ مِنَ الموتِ إلى الحياةِ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 5:28; أفَسُس 5:14; أفَسُس 2:5"> الحقَّ الحقَّ أقولُ لكُم:<br />سَتَجِيءُ ساعةٌ، بل جاءَتِ الآنَ،<br />يسمَعُ فيها الأمواتُ صَوتَ اَبنِ اللهِ،<br />وكُلُّ مَنْ يُصغي إلَيهِ يَحيا.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 11:26; يُوحَنّا 1:4; الخُرُوج 3:14"> فكما أنَّ الآبَ هوَ في ذاتِهِ مَصدَرُ الحياةِ،<br />فكذلِكَ أعطى الابنَ أنْ يكونَ في ذاتِهِ مَصدَرَ الحياةِ</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 10:42; يُوحَنّا 5:22; اعمال الرسل 17:31"> وأعطاهُ أنْ يَدينَ أيضًا<br />لأنَّهُ اَبنُ الإنسانِ.</span>
<span className="v" id="v28" data-v="28" data-xref="رُؤيا يُوحَنّا 20:12; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:14–17; إشَعْياء 26:19"> لا تتعجَّبوا من هذا.<br />ستَجيءُ ساعةٌ يَسمَعُ فيها صوتَهُ<br />جميعُ الّذينَ في القُبورِ،</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="دانيال 12:2–3; ﺭﻭﻣﻴﺔ 2:6–10; اعمال الرسل 24:15"> فيَخرُجُ مِنها الّذينَ عَمِلوا الصّالحاتِ ويَقومونَ إلى الحياةِ،<br />والّذينَ عَمِلوا السَّيِّئاتِ يَقومونَ إلى الدَّينونَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 6:38; يُوحَنّا 5:19; يُوحَنّا 4:34"> أنا لا أقدِرُ أنْ أعمَلَ شيئًا مِنْ عِندي.<br />فكما أسمَعُ مِنَ الآبِ أحكُمُ، وحُكمي عادِلٌ<br />لأنِّي لا أطلُبُ مَشيئَتي، بل مشيئَةَ الّذي أرسَلَني.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الأمثال 27:2; يُوحَنّا 8:54; رُؤيا يُوحَنّا 3:14"> لَو كُنتُ أشهَدُ لِنَفسي، لكانَت شَهادَتي باطِلَةً.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="يُوحَنّا 1:33; يُوحَنّا 12:28–30; يُوحَنّا 5:36–37"> ولكنَّ غَيري يَشهَدُ لي،<br />وأنا أعرِفُ أنَّ شَهادتَهُ صَحيحَةٌ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 1:29–34; يُوحَنّا 3:26–36; يُوحَنّا 1:15–27"> أرسَلْتُم رُسُلًا إلى يوحنَّا،<br />فشَهِدَ يوحنَّا لِلحقِّ.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="لُوقا 19:41–42; يُوحَنّا 20:31; رسالَةُ يُوحَنّا الأُولَى 5:9"> لا أعتمِدُ على شهادَةِ إنسانٍ،<br />ولكنِّي أقولُ هذا لِخلاصِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="رسالَةُ بُطرُسَ الثّانيَة 1:19; يُوحَنّا 1:7–8; ﻣﺘﻰ 11:11"> كانَ يوحنَّا سِراجًا مُنيرًا ساطِعًا،<br />فَرَضيتُم أنْ تَبتَهِجوا فَتْرَةً بِنُورِهِ.</span>
<span className="v" id="v36" data-v="36" data-xref="يُوحَنّا 10:25; يُوحَنّا 10:37–38; يُوحَنّا 15:24"> لي شهادَةٌ أعظمُ مِنْ شهادَةِ يوحنَّا:<br />لي أعمالي الّتي أعطاني الآبُ أنْ أعمَلَها،<br />وهذِهِ الأعمالُ الّتي أعمَلُها<br />هيَ نَفْسُها تَشهَدُ لي بأنَّ الآبَ أرسَلَني.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="يُوحَنّا 8:18; يُوحَنّا 1:18; ﻣﺘﻰ 3:17"> والآبُ الّذي أرسَلَني هوَ يَشهَدُ لي.<br />ما سَمِعْتُم صَوتَهُ مِنْ قَبلُ، ولا رأيتُم وجهَهُ،</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:14; التَّثْنِيَة 6:6–9; يُوحَنّا 12:44–48"> وكلامُهُ لا يَثبُتُ فيكُم،<br />لأنَّكُم لا تُؤمِنونَ بالّذي أرسَلَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="لُوقا 24:27; لُوقا 24:44; رسالَةُ بُطرُسَ الثّانيَة 1:19–21"> تَفحَصونَ الكُتُبَ المُقدَّسَةَ،<br />حاسبينَ أنَّ لكُم فيها الحياةَ الأبديَّةَ،<br />هيَ تَشهَدُ لي،</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 3:19; يُوحَنّا 1:11; المَزامِير 81:11"> ولكنَّكُم لا تُريدونَ أنْ تَجيئوا إليَّ لِتكونَ لكُمُ الحياةُ.</span>

<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="يُوحَنّا 7:18; يُوحَنّا 5:34; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:6"> أنا لا أطلُبُ مَجدًا مِنْ عِندِ النَّاسِ،</span>

<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:20; رسالَةُ يُوحَنّا الأُولَى 3:17; يُوحَنّا 2:25"> لأنِّي عَرَفْتُكُم فعَرَفْتُ أنَّ محبَّةَ اللهِ<br />لا محلَّ لها في قُلوبِكُم.</span>

</div>
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="يُوحَنّا 12:43; ﺭﻭﻣﻴﺔ 2:29; فِيلِبِّي 2:3"> وكيفَ تُؤمِنونَ ما دُمتُم تَطلُبونَ المَجدَ بَعضُكُم مِنْ بَعضٍ،<br />والمَجدُ الّذي هوَ مِنَ اللهِ الواحدِ لا تَطلُبونَهُ؟</span>
