## اعتقال يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 14:32; ﻣﺘﻰ 26:36; صموئيلَ الثّانِي 15:23"> قالَ يَسوعُ هذا الكلامَ وخرَجَ معَ تلاميذِهِ، فعبَرَ وادي قَدْرونَ وكانَ هُناكَ بُستانٌ، فدَخَلَهُ هوَ وتلاميذُهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 21:37; لُوقا 22:39; مَرْقُس 11:11–12"> وكانَ يَهوذا الّذي أسلَمَهُ يَعرِفُ هذا المكانَ، لأنَّ يَسوعَ كانَ يَجتَمِعُ فيهِ كثيرًا معَ تلاميذِهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 18:12; لُوقا 22:47–53; المَزامِير 22:12"> فجاءَ يَهوذا إلى هُناكَ بِجُنودٍ وحَرَسٍ أرسَلَهُم رُؤساءُ الكَهنَةِ والفَرِّيسيُّونَ، وكانوا يَحمِلونَ المَصابيحَ والمشاعِلَ والسِّلاحَ.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 13:1; يُوحَنّا 18:7; المَزامِير 27:3"> فتقَدَّمَ يَسوعُ وهوَ يَعرِفُ ما سيَحدُثُ لَه وقالَ لهُم: ”مَنْ تَطلُبونَ؟“</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 21:11; يُوحَنّا 19:19; ﻣﺘﻰ 2:23"> أجابوا: ”يَسوعَ النّاصِريَّ“. فقَالَ لهُم: ”أنا هوَ“.<br />وكانَ يَهوذا الّذي أسلَمَهُ واقِفًا معَهُم.</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 27:2; المَزامِير 70:2–3; المُلُوكِ الثّانِي 1:9–15"> فلمَّا قالَ لهُم يَسوعُ: ”أنا هوَ“، تَراجَعوا ووقَعوا إلى الأرضِ.</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 18:4"> فسألَهُم يَسوعُ ثانيَةً: ”مَنْ تَطلُبونَ؟“ أجابوا: ”يَسوعَ النّاصِريَّ“.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 10:28; ﻣﺘﻰ 26:56; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:9"> فقَالَ لهُم يَسوعُ: ”قُلتُ لكُم: أنا هوَ. فإذا كُنتُم تَطلُبوني، فَدَعُوا هَؤُلاءِ يَذهَبونَ“.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 17:12"> فتَمَّ ما قالَ يَسوعُ: ”ما خَسِرتُ أحدًا مِنَ الّذينَ وهَبتَهُم لي“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="مَرْقُس 14:47; ﻣﺘﻰ 26:51–54; لُوقا 22:49–51"> وكانَ سِمعانُ بُطرُسُ يَحمِلُ سَيفًا، فاَستَلَّهُ وضرَبَ خادِمَ رَئيسِ الكَهنَةِ فَقطَعَ أُذُنَهُ اليُمنى، وكانَ اَسمُ الخادِمِ مَلْخَسَ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 20:22; ﻣﺘﻰ 26:39; يُوحَنّا 20:17"> فقَالَ يَسوعُ لبُطرُسَ: ”رُدَّ سَيفَك إلى غِمدِهِ! ألا أشرَبُ كأسَ الآلامِ الّتي جَعَلها لي الآبُ“.</span>
</div>
</div>

## يسوع عند حنان وقيافا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 26:57; يُوحَنّا 18:3; اعمال الرسل 21:37"> فقَبَضَ الجنودُ وقائِدُهُم وحرَسُ الهَيكَلِ على يَسوعَ وقَيَّدوهُ</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 18:24; يُوحَنّا 11:51; ﻣﺘﻰ 26:3"> وأخذوهُ أوَّلًا إلى حَنّانَ، وهوَ حَمُو قَيافا رَئيسِ الكَهنَةِ في تِلكَ السَّنةِ.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 11:49–52"> وقَيافا هذا هوَ الّذي أشارَ على اليَهودِ فقَالَ: ”أنْ يَموتَ رَجُلٌ واحدٌ فِدى الشَّعبِ خَيرٌ لكُم“.</span>
</div>
</div>

## بطرس ينكر يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="لُوقا 22:54; مَرْقُس 14:54; ﻣﺘﻰ 26:58–68"> وكانَ سِمعانُ بُطرُسُ وتِلميذٌ آخَرُ يَتبَعانِ يَسوعَ. وكانَ هذا التِّلميذُ مَعروفًا مِنْ رَئيسِ الكَهنَةِ، فدخَلَ دارَ رَئيسِ الكَهنَةِ معَ يَسوعَ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 26:69; مَرْقُس 14:66–68; لُوقا 22:55–57"> أمَّا بُطرُسُ فوقَفَ في الخارِجِ عِندَ البابِ. وعادَ التِّلميذُ الآخرُ الّذي يَعرِفُهُ رَئيسُ الكَهنَةِ إلى الخارِجِ وكَلَّمَ الفتاةَ الّتي تَحرُسُ البابَ وأدخَلَ بُطرُسَ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 26:69–70; يُوحَنّا 21:15; لُوقا 22:56–57"> فقَالَتِ الفتاةُ لِبُطرُسَ: ”أما أنتَ أيضًا مِنْ تلاميذِ هذا الرَّجُلِ؟“ فأجابَها بُطرُسُ: ”ما أنا مِنهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="مَرْقُس 14:54; يُوحَنّا 18:25; مَرْقُس 14:67"> وكانَ البَرْدُ شَديدًا، فأوقَدَ الخَدَمُ والحرَسُ نارًا وأخذوا يتَدَفّأونَ، ووقَفَ بُطرُسُ يتَدَفّأُ معَهُم.</span>
</div>
</div>

## يسوع عند حنان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 14:55–65; لُوقا 20:20; لُوقا 22:63–71"> وسألَ رَئيسُ الكَهنَةِ يَسوعَ عَنْ تلاميذِهِ وتَعليمِهِ،</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 7:26; إشَعْياء 48:16; ﻣﺘﻰ 4:23"> فأجابَهُ يَسوعُ: ”كَلَّمْتُ النَّاسَ عَلانيةً، وعَلَّمتُ دائِمًا في المَجامِعِ وفي الهَيكَلِ حَيثُ يَجتَمِعُ اليَهودُ كلُّهُم، وما قُلتُ شيئًا واحدًا في الخِفيَةِ.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 26:59–60; اعمال الرسل 24:12–13; لُوقا 22:67"> فلِماذا تسأَلُني؟ إِسأَلِ الّذينَ سَمِعوني عمَّا كَلَّمتُهُم بِه، فَهُم يَعرِفونَ ما قُلتُ“.</span>
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 19:3; مَرْقُس 14:65; لُوقا 22:63–64"> فلمَّا قالَ يَسوعُ هذا الكلامَ، لَطَمَهُ واحِدٌ مِنَ الحَرَسِ كانَ بِجانِبِهِ وقالَ لَه: ”أهكذا تُجيبُ رَئيسَ الكَهنَةِ؟“</span>
<span className="v" id="v23" data-v="23" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:20–23; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:1; ﻣﺘﻰ 5:39"> فأجابَهُ يَسوعُ: ”إنْ كُنتُ أخطأتُ في الكلامِ، فقُلْ لي أينَ الخَطَأُ؟ وإنْ كُنتُ أصَبْتُ، فلِماذا تَضرِبُني؟“</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 18:13; ﻣﺘﻰ 26:57"> فأرسلَهُ حنّانُ مُوثَقًا إلى قَيافا رَئيسِ الكَهنَةِ.</span>
</div>
</div>

## بطرس ينكر المسيح ثانيةً

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="مَرْقُس 14:67–72; يُوحَنّا 18:17–18; التَّكوين 18:15"> وبَينَما سِمعانُ بُطرُسُ واقِفٌ هُناكَ يَتَدَفّأُ قالوا لَه: ”أما أنتَ مِنْ تلاميذِهِ؟“ فأنكَرَ وقالَ: ”ما أنا مِنهُم!“</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 18:1; يُوحَنّا 18:10; ﻣﺘﻰ 26:73"> فقَالَ لَه واحِدٌ مِنْ خَدَمِ رَئيسِ الكَهنَةِ، وكانَ مِنْ أقرباءِ الرَّجُلِ الّذي قطَعَ بُطرُسُ أذنَهُ: ”أما رأيتُكَ معَهُ في البُستانِ؟“</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 13:38; مَرْقُس 14:71–72; لُوقا 22:60–62"> فأنكَرَ أيضًا وفي الحالِ صاحَ الدّيكُ.</span>
</div>
</div>

## يسوع عند بيلاطس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 18:33; يُوحَنّا 19:9; ﻣﺘﻰ 27:27"> وأخذوا يَسوعَ مِنْ عِندِ قَيافا إلى قَصرِ الحاكِمِ. وكانَ الوقتُ صَباحًا. فاَمتَنَعَ اليَهودُ مِنْ دُخولِ القَصرِ لِئَلَّا يَتنَجَّسوا، فلا يتَمكَّنوا مِنْ أكلِ عَشاءِ الفِصحِ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 27:23; مَرْقُس 15:2–5; ﻣﺘﻰ 27:11–14"> فخَرَجَ إلَيهِم بِيلاطُسُ وسألَهُم: ”بِماذا تَتَّهِمونَ هذا الرَّجُلَ؟“</span>
<span className="v" id="v30" data-v="30" data-xref="اعمال الرسل 3:13; لُوقا 20:19–26; مَرْقُس 10:33"> فأجابوا: ”لَولا أنَّهُ مُجرِمٌ، لما أسلَمْناهُ إلَيكَ“.</span>
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 19:6–7; يُوحَنّا 19:15; اعمال الرسل 25:18–20"> فقَالَ لهُم بِيلاطُسُ: ”خُذوهُ أنتُم وحاكِموهُ حسَبَ شريعَتِكُم“. فأجابوا: ”لا يَجوزُ لنا أنْ نَحكُمَ على أحدٍ بالقَتْلِ“.</span>
<span className="v" id="v32" data-v="32" data-xref="يُوحَنّا 3:14; ﻣﺘﻰ 20:19; ﻣﺘﻰ 26:2"> فتَمَّ ما قالَ يَسوعُ مُشيرًا إلى المِيتَةِ الّتي يَموتُها.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="لُوقا 23:3–4; يُوحَنّا 18:37; يُوحَنّا 19:3"> فعادَ بِيلاطُسُ إلى قَصرِ الحاكِمِ ودَعا يَسوعَ وقالَ لَه: ”أأنتَ مَلِكُ اليَهودِ؟“</span>
<span className="v" id="v34" data-v="34"> فأجابَهُ يَسوعُ: ”أتَقولُ هذا مِنْ عِندِكَ، أمْ قالَهُ لكَ آخَرونَ؟“</span>
<span className="v" id="v35" data-v="35" data-xref="اعمال الرسل 18:14–16; نَحَمْيا 4:2; ﺭﻭﻣﻴﺔ 3:1–2"> فقَالَ بِيلاطُسُ: ”أيهودِيٌّ أنا؟ شَعبُكَ ورُؤساءُ الكَهنَةِ أسلَموكَ إليَّ. فماذا فعَلْتَ؟“</span>
<span className="v" id="v36" data-v="36" data-xref="لُوقا 17:20–21; كُولُوسِي 1:12–14; ﺭﻭﻣﻴﺔ 14:17"> أجابَهُ يَسوعُ: ”ما مَملكَتي مِنْ هذا العالَمِ. لَو كانَت مَملكَتي مِنْ هذا العالَمِ، لَدافَعَ عنِّي أتباعي حتّى لا أُسلَمَ إلى اليَهودِ. لا! ما مَملكَتي مِنْ هُنا“.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:6; يُوحَنّا 8:47; يُوحَنّا 8:14"> فقَالَ لَه بِيلاطُسُ: ”أمَلِكٌ أنتَ، إذَنْ؟“ أجابَهُ يَسوعُ: ”أنتَ تَقولُ إنِّي مَلِكٌ. أنا وُلِدْتُ وجِئتُ إلى العالَمِ حتّى أشهَدَ لِلحَقِّ. فمَنْ كانَ مِنْ أبناءِ الحَقِّ يَستَمِعُ إلى صَوتي“.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 19:4; لُوقا 23:4; يُوحَنّا 19:6"> فقَالَ لَه بِيلاطُسُ: ”ما هوَ الحقُّ؟“ قالَ هذا وخرَجَ ثانيَةً إلى اليَهودِ وقالَ لهُم: ”لا أجِدُ سَبَبًا لِلحُكمِ علَيهِ.</span>
<span className="v" id="v39" data-v="39" data-xref="ﻣﺘﻰ 27:15–18; لُوقا 23:16–23; مَرْقُس 15:6–14"> ولكِنَّ العادَةَ عِندَكُم أنْ أُطلِقَ لكُم سَجينًا في عيدِ الفِصحِ. أتُريدونَ أنْ أُطلِقَ لكُم مَلِكَ اليَهودِ؟“</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 23:25; مَرْقُس 15:15; ﻣﺘﻰ 27:16"> فصاحوا: ”لا تُطلِقْ هذا، بل باراباسَ“. وكانَ باراباسُ لِصًّا.</span>
</div>
