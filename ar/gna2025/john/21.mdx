## يسوع يظهر لسبعة من تلاميذه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 6:1; يُوحَنّا 6:23; ﻣﺘﻰ 28:16"> وظهَرَ يَسوعُ لتلاميذِهِ مَرَّةً أُخرى على شاطئِ بُحيرةِ طبَريَّةَ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 4:21–22; يُوحَنّا 1:45–1; يُوحَنّا 4:46"> وكانَ ذلِكَ حينَ اَجتمَعَ سِمعانُ بُطرُسُ، وتوما المُلَقَّبُ بالتَوأمِ، ونَثَنائِيلُ الّذي مِنْ قانا الجليلِ، واَبنا زَبَدي، وتِلميذانِ آخَرانِ مِنْ تلاميذِ يَسوعَ.</span>
<span className="v" id="v3" data-v="3" data-xref="لُوقا 5:5; ﻣﺘﻰ 4:18–20; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:7–9"> فقَالَ لهُم سِمعانُ بُطرُسُ: ”أنا ذاهِبٌ لِلصَّيدِ“. فقَالوا لَه: ”ونَحنُ نَذهَبُ معَكَ“. فخَرَجوا وركِبوا القارِبَ، ولكِنَّهُم في تِلكَ اللَّيلَةِ ما أمسَكوا شيئًا مِنَ السَّمَكِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 20:14; لُوقا 24:15–16; لُوقا 24:31"> وفي الصَّباحِ وقَفَ يَسوعُ على الشَّاطئِ، فما عَرَفَ التَّلاميذُ أنَّهُ يَسوعُ.</span>
<span className="v" id="v5" data-v="5" data-xref="فِيلِبِّي 4:19; فِيلِبِّي 4:11–13; رسالَةُ يُوحَنّا الأُولَى 2:13"> فقَالَ لهُم: ”أيُّها الشُّبانُ، أمَعكُم شَيءٌ يُؤكلُ؟“ فأجابوهُ: ”لا“.</span>
<span className="v" id="v6" data-v="6" data-xref="لُوقا 5:4–7; اعمال الرسل 2:41; يُوحَنّا 2:5"> قالَ لهُم: ”ألقُوا الشَّبكةَ إلى يَمينِ القارِبِ تَجدوا سَمكًا“. فألقَوا الشَّبكَةَ وما قَدِروا أنْ يُخرِجوها، لِكَثرَةِ ما فيها مِنَ السَّمكِ.</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 13:23; يُوحَنّا 20:2; ﻣﺘﻰ 14:28–29"> فقَالَ التِّلميذُ الّذي كانَ يُحبُّهُ يَسوعُ لِبُطرُسَ: ”هذا هوَ الرَّبُّ!“ فلمَّا سَمِعَ سِمعانُ بُطرُسُ قولَهُ: ”هذا هوَ الرَّبُّ“، لَبِسَ ثَوبَهُ لأنَّهُ كانَ عُريانًا، وألقى نَفسَه في الماءِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 3:11"> وجاءَ التَّلاميذُ الآخَرونَ بالقارِبِ، يَجُرُّونَ الشَّبكَةَ بِما فيها مِنَ السَّمَكِ، وكانوا لا يَبعُدونَ إلَّا مئتَي ذِراعٍ عَنِ البَرِّ.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 21:13; المُلُوكِ الأوَّلُ 19:5–6; يُوحَنّا 21:10"> فلمَّا نَزَلوا إلى البَرِّ رأَوا جَمرًا علَيهِ سمَكٌ، وخُبزًا.</span>
<span className="v" id="v10" data-v="10"> فقَالَ لهُم يَسوعُ: ”هاتوا مِنَ السَّمَكِ الّذي أمسكْتُموهُ الآنَ“.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 5:6–8; اعمال الرسل 2:41"> فصَعِدَ سِمعانُ بُطرُسُ إلى القارِبِ وجذَبَ الشَّبكةَ إلى البَرِّ وكانَت اَمتَلأت بِمئَةٍ وثلاثٍ وخَمسينَ سَمكَةً كَبيرةً، مِنْ دونِ أنْ تتَمزَّقَ معَ هذا العَدَدِ الكثيرِ.</span>
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 10:41; لُوقا 9:45; يُوحَنّا 16:19"> فقَالَ لهُم يَسوعُ: ”تعالَوا كُلُوا!“ وما جَرُؤَ أحَدٌ مِنَ التَّلاميذِ أنْ يسألَهُ مَنْ أنتَ، لأنَّهُم عَرَفوا أنَّهُ الرَّبُّ.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 21:9; اعمال الرسل 10:41; لُوقا 24:42–43"> ودَنا يَسوعُ، فأخَذَ الخُبزَ وناوَلَهُم، وكذلِكَ ناوَلَهُم مِنَ السَّمَكِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 20:26; يُوحَنّا 20:19"> هذِهِ مرَّةٌ ثالِثَةٌ ظهَرَ فيها يَسوعُ لتلاميذِهِ بَعدَ قيامتِهِ مِنْ بَينِ الأمواتِ.</span>
</div>
</div>

## يسوع وبطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 40:11; رسالَةُ يُوحَنّا الأُولَى 4:19; ﺭﻭﻣﻴﺔ 15:1"> وبَعدَما أكلُوا، قالَ يَسوعُ لِسِمعانَ بُطرُسَ: ”يا سِمعانُ بنَ يوحنَّا، أتُحبُّني أكثرَ مِما يُحبُّني هَؤُلاءِ؟“ فأجابَهُ: ”نعم، يا ربُّ. أنتَ تَعرِفُ أنِّي أحِبُّكَ“. فقَالَ لَه: ”إرعَ خِرافي“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 20:28; رسالَةُ بُطرُسَ الأُولَى 5:2; رسالَةُ بُطرُسَ الأُولَى 2:25"> وسألَهُ مرَّةً ثانيَةً: ”يا سِمعانُ بنَ يوحنَّا، أتُحِبُّني؟“ فأجابَهُ: ”نعم، يا ربُّ، أنتَ تَعرِفُ أنِّي أُحِبُّكَ“. فقَالَ لَه: ”إرعَ خِرافي“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 13:38; يُوحَنّا 16:30; مَرْقُس 14:72"> وسألَهُ مرَّةً ثالِثَةً: ”يا سِمعانُ بنَ يوحنَّا، أتُحِبُّني؟“ فحَزِنَ بُطرُسُ لأنَّ يَسوعَ سألَهُ مرَّةً ثالِثَةً: أتُحِبُّني؟ فقَالَ: ”يا ربُّ، أنتَ تَعرِفُ كُلَّ شيءٍ، وتَعرِفُ أنِّي أحِبُّكَ“. قال لَه يَسوعُ: ”إرعَ خِرافي.</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 13:36; اعمال الرسل 12:3–4; اعمال الرسل 21:11"> الحقَّ الحقَّ أقولُ لكَ: كُنتَ، وأنتَ شابٌّ، تَشُدُّ حِزامَكَ بِيَدَيكَ وتذهَبُ إلى حَيثُ تُريدُ. فإذا صِرتَ شيخًا مَدَدْتَ يَدَيكَ وشَدَّ غيرُكَ لكَ حِزامَكَ وأخذَكَ إلى حيثُ لا تُريدُ“.</span>
<span className="v" id="v19" data-v="19" data-xref="رسالَةُ بُطرُسَ الثّانيَة 1:14; يُوحَنّا 21:22; ﻣﺘﻰ 10:38"> بِهذا الكلامِ أشارَ يَسوعُ إلى المِيتَةِ الّتي سيَموتُها بُطرُسُ، فيُمَجِّدُ بِها اللهَ. ثُمَّ قالَ لَه: ”إِتبَعْني“.</span>
</div>
</div>

## التلميذ الّذي كان يحبه يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 21:7; يُوحَنّا 20:2; يُوحَنّا 21:24"> واَلتَفتَ بُطرُسُ، فرَأى التِّلميذَ الّذي كانَ يُحبُّهُ يَسوعُ يَمشي خَلفَهُما، وهوَ الّذي مالَ على صَدرِ يَسوعَ وقتَ العَشاءِ وقالَ لَه: ”يا ربُّ، مَنِ الّذي سيُسلِّمُكَ؟“</span>
<span className="v" id="v21" data-v="21" data-xref="لُوقا 13:23–24; اعمال الرسل 1:6–7; ﻣﺘﻰ 24:3–4"> فلمَّا رآهُ بُطرُسُ قالَ لِيَسوعَ: ”يا ربُّ، وهذا ما هوَ مَصيرُهُ؟“</span>
<span className="v" id="v22" data-v="22" data-xref="رُؤيا يُوحَنّا 2:25; رُؤيا يُوحَنّا 3:11; مَرْقُس 9:1"> فأجابَهُ يَسوعُ: ”لَو شِئتُ أنْ يَبقى إلى أنْ أَجيءَ، فماذا يَعنيكَ؟ إتبَعْني أنتَ!“</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 29:29; أيُّوب 33:13; دانيال 4:35"> فشاعَ بَينَ الإخوةِ أنَّ هذا التِّلميذَ لا يَموتُ، معَ أنَّ يَسوعَ ما قالَ لبُطرُسَ إنَّهُ لا يَموتُ، بل قالَ لَه: ”لَو شِئتُ أنْ يَبقى إلى أنْ أَجيءَ، فَماذا يَعنيكَ؟“</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 19:35; يُوحَنّا 15:27; رسالَةُ يُوحَنّا الأُولَى 5:6"> وهذا التِّلميذُ هوَ الّذي يَشهَدُ بِهذِهِ الأمورِ ويُدوِّنُها، ونَحنُ نَعرِفُ أنَّ شَهادَتَهُ صادِقَةٌ.</span>
</div>
</div>
