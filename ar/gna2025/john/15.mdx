## يسوع الكرمة الحقيقية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 2:21; إشَعْياء 27:2–3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:9"> ”أنا الكَرمَةُ الحَقيقيَّةُ وأبي الكرّامُ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 3:10; العِبرانِيّين 12:10–11; ﻣﺘﻰ 15:13"> كُلُّ غُصنٍ مِنِّي لا يَحمِلُ ثَمرًا يَقطَعُه، وكُلُّ ما يُثمِرُ يُنقِّيهِ ليَكثُرَ ثَمَرُهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 17:17; أفَسُس 5:26; رسالَةُ بُطرُسَ الأُولَى 1:22"> أنتُمُ الآنَ أنقياءُ بِفَضلِ ما كَلَّمتُكُم بِه.</span>
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:6; غَلاطِيَّة 2:20; يُوحَنّا 8:31"> أُثبتوا فيَّ وأنا فيكُم. وكما أنَّ الغُصنَ لا يُثمِرُ مِنْ ذاتِهِ إلَّا إذا ثَبَتَ في الكرمَةِ، فكذلِكَ أنتُم: لا تُثمِرونَ إلَّا إذا ثَبَتُّم فيَّ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="فِيلِبِّي 4:13; كُولُوسِي 1:10; يُوحَنّا 15:16"> أنا الكرمَةُ وأنتُمُ الأغصانُ: مَنْ ثبَتَ فِيَّ وأنا فيهِ يُثمِرُ كثيرًا. أمَّا بِدوني فلا تَقدِرونَ على شَيءٍ.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 7:19; العِبرانِيّين 6:7–8; ﻣﺘﻰ 13:41"> مَنْ لا يَثبُتُ فيَّ يُرمى كالغُصْنِ فيَيبَسُ. والأغصانُ اليابِسَةُ تُجمَعُ وتُطرَحُ في النَّارِ فتَحتَرِقُ.</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 37:4; رسالَةُ يُوحَنّا الأُولَى 5:14; يُوحَنّا 14:13"> إذا ثَبَتُّم فيَّ وثبَتَ كلامي فيكُم، تَطلُبونَ ما تَشاؤونَ فتنالونَهُ.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 8:31; ﻣﺘﻰ 5:16; فِيلِبِّي 1:11"> بِهذا يَتمَجَّدُ أبي: أنْ تَحمِلوا ثَمَرًا كثيرًا فتكونوا تلاميذي.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 17:23–24; يُوحَنّا 17:26; رسالَةُ يُوحَنّا الأُولَى 2:28"> أنا أحِبُّكُم مِثلَما أحبَّني الآبُ، فاَثبُتوا في مَحبَّتي.</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 14:15; رسالَةُ يُوحَنّا الأُولَى 5:3; رسالَةُ يُوحَنّا الأُولَى 3:21–24"> إذا عَمِلتُم بِوصايايَ تَثبُتونَ في مَحَبَّتي، كما عَمِلتُ بوصايا أبي وأثبُتُ في مَحبَّتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 17:13; ﺭﻭﻣﻴﺔ 15:13; يُوحَنّا 16:24"> قُلتُ لكُم هذا ليَدومَ فيكُم فَرَحي، فيكونُ فرَحُكُم كامِلًا.</span>
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:21; رسالَةُ بُطرُسَ الأُولَى 4:8; يُوحَنّا 13:34"> هذِهِ هِيَ وصِيَّتي: أحِبُّوا بَعضُكُم بَعضًا مِثلَما أحبَبْتُكم.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 10:11; أفَسُس 5:2; ﺭﻭﻣﻴﺔ 5:6–8"> ما مِنْ حُبٍّ أعظَمَ مِنْ هذا: أنْ يُضَحِّيَ الإنسانُ بِنَفسِهِ في سَبيلِ أحبّائِهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 12:50; رسالَةُ يُوحَنّا الأُولَى 5:3; يَعقُوب 2:23"> وأنتُم أحبّائي إذا عَمِلتُم بِما أوصِيكُم بِه.</span>
<span className="v" id="v15" data-v="15" data-xref="المَزامِير 25:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:9–12; ﻣﺘﻰ 13:11"> أنا لا أدعوكُم عَبيدًا بَعدَ الآنَ، لأنَّ العَبدَ لا يَعرِفُ ما يَعمَلُ سيِّدُهُ، بل أدعوكُم أحِبّائي، لأنِّي أخبَرتُكُم بِكُلِّ ما سَمِعتُهُ مِنْ أبي.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 15:19; إرْمِيا 1:5–7; أفَسُس 2:10"> ما اَختَرْتُموني أنتُم، بل أنا اَختَرْتُكُم وأقَمْتُكُم لِتَذهبوا وتُثْمِروا ويَدومَ ثَمَرُكُم، فيُعطِيَكُمُ الآبُ كُلَّ ما تَطلُبونَهُ باَسمي.</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 15:12; رسالَةُ يُوحَنّا الأُولَى 3:14–17; رسالَةُ بُطرُسَ الأُولَى 2:17"> وهذا ما أوصيكُم بِه: أنْ يُحِبَّ بَعضُكُم بَعضًا.</span>
</div>

## العالم ويسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:13; يُوحَنّا 7:7; مَرْقُس 13:13"> ”إنْ أبغَضَكُمُ العالَمُ، فتَذكَّروا أنَّهُ أبغَضَني قَبلَ أنْ يُبغِضَكُم.</span>
<span className="v" id="v19" data-v="19" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:4–5; يُوحَنّا 17:14–16; أفَسُس 1:4–11"> لَو كُنتُم مِنَ العالَمِ، لأحبَّكُمُ العالَمُ كأهلِهِ. ولأنِّي اَختَرْتُكم مِنْ هذا العالَمِ وما أنتُم مِنهُ، لذلِكَ أبغَضَكُمُ العالَمُ.</span>
<span className="v" id="v20" data-v="20" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:12; يُوحَنّا 13:16; ﻣﺘﻰ 10:24"> تَذكَّروا ما قُلتُهُ لكُم: ما كانَ خادِمٌ أعظَمَ مِنْ سيِّدِهِ. فإذا اَضطَهَدوني يَضطَهِدونَكُم، وإذا سَمِعوا كلامي يَسمَعونَ كلامَكُم.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 10:22; يُوحَنّا 16:3; ﻣﺘﻰ 10:18"> هُم يَفعَلونَ بكُم هذا كُلَّهُ مِنْ أجلِ اَسمي، لأنَّهُم لا يَعرِفونَ الّذي أرسَلَني.</span>
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 9:41; يَعقُوب 4:17; اعمال الرسل 17:30"> لَو أنِّي ما جِئْتُ وكَلَّمْتُهُم، لما كانَت علَيهِم خَطيئةٌ. أمَّا الآنَ، فلا عُذْرَ لهُم مِنْ خَطيئَتِهِم.</span>
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 8:40–42; رسالَةُ يُوحَنّا الثّانيَة 1:9; رسالَةُ يُوحَنّا الأُولَى 2:23"> مَنْ أبغَضَني أبغَضَ أبي.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 5:36; يُوحَنّا 14:9; يُوحَنّا 7:31"> لَو أنِّي ما عَمِلتُ بَينَهُم أعمالًا ما عَمِلَ مِثلَها أحدٌ، لما كانَت لهُم خَطيئَةٌ. لكِنَّهُمُ الآنَ رَأوا، ومعَ ذلِكَ أبغَضوني وأبغَضوا أبي.</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 109:3; المَزامِير 69:4; المَزامِير 35:19"> وكانَ هذا ليَتِمَّ ما جاءَ في شَريعَتِهِم: أبغَضوني بِلا سَبَبٍ.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 14:26; يُوحَنّا 14:16–17; يُوحَنّا 16:13–15"> ومتى جاءَ المُعَزِّي الّذي أُرسِلُهُ إلَيكُم مِنَ الآبِ، رُوحُ الحقِّ المُنبَثِقُ مِنَ الآبِ، فهوَ يَشهَدُ لي.</span>
<span className="v" id="v27" data-v="27" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:14; يُوحَنّا 21:24; اعمال الرسل 10:39–42"> وأنتُم أيضًا ستَشهَدونَ، لأنَّكُم مِنَ البَدءِ مَعي.</span>
