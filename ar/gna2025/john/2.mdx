## عرس قانا الجليل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 4:46; الأمثال 31:10–12; يُوحَنّا 21:2"> وفي اليومِ الثَّالثِ كانَ في قانا الجَليلِ عُرسٌ، وكانَت أُمُّ يَسوعَ هُناكَ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 1:40–49; ﻣﺘﻰ 10:40–42; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:39"> فدُعيَ يَسوعُ وتلاميذُهُ إلى العُرسِ.</span>
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 104:15; فِيلِبِّي 4:6; إشَعْياء 24:11"> ونفَدَتِ الخَمرُ، فقَالَت لَه أُمُّهُ: ”ما بَقِيَ عِندَهُم خمرٌ“.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 7:6; يُوحَنّا 13:1; يُوحَنّا 7:30"> فأجابَها: ”ما لي ولَكِ، يا اَمرأةُ، ما جاءَت ساعَتي بَعدُ“.</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 15:14; التَّكوين 6:22; العِبرانِيّين 11:8"> فقَالَت أُمُّهُ لِلخدَمِ: ”إعمَلوا ما يقولُ لَكُم“.</span>
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 3:25; العِبرانِيّين 9:10; أفَسُس 5:26"> وكانَ هُناكَ سِتَّةُ أجرانٍ مِنْ حجَرٍ يتَطَهَّرُ اليَهودُ بِمائِها على عادَتِهِم، يسَعُ كُلُّ واحدٍ مِنها مِقدارَ مكيالَينِ أو ثلاثَةٍ.</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الثّانِي 4:2–6; يُوحَنّا 2:5; يُوحَنّا 2:3"> فقَالَ يَسوعُ لِلخدَمِ: ”إملأُوا الأجرانَ بالماءِ“. فمَلأوها حتّى فاضَت</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 2:9; الأمثال 3:5–6; ﺭﻭﻣﻴﺔ 13:7"> فقَالَ لهُم: ”إِستَقوا الآنَ وناوِلوا رَئيسَ الوَليمَةِ“. فناوَلُوهُ.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 4:46; يُوحَنّا 7:17; المَزامِير 119:100"> فلمَّا ذاقَ الماءَ الّذي صارَ خَمرًا، وكانَ لا يَعرِفُ مِنْ أينَ جاءَتِ الخَمرُ، لكِنَّ الخدَمَ الّذينَ اَستَقَوا مِنهُ كانوا يَعرِفونَ، دَعا العَريسَ</span>
<span className="v" id="v10" data-v="10" data-xref="رُؤيا يُوحَنّا 7:16–17; الأمثال 9:1–6; لُوقا 16:25"> وقالَ لَه: ”جَميعُ النَّاسِ يُقدِّمونَ الخَمرَ الجيِّدةَ أوّلًا، حتّى إذا سكِرَ الضُّيوفُ، قَدَّموا الخَمرَ الرَّديئةَ. أمَّا أنتَ فأخَّرتَ الخَمرَ الجيِّدةَ إلى الآنَ!“</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 20:30–31; يُوحَنّا 1:14; يُوحَنّا 3:2"> هذِهِ أُولى آياتِ يَسوعَ، صنَعَها في قانا الجَليلِ. فأظهَرَ مَجدَهُ، فآمَنَ بِه تلاميذُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 12:46; ﻣﺘﻰ 13:55–56; مَرْقُس 6:3"> ونزَلَ يَسوعُ بَعدَ ذلِكَ إلى كَفْرناحومَ ومعَهُ أُمُّهُ وإخوَتُهُ وتلاميذُهُ، فأقاموا فيها أيّامًا قَليلةً.</span>
</div>
</div>

## يسوع يطرد الباعة من الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="لُوقا 2:41; يُوحَنّا 11:55; يُوحَنّا 2:23"> واَقترَبَ عِيدُ الفِصحِ عِندَ اليَهودِ، فصَعِدَ يَسوعُ إلى أُورُشليمَ.</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 19:45–46; مَرْقُس 11:15; مَرْقُس 11:17"> ورأى في الهَيكَلِ باعَةَ البقَرِ والغنَمِ والحَمامِ، والصَّيارِفَةَ جالِسينَ إلى مناضِدِهِم،</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 18:6; زَكَريّا 4:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:4"> فجَدَلَ سَوطًا مِنْ حِبالٍ وطرَدَهُم كُلَّهُم مِنَ الهَيكَلِ معَ الغنَمِ والبقَرِ، وبَعثَرَ نُقودَ الصَّيارِفَةِ وقَلَبَ مناضِدَهُم،</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 21:13; إرْمِيا 7:11; مَرْقُس 11:17"> وقالَ لِباعَةِ الحَمامِ: ”إِرفَعوا هذا مِنْ هُنا، ولا تَجعَلوا مِنْ بَيتِ أبي بَيتًا لِلتِّجارَةِ“.</span>
<span className="v" id="v17" data-v="17" data-xref="المَزامِير 69:9; المَزامِير 119:139"> فتذكَّرَ تلاميذُهُ هذِهِ الآيةَ: ”الغَيرةُ على بَيتِكَ، يا اللهُ، ستأكُلُني“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 21:23; مَرْقُس 8:11; مَرْقُس 11:27–28"> فقَالَ لَه اليَهودُ: ”أرِنا آيةً تُجيزُ عَملَكَ هذا؟“</span>
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 14:58; مَرْقُس 15:29; ﻣﺘﻰ 27:40"> فأجابَهُم يَسوعُ: ”إِهْدِمُوا هذا الهَيكَلَ، وأنا أبْنِيهِ في ثلاثةِ أيامٍ“.</span>
<span className="v" id="v20" data-v="20"> فقَالَ اليَهودُ: ”بُنِيَ هذا الهَيكَلُ في سِتٍّ وأربعينَ سنَةً، فكيفَ تَبنيهِ أنتَ في ثلاثَةِ أيّامٍ؟“</span>
<span className="v" id="v21" data-v="21" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:16; أفَسُس 2:20–22"> وكانَ يَسوعُ يَعني بِالهَيكَلِ جسَدَهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 12:16; يُوحَنّا 16:4; لُوقا 24:7–8"> فلمَّا قامَ مِنْ بَينِ الأمواتِ، تذكَّرَ تلاميذُهُ هذا الكلامَ، فآمَنوا بالكِتابِ والكلامِ الّذي قالَهُ يَسوعُ.</span>
</div>
</div>

## المسيح يعرف البشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 7:31; لُوقا 8:13; يُوحَنّا 6:14"> ولمَّا كانَ في أُورُشليمَ مُدَّةَ عيدِ الفِصحِ، آمنَ بِه كثيرٌ مِنَ النَّاسِ حينَ رأَوا الآياتِ الّتي صَنَعَها.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 6:64; العِبرانِيّين 4:13; اعمال الرسل 1:24"> لكِنَّ يَسوعَ ما اَطمأَنَّ إلَيهِم، لأنَّهُ كانَ يَعرِفُهُم كُلَّهُم،</span>
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 13:11; ﻣﺘﻰ 9:4; يُوحَنّا 6:64"> ولا يَحتاجُ إلى مَنْ يُخبِرُهُ عَنْ أحَدٍ، لأنَّهُ كانَ يَعلَمُ ما في قَلبِ الإنسانِ.</span>
