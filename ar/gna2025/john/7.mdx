## يسوع وإخوته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 5:16–18; يُوحَنّا 4:3; ﻣﺘﻰ 21:38"> وسارَ يَسوعُ بَعدَ ذلِكَ في الجليلِ، وما شاءَ أنْ يَسيرَ في اليَهوديَّةِ، لأنَّ اليَهودَ كانوا يُريدونَ أنْ يَقتُلوهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="زَكَريّا 14:16–19; التَّثْنِيَة 16:13–16; الخُرُوج 23:16–17"> ولمَّا اَقتَرَبَ عِيدُ المظالِّ عِندَ اليَهودِ،</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 12:46; يُوحَنّا 7:5; لُوقا 8:19"> قالَ لَه إخوتُهُ: ”أُترُكْ هذا المكانَ واَذهَبْ إلى بلادِ اليَهوديَّةِ حتّى يَرى التَّلاميذُ أعمالَكَ،</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 18:20; الأمثال 18:1–2; ﻣﺘﻰ 23:5"> فلا أحَدٌ يَعمَلُ في الخِفيةِ إذا أرادَ أنْ يَعرِفَهُ النَّاسُ. وما دُمتَ تَعمَلُ هذِهِ الأعمالَ، فأَظهِرْ نَفسَكَ للعالَمِ“.</span>
<span className="v" id="v5" data-v="5" data-xref="مَرْقُس 3:21; يُوحَنّا 1:11–13; ميخا 7:5–6"> وكانَ إخوتُهُ أنفُسُهُم لا يُؤمِنونَ بِه.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 7:8; يُوحَنّا 2:4; يُوحَنّا 7:30"> فقَالَ لهُم يَسوعُ: ”ما جاءَ وقتي بَعدُ. وأمَّا أنتُم، فالوَقتُ في كُلِّ حينٍ وَقتُكُم.</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 15:18–19; يُوحَنّا 3:19; غَلاطِيَّة 4:16"> أنتُم لا يُبغِضُكُمُ العالَمُ، ولكنَّهُ يُبغِضُني لأنِّي أشهَدُ على فسادِ أعمالِهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 7:6; يُوحَنّا 8:20; يُوحَنّا 7:30"> إِصْعَدُوا أنتُم إلى العيدِ، فأنا لا أصعَدُ إلى هذا العيدِ، لأنَّ وَقتي ما جاءَ بَعدُ“.</span>
<span className="v" id="v9" data-v="9"> قالَ لهُم هذا وبَقِيَ في الجليلِ.</span>
</div>
</div>

## يسوع في عيد المظال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="عامُوس 5:13; ﻣﺘﻰ 10:16; إشَعْياء 42:2–3"> ولمَّا صَعِدَ إخوتُهُ إلى العيدِ، صَعِدَ بَعدَهُم في الخِفيَةِ لا في العَلانيَةِ.</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 11:56"> فكانَ اليَهودُ يَبحثونَ عَنهُ في العيدِ ويسألونَ: ”أينَ هوَ؟“</span>
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 7:40–43; يُوحَنّا 9:16; يُوحَنّا 7:32"> وتَهامَسَ النَّاسُ علَيهِ، فقَالَ بَعضُهُم: ”هوَ رَجُلٌ صالِحٌ“. وقالَ آخرونَ: ”لا، هوَ يُضلِّلُ الشَّعبَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 9:22; يُوحَنّا 19:38; يُوحَنّا 20:19"> وما تَحدَّثَ عَنهُ أحدٌ جَهارًا خَوفًا مِنْ رُؤساءِ اليَهودِ.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 7:28; لُوقا 19:47; يُوحَنّا 7:37"> وفي مُنتَصَفِ أيّامِ العيدِ، صَعِدَ يَسوعُ إلى الهَيكَلِ وأخَذَ يُعَلِّمُ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 4:22; ﻣﺘﻰ 7:28–29; ﻣﺘﻰ 13:54"> فتَعجَّبَ اليَهودُ وقالوا: ”كيفَ يَعرِفُ الكُتُبَ المُقَدَّسَةَ، وما تَعَلَّمَ؟“</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 14:10; يُوحَنّا 14:24; يُوحَنّا 17:8"> فأجابَهُم يَسوعُ: ”ما تَعليمي مِنْ عِندي، بل مِنْ عِندِ الّذي أرسَلَني.</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 8:31–32; يُوحَنّا 8:43; إرْمِيا 31:33–34"> إذا أرادَ أحدٌ أنْ يَعمَلَ بِمَشيئةِ اللهِ، عرَفَ هَلْ هذا التَّعليمُ مِنْ عِندِ اللهِ أو أنِّي أتكَلَّمُ مِنْ عِندي.</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 5:41; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:31–33; يُوحَنّا 13:31–32"> فالّذي يتكَلَّمُ مِنْ عندِهِ يَطلبُ المجدَ لنَفسِهِ، ولكِنْ مَنْ يَطلبُ المَجدَ لِلَّذي أرسَلَهُ، فهوَ صادِقٌ لا غُشَّ فيهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 1:17; يُوحَنّا 11:53; ﻣﺘﻰ 12:14"> أما أعطاكُم موسى الشَّريعةَ؟ ولا أحَدَ مِنكُم يَعمَلُ بِها. لماذا تُريدونَ أنْ تَقتُلوني؟“</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 10:20; يُوحَنّا 8:48; يُوحَنّا 8:52"> فأجابَتِ الجُموعُ: ”أنتَ فيكَ شَيطانٌ، فمَنْ يُريدُ أنْ يَقتُلَكَ؟“</span>
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 7:23; يُوحَنّا 5:2–11"> فقَالَ يَسوعُ: ”ما عَمِلتُ إلَّا عَملًا واحدًا، فتَعَجَّبتُم كُلُّكُم.</span>
<span className="v" id="v22" data-v="22" data-xref="اللّاويِّين 12:3; التَّكوين 17:10–14; ﺭﻭﻣﻴﺔ 4:9–11"> أمَركُم موسى بالخِتانِ، وما كانَ الخِتانُ مِنْ موسى بل مِنَ الآباءِ، فأخَذتُم تَختُنونَ الإنسانَ يومَ السَّبتِ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 12:5; يُوحَنّا 5:14–16; ﻣﺘﻰ 12:2"> فإذا كُنتُم تَختُنونَ الإنسانَ يومَ السَّبتِ لِئَلَّا تُخالفوا شريعةَ موسى، فكيفَ تَغضبونَ علَيَّ لأنِّي شَفيتُ إنسانًا بأكمَلِهِ يومَ السَّبتِ؟</span>
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 11:3–4; يَعقُوب 2:9; يُوحَنّا 8:15"> لا تَحكُموا على الظّاهِرِ، بلِ اَحكُمُوا بالعَدلِ“.</span>
</div>
</div>

## هل يسوع هو المسيح؟

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 7:10–11"> فقَالَ بَعضُ أهالي أُورُشليمَ: ”أما هذا هوَ الّذي يُريدونَ أنْ يَقتُلوهُ؟</span>
<span className="v" id="v26" data-v="26" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:7–8; يُوحَنّا 7:48; يُوحَنّا 12:42"> ها هوَ يتكَلَّمُ جَهارًا ولا يَقولونَ لَه شَيئًا. فهلِ اَقتَنَعَ الرُؤساءُ أنَّهُ المَسيحُ؟</span>
<span className="v" id="v27" data-v="27" data-xref="ميخا 5:2; يُوحَنّا 7:41–42; يُوحَنّا 6:42"> لكِنَّنا نَعرِفُ مِنْ أينَ جاءَ هذا الرَّجُلُ. وأمَّا المَسيحُ، فلا يَعرِفُ أحَدٌ حينَ يَجيءُ مِنْ أينَ جاءَ“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 8:42; يُوحَنّا 8:19; يُوحَنّا 8:26"> فقَالَ يَسوعُ بأعلى صوتِهِ وهوَ يُعَلِّمُ في الهَيكَلِ: ”أنتُم تَعرِفوني وتَعرِفونَ مِنْ أينَ أنا، لكنِّي ما جِئتُ مِنْ عِندي. ذاكَ الّذي أرسَلَني هوَ حقٌّ، وأنتُم لا تَعرِفونَهُ.</span>
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 8:55; ﻣﺘﻰ 11:27; يُوحَنّا 17:18"> وأمَّا أنا فأعرِفُهُ، لأنِّي مِنْ عِندِهِ جِئتُ، وهوَ الّذي أرسَلَني“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 8:20; يُوحَنّا 10:39; يُوحَنّا 7:6"> فأرادوا أن يمسكوه، فما مَدَّ أَحدٌ يدًا علَيهِ، لأَنَّ ساعَتَهُ ما جاءَتْ بَعدُ.</span>
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 12:42; ﻣﺘﻰ 12:23; يُوحَنّا 10:41–42"> ولكنَّ كثيرينَ مِنَ الجَمعِ آمنوا بِه وقالوا: ”أيَعمَلُ المَسيحُ مِنَ الآياتِ حينَ يَجيءُ أكثرَ مِمَّا عَمِلَ هذا الرَّجُلُ؟“</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="يُوحَنّا 18:3; لُوقا 22:52–53; يُوحَنّا 11:47–48"> وسَمِعَ الفَرِّيسيُّونَ أنَّ النَّاسَ يَتَهامَسونَ بِهذِهِ الأُمورِ على يَسوعَ، فأرسلوا هُم ورُؤساءُ الكَهنَةِ حَرَسًا ليُمسِكوهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 16:5; يُوحَنّا 13:33; مَرْقُس 16:19"> فقَالَ يَسوعُ: ”سأبقى مَعكُم وَقتًا قليلًا، ثُمَّ أمضي إلى الّذي أرسَلَني.</span>
<span className="v" id="v34" data-v="34" data-xref="الأمثال 1:24–31; يُوحَنّا 13:33–36; يُوحَنّا 14:6"> سَتَطلُبوني فَلا تَجدوني، وحَيثُ أكونُ أنا لا تَقدِرونَ أنتُم أنْ تَجيئوا“.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="يَعقُوب 1:1; إشَعْياء 11:12; صَفَنْيا 3:10"> فقَالَ اليَهودُ في ما بَينَهُم: ”إلى أينَ يذهَبُ هذا، فلا نَقدِرُ أنْ نَجِدَهُ؟ أيذهَبُ إلى اليَهودِ المُشتَّتينَ بَينَ اليونانيّينَ لِيُعَلِّمَ اليونانيّينَ؟</span>
<span className="v" id="v36" data-v="36" data-xref="يُوحَنّا 3:9; يُوحَنّا 7:34; يُوحَنّا 12:34"> ما مَعنى قولِهِ: ستَطلُبوني ولا تَجدُوني، وحيثُ أكونُ أنا لا تَقدِرونَ أنتُم أنْ تَجيئوا؟“</span>
</div>
</div>

## أنهار ماء حي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="إشَعْياء 55:1; يُوحَنّا 4:14; يُوحَنّا 6:35"> ووقَفَ يَسوعُ في آخرِ يومٍ مِنَ العيدِ وهوَ أعظَمُ أيّامِهِ، فقَالَ بأعلى صوتِهِ: ”إنْ عَطِشَ أحَدٌ، فليَجئْ إليَّ ليَشرَبَ.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 4:14; إشَعْياء 44:3; إشَعْياء 58:11"> ومَنْ آمنَ بي، كما قالَ الكِتابُ، تَفيضُ مِنْ صَدرِهِ أنهارُ ماءٍ حيٍّ“.</span>
<span className="v" id="v39" data-v="39" data-xref="اعمال الرسل 2:17; يُوئيل 2:28; يُوحَنّا 14:16–17"> وعَنى بِكلامِهِ الرُّوحَ الّذي سَينالُهُ المُؤمِنونَ بِه، فما كانَ الرُّوحُ أُعطِيَ حتّى الآنَ، لأنَّ يَسوعَ ما تَمجَّدَ بَعدُ.</span>
</div>
</div>

## انقسام الرأي في يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="ﻣﺘﻰ 21:11; يُوحَنّا 1:21; يُوحَنّا 6:14"> فقَالَ كَثيرونَ مِمَّنْ سَمِعوا كلامَ يَسوعَ: ”بالحقيقةِ هذا هوَ النَّبيُّ“.</span>
<span className="v" id="v41" data-v="41" data-xref="يُوحَنّا 7:52; يُوحَنّا 1:46; يُوحَنّا 1:49"> وقالَ غيرُهُم: ”هذا هوَ المَسيحُ!“ وقالَ آخرونَ: ”أمِنَ الجليلِ يَجيءُ المَسيحُ؟</span>
<span className="v" id="v42" data-v="42" data-xref="ميخا 5:2; ﻣﺘﻰ 2:5; لُوقا 2:4"> أما قالَ الكِتابُ إنَّ المَسيحَ يَجيءُ مِنْ نَسلِ داودَ، ومِنْ بَيتَ لحمَ مدينةِ داودَ؟“</span>
<span className="v" id="v43" data-v="43" data-xref="يُوحَنّا 9:16; يُوحَنّا 10:19; يُوحَنّا 7:12"> فاَنقَسَمَ رأيُ النَّاسِ فيهِ.</span>
<span className="v" id="v44" data-v="44" data-xref="يُوحَنّا 7:30; اعمال الرسل 23:11; يُوحَنّا 8:20"> وأرادَ بَعضُهُم أنْ يُمسكوهُ، فما مَدَّ أحَدٌ يَدًا علَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="يُوحَنّا 7:32; اعمال الرسل 5:21–27"> ورجَعَ الحرَسُ إلى رُؤساءِ الكَهنَةِ والفَرِّيسيِّينَ، فقَالَ لهُم هَؤُلاءِ: ”لماذا ما جِئتُمْ بِه؟“</span>
<span className="v" id="v46" data-v="46" data-xref="ﻣﺘﻰ 7:28–29; لُوقا 4:22; يُوحَنّا 7:26"> فأجابَ الحرَسُ: ”ما تكَلَّمَ إنسانٌ مِنْ قَبلُ مِثلَ هذا الرَّجُلِ!“</span>
<span className="v" id="v47" data-v="47" data-xref="يُوحَنّا 7:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:8; المُلُوكِ الثّانِي 18:32"> فقَالَ لهُمُ الفَرِّيسيُّونَ: ”أخَدَعكُم أنتُم أيضًا؟</span>
<span className="v" id="v48" data-v="48" data-xref="يُوحَنّا 12:42; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:8"> أرأيتُم واحدًا مِنَ الرُؤساءِ أوِ الفَرِّيسيِّينَ آمنَ بِه؟</span>
<span className="v" id="v49" data-v="49" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:20–21; يُوحَنّا 9:40; يُوحَنّا 9:34"> أمَّا هَؤُلاءِ العامَّةُ مِنَ النَّاسِ الّذينَ يَجهَلونَ الشَّريعةَ، فهُم مَلعونونَ“.</span>
</div>
<div className="bp">
