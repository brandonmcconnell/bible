## سليمان يطلب الحكمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المُلُوكِ الأوَّلُ 9:24; المُلُوكِ الأوَّلُ 11:1; المُلُوكِ الأوَّلُ 6:1–15"> وصاهرَ سُليمانُ فِرعَونَ مَلِكَ مِصْرَ وتزَوَّجَ اَبنَتَهُ، وجاءَ بها إلى مدينةِ داوُدَ إلى أنْ أكمَلَ بِناءَ قصرِهِ وهَيكلِ الرّبِّ وسُورِ أورُشليمَ المُحيطِ بِها.</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الأوَّلُ 22:43; أخبارِ الأيّامِ الثّانِي 33:17; التَّثْنِيَة 12:2–5"> وكانَ الشَّعبُ يُقَدِّمونَ ذبائِحَهُم على رُؤوسِ التِّلالِ، لأنَّهُ لم يكُن هَيكلٌ لاَسم الرّبِّ إلى تِلكَ الأيّامِ.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:3; المَزامِير 31:23; التَّثْنِيَة 30:16"> وأحبَّ سُليمانُ الرّبَّ وتَبِعَ فرائِضَ داوُدَ أبيهِ، غَيرَ أنَّهُ كانَ يُقَدِّمُ الذَّبائحَ ويَحرُقُ البَخورَ على رُؤوسِ التِّلالِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="أخبارِ الأيّامِ الأوّلُ 16:39; أخبارِ الأيّامِ الثّانِي 1:3; أخبارِ الأيّامِ الثّانِي 7:5"> وذهَبَ مرَّةً إلى جبعونَ لِيُقَدِّمَ الذَّبائحَ هُناكَ، لأنَّ فيها أعظَمَ المذابِحِ، وعلى ذلِكَ المذبَحِ أصعَدَ ألفَ مُحرَقةٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الأوَّلُ 9:2; يَعقُوب 1:5–6; العَدَد 12:6"> وفي جبعونَ تجلَّى الرّبُّ لِسُليمانَ في الحُلُم ليلًا وقالَ لَه: ”أُطْلُبْ ما تُريدُ“.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 9:4; المُلُوكِ الأوَّلُ 1:48; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:11"> فقالَ سُليمانُ: ”أنتَ أظهَرتَ إلى عبدِكَ داوُدَ أبي رَحمةً عظيمةً لأنَّهُ سَلَكَ أمامَكَ بِأمانةٍ وحَقٍّ واَستِقامَةِ قلبٍ، وثابَرتَ على تِلكَ الرَّحمةِ العظيمةِ فرَزَقتَهُ اَبنًا يَجلِسُ على عرشِهِ كما هوَ اليومَ.</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الأوّلُ 29:1; العَدَد 27:17; إرْمِيا 1:6"> والآنَ أيُّها الرّبُّ إلهي، أنتَ مَلَّكتَني مكانَ داوُدَ أبي وأنا صغيرٌ لا خِبرَةَ لي في الحُكْمِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 15:5; التَّكوين 13:16; أخبارِ الأيّامِ الأوّلُ 21:2"> وها أنا وسَطَ شعبِكَ الّذي اَختَرتَهُ وهوَ شعبٌ عظيمٌ لا يُحصى ولا يُعَدُّ لكثرَتِه.</span>
<span className="v" id="v9" data-v="9" data-xref="يَعقُوب 1:5; الأمثال 16:16; العِبرانِيّين 5:14"> فاَمنَحْني عقلًا مُدرِكًا لأحكُمَ شعبَكَ وأُمَيِّزَ الخَيرَ مِنَ الشَّرِّ، وإلَّا فكيفَ أقدِرُ أنْ أحكُمَ شعبَكَ هذا الكثيرَ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الأمثال 15:8"> فحَسُنَ في عينَي الرّبِّ طَلَبُ سُليمانَ،</span>
<span className="v" id="v11" data-v="11" data-xref="ﺭﻭﻣﻴﺔ 8:26; يَعقُوب 4:2–3; الأمثال 16:31"> فقالَ لَه: ”لأنَّكَ طَلَبتَ هذا ولم تطلُبْ لكَ طُولَ العُمرِ والغِنى، ولا موتَ أعدائِكَ، بلِ القُدرَةَ على تمييزِ ما هوَ حَقٌّ،</span>
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:14–15; الجامِعَة 1:16; ﻣﺘﻰ 12:42"> فأنا أُلبِّي طلَبَكَ، فأُعطيكَ عقلًا حكيمًا راجِحًا لم يكُنْ مِثلُهُ لأحدٍ قَبلَكَ ولا يكونُ مِثلُهُ لأحدٍ بَعدَكَ.</span>
<span className="v" id="v13" data-v="13" data-xref="أفَسُس 3:20; ﻣﺘﻰ 6:33; الأمثال 3:16"> وأُعطيكَ أيضًا ما لم تَطلُبْهُ: الغِنى والمَجدَ، فلا يكونُ لكَ مَثيلٌ في المُلوكِ كُلَّ أيّامِكَ.</span>
<span className="v" id="v14" data-v="14" data-xref="الأمثال 3:2; المَزامِير 91:16; المُلُوكِ الأوَّلُ 15:5"> وأُطِيلُ عُمرَكَ إذا سَلكْتَ في طريقي، حافِظًا فرائِضي ووصايايَ كما سَلَكَ داوُدُ أبُوكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 41:7; المُلُوكِ الأوَّلُ 8:65; أسْتِير 1:3"> وحينَ أفاقَ سُليمانُ مِنْ نَومِه، عرَفَ أنَّ ما رَآهُ كانَ حُلُمًا. فجاءَ إلى أورُشليمَ ووَقَفَ أمامَ تابوتِ العَهدِ وأصعَدَ مُحرَقاتٍ وقَدَّمَ ذبائحَ سلامةٍ، وأقامَ مَأدُبةً لِجميعِ رِجالِ حاشيتِهِ.</span>
</div>
</div>

## سليمان يظهر حكمته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 18:16; القُضاة 11:1; العَدَد 27:2"> ثُمَّ جاءَت إليهِ زانِيتانِ ووَقَفَتا أمامَهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 13:7; التَّكوين 43:20"> فقالَت إحداهُما: ”آهِ يا سيِّدي. أُقيمُ معَ هذِهِ المَرأةِ في بَيتٍ واحدٍ، فوَلَدتُ أنا في البَيتِ</span>
<span className="v" id="v18" data-v="18"> وبَعدَ يومَينِ ولَدَت هذِهِ المَرأةُ أيضًا. وكُنا معًا ولا أحدَ في البَيتِ غيرُنا.</span>
<span className="v" id="v19" data-v="19"> فماتَ اَبنُ هذِهِ المَرأةِ في اللَّيلِ، لأنَّها نامَت علَيهِ.</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 3:20; ﻣﺘﻰ 13:25; المَزامِير 139:11"> فلمَّا قامَت عِندَ نِصفِ اللَّيلِ أخذَتِ اَبني مِنْ جانبي وأنا نائمةٌ، واَستَبدَلَتهُ بِاَبنِها المَيْتِ.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 21:7; صَمُوئيلَ الأوَّلُ 1:23; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:3–4"> وقُمْتُ في الصَّباحِ لأُرضِعَ اَبني فوَجَدتُهُ مَيْتًا، وعِندَما تَفرَّستُ فيه رَأيتُ أنَّهُ لم يكُنِ اَبني“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="المُلُوكِ الأوَّلُ 3:23–24"> فقالَتِ الثَّانيةُ: ”لا، بلِ الحَيُّ اَبني والمَيتُ اَبنُكِ“. فأجابَتها: ”لا، بَلِ المَيتُ اَبنُكِ والحَيُّ اَبني“. هكذا تَجادَلَتا أمامَ المَلِكِ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23"> فقالَ المَلِكُ: ”هذِهِ تقولُ: الحَيُّ اَبني والمَيتُ اَبنُكِ. وتِلكَ تقولُ: لا، بلِ المَيتُ اَبنُكِ والحَيُّ اَبني“.</span>
<span className="v" id="v24" data-v="24"> ثُمَّ قالَ المَلِكُ: ”هاتوا سَيفًا“. فأعطَوهُ،</span>
<span className="v" id="v25" data-v="25" data-xref="الأمثال 25:8"> فقالَ: ”أُشطُروا الصَّبيَّ الحَيَّ شَطرَينِ وأَعطوا كُلَّ واحدةٍ شَطرًا“.</span>
<span className="v" id="v26" data-v="26" data-xref="إشَعْياء 49:15; إرْمِيا 31:20; رسالَةُ يُوحَنّا الأُولَى 3:17"> فقالَت والِدةُ الصَّبيِّ الحَيِّ مُتلَهِّفةً على اَبنِها: ”آهِ يا سيِّدي. أعطوها الصَّبيَّ حَيًّا ولا تَقتُلوهُ“. فقالَتِ الثَّانيةُ: ”بل لا يكونُ لي ولا لكِ. أُشطُروهُ“.</span>
<span className="v" id="v27" data-v="27"> فقالَ المَلِكُ: ”أعطوا الصَّبيَّ الحَيَّ لِتلكَ المَرأةِ لأنَّها أُمُّهُ، ولا تَقتُلوهُ“.</span>
<span className="v" id="v28" data-v="28" data-xref="دانيال 2:47; المُلُوكِ الأوَّلُ 3:9–12; كُولُوسِي 2:3"> فسَمِعَ جميعُ بَني إِسرائيلَ بِالحُكْمِ الّذي حكَمَ بهِ المَلِكُ، فهابوهُ لأنَّهُم عرَفوا أنَّ اللهَ منَحَهُ الحِكْمةَ لِيَحكُمَ بِالعَدلِ.</span>
</div>
</div>
