## داود في شيخوخته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يَشُوع 23:1–2; أخبارِ الأيّامِ الأوّلُ 29:27–28; أخبارِ الأيّامِ الأوّلُ 23:1"> وشاخَ المَلِكُ داوُدُ وكبُرَ في السِّنِ. وكانوا يُكثِرونَ علَيهِ الأغطيةَ فلا يَدفَأُ.</span>
<span className="v" id="v2" data-v="2" data-xref="الجامِعَة 4:11; التَّكوين 16:5; ميخا 7:5"> فقالَ لَه رِجالُ حاشيتِهِ: ”دَعْنا نبحَثُ لِسَيِّدِنا المَلِكِ عَنْ فتاةٍ عذراءَ تَخدُمُهُ وتُؤانِسُهُ وبَينَ ذِراعَيهِ تنامُ فيَدْفَأُ“.</span>
<span className="v" id="v3" data-v="3" data-xref="يَشُوع 19:18; أسْتِير 2:4; أسْتِير 2:2"> فبَحَثوا لَه عَنْ فتاةٍ جميلةٍ في جميعِ أرضِ إِسرائيلَ، فوَجَدوا أبيشَجَ الشُّونَميَّةَ فجاؤوا بها إليهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 1:25"> وكانَتِ الفتاةُ رائِعةَ الجَمالِ، فكانَت تُؤانِسُهُ وتَخدُمُهُ، لكنَّهُ لم ينَمْ معَها.</span>
</div>
</div>

## أدونيا يطمح إلى الملك

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="صموئيلَ الثّانِي 15:1; صموئيلَ الثّانِي 3:4; التَّثْنِيَة 17:15–16"> وبلَغَ الطُّموحُ بأدونيَّا بنِ داوُدَ واَبنِ حَجِّيتَ إلى حَدِّ قولِهِ: ”سأكونُ مَلِكًا“. واَتَّخَذَ لَه مَركباتٍ وخَيلًا وخَمسينَ رَجُلًا يَخدِمونَهُ</span>
<span className="v" id="v6" data-v="6" data-xref="صموئيلَ الثّانِي 3:3–4; صَمُوئيلَ الأوَّلُ 3:13; العِبرانِيّين 12:5–6"> ولم يشَأْ داوُدُ أبوهُ أنْ يُزعِلَهُ بِأنْ يقولَ لَه يومًا: ”لماذا تفعَلُ هذا؟“ وكانَ هوَ أيضًا جميلَ الصُّورةِ جِدًّا، وكانَت أمُّهُ ولَدَتْهُ بَعدَ أبشالومَ.</span>
<span className="v" id="v7" data-v="7" data-xref="صموئيلَ الثّانِي 20:25; المُلُوكِ الأوَّلُ 2:22; صَمُوئيلَ الأوَّلُ 22:20–23"> وفاوَضَ يُوآبَ اَبْنَ صُرويَةَ وأبياثارَ الكاهنَ ففازَ بِتَأييدِهِما.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الأوَّلُ 4:18; صموئيلَ الثّانِي 23:8–39; صموئيلَ الثّانِي 20:25"> وأمَّا صادوقُ الكاهنُ، وبَنايا بنُ يوياداعَ، وناثانُ النَّبيُّ، وشِمْعي ورِبعي، وكِبارُ قادةِ داوُدَ فلم يُؤيِّدوه.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="صموئيلَ الثّانِي 17:17; صموئيلَ الثّانِي 13:23–27; صموئيلَ الثّانِي 15:11–12"> وذبَحَ أدونيَّا غنَمًا وبقَرًا وعُجولًا مُسَمَّنةً عِندَ الحجَرِ الزَّاحِلِ الّذي بِجانِبِ عينِ روجَلَ، ودَعا جميعَ إخوَتِهِ الأمَراءِ وجميعَ رِجالِ يَهوذا الّذينَ في حاشيةِ المَلِكِ،</span>
<span className="v" id="v10" data-v="10" data-xref="صموئيلَ الثّانِي 12:24; المُلُوكِ الأوَّلُ 1:19; صموئيلَ الثّانِي 12:1–15"> وأمَّا ناثانُ النَّبيُّ، وبَنايا، والقادةُ، وسُليمانُ أخوهُ فلم يَدْعُهُم.</span>
</div>
</div>

## سليمان ينادى به ملكًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="صموئيلَ الثّانِي 3:4; صموئيلَ الثّانِي 7:12–17; صموئيلَ الثّانِي 12:24–25"> فقالَ ناثانُ لِبَتْشابَعَ أمِّ سُليمانَ: ”أما سَمِعْتِ أنَّ أدونيَّا اَبنَ حَجِّيتَ صارَ مَلِكًا، وسيِّدُنا داوُدُ لا يعلَمُ؟</span>
<span className="v" id="v12" data-v="12" data-xref="الأمثال 11:14; اعمال الرسل 27:31; التَّكوين 19:17"> والآنَ أُقَدِّمُ إليكِ نصيحةً تُنقِذينَ بِها حياتَكِ وحياةَ سُليمانَ اَبنِكِ.</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الأوَّلُ 1:30; المُلُوكِ الأوَّلُ 1:17; المَزامِير 132:11–12"> إذْهَبي إلى المَلِكِ داوُدَ وقولي لَه: أما حلَفْتَ لي يا سيِّدي المَلِكُ وقُلتَ: سُليمانُ اَبْنُكِ يَملِكُ مِنْ بَعدي ويجلِسُ على عرشي، فكيفَ صارَ أدونيَّا مَلِكًا</span>
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:1; المُلُوكِ الأوَّلُ 1:17–27"> وبَينَما تُكَلِّمينَهُ أجيءُ وأدعَمُ كلامَكِ“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الأوَّلُ 1:1"> فذَهَبَت بَتْشابَعُ إلى المَلِكِ في غُرفةِ نومِهِ، وكانَ في عِزِّ شَيخوخَتِهِ وأبيشَجُ الشُّونَمِيَّةُ تَخدُمُهُ.</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 25:23; صَمُوئيلَ الأوَّلُ 24:8; صَمُوئيلَ الأوَّلُ 20:41"> فرَكَعَت بِتْشابَعُ أمامَهُ، فسألَها: ”ماذا تُريدينَ؟“</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الأوَّلُ 1:13; المُلُوكِ الأوَّلُ 1:30; رسالَةُ بُطرُسَ الأُولَى 3:6"> فأجابَت: ”يا سيِّدي، أنتَ حلَفْتَ لي بِالرّبِّ إلهِكَ وقُلتَ: سُليمانُ اَبْنُكِ يملِكُ مِنْ بَعدي ويجلِسُ على عرشي.</span>
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الأوَّلُ 1:24; المُلُوكِ الأوَّلُ 1:11; اعمال الرسل 3:17"> لكنَّ أدونيَّا صارَ مَلِكًا دونَ عِلْمِكَ، يا سيِّدي المَلِكُ.</span>
<span className="v" id="v19" data-v="19" data-xref="المُلُوكِ الأوَّلُ 1:25; المُلُوكِ الأوَّلُ 1:7–10"> وذبَحَ كثيرًا مِنَ البقَرِ والعُجولِ المُسَمَّنةِ والغنَمِ ودَعا جميعَ إخوَتِهِ الأمَراءِ وأبياثارَ الكاهنَ ويوآبَ قائدَ الجيشِ، ما عَدا سُليمانَ اَبْنَكَ</span>
<span className="v" id="v20" data-v="20" data-xref="أخبارِ الأيّامِ الأوّلُ 29:1; المَزامِير 123:2; أخبارِ الأيّامِ الأوّلُ 22:8–10"> وجميعُ بَني إِسرائيلَ ينتَظِرونَ يا سيِّدي المَلِكُ لِتُعلِمَهُم مَنْ يجلِسُ على عرشِكَ مِنْ بَعدِكَ.</span>
<span className="v" id="v21" data-v="21" data-xref="المُلُوكِ الأوَّلُ 2:10; التَّثْنِيَة 31:16; التَّكوين 15:15"> وإلَّا، إذا رقَدْتَ يا سيِّدي المَلِكُ معَ آبائِكَ، فأنا واَبْني سُليمانُ ندفعُ الثَّمَنَ“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="أيُّوب 1:16–18; التَّكوين 24:15; دانيال 9:20"> وبَينَما هيَ تَتكَلَّمُ معَ المَلِكِ جاءَ ناثانُ النَّبيُّ.</span>
<span className="v" id="v23" data-v="23" data-xref="المُلُوكِ الأوَّلُ 1:16; رسالَةُ بُطرُسَ الأُولَى 2:17; ﺭﻭﻣﻴﺔ 13:7"> فقيلَ لِلمَلِكِ: ”ها ناثانُ النَّبيُّ“. فدَخَلَ واَنْحَنى حتّى الأرضِ</span>
<span className="v" id="v24" data-v="24" data-xref="المُلُوكِ الأوَّلُ 1:17–18; المُلُوكِ الأوَّلُ 1:5; المُلُوكِ الأوَّلُ 1:13–14"> وقالَ: ”يا سيِّدي المَلِكُ، هل قُلتَ: يَملِكُ أدونيَّا مِنْ بَعدي ويجلِسُ على عرشي؟</span>
<span className="v" id="v25" data-v="25" data-xref="صَمُوئيلَ الأوَّلُ 10:24; المُلُوكِ الأوَّلُ 1:19; المُلُوكِ الأوَّلُ 1:9"> ففي هذا اليومِ ذبَحَ الكثيرَ مِنَ البقَرِ والعُجولِ المُسَمَّنَةِ والغنَمِ، ودعا جميعَ إخوَتِهِ الأمراءِ، وقادةَ الجيشِ، وأبياثارَ الكاهنَ. وها هُم يَأكُلونَ ويشربونَ أمامَهُ ويهْتِفونَ: لِيحيَ المَلِكُ أدونيَّا.</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الأوَّلُ 1:8; المُلُوكِ الأوَّلُ 1:10; صموئيلَ الثّانِي 7:2"> وأمَّا أنا، وصادوقُ الكاهنُ، وبَنايا بنُ يوياداعَ، وسُليمانُ اَبْنُكَ، فما دَعانا.</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 15:15; المُلُوكِ الأوَّلُ 1:24; المُلُوكِ الثّانِي 4:27"> فهل وافَقْتَ على هذا، يا سيِّدي المَلِكُ، ولم تُعلِمْ حتّى رِجالَ حاشيتِكَ مَنْ يجلِسُ على عرشِكَ مِنْ بَعدِكَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28"> فأجابَ المَلِكُ: ”أدعوا لي بَتْشابَعَ“. فدَخَلَت ووَقَفَت أمامَهُ.</span>
<span className="v" id="v29" data-v="29" data-xref="صموئيلَ الثّانِي 4:9; المَزامِير 136:24; صَمُوئيلَ الأوَّلُ 14:45"> فحَلَفَ وقالَ: ”حَيٌّ هوَ الرّبُّ الّذي خلَّصَني مِنْ كُلِّ ضيقٍ</span>
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الأوَّلُ 1:13; المُلُوكِ الأوَّلُ 1:17"> أحلِفُ لكِ اليومَ بِالرّبِّ إلهِ إِسرائيلَ كما حلَفْتُ لكِ مِنْ قَبلُ وقُلتُ: سُليمانُ اَبْنُكِ يَملِكُ بَعدي ويجلِسُ مكاني على عرشي“.</span>
<span className="v" id="v31" data-v="31" data-xref="دانيال 2:4; دانيال 5:10; دانيال 3:9"> فاَنْحَنَت لَه حتّى الأرضِ وقالَت: ”لِيحيَ سيِّدي المَلِكُ داوُدُ إلى الأبدِ“.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="المُلُوكِ الأوَّلُ 1:8; المُلُوكِ الأوَّلُ 1:38; المُلُوكِ الأوَّلُ 1:26"> وقالَ المَلِكُ داوُدُ: ”لِيَحضَرْ أمامي صادوقُ الكاهنُ، وناثانُ النَّبيُّ، وبَنايا بنُ يوياداعَ“. فحَضَروا.</span>
<span className="v" id="v33" data-v="33" data-xref="أخبارِ الأيّامِ الثّانِي 32:30; المُلُوكِ الأوَّلُ 1:38; أخبارِ الأيّامِ الثّانِي 33:14"> فقالَ لهُم: ”خُذوا مَعكُم رِجالَ حاشيتي وأركِبوا سُليمانَ اَبْني على بَغلَتي، واَنْزِلوا بهِ إلى جيحونَ،</span>
<span className="v" id="v34" data-v="34" data-xref="المُلُوكِ الأوَّلُ 1:25; صموئيلَ الثّانِي 15:10; صَمُوئيلَ الأوَّلُ 10:1"> حَيثُ يَمسَحُهُ صادوقُ الكاهنُ وناثانُ النَّبيُّ مَلِكًا على إِسرائيلَ. واَهْتِفُوا بِالبوقِ وقولوا: لِيحيَ المَلِكُ سُليمانُ.</span>
<span className="v" id="v35" data-v="35" data-xref="المُلُوكِ الأوَّلُ 1:17; المَزامِير 72:1–2; المُلُوكِ الأوَّلُ 1:13"> واَصعَدوا وراءَهُ، فيَجيءُ ويجلِسُ على عرشي لِيَملِكَ مكاني. فأنا أوْصَيتُ أنْ يكونَ سُليمانُ رئيسًا على إِسرائيلَ ويَهوذا“.</span>
<span className="v" id="v36" data-v="36" data-xref="إرْمِيا 28:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:16; المَزامِير 72:19"> فأجابَهُ بَنايا بنُ يوياداعَ: ”نعم. هذِهِ إرادةُ الرّبِّ إلهِكَ يا سيِّدي المَلِكُ.</span>
<span className="v" id="v37" data-v="37" data-xref="صَمُوئيلَ الأوَّلُ 20:13; المُلُوكِ الأوَّلُ 1:47; يَشُوع 1:17"> وكما كانَ الرّبُّ معَكَ، فليكُنْ معَ سُليمانَ أيضًا ويجعَلْ عرشَهُ أعظَمَ مِنْ عرشِكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="صموئيلَ الثّانِي 8:18; المُلُوكِ الأوَّلُ 1:8; المُلُوكِ الأوَّلُ 1:33"> فنَزلَ صادوقُ الكاهنُ، وناثانُ النَّبيُّ، وبَنايا بنُ يوياداعَ، والحرَسُ المَلكيُّ، وأركبوا سُليمانَ على بَغلَةِ المَلِكِ داوُدَ وساروا بهِ إلى جيحونَ.</span>
<span className="v" id="v39" data-v="39" data-xref="المَزامِير 89:20; المُلُوكِ الأوَّلُ 1:34; أخبارِ الأيّامِ الأوّلُ 29:22"> وأخذَ صادوقُ الكاهنُ وِعاءَ الزَّيتِ مِنْ خَيمَةِ الاجتِماعِ ومسَحَ سُليمانَ، فهَتَفَ الشَّعبُ بِالبوقِ ونادَوا: ”لِيحيَ المَلِكُ سُليمانُ“.</span>
<span className="v" id="v40" data-v="40" data-xref="المُلُوكِ الثّانِي 11:14; زَكَريّا 9:9; المُلُوكِ الثّانِي 11:20"> وصَعِدوا وراءَهُ وهُم يَعزِفونَ بِالنَّاي ويُهَلِّلونَ فرَحًا، حتّى تصَدَّعَتِ الأرضُ مِنْ أصواتِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="أيُّوب 15:21–22; لُوقا 17:26–29; اعمال الرسل 21:31"> فسَمِعَ أدونيَّا وجميعُ المَدعُوّينَ الّذينَ عِندَهُ، وكانوا اَنْتَهَوا مِنَ الأكلِ. ولمَّا سَمعَ يوآبُ صوتَ البوقِ سألَ: ”ما هذا الصَّوتُ الّذي تَضِجُّ بهِ المدينةُ؟“</span>
<span className="v" id="v42" data-v="42" data-xref="صموئيلَ الثّانِي 15:36; صموئيلَ الثّانِي 17:17; صموئيلَ الثّانِي 18:27"> وبَينَما هوَ يَتكَلَّمُ أقبَلَ يوناثانُ بنُ أبياثارَ الكاهنُ، فقالَ لَه أدونيَّا: ”أُدْخُلْ، فأنتَ رَجُلٌ صالِحٌ وتُبَشِّرُ بِالخيرِ“.</span>
<span className="v" id="v43" data-v="43"> فأجابَهُ يوناثانُ: ”لا، فسيِّدُنا المَلِكُ أعطى العرشَ لِسُليمانَ</span>
<span className="v" id="v44" data-v="44"> وأرسَلَ معَهُ صادوقَ الكاهنَ، وناثانَ النَّبيَّ، وبَنايا بنَ يوياداعَ، والحرَسَ المَلِكي، فأركبوهُ على بَغلَةِ المَلِكِ</span>
<span className="v" id="v45" data-v="45" data-xref="المُلُوكِ الأوَّلُ 1:40; دانيال 5:26–28; صَمُوئيلَ الأوَّلُ 4:5"> وساروا بهِ إلى جيحونَ، حَيثُ مسَحَهُ صادوقُ الكاهنُ وناثانُ النَّبيُّ مَلِكًا. ثُمَّ صَعِدوا مِنْ هُناكَ مُهَلِّلينَ فرَحًا، فضَجَّتِ المدينةُ، وهذا هوَ الصَّوتُ الّذي سَمِعتُم.</span>
<span className="v" id="v46" data-v="46" data-xref="أخبارِ الأيّامِ الأوّلُ 29:23; المَزامِير 132:11; المُلُوكِ الأوَّلُ 1:13"> وجلَسَ سُليمانُ على عرشِ المُلْكِ.</span>
<span className="v" id="v47" data-v="47" data-xref="التَّكوين 47:31; المُلُوكِ الأوَّلُ 1:37; عَزْرا 6:10"> ودخَلَ رِجالُ الحاشيةِ لِلدُّعاءِ لِسيِّدِنا المَلِكِ داوُدَ وقالوا: لِيَجعَلْ إلهُكَ اَسمَ سُليمانَ أعظَمَ مِنِ اَسمِكَ، وعرشَهُ أعظَمَ مِنْ عرشِكَ. فسَجَدَ المَلِكُ على سريرِهِ</span>
<span className="v" id="v48" data-v="48" data-xref="المُلُوكِ الأوَّلُ 3:6; المَزامِير 132:11–12; المَزامِير 145:2"> وقالَ: تَبارَكَ الرّبُّ إلهُ إِسرائيلَ الّذي رزَقَني اليومَ مَنْ يَجلِسُ على عرشي، وعيناي مَفتوحَتانِ“.</span>

<span className="v" id="v50" data-v="50" data-xref="المُلُوكِ الأوَّلُ 2:28; الخُرُوج 21:14; الخُرُوج 27:2"> وخافَ أدونيَّا مِنْ سُليمانَ فلَجَأ إلى خَيمَةِ الاجتماعِ وتمَسَّكَ بِقرونِ المذبَحِ.</span>
<span className="v" id="v51" data-v="51"> فقيلَ لِسُليمانَ: ”أدونيَّا خائفٌ مِنكَ، فتَمَسَّكَ بِقرونِ المذبَحِ وقالَ: لِيَحلِفْ ليَ المَلِكُ سُليمانُ أنَّهُ لا يقتلُني بِالسَّيفِ“.</span>
<span className="v" id="v52" data-v="52" data-xref="صَمُوئيلَ الأوَّلُ 14:45; صموئيلَ الثّانِي 14:11; الأمثال 13:6"> فقالَ سُليمانُ: ”إنْ أحسَنَ فلا تَسقُط شَعرةٌ مِنهُ إلى الأرضِ، وإنْ أساءَ فموتًا يموتُ“.</span>
