## نقل تابوت العهد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أخبارِ الأيّامِ الثّانِي 5:2–10; أخبارِ الأيّامِ الأوّلُ 15:25; صموئيلَ الثّانِي 6:12–17"> ثُمَّ دَعا المَلِكُ سُليمانُ شُيوخَ إِسرائيلَ وجميعَ رُؤساءِ الأسباطِ وزُعَماءَ عَشائِرِ بَني إِسرائيلَ إلى أورُشليمَ، لِيَحمِلوا تابوتَ عَهدِ الرّبِّ مِنْ صِهيَونَ مدينةِ داوُدَ.</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 23:34; أخبارِ الأيّامِ الثّانِي 7:8–10; زَكَريّا 14:16–19"> فاَجتَمَعوا كُلُّهُم إليهِ في العيدِ، في شهرِ أيثانيمَ وهوَ الشَّهرُ السَّابِعُ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 31:9; يَشُوع 3:6; العَدَد 4:15"> وحمَلَ الكهَنةُ تابوتَ العَهدِ</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الأوَّلُ 3:4; أخبارِ الأيّامِ الثّانِي 1:3; الخُرُوج 40:2–33"> وأصعَدوهُ معَ خَيمَةِ الاجتماعِ وكُلِّ أدواتِ الهَيكلِ الّتي في الخَيمةِ.</span>
<span className="v" id="v5" data-v="5" data-xref="صموئيلَ الثّانِي 6:13; أخبارِ الأيّامِ الأوّلُ 16:1; المُلُوكِ الأوَّلُ 8:62–63"> وكانَ المَلِكُ سُليمانُ وكُلُّ الّذينَ اَجتَمَعوا إليهِ مِنْ بَني إِسرائيلَ أمامَ التَّابوتِ يذبَحونَ مِنَ الغنَمِ والبقَرِ ما لا يُحصى ولا يُعَدُّ لِكَثرَتِهِ.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 6:19; صموئيلَ الثّانِي 6:17; المُلُوكِ الأوَّلُ 6:27"> وأدخَلَ الكَهنةُ تابوتَ عَهدِ الرّبِّ إلى مكانِهِ في مِحرابِ الهَيكلِ، في قُدسِ الأقداسِ، تَحتَ أجنِحةِ الكَروبَينِ</span>
<span className="v" id="v7" data-v="7"> اللَّذينِ كانا يَبسُطانِ أجنِحَتَهُما على مَوضِعِ التَّابوتِ فيُظَلِّلانِهِ معَ قُضبانِهِ الّتي يُحمَلُ بِها.</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 25:13–15; الخُرُوج 37:4–5; يَشُوع 4:9"> وكانتِ القُضبانُ طويلَةً، أطرافُها تُرَى مِنْ قُدسِ الأقداسِ أمامَ المِحرابِ، لا مِنَ الخارِجِ. وهيَ هُناكَ إلى هذا اليومِ.</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 25:21; العِبرانِيّين 9:4; الخُرُوج 40:20"> ولم يكُن في التَّابوتِ إلَّا لوحا الحجَرِ اللَّذانِ وضَعَهُما فيهِ موسى في حوريبَ، حَيثُ عاهدَ الرّبُّ بَني إِسرائيلَ عِندَ خُروجِهِم مِنْ أرضِ مِصْرَ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 40:34–35; أخبارِ الأيّامِ الثّانِي 7:1–2"> ولمَّا خرَجَ الكهَنةُ مِنَ المكانِ المُقدَّسِ، ملأَ السَّحابُ هَيكلَ بَيتِ الرّبِّ</span>
<span className="v" id="v11" data-v="11" data-xref="حِزْقِيال 43:4–5; حِزْقِيال 44:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:18"> فلم يَقدِرِ الكهَنةُ أنْ يُؤدُّوا الخِدمةَ، لأنَّ مَجدَ الرّبِّ ملأَ الهَيكلَ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 97:2; أخبارِ الأيّامِ الثّانِي 6:1–11; اللّاويِّين 16:2"> ثُمَّ قالَ سُليمانُ:<br />في السَّحابِ يسكُنُ الرّبُّ</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="صموئيلَ الثّانِي 7:13; المَزامِير 132:13–14; العِبرانِيّين 9:24"> وها أنا يا ربُّ، بَيتًا بنَيتُ،<br />مكانًا لِسُكناكَ إلى الأبدِ.</span>
</div>

## خطاب سليمان إلى الشعب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="صموئيلَ الثّانِي 6:18; المُلُوكِ الأوَّلُ 8:55–56; أخبارِ الأيّامِ الثّانِي 30:18–20"> واَلتَفَتَ المَلِكُ إلى جماعةِ إِسرائيلَ وباركَهُم وهُم وُقوفٌ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 1:68; أخبارِ الأيّامِ الأوّلُ 29:20; أخبارِ الأيّامِ الأوّلُ 29:10"> وقالَ: ”تَبارَكَ الرّبُّ إلهُ إِسرائيلَ الّذي أتَمَّ بِيدِهِ ما وعَدَ بهِ أبي داوُدَ وقالَ:</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 16:1; المُلُوكِ الأوَّلُ 8:29; صموئيلَ الثّانِي 7:4–8"> مُنذُ يومِ أخرَجتُ شعبي إِسرائيلَ مِنْ مِصْرَ لم أختَرْ مدينةً مِنْ جميعِ مُدُنِ أسباطِ إِسرائيلَ لِيُبنى لي فيها هَيكلٌ يكونُ اَسمي فيهِ، لكنِّي اَختَرتُكَ يا داوُدُ لِتكونَ رئيسًا على بَني إِسرائيلَ شعبي“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="صموئيلَ الثّانِي 7:2–3; أخبارِ الأيّامِ الأوّلُ 22:7; أخبارِ الأيّامِ الأوّلُ 28:2"> ثُمَّ قالَ سُليمانُ: ”وكانَ في نيَّةِ داوُدَ أبي أنْ يَبنيَ هَيكَلًا لاَسمِ الرّبِّ إلهِ إِسرائيلَ،</span>
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الثّانِي 6:7–9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12"> فقالَ لَه الرّبُّ: نَوَيتَ في قلبِكَ أنْ تَبنيَ هَيكلًا لاَسمي، فحَسَنٌ ما نَوَيتَ.</span>
<span className="v" id="v19" data-v="19" data-xref="صموئيلَ الثّانِي 7:5; أخبارِ الأيّامِ الأوّلُ 22:8–10; أخبارِ الأيّامِ الأوّلُ 28:6"> لكِنْ لا أنتَ تَبنيهِ، بل اَبنُكَ الّذي يخرُجُ مِنْ صُلبِكَ.</span>
<span className="v" id="v20" data-v="20" data-xref="أخبارِ الأيّامِ الأوّلُ 28:5–6; فِيلِبِّي 1:6; المُلُوكِ الأوَّلُ 8:15"> وتَمَّمَ الرّبُّ قولَهُ وقُمتُ أنا مكانَ داوُدَ أبي وجلَستُ على عرشِ إِسرائيلَ، كما قالَ الرّبُّ، وبَنَيتُ الهَيكلَ لاَسمِ الرّبِّ إلهِ إِسرائيلَ،</span>
<span className="v" id="v21" data-v="21" data-xref="المُلُوكِ الأوَّلُ 8:9; التَّثْنِيَة 31:26; المُلُوكِ الأوَّلُ 8:5–6"> وأعدَدْتُ مكانًا لِتابوتِ العَهدِ الّذي قطَعَهُ لآبائِنا حينَ أخرَجَهُم مِنْ أرضِ مِصْرَ“.</span>
</div>
</div>

## صلاة سليمان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="عَزْرا 9:5; المُلُوكِ الأوَّلُ 8:54; الخُرُوج 9:33"> ثُمَّ نهَضَ سُليمانُ أمامَ مذبَحِ الرّبِّ بِحُضورٍ مِنْ بَني إِسرائيلَ وبسَطَ يدَيهِ نحوَ السَّماءِ</span>
<span className="v" id="v23" data-v="23" data-xref="صموئيلَ الثّانِي 7:22; التَّثْنِيَة 7:9; دانيال 9:4"> وقالَ: ”أيُّها الرّبُّ إلهُ إِسرائيلَ، لا إلهَ مِثلُكَ في السَّماءِ ولا في الأرضِ، حافِظُ العَهدِ وصانِعُ الرَّحمةِ لِعبيدِكَ الّذينَ يسلُكونَ طُرُقَكَ بِكُلِّ قلوبِهِم،</span>
<span className="v" id="v24" data-v="24" data-xref="صموئيلَ الثّانِي 7:16; المُلُوكِ الأوَّلُ 8:15; صموئيلَ الثّانِي 7:12"> أنتَ يا مَنْ أتَمَّ بِيَدِهِ اليومَ ما بهِ وعَدَ عبدَهُ داوُدَ أبي.</span>
<span className="v" id="v25" data-v="25" data-xref="المُلُوكِ الأوَّلُ 2:4; المُلُوكِ الأوَّلُ 9:4–6; إرْمِيا 33:17–26"> والآنَ أيُّها الرّبُّ إلهُ إِسرائيلَ اَحفَظْ لِعبدِكَ داوُدَ أبي عَهدَكَ لَه: لا ينقَطِعُ مِنْ نَسلِكَ رَجُلٌ يجلِسُ على عرشِ إِسرائيلَ، إذا لَزِمَ بَنوكَ الطَّريقَ القويمَ كما سَلكْتَ أنتَ أمامي.</span>
<span className="v" id="v26" data-v="26" data-xref="صموئيلَ الثّانِي 7:25–29; إشَعْياء 41:17; صَمُوئيلَ الأوَّلُ 1:17"> والآنَ يا إلهَ إِسرائيلَ، لِيَتَحَقَّقِ القولُ الّذي وعَدْتَ بهِ عبدَكَ داوُدَ أبي.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="أخبارِ الأيّامِ الثّانِي 2:6; إشَعْياء 66:1; إرْمِيا 23:24"> ”ولكِنْ هل تسكُنُ يا اللهُ حَقًا على الأرضِ؟ حتّى السَّماواتُ وسماواتُ السَّماواتِ لا تَتَّسِعُ لكَ، فكيفَ هذا الهَيكلُ الّذي بَنَيتُهُ لكَ؟</span>
<span className="v" id="v28" data-v="28" data-xref="المَزامِير 88:1–2; المَزامِير 5:1; لُوقا 18:1"> إلتَفِتْ إلى صلاتي وتضَرُّعي، أنا عبدُكَ، أيُّها الرّبُّ إلهي، واَسمَعْ دُعائي وصلاتي أمامَكَ اليومَ.</span>
<span className="v" id="v29" data-v="29" data-xref="أخبارِ الأيّامِ الثّانِي 6:20; المُلُوكِ الأوَّلُ 8:16; أخبارِ الأيّامِ الثّانِي 7:15–16"> لِتكُنْ عيناكَ مَفتوحَتَينِ ليلًا نهارًا على هذا الهَيكلِ، على المَوضِعِ الّذي قُلتَ يكونُ اَسمُك فيهِ لِتَسمَعَ صلاتي، أنا عبدُكَ،</span>
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الأوَّلُ 8:39; المُلُوكِ الأوَّلُ 8:34; المُلُوكِ الأوَّلُ 8:36"> واَسْتَجِبْ إلى تَضَرُّعِ عبدِكَ وبَني إِسرائيلَ شعبِكَ الّذينَ يُصَلُّونَ شاخِصينَ إلى هذا المَوضِعِ، واَسمَعْ أنتَ مِنْ مُقامِكَ في السَّماءِ وإذا سَمِعْتَ فاَغْفِرْ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 22:8–11; أخبارِ الأيّامِ الثّانِي 6:22–23; اللّاويِّين 5:1"> ”إذا اَتُّهِمَ أحدُهُم بِالإساءَةِ إلى آخَرَ وجيءَ بهِ إلى مذبَحِكَ في هذا الهَيكلِ لِيَحلِفَ أنَّهُ بَريءٌ،</span>
<span className="v" id="v32" data-v="32" data-xref="التَّثْنِيَة 25:1; إشَعْياء 3:10–11; ﺭﻭﻣﻴﺔ 2:6–10"> فاَسمَعْ أنتَ مِنَ السَّماءِ واَحكُمْ بَينَ عبيدِكَ، فتَدينَ المُذنِبَ وتُعاقِبَهُ والبَريءَ تُكافِئُهُ على بَراءَتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="اللّاويِّين 26:17; التَّثْنِيَة 28:25; ﻳﻮﻧﺎﻥ 3:10"> ”وإذا اَنْهَزَمَ بَنو إِسرائيلَ شعبكَ أمامَ أعدائِهِم بِسبَبِ خطيئَتِهِم إليكَ، ثُمَّ تابوا إليكَ واَعتَرَفوا باَسمِكَ وصلَّوا إليكَ وتضَرَّعوا في هذا الهَيكلِ،</span>
<span className="v" id="v34" data-v="34" data-xref="إرْمِيا 31:4–9; دانيال 9:12; التَّكوين 13:15"> فَاَسمَعْ مِنَ السَّماءِ واَغْفِرْ خطيئَتَهم ورُدَّهُم إلى الأرضِ الّتي أعطَيتَها لآبائِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="اللّاويِّين 26:19; التَّثْنِيَة 11:17; المُلُوكِ الأوَّلُ 8:33"> ”وإذا حبَسْتَ مِنَ السَّماءِ عَنهُمُ المطَرَ، بِسبَبِ خطيئَتِهِم إليكَ، وصَلَّوا شاخِصينَ إلى هذا المَوضِعِ واَعتَرَفوا باَسمِكَ وعَنْ خطيئَتِهِم رَجَعوا بَعدَما أذلَلتَهُم،</span>
<span className="v" id="v36" data-v="36" data-xref="المَزامِير 27:11; صَمُوئيلَ الأوَّلُ 12:23; إرْمِيا 6:16"> فاَسمَعْ مِنَ السَّماءِ واَغْفِرْ خطيئةَ عَبيدِكَ وشعبِكَ بَني إِسرائيلَ، واَهدِهِمِ الطَّريقَ الصَّالحَ، وأنزِلْ مطَرًا على أرضِكَ هذِهِ الّتي أعطَيتَها لِشعبِك مِيراثًا.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="التَّثْنِيَة 28:21–22; اللّاويِّين 26:16; أخبارِ الأيّامِ الثّانِي 20:9"> ”وإذا حدَثَ في الأرضِ جُوعٌ أو وَبَاءٌ أو جَفافٌ أو يَرَقانٌ أو جَرادٌ أو سِلٌّ، وإذا حاصَرَهُم أعداؤُهُم في أيَّةِ مدينةٍ مِنْ مُدُنِهِم، أو بِهِم نزَلَت آفَةٌ أو داءٌ،</span>
<span className="v" id="v38" data-v="38" data-xref="المَزامِير 73:21–22; المُلُوكِ الأوَّلُ 8:22; فِيلِبِّي 4:6"> وإذا أيُّ واحدٍ مِنْ بَني إِسرائيلَ شعبِكَ نَدِمَ في قلبِهِ على ما فعَلَ مِنْ سُوءٍ، فصلَّى إليكَ وتضَرَّعَ باسِطًا يدَيهِ نحوَ هذا الهَيكلِ،</span>
<span className="v" id="v39" data-v="39" data-xref="أخبارِ الأيّامِ الأوّلُ 28:9; إرْمِيا 17:10; اعمال الرسل 1:24"> فاَسمَعْ مِنَ السَّماءِ، مِنْ مُقامِكَ، واَغْفِرْ وأعِنْ مُجازِيًا كُلَّ واحدٍ بِحسَبِ عمَلِهِ ومَعرِفَتِكَ ما في قلبِهِ، لأنَّكَ أنتَ وحدَك تعرِفُ قُلوبَ جميعِ بَني البشَرِ،</span>
<span className="v" id="v40" data-v="40" data-xref="المَزامِير 130:4; رُؤيا يُوحَنّا 19:5; التَّثْنِيَة 6:2"> لِيَخافوكَ كُلَّ أيّامِ حياتِهِم على هذِهِ الأرضِ الّتي أعطَيتَها لآبائِنا.</span>
</div>

<span className="v" id="v42" data-v="42" data-xref="التَّثْنِيَة 3:24; المُلُوكِ الثّانِي 17:36; الخُرُوج 3:13–16"> لأنَّ النَّاسَ يسمَعونَ باَسمِكَ العظيمِ ويدِكَ القديرةِ وذِراعِكَ المَمدودةِ وصلَّى في هذا الهَيكلِ</span>
<span className="v" id="v43" data-v="43" data-xref="المَزامِير 102:15; صَمُوئيلَ الأوَّلُ 17:46; المَزامِير 22:27"> فاَسمَعْ مِنَ السَّماءِ، مِنْ مُقامِكَ، واَعمَلْ بِجميعِ ما يلتَمِسُ مِنكَ هذا الغريبُ، لِيَعرِفَ جميعُ أُمَمِ الأرضِ اَسمَكَ ويَخافوكَ مِثلَ إِسرائيلَ شعبِكَ، ويعلَموا أنَّ اَسمَكَ على هذا الهَيكلِ الّذي بَنَيتُهُ لكَ.</span>

<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="أخبارِ الأيّامِ الثّانِي 6:34; المَزامِير 78:67–69; المُلُوكِ الأوَّلُ 8:16"> ”وإذا خرَجَ شعبُكَ لِمُحارَبةِ أعدائِهِم وَفقَ مشيئَتِكَ، وصلَّوا إليكَ جِهَةَ المدينةِ الّتي اَختَرْتَها والهَيكلِ الّذي بَنَيتُهُ لاَسمِكَ،</span>
<span className="v" id="v45" data-v="45" data-xref="التَّكوين 18:25; المَزامِير 9:4; إرْمِيا 5:28"> فاَسمَعْ مِنَ السَّماءِ صلاتَهُم وتضَرُّعَهُم واَنصُرْ حَقَّهُم.</span>

<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="الجامِعَة 7:20; الأمثال 20:9; رسالَةُ يُوحَنّا الأُولَى 1:8–10"> ”وإذا خطِئوا إليك، كما لا بُدَّ أنْ يَخطَأَ كُلُّ إنسانٍ، وغَضِبْتَ علَيهِم وهَزمتَهُم أمامَ أعدائِهِم، فسَبَوهُم إلى بِلادٍ بعيدةٍ أو قريبةٍ،</span>
<span className="v" id="v47" data-v="47" data-xref="المَزامِير 106:6; نَحَمْيا 1:6–11; نَحَمْيا 9:26–30"> ثُمَّ رَجِعوا إلى نُفوسِهِم في أرضِ سَبْيِهِم، فتابوا وتضَرَّعوا إليكَ وقالوا: خَطِئنا وأثِمْنا وأجرَمْنا،</span>
<span className="v" id="v48" data-v="48" data-xref="دانيال 6:10; إرْمِيا 29:12–14; التَّثْنِيَة 4:29"> وأقبَلوا إليكَ بِكُلِ قلوبِهِم ونُفوسِهِم، وصلَّوا إليكَ جِهَةَ أرضِهِمِ الّتي أعطَيتَها لآبائِهِم والمدينةِ الّتي اَختَرتَها والهَيكلِ الّذي بنَيتُهُ لاَسمِكَ،</span>
<span className="v" id="v49" data-v="49" data-xref="المُلُوكِ الثّانِي 19:19; المُلُوكِ الأوَّلُ 8:45; زَكَريّا 1:15–16"> فاَسمَعْ مِنَ السَّماءِ، مِنْ مكانِ سُكناكَ، صلاتَهُم وتضَرُّعَهُم واَستَجِب لهُم،</span>
<span className="v" id="v50" data-v="50" data-xref="المَزامِير 106:46; أخبارِ الأيّامِ الثّانِي 30:9; اعمال الرسل 7:10"> واَغفِرْ لهُم جميعَ إساءاتِهِم إليكَ، واَرحَمْهُم أمامَ الّذينَ سَبَوهُم فيَرحَموهُم.</span>
<span className="v" id="v51" data-v="51" data-xref="نَحَمْيا 1:10; إرْمِيا 11:4; التَّثْنِيَة 4:20"> فهُم شعبُكَ وميراثُكَ الّذينَ أخرَجتَهُم مِنْ مِصْرَ، مِنْ وسَطِ أتونِ العذابِ.</span>
<span className="v" id="v52" data-v="52" data-xref="المَزامِير 145:18; المُلُوكِ الأوَّلُ 8:29; المَزامِير 86:5"> لِتكُنْ عيناكَ مَفتوحَتَينِ إلى تضَرُّعِ عَبدِكَ وبَني إِسرائيلَ شعبِكَ، واَستَمِعْ إليهِم في كُلِّ ما يلتَمِسونَهُ مِنكَ.</span>
<span className="v" id="v53" data-v="53" data-xref="الخُرُوج 19:5–6; التَّثْنِيَة 14:2; التَّثْنِيَة 32:9"> فأنتَ اَختَرتَهُم لكَ شعبًا مِنْ بَينِ جميعِ شعوبِ الأرضِ، كما تكَلَّمْتَ على لِسانِ موسى عَبدِكَ، حينَ أخرَجتَ آباءَنا مِنْ مِصْرَ أيُّها الرّبُّ الإلهُ“.</span>

## سليمان يبارك شعبه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v54" data-v="54" data-xref="أخبارِ الأيّامِ الثّانِي 7:1; أخبارِ الأيّامِ الثّانِي 6:12–13; لُوقا 22:45"> فلمَّا أتَمَّ سُليمانُ صلاتَهُ، قامَ مِنْ أمامِ مذبَحِ الرّبِّ حَيثُ كانَ راكعًا، ويَداهُ مَرفوعَتانِ إلى السَّماءِ،</span>
<span className="v" id="v55" data-v="55" data-xref="المُلُوكِ الأوَّلُ 8:14; صموئيلَ الثّانِي 6:18; العَدَد 6:23–26"> ووقَفَ وبارَكَ الحاضِرينَ بِصوتٍ عالٍ فقالَ:</span>
<span className="v" id="v56" data-v="56" data-xref="يَشُوع 23:14–15; يَشُوع 21:44–45; لُوقا 21:33"> ”تبارَكَ الرّبُّ الّذي وهبَ الرَّاحةَ لإِسرائيلَ شعبِهِ، كما تكَلَّمَ على لِسانِ موسى عَبدِهِ، ولم تسقُطْ كلِمَةٌ واحدةٌ مِنْ جميعِ أقوالِهِ الصَّالحةِ.</span>
<span className="v" id="v57" data-v="57" data-xref="يَشُوع 1:5; التَّثْنِيَة 31:6; التَّثْنِيَة 31:8"> لِيكُنِ الرّبُّ إلهُنا معَنا كما كانَ معَ آبائِنا، ولا يَخذُلُنا ولا يَرفُضُنا،</span>
<span className="v" id="v58" data-v="58" data-xref="المَزامِير 119:36; حِزْقِيال 36:26–27; إرْمِيا 31:33"> وليَمِلْ بِقلوبِنا إليهِ لِنَسلُكَ جميعَ طُرقِهِ ونعمَلَ بِوَصاياهُ وفرائِضِهِ وأحكامِهِ الّتي بِها أمرَ آباءَنا.</span>
<span className="v" id="v59" data-v="59" data-xref="المَزامِير 141:2; يُوحَنّا 17:9; يُوحَنّا 17:20–24"> ولتكُنْ صلاتي إلى الرّبِّ إلهِنا قريبةً مِنهُ نهارًا وليلًا، حتّى يستَجيبَ لِحاجةِ عَبدِهِ ولِحاجةِ بَني إِسرائيلَ شعبِهِ كُلَّ يومٍ في يومِهِ،</span>
<span className="v" id="v60" data-v="60" data-xref="التَّثْنِيَة 4:35; إرْمِيا 10:10–12; المُلُوكِ الأوَّلُ 18:39"> لِتَعلَمَ جميعُ شعوبِ الأرضِ أنَّ الرّبَّ هوَ الإلهُ، ولا أحدَ سِواهُ.</span>
<span className="v" id="v61" data-v="61" data-xref="المُلُوكِ الأوَّلُ 11:4; المُلُوكِ الثّانِي 20:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1"> فلتكُنْ قُلوبُكُم مُخلِصَةً لِلرّبِّ إلهِنا، لِتَسلُكوا في فرائِضِهِ وتعمَلوا بِوَصاياهُ، كما تفعَلونَ اليومَ“.</span>

## تدشين الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v62" data-v="62" data-xref="صموئيلَ الثّانِي 6:17–19; عَزْرا 6:16–17; أخبارِ الأيّامِ الثّانِي 7:4–10"> ثُمَّ قدَّمَ المَلِكُ وجميعُ بَني إِسرائيلَ معَهُ أمامَ الرّبِّ</span>
<span className="v" id="v63" data-v="63" data-xref="عَزْرا 6:16–17; أخبارِ الأيّامِ الثّانِي 15:11; نَحَمْيا 12:27"> ذبائِحَ سلامةٍ لِلرّبِّ اثنَينِ وعِشرينَ ألفًا مِنَ البقَرِ ومئةً وعِشرينَ ألفًا مِنَ الغنَمِ، ودَشَّنوا الهَيكلَ.</span>
<span className="v" id="v64" data-v="64" data-xref="أخبارِ الأيّامِ الثّانِي 4:1; أخبارِ الأيّامِ الثّانِي 7:7"> وفي ذلِكَ اليومِ كرَّسَ المَلِكُ لِلرّبِّ ساحةَ الدَّارِ الّتي أمامَ الهَيكلِ، لأنَّهُ هُناكَ قدَّمَ المُحرَقةَ والتَّقدِمةَ وذبائِحَ السَّلامةِ. لأنَّ مذبَحَ النُّحاسِ الصَّغيرَ الّذي كانَ أمامَ الرّبِّ داخلَ الهَيكلِ لم يكُنْ يسَعُها.</span>
