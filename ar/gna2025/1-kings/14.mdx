## مرض اَبن يربعام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 20:5; المُلُوكِ الأوَّلُ 13:33–34; صَمُوئيلَ الأوَّلُ 31:2"> في تِلكَ الأيّامِ مَرِضَ أبيَّا بنُ يَرُبعامَ،</span>
<span className="v" id="v2" data-v="2" data-xref="صَمُوئيلَ الأوَّلُ 28:8; المُلُوكِ الأوَّلُ 11:29–38; صموئيلَ الثّانِي 14:2"> فقالَ يَرُبعامُ لِزَوجَتِهِ: ”قومي تنَكَّري حتّى لا يَعلمَ أحدٌ أنَّكِ زَوجَتي، واَذهَبي إلى شيلُوه حَيثُ يُقيمُ أخيَّا النَّبيُّ الّذي تنبَّأَ لي أنِّي سأكونُ مَلِكًا لِهذا الشَّعبِ.</span>
<span className="v" id="v3" data-v="3" data-xref="صَمُوئيلَ الأوَّلُ 9:7–8; المُلُوكِ الأوَّلُ 13:7; المُلُوكِ الثّانِي 4:42"> وخُذي في يَدِكِ عشَرَةَ أرغِفةٍ وكَعكًا وجَرَّةَ عسَلٍ وروحي إليهِ وهوَ يُخبِرُكِ عَمَّا يُصيبُ الولَدَ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الأوَّلُ 11:29; صَمُوئيلَ الأوَّلُ 3:2; صَمُوئيلَ الأوَّلُ 4:15"> فذَهبَت إلى شيلُوه ودخَلَت بَيتَ أخيَّا النَّبيِّ، وكانَ لا يُبصِرُ لأنَّ عَينَيهِ ضَعُفَتا مِنَ الشَّيخوخةِ.</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الثّانِي 6:8–12; الأمثال 21:30; المُلُوكِ الثّانِي 4:27"> فقالَ الرّبُّ لأخيَّا: ”هذِهِ زَوجَةُ يَرُبعامَ قادِمةٌ إليكَ تَستَخبِرُكَ عَنِ اَبنِها المريضِ. فكَلِّمْها بِكذا وكذا“. فلمَّا دخَلَت، وهيَ مُتنَكِّرَةٌ</span>
<span className="v" id="v6" data-v="6" data-xref="دانيال 5:17–28; اعمال الرسل 5:9–10; لُوقا 20:20–23"> سَمِعَ وَقْعَ خطَواتِها في البابِ فقالَ لها: ”أُدخُلي يا زَوجَةَ يَرُبعامَ. لِماذا أنتِ مُتَنكِّرَةٌ، ولكِ عِندي خَبَرٌ سَيِّئٌ؟</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الأوَّلُ 16:2; صموئيلَ الثّانِي 12:7–8; صَمُوئيلَ الأوَّلُ 15:16"> عودي وقولي لِيَرُبعامَ: هذا ما قالَ الرّبُّ إلهُ إِسرائيلَ: إختَرتُكَ مِنْ عامَّةِ الشَّعبِ وجعَلتُكَ رئيسًا على مَملكَةِ إِسرائيلَ،</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الأوَّلُ 15:5; المُلُوكِ الأوَّلُ 11:33–38; أخبارِ الأيّامِ الثّانِي 28:1"> واَنتَزَعْتُ المُلْكَ مِنْ بَيتِ داوُدَ وأعطَيتُهُ لكَ. لكِنَّكَ لم تكُنْ كَعَبدي داوُدَ الّذي عَمِلَ بِوَصايايَ، وتَبِعَني بِكُلِّ قلبِهِ، ولم يعمَلْ إلَّا ما هوَ قَويمٌ في نظَري،</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 23:35; نَحَمْيا 9:26; المَزامِير 50:17"> بل زادَ عمَلُكَ سُوءًا على كُلِّ مَنْ كانَ قَبلَكَ. فنَبَذتَني وأغَظتَني حينَ صَنَعْتَ لِنَفْسِكَ أصنامًا ومَسبوكاتٍ لِلعِبادةِ.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 32:36; المُلُوكِ الأوَّلُ 21:21; المُلُوكِ الثّانِي 14:26"> لذلِكَ أجلِبُ الشَّرَّ على بَيتِكَ، فأُبيدُ لكَ كُلَّ ذَكَرٍ، مِنَ الكبيرِ إلى الصَّغيرِ، وأُفنيهِم عَنْ آخِرِهِم، كما يُكنَّسُ البَعْرُ،</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الأوَّلُ 16:4; إرْمِيا 15:3; المُلُوكِ الأوَّلُ 21:23–24"> ومَنْ ماتَ لكَ في المدينةِ تأكُلُهُ الكلابُ، ومَنْ ماتَ في البرِّيَّةِ تأكُلُهُ طَيرُ السَّماءِ. هوَ الرّبُّ تكلَّمَ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المُلُوكِ الأوَّلُ 14:3; المُلُوكِ الأوَّلُ 14:16–17; المُلُوكِ الثّانِي 1:6"> ثُمَّ قالَ أخيَّا لِزَوجَةِ يَرُبعامَ: ”عُودي إلى بَيتِكِ، وعِندَ دُخولِكِ المدينةَ يموتُ الولَدُ،</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الثّانِي 19:3; أخبارِ الأيّامِ الثّانِي 12:12; أيُّوب 19:28"> فيَندُبُهُ شَعب إِسرائيلَ ويقبِرونَهُ، وهوَ الوحيدُ الّذي يُقبَرُ مِنْ بَيتِ يَرُبعامَ. لأنَّهُ الوحيدُ مِنْ بَيتِ يَرُبعامَ الّذي فيهِ شيءٌ يُرضي الرّبَّ إلهَ إِسرائيلَ،</span>
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الأوَّلُ 15:27–29; حِزْقِيال 7:2–7; حِزْقِيال 12:22–28"> وسيُقيمُ الرّبُّ مَلِكًا على إِسرائيلَ لِيُبيدَ بَيتَ يَرُبعامَ،</span>
<span className="v" id="v15" data-v="15" data-xref="يَشُوع 23:15–16; المَزامِير 52:5; المُلُوكِ الثّانِي 15:29"> ويضرِبُ الرّبُّ شعبَ إِسرائيلَ فيَهتَزُّونَ كالقصَبِ في الماءِ، ويقتَلِعُهُم مِنْ هذِهِ الأرضِ الصَّالحةِ الّتي أعطاها آباءَهُم ويُبَدِّدُهُم إلى عِبرِ الفُراتِ، لأنَّهُم أغاظوهُ بِإقامةِ أنصابٍ لأشيرةِ،</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الأوَّلُ 15:34; المُلُوكِ الأوَّلُ 16:2; المُلُوكِ الأوَّلُ 13:34"> ويتَخَلَّى عَنْ إِسرائيلَ لأنَّ يَرُبعامَ خطِئَ وجعَلَ شعبَ إِسرائيلَ يَخطَأُ“.</span>
</div>
</div>

## موت يربعام وابنه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الأوَّلُ 15:33; المُلُوكِ الأوَّلُ 16:15; المُلُوكِ الأوَّلُ 16:23"> فرجَعَت زَوجَةُ يَرُبعامَ إلى تِرْصَةَ، وعِندَ دُخولِها عتَبةَ بابِ بَيتِها ماتَ الولَدُ.</span>
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الأوَّلُ 14:13"> فدَفنَهُ شعبُ إِسرائيلَ، وندَبوهُ كما تكلَّمَ الرّبُّ على لِسانِ عَبدِهِ أخيَّا النَّبيِّ.</span>
<span className="v" id="v19" data-v="19" data-xref="أخبارِ الأيّامِ الثّانِي 13:2–20; أخبارِ الأيّامِ الأوّلُ 27:24; المُلُوكِ الأوَّلُ 16:20"> وما بَقيَ مِنْ أخبارِ يَرُبعامَ: كيفَ حاربَ وكيفَ مَلَكَ مُدَوَّنَةٌ في سِفرِ الأخبارِ لِمُلوكِ إِسرائيلَ.</span>
<span className="v" id="v20" data-v="20" data-xref="أيُّوب 14:12; المَزامِير 3:5; المُلُوكِ الأوَّلُ 11:43"> وملَكَ يَرُبعامُ اَثنَتَينِ وعِشرينَ سنَةً وماتَ، فمَلَكَ نادابُ اَبنُهُ مكانَهُ.</span>
</div>
</div>

## رحبعام ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="أخبارِ الأيّامِ الثّانِي 12:13; المُلُوكِ الأوَّلُ 11:36; المُلُوكِ الأوَّلُ 14:31"> وكانَ رَحُبعامُ بنُ سُليمانَ اَبنَ إحدى وأربَعينَ سنَةً حينَ مَلَكَ في يَهوذا، وملَكَ سَبعَ عشْرَةَ سنَةً بِأُورُشليمَ المدينةِ الّتي اَختارَها الرّبُّ مِنْ جميعِ مُدُنِ إِسرائيلَ لِيَجعَلَ اَسمَهُ هُناكَ. واَسمُ أمِّهِ نِعمَةُ العَمُّونيَّةُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="أخبارِ الأيّامِ الثّانِي 12:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:22; المَزامِير 78:58"> وفعَلَ شعبُ يَهوذا الشَّرَّ أمامَ الرّبِّ، وأثاروا غضَبَهُ بِخَطاياهُم أكثَر مِمَّا أثارَهُ جميعُ آبائِهِم.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 12:2; إشَعْياء 57:5; المُلُوكِ الثّانِي 17:9–10"> فبَنَوا لأنْفُسِهِم مذابِحَ وأنصابًا ورُموزًا لأشيرةِ على كُلِّ تَلَّةٍ عاليةٍ وتَحتَ كُلِّ شجَرةٍ خضراءَ.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّثْنِيَة 23:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:9; المُلُوكِ الأوَّلُ 15:12"> كذلِكَ في أرضِهِم كانَ رِجالٌ ونِساءٌ يُمارِسونَ البَغاءَ المُكَرَّسَ، واَرتكَبوا جميعَ رجاساتِ الأُمَمِ الّتي طرَدَها الرّبُّ مِنْ أمامِ بَني إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="المُلُوكِ الأوَّلُ 11:40; أخبارِ الأيّامِ الثّانِي 12:2–4; أخبارِ الأيّامِ الثّانِي 12:9"> وفي السَّنةِ الخامِسةِ لِلمَلِكِ رَحُبعامَ صَعِدَ شيشَقُ مَلِكُ مِصْرَ لِمُحاربَةِ أورُشليمَ،</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الأوَّلُ 15:18; أخبارِ الأيّامِ الثّانِي 9:15–16; المُلُوكِ الأوَّلُ 7:51"> فنَهَبَ كُلَّ ما في خزائِنِ هَيكلِ الرّبِّ وقصرِ المَلِكِ، وأخذَ تُروسَ الذَّهبِ الّتي صنَعَها سُليمانُ.</span>
<span className="v" id="v27" data-v="27" data-xref="صَمُوئيلَ الأوَّلُ 8:11; صَمُوئيلَ الأوَّلُ 22:17; المُلُوكِ الأوَّلُ 18:46"> فصَنَعَ رَحُبعامُ مكانَها تُروسًا مِنْ نُحاسٍ، وسَلَّحَ بِها حُرَّاسَ قصرِهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="أخبارِ الأيّامِ الثّانِي 12:11"> وكانَ إذا دخَلَ المَلِكُ هَيكلَ الرّبِّ يَحمِلُها الحرَسُ إلى هُناكَ، ثُمَّ يَرُدُّونَها إلى غُرفةِ الحِراسةِ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="المُلُوكِ الأوَّلُ 22:45; المُلُوكِ الأوَّلُ 15:23; المُلُوكِ الأوَّلُ 14:19"> وما بَقيَ مِنْ أخبارِ رَحُبعامَ وكُلِّ أعمالِهِ مُدوَّنَةٌ في سِفرِ أخبارِ الأيّامِ لِمُلوكِ يَهوذا.</span>
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الأوَّلُ 12:24; المُلُوكِ الأوَّلُ 12:21; المُلُوكِ الأوَّلُ 15:6–7"> وكانَ بَينَ رَحُبعامَ ويَرُبعامَ حربٌ مُستَديمةٌ</span>
<span className="v" id="v31" data-v="31" data-xref="أخبارِ الأيّامِ الثّانِي 12:16; ﻣﺘﻰ 1:7; المُلُوكِ الأوَّلُ 15:3"> وماتَ رَحُبعامُ ودُفِنَ معَ آبائِهِ في مدينةِ داوُدَ، وملَكَ أبيامُ اَبْنُهُ مكانَهُ.</span>
</div>
