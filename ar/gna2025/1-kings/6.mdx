## بناء الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أخبارِ الأيّامِ الثّانِي 3:1–2; اعمال الرسل 7:47; أخبارِ الأيّامِ الأوّلُ 29:19"> وفي السَّنةِ الأربَعِ مئَةٍ والثَّمانينَ لِخُروجِ بَني إِسرائيلَ مِنْ أرضِ مِصْرَ، في السَّنةِ الرَّابِعةِ مِنْ مُلْكِ سُليمانَ على إِسرائيلَ، في زِيو وهوَ الشَّهرُ الثَّاني، بدأ سُليمانُ بِبِناءِ هَيكلِ الرّبِّ.</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 40:1–41; حِزْقِيال 41:1–15; رُؤيا يُوحَنّا 21:16–17"> وكانَ الهَيكلُ سِتِّينَ ذِراعًا في الطُّولِ، وعِشرينَ في العَرضِ، وثَلاثينَ ذِراعًا في العُلُوِّ.</span>
<span className="v" id="v3" data-v="3" data-xref="أخبارِ الأيّامِ الأوّلُ 28:11; أخبارِ الأيّامِ الثّانِي 3:3–4; حِزْقِيال 41:15"> وكانَ الرِّواقُ أمامَ الهَيكلِ عِشرينَ ذِراعًا طولًا، على مُحاذاةِ عَرضِ الهَيكلِ، وعَشْرَ أذرُعٍ عَرضًا أمامَهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 40:16; حِزْقِيال 41:26; حِزْقِيال 41:16"> وصنَعَ لَه نَوافِذَ مُشَبَّكةً بِعَوارِضَ مَتينَةٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الأوَّلُ 6:16; المُلُوكِ الأوَّلُ 6:19–21; المُلُوكِ الأوَّلُ 6:31"> وبَنى على جَوانِبِهِ مِنْ حَولِهِ مُلحَقًا بِثَلاثِ طبقاتٍ يُحيطُ بِالقاعةِ والمِحرابِ، وصنَعَ غُرُفاتٍ على مُحيطِهِ.</span>
<span className="v" id="v6" data-v="6"> فالطَّبقَةُ السُّفلى عَرضُها خَمسُ أذرُعٍ، والوُسْطى عَرضُها سِتُّ أذرُعٍ، والثَّالِثَةُ عَرضُها سَبعُ أذرُعٍ، لأنَّه ترَكَ مساحةً في خارِجِ جُدرانِ كُلِّ طَبقَةٍ على مُحيطِها، حتّى تَثبُتَ في مكانِها مِنْ دونِ إدخالِ عَوارِضِها في داخِلِ الجُدرانِ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 27:5–6; رسالَةُ بُطرُسَ الأُولَى 2:5; الخُرُوج 20:25"> وبَنى البَيتَ بِحجارةٍ أُعِدَّت في المَقلَعِ، فلم يكُنْ أحدٌ يسمَعُ صوتَ مِطرَقَةٍ ولا إزميلٍ ولا أداةٍ مِنْ حديدٍ في الهَيكلِ عِندَ بِنائِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 41:6–7"> وكانَ بابُ الغُرفَةِ السُّفلى على الجانِبِ الأيمَنِ مِنَ الهَيكلِ، وكانَ يُصعَدُ في دَرَجٍ لَولَبيٍّ إلى الطَّبقَةِ الوُسطى، ومِنَ الوُسطى إلى الثَّالِثَةِ.</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الأوَّلُ 6:14; المُلُوكِ الأوَّلُ 6:38"> فبَنى سُليمانُ الهَيكلَ وأكمَلَهُ وسَقَفَهُ بِجُذوعٍ وألواحٍ مِنَ الأرزِ.</span>
<span className="v" id="v10" data-v="10"> وبَنى الطَّبَقاتِ على جَوانِبِ الهَيكلِ كُلِّهِ، عُلُوُّ كُلِّ طَبقَةٍ خَمسُ أذرُعٍ، ووَصَلها بالهَيكلِ بِخشَبِ الأرْزِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11"> وقالَ الرّبُّ لِسُليمانَ:</span>
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الأوّلُ 22:10; كُولُوسِي 1:23; المُلُوكِ الأوَّلُ 3:14"> ”إذا عَمِلْتَ بِفَرائِضي وأحكامي وحَرِصْتَ على اَتِّباعِ جميعِ وصايايَ، فأنا أفي لكَ بِوَعدي الّذي وَعَدْتُ بهِ أباكَ داوُدَ.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّثْنِيَة 31:6; الخُرُوج 25:8; اللّاويِّين 26:11"> وأُقيمُ في هذا الهيكلِ الّذي تَبنيهِ وأكونُ معَ شَعبي بَني إِسرائيلَ، ولا أترُكُهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الأوَّلُ 6:9; المُلُوكِ الأوَّلُ 6:38; اعمال الرسل 7:47–48"> فبَنى سُليمانُ الهَيكلَ وأكمَلَهُ.</span>
</div>
</div>

## الترتيب الداخلي: قدس الأقداس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الأوَّلُ 7:7"> وعلى جُدرانِهِ مِنَ الدَّاخِلِ بَنى ألواحَ أرْزٍ، وصَفَّحَ داخِلَه مِنَ الأرضِ إلى السَّقفِ بالخشَبِ، وفرَشَ أرضَهُ بألواحِ السَّرْوِ.</span>
<span className="v" id="v16" data-v="16" data-xref="أخبارِ الأيّامِ الثّانِي 3:8; المُلُوكِ الأوَّلُ 8:6; العِبرانِيّين 9:3"> وفي مُؤخَّرِ الهَيكلِ على مسافَةِ عِشرينَ ذِراعًا بَنى ألواحَ أرْزٍ مِنَ الأرضِ إلى السَّقفِ، بَناها في داخِلِهِ مِحرابًا هوَ قُدْسُ الأقداسِ.</span>
<span className="v" id="v17" data-v="17"> فكانَ مُقَدَّمُ الهَيكلِ أربَعينَ ذِراعًا.</span>
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الأوَّلُ 7:24"> وكانَ على الهَيكلِ مِنَ الدَّاخِلِ أرزٌ مَنقوشٌ على شكلِ يَقطينٍ وزُهورٍ مُتَفَتِّحَةٍ، فلم يكُنْ يَبينُ حجَرٌ لأنَّ كُلَّهُ مِنَ الدَّاخِلِ كانَ أرْزًا.</span>
<span className="v" id="v19" data-v="19" data-xref="المُلُوكِ الأوَّلُ 8:6–10; المَزامِير 28:2; المُلُوكِ الأوَّلُ 6:5"> وهَيَّأ المِحرابَ في باطِنِ الهَيكلِ لِيضَعَ هُناكَ تابوتَ عَهدِ الرّبِّ</span>
<span className="v" id="v20" data-v="20" data-xref="المُلُوكِ الأوَّلُ 6:2–3; المُلُوكِ الأوَّلُ 6:22; المُلُوكِ الأوَّلُ 7:48"> وكانَ طولُ المِحرابِ عِشرينَ ذِراعًا، وعَرضُهُ عِشرينَ ذِراعًا، وعُلُوُّهُ عِشرينَ ذِراعًا، وغَطَّاهُ بِذَهَبٍ خالصٍ. وصنَعَ مذبَحًا مِنَ الأرْزِ.</span>
<span className="v" id="v21" data-v="21" data-xref="أخبارِ الأيّامِ الثّانِي 3:7–9; أخبارِ الأيّامِ الثّانِي 3:14–16; المُلُوكِ الأوَّلُ 6:5"> وغطَّى سُليمانُ داخِلَ الهَيكلِ بِذهَبٍ خالصٍ، ومَدَّ سَلاسِلَ ذهَبٍ أمامَ المِحرابِ،</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 30:1; الخُرُوج 30:3; أخبارِ الأيّامِ الثّانِي 3:7–14"> وغطَّى بالذَّهَبِ الهَيكلَ كُلَّهُ ومذبَحَ المِحرابِ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 37:7–9; التَّكوين 3:24; رسالَةُ بُطرُسَ الأُولَى 1:12"> وصنَعَ في المِحرابِ كَروبيم مِنْ خشَبِ الزَّيتونِ، عُلُوُّ كُلِّ واحدٍ عَشْرُ أذرُعٍ،</span>
<span className="v" id="v24" data-v="24"> والجَناحُ الواحدُ مِنَ الكَروبِ الواحدِ خَمسُ أذرُعٍ، والجَناحُ الآخَرُ خَمسُ أذرُعٍ، فَمِنْ طَرَفِ الجَناحِ الواحدِ إلى طَرَفِ الجَناحِ الآخَرِ عَشْرُ أذرُعٍ.</span>
<span className="v" id="v25" data-v="25"> والكَروبُ الآخَرُ عَشْرُ أذرُعٍ، قياسٌ واحدٌ وصَوغٌ واحدٌ لِلكَروبَينِ.</span>
<span className="v" id="v26" data-v="26"> وعُلُوُّ الكَروبِ الواحدِ عَشْرُ أذرُعٍ، وكذلِكَ الكَروبُ الآخَرُ.</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 37:9; الخُرُوج 25:20; أخبارِ الأيّامِ الثّانِي 5:8"> ووَضَعَ الكَروبَينِ وسَطَ المكانِ الدَّاخلي. وكانَت أجنِحةُ الكَروبَينِ مُفَرَّشةً، فمَسَّ جَناحُ الواحدِ الحائِطَ الواحدَ، وجَناحُ الآخَرِ الحائِطَ الآخَرَ، وتَماسَّت أجنِحَتُهُما في وسَطِ الهَيكلِ.</span>
<span className="v" id="v28" data-v="28"> وغطَّى الكَروبَينِ بالذَّهبِ.</span>
<span className="v" id="v29" data-v="29" data-xref="المُلُوكِ الأوَّلُ 6:32; المَزامِير 92:12–15; أخبارِ الأيّامِ الثّانِي 4:2–5"> ونَقَشَ على مدارِ جُدرانِ الهَيكلِ صُوَرَ كَروبيمَ ونَخيلٍ وزُهورٍ مُتَفتِّحةٍ في الهَيكلِ وما يَتَّصِلُ بهِ.</span>
<span className="v" id="v30" data-v="30" data-xref="رُؤيا يُوحَنّا 21:18–21; إشَعْياء 60:17; إشَعْياء 54:11–12"> وغطَّى بالذَّهبِ أرضَ الهَيكلِ داخِلًا وخارِجًا.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="العِبرانِيّين 10:19–20; أفَسُس 3:18; يُوحَنّا 14:6"> وصنَعَ لِبابِ المِحرابِ مِصراعَينِ مِنْ خشَبِ الزَّيتونِ، وكانَ أعلى المَدخَلِ بارِزًا وعلى شَكلِ قَوسٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="المُلُوكِ الأوَّلُ 6:18; المُلُوكِ الأوَّلُ 6:29"> ونَقَشَ على المِصراعَينِ صُوَرَ كَروبيمَ ونَخيلٍ وزُهورٍ مُتَفتِّحَةٍ وغطَّاهُما بِذهَبٍ، ورَصَّعَ الكَروبيمَ والنَّخيلَ بِالذَّهبِ.</span>
<span className="v" id="v33" data-v="33"> وكذلِكَ صنَعَ لِبابِ القاعةِ قَوائِمَ مِنْ خشَبِ الزَّيتونِ مُربَّعةَ الإطارِ،</span>
<span className="v" id="v34" data-v="34" data-xref="حِزْقِيال 41:23–25; المُلُوكِ الأوَّلُ 5:8"> ومِصراعَينِ مِنْ خشَبِ السَّرْوِ، كُلُّ مِصْراعٍ مِنهُما دَفَّتانِ تَنطَويانِ،</span>
<span className="v" id="v35" data-v="35"> ونقَشَ علَيهِما كَروبيمَ ونَخيلًا وزُهورًا مُتَفتِّحَةً، وغطَّاهُما بِذهَبٍ مُطَعَّمٍ على النَّقشِ.</span>
<span className="v" id="v36" data-v="36" data-xref="المُلُوكِ الأوَّلُ 7:12; أخبارِ الأيّامِ الثّانِي 7:7; الخُرُوج 27:9–19"> وبَنى الدَّارَ الدَّاخِليَّةَ ثَلاثةَ صُفوفٍ مِنَ الحجارةِ المَنحوتَةِ وصفًّا مِنْ عَوارِضِ الأرْزِ.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="المُلُوكِ الأوَّلُ 6:1; أخبارِ الأيّامِ الثّانِي 3:2"> وفي السَّنةِ الرَّابِعةِ، في شهرِ زيو، أسَّسَ هَيكلَ الرّبِّ.</span>
<span className="v" id="v38" data-v="38" data-xref="عَزْرا 6:14–15; المُلُوكِ الأوَّلُ 7:1; زَكَريّا 4:9"> وفي السَّنةِ الحاديةَ عشْرَةَ، في شهرِ بولَ وهوَ الشَّهرُ الثَّامِنُ، أكمَلَ الهَيكلَ بجميعِ أقسامِهِ ومُواصَفاتِهِ. فاَستَغرَقَ بِناؤُهُ سَبْعَ سِنينَ.<br />قصر سليمان</span>
