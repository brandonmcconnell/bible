<div className="bp">

<div className="bp">

<div className="bp">

</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="ميخا 1:4; صموئيلَ الثّانِي 22:8; إرْمِيا 4:24"> مِنَ الرّبِّ ترتعشُ الجِبالُ،<br />ومِنهُ تموجُ جميعُ التِّلالِ.<br />مِنْ أمامِهِ تتراجَعُ الأرضُ،<br />يتراجعُ العالَمُ وسُكَّانُهُ</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إرْمِيا 10:10; مَلاخِي 3:2; التَّثْنِيَة 32:22–23"> مَنْ يقِفُ أمامَ سُخْطِ الرّبِّ<br />ومَنْ يقومُ لَدى اَشتدادِ غضَبِهِ؟<br />يصُبُّ غَيظَهُ كالنَّارِ،<br />ومِنهُ تتفتَّتُ الصُّخورُ.</span>

<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 10:27; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:19; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:25"> الرّبُّ صالِحٌ لِمَنْ يَلوذُ بهِ<br />ويَصونُ المُحتَمي بهِ في الضِّيقِ.</span>

<div className="bp">

<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ناحُوم 1:9; أخبارِ الأيّامِ الثّانِي 13:7; المُلُوكِ الثّانِي 18:13–14"> مِنكِ يا نينَوى خرجَ المُفكِّرُ بالسُّوءِ على الرّبِّ والمُبشِّرُ بالهلاكِ.</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 30:19; يُوئيل 2:19; إشَعْياء 17:14"> وهذا ما قالَ الرّبُّ: ”معَ أنَّ الأشُّوريِّينَ أقوياءُ وكثيرونَ، فسَيُقطَعونَ ويُبادونَ. أذللْتُكَ يا شعبي، فلا أعودُ أُذِلُّكَ،</span>
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 10:27; إشَعْياء 9:4; إرْمِيا 2:20"> بل أكسِرُ الآنَ نيرَ أشُّورَ عنكَ وأقطعُ قُيودَكَ“.</span>
</div>
<div className="bp">
