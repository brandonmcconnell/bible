## بطرس وكورنيليوس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اعمال الرسل 27:1; ﻣﺘﻰ 8:5–13; ﻣﺘﻰ 27:27"> وكانَ في قيصَرِيَّةَ رَجُلٌ اَسمُهُ كُورنيليوسُ، ضابِطٌ مِنَ الفِرقَةِ الإيطالِيَّةِ في الجَيشِ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 7:7–8; اعمال الرسل 10:4; اعمال الرسل 10:22"> كانَ تَقِيًّا يَخافُ اللهَ هوَ وجميعُ أهلِ بَيتِهِ، ويُحسِنُ إلى الشَّعبِ بِسَخاءٍ، ويُداوِمُ على الصَّلاةِ للهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="دانيال 9:20–21; اعمال الرسل 3:1; اعمال الرسل 5:19"> فرأَى نحوَ السّاعَةِ الثَّالِثَةِ مِنَ النَّهارِ في رُؤيا واضِحَةٍ مَلاكَ اللهِ يدخُلُ علَيهِ ويُناديهِ: ”يا كُورنيليوسُ!“</span>
<span className="v" id="v4" data-v="4" data-xref="فِيلِبِّي 4:6; العِبرانِيّين 6:10; اعمال الرسل 10:31"> فنظَرَ إلَيهِ في خَوفٍ وقالَ: ”ما الخبَرُ، يا سيِّدي؟“ فقَالَ لَه المَلاكُ: ”صَعِدَت صَلَواتُكَ وأعمالُكَ الخَيريَّةُ إلى اللهِ، فتَذَكَّركَ.</span>
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 16:9; اعمال الرسل 9:38; اعمال الرسل 15:7"> فأرسِلِ الآنَ رِجالًا إلى يافا وَجِئْ بسِمعانَ الّذي يُقالُ لَه بُطرُسُ.</span>
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 9:43; اعمال الرسل 11:13–14; ﺭﻭﻣﻴﺔ 10:14–17"> فهوَ نازِلٌ عِندَ دبّاغٍ اَسمُهُ سِمعانُ وبَيتُهُ على شاطئِ البحرِ“.</span>
<span className="v" id="v7" data-v="7" data-xref="فِلِيْمُون 1:16; صَمُوئيلَ الأوَّلُ 14:6–7; ﻣﺘﻰ 8:9–10"> فلمَّا اَنصَرَفَ المَلاكُ الّذي كانَ يُكَلِّمُهُ، دَعا اَثنَينِ مِنْ خَدَمِهِ وجُنديًّا تَقيًّا مِنْ أخِصّائِهِ،</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 9:36; المَزامِير 119:59–60; اعمال الرسل 10:33"> وأخبرَهُم بِكُلِّ ما جَرى، وأرسَلَهُم إلى يافا.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="صَفَنْيا 1:5; المَزامِير 55:17; صَمُوئيلَ الأوَّلُ 9:25"> فساروا في الغَدِ. وبَينَما هُم يَقتَرِبونَ مِنْ يافا، صَعِدَ بُطرُسُ إلى السَّطحِ نحوَ الظُّهرِ ليُصلِّيَ،</span>
<span className="v" id="v10" data-v="10" data-xref="اعمال الرسل 22:17; رُؤيا يُوحَنّا 4:2–3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:2–4"> فجاعَ وأرادَ أنْ يأكُلَ. ولمَّا أخَذوا يُهيِّئونَ لَه الطَّعامَ وقَعَ في غَيبوبةٍ،</span>
<span className="v" id="v11" data-v="11" data-xref="أفَسُس 3:6; يُوحَنّا 1:51; ﺭﻭﻣﻴﺔ 16:25–26"> فرأى السَّماءَ مَفتوحَةً، وشيئًا يُشبِهُ قِطعةَ قماشٍ كبيرةً مَعقودَةً بأطرافِها الأربعةِ تَتَدلَّى إلى الأرضِ.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 7:8–9; يُوحَنّا 7:37; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:9–11"> وكانَ علَيها مِنْ جميعِ دَوابِّ الأرضِ وزحّافاتِها وطُيورِ السَّماءِ.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 4:31–34; إرْمِيا 35:2–5; اعمال الرسل 10:10"> وجاءَهُ صَوتٌ يقولُ لَه: ”يا بُطرُسُ، قُمِ اَذبَحْ وكُلْ“.</span>
<span className="v" id="v14" data-v="14" data-xref="اللّاويِّين 20:25; حِزْقِيال 4:14; دانيال 1:8"> فقَالَ بُطرُسُ: ”لا يا ربُّ! ما أكَلتُ في حياتي نَجِسًا أو دَنِسًا“.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:25; ﻣﺘﻰ 15:11; تِيطُس 1:15"> فقَالَ لَه الصَّوتُ ثانيَةً: ”ما طَهَّرهُ اللهُ لا تَعتَبِرْهُ أنتَ نَجِسًا!“</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 41:32; يُوحَنّا 21:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:1"> وحدَثَ هذا ثلاثَ مَرَّاتٍ، ثُمَّ اَرتفَعَ الشيءُ في الحالِ إلى السَّماءِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 13:12; اعمال الرسل 10:7–19; اعمال الرسل 5:24"> وبَينَما بُطرُسُ في حَيرَةٍ يُسائِلُ نَفسَهُ ما مَعنى هذِهِ الرُّؤيا الّتي رآها، كانَ الرِّجالُ الّذينَ أرْسَلَهُم كُورنيليوسُ سألوا عَنْ بَيتِ سِمعانَ ووَقَفوا بالبابِ</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 10:5–6; اعمال الرسل 11:11"> ونادُوا مُستَخبِرينَ: ”هَلْ سِمعانُ الّذي يُقالُ لَه بُطرُسُ نازِلٌ هُنا؟“</span>
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 8:29; اعمال الرسل 11:12; اعمال الرسل 21:4"> كانَ بُطرُسُ لا يَزالُ يُفَكِّرُ في الرُّؤيا، فقَالَ لَه الرُّوحُ: ”هُنا ثلاثةُ رِجالٍ يَطلُبونَكَ،</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 15:7–9; زَكَريّا 2:9–11; إشَعْياء 48:16"> فقُمْ واَنزِلْ إلَيهِم واَذهَبْ مَعَهُم ولا تَخَفْ، لأنِّي أنا أرسَلْتُهُم“.</span>
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 1:38–39; مَرْقُس 10:51; اعمال الرسل 10:29"> فنَزَلَ بُطرُسُ وقالَ لهُم: ”أنا هوَ الّذي تَطلُبونَهُ. لِماذا جِئتُم؟“</span>
<span className="v" id="v22" data-v="22" data-xref="اعمال الرسل 11:14; اعمال الرسل 10:2; اعمال الرسل 22:12"> أجابوا: ”أرسَلَنا الضَّابِطُ كُورنيليوسُ، وهوَ رَجُلٌ بارٌّ يَخافُ اللهَ ويَشهَدُ على فَضلِهِ جميعُ اليَهودِ، لأنَّ مَلاكًا طاهِرًا أبلَغَهُ أنْ يَجيءَ بِكَ إلى بَيتِهِ ليَسمَعَ ما عِندَك مِنْ كلامٍ“.</span>
<span className="v" id="v23" data-v="23" data-xref="اعمال الرسل 10:45; اعمال الرسل 11:12; رسالَةُ بُطرُسَ الأُولَى 4:9"> فدعاهُمْ بُطرُسُ وأنزَلَهُم عِندَهُ.<br />وفي الغدِ، قامَ وذهَبَ مَعهُم يُرافِقُهُ بَعضُ الإخوةِ مِنْ يافا،</span>
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 8:40; زَكَريّا 8:20–23; زَكَريّا 3:10"> فوصَلَ إلى قَيصَريَّةَ في اليومِ التَّالي. وكانَ كُورنيليوسُ يَنتَظِرُهُم معَ الّذينَ دَعاهُم مِنْ أنسِبائِهِ وأخَصِّ أصدقائِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="رُؤيا يُوحَنّا 19:10; رُؤيا يُوحَنّا 22:8–9; دانيال 2:46"> فلمَّا دخَلَ بُطرُسُ، اَستَقْبَلَهُ كُورنيليوسُ واَرتَمى ساجِدًا لَه.</span>
<span className="v" id="v26" data-v="26" data-xref="رُؤيا يُوحَنّا 19:10; رُؤيا يُوحَنّا 22:8–9; اعمال الرسل 14:14–15"> فأنهَضَهُ بُطرُسُ وقالَ لَه: ”قُمْ، ما أنا إلَّا بَشَرٌ مِثلُكَ!“</span>
<span className="v" id="v27" data-v="27" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:12; يُوحَنّا 4:35; اعمال الرسل 10:24"> ودخَلَ وهوَ يُحادِثُهُ، فوَجَدَ جَمعًا كَبيرًا مِنَ النَّاسِ،</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 4:9; أفَسُس 3:6–7; غَلاطِيَّة 2:12–14"> فقَالَ لهُم: ”تَعرِفونَ أنَّ اليَهودِيَّ لا يَحِلُّ لَه أنْ يُخالِطَ أجنَبِيًّا، أو يَدخُلَ بَيتَهُ. لكِنَّ اللهَ أراني أنْ لا أحسُبَ أحدًا مِنَ النَّاسِ نَجِسًا أو دَنِسًا.</span>
<span className="v" id="v29" data-v="29" data-xref="المَزامِير 119:60; رسالَةُ بُطرُسَ الأُولَى 3:15; اعمال الرسل 10:19–20"> فلمَّا دَعَوتُموني جِئْتُ مِنْ غَيرِ اَعتِراضٍ. فأسألُكُم: لِماذا دَعَوتُموني؟“</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="اعمال الرسل 10:3; دانيال 9:20–21; نَحَمْيا 9:1–3"> فقَالَ كُورنيليوسُ: ”كُنتُ مِنْ أربعةِ أيّامٍ أُصَلّي في بَيتي عِندَ السّاعَةِ الثَّالِثَةِ بَعدَ الظُّهرِ، فرَأيتُ رَجُلًا علَيهِ ثِيابٌ بَرّاقةٌ يَقِفُ أمامي</span>
<span className="v" id="v31" data-v="31" data-xref="اعمال الرسل 10:4; دانيال 9:23; لُوقا 1:13"> ويَقولُ لي: ”يا كُورنيليوسُ! سَمِعَ اللهُ صَلواتِكَ وذكَرَ أعمالَكَ الخَيريَّةَ،</span>
<span className="v" id="v32" data-v="32"> فأرسِلْ إلى يافا، واَستَدْعِ سِمعانَ الّذي يُقالُ لَه بُطرُسُ، فهوَ نازِلٌ في بَيتِ سِمعانَ الدَّبّاغِ على شاطئِ البحرِ“.</span>
<span className="v" id="v33" data-v="33" data-xref="يَعقُوب 1:19; غَلاطِيَّة 4:14; أخبارِ الأيّامِ الثّانِي 30:12"> فأرسَلتُ إلَيكَ في الحالِ، وأنتَ أحسَنْتَ في مَجيئِكَ. ونَحنُ الآنَ جَميعًا في حَضرَةِ اللهِ لنَسمَعَ كُلَّ ما أمَرَكَ بِه الرَّبُّ“.</span>
</div>
</div>

## عظة بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="ﺭﻭﻣﻴﺔ 2:11; أخبارِ الأيّامِ الثّانِي 19:7; التَّثْنِيَة 10:17"> فقَالَ بُطرُسُ: ”أرى أنَّ اللهَ في الحَقيقةِ لا يُفَضِّلُ أحدًا على أحدٍ،</span>
<span className="v" id="v35" data-v="35" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:29; اعمال الرسل 10:2; ﺭﻭﻣﻴﺔ 10:12–13"> فمَنْ خافَهُ مِنْ أيَّةِ أُمَّةٍ كانَت وعَمِلَ البرَّ كانَ مَقبولًا عِندَهُ.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 28:18; اعمال الرسل 2:36; رُؤيا يُوحَنّا 17:14"> أرسَلَ كَلِمَتَهُ إلى بَني إِسرائيلَ يُعلِنُ بِشارَةَ السَّلامِ بِيَسوعَ المَسيحِ الّذي هوَ رَبُّ العالَمينَ.</span>
<span className="v" id="v37" data-v="37" data-xref="لُوقا 4:14; اعمال الرسل 2:22; اعمال الرسل 28:22"> وأنتُم تَعرِفونَ ما جرى في اليَهوديَّةِ كُلِّها، اَبتداءً مِنَ الجَليلِ بَعدَ المَعموديَّةِ الّتي دعا إلَيها يوحنَّا،</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 4:18; اعمال الرسل 2:22; إشَعْياء 61:1–3"> وكيفَ مسَحَ اللهُ يَسوعَ النّاصِريَّ بالرُّوحِ القُدُسِ والقُدرَةِ، فسارَ في كُلِّ مكانٍ يَعمَلُ الخَيرَ ويَشفي جميعَ الّذينَ اَستولى علَيهِم إبليسُ، لأنَّ اللهَ كانَ معَهُ.</span>
<span className="v" id="v39" data-v="39" data-xref="لُوقا 24:48; اعمال الرسل 1:8; اعمال الرسل 10:41"> ونَحنُ شُهودٌ على كُلِّ ما عَمِلَ مِنَ الخَيرِ في بِلادِ اليَهودِ وفي أُورُشليمَ. وهوَ الّذي صَلَبوهُ وقَتَلوهُ.</span>
<span className="v" id="v40" data-v="40" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:21; اعمال الرسل 13:30–31; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:3–4"> ولكِنَّ اللهَ أقامَهُ في اليومِ الثَّالِثِ وأعطاهُ أنْ يَظهَرَ،</span>
<span className="v" id="v41" data-v="41" data-xref="يُوحَنّا 14:22; اعمال الرسل 10:39; يُوحَنّا 21:13"> لا لِلشَعبِ كُلِّهِ، بل لِلشُّهودِ الَّذينَ اَختارَهُمُ اللهُ مِنْ قَبلُ، أي لنا نَحنُ الّذينَ أكلوا وشَرِبوا معَهُ بَعدَ قيامَتِهِ مِنْ بَينِ الأمواتِ.</span>
<span className="v" id="v42" data-v="42" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:10; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:1; رسالَةُ بُطرُسَ الأُولَى 4:5"> وأوصانا أنْ نُبَشِّرَ الشَّعبَ ونَشْهَدَ أنَّ اللهَ جَعَلَهُ ديَّانًا لِلأحياءِ والأموات.</span>
<span className="v" id="v43" data-v="43" data-xref="زَكَريّا 13:1; اعمال الرسل 2:38; إشَعْياء 53:11"> ولَه يَشهَدُ جميعُ الأنبِياءِ بِأَنَّ كُلَّ مَنْ آمنَ بِه يَنالُ باَسمِهِ غُفرانَ الخَطايا“.</span>
</div>
</div>

## حلول الروح القدس على غير اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="اعمال الرسل 11:15; اعمال الرسل 19:6; اعمال الرسل 15:8"> وبَينَما بُطرُسُ يَتكَلَّمُ، نزَلَ الرُّوحُ القُدُسُ على جميعِ الّذينَ يَسمَعونَ كلامَهُ.</span>

<span className="v" id="v46" data-v="46" data-xref="اعمال الرسل 19:6; اعمال الرسل 2:4; مَرْقُس 16:17"> لأنَّهُم سَمِعوهُم يتكَلَّمونَ بِلُغاتٍ غَيرِ لُغَتِهِم ويُعَظِّمونَ اللهَ. فقَالَ بُطرُسُ:</span>
