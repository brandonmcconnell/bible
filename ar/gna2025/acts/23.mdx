<span className="v" id="v1" data-v="1" data-xref="اعمال الرسل 24:16; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:12"> فنظَرَ بولُسُ إلى المجلِسِ وقالَ: ”أيُّها الإخوةُ، عِشتُ للهِ بِضميرٍ صالِحٍ إلى هذا اليومِ“.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 18:22; اعمال الرسل 24:1; ميخا 5:1"> فأمَرَ حنانيَّا، رَئيسُ الكَهنَةِ، أعوانَهُ بِأنْ يَضرِبوا بولُسَ على فَمِهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 25:1–2; يُوحَنّا 7:51; ﻣﺘﻰ 23:27–28"> فقَالَ لَه بولُسُ: ”ضَرَبكَ اللهُ، أيُّها الحائِطُ المُبَيَّضُ! أتَجلِسُ لتُحاكِمَني بِحَسبِ الشَّريعةِ، وتأمُرُ بِضَربي فتُخالِفَ الشَّريعةَ؟“</span>
<span className="v" id="v4" data-v="4"> فقَالَ الحاضِرونَ لِبولُسَ: ”أنتَ تَشتُمُ رَئيسَ كَهنَةِ اللهِ!“</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 22:28; الجامِعَة 10:20; يَهُوذا 1:8–9"> فقَالَ بولُسُ: ”ما كُنتُ أعرِفُ، أيُّها الإخوةُ، أنَّهُ رَئيسُ الكَهنَةِ. فالكُتُبُ المُقَدَّسَةُ تَقولُ: ”لا تَلعَنْ رَئيسَ شَعبِكَ“.</span>
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 24:15; اعمال الرسل 24:21; اعمال الرسل 26:5–8"> وكانَ بولُسُ يَعرِفُ أنّ بَعضَهُم صدّوقيُّونَ والبَعضَ الآخَرَ فَرِّيسيّونَ، فَصاحَ في المَجلِسِ: ”أيُّها الإخوةُ، أنا فَرِّيسيٌّ اَبنُ فَرِّيسيٍّ. وأنا أحاكَمُ الآنَ لأنِّي أرجو قيامَةَ الأمواتِ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 55:9; يُوحَنّا 7:40–43; ﻣﺘﻰ 10:34"> فلمَّا قالَ هذا، وقَعَ الخِلافُ بَينَ الفَرِّيسيِّينَ والصَدُّوقيِّينَ واَنقَسَمَ المَجلِسُ،</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 20:27; ﻣﺘﻰ 22:23; مَرْقُس 12:18"> لأنَّ الصَدُّوقيِّينَ لا يُؤمِنونَ بِقيامَةِ الأمواتِ وبوجودِ الملائِكةِ والأرواحِ، وأمَّا الفَرِّيسيُّونَ فيُؤمِنونَ بِهذا كُلِّهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 22:7; اعمال الرسل 22:17–18; يُوحَنّا 12:29"> فاَرتَفَعَ الصّياحُ، وقامَ بَعضُ مُعَلِّمي الشَّريعَةِ مِنَ الفَرِّيسيِّينَ وأخذوا يَحتجُّونَ بِشدَّةٍ قالوا: ”لا نَجِدُ جُرْمًا على هذا الرَّجُلِ. رُبَّما كَلَّمَهُ رُوحٌ أو ملاكٌ“.</span>
<span className="v" id="v10" data-v="10" data-xref="يَعقُوب 1:19; يَعقُوب 3:14–2; اعمال الرسل 22:24"> واَشتَدَّ الخِلافُ حتّى خافَ القائِدُ أنْ يُمَزِّقوا بولُسَ تَمزيقًا، فأمَرَ جُنودَهُ بِأنْ يَنزِلوا إلَيهِ ويَنتَزِعوهُ مِنْ بَينِهِم، ويأخذوهُ إلى القَلْعَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 18:9; اعمال الرسل 19:21; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:17"> وفي اللَّيلةِ الثَّانِيَةِ ظهَرَ الرَّبُّ لِبولُسَ وقالَ لَه: ”تَشَجَّعْ! فمِثلَما شَهِدتَ لي في أُورُشليمَ، هكذا يَجبُ أنْ تَشهَدَ لي في رومَةَ“.</span>
</div>

## مؤامرة لاغتيال بولس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 23:30; إشَعْياء 8:9–10; اعمال الرسل 23:21"> ولمَّا طلَعَ الصَّباحُ تآمَرَ اليَهودُ، فحَلَفوا لا يأكلونَ ولا يَشرَبونَ حتّى يَقتُلوا بولُسَ.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 16:2; صموئيلَ الثّانِي 15:31; صموئيلَ الثّانِي 15:12"> وكانَ المُتآمرونَ أكثَرَ مِنْ أربعينَ.</span>
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 23:12; إرْمِيا 8:12; إشَعْياء 3:9"> فجاؤُوا إلى رُؤساءِ الكَهنَةِ والشُّيوخِ وقالوا: ”حَلَفْنا لا نَذوقُ شيئًا حتّى نَقتُلَ بولُسَ.</span>
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 59:7; المَزامِير 37:32–33; الأمثال 1:16"> فاَطلُبوا أنتُم وأعضاءُ المَجلِسِ مِنْ قائِدِ الحامِيَةِ أنْ يُرسِلَ بولُسَ إلَيكُم، بِحُجَّةِ أنَّكُم تُريدونَ تَدقيقَ النَّظَرِ في أمرِهِ، ونكونُ نَحنُ على اَستِعدادٍ لقَتلِهِ وهوَ في طريقِهِ إلَيكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 23:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:19; اعمال الرسل 23:32"> وسَمِعَ اَبنُ أُختِ بولُسَ بالكَمينِ، فذهَبَ إلى القَلعَةِ ودخَلَ على بولُسَ وأخبَرَهُ بالأمرِ.</span>
<span className="v" id="v17" data-v="17" data-xref="الأمثال 22:3; ﻣﺘﻰ 8:8–9; ﻣﺘﻰ 10:16"> فدَعا بولُسُ واحدًا مِن الضُّبّاطِ وقالَ لَه: ”خُذْ هذا الشَّابَّ إلى القائِدِ، لأنَّ عِندَهُ ما يَقولُهُ لَه“.</span>
<span className="v" id="v18" data-v="18" data-xref="أفَسُس 3:1; أفَسُس 4:1; اعمال الرسل 28:17"> فأخذَهُ الضَّابِطُ إلى القائِدِ وقالَ: ”دَعاني السَّجينُ بولُسُ وطلَبَ أنْ أَجيءَ بِهذا الشَّابِّ إلَيكَ، لأنَّ عِندَهُ ما يَقولُهُ لكَ“.</span>
<span className="v" id="v19" data-v="19" data-xref="إرْمِيا 31:32; أسْتِير 9:12; أسْتِير 7:2"> فأمسَكَ القائِدُ بِيدِهِ واَنفرَدَ بِه وسألَهُ: ”ما عِندَكَ؟ أخْبِرني“.</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 23:14–15; اعمال الرسل 23:1; المَزامِير 12:2"> فقَالَ: ”إِتَّفَقَ اليَهودُ على أنْ يَطلبوا إلَيكَ أنْ تُنزِلَ بولُسَ غَدًا إلى المَجلِسِ بِحُجَّةِ أنَّهُم يُريدونَ تَدقيقَ النَّظَرِ في أمرِهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 23:12–14; اعمال الرسل 14:5–6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:26"> فلا تُصَدِّقْهُم، لأنَّ أكثرَ مِنْ أربعينَ رَجُلًا مِنهُم أقاموا لَه كمينًا، وحَلَفوا لا يأكلونَ ولا يَشرَبونَ حتّى يَقتُلوهُ. وهُمُ الآنَ على أتَمِّ الاَستعدادِ، يَنتَظِرونَ أنْ تَستَجيبَ لهُم“.</span>
<span className="v" id="v22" data-v="22" data-xref="مَرْقُس 1:44; يَشُوع 2:14"> فقَالَ القائِدُ لِلشَّابِّ: ”لا تَقُلْ لأحدٍ أنَّكَ أخبَرتَني بِهذا الأمرِ“. وصرَفَهُ مِنْ عِندِهِ.</span>
</div>
</div>

## بولس عند الحاكم فيلكس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 14:25; اعمال الرسل 23:33; لُوقا 12:38"> ثُمَّ دَعا القائِدُ اَثنَينِ مِنْ ضُبَّاطِهِ وقالَ لَهُما: ”أعِدَّا مِئتَي جُنديٍّ للذَهابِ في السَّاعةِ التَّاسِعةِ مِنَ اللَّيلِ إلى قيصرِيَّةَ، ومعَهُم سَبعونَ فارِسًا ومِئتا مُسلَّحٍ بالرِّماحِ.</span>
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 25:14; اعمال الرسل 23:26; اعمال الرسل 24:10"> وأعِدُّوا لبولُسَ خَيلًا تَحمِلُهُ سالِمًا إلى الحاكِمِ فيلِكْسَ“.</span>
<span className="v" id="v25" data-v="25"> وكتَبَ إلى الحاكِمِ بِرسالةٍ يَقول فيها:</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 15:23; اعمال الرسل 26:25; لُوقا 1:3"> ”مِنْ كلوديوسَ ليسياسَ إلى صاحبِ العِزَّةِ فيلِكْسَ، سلامٌ:</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 22:25–29; اعمال الرسل 21:31–33; اعمال الرسل 24:6"> أمسَكَ اليَهودُ هذا الرَّجُلَ وكادوا يَقتُلونَهُ. فلمَّا عرَفْتُ أنَّهُ رومانيٌّ، أسرَعْتُ إلَيهِ معَ جُنودي فأنقَذْتُهُ.</span>
<span className="v" id="v28" data-v="28" data-xref="اعمال الرسل 22:30"> وأردتُ أنْ أعرِفَ بِما يتَّهِمونَهُ، فجِئتُ بِه إلى مجلِسِهِم،</span>
<span className="v" id="v29" data-v="29" data-xref="اعمال الرسل 26:31; اعمال الرسل 25:25; اعمال الرسل 18:15"> فتَبيَّنَ لي أنَّهُم يتَّهِمونَهُ بِمسائِلَ في شَريعتِهِم، وأنْ لا شيءَ مِمَّا يتَّهِمونَهُ بِه يَستَوجِبُ الموتَ أوِ السِّجنَ.</span>
<span className="v" id="v30" data-v="30" data-xref="اعمال الرسل 23:35; اعمال الرسل 25:16; اعمال الرسل 25:5–6"> ثُمَّ سَمِعتُ أنَّهُم وضَعوا خُطَّةً لاَغتِيالِهِ، فأرسلْتُهُ إلَيكَ في الحالِ، وأبلَغتُ خُصومَهُ أنْ يَرفَعوا إلَيكَ دعواهُم علَيهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:3–4; لُوقا 7:8; اعمال الرسل 23:23–24"> وقامَ الجُنودُ بِتَنفيذِ أوامِرِ القائِدِ، فأخذوا بولُسَ ليلًا إلى أنْتيباتْريسَ.</span>
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 23:23; اعمال الرسل 23:10"> وفي الغَدِ رَجَعوا إلى القَلعَةِ وتَرَكوا الفُرسانَ يُواصِلونَ السَّفَرَ معَهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="اعمال الرسل 23:26; اعمال الرسل 23:23–24; اعمال الرسل 28:16"> فلمَّا وصَلَ الفُرسانُ إلى قيصرِيَّةَ، أدخَلوا بولُسَ على الحاكِمِ وسَلَّموا إلَيهِ الرِّسالَةَ.</span>
<span className="v" id="v34" data-v="34" data-xref="اعمال الرسل 21:39; اعمال الرسل 25:1; دانيال 2:49"> فقرأها وسألَ بولُسَ عَنِ الوِلايةِ الّتي يَنتَمي إلَيها. فلمَّا عرَفَ أنَّهُ مِنْ كِيليكِيَّةَ،</span>
<span className="v" id="v35" data-v="35" data-xref="اعمال الرسل 23:30; اعمال الرسل 25:16; ﻣﺘﻰ 27:27"> قالَ لَه: ”سأسمَعُ مِنكَ متى حضَرَ خُصومُكَ“. وأمَرَ جُنودَهُ بِأنْ يَحرُسوا بولُسَ في قَصْرِ هيرودُسَ.</span>
