## في أيقونية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اعمال الرسل 18:4; اعمال الرسل 13:51; اعمال الرسل 13:46"> ودَخَلَ بولُسُ وبَرنابا كعادَتِهِما مَجمعَ اليَهودِ في أيقونيةَ وتكَلَّما كَلامًا جعَلَ كثيرًا مِنَ اليَهودِ واليونانيِّينَ يُؤمِنونَ.</span>
<span className="v" id="v2" data-v="2" data-xref="اعمال الرسل 13:50; اعمال الرسل 21:27–30; اعمال الرسل 14:19"> لكِنَّ اليَهودَ الّذينَ رَفَضوا أنْ يُؤمِنوا حَرَّضوا غيرَ اليَهودِ على الإخوَةِ وأفسَدوا قُلوبَهُم.</span>
<span className="v" id="v3" data-v="3" data-xref="العِبرانِيّين 2:4; أفَسُس 6:18–20; اعمال الرسل 20:32"> ولكِنَّ بولُسَ وبَرنابا أقاما هُناكَ مُدَّةً طويلةً يُجاهِرانِ بالرَّبِّ. وكانَ الرَّبُّ يشهَدُ لِكلامِهما على نِعمَتِه بما أجرى على أيدِيهما من العجائِبِ والآياتِ.</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 28:24; اعمال الرسل 14:14; لُوقا 11:21–23"> واَنقَسَم أهلُ المدينةِ، فكانَ بعضُهُم معَ اليهودِ وبعضُهُم معَ الرَّسولَين.</span>
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 14:19; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:11; اعمال الرسل 17:5"> ولمَّا عزَمَ اليهودُ وغيرُ اليهودِ، ومعَهُم رُؤساؤُهُم، أن يُهينوا الرَّسولَينِ ويَرجُموهُما،</span>
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 14:20–21; ﻣﺘﻰ 10:23; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:11"> شعرا بالأمرِ فهرَبا إلى لِسترَةَ ودَرْبَة والمدنِ المُجاورةِ لهُما في ولايةِ ليقونِيةَ،</span>
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 14:21; اعمال الرسل 16:10; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:2"> وأخذا يُبشِّرانِ هُناكَ.</span>
</div>
</div>

## في لسترة ودربة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 3:2; يُوحَنّا 9:1–2; يُوحَنّا 5:3–5"> وكانَ في لِسْترَةَ رَجُلٌ عاجِزٌ كسيحٌ مُنذُ مَولِدِهِ، ما مَشى في حياتِهِ مرَّةً.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 3:4; ﻣﺘﻰ 15:28; ﻣﺘﻰ 9:28–29"> وبَينَما هوَ يُصغي إلى كلامِ بولُسُ، نظَرَ إلَيهِ بولُسَ فرَأى فيهِ مِنَ الإيمانِ ما يَدعو إلى الشِّفاءِ،</span>
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 35:6; يُوحَنّا 5:8–9; يُوحَنّا 14:12"> فقَالَ لَه بأعلى صوتِهِ: ”قُمْ وقِفْ مُنتصِبًا على رِجْلَيكَ!“ فنهَضَ يَمشي.</span>
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 28:6; اعمال الرسل 8:10; اعمال الرسل 12:22"> فلمَّا رَأى الجُموعُ ما عَمِلَ بولُسُ، صاحوا بلُغتِهِم اللِّيقونيَّةِ: ”تَشبَّهَ الآلهَةُ بالبَشرِ ونَزَلوا إلَينا!“</span>
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 19:35"> وسَمُّوا بَرنابا زَيُوسَ، وبولُسَ هَرْمَسَ لأنَّهُ كانَ يَتَولَّى الكلامَ.</span>
<span className="v" id="v13" data-v="13" data-xref="دانيال 2:46; اعمال الرسل 10:25"> وجاءَ كاهِنُ زَيُوسَ الّذي كانَ مَعبَدُهُ عِندَ مَدخَلِ المدينةِ بِثِيرانٍ وأكاليلَ مِنْ زَهرٍ إلى الأبوابِ، يُريدُ أنْ يُقَدِّمَ ذَبيحةً معَ الجُموعِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 26:65; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:5–6; المُلُوكِ الثّانِي 5:7"> فلمَّا سَمِعَ بَرنابا وبولُسُ، مزَّقا ثيابَهُما وأسرَعا إلى الجُموعِ يَصيحانِ بِهِم:</span>
<span className="v" id="v15" data-v="15" data-xref="اعمال الرسل 10:26; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:4; يَعقُوب 5:17"> ”ماذا تَفعَلونَ، أيُّها النَّاسُ؟ نَحنُ بشَرٌ ضُعفاءُ مِثلُكُم، جِئنا نُبَشِّرُكُم لتَتْرُكوا هذِهِ الأباطيلَ وتَهتَدوا إلى اللهِ الحيِّ الّذي خلَقَ السَّماءَ والأرضَ والبحرَ وكُلَّ شيءٍ فيها.</span>
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 17:30; المَزامِير 81:12; ﺭﻭﻣﻴﺔ 1:28"> ترَكَ جميعَ الأُممِ في العُصورِ الماضيةِ تَسلُكُ طريقَها،</span>
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 1:19–20; يُوئيل 2:23; أيُّوب 5:10"> ولكنَّهُ كانَ يَشهَدُ لِنَفسِهِ بِما يَعمَلُ مِنَ الخيرِ: أنزَلَ المطَرَ مِنَ السَّماءِ وأعطى المواسِمَ في حِينِها، ورَزَقكُمُ القُوتَ ومَلأَ قُلوبَكُم بالسُّرورِ“.</span>
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 44:16–17; التَّكوين 19:9; التَّكوين 11:6"> فما تَمكَّنَ الرَّسولانِ بهذا الكلامِ أنْ يَمنَعا الجُموعَ مِنْ تَقديمِ ذَبيحةٍ لهُما إلَّا بَعدَ جَهدٍ كثيرٍ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:25; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:11; اعمال الرسل 13:50–51"> وجاءَ بَعضُ اليَهودِ مِنْ أنطاكيةَ وأيقونيَةَ واَستَمالوا الجُموعَ، فرَجَموا بولُسَ وجَرُّوهُ إلى خارِجِ المدينةِ وهُم يَحسبونَ أنَّهُ ماتَ.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:9–10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:9; اعمال الرسل 14:6"> فلمَّا أحاطَ بِه التَّلاميذُ، قامَ ودخَلَ المدينةَ. وفي الغَدِ خرَجَ معَ بَرنابا إلى دَرْبَةَ.</span>
</div>
</div>

## العودة إلى أنطاكية في سورية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 14:19; اعمال الرسل 14:1; اعمال الرسل 13:51"> وبَشَّرَ بولُسُ وبَرنابا في دَرْبَةَ وكسَبا كثيرًا مِنَ التَّلاميذِ. ثُمَّ رَجَعا إلى لِسترَةَ، ومِنها إلى أيقونِيَةَ وأنطاكيةَ،</span>
<span className="v" id="v22" data-v="22" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:10; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:12; يُوحَنّا 16:33"> يُشدِّدانِ عَزائمَ التَّلاميذِ ويُشَجِّعانِهِم على الثَّباتِ في إيمانِهم، ويَقولانِ لهُم: ”لا بُدَّ مِنْ أنْ نَجتازَ كثيرًا مِنَ المَصاعبِ لنَدخُلَ مَلكوتَ اللهِ“.</span>
<span className="v" id="v23" data-v="23" data-xref="تِيطُس 1:5; اعمال الرسل 11:30; يَعقُوب 5:14"> وكانا يُعَيّنانِ لهُم قُسوسًا في كُلِّ كنيسةٍ، ثُمَّ يُصلِّيانِ ويَصومانِ ويَستَودِعانِهِمِ الرَّبَّ الّذي آمَنوا بِه.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 13:13–14; اعمال الرسل 15:38"> واَجتازَ الرَّسولانِ مُقاطَعةَ بِسيدِيَّةَ حتّى وصَلا إلى بَمْفيليَّةَ.</span>
<span className="v" id="v25" data-v="25"> وبَشَّرا بِكلامِ اللهِ في بَرجَةَ. ثُمَّ نَزَلا إلى أتَّالِيَةَ،</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 15:40; اعمال الرسل 11:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:12"> ومِنها سافَرا في البحرِ إلى أنطاكيةَ الّتي خَرَجا مِنها، تَرعاهُما نِعمةُ اللهِ مِنْ أجلِ العَمَلِ الّذي قاما بِه.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:9; اعمال الرسل 15:12; كُولُوسِي 4:3"> فلمَّا وصَلا إلى أنطاكيةَ جَمَعا الكنيسةَ، وأخبَرا بِكُلِّ ما أجْرى اللهُ على أيديهِما، وكيفَ فتَحَ بابَ الإيمانِ لِغيرِ اليَهودِ.</span>
<span className="v" id="v28" data-v="28" data-xref="اعمال الرسل 11:26; اعمال الرسل 15:35"> وأقاما هُناكَ مُدَّةً معَ التَّلاميذِ.</span>
</div>
</div>
