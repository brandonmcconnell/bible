## في مكدونية واليونان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:5; اعمال الرسل 19:21; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:3"> ولمَّا سكَنَ الهِياجُ في أفسُسَ دَعا بولُسُ التَّلاميذَ، فودَّعَهُم بِكَلِمَةِ تَشجيعٍ وسافَرَ إلى مكدونِيَّةَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:3; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:1; كُولُوسِي 1:28"> وسارَ في تِلكَ الأنحاءِ يُشَجِّعُ بِكلامِهِ الكثيرِ جماعَةَ المُؤمنينَ. ثُمَّ جاءَ إلى اليونانِ.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 20:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:26; اعمال الرسل 9:23–24"> فأقامَ فيها ثلاثةَ أشهُرٍ. وبَينَما هوَ يَستَعِدُّ للسَّفَرِ في البحرِ إلى سوريَّةَ، تآمَرَ اليَهودُ لِقَتلِهِ، فرأى أنْ يَرجِعَ بِطريقِ مكدونِيَّةَ.</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 19:29; أفَسُس 6:21; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:20"> فرافَقَهُ سوباتَرُسُ بنُ بِرُّسَ مِنْ أهلِ بيريَّةَ وأرَسْترْخسُ وسكُوندُسُ مِنْ أهلِ تسالونيكي، وغايوسُ مِنْ أهلِ دَربَةَ، وتيموثاوسُ وتيخيكُسُ وتُروفيمُسُ.</span>
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 16:8; اعمال الرسل 16:10–11; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:13"> فسَبَقونا إلى ترواسَ واَنتَظَرونا هُناكَ.</span>
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 12:3; اعمال الرسل 16:12; الخُرُوج 23:15"> أمَّا نَحنُ فسافَرنا في البحرِ مِنْ فيلبّي بَعدَ عيدِ الفَطيرِ، ولَحِقْنا بِهِم بَعدَ خَمسةِ أيّامٍ إلى تَرواسَ، فقَضَينا فيها سَبعَةَ أيّامٍ.</span>
</div>
</div>

## زيارة بولس الأخيرة لترواس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:2; اعمال الرسل 2:42; يُوحَنّا 20:19"> وفي يومِ الأحدِ اَجتَمَعْنا لِكَسرِ الخُبزِ، فأخَذَ بولُسُ يَعِظُ الحاضِرينَ. فأطالَ الكلامَ إلى مُنتَصَفِ اللَّيلِ، لأنَّهُ كانَ يُريدُ السَّفَرَ في الغدِ.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 1:13; لُوقا 22:12"> وكانَ في الغُرفةِ العُليا الّتي اَجتَمَعْنا فيها مَصابيحُ كثيرةٌ.</span>
<span className="v" id="v9" data-v="9" data-xref="مَرْقُس 13:36; ﻣﺘﻰ 26:40–41; ﻳﻮﻧﺎﻥ 1:5–6"> وهُناكَ فتى اَسمُهُ أفتيخوسُ جالِسًا عِندَ النَّافِذَةِ. فأخَذَهُ النُّعاسُ، وبولُسُ يَستَرسِلُ في الكلامِ، حتّى غَلَبَ علَيهِ النّومُ، فوقَعَ مِنَ الطَبَقَةِ الثَّالِثَةِ إلى أسفلُ وحُمِلَ مَيْتًا.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="مَرْقُس 5:39; ﻣﺘﻰ 9:23–24; المُلُوكِ الأوَّلُ 17:21–22"> فنَزَلَ بولُسُ واَرتَمى علَيهِ وحَضَنَهُ، وقالَ: ”لا تَقلَقوا، فهوَ حيٌّ“.</span>
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 20:7; اعمال الرسل 20:9"> وصَعِدَ إلى الغُرفَةِ العُليا وكسَرَ الخُبزَ وأكَلَ. وحَدَّثَهُم طويلًا إلى الفَجرِ ومَضى.</span>
<span className="v" id="v12" data-v="12" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:16; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:4"> فجاؤُوا بِالفَتى حَيًّا، فكانَ لهُم عزاءٌ كبيرٌ.</span>
</div>
</div>

## من ترواس إلى ميليتس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 6:46; مَرْقُس 1:35; مَرْقُس 6:31–33"> أمَّا نحنُ فتوجَّهْنا إلى السَّفينةِ، فأقلَعْنا إلى أسُّوسَ لِنأخُذَ بولُسَ مَعنا مِنْ هُناكَ كما طَلَبَ مِنَّا، لأنَّهُ أرادَ السَّفَرَ إلَيها في البَرِّ.</span>
<span className="v" id="v14" data-v="14"> فلمَّا لَحِقَ بِنا إلى أسُّوسَ، أصعَدناهُ إلى السَّفينةِ وجِئنا إلى ميتيلينَةَ.</span>
<span className="v" id="v15" data-v="15" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:20; اعمال الرسل 20:17"> ثُمَّ أبحَرْنا مِنها في اليومِ الثَّاني، فأشرَفْنا على خِيوسَ. وسِرْنا في اليومِ الثَّالثِ بمُحاذاةِ ساموسَ، ثُمَّ وصَلْنا في اليومِ الرَّابعِ إلى ميليتُسَ</span>
<span className="v" id="v16" data-v="16" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:8; اعمال الرسل 2:1; اعمال الرسل 19:21"> لأنَّ بولُسَ رأى أن يَتَجاوَزَ أفسُسَ في البحرِ لِئَلَّا يَتأخَّرَ في آسيَّةَ وهوَ يُريدُ السُّرعَةَ لَعلَّه يَصِلُ إلى أُورُشليمَ في يومِ الخمسينَ.</span>
</div>
</div>

## بولس يودع شيوخ أفسس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="تِيطُس 1:5; اعمال الرسل 11:30; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:17"> وأرسَلَ بولُسُ مِنْ ميليتُسَ يَستَدعي شُيوخَ كنيسةِ أفسُسَ.</span>
<span className="v" id="v18" data-v="18" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 1:5–6; اعمال الرسل 18:19; اعمال الرسل 19:1"> فلمَّا جاؤُوا إلَيهِ، قالَ لهُم: ”تَعرِفونَ كيفَ عِشْتُ مَعكُم طَوالَ المُدَّةِ الّتي أقَمتُ فيها بَينَكُم، مِنْ أوَّلِ يومٍ جِئتُ فيهِ إلى آسيَّةَ.</span>
<span className="v" id="v19" data-v="19" data-xref="كُولُوسِي 3:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:4; ﺭﻭﻣﻴﺔ 12:11"> فكُنْتُ أخدُمُ الرَّبَّ بِكُلِّ تواضُعٍ وبكثيرٍ مِنَ الدُّموعِ، أُقاسي المِحَنَ الّتي أنزَلَتْها بي مكايِدُ اليَهودِ.</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 20:27; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:2; اعمال الرسل 5:42"> وما قَصَّرتُ في شيءٍ يُفيدُكُم، بل كُنتُ أعِظُكُم بِه، وأعلِّمُكُم في الأماكِنِ العامةِ وفي البُيوتِ.</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 2:38; اعمال الرسل 11:18; رسالَةُ يُوحَنّا الأُولَى 5:11–13"> وناشَدتُ اليَهودَ واليونانيِّينَ أنْ يتوبوا إلى اللهِ ويُؤمنوا بِرَبِّنا يَسوعَ.</span>
<span className="v" id="v22" data-v="22" data-xref="يَعقُوب 4:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:14; رسالَةُ بُطرُسَ الثّانيَة 1:14"> وأنا اليومَ ذاهِبٌ إلى أُورُشليمَ بِدافِعٍ مِنَ الرُّوحِ القُدُسِ، لا أعرِفُ ما يُصادِفُني هُناكَ.</span>
<span className="v" id="v23" data-v="23" data-xref="اعمال الرسل 9:16; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 3:3; اعمال الرسل 21:4"> غيرَ أنَّ الرُّوحَ القُدُسَ كانَ يُحذِّرُني في كُلِّ مدينةٍ أنَّ القُيودَ والمَشقّاتِ تَنتظِرُني.</span>
<span className="v" id="v24" data-v="24" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:6–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:10"> ولكنِّي لا أحسَبُ أنَّ حياتي لها أيَّةُ قيمَةٍ عِندي، ما دُمتُ أقومُ بِمهمَّتي وأُتمِّمُ العمَلَ الّذي تَسلَّمتُهُ مِنَ الرَّبِّ يَسوعَ، فأُنادي بِبشارَةِ نِعمَةِ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="اعمال الرسل 28:31; اعمال الرسل 20:38; ﻣﺘﻰ 4:23"> وأنا أعرِفُ أنَّكُم لن تَرَوا وَجهي بَعدَ اليومِ، أنتُمُ الّذينَ سِرتُ بَينَهُم كُلِّهِم أُبشِّرُ بِمَلكوتِ اللهِ.</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 18:6; حِزْقِيال 3:18–21; حِزْقِيال 33:2–9"> لذلِكَ أَشهَدُ اليومَ لَدَيكُم أنِّي بَريءٌ مِنْ دَمِكُم جميعًا،</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 20:20; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:4; غَلاطِيَّة 1:7–10"> لأنِّي ما قَصَّرتُ في إبلاغِكم مَشيئةَ اللهِ كُلَّها.</span>
<span className="v" id="v28" data-v="28" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:2–3; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 4:16; إشَعْياء 40:11"> فاَسهَروا على أنفُسِكُم وعلى الرَّعِيَّةِ الّتي أقامَكُمُ الرُّوحُ القُدُسُ فيها أساقِفَةً لِترعَوا كنيسةَ اللهِ الّتي اَكتسَبَها بدَمِهِ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 7:15; رسالَةُ بُطرُسَ الثّانيَة 2:1; يُوحَنّا 10:12"> وأنا أعرِفُ أنَّ الذِّئابَ الخاطفَةَ سَتدخُلُ بينَكُم بَعدَ رحيلي ولا تُشفِقُ على الرَّعِيَّةِ،</span>
<span className="v" id="v30" data-v="30" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:19–20; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:3–4; الأمثال 23:33"> ويقومُ مِنْ بَينِكُم أنتُم أُناسٌ يَنطِقونَ بالأكاذيبِ ليُضَلِّلوا التَّلاميذَ فيَتْبعُوهُم.</span>
<span className="v" id="v31" data-v="31" data-xref="اعمال الرسل 19:8; اعمال الرسل 20:19; اعمال الرسل 19:10"> فتَنبَّهوا وتَذكَّروا أنِّي بِدُموعي نَصَحتُ كُلَّ واحدٍ مِنكُم، ليلًا ونهارًا، مُدَّةَ ثلاثِ سَنواتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="كُولُوسِي 1:12; أفَسُس 1:14; كُولُوسِي 3:24"> والآنَ أستودِعُكُمُ اللهَ وكَلِمَةَ نِعمَتِهِ، فهوَ القادِرُ على أنْ يُقوِّيَكُم ويَمنَحكُمُ الميراثَ معَ جميعِ القِدّيسينَ.</span>
<span className="v" id="v33" data-v="33" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:18"> ما اَشتهيَتُ يومًا فِضَّةَ أحدٍ أو ذَهبَهُ أو ثيابَهُ،</span>
<span className="v" id="v34" data-v="34" data-xref="اعمال الرسل 18:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:12; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:8–9"> وأنتُم تَعرِفونَ أنِّي بِهاتَينِ اليدَينِ اَشتَغَلْتُ وحَصَلْتُ على ما نَحتاجُ إلَيهِ أنا ورِفاقي.</span>
<span className="v" id="v35" data-v="35" data-xref="العِبرانِيّين 13:16; الأمثال 19:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:6–12"> وأرَيتُكُم في كُلِّ شيءٍ كيفَ يَجبُ علَينا بالكَدِّ والعَمَلِ أنْ نُساعِدَ الضُّعفاءَ، مُتذَكِّرينَ كلامَ الرَّبِّ يَسوعَ: ”تَبارَكَ العَطاءُ أكثرَ مِنَ الأخذِ“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="اعمال الرسل 21:5; لُوقا 22:41; فِيلِبِّي 4:6"> ولمَّا ختَمَ بولُسُ كلامَهُ، سجَدَ مَعهُم كُلِّهِم وصَلّى.</span>
<span className="v" id="v37" data-v="37" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:20; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:26"> وبكَوا كثيرًا وعانَقُوا بولُسَ وقَبَّلوهُ.</span>
<span className="v" id="v38" data-v="38" data-xref="اعمال الرسل 15:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:11; اعمال الرسل 20:25"> وكانَ أكثَرَ ما أحزَنَهُم قَولُهُ لهُم: ”لن تَرَوا وَجهي بَعدَ اليومِ“. ثُمَّ شَيَّعوهُ إلى السَّفينةِ.</span>
