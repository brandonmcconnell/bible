## اختيار برنابا وبولس للتبشير

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اعمال الرسل 4:36; اعمال الرسل 15:35; اعمال الرسل 11:22–27"> وكانَ في كنيسةِ أنطاكيةَ أنبياءُ ومُعَلِّمونَ هُم: بَرنابا وشَمعونُ الّذي يُدعى نِيجَرَ، ولوقُيوسُ القيرينيُّ، ومَنايِنُ وهوَ صَديقُ الوالي هيرودُسَ مِنَ الطُّفولَةِ، وشاوُلُ.</span>
<span className="v" id="v2" data-v="2" data-xref="غَلاطِيَّة 1:15; اعمال الرسل 9:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:11"> وبَينَما هُم يَخدِمونَ الرَّبَّ ويَصومونَ قالَ لهُمُ الرُّوحُ القُدُسُ: ”خَصِّصوا لي بَرنابا وشاوُلَ لِعَمَلٍ دَعوتُهُما إلَيهِ“.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 6:6; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:2; اعمال الرسل 13:2"> فصاموا وصَلُّوا، ثُمَّ وضَعوا أيديَهُم علَيهِما وصَرَفوهُما.</span>
</div>
</div>

## في قبرص

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 4:36; اعمال الرسل 13:2; اعمال الرسل 11:19"> فأرسَلَهُما الرُّوحُ القُدُسُ، فنَزَلا إلى سَلوكيةَ ومِنها سافَرا في البحرِ إلى قُبرُصَ.</span>
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 13:14; اعمال الرسل 13:46; اعمال الرسل 14:1"> فلمَّا وصَلا إلى سَلاميسَ بَشّرا بِكلامِ اللهِ في مَجامِعِ اليَهودِ، وكانَ يُعاوِنهُما يوحنَّا.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 7:15; رسالَةُ يُوحَنّا الأُولَى 4:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:13"> فاَجتازا الجَزيرَةَ كُلَّها حتّى مدينةِ بافوسَ، فوَجَدا فيها ساحِرًا يَهوديًّا يَدَّعي النُّبوَّةَ كذبًا اَسمُهُ بَريَشوعُ،</span>
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 19:38; اعمال الرسل 13:12; اعمال الرسل 18:12"> مِنْ حاشيةِ سَرجِيوسَ بولُسَ حاكِمِ الجزيرَةِ. وكانَ هذا رَجُلًا عاقِلًا فدَعا بَرنابا وشاوُلَ وطلَبَ إلَيهِما أنْ يَسمَعَ كلامَ اللهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:8; اعمال الرسل 13:6–7; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:14–15"> ولكِنَّ عليمًا السّاحِرَ، وهذا مَعنى اَسمِهِ، عارَضَهُما وحاوَلَ أنْ يُبعِدَ الحاكِمَ عَن الإيمانِ.</span>
<span className="v" id="v9" data-v="9" data-xref="ميخا 3:8; اعمال الرسل 4:8; اعمال الرسل 2:4"> فاَمتَلأ شاوُلُ، واَسمُهُ أيضًا بولُسُ، مِنَ الرُّوحِ القُدُسِ، فنَظَرَ إلى السّاحِرِ</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 8:44; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3; ﻣﺘﻰ 13:38"> وقالَ لَه: ”يا اَبنَ إبليسَ، يا عدوَّ كُلِّ بِرٍّ، أيُّها المُمتَلِئُ بِكُلِّ خُبثٍ وغِشٍّ! أما تكِفُّ عَنْ تَعويجِ سُبُلِ الرَّبِّ القَويمَةِ؟</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 9:3; المَزامِير 32:4; العِبرانِيّين 10:31"> ستَضرِبُكَ يدُ الرَّبِّ، فتكونُ أعمى لا تُبصِرُ نُورَ الشَّمسِ إلى حينٍ“. فسَقَطَت على عَينَيهِ في الحالِ غَشاوَةٌ سَوداءُ، فأخَذَ يَدورُ حَولَ المكانِ مُلتَمِسًا مَنْ يَقودُهُ بِيَدِهِ.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:4–5; اعمال الرسل 13:7; لُوقا 4:22"> فآمنَ الحاكِمُ عِندَما شاهَدَ ما جَرى، وأدهَشَهُ تَعليمُ الرَّبِّ.</span>
</div>
</div>

## في أنطاكية بسيدية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="اعمال الرسل 15:38; اعمال الرسل 2:10; اعمال الرسل 14:24–25"> ثُمَّ أبحَرَ بولُسُ ورَفيقاهُ مِنْ بافوسَ إلى بَرجةَ في بَمفيليةَ، ففارَقَهُما يوحنَّا ورَجَعَ إلى أُورُشليمَ.</span>
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 16:13; اعمال الرسل 14:19; اعمال الرسل 13:5"> أمَّا هُما فتَوجَّها مِنْ بَرجَةَ إلى أنطاكيةَ في بِسيدِيَّةَ. ودَخَلا المَجمعَ يومَ السَّبتِ وجَلسا.</span>
<span className="v" id="v15" data-v="15" data-xref="العِبرانِيّين 13:22; اعمال الرسل 15:21; مَرْقُس 5:22"> وبَعدَ تلاوةِ فَصلٍ مِنْ شَريعةِ موسى وكُتبِ الأنبياءِ، أرسَلَ إلَيهِما رُؤساءُ المَجمَعِ يَقولونَ: ”أيُّها الأخوانِ، إنْ كانَ عِندَكُما ما تَعِظانِ بِه الشَّعبَ، فتكَلَّما“.</span>
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 12:17; اعمال الرسل 13:26; اعمال الرسل 10:2"> فقامَ بولُسُ وأشارَ بيدِهِ وقالَ: ”يا بَني إِسرائيلَ، ويا أيُّها الّذينَ يَتَّقونَ اللهَ، اَسمَعوا:</span>
<span className="v" id="v17" data-v="17" data-xref="التَّثْنِيَة 7:6–8; إشَعْياء 63:9–14; المَزامِير 105:23–24"> إلهُ هذا الشَّعبِ، شَعبِ إِسرائيلَ، اَختارَ آباءَنا ورفَع قَدْرَ هذا الشَّعبِ طَوالَ غُربَتِهِ في أرضِ مِصْرَ. ثُمَّ أخرَجَهُم مِنها بِقُوَّةِ ذِراعِهِ</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 7:36; المَزامِير 95:8–11; المَزامِير 106:13–29"> واَحتَمَلَ ما كانَ مِنهُم نحوَ أربعينَ سنَةً في البرِّيَّةِ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 7:1; المَزامِير 78:55; اعمال الرسل 7:45"> وأبادَ سَبعَ أُمَمٍ في أرضِ كَنعانَ وأورَثَهُم أرضَها،</span>
<span className="v" id="v20" data-v="20" data-xref="القُضاة 2:16; صَمُوئيلَ الأوَّلُ 3:20; المُلُوكِ الثّانِي 23:22"> واَستَغرَقَ ذلِكَ نحوَ أربع مِئةٍ وخمسينَ سنَةً. ثُمَّ جعَلَ لهُم قُضاةً حتّى عَهدِ النَّبيِّ صَموئيلَ.</span>
<span className="v" id="v21" data-v="21" data-xref="صَمُوئيلَ الأوَّلُ 10:1; صَمُوئيلَ الأوَّلُ 15:1; صَمُوئيلَ الأوَّلُ 12:12–19"> ثُمَّ طَلَبوا مِنَ اللهِ مَلِكًا، فأعطاهُم شاوُلَ بنَ قَيسٍ، مِنْ عَشيرَةِ بَنيامّينَ، طَوالَ أربَعينَ سنَةً.</span>
<span className="v" id="v22" data-v="22" data-xref="صَمُوئيلَ الأوَّلُ 13:13–14; المُلُوكِ الأوَّلُ 15:5; المَزامِير 78:70–72"> ثُمَّ عَزَلَهُ وأقامَ داوُدَ مَلِكًا علَيهِم وشَهِدَ لَه بِقولِهِ: وجَدتُ داوُدَ بنَ يسّى رَجُلًا يَرتَضيهِ قلبي، وسيَعمَلُ كُلَّ ما أُريدُ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 1:1; المَزامِير 132:11; رُؤيا يُوحَنّا 22:16"> وأخرَجَ اللهُ مِنْ نَسلِ داوُدَ حسَبَ الوَعدِ يَسوعَ مُخلِّصًا لِشَعبِ إِسرائيلَ.</span>
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 1:22; يُوحَنّا 1:6–8; لُوقا 3:2–20"> وقَبْلَ مَجيءِ يَسوعَ، دعا يوحنَّا جميعَ شَعبِ إِسرائيلَ إلى مَعموديَّةِ التَّوبَةِ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 3:11; مَرْقُس 1:7; يُوحَنّا 1:26–27"> وقالَ وهوَ يُنهي سَعيَهُ: أتَظُنُّونَ أنِّي أنا هوَ؟ لا! فذاكَ يَجيءُ بَعدي، وما أنا أهلٌ لأنْ أحُلَّ رِباطَ حِذائِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="لُوقا 1:77; ﺭﻭﻣﻴﺔ 1:16; لُوقا 1:69"> يا إخوتي، يا أبناءَ إبراهيمَ، ويا أيُّها الحاضِرونَ مَعكُم مِنَ الّذينَ يَتَّقونَ اللهَ، إلَينا أرسَلَ اللهُ كَلِمَةَ الخلاصِ.</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 3:17; لُوقا 24:24–27; لُوقا 24:20"> فلا أهلُ أُورُشليمَ ورُؤساؤُهُم عَرَفوا المَسيحَ، ولا هُم فَهِموا ما يُتلى مِنْ أقوالِ الأنبِياءِ في كُلِّ سبتٍ، فتَمَّموها بِالحُكمِ علَيهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 23:21–25; لُوقا 23:14–16; يُوحَنّا 19:4"> ومعَ أنَّهُم ما وَجَدوا جُرمًا يَستَوجِبُ بِه الموتَ، طَلَبوا مِنْ بيلاطُسَ أنْ يَقتُلَه.</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 23:53; يُوحَنّا 19:28; يُوحَنّا 19:30"> وبَعدَما تَمَّموا كُلَّ ما كتَبَهُ الأنبياءُ في شأنِهِ، أنزَلوهُ عَن الصَّليبِ ووَضعوهُ في القَبرِ.</span>
<span className="v" id="v30" data-v="30" data-xref="اعمال الرسل 2:24; اعمال الرسل 17:31; ﻣﺘﻰ 28:6"> ولكِنَّ اللهَ أقامَهُ مِنْ بَينِ الأمواتِ،</span>
<span className="v" id="v31" data-v="31" data-xref="لُوقا 24:48; اعمال الرسل 1:3; اعمال الرسل 1:11"> فظَهرَ أيّامًا كثيرةً لِلَّذينَ صَعِدوا معَهُ مِنَ الجَليلِ إلى أُورُشليمَ. وهُم الآنَ شُهودٌ لَه عِندَ الشَّعبِ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="ﺭﻭﻣﻴﺔ 4:13; اعمال الرسل 26:6; حِزْقِيال 34:23"> ونَحنُ نُبَشِّرُكُم بِأَنَّ ما وعَدَ اللهُ بِه آباءَنا</span>
<span className="v" id="v33" data-v="33" data-xref="المَزامِير 2:7; العِبرانِيّين 5:5; العِبرانِيّين 1:5–6"> تَمَّ لنا، نَحنُ أبناءَهُم، حينَ أقامَ يَسوعَ مِنْ بَينِ الأمواتِ، كما جاءَ في المزمورِ الثاني:<br />أنتَ اَبني، وأنا اليومَ ولَدتُكَ.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="إشَعْياء 55:3; المَزامِير 89:2–4; ﺭﻭﻣﻴﺔ 6:9"> وأنْ يكونَ اللهُ أقامَهُ مِنْ بَينِ الأمواتِ ولن يَعودَ إلى الفَسادِ، فهذا وارِدٌ في قَولِهِ: سأُعطيكُمُ البَرَكاتِ المُقدَّسَةَ الأكيدةَ الّتي وَعَدتُ بِها داوُدَ.</span>
<span className="v" id="v35" data-v="35" data-xref="المَزامِير 16:10; اعمال الرسل 2:27–31; المَزامِير 49:9"> ولذلِكَ قالَ في مزمورٍ آخَرَ:<br />لن تَترُكَ قُدُّوسَكَ يرى الفسادَ.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="اعمال الرسل 2:29; صموئيلَ الثّانِي 7:12; المُلُوكِ الأوَّلُ 2:10"> لكِنَّ داوُدَ، بَعدَما عَمِلَ بِمَشيئةِ اللهِ في أيّامِهِ، رقَدَ ودُفِنَ بِجِوارِ آبائِهِ، فرَأَى الفَسادَ.</span>
<span className="v" id="v37" data-v="37" data-xref="اعمال الرسل 13:30; اعمال الرسل 2:24"> وأمَّا الّذي أقامَهُ اللهُ، فَما رأى الفَسادَ.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="لُوقا 24:47; رسالَةُ يُوحَنّا الأُولَى 2:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:18–21"> فاَعلَموا، يا إخوَتي، أنَّنا بيَسوعَ نُبَشِّرُكُم بِغُفرانِ الخَطايا،</span>
<span className="v" id="v39" data-v="39" data-xref="ﺭﻭﻣﻴﺔ 10:4; غَلاطِيَّة 2:16; ﺭﻭﻣﻴﺔ 8:3"> وأنَّ مَنْ آمَنَ بِه يَتَبرَّرُ مِنْ كُلِّ ما عَجِزَت شريعةُ موسى أنْ تُبَرِّرَهُ مِنهُ.</span>
<span className="v" id="v40" data-v="40" data-xref="مَلاخِي 4:1; العِبرانِيّين 3:12; مَلاخِي 3:2"> فاَنتَبِهوا لِئَلَّا يَحِلَّ بِكُم ما قيلَ في كُتُبِ الأنبياءِ:</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="حَبَقُّوق 1:5; رسالَةُ بُطرُسَ الأُولَى 4:17; ﺭﻭﻣﻴﺔ 11:7–14"> ”أنظُروا، أيُّها المُستَهزِئونَ!<br />تَعجَّبوا واَهلكوا.<br />فأنا أعمَلُ في أيّامِكُم عَمَلًا<br />إنْ أخبَركُم بِه أحَدٌ لا تُصَدِّقونَ“.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="اعمال الرسل 28:28; اعمال الرسل 10:33; ﻣﺘﻰ 19:30"> وبَينَما بولُسُ وبَرنابا خَارِجانِ مِنَ المَجمَعِ، طَلَبوا إلَيهِما أنْ يُحدِّثاهُم بهذِهِ الأمورِ في السَّبتِ القادِمِ.</span>
<span className="v" id="v43" data-v="43" data-xref="اعمال الرسل 11:23; اعمال الرسل 14:22; أفَسُس 2:8"> وتَبِعَهُما بَعدَ الاجتماعِ كَثيرٌ مِنَ اليَهودِ والدُّخلاءِ الّذينَ يَعبُدونَ اللهَ، فكَلَّماهُم وشَجَّعاهُم على الثَّباتِ في نِعمَةِ اللهِ.</span>
</div>
<div className="bp">
