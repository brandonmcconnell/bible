## مجمع أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="غَلاطِيَّة 5:6; اعمال الرسل 15:5; اللّاويِّين 12:3"> ونزَلَ جماعةٌ مِنَ اليَهودِيَّةِ وأخَذوا يُعَلِّمونَ الإخوةَ، فيَقولونَ: ”لا خَلاصَ لكُم إلَّا إذا اَختَتَنتُم على شريعةِ موسى“.</span>
<span className="v" id="v2" data-v="2" data-xref="اعمال الرسل 15:22–23; اعمال الرسل 15:6–7; غَلاطِيَّة 2:1–2"> فوَقَعَ بَينَهُم وبَينَ بولُسَ وبَرنابا خِلافٌ وجِدالٌ شديدٌ. فأجمَعوا على أنْ يَصعَدَ بولُسُ وبَرنابا وغيرُهُما مِنَ المُؤمنينَ إلى أُورُشليمَ ليُراجِعوا الرُّسُلَ والشُّيوخَ في هذِهِ المَسألَةِ.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 14:27; اعمال الرسل 21:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:11"> وبَعدَما شَيَّعتْهُمُ الكَنيسةُ اَجتازوا فِينيقيةَ والسّامِرَةَ يُخبِرونَ كيفَ اَهتَدى غَيرُ اليَهودِ، فيَفْرَح الإخوةُ كثيرًا.</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 14:27; اعمال الرسل 15:12; اعمال الرسل 15:3"> فلمَّا وصَلوا إلى أُورُشليمَ رَحَّبتْ بِهِم الكَنيسةُ والرُّسُلُ والشُّيوخُ، فأخبروهُم بِكُلِّ ما أجرَى اللهُ على أيْديهِم.</span>
<span className="v" id="v5" data-v="5" data-xref="غَلاطِيَّة 5:1–3; فِيلِبِّي 3:5–8; اعمال الرسل 15:1"> فقامَ بَعضُ المُؤمنينَ الّذينَ كانوا مِنْ قَبلُ على مَذهَبِ الفَرِّيسيِّينَ وقالوا: ”يَجبُ أن يَختَتِنَ غَيرُ اليَهودِ ويَعمَلوا بِشَريعةِ موسى“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="الأمثال 15:22; العِبرانِيّين 13:7; العِبرانِيّين 13:17"> فاَجتَمَعَ الرُّسُلُ والشُّيوخُ لِلنَظرِ في هذِهِ المَسأَلةِ.</span>
<span className="v" id="v7" data-v="7" data-xref="ﺭﻭﻣﻴﺔ 10:17–18; غَلاطِيَّة 2:7–9; أخبارِ الأيّامِ الأوّلُ 28:4–5"> وبَعدَ جِدالٍ طويلٍ قامَ بُطرُسُ وقالَ لهُم: ”أيُّها الإخوةُ، تَعرِفونَ أنَّ اللهَ اَختارَني مِنْ بَينِكُم مِنْ زَمنٍ بعيدٍ لِيَسمَعَ غيرُ اليَهودِ مِنْ فَمي كلامَ البِشارَةِ ويُؤمنوا.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 1:24; اعمال الرسل 10:47; اعمال الرسل 14:3"> واللهُ الّذي يَعرِفُ ما في القُلوبِ، شَهِدَ على رِضاهُ عَنهُم، فوَهَبَ لهُمُ الرُّوحَ القُدُسَ كما وهَبَهُ لنا،</span>
<span className="v" id="v9" data-v="9" data-xref="أفَسُس 3:6; غَلاطِيَّة 5:6; اعمال الرسل 10:28"> فما فَرَّقَ بَينَنا وبَينَهُم في شيءٍ. فهوَ طَهَّرَ قُلوبَهُم بالإيمانِ.</span>
<span className="v" id="v10" data-v="10" data-xref="غَلاطِيَّة 5:1; ﻣﺘﻰ 23:4; غَلاطِيَّة 4:9"> فلماذا تُجرِّبونَ اللهَ الآنَ بأنْ تَضَعُوا على رِقابِ التَّلاميذِ نِيرًا عَجَزَ آباؤُنا وعَجَزْنا نَحنُ عَنْ حَمْلِهِ؟</span>
<span className="v" id="v11" data-v="11" data-xref="ﺭﻭﻣﻴﺔ 3:24; تِيطُس 3:4–7; تِيطُس 2:11"> خُصوصًا ونَحنُ نُؤمِنُ أنَّنا نَخلُصُ بِنعمَةِ الرَّبِّ يَسوعَ كما هُمْ يَخلُصُونَ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 14:27; اعمال الرسل 15:4; اعمال الرسل 21:19"> فسكَتَ المُجتمِعونَ كُلُّهُم واَستَمَعوا إلى بَرنابا وبولُسَ وهُما يَروِيانِ لهُمُ الآياتِ والعَجائِبَ الّتي أجراها اللهُ على أيديهِما بَينَ غَيرِ اليَهودِ.</span>
<span className="v" id="v13" data-v="13" data-xref="اعمال الرسل 12:17; غَلاطِيَّة 2:12; يَعقُوب 1:1"> وبَعدَما خَتَما كلامَهُما، قالَ يَعقوبُ: ”إِستَمِعوا لي، أيُّها الإخوةُ:</span>
<span className="v" id="v14" data-v="14" data-xref="رسالَةُ بُطرُسَ الثّانيَة 1:1; اعمال الرسل 15:7–9; لُوقا 2:31–32"> أخبَركُم سِمعانُ كيفَ اَهتَمَّ اللهُ في أوَّلِ الأمرِ أنْ يتَّخِذَ مِنْ بَينِ الأُمَمِ شَعبًا لاَسمِهِ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 15:8–12; اعمال الرسل 13:47"> وهذا يُوافِقُ كلامَ الأنبياءِ في الكُتُبِ المُقدَّسَةِ:</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="عامُوس 9:11–12; المَزامِير 89:35–49; لُوقا 1:31–33"> ”سأرجِعُ بَعدَ هذا،<br />فأبني خَيمةَ داوُدَ المُتَهدِّمَةَ:<br />أبني أنقاضَها وأُقيمُها</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="زَكَريّا 2:11; زَكَريّا 8:20–23; إشَعْياء 65:1"> ليَسعى سائرُ النَّاسِ إلى الرَّبِّ وجميعُ الشُّعوبِ الّتي تَحمِلُ اَسمي.<br />هذا ما يَقولُ الرَّبُّ الّذي صَنَعَ هذا كُلَّه،</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="أفَسُس 1:4; أفَسُس 3:9; إشَعْياء 46:9–10"> وجعَلَهُ مَعروفًا مِنْ قَديمِ الزَّمانِ“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 15:28; غَلاطِيَّة 1:7–10; اعمال الرسل 15:24"> ولذلِكَ أرى أنْ لا نُثَقِّلَ على الّذينَ يَهتدُونَ إلى اللهِ مِنْ غَيرِ اليَهودِ،</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 15:29; رُؤيا يُوحَنّا 2:20; اللّاويِّين 3:17"> بل نكتُبَ إلَيهِم أنْ يَمتَنعوا عَنْ ذَبائحِ الأصنامِ النَّجِسَةِ والزِّنى والحيوانِ المَخنوقِ والدَّمِ.</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 13:15; اعمال الرسل 13:27; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:14–15"> فَلِشريعةِ موسى مِنْ قديمِ الزَّمانِ مُعَلِّمونَ في كُلِّ مدينةٍ يَقرأونَها كُلَّ سَبتٍ في المَجامعِ“.</span>
</div>
</div>

## رسالة إلى المؤمنين من غير اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اعمال الرسل 15:27; اعمال الرسل 15:40; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 1:1"> فأجمَعَ الرُّسُلُ والشُّيوخُ والكنيسةُ كُلُّها على اَختِيارِ رَجُلَينِ مِنهُم يُرسلونَهُما إلى أنطاكيةَ معَ بَرنابا وبولُسَ. فاَختاروا يَهوذا الّذي يُقالُ لَه بَرسابا، وسيلا، وكانا مِمَّنْ لهُم مكانَةٌ رَفيعَةٌ بَينَ الإخوةِ،</span>
<span className="v" id="v23" data-v="23" data-xref="اعمال الرسل 23:26; يَعقُوب 1:1; غَلاطِيَّة 1:21"> وسَلَّموا إلَيهِم هذِهِ الرِّسالةَ:<br />”مِنْ إخوتِكُمُ الرُّسلِ والشُّيوخِ إلى الإخوَةِ المُهتدينَ مِنْ غَيرِ اليَهودِ في أنطاكيةَ وسوريةَ وكيليكيةَ، سلامٌ:</span>
<span className="v" id="v24" data-v="24" data-xref="غَلاطِيَّة 5:12; غَلاطِيَّة 5:10; اعمال الرسل 15:1"> سَمِعنا أنَّ بَعضَ النَّاسِ جاؤُوا إلَيكُم مِنْ غيرِ توكيلٍ مِنّا، فأزعَجوكُم وأقلَقوا أفكارَكُم.</span>
<span className="v" id="v25" data-v="25" data-xref="رسالَةُ بُطرُسَ الثّانيَة 3:15; اعمال الرسل 1:14; اعمال الرسل 15:2"> فأجمَعْنا على أنْ نَختارَ رَجُلَينِ نُرسِلُهُما إلَيكُم معَ الحَبيبَينِ بَرنابا وبولُسَ،</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 14:19; اعمال الرسل 9:23–25; فِيلِبِّي 2:29–30"> رجُلَينِ كَرَّسا حياتَهُما لِخِدمَةِ رَبِّنا يَسوعَ المَسيحِ.</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 15:22; اعمال الرسل 15:32; رسالَةُ يُوحَنّا الثّالثَة 1:13"> فأرسَلْنا يَهوذا وسيلا ليَقولا لكُم مُشافَهَةً ما نكتُبُ بِه إلَيكُم.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 16:13; رُؤيا يُوحَنّا 2:24; ﻣﺘﻰ 11:30"> فالرُّوحُ القُدُسُ ونَحنُ رأينا أنْ لا نُحَمِّلَكُم مِنَ الأثقالِ إلَّا ما لا بُدَّ مِنهُ، وهوَ</span>
<span className="v" id="v29" data-v="29" data-xref="اعمال الرسل 15:20; اعمال الرسل 21:25; رُؤيا يُوحَنّا 2:14"> أنْ تَمتَنِعوا عَنْ ذَبائِحِ الأصنامِ، وعَنِ الدَّمِ والحيوانِ المخنوقِ والزِّنى. فإذا صُنتُم أنفُسَكُم مِنها، فحَسَنًا تَفعَلونَ. واللهُ مَعكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="اعمال الرسل 16:4; اعمال الرسل 6:2; اعمال الرسل 23:33"> فاَنْصَرَفوا ونَزَلوا إلى أنطاكيةَ، فدَعوا جَماعةَ المُؤمنينَ وسَلَّموا إليهِمِ الرِّسالَةَ.</span>
<span className="v" id="v31" data-v="31" data-xref="غَلاطِيَّة 2:4–5; فِيلِبِّي 3:3; غَلاطِيَّة 5:1"> فلمَّا قرَأوها فَرِحوا كثيرًا بِما جاءَ فيها مِنْ تَشجيعٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 11:23; اعمال الرسل 14:22; اعمال الرسل 13:1"> وألقى يَهوذا وسيلا، وكانا هُما أيضًا نَبيَّينِ، عِظةً طويلةً شَجَّعا بِها الإخوةَ وشدَّدا عَزائِمَهُم.</span>
<span className="v" id="v33" data-v="33" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:11; اعمال الرسل 16:36; العِبرانِيّين 11:31"> وبَعدَما أقاما بَعضَ الوقتِ في أنطاكيةَ، صرَفَهُما الإخوةُ بِسلامٍ إلى الّذينَ أرسَلوهُما.</span>
<span className="v" id="v34" data-v="34"> &#91;ولكِنَّ سيلا رَأى أنْ يَبقى هُناكَ، فرَجَعَ يَهوذا وحدَهُ&#93;.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:2; كُولُوسِي 1:28; اعمال الرسل 13:1"> وأقامَ بولُسُ وبَرنابا في أنطاكِيَةَ، يُعَلِّمانِ ويُبَشِّرانِ بِكلامِ الرَّبِّ ومعَهُما آخرونَ كثيرونَ.</span>
</div>

## بولس يفارق برنابا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="اعمال الرسل 14:6; اعمال الرسل 13:4; اعمال الرسل 13:51"> وبَعدَ أيّامٍ قَليلةٍ، قالَ بولُسُ لبَرنابا: ”تعالَ نَرجِعُ لِنَتَفقَّدَ الإخوةَ في كُلِّ مدينةٍ بَشَّرْنا فيها بِكلامِ الرَّبِّ، ونَطَّلِعَ على أحوالِهِم“.</span>
<span className="v" id="v37" data-v="37" data-xref="اعمال الرسل 12:12; فِلِيْمُون 1:24; اعمال الرسل 13:5"> فأرادَ بَرنابا أنْ يُرافِقَهما يوحنَّا المُلقَّبُ بمَرقُسَ،</span>
<span className="v" id="v38" data-v="38" data-xref="اعمال الرسل 13:13; الأمثال 25:19; لُوقا 14:27–34"> ولكنَّ بولُسَ رَأى أنْ لا يُرافِقَهُما مَنْ فارَقَهُما في بَمفيلِيَّةَ وما شارَكَهُما في العمَلِ.</span>
<span className="v" id="v39" data-v="39" data-xref="كُولُوسِي 4:10; الجامِعَة 7:20; اعمال الرسل 4:36"> فوقَعَ بَينَهُما نِزاعٌ حتّى اَفتَرَقا. فأخَذَ بَرنابا مَرقُسَ وسافَرَ في البحرِ إلى قُبرُصَ.</span>
<span className="v" id="v40" data-v="40" data-xref="اعمال الرسل 15:22; اعمال الرسل 14:26; اعمال الرسل 11:23"> وأمَّا بولُسُ، فاَختارَ سيلا وخرَجَ مِنْ أنطاكيةَ، بَعدَما اَستَودَعَه الإخوةُ نِعمةَ الرَّبِّ.</span>
<span className="v" id="v41" data-v="41" data-xref="اعمال الرسل 15:23; غَلاطِيَّة 1:21; اعمال الرسل 15:32"> فاَجتازَ سُورِيَّةَ وكيليكيةَ يُقوِّي إيمانَ الكَنائسِ.</span>
</div>
