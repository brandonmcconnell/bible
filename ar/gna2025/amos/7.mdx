## رؤيا الجراد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوئيل 1:4; عامُوس 8:1; يُوئيل 2:25"> هذا ما أراني السَّيِّدُ الرّبُّ: كانَ يَجبلُ جَرادًا بعدَ أن اَقتَطَعَ المَلِكُ حِصَّتَهُ وحَلَّ طُلوعُ النَّبْتِ المُتَأخِّرِ،</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 11:13; إشَعْياء 37:4; إرْمِيا 42:2"> فلمَّا فرَغَ الجَرادُ مِنْ أكلِ عُشبِ الأرضِ، قُلتُ: ”أيُّها السَّيِّدُ الرّبُّ، عَفوكَ! كيفَ يقومُ لِبَني يَعقوبَ قائمةٌ؟ فهُم شعبٌ صغيرٌ“.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 32:36; ﻳﻮﻧﺎﻥ 3:10; هُوشَع 11:8"> فنَدِمَ الرّبُّ على ذلِكَ وقالَ: ”هذا لا يكونُ“.</span>
</div>
</div>

## رؤيا النار

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="عامُوس 7:1; إشَعْياء 66:15–16; العِبرانِيّين 1:7"> وهذا ما أراني السَّيِّدُ الرّبُّ: كانَ السَّيِّدُ الرّبُّ يدعو إلى الهَلاكِ بالنَّارِ، فأكلتِ النَّارُ الغَمرَ العظيمَ وأخذَت تأكلُ الأرضَ اليابِسَةَ.</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 85:4; إشَعْياء 1:9; إرْمِيا 30:19"> فقُلتُ: ”أيُّها السَّيِّدُ الرّبُّ. توقَّفْ! كيفَ يقومُ لبَني يَعقوبَ قائِمةٌ؟ فهُم شعبٌ صغيرٌ“.</span>
<span className="v" id="v6" data-v="6" data-xref="عامُوس 7:3; القُضاة 2:18; المَزامِير 90:13"> فندِمَ السَّيِّدُ الرّبُّ على ذلِكَ وقالَ: ”هذا أيضًا لا يكونُ“.</span>
</div>
</div>

## رؤيا الشاقول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الثّانِي 21:13; رُؤيا يُوحَنّا 21:15; إشَعْياء 28:17"> وهذا ما أراني: كانَ الرّبُّ واقِفًا على حائطٍ مَبنيٍّ على الشَّاقولِ، وبيدِهِ شاقولٌ.</span>
<span className="v" id="v8" data-v="8" data-xref="عامُوس 8:2; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:8; حِزْقِيال 7:2–9"> فقالَ لي: ”ماذا ترى يا عاموسُ؟“ فقُلتُ: ”شاقولًا“. فقالَ: ”سأجعَلُ الشَّاقولَ في وسَطِ شعبي إسرائيلَ ولا أعودُ أَغُضُّ النَّظرَ عَنِ اعوجاجِهِم مِنْ بَعدُ.</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الثّانِي 15:8–10; التَّكوين 46:1; عامُوس 8:14"> فأُدَمِّرُ مَعابِدَ ذُرِّيَّةِ إسحقَ على المُرتفَعاتِ وأخرِّبُ كُلَّ مَوضِعٍ مُقَدَّسٍ في أرضِ إسرائيلَ، وأقومُ على بَيتِ يرُبْعامَ بالسَّيفِ“.</span>
</div>
</div>

## عاموس يُطرد من بيت إيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إرْمِيا 38:4; المُلُوكِ الأوَّلُ 12:31–32; إرْمِيا 26:8–11"> فأرسلَ أمَصْيا، كاهنُ بَيت إيلَ، إلى يرُبْعامَ مَلِكِ إِسرائيلَ يقولُ لَه: ”يَجُرُّ عاموسُ علَيكَ الشُّؤْمَ في بَني إسرائيلَ، والبلادُ لا تقدرُ أن تَحتمِلَ جميعَ كلامِهِ.</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 17:6; المَزامِير 56:5; عامُوس 6:7–8"> فهوَ يقولُ: سيموتُ يرُبْعامُ بالسَّيفِ، ويُسبَى إسرائيلُ عَنْ أرضِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14; ﻣﺘﻰ 8:34; صَمُوئيلَ الأوَّلُ 9:9"> وقالَ أمَصْيا لعاموسَ: ”أيُّها الّذي يرى رؤًى! إذهبِ اَهربْ إلى أرضِ يَهوذا، وهُناكَ تنبَّأْ وكُلْ خُبزَكَ،</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الأوَّلُ 12:32; عامُوس 2:12; المُلُوكِ الأوَّلُ 12:29"> وأمَّا بَيتُ إيلَ فلا تَعُدْ تتَنَبَّأُ فيها لأنَّها معبَدُ المَلِكِ وبيتُ مُلْكِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="عامُوس 1:1; المُلُوكِ الثّانِي 4:38; المُلُوكِ الثّانِي 2:5"> فقالَ عاموسُ لأمَصْيا: ”ما أنا نبيٌّ ولا اَبنُ نبيٍّ، إنَّما أنا راعي غنَمٍ وقاطفُ جُمَّيزٍ،</span>
<span className="v" id="v15" data-v="15" data-xref="صموئيلَ الثّانِي 7:8; حِزْقِيال 2:3–4; إرْمِيا 1:7"> أخذَني الرّبُّ مِنْ وراءِ الغنَمِ وقالَ لي: إِذهَبْ تَنَبَّأْ لِشعبي إِسرائيلَ.</span>
<span className="v" id="v16" data-v="16" data-xref="حِزْقِيال 20:46; ميخا 2:6; حِزْقِيال 21:2"> فالآنَ إِسمَعْ كلِمةَ الرّبِّ: أنتَ تقولُ لا تتَنَبَّأْ شَرًّا على بَني إِسرائيلَ ولا تتكَلَّمْ سُوءًا على بَيتِ إسحقَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="هُوشَع 9:3; المُلُوكِ الثّانِي 17:6; هُوشَع 4:13–14"> لذلِكَ هذا ما قالَ الرّبُّ: تَزني امرأتُكَ في المدينةِ ويسقطُ بَنوكَ وبَناتُكَ بالسَّيفِ، وتُقسَمُ أرضُكَ بالحَبلِ، وتموتُ أنتَ في أرضٍ نجِسَةٍ، ويُسبَى شعبُ إِسرائيلَ عَنْ أرضِهِ“.</span>
</div>
</div>
