## الساكنون في أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 48:2; إشَعْياء 52:1; نَحَمْيا 11:18"> وسكَنَ زُعماءُ الشَّعبِ في أورُشليمَ، أمَّا سائرُ الشَّعبِ فألقَوا القُرْعةَ لاَختيارِ واحدٍ مِنْ عشَرَةٍ ليَسكُنَ في أورُشليمَ، المدينةِ المقدَّسةِ، على أنْ يَسكُنَ التِّسعةُ الباقونَ في المُدُنِ الأخرى.</span>
<span className="v" id="v2" data-v="2" data-xref="القُضاة 5:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:16–17; أيُّوب 29:13"> وباركَ الشَّعبُ جميعَ الّذين قدَّموا أنفسَهُم طَوعًا لِيَسكنوا في أورُشليمَ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="عَزْرا 2:1; عَزْرا 2:43; أخبارِ الأيّامِ الأوّلُ 9:1–34"> ففي المُدُنِ الأخرى بيَهوذا سكَنَ الشَّعبُ والكهَنةُ واللَّاويُّونَ وخُدَّامُ الهَيكلِ وبَنو عبيدِ سليمانَ، كُلُّ واحدٍ في مُلْكِهِ في مدينتهِ. هؤلاءِ أعيانُ البِلادِ الّذينَ سكنوا في أورُشليمَ،</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 38:29; ﻣﺘﻰ 1:3; لُوقا 3:33"> وهُم مِنْ بَني يَهوذا وبَني بَنيامينَ. فمِنْ بَني يَهوذا عَثايا بنُ عُزِّيَّا بنِ زكريَّا بنِ أمَرْيا بنِ شَفَطْيا بنِ مَهْلَلْئيلَ مِنْ بَني فارِصَ</span>
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الأوّلُ 9:5; نَحَمْيا 3:15; العَدَد 26:20"> ومَعْسِيا بنُ باروخَ بنِ كلحوزَةَ بنِ حَزايا بنِ عَدايا بنِ يوياريبَ بنِ زكريَّا بنِ الشِّيلونيّ.</span>
<span className="v" id="v6" data-v="6"> جميعُ بَني فارصَ السَّاكنينَ في أورُشليمَ أربعُ مئَةٍ وثمانيةٌ وسِتُّونَ رجلًا مِنَ الأشدَّاءِ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الأوّلُ 9:7–9"> وهؤلاءِ بَنو بَنيامينَ: سلُّو بنُ مَشُلَّامَ بنِ يوعيدَ بنِ فَدايا بن قُولايا بنِ مَعْسِيا بنِ إيثيئيلَ بنِ يَشَعْيا،</span>
<span className="v" id="v8" data-v="8"> وبَعدَهُ جَبَّايُ سلَّايُ تِسعُ مئَةٍ وثمانيةٌ وعشرونَ</span>
<span className="v" id="v9" data-v="9" data-xref="أخبارِ الأيّامِ الأوّلُ 9:7"> وكانَ يوئيلُ بنُ زكري وكيلًا علَيهِم ويَهوذا بنُ هَسْنوآةَ ثانيًا على المدينةِ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="أخبارِ الأيّامِ الأوّلُ 9:10; نَحَمْيا 12:6; نَحَمْيا 12:19"> وسكنَ أورُشليمَ مِنَ الكهَنةِ يَدَعْيا بنُ يوياريبَ وياكينُ</span>
<span className="v" id="v11" data-v="11" data-xref="أخبارِ الأيّامِ الأوّلُ 9:11; أخبارِ الأيّامِ الأوّلُ 6:7–14; العَدَد 3:32"> وسرايا بنُ حِلْقيَّا بنِ مَشُلامَ بنِ صادوقَ بنِ مرايوثَ بنِ أخيطوبَ، رئيسُ كَهنةِ هَيكلِ اللهِ،</span>
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الأوّلُ 9:12–13"> وهُم معَ أنسِبائِهِم ثماني مئَةٍ واَثنانِ وعشرونَ خادِمًا في هَيكلِ اللهِ، وعَدايا بنُ يروحامَ بنِ فلَلْيا بنِ أمْصي بنِ زكريَّا بنِ فشحورَ بنِ مَلْكيَّا،</span>
<span className="v" id="v13" data-v="13"> وهوَ معَ أنسبائِهِ مِنْ رؤساءِ العشائِرِ مئتانِ واَثنانِ وأربعونَ، وعمْشِسايُ بنُ عَزَرْئيلَ بنِ أخزايَ بنِ مَشْليموثَ بنِ إِمِّيرَ،</span>
<span className="v" id="v14" data-v="14"> وهوَ معَ أنسبائِهِ مئَةٌ وثمانيةٌ وعشرونَ رَجلًا مِنَ الجبابرةِ الأشدَّاءِ والوكيلُ علَيهِم زَبْديئيلُ بنُ هَجْدوليمَ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="أخبارِ الأيّامِ الأوّلُ 9:14; أخبارِ الأيّامِ الأوّلُ 9:19"> وكانَ مِنَ اللَّاويِّينَ الّذينَ سكنوا أورُشليمَ شَمَعْيا بنُ حشوبَ بنِ عَزْريقامَ بنِ حَشَبْيا بنِ بُوني</span>
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 6:2–3; نَحَمْيا 8:7; عَزْرا 8:33"> وشَبْتايُ ويوزابادُ، وهُما مِنْ رؤساءِ اللَّاويِّينَ القيِّمينَ على العمَلِ الخارجيِّ لهَيكلِ اللهِ.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الأوّلُ 9:15; أخبارِ الأيّامِ الأوّلُ 25:1–6; نَحَمْيا 12:8–9"> ومَتَّنْيا بنُ مِيخا بنِ زَبْدي بنِ آسافَ، وهوَ رئيسُ جَوقةِ التَّهليلِ بحَمدِ اللهِ في الصَّلاةِ، وبقْبُقيا مساعِدُهُ مِنْ بَينِ أنسبائِهِ وعَبْدا بنُ شمُّوعَ بنِ جَلالَ بنِ يَدُوثُونَ.</span>
<span className="v" id="v18" data-v="18" data-xref="نَحَمْيا 11:1; رُؤيا يُوحَنّا 21:2; دانيال 9:24"> وجميعُ هؤلاءِ اللَّاويِّينَ السَّاكنينَ في أورُشليمَ، المدينةِ المُقدَّسةِ، مِئتانِ وثمانيةٌ وأربعونَ،</span>
<span className="v" id="v19" data-v="19" data-xref="نَحَمْيا 7:45; أخبارِ الأيّامِ الأوّلُ 9:17–22; المَزامِير 84:10"> والبوَّابونَ عقُّوبُ وطلمونُ وأنسباؤُهُما حارِسو الأبوابِ مئَةٌ واَثنانِ وسَبعونَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="نَحَمْيا 11:3"> وكانَ سائرُ بَني إِسرائيلَ مِنَ الكَهنَةِ واللَّاويِّينَ يَسكنونَ في مُدُنِ يَهوذا، كُلُّ واحدٍ في مُلْكِهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="نَحَمْيا 3:26; أخبارِ الأيّامِ الثّانِي 27:3; نَحَمْيا 3:31"> وأمَّا خُدَّامُ الهَيكلِ فَسكنوا في تَلَّةِ عَوفلَ بأورُشليمَ، وكانَ صيحا وجِشفا وكيلَينِ علَيهِم.</span>
<span className="v" id="v22" data-v="22" data-xref="نَحَمْيا 12:42; أخبارِ الأيّامِ الأوّلُ 25:1–6; عَزْرا 8:19"> وكانَ وكيل اللَّاويِّينَ في أورُشليمَ على خِدمَةِ هَيكلِ اللهِ عُزِّي بنَ بانيَ بنِ حشَبْيا بنِ متَّنْيا بنِ مِيخا مِنْ بَني آسافَ المُغنِّينَ</span>
<span className="v" id="v23" data-v="23" data-xref="عَزْرا 7:20–24; نَحَمْيا 12:47; عَزْرا 6:8–9"> لأنَّ وصيَّةَ المَلِكِ فرَضَتْ على هؤلاءِ المُغنّينَ أنْ يتناوبوا على الغِناءِ في هَيكلِ الرّبِّ كُلَّ يومٍ.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 38:30; أخبارِ الأيّامِ الأوّلُ 18:17; أخبارِ الأيّامِ الأوّلُ 23:28"> وكانَ فَتَحْيا بنُ مَشيزَبْئيلَ مِنْ بَني زارَحَ بنِ يَهوذا مُستشارَ مَلِكِ الفُرسِ في كُلِّ أمورِ الشَّعبِ.</span>
</div>
</div>

## الساكنون في المدن الأخرى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يَشُوع 14:15; يَشُوع 15:21–22"> وفي القُرى معَ حُقولِها سكنَ مِنْ بَني يَهوذا في قريةِ أربعَ وتوابعِها وديبونَ وتوابِعها ويَقبْصَئيل وقُراها.</span>
<span className="v" id="v26" data-v="26" data-xref="يَشُوع 15:26–27; يَشُوع 19:2"> وسكنَ مِنهُم في قُرى يشوعَ ومُولادةَ وبَيتَ فالَطَ</span>
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 26:33; يَشُوع 15:28; القُضاة 20:1"> وحَصرِ شوعالَ وبِئرَ سبعَ وتوابِعِها.</span>
<span className="v" id="v28" data-v="28" data-xref="صَمُوئيلَ الأوَّلُ 27:6; يَشُوع 15:31"> وفي صَقلغَ ومكونَةَ وتوابِعها،</span>
<span className="v" id="v29" data-v="29" data-xref="يَشُوع 19:41; يَشُوع 15:32–33; يَشُوع 15:35"> وفي عينَ رمُّونَ وصرعةَ ويرموثَ</span>
<span className="v" id="v30" data-v="30" data-xref="يَشُوع 10:3; يَشُوع 15:8; يَشُوع 18:16"> وزانوحَ وعَدلَّامَ وقُراهما، وفي لَخيشَ وحقولِها وعزيقةَ وقُراها وبذلِكَ سكنوا مِنْ بِئرَ سبعَ إلى وادي هِنُّومَ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 12:8; يَشُوع 8:9; يَشُوع 18:13"> وسكنَ بَنو بَنيامينَ مِنْ جَبعَ إلى مِخْماسَ وعيَّا وبيتَ إيلَ وتوابِعها</span>
<span className="v" id="v32" data-v="32" data-xref="يَشُوع 21:18; إشَعْياء 10:30; صَمُوئيلَ الأوَّلُ 21:1"> وعناثوثَ ونوبٍ وعَنَنْيةَ</span>
<span className="v" id="v33" data-v="33" data-xref="صموئيلَ الثّانِي 4:3; صَمُوئيلَ الأوَّلُ 7:17; يَشُوع 18:25"> وحاصورَ ورامةَ وجتَّايِمَ</span>
<span className="v" id="v34" data-v="34" data-xref="صَمُوئيلَ الأوَّلُ 13:18"> وحاديدَ وصَبوعِيمَ ونبَلَّاطَ</span>
<span className="v" id="v35" data-v="35" data-xref="أخبارِ الأيّامِ الأوّلُ 8:12; نَحَمْيا 7:37; أخبارِ الأيّامِ الأوّلُ 4:14"> ولودٍ وأونو وادي الصُّنَّاعِ</span>
<span className="v" id="v36" data-v="36" data-xref="يَشُوع 21:1–45; التَّكوين 49:7; أخبارِ الأيّامِ الأوّلُ 6:54–81"> وكانَ مِنَ اللَّاويِّينَ في أرضِ يَهوذا مَنِ اَنتقلوا للسَّكنِ في أرضِ بنيامينَ.</span>
</div>
