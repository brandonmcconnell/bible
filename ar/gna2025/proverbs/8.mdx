## تمجيد الحكمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الأمثال 1:20–21; ﻣﺘﻰ 3:3; يُوحَنّا 7:37"> إسمَعِ الحِكمةَ تُنادي،<br />وجَهارًا تَرفَعُ صوتَها.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="الأمثال 9:14; الأمثال 9:3"> مِنْ على رُؤوسِ القِمَمِ<br />وفي الدُّروبِ ومَفارِقِ الطُّرُقاتِ،</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 18:20; اعمال الرسل 5:20; أيُّوب 29:7"> وبجانِبِ الأبوابِ عِندَ مَدخَلِ المدينةِ،<br />وفي النَّوافذِ ها هيَ تَصيحُ:</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 11:15; يُوحَنّا 3:16; المَزامِير 50:1"> أنا أُناديكُم أيُّها النَّاسُ<br />وأقولُ لكُم يا بني آدمَ:</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="الأمثال 1:22; اعمال الرسل 26:18; رُؤيا يُوحَنّا 3:17–18"> على الجُهَّالِ أنْ يتَعَقَّلوا،<br />وعلى البُلَداءِ أنْ يتَعَلَّموا الفهْمَ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="كُولُوسِي 1:26; ﻣﺘﻰ 7:28–29; الأمثال 23:16"> بالكَلامِ الصَّريحِ أنطِقُ فاَسمَعوا،<br />وبالاَستقامةِ أفتَحُ شفَتَيَّ،</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 37:30; يُوحَنّا 8:14; يُوحَنّا 17:17"> لأنَّ فَمي يلهَجُ بالصِّدْقِ،<br />وشفَتَيَّ تَمقُتانِ الكَذِبَ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 7:46; المَزامِير 12:6; الأمثال 8:13"> كَلامُ فَمي كُلُّه صِدْقٌ،<br />لا اَلتواءَ فيهِ ولا اَعْوِجاجَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الأمثال 14:6; يَعقُوب 1:5; يُوحَنّا 7:17"> كُلُّهُ مُقْنِع لدَى الفهيمِ،<br />وقويمٌ عِندَ مَنْ وجَدَ المعرِفةَ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الأمثال 2:4–5; المَزامِير 119:127; المَزامِير 119:72"> خُذوا مَشورتي لا الفِضَّةَ،<br />واَخْتاروا المعرِفةَ لا الذَّهَبَ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="الأمثال 3:14–15; الأمثال 4:5–7; الأمثال 16:16"> فالحِكمةُ خَيرٌ مِنَ الّلآلئِ،<br />وكُلُّ الجواهِرِ لا تُساوِيها.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 28:26; الخُرُوج 35:30–4; ﺭﻭﻣﻴﺔ 11:33"> تقولُ: أنا معَ رَجاحةِ العقلِ،<br />وليَ المعرِفَةُ وحُسْنُ التَّدبيرِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 97:10; الأمثال 16:6; صَمُوئيلَ الأوَّلُ 2:3"> مَنْ يَخَفِ الرّبَّ يُبغِضِ الشَّرَّ.<br />ويُبغضِ الجاهَ والزَّهوَ<br />وطريقَ الرُّعاعِ وكَلامَ الكَذِبِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الجامِعَة 7:19; إشَعْياء 9:6; إشَعْياء 40:14"> ليَ المشورَةُ وحُسْنُ الرَّأيِ.<br />أنا الفِطنةُ، والجبَروتُ لي.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 13:1; دانيال 2:21; ﻣﺘﻰ 28:18"> بيَ المُلوكُ يَملِكونَ<br />والحُكَّامُ يحكُمونَ بالعَدلِ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16"> بيَ الرُّؤساءُ يَرأسونَ<br />والأمَراءُ يقضونَ في الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 14:23; يُوحَنّا 14:21; المَزامِير 91:14"> أُحِبُّ الّذينَ يُحِبُّونَني،<br />ومَنْ بكَّرَ في طَلَبي وجَدَني.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الأمثال 3:16; ﻣﺘﻰ 6:33; لُوقا 16:11–12"> الغِنى والكرامةُ معي<br />ودوامُ الثَّروةِ والرَّخاءُ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الأمثال 3:14; الأمثال 10:20; الجامِعَة 7:12"> ثمَري خَيرٌ مِنَ الذَّهَبِ الإبريزِ<br />وغَلَّتي خَيرٌ مِنَ الفِضَّةِ النَّقِيَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 5:32; يُوحَنّا 10:27–28; المَزامِير 32:8"> أسيرُ في طريقِ الحَقِّ<br />وأجتازُ مَسالِكَ الإنصافِ،</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:4; الأمثال 24:4; ﻣﺘﻰ 25:46"> فأُورِّثُ مَنْ يُحِبُّونَني<br />وأملأُ خَزائِنَهم كُنُوزًا.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="كُولُوسِي 1:17; يُوحَنّا 1:1–2; الأمثال 3:19"> الرّبُّ اَقتناني أوَّلَ ما خَلَقَ<br />مِنْ قديمِ أعمالِهِ في الزَّمانِ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 17:24; المَزامِير 2:6; التَّكوين 1:26"> مِنَ الأزلِ صنَعَني، مِنَ البَدءِ،<br />مِنْ قَبلِ أنْ كانتِ الأرضُ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الأمثال 3:20; التَّكوين 1:2; رسالَةُ يُوحَنّا الأُولَى 4:9"> وُجِدتُ وما كانَ غَمْرٌ،<br />ولا مياهٌ في قلبِ الينابيعِ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 90:2; المَزامِير 102:25–28; أيُّوب 38:4–11"> قَبلَ أنْ تُخْلَقَ الجِبالُ<br />وقَبلَ التِّلالِ وُجِدْتُ،</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 1:1–31"> حينَ لم تكُنْ أرضٌ ولا مياهٌ،<br />ولا حَفنةٌ مِنْ تُرابِ الكَونِ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 40:22; أيُّوب 26:10; الأمثال 3:19"> وكُنتُ حينَ كوَّنَ السَّماواتِ<br />وحَوَّقَ حولَ وجهِ الغَمْرِ،</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28"> وثَبَّتَ الغُيومَ في العَلاءِ<br />وفَجَّرَ ينابيعَ المياهِ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="المَزامِير 104:9; إرْمِيا 5:22; التَّكوين 1:9–10"> وكُنتُ حينَ حَوَّطَ البحرَ<br />فلا تَعبُرُ المياهُ حُدودَهُ،<br />وحينَ أرسى أساساتِ الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 1:1–3; يُوحَنّا 1:18; إشَعْياء 42:1"> وكُنتُ حيالَهُ بأمانٍ،<br />وفي بَهجةٍ يومًا بَعدَ يومٍ،<br />ضاحِكًا أمامَهُ كُلَّ حينٍ،</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="المَزامِير 16:3; يُوحَنّا 13:1; المَزامِير 40:6–8"> ضاحِكًا في أرضِهِ الآهلةِ،<br />ومُبتَهجًا معَ بَني البشَرِ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="لُوقا 11:28; المَزامِير 119:1–2; المَزامِير 1:1–4"> إسمَعوا لي أيُّها الأبناءُ،<br />فهَنيئًا لِمَنْ يَتبَعُ طُرُقي</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="الأمثال 4:1; المَزامِير 81:11–12; الأمثال 5:1"> إسمَعوا المَشورَةَ وكُونوا حُكَماءَ،<br />ولا تَنبِذُوا ما أقُولُهُ لكُم.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="لُوقا 11:28; ﻣﺘﻰ 7:24; الأمثال 3:13"> هَنيئًا لِمَنْ يَستَمِعُ إليَّ<br />ساهِرًا عِندَ بابي كُلَّ يومٍ.<br />ناطرًا بجانبِ مَدخَلِ داري.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="يُوحَنّا 3:36; رسالَةُ يُوحَنّا الأُولَى 5:11–12; الأمثال 12:2"> مَنْ وجَدَني وجَدَ الحياةَ<br />ونالَ رِضىً مِنَ الرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="الأمثال 15:32; الأمثال 1:31; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:22"> مَنْ أخطأني أضَرَّ نفْسَهُ،<br />ومَنْ أبغَضَني أحبَّ الموتَ.</span>
</div>
