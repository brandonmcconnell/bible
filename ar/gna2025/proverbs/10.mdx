## أمثال سليمان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الأمثال 1:1; الأمثال 29:15; الأمثال 17:25"> الاَبنُ الحكيمُ يُفَرِّحُ أباهُ<br />والاَبنُ البليدُ حسْرَةٌ لأُمِّهِ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="الأمثال 11:4; الأمثال 12:28; المَزامِير 49:6–10"> كُنوزُ الشَّرِّ لا تُعيلُ العيالَ،<br />والصِّدْقُ يُنقِذُ مِنَ الموتِ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 37:3; ﻣﺘﻰ 6:30–33; المَزامِير 37:25"> الرّبُّ لا يُقاوِمُ رَغبةَ الصِّدِّيقينَ،<br />وأمَّا هوى الأشرارِ فيَصُدُّهُ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الأمثال 21:5; الأمثال 20:4; الأمثال 19:15"> مَنْ عَمِلَ بيدٍ مُرتخيةٍ يَفتَقِرْ،<br />ومنْ عَمِلَ بيدٍ مُجتَهِدَةٍ يَغتَنِ</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="الأمثال 6:8; الأمثال 30:25; الأمثال 17:2"> مَنْ يَكُنْ عاقِلًا يحصُدْ في الصَّيفِ،<br />وأمَّا المَعتوهُ فينامُ في الحصادِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 28:2; الأمثال 28:20; أيُّوب 29:13"> البركاتُ تَغمُرُ رأسَ الصِّدِّيقِ،<br />والعُنْفُ يَملأُ فَمَ الشِّرِّيرِ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 112:6; لُوقا 1:48; مَرْقُس 14:9"> ذِكْرُ الصِّدِّيقِ برَكةٌ،<br />واَسمُ الشِّرِّيرِ يَبلَى.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="ﻣﺘﻰ 7:24–25; الأمثال 12:1; يَعقُوب 3:13"> الحكيمُ القلبِ يقبَلُ الوصايا،<br />والأحمقُ في كَلامِهِ يتَهوَّرُ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الأمثال 28:18; ﻣﺘﻰ 10:26; المَزامِير 26:11–12"> مَنْ يَسلُكْ بنزاهةٍ يَسلُكْ مُطمَئِنًّا<br />ومَنْ يُعوِّجْ طُرُقَهُ يَفتَضِحْ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 35:19; الأمثال 6:13; رسالَةُ يُوحَنّا الثّالثَة 1:10"> مَنْ يَغمِزُ بالعينِ يُثيرُ الغضَبَ،<br />والأحمقُ في كَلامِهِ يتَهوَّرُ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="الأمثال 13:14; أفَسُس 4:29; الأمثال 10:6"> فَمُ الصِّدِّيقِ يُنبوعُ حياةٍ،<br />وفمُ الشِّرِّيرِ يَملأُهُ العُنْفُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:4–7; الأمثال 17:9"> البُغضُ يُثيرُ الخُصوماتِ،<br />والمَحَبَّةُ تستُرُ كُلَّ الذُّنوبِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الأمثال 26:3; الأمثال 6:32; إشَعْياء 50:4"> الحِكمةُ تُوجَدُ في كلامِ الفهيمِ،<br />والعصا لِظَهرِ مَنْ يُعوِزُهُ الفهْمُ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الأمثال 18:7; الأمثال 10:8; الأمثال 13:3"> الحُكَماءُ يَصونونَ المَعرِفةَ،<br />وكلامُ الأحمقِ هلاكٌ قريبٌ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="الأمثال 18:11; الأمثال 19:7; المَزامِير 52:7"> مالُ الغَنيِّ حِصْنُهُ الحَصينُ،<br />وفَقْرُ الفُقراءِ هلاكُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 6:23; إشَعْياء 3:10–11; العِبرانِيّين 6:10"> رِزْقُ الصِّدِّيقِ للحياةِ،<br />وغَلَّةُ الشِّرِّيرِ الخَطيئَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="رسالَةُ بُطرُسَ الثّانيَة 1:5–11; لُوقا 11:28; الأمثال 6:23"> مَنْ حَفِظَ المَشورَةَ فسَبيلُهُ الحياةُ،<br />ومَنْ أهمَلَ التَّوبيخَ فهو ضالٌّ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الأمثال 26:24–26; المَزامِير 55:21; صموئيلَ الثّانِي 3:27"> شفَتا الكَذِبِ تُخفيانِ البُغضَ،<br />والبليدُ يُجاهِرُ بالنَّميمةِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يَعقُوب 1:19; يَعقُوب 3:2; الأمثال 17:27–28"> كثرَةُ الكلامِ لا تَخلو مِنَ الخطأِ،<br />ومَنْ ضبَطَ شفَتَيهِ فهوَ عاقِلٌ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="إرْمِيا 17:9; الأمثال 12:18; التَّكوين 6:5"> لِسانُ الصِّدِّيقِ فِضَّةٌ نقِيَّةٌ،<br />وقلبُ الشِّرِّيرِ لا نَفْعَ فيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="هُوشَع 4:6; الأمثال 17:16; الأمثال 1:29"> أقوالُ الصِّدِّيقِ تحفَظُ الكثيرينَ.<br />وحماقةُ ناقِصِ الفهْمِ تُميتُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 107:38; التَّكوين 26:12; صَمُوئيلَ الأوَّلُ 2:7–8"> برَكَةُ الرّبِّ هيَ الّتي تُغْني،<br />وكَثرةُ التَّعَبِ لا تزيدُ شيئًا.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الأمثال 15:21; الأمثال 2:14; الأمثال 14:9"> البليدُ يَلهو بالرَّذيلَةِ،<br />أمَّا الفهيمُ فبِالحِكمةِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="المَزامِير 145:19; إشَعْياء 66:4; ﻣﺘﻰ 5:6"> ما يَخافُهُ الشِّرِّيرُ يقَعُ علَيهِ،<br />وما يتَمَنَّاهُ الصِّدِّيقُ يَنالُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 15:5; المَزامِير 58:9; ﻣﺘﻰ 7:24–27"> تَعبُرُ الزَّوبعةُ فيَزولُ الشِّرِّيرُ،<br />أمَّا الصِّدِّيقُ فيَسودُ إلى الأبدِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="الأمثال 25:13; الأمثال 26:6; ﻣﺘﻰ 25:26"> كالحامِضِ لِلأسنانِ والدُّخانِ للعينِ<br />كذلِكَ البطَّالُ للّذينَ يُرسلونَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الأمثال 9:11; الأمثال 3:2; المَزامِير 91:16"> مَنْ يَخَفِ الرّبَّ فأيّامُهُ تَطولُ،<br />أمّا سِنُوُّ الأشرارِ فتَقصُرُ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="أيُّوب 8:13; ﺭﻭﻣﻴﺔ 15:13; المَزامِير 73:24–26"> رَجاءُ الصِّدِّيقينَ فرَحٌ لهُم،<br />وأملُ الأشرارِ يَبيدُ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="الأمثال 21:15; ﺭﻭﻣﻴﺔ 2:8–9; فِيلِبِّي 4:13"> الرّبُّ حِصنٌ للصَّالحينَ<br />وهلاكٌ لِفاعِلي الإثْمِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 37:22; المَزامِير 125:1; المَزامِير 37:9–10"> الصِّدِّيقُ لا يضْطَربُ أبدًا،<br />والشِّرِّيرُ لا يَستَقِرُّ في الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="المَزامِير 37:30; الأمثال 10:13; الأمثال 10:20–21"> فَمُ الصِّدِّيقِ يُنبِتُ الحكمةَ،<br />أمَّا لِسانُ الكَذِبِ فيُقطَعُ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="الأمثال 15:28; الأمثال 12:6; الأمثال 11:11"> شفَتا الصِّدِّيق تَبوحانِ بما يُرضي،<br />وفَمُ الشِّرِّيرِ يَنطِقُ بالأكاذيبِ.</span>
</div>
