## نصيحة أم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الأمثال 6:20; الأمثال 1:8; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:5"> هذا كَلامُ لَمُوئيلَ مَلكِ مسَّا، أدَّبَتْهُ بهِ أمُّهُ:</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 49:15; صَمُوئيلَ الأوَّلُ 1:11; صَمُوئيلَ الأوَّلُ 1:28"> ماذا أقولُ لكَ يا اَبني؟<br />ماذا أقولُ يا اَبنَ أحشائي؟<br />يا مَنْ نَذَرْتُ لأجلِهِ نُذوري؟</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 17:17; نَحَمْيا 13:26; الأمثال 7:26–27"> لا تُبَدِّدْ حَيلَكَ على النِّساءِ،<br />ولا على ما يُسَبِّبُ هلاكَ المُلوكِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الأمثال 20:1; الجامِعَة 10:17; إشَعْياء 5:22"> لا يَليقُ بالمُلوكِ، يا لَمُوئيلُ، لا يَليقُ بالمُلوكِ أنْ يَشرَبوا الخمرَ، ولا بالعُظَماءِ أن يُدمِنوا المُسكِرَ،</span>
<span className="v" id="v5" data-v="5" data-xref="حَبَقُّوق 2:5"> لِئلَّا يَسكروا فَيَنسَوا حُقوقَ النَّاسِ، ويُهمِلوا دَعوى المَساكينِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:23; المَزامِير 104:15; صَمُوئيلَ الأوَّلُ 1:10"> أعطِ المُسكِرَ للهالِكينَ، والخمرَ لِأصحابِ النُّفوسِ المُّرَّةِ،</span>
<span className="v" id="v7" data-v="7" data-xref="أفَسُس 5:18"> فيَسكروا ويَنسَوا فَقْرَهُم ولا يَذكُروا تَعاسَتَهُم بَعدُ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 82:3–4; الأمثال 24:11–12; أسْتِير 4:13–16"> إفتَحْ فَمَكَ دِفاعًا عَنِ المُتألِّمينَ، وعَنْ حُقوقِ جميعِ المُهمَلينَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 1:17; اللّاويِّين 19:15; عامُوس 5:11–12"> إفتَحْ فَمَكَ واَحكُمْ بالعَدلِ وأنصِفِ المِسكينَ والبائِسَ.</span>
</div>
</div>

## المرأة القديرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الأمثال 19:14; الأمثال 12:4; الأمثال 18:22"> المَرأةُ القديرةُ مَنْ يَجِدُها؟<br />قيمَتُها تفوقُ اللَّآلئَ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:1–7; المُلُوكِ الثّانِي 4:22–23; المُلُوكِ الثّانِي 4:9–10"> إليها يَطْمَئِنُّ قلبُ زَوجِها،<br />فلا يُعوِزُهُ مَغنَمٌ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="صَمُوئيلَ الأوَّلُ 25:18–22; صَمُوئيلَ الأوَّلُ 25:26–27"> تَحمِلُ لَه الخَيرَ مِنْ دونِ الشّرِّ،<br />طُولَ أيّامِ حياتِها.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="تِيطُس 2:5; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:10; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:10–12"> تَطلُبُ صُوفًا وكتَّانًا<br />وتَعملُ بِيَدَينِ راضيَتَيْنِ،</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الأمثال 31:24; المُلُوكِ الأوَّلُ 9:26–28; حِزْقِيال 27:3–36"> فتكونُ كالسُّفُنِ التِّجاريَّةِ،<br />تَجلِبُ طَعامَها مِنْ بعيدٍ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 12:11; مَرْقُس 1:35; لُوقا 12:42"> تقومُ قبلَ طُلوعِ الفَجرِ<br />لتُهَيِّئَ لأهلِ بَيتِها طَعامًا، ولِخادِماتِها عمَلًا.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 13:44; نَشِيدِ الأنشاد 8:12; يَشُوع 15:18"> تتَأمَّلُ حَقلًا فتَشتَريهِ،<br />وبِثَمَرِ يَدَيها تَغرِسُ كرمًا.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="أيُّوب 38:3; رسالَةُ بُطرُسَ الأُولَى 1:13; أفَسُس 6:10"> تتَزَنَّرُ بالقُوَّةِ وتُشَدِّدُ ذِراعَيها.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:9; ﻣﺘﻰ 25:3–10; المَزامِير 127:2"> تَرَى أنَّ تِجارَتَها رابِحةٌ،<br />فسِراجُها لا يَنطَفِئُ في اللَّيلِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 35:25–26"> تُلقي يَدَيها على المِكَبِّ،<br />وتُمسِكُ بِكفَّيها المِغْزَلَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="أفَسُس 4:28; العِبرانِيّين 13:16; الأمثال 22:9"> تَبسُطُ كفَّيها إلى المِسكينِ،<br />وتمُدُّ يَدَيها إلى البائِسِ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 45:22; صموئيلَ الثّانِي 1:24; الأمثال 25:20"> لا تَخافُ على أهلِ بَيتِها مِنَ الثَّلْجِ،<br />فكُلُّ واحدٍ يَلبَسُ حُلَّتَينِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="رُؤيا يُوحَنّا 19:8; الأمثال 7:16; التَّكوين 41:42"> تَصنَعُ لِنَفْسِها أغطيَةً مُوَشَّاةً،<br />وتَلبَسُ الدِّيباجَ والأرجُوانَ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الأمثال 12:4; راعُوث 4:1; التَّثْنِيَة 16:18"> زَوجُها مَعروفٌ في المَجالِسِ،<br />وهوَ مِنْ مَشايخِ البلدِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="حِزْقِيال 27:16; لُوقا 16:19; الأمثال 31:19"> تَصنَعُ قُمصانًا وتَبيعُها،<br />وتَعرِضُ أحزِمَةً على التَّاجِرِ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="أيُّوب 40:10; ﺭﻭﻣﻴﺔ 13:14; أيُّوب 29:14"> تَلبَسُ العِزَّ والبَهاءَ،<br />وتَضحَكُ على الأيّامِ الآتيةِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="أفَسُس 4:29; الأمثال 16:24; الأمثال 10:31"> تَفتَحُ فَمَها بالحِكمةِ،<br />وفي لِسانِها كَلامُ المَعروفِ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:10; تِيطُس 2:4; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:6"> تُراقبُ سُلوكَ أهلِ بَيتِها،<br />ولا تَأكُلُ خبزَ البِطالَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:15–17; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:5; المُلُوكِ الأوَّلُ 2:19"> يَنهَضُ أولادُها فيُبارِكونَها،<br />وزَوجُها يَمدَحُها فيقولُ:</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="نَشِيدِ الأنشاد 6:8–9; أفَسُس 5:27"> ”نِساءٌ كثيراتٌ أظهَرنَ قُدْرةً،<br />وأنتِ تَفَوَّقْتِ علَيهِنَّ جميعًا“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="الأمثال 1:7; رسالَةُ بُطرُسَ الأُولَى 3:4–5; الأمثال 11:22"> الحُسْنُ غُرورٌ والجَمالُ باطِلٌ،<br />وللمَرأةِ الّتي تَخافُ الرّبَّ كُلُّ المديحِ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="المَزامِير 128:2; العِبرانِيّين 6:10; ﻣﺘﻰ 7:16"> فأعطُوها مِنْ ثمَرِ يَدَيها،<br />ولتَمدَحْها أعمالُها في المَجالِسِ.</span>
</div>
</div>
