## غاية الأمثال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الأمثال 10:1; أخبارِ الأيّامِ الأوّلُ 22:9; أخبارِ الأيّامِ الأوّلُ 28:5"> هذِهِ أمثالُ سُليمانَ بنِ داوُدَ مَلِكِ إِسرائيلَ،</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="الأمثال 7:4; التَّثْنِيَة 4:5–6; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:15–17"> وغايتُها أنْ تُعَرِّفَكَ الحِكمةَ وحُسنَ الرَّأيِ وتُبَيِّنَ لكَ الكَلامَ المُبينَ،</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="الأمثال 2:1–9; أيُّوب 22:22; الأمثال 8:10–11"> وأنْ تُعَلِّمَكَ مَشورَةَ العُقلاءِ<br />والحَقَّ والعَدلَ والاستِقامةَ،</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الأمثال 8:5; الأمثال 8:17; المَزامِير 119:9"> وأنْ تهَبَ الفِتيانَ تعقُّلًا<br />والشَّبابَ معرِفةً وتدبيرًا.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="الأمثال 12:1; الأمثال 9:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:15"> يسمَعُها الحكيمُ فيزدادُ عِلْمًا<br />والفهيمُ فيكتَسِبُ هِدايةً</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 78:2; مَرْقُس 4:11; المَزامِير 49:4"> ويتَبَيَّنُ الأمثالَ وسِرَّ مَعانيها<br />وأقوالَ الحُكماءِ وأحاجيَّهُم،</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="أيُّوب 28:28; المَزامِير 111:10–1; الجامِعَة 12:13"> فرأسُ المعرِفةِ مخافةُ الرّبِّ،<br />والحَمقى يحتَقرونَ الحِكمةَ والفهْمَ.</span>
</div>
</div>

## نصائح للشباب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الأمثال 6:20; الأمثال 30:17; الأمثال 5:1–2"> إسمَعْ، يا اَبني مَشورَةَ أبيكَ<br />ولا تُهمِلْ نَصيحةَ أُمِّكَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="دانيال 5:29; الأمثال 4:9; التَّكوين 41:42"> إكليلُ نِعمةٍ هُما لِرأسِكَ<br />وقلائِدُ بركَةٍ لعُنُقِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="أفَسُس 5:11; الأمثال 16:29; المَزامِير 1:1"> إنْ أغراكَ الخاطِئونَ يا اَبني،<br />فإيَّاكَ أنْ تَقبلَ إغراءَهُم.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 5:26; الأمثال 1:18; الأمثال 12:6"> إنْ قالوا: تعالَ نكمُنْ لِلقَتلِ<br />ونتَرَقَّبْ ضَحيَّةً بريئةً،</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 124:3; المَزامِير 28:1; ﺭﻭﻣﻴﺔ 3:13"> فنبتَلِعُهُم كالهاويةِ وهُم أحياءُ<br />كالقبرِ وهُم هابِطونَ أصِحَّاءُ،</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9–10; لُوقا 12:15; أيُّوب 24:2–3"> فنَحظى بكلِّ شيءٍ نفيسٍ<br />ونَملأَ بُيوتَنا غنائِمَ،</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14"> ويكونَ حَظُّكَ كحظِّنا،<br />ولجميعِنا كيسٌ واحدٌ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="المَزامِير 119:101; المَزامِير 1:1; الأمثال 4:14–15"> لا تَسِرْ حيثُما يَسيرونَ.<br />واَمنَعْ قدَمَيكَ عن سُبُلِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 59:7; الأمثال 6:18; الأمثال 4:16"> خطَواتُهُم تسيرُ إلى الشَّرِّ<br />وتُسْرِعُ إلى إراقةِ الدِّماءِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="أيُّوب 35:11; الأمثال 7:23; إرْمِيا 8:7"> يَنصِبُونَ شِباكَهُم عبَثًا<br />أمامَ عَينَي أيٍّ مِنَ الطُّيورِ</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 9:16; أسْتِير 7:10; الأمثال 5:22–23"> لأنَّ دَمَهُم هوَ الّذي يُراقُ<br />وبِكمينِهِم هُمُ الّذينَ يَهلكونَ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الأمثال 15:27; رسالَةُ بُطرُسَ الثّانيَة 2:3; يَعقُوب 5:1–4"> تِلكَ حالُ الطَّامِعينَ في الكَسْبِ،<br />يَنالونَهُ على حِسابِ حياتِهِم.</span>
</div>
</div>

## نداء الحكمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="الأمثال 9:3; كُولُوسِي 2:3; لُوقا 11:49"> الحِكمةُ تَصيحُ في الشَّوارعِ،<br />وفي السَّاحاتِ تُرسِلُ صوتَها.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 5:20; الأمثال 9:3; ﻣﺘﻰ 13:2"> في رُؤوسِ الأسواقِ تُنادي،<br />وعِندَ مَداخِلِ المدينةِ تقولُ:</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الأمثال 1:29; المَزامِير 1:1; الأمثال 5:12"> إلى متى يَعشقُ الجُهَّالُ الجَهلَ،<br />ويمدَحُ السَّاخِرونَ السُّخريةَ؟<br />إلى متى يُبغِضُ البُلَداءُ المعرِفةَ؟</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="اعمال الرسل 3:19; حِزْقِيال 33:11; حِزْقِيال 18:27–30"> إرجعُوا واَسمَعوا تَوبيخي،<br />فأفيضَ مِنْ روحي علَيكُم<br />وأجعَلَكُم تعرِفونَ كَلامي.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 65:12; ﺭﻭﻣﻴﺔ 10:21; إرْمِيا 7:13"> دَعَوتُكمُ فرَفَضتُم دَعوَتي،<br />ومدَدْتُ يَدي فلم يَلتَفِت أحدٌ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 81:11; لُوقا 7:30; الأمثال 1:30"> طَرَحْتُم كُلَّ عِظَةٍ مِنِّي<br />وأبَيتُم قُبولَ تَوبيخي،</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="المَزامِير 2:4; الأمثال 10:24; المَزامِير 37:13"> لذلِكَ أضحكُ في بَليَّتِكُم<br />وأشمَتُ إنْ حلَّ بِكُم رُعبٌ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الأمثال 3:25–26; لُوقا 21:34–35; المَزامِير 58:9"> وإنْ جاءَ الرُّعبُ كالعاصفةِ<br />وأتَت بَليَّتُكُم كالزَّوبَعةِ<br />وحَلّت بِكُم شِدَّةٌ وضيقٌ</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="ميخا 3:4; يَعقُوب 4:3; إرْمِيا 11:11"> تَدعونَني فلا أُجيبُ،<br />وتُبكِّرونَ إليَّ فلا تَجِدونَني.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="الأمثال 1:22; أيُّوب 21:14–15; الأمثال 6:23"> ولأنَّكُم أبغَضتُم كُلَّ مَعرفةٍ<br />وما اَختَرتُم مخافَةَ الرّبِّ،</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 81:11; الأمثال 1:25; لُوقا 14:18–20"> وأبَيتُم قُبولَ مَشورَتي<br />واَسْتَهَنتُم بكُلِّ توبيخٍ مِنِّي،</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الأمثال 14:14; إرْمِيا 6:19; الأمثال 22:8"> فسَتأكُلونَ ثمَرَةَ ضَلالِكُم<br />وتَشبَعونَ مِن ثمَرِ مَعصيتِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="الأمثال 8:36; إرْمِيا 2:19; يُوحَنّا 3:36"> فالجُهَّالُ يَقتُلُهُم هَذَيانُهُم،<br />والكُسالَى في لَهوِهِم يَبيدونَ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="المَزامِير 25:12–13; المَزامِير 112:7–8; إشَعْياء 26:3"> مَنْ يَسمَعْ لي يسكُنْ مُطمَئِنًّا،<br />ويَأمَنْ ولا يُرعِبُهُ شَرٌّ.</span>
</div>
</div>
