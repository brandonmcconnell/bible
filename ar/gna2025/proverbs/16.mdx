</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="الأمثال 21:2; صَمُوئيلَ الأوَّلُ 16:7; لُوقا 16:15"> سُلوكُ الإنسانِ مُبَرَّرٌ في عَينَيهِ،<br />والرّبُّ يَزِنُ ما في النُّفوسِ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 37:4–5; الأمثال 3:6; المَزامِير 55:22"> فَوِّضْ إلى الرّبِّ أعمالَكَ،<br />فتَتِمُّ جميعُ مَقاصِدِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:9; ﺭﻭﻣﻴﺔ 9:22; إشَعْياء 43:21"> الرّبُّ صنَعَ الجميعَ لِغايةٍ ما،<br />حتّى الشِّرِّيرَ لِسُوءِ المَصيرِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="الأمثال 6:16–17; الأمثال 11:21; الأمثال 8:13"> المُتكبِّرُ يَمقُتُهُ الرّبُّ،<br />وإلى الأبدِ لا عُذْرَ لهُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="الأمثال 14:16; أيُّوب 28:28; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1"> بالرَّحمةِ والحَقِّ تَكفيرُ الذُّنوبِ،<br />وبمَخافَةِ الرّبِّ اَبتِعادٌ عَنِ الشَّرِّ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﺭﻭﻣﻴﺔ 8:31; رسالَةُ يُوحَنّا الأُولَى 3:22; كُولُوسِي 1:10"> يَرضَى الرّبُّ عَنْ سُلوكِ الإنسانِ،<br />فيَجعَلُ حتّى أعداءَهُ يُسالِمُونَه.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الأمثال 15:16; المَزامِير 37:16; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:6–9"> القليلُ القليلُ معَ العَدلِ،<br />ولا الرِّزْقُ الكثيرُ بغيرِ إنصافٍ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الأمثال 19:21; المَزامِير 37:23; الأمثال 16:1"> قلبُ الإنسانِ يَرسُمُ طريقَهُ،<br />والرّبُّ يُثَبِّتُ خَطَواتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 32:1–2; التَّكوين 44:5; التَّكوين 44:15"> القِسمةُ والنَّصيبُ في كَلامِ المَلِكِ،<br />وفَمُهُ لا يَغشُّ في أحكامِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="الأمثال 11:1; الأمثال 20:10; الأمثال 20:23"> للرّبِّ قَبَّانُ العَدلِ وموَازينُهُ،<br />وكُلُّ مِعيارٍ سليمٍ مِعيارُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الأمثال 25:5; الأمثال 29:14; لُوقا 12:48"> فِعْلُ الشَّرِّ يَمقُتُهُ المُلوكُ،<br />لأنَّ العرشَ بالعَدلِ يَثبُتُ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الأمثال 14:35; المَزامِير 101:5–7; الأمثال 22:11"> الكَلامُ الصَّادِقُ يُرضي المُلوكَ،<br />وهُم يُحِبُّونَ القولَ المُستقيمَ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الأمثال 20:2; الأمثال 19:12; الجامِعَة 10:4"> غضَبُ المَلِكِ رسولُ الموتِ،<br />والرَّجلُ الحكيمُ يَستَعطِفُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="المَزامِير 72:6; أيُّوب 29:23–24; هُوشَع 6:3"> في نورِ وجهِ المَلِكِ حياةٌ،<br />ورِضاهُ سَحابُ مطَرٍ ربيعيٍّ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الأمثال 8:10–11; ﻣﺘﻰ 16:26; الأمثال 8:19"> إقتِناءُ الحكمةِ خَيرٌ مِنَ الذَّهَبِ،<br />واَقْتِناءُ الفِطنَةِ أفضَلُ مِنَ الفِضَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 35:8; الأمثال 19:16; الأمثال 10:9"> مَنْ يَسلُكْ مُستَقيمًا يَبتَعِدْ مِنَ الشَّرِّ،<br />ومَنْ يُبْصِرْ طريقَهُ يَحفَظْ نفْسَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الأمثال 11:2; إشَعْياء 2:11–12; الأمثال 18:12"> قَبلَ الخَيبةِ الكِبرياءُ،<br />وقبلَ السُّقوطِ تَشامخُ الرُّوحِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 5:3; إشَعْياء 57:15; المَزامِير 138:6"> تَواضُعُ الرُّوحِ معَ المَساكينِ،<br />ولا اَقْتِسامُ الغَنيمةِ معَ المُتكبِّرينَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="إرْمِيا 17:7–8; المَزامِير 2:12; إشَعْياء 30:18"> المُتَعَقِّلُ في أمرِهِ يَجِدُ خَيرًا،<br />ومنْ يَتَّكِلْ على الرّبِّ يَهنَأْ.</span>
</div>
<div className="bp">

</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الأمثال 13:14; يُوحَنّا 6:63; يُوحَنّا 5:24"> العقلُ يُنبوعُ حياةٍ لِصاحِبِهِ،<br />ومَشورَةُ الأحمقِ حماقةٌ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الأمثال 15:28; ﻣﺘﻰ 12:34–35; كُولُوسِي 3:16"> قلبُ الحكيمِ يُرشِدُ فَمَهُ،<br />ويَزيدُ كَلامَهُ عِلْمًا.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الأمثال 12:18; الأمثال 24:13–14; الأمثال 25:11–12"> الكَلامُ الجميلُ شَهْدُ عسَلٍ،<br />يَحلُو لِلنَّفْسِ ويَشفي العِظامَ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="الأمثال 14:12; الأمثال 12:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:5"> طريقٌ مُستَقيمَةٌ في عَينَيكَ،<br />قد يُؤدِّي آخِرُها إلى الموتِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="الجامِعَة 6:7; الأمثال 14:23; الأمثال 9:12"> جُوعُ العامِلِ يَعمَلُ لِصالِحِهِ،<br />لأنَّ جُوعَهُ يحُثُّهُ على العمَلِ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="يَعقُوب 3:6; الأمثال 6:14; إشَعْياء 5:18"> اللَّئيمُ يَنبُشُ عَنِ السُّوءِ،<br />وكَلامُهُ نارٌ مُتَّقِدَةٌ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="الأمثال 17:9; الأمثال 18:8; الأمثال 15:18"> الكذَّابُ يزرَعُ الخِصامَ،<br />والنَمَّامُ يُفَرِّقُ الأصحابَ.</span>

<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="الأمثال 2:12–15; الأمثال 12:26; الأمثال 1:10–14"> الرَّجلُ العَنيفُ يُغري صاحِبَهُ،<br />ويَقودُهُ في طريقٍ غيرِ صالِحٍ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 13:15; ميخا 7:3; ﻣﺘﻰ 14:7–8"> يُغمِضُ عَينَيهِ لِيَحسُبَ أكاذيبَهُ،<br />ويَعَضُّ شفَتَيهِ وكُلُّهُ سُوءٌ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الأمثال 20:29; اللّاويِّين 19:32; الأمثال 3:1–2"> الشَّيبَةُ إكليلُ فَخرٍ<br />للسَّائرينَ في طريقِ الصَّلاحِ.</span>

<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="يَعقُوب 1:19; الأمثال 15:18; الأمثال 25:28"> البطيءُ الغضَبِ خَيرٌ مِنَ الجَبَّارِ،<br />ومالِكُ نَفسِهِ مِمَّنْ يأخُذُ مدينَةً.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="الأمثال 18:18; ﻳﻮﻧﺎﻥ 1:7; اعمال الرسل 1:26"> القُرعةُ تُلقَى لِتَحكُمَ في الأمورِ،<br />ومِنَ الرّبِّ جميعُ أحكامِها.</span>
</div>
