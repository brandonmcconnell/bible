## المواهب الروحيّة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:4–11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:37; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:1–18"> وأمَّا المَواهِبُ الرُّوحِيَّةُ، أيُّها الإخوَةُ، فلا أُريدُ أنْ تَجهَلوا حَقيقَتَها.</span>
<span className="v" id="v2" data-v="2" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:3; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 1:9; المَزامِير 115:5"> تَعرِفونَ أنَّكُم، عِندَما كُنتُم وثَنيّينَ، كُنتُم تَندَفِعونَ إلى الأوثانِ البُكْمِ على غَيرِ هُدًى.</span>
<span className="v" id="v3" data-v="3" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:2–3; ﻣﺘﻰ 16:16–17; ﺭﻭﻣﻴﺔ 10:9"> أمَّا الآنَ فاَعلَموا أنَّ ما مِنْ أحدٍ إذا ألهَمَهُ رُوحُ اللهِ يَقولُ إنَّ يَسوعَ مَلعونٌ مِنَ اللهِ، ولا يَقدِرُ أحدٌ أنْ يَقولَ إنَّ يَسوعَ رَبٌّ إلَّا بإلهامٍ مِنَ الرُّوحِ القُدُسِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:28; ﺭﻭﻣﻴﺔ 12:4–6"> فالمَواهِبُ الرُّوحِيَّةُ على أنواعٍ، ولكِنَّ الرُّوحَ الّذي يَمنَحُها واحدٌ.</span>
<span className="v" id="v5" data-v="5" data-xref="أفَسُس 4:11–12; ﺭﻭﻣﻴﺔ 12:6–8; ﺭﻭﻣﻴﺔ 14:8–9"> والخِدمَةُ على أنواعٍ، ولكِنَّ الرَّبَّ واحدٌ.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:11; يُوحَنّا 5:17"> والأعمالُ على أنواعٍ، ولكِنَّ اللهَ الّذي يَعمَلُ كُلَّ شيءٍ في الجميعِ واحدٌ.</span>
<span className="v" id="v7" data-v="7" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:10–11; ﺭﻭﻣﻴﺔ 12:6–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:19"> كُلُّ واحدٍ يَنالُ مَوهِبَةً يتَجَلّى فيها الرُّوحُ لِلخَيرِ العامِ.</span>
<span className="v" id="v8" data-v="8" data-xref="الأمثال 2:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:6–10; المَزامِير 143:10"> فهذا يَنالُ مِنَ الرُّوحِ كلامَ الحِكمَةِ، وذاكَ يَنالُ مِنَ الرُّوحِ نَفسِهِ كلامَ المَعرِفَةِ.</span>
<span className="v" id="v9" data-v="9" data-xref="يَعقُوب 5:14–15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:13; أفَسُس 2:8"> والرُّوحُ الواحدُ نَفسُهُ يَهَبُ أحَدَهُمُ الإيمانَ، والآخرَ موهِبَةَ الشِّفاءِ،</span>
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:39; رسالَةُ يُوحَنّا الأُولَى 4:1; مَرْقُس 16:17"> وسِواهُ القُدرَةَ على صُنعِ المُعْجزاتِ، والآخرَ النُّبوءَةَ، وسِواهُ التَّمييزَ بَينَ الأرواحِ، والآخرَ التَّكَلُّمَ بلُغاتٍ مُتَنوِّعَةٍ، والآخرَ تَرجَمَتها.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:4; العِبرانِيّين 2:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:13"> وهذا كُلُّهُ يَعمَلُهُ الرُّوحُ الواحدُ نَفسُهُ مُوزِّعًا مواهِبَهُ على كُلِّ واحدٍ كما يَشاءُ.</span>
</div>
</div>

## جسد واحد وأعضاء كثيرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﺭﻭﻣﻴﺔ 12:4–5; كُولُوسِي 3:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:27"> وكما أنَّ الجَسَدَ واحدٌ ولَه أعضاءٌ كثيرَةٌ هِيَ على كَثْرَتِها جَسَدٌ واحدٌ، فكذلِكَ المَسيحُ.</span>
<span className="v" id="v13" data-v="13" data-xref="أفَسُس 4:5; كُولُوسِي 3:11; غَلاطِيَّة 3:28"> فنَحنُ كُلُّنا، أيَهودًا كُنَّا أم غَيرَ يَهودٍ، عبيدًا أم أحرارًا، تَعَمَّدنا بِرُوحٍ واحدٍ لِنكونَ جَسَدًا واحدًا، واَرتَوَيْنا مِنْ رُوحٍ واحدٍ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:27–28; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:12; أفَسُس 4:25"> وما الجَسَدُ عُضوًا واحدًا، بَلْ أعضاءٌ كثيرةٌ.</span>
<span className="v" id="v15" data-v="15" data-xref="القُضاة 9:8–15; المُلُوكِ الثّانِي 14:9"> فَلَو قالَتِ الرِّجْلُ: ”ما أنا يَدًا، فَما أنا مِنَ الجَسَدِ؟</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 12:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:22; فِيلِبِّي 2:3"> ولَو قالَتِ الأُذُنُ: ”ما أنا عينًا، فما أنا مِنَ الجَسَدِ“، أتَبطُلُ أنْ تكونَ عُضوًا في الجَسَدِ؟</span>
<span className="v" id="v17" data-v="17" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:21; المَزامِير 139:13–16; الأمثال 20:12"> فلَو كانَ الجَسَدُ كُلُّهُ عينًا، فأينَ السَّمعُ؟ ولَو كانَ الجَسَدُ كُلُّهُ أذُنًا، فأينَ الشَّمُّ؟</span>
<span className="v" id="v18" data-v="18" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:28; ﺭﻭﻣﻴﺔ 12:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:11"> ولكِنَّ اللهَ جعَلَ كُلَّ عُضوٍ في الجَسَدِ كما شاءَ.</span>
<span className="v" id="v19" data-v="19"> فَلَو كانَت كُلُّها عُضوًا واحدًا فأينَ الجَسَدُ؟</span>
<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:12"> ولكِنَّ الأعضاءَ كثيرةٌ والجَسَد واحدٌ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="عَزْرا 10:1–5; نَحَمْيا 4:16–21; أيُّوب 29:11"> فلا تَقدِرُ العينُ أن تَقولُ لِليَدِ: ”لا أحتاجُ إلَيكِ“. ولا الرأْسُ لِلرِّجلَينِ: ”لا أحتاجُ إلَيكُما!“</span>
<span className="v" id="v22" data-v="22" data-xref="الأمثال 14:28; الجامِعَة 9:14–15; الجامِعَة 4:9–12"> فما نَحسبهُ أضعفَ أعضاءِ الجَسَدِ هوَ ما كانَ أشَدَّها ضرورَةً،</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 3:21; التَّكوين 3:7"> وما نَحسبهُ أقَلَّها كَرامَةً هوَ الّذي نَخُصُّهُ بِمَزيدٍ مِنَ التَّكريمِ، وما نَستَحي بِه هوَ الّذي نَخُصُّهُ بِمَزيدٍ مِنَ الوَقارِ.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 3:11; التَّكوين 2:25"> أمَّا الأعضاءُ الكريمَةُ، فلا حاجَةَ بِها إلى ذلِكَ. ولكِنَّ اللهَ صنَعَ الجَسَدَ بِطريقةٍ تَزيدُ في كرامةِ الأعضاءِ الّتي بِلا كرامةٍ،</span>
<span className="v" id="v25" data-v="25" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:10–12; يُوحَنّا 17:21–26"> لِئَلَّا يَقعَ في الجَسَدِ شِقاقٌ، بَلْ لِتَهتَمَّ الأعضاءُ كُلُّها بَعضُها بِبَعضٍ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﺭﻭﻣﻴﺔ 12:15; غَلاطِيَّة 6:2; العِبرانِيّين 13:3"> فإذا تألَّمَ عُضوٌ تألَّمَت معَهُ جميعُ الأعضاءِ، وإذا أُكرِمَ عُضوٌ فَرِحَتْ معَهُ سائِرُ الأعضاءِ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="أفَسُس 4:12; كُولُوسِي 1:24; أفَسُس 5:30"> فأنتُم جَسَدُ المَسيحِ، وكُلُّ واحدٍ مِنكُم عُضوٌ مِنهُ.</span>
<span className="v" id="v28" data-v="28" data-xref="أفَسُس 4:11–13; ﺭﻭﻣﻴﺔ 12:6–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:18"> واللهُ أقامَ في الكنيسةِ الرُّسُلَ أوّلًا والأنبياءَ ثانيًا والمُعَلِّمينَ ثالثًا، ثُمَّ منَحَ آخَرينَ القُدرَةَ على صُنعِ المُعجِزاتِ ومَواهِبَ الشِّفاءِ والإسعافِ وحُسْنِ الإدارَةِ والتَّكَلُّمِ بِلُغاتٍ مُتنوِّعَةٍ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:4–11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:14–20"> فهَلْ كُلُّهُم رُسُلٌ وكُلُّهُم أنبياءُ وكُلُّهُم مُعَلِّمونَ وكُلُّهُم يَصنَعونَ المُعجِزاتِ</span>
<span className="v" id="v30" data-v="30" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:10"> وكُلُّهُم يَملِكونَ مَوهِبَةَ الشِّفاءِ وكُلُّهُم يَتكَلَّمونَ بِلُغاتٍ وكُلُّهُم يُتَرجِمونَ؟</span>
<span className="v" id="v31" data-v="31" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:39; ﻣﺘﻰ 5:6"> فاَرغَبوا في المواهِبِ الحُسنى، وأنا أَدُلُّكُم على أفضَلِ الطُّرُقِ.</span>
