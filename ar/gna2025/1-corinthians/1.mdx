## سلام وصلاة شكر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﺭﻭﻣﻴﺔ 1:1; أفَسُس 1:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:1"> مِنْ بولُسَ الّذي شاءَ اللهُ أنْ يَدعُوَهُ ليكونَ رَسولَ المَسيحِ يَسوعَ، ومِنَ الأخِ سُوسْتانيسَ،</span>
<span className="v" id="v2" data-v="2" data-xref="ﺭﻭﻣﻴﺔ 1:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:9–11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:30"> إلى كَنيسةِ اللهِ في كُورِنثوسَ، إلى الّذينَ قَدَّسَهُمُ اللهُ في المَسيحِ يَسوعَ ودَعاهُم ليكونوا قِدِّيسينَ معَ جميعِ الّذينَ يَدْعُونَ في كُلِّ مكانٍ باَسمِ ربِّنا يَسوعَ المَسيحِ، رَبِّهِم ورَبِّنا.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="ﺭﻭﻣﻴﺔ 1:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:2; رسالَةُ بُطرُسَ الأُولَى 1:2"> علَيكُمُ النِّعمةُ والسَّلامُ مِنَ اللهِ أبينا ومِنَ الرَّبِّ يَسوعَ المَسيحِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="ﺭﻭﻣﻴﺔ 1:8; يُوحَنّا 14:26; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:14"> أشكُرُ إلهي لأجلِكُم دومًا على النِّعمَةِ الّتي وهَبَها لكم في المَسيحِ يَسوعَ،</span>
<span className="v" id="v5" data-v="5" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:7; كُولُوسِي 2:3"> فَصِرتُم بِه أغنياءَ في كُلِّ شيءٍ، في أساليبِ الكلامِ وأنواعِ المَعرِفَةِ.</span>
<span className="v" id="v6" data-v="6" data-xref="رُؤيا يُوحَنّا 1:2; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:8; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:6"> على قَدْرِ ما رسَخَتْ فيكُم شَهادةُ المَسيحِ،</span>
<span className="v" id="v7" data-v="7" data-xref="فِيلِبِّي 3:20; رسالَةُ بُطرُسَ الثّانيَة 3:12; العِبرانِيّين 9:28"> حتّى إنَّهُ لا تُعوزُكُم موهِبَةٌ مِنَ المَواهبِ، وأنتُم تَنتَظِرونَ ظُهورَ رَبِّنا يَسوعَ المَسيحِ،</span>
<span className="v" id="v8" data-v="8" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:23–24; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 3:13; كُولُوسِي 1:22"> وهوَ الّذي يَحفَظُكم ثابِتينَ إلى النِّهايَةِ حتّى لا يكونَ علَيكُم لومٌ في يومِ رَبِّنا يَسوعَ المَسيحِ.</span>
<span className="v" id="v9" data-v="9" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:3; العِبرانِيّين 10:23; التَّثْنِيَة 7:9"> أمينٌ هوَ اللهُ الّذي دَعاكُم إلى شَرِكةِ اَبنِهِ يَسوعَ المَسيحِ رَبِّنا.</span>
</div>

## خلاف بين المؤمنين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="فِيلِبِّي 2:1–4; أفَسُس 4:1–7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:25"> أناشِدُكُم أيُّها الإخوةُ، باَسمِ ربِّنا يَسوعَ المَسيحِ، أنْ تكونوا جميعًا مُتَّفِقينَ في الرَّأيِ وأنْ لا يكونَ بَينَكُم خِلافٌ، بَلْ كونوا على وِفاقٍ تامٍ، لكُم رُوحٌ واحدٌ وفِكرٌ واحدٌ.</span>
<span className="v" id="v11" data-v="11" data-xref="الأمثال 18:6; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:4; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:23–25"> فأهلُ بَيتِ خُلُوَةَ أخبَروني أيُّها الإخوةُ أنَّ بَينَكُم خِلافًا،</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:21–23; يُوحَنّا 1:42; ﻣﺘﻰ 23:9–10"> أعني أنَّ كُلَّ واحدٍ مِنكُم يَقولُ ”أنا معَ بولسَ“ و”أنا معَ أبُلُّوسَ“ و”أنا معَ بُطرُسَ“ و”أنا معَ المَسيحِ“.</span>
<span className="v" id="v13" data-v="13" data-xref="أفَسُس 4:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:4; اعمال الرسل 2:38"> فهَلِ المَسيحُ اَنقَسَمَ؟ هَلْ بولُسُ هوَ الّذي صُلِبَ مِنْ أجلِكُم؟ أو باَسمِ بولُسَ تَعَمَّدتُم؟</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 18:8; ﺭﻭﻣﻴﺔ 16:23; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:4"> أشكُرُ اللهَ على أنِّي ما عَمَّدتُ أحدًا مِنكُم غيرَ كِريسبُسَ وغايُسَ،</span>
<span className="v" id="v15" data-v="15" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:2; يُوحَنّا 3:28–29; يُوحَنّا 7:18"> فلا يَقدِرُ أحَدٌ أنْ يقولَ إنَّكُم باَسمي تَعمَّدتُم.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:17; اعمال الرسل 16:33"> نعم، عَمَّدتُ أيضًا عائِلَةَ اَستفاناسَ. وما عَدا هَؤُلاءِ، فلا أذكُرُ أنِّي عَمَّدتُ أحدًا.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:13; رسالَةُ بُطرُسَ الثّانيَة 1:16"> فالمَسيحُ أرسَلَني لا لأُعَمِّدَ، بَلْ لأُعلِنَ البِشارَةَ غَيرَ مُتَّكِلْ على حكِمَةِ الكلامِ لِئَلَّا يَفقُدَ موتُ المَسيحِ على الصَّليبِ قوَّتَهُ.</span>
</div>
</div>

## المسيح قدرة الله وحكمته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14; ﺭﻭﻣﻴﺔ 1:16; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:10"> فالبِشارةُ بالصَّليبِ ”حماقَةٌ“ عِندَ الّذينَ يَسلُكونَ طريقَ الهلاكِ. وأمَّا عِندَنا نَحنُ الّذينَ يَسلُكونَ طريقَ الخلاصِ، فهوَ قُدرَةُ اللهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 29:14; إرْمِيا 8:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:19"> فالكِتابُ يقولُ: ”سأمحو حِكْمَةَ الحُكَماءِ، وأُزيلُ ذَكاءَ الأذكياءِ“.</span>
<span className="v" id="v20" data-v="20" data-xref="ﺭﻭﻣﻴﺔ 1:22; أيُّوب 12:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:19"> فأينَ الحكيمُ؟ وأينَ العَلَّامةُ؟ وأينَ المُجادِلُ في هذا الزَّمانِ؟ أما جعَلَ اللهُ حِكمَةَ العالَمِ حماقةً؟</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="لُوقا 10:21; ﺭﻭﻣﻴﺔ 1:20–22; ﻣﺘﻰ 11:25"> فلَمَّا كانَت حِكمَةُ اللهِ أنْ لا يَعرِفَهُ العالَمُ بالحِكمَةِ، شاءَ اللهُ أنْ يُخلِّصَ المُؤمنينَ بِه ”بِحماقَةِ“ البِشارَة.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 16:1–4; ﻣﺘﻰ 12:38–39; مَرْقُس 8:11"> وإذا كانَ اليَهودُ يَطلُبونَ المُعجزاتِ، واليونانِيّونَ يَبحَثونَ عَنِ الحِكمَةِ،</span>
<span className="v" id="v23" data-v="23" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:2; رسالَةُ بُطرُسَ الأُولَى 2:8"> فنَحنُ نُنادي بِالمسيحِ مَصلُوبًا، وهذا عَقبَةٌ لليَهودِ وحماقَةٌ في نظَرِ الوثنيِّينَ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:30; كُولُوسِي 2:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:18"> وأمَّا لِلَّذينَ دَعاهُمُ اللهُ مِنَ اليَهودِ واليونانيّينَ، فالمَسيحُ هوَ قُدرَةُ اللهِ وحِكمَةُ اللهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﺭﻭﻣﻴﺔ 11:33–36; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:18; زَكَريّا 4:6–7"> فما يَبدو أنَّهُ حماقَةٌ مِنَ اللهِ هوَ أحكمُ مِنْ حِكمَةِ النَّاسِ، وما يَبدو أنَّهُ ضُعفٌ مِنَ اللهِ هوَ أقوَى مِنْ قُوَّةِ النَّاسِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="يَعقُوب 2:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:18–20; يَعقُوب 1:9–11"> تذَكَّروا أيُّها الإخوةُ كيفَ كُنتُم حينَ دَعاكُمُ اللهُ، فما كانَ فيكُم كثيرٌ مِنَ الحُكماءِ بِحكمَةِ البشَرِ ولا مِنَ الأقوِياءِ أو الوُجَهاءِ.</span>
<span className="v" id="v27" data-v="27" data-xref="يَعقُوب 2:5; ﻣﺘﻰ 11:25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:20"> إلَّا أنَّ اللهَ اَختارَ ما يَعتَبِرُهُ العالَمُ حماقَةً ليُخزِيَ الحكماءَ، وما يعتبرُهُ العالمُ ضُعفًا ليُخزِيَ الأقوِياءَ.</span>
<span className="v" id="v28" data-v="28" data-xref="أيُّوب 34:24; ﺭﻭﻣﻴﺔ 4:17; إشَعْياء 2:11"> واَختارَ اللهُ ما يَحتَقِرُهُ العالَمُ ويَزدَريهِ ويَظُنُّهُ لا شيءَ، لِيُزيلَ ما يَظُنُّهُ العالَمُ شيئًا،</span>
<span className="v" id="v29" data-v="29" data-xref="ﺭﻭﻣﻴﺔ 3:27; إرْمِيا 9:23; أفَسُس 2:9"> حتّى لا يَفتَخِرَ بشَرٌ أمامَ اللهِ.</span>
<span className="v" id="v30" data-v="30" data-xref="أفَسُس 1:7; فِيلِبِّي 3:9; ﺭﻭﻣﻴﺔ 8:1"> وأمَّا أنتُم، فَبِفَضلِه صِرتُم في المَسيحِ يَسوعَ الّذي هوَ لَنا مِنَ اللهِ حِكمَةً وبِرًّا وقَداسَةً وفِداءً،</span>
<span className="v" id="v31" data-v="31" data-xref="إرْمِيا 9:23–24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:17; فِيلِبِّي 3:3"> كما جاءَ في الكِتابِ: ”مَنْ أرادَ أنْ يَفتَخِرَ، فلْيَفتَخِرْ بالرَّبِّ“.</span>
