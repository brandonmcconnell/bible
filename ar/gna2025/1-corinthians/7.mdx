## الزواج والبتولية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:26–27; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:37–38"> وأمَّا مِنْ جِهَةِ ما كَتَبتُم بِه إليَّ، فخَيرٌ لِلرَّجُلِ أنْ لا يَمَسَّ اَمرَأةً.</span>
<span className="v" id="v2" data-v="2" data-xref="أفَسُس 5:28; أفَسُس 5:33; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:9"> ولكِنْ، خَوفًا مِنَ الزِّنى، فلْيكُنْ لِكُلِّ رَجُلٍ اَمرأتُهُ ولِكُلِّ اَمرأةٍ زَوجُها،</span>
<span className="v" id="v3" data-v="3" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:7; الخُرُوج 21:10"> وعلى الزَّوجِ أنْ يوفيَ اَمرأتَهُ حَقَّها، كما على المرأةِ أنْ توفيَ زَوجَها حَقَّهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="هُوشَع 3:3; ﻣﺘﻰ 19:9; مَرْقُس 10:11–12"> لا سُلطَةَ لِلمرأةِ على جَسِدَها، فهوَ لِزَوجِها. وكذلِكَ الزَّوجُ لا سُلطَةَ لَه على جَسَدِهِ، فهوَ لاَمرأتِهِ.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 19:15; يُوئيل 2:16; زَكَريّا 12:12–14"> لا يَمتَنِعْ أحدُكُما عَنِ الآخَرِ إلَّا على اَتفاقٍ بَينَكما وإلى حينٍ، حتّى تتَفَرَّغا لِلصَّلاةِ. ثُمَّ عودا إلى الحياةِ الزَّوجِيَّةِ العادِيَّةِ لِئَلَّا يُعوزَكُم ضَبطُ النَّفسِ، فتَقَعوا في تَجرِبَةِ إبليسَ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:12"> أقولُ لكُم هذا لا على سَبيلِ الأمْرِ، بَلْ على سَبيلِ السَّماحِ،</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 19:11–12; ﺭﻭﻣﻴﺔ 12:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:5"> فأنا أتَمنّى لَو كانَ جميعُ النَّاسِ مِثْلي. ولكِنْ لِكُلِّ إنسانٍ هِبَةٌ خَصَّهُ اللهُ بِها، فبَعضُهُم هذِهِ وبَعضُهُم تِلكَ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:32; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:34–35; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:26–27"> وأقولُ لِغَيرِ المُتَزوِّجينَ والأرامِلِ إنَّهُ خَيرٌ لهُم أنْ يَبْقَوا مِثْلي.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:2; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:14; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:11"> أمَّا إذا كانوا غَيرَ قادِرينَ على ضَبطِ النَّفسِ، فَليَتزوَّجوا. فالزَّواجُ أفضَلُ مِنَ التَّحَرُّقِ بالشَّهوَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 5:32; لُوقا 16:18; ﻣﺘﻰ 19:3–9"> وأمَّا المُتَزوِّجونَ فوَصيَّتي لهُم، وهيَ مِنَ الرَّبِّ لا مِنِّي، أنْ لا تُفارِقَ المرأةُ زَوجَها،</span>
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 3:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:10; إشَعْياء 50:1"> وإنْ فارَقَتْهُ، فلْتَبقَ بِغَيرِ زَوجٍ أو فَلتُصالِحْ زَوجَها، وعلى الزَّوجِ أنْ لا يُطَلِّقَ اَمرأتَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:25"> وأمَّا الآخَرونَ، فأقولُ لهُم أنا لا الرَّبُّ: إذا كانَ لأخٍ مُؤمنٍ اَمرَأةٌ غَيرُ مُؤمِنَةٍ رَضِيَتْ أنْ تَعيشَ معَهُ، فَلا يُطَلِّقْها.</span>
<span className="v" id="v13" data-v="13"> وإذا كانَ لاَمرأةٍ مُؤمِنَةٍ زَوجٌ غَيرُ مُؤمِنٍ يَرضى أنْ يَعيشَ مَعَها، فَلا تُطَلِّقْهُ.</span>
<span className="v" id="v14" data-v="14" data-xref="مَلاخِي 2:15–16; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 4:5; تِيطُس 1:15"> فالزَّوجُ غَيرُ المُؤمِنِ يتَقَدَّسُ باَمرأتِهِ المُؤمِنَةِ، والمرأةُ غَيرُ المُؤمِنَةِ تتَقَدَّسُ بِزَوجِها المُؤمِنِ، وإلَّا كانَ أولادُكُم أنجاسًا، معَ أنَّهُم مُقَدَّسونَ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 14:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:11; يَعقُوب 3:17–18"> وإنْ أرادَ غَيرُ المُؤمِنِ أو غَيرُ المُؤمِنَةِ أنْ يُفارِقَ فليُفارِقْ، ففي مِثلِ هذِهِ الحالِ لا يكونُ المُؤمِنُ أوِ المُؤمِنَةُ خاضِعَينِ لِرِباطِ الزَّواجِ، لأنَّ اللهَ دَعاكُم أنْ تَعيشوا بِسَلامٍ.</span>
<span className="v" id="v16" data-v="16" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:1–2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:22; ﺭﻭﻣﻴﺔ 11:14"> فكَيفَ تَعلَمينَ أيَّتُها المرأةُ المُؤمِنَةُ أنَّكِ سَتُخَلِّصينَ زَوجَكِ؟ وكيفَ تَعلَمُ أيُّها الرَّجُلُ المُؤمِنُ أنَّكَ سَتُخلِّصُ زَوجَتَكَ؟</span>
</div>

## بقاء المؤمن على حاله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:17; ﺭﻭﻣﻴﺔ 12:3–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:24"> فلْيَسلُكْ كُلُّ واحِدٍ في حَياتِهِ حسَبَ ما قسَمَ لَه الرَّبُّ وكما كانَت علَيهِ حالُهُ عِندَما دَعاهُ اللهُ، وهذا ما أفرِضُهُ في الكنائِسِ كُلِّها.</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 15:5; اعمال الرسل 15:28; اعمال الرسل 15:24"> فمَنْ دَعاهُ اللهُ وهوَ مَختونٌ، فَلا يُحاوِل أنْ يَستُرَ خِتانَهُ، ومَنْ دَعاهُ الرَّبُّ وهوَ غير مَختونٍ، فَلا يَختَتِنْ.</span>
<span className="v" id="v19" data-v="19" data-xref="غَلاطِيَّة 6:15; غَلاطِيَّة 5:6; رسالَةُ يُوحَنّا الأُولَى 2:3–4"> لا الخِتانُ لَه مَعنى ولا عَدَمُ الخِتانِ، بَلِ الخَيرُ كُلُّ الخَيرِ في العمَلِ بوَصايا اللهِ.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:17; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:11; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:12"> فَعَلى كُلِّ واحِدٍ أنْ يَبْقى مِثلَما كانَت علَيهِ حالُهُ عِندَما دَعاهُ اللهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:18–24; كُولُوسِي 3:11; لُوقا 10:40–41"> فإنْ كُنتَ عَبدًا عِندَما دَعاكَ اللهُ فلا تَهتَمَّ. ولكِنْ إنْ كانَ بإمكانِكَ أنْ تَصيرَ حُرًّا، فالأَولى بِكَ أنْ تَغتَنِمَ الفُرصَةَ.</span>
<span className="v" id="v22" data-v="22" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:16; فِلِيْمُون 1:16; كُولُوسِي 3:22–24"> فمَنْ دَعاهُ الرَّبُّ وهوَ عَبدٌ كانَ لِلرَّبِّ حُرًّا، وكذلِكَ مَنْ دَعاهُ المَسيحُ وهوَ حُرٌّ كانَ لِلمَسيحِ عَبدًا.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:20; رسالَةُ بُطرُسَ الأُولَى 1:18–19; تِيطُس 2:14"> واللهُ اَشتَراكُم ودَفَعَ الثَّمَنَ، فلا تَصيروا عَبيدًا لِلنَّاسِ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:20; كُولُوسِي 3:23–24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:17"> فليَبقَ كُلُّ واحِدٍ مِنكُم، أيُّها الإخوةُ، أمامَ اللهِ مِثلَما كانَت علَيهِ حالُهُ عِندَما دَعاهُ.</span>
</div>
</div>

## غير المتزوجين والأرامل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:40"> وأمَّا غَيرُ المُتزوِّجينَ فَلا وَصِيَّةَ لهُم عِندي مِنَ الرَّبِّ، ولكِنِّي أُعطي رَأْيي كرَجُلٍ جَعَلَتْه رَحمَةُ الرَّبِّ مَوضِعَ ثِقَةٍ،</span>
<span className="v" id="v26" data-v="26" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:8; رسالَةُ بُطرُسَ الأُولَى 4:17"> فأقولُ إنَّهُ مِنَ الخَيرِ، نظَرًا إلى ما في الوَقتِ الحاضِرِ مِنْ ضيقٍ، أنْ يَبقى الإنسانُ على حالِهِ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:12–14"> هَلْ أنتَ مُقتَرِنٌ باَمرأةٍ؟ إذًا، لا تَطْلُبِ الانفِصالَ عَنها. هَل أنتَ غيرُ مُقتَرِنٍ باَمرأةٍ؟ إذًا، لا تَطلُبِ الزَّواجَ باَمرأةٍ،</span>
<span className="v" id="v28" data-v="28" data-xref="العِبرانِيّين 13:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:32–36; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:26"> وإذا تَزَوَّجْتَ فأنتَ لا تُخطِئُ، ولكِنَّ الّذينَ يَتَزوَّجونَ يَجدِونَ مَشقَّةً في هُمومِ الحياةِ، وأنا أُريدُ أن أُبعِدَها عَنكُم.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:31; ﺭﻭﻣﻴﺔ 13:11–12"> أقولُ لكُم، أيُّها الإخوةُ، إنَّ الزَّمانَ يَقصُرُ. فلْيكُنِ الّذينَ لهُم نِساءٌ كأنَّ لا نِساءَ لهُم،</span>
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 30:5; لُوقا 6:25; لُوقا 6:21"> والّذينَ يَبكونَ كأنَّهُم لا يَبكونَ، والّذينَ يَفرَحونَ كأنَّهُم لا يَفرَحونَ، والّذينَ يَشتَرونَ كأنَّهُم لا يَملِكونَ،</span>
<span className="v" id="v31" data-v="31" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:17; رسالَةُ بُطرُسَ الأُولَى 4:7; يَعقُوب 4:14"> والّذينَ يَتعاطَوْنَ أُمورَ هذا العالَمِ كأنَّهُم لا يَتعاطَوْنَ، لأنَّ صورَةَ هذا العالَمِ في زَوالٍ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 6:25–34; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:5; فِيلِبِّي 4:6"> أُريدُ أن تكونوا مِنْ دُونِ هَمٍّ. فغَيرُ المُتَزوِّجِ يَهتَّمُ بأمورِ الرَّبِّ وكيفَ يُرضي الرَّبَّ،</span>
<span className="v" id="v33" data-v="33" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:7; كُولُوسِي 3:19; أفَسُس 5:25–33"> والمُتَزَوِّجُ يَهتَمُّ بأمورِ العالَمِ وكيفَ يُرضي اَمرأتَهُ،</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 10:40–42; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:20; لُوقا 2:36–37"> فهوَ مُنقَسِمٌ. وكذلِكَ العَذراءُ والمرأةُ الّتي لا زَوجَ لها تَهتَمّانِ بأُمورِ الرَّبِّ وكيفَ تَنالانِ القَداسَةَ جسَدًا ورُوحًا، وأمَّا المُتَزوِّجةُ فتَهتَمُّ بأُمورِ العالَمِ وكيفَ تُرضي زَوجَها.</span>
<span className="v" id="v35" data-v="35" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:33–34; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:36; لُوقا 8:14"> أقولُ هذا لِخَيرِكُم، لا لأُلقيَ علَيكُم قَيدًا، بَلْ لِتَعمَلوا ما هوَ لائِقٌ وتَخدُموا الرَّبَّ مِنْ دونِ اَرتِباكٍ.</span>
