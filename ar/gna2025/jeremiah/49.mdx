## حكم الرّبّ على بني عمون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="عامُوس 1:13–15; صَفَنْيا 2:8–11; حِزْقِيال 25:2–10"> هذا ما قالَ الرّبُّ على بَني عَمُّونَ: ”أمَا لِشعب إِسرائيلَ بَنونَ، أو لا وارِثَ لهُم؟ فما بالُ شعبِ الإلهِ مَلكومَ وَرِثَ أرضَ جادَ وسكَنَ في مُدُنِها</span>
<span className="v" id="v2" data-v="2" data-xref="إرْمِيا 4:19; حِزْقِيال 21:20; التَّثْنِيَة 3:11"> لذلِكَ تأتي أيّامٌ، يقولُ الرّبُّ، أسمَعُ فيها هُتافَ القِتالِ في رَبَّةَ عاصِمةِ بَني عَمُّونَ، فتَصيرُ تَلًا مِنَ الأنقاضِ، وتُحرَقُ قُراها بِالنَّارِ، ويَسلُبُ بَنو إِسرائيلَ سَالِبيهِم، يقولُ الرّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="إرْمِيا 48:7; إرْمِيا 48:37; إشَعْياء 15:2"> وَلْوِلي يا حَشبونُ لأنَّ عايَ دُمِّرَت. أصرُخنَ يا بَناتِ رَبَّةَ وتَحَزَّمْنَ بِالمُسوحِ واَندُبْنَ وطُفْنَ بَينَ الأسيجَةِ، لأنَّ الإلهَ مَلكومَ يَذهَبُ إلى السَّبْي، هوَ وكَهنَتُهُ ورُؤساؤُهُ جميعًا.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:17; إرْمِيا 9:23; إرْمِيا 21:13"> ما بالُكِ تَفتَخرينَ بالأوديَةِ، وواديكِ يَفيضُ دَمًا أيَّتُها الابنَةُ العاصيَةُ، المُتَوكِّلَةُ على كُنوزِها، القائِلَةُ: مَنْ يَهجُمُ عليَّ؟</span>
<span className="v" id="v5" data-v="5" data-xref="إرْمِيا 46:5; إرْمِيا 49:29; إرْمِيا 15:8"> سأَجلِبُ علَيكِ الرُّعْبَ مِنْ حَوالَيكِ يقولُ السَّيِّدُ الرّبُّ القديرُ، فيُطْرَدُ أَهلُكِ ويَهيمُ كُلُّ واحدٍ على وجهِهِ ولا أحدَ يَجمَعُ الشَّارِدينَ.</span>
<span className="v" id="v6" data-v="6" data-xref="إرْمِيا 49:39; إرْمِيا 48:47; حِزْقِيال 16:53"> وبَعدَ ذلِكَ أُعيدُ أمجادَ بَني عَمُّونَ يقولُ الرّبُّ“.</span>
</div>
</div>

## حكم الرّبّ على أدوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="عامُوس 1:11–12; إرْمِيا 25:21; التَّكوين 36:15"> وقالَ الرّبُّ القديرُ على أدومَ: ”أمَا مِنْ حِكمَةٍ بَعدُ في تيمانَ؟ هل نَفَدَ الفَهْمُ في بَنيها وزالت حِكمَتُهُم؟</span>
<span className="v" id="v8" data-v="8" data-xref="إرْمِيا 25:23; إرْمِيا 46:21; إرْمِيا 49:30"> أُهرُبوا. وَلُّوا. إختَبِئوا يا سُكَّانَ دَدانَ! سأجلِبُ النَّكبَةَ على الأدوميِّينَ بَني عيسو لأنَّ وقتَ عِقابِهِم حانَ.</span>
<span className="v" id="v9" data-v="9" data-xref="عُوبَدْيا 1:5–6; إشَعْياء 17:6"> لو جاءَكِ القاطِفونَ يا أدومُ لكانوا تَركوا في الكَرْمِ بَقيَّةً، أوِ اللُّصوصُ ليلًا لكانوا قَنَعوا بِنَهْبِ ما يكفيهِم.</span>
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 17:14; مَلاخِي 1:3–4; إشَعْياء 45:3"> أمَّا أنا فأُعرِّي بَني عيسو وأَكشِفُ خَفاياهُم فلا يَقدِرونَ أنْ يَختَبِئوا. وأُدَمِّرُهُم.هُم وذُرِّيَّتُهُم وأنسِباؤُهُم وجيرانُهُم، حتّى لا يَبقى مِنهُم أحدٌ.</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 68:5; هُوشَع 14:3; يَعقُوب 1:27"> أُترُكْ أيتامَكَ يا أدومُ فأنا أحييهِم، ولتَتَوكَّلْ أرامِلُكَ عليَّ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 25:28–29; عُوبَدْيا 1:16; إرْمِيا 25:15"> وقالَ الرّبُّ: ”الّذينَ لا يَستَحِقُّونَ أنْ يَشرَبوا كأسَ العِقابِ شَرِبوها الآنَ، فلِماذا أُبَرِّئُكَ أنتَ يا أدومُ؟ لا أُبرِّئك، بل تَشرَبُها أنتَ أيضًا،</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 22:16; إشَعْياء 34:6; التَّكوين 36:33"> لأنِّي بذاتي أقسَمْتُ، يقولُ الرّبُّ: تَصيرُ بُصرَةُ العاصِمةُ مَثارَ رُعْبٍ وعارًا وخَرابًا ولَعنَةً، وتَصيرُ جميعُ قُراها خرائِبَ أبديَّةً“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 30:4; إشَعْياء 37:7; حِزْقِيال 7:25–26"> سَمِعْتُ نَبَأً مِنَ الرّبِّ حَمَلَهُ سفيرٌ إلى الأمَمِ: إجتَمِعوا واَهجُموا على أدومَ واَنهَضوا لِلقِتالِ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 1:51; عُوبَدْيا 1:2; ميخا 7:10"> سأجعَلُكِ صغيرةً في الأمَمِ، حقيرةً بَينَ البشَرِ.</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 49:25; عامُوس 9:2; أيُّوب 39:27"> ضَلَّلَكِ بَطشُكِ وكِبرِياءُ قلبِكِ، أنتِ أيَّتُها السَّاكِنَةُ في شُقوقِ الصَّخرِ بِأعالي الجِبالِ، ومهما تُعَلِّينَ عِشَّكِ كَالنَّسرِ، فأنا أُوقِعُكِ مِنْ هُناكَ.</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 35:7; إرْمِيا 50:13; إرْمِيا 49:13"> وتكونينَ خرابًا، فكُلُّ مَنْ يَمُرُّ بِكِ يُدهَشُ ويَصفِرُ صفيرَ الهُزءِ على جميعِ نكَباتِكِ.</span>
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 49:33; التَّثْنِيَة 29:23; إرْمِيا 50:40"> وكما في سَدومَ وعَمورَةَ والمُدُنِ المُجاوِرَةِ بَعدَ دَمارِها لا يَسكُنُ فيكِ إنسانٌ ولا يتَغَرَّبُ بشَرٌ يقولُ الرّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="إرْمِيا 12:5; إرْمِيا 30:21; إرْمِيا 50:44–46"> ”ها أنا كأسَدٍ يَصعَدُ مِنْ غَورِ الأردُنِّ على مُرتَفَعِ قُطعانٍ خَصيبٍ انقَضَّ بَغتَةً وأطرُدُهُم مِنها وأُقيمُ علَيها الّذي أختارُهُ. فَمَن مِثلي؟ ومَنْ يُحاكِمُني؟ ومَنِ الرَّاعي الّذي يَقِفُ في وجهي؟</span>
<span className="v" id="v20" data-v="20" data-xref="إرْمِيا 50:45; مَلاخِي 1:3–4; إرْمِيا 49:7"> فاَسمَعوا ما نَوَيتُ بهِ أنا الرّبُّ على بَني أدومَ وما فكَّرْتُ بهِ على سُكَّانِ مدينةِ تيمانَ: حتّى صِغارُهُم يُجَرُّونَ جَرًّا ويَستَولي الرُّعْبُ على مساكنِهِم.</span>
<span className="v" id="v21" data-v="21" data-xref="إرْمِيا 50:46; حِزْقِيال 32:10; إشَعْياء 14:4–15"> مِنْ صوتِ سُقوطِهِم تتَزَلزَلُ الأرضُ، وصُراخُهُم يُسمَعُ في بَحرِ القَصَبِ.</span>
<span className="v" id="v22" data-v="22" data-xref="إرْمِيا 48:40–41; هُوشَع 8:1; إشَعْياء 13:8"> ها هوَ العَدُوُّ كنَسرٍ يَرتَفِعُ ويَطيرُ ويَنشُرُ جناحَيهِ على بُصرَةَ، فتَصيرُ قُلوبُ جبابِرةِ بَني أَدومَ في ذلِكَ اليومِ كقلبِ اَمرأةٍ تُعاني المَخاضَ“.</span>
</div>
</div>

## حكم الرّبّ على دمشق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 10:9; المُلُوكِ الثّانِي 18:34; التَّكوين 15:2"> وقالَ الرّبُّ على دِمشْقَ: ”تَخزَى حماةُ وأرفادُ لأنَّهُما سَمِعَتا نَبَأً فظيعًا، وتَذوبانِ مِنَ الرُّعْبِ وتَضطَرِبانِ كالبحرِ الّذي لا يُمكِنُ أنْ يَهدَأَ.</span>
<span className="v" id="v24" data-v="24"> تَخورُ قوى دِمشْقَ فتُولِّي هارِبَةً. يَدُبُّ فيها الذُّعْرُ ويَستَولي علَيها الغَمُّ والوجَعُ كاَمرَأةٍ في المَخاضِ.</span>
<span className="v" id="v25" data-v="25" data-xref="إرْمِيا 33:9; إرْمِيا 51:41; إشَعْياء 1:26"> وَيحَها كيفَ تُهجَرُ تِلكَ المدينةُ المَرِحَةُ</span>
<span className="v" id="v26" data-v="26" data-xref="إرْمِيا 50:30; عامُوس 4:10; إرْمِيا 11:22"> وكَيفَ يَسقُطُ شُبَّانُها في ساحاتِها ويَهلِكُ جُنودُها في ذلِكَ اليومِ.</span>
<span className="v" id="v27" data-v="27" data-xref="عامُوس 1:3–5; المُلُوكِ الأوَّلُ 15:18–20; إرْمِيا 43:12"> وأُشعِلُ نارًا في سُورِ دِمشْقَ فتَأْكُلُ قُصورَ بَنهَدَدَ مَلِكِها.</span>
</div>
</div>

## الحكم على قبائل العرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="القُضاة 6:3; التَّكوين 25:13; إشَعْياء 11:14"> وقالَ الرّبُّ على بَني قيدارَ ومَمالِكِ حاصورَ الّتي قاتَلَها نبوخذنَصَّرُ مَلِكُ بابِلَ: ”قوموا اَصعَدوا إلى بَني قيدارَ ودَمِّروا أبناءَ المَشرِقِ</span>
<span className="v" id="v29" data-v="29" data-xref="إرْمِيا 46:5; إرْمِيا 6:25; حَبَقُّوق 3:7"> خُذوا خيامَهُم وغنَمَهُم وستائِرَهُم وجميعَ أدواتِهِم وجِمالَهُم، ونادوا علَيهِم بِالرُّعْبِ مِنْ كُلِّ جهةٍ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="إرْمِيا 25:9; إرْمِيا 27:6; إشَعْياء 10:7"> أُهرُبوا سَريعًا واَختَبِئوا يا سُكَّانَ حاصورَ، لأنَّ نبوخذنَصَّرَ مَلِكَ بابِلَ نَوى علَيكُم شَرًّا ودَبَّرَ مكيدَةً.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="حِزْقِيال 38:11; العَدَد 23:9; التَّثْنِيَة 33:28"> قوموا اَصعَدوا إلى أُمَّةٍ مُطمَئِنَّةٍ ساكِنةٍ في أمانٍ، لا أبوابَ لها ولا أقفالَ وسُكَّانُها في عُزلَةٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="إرْمِيا 9:26; إرْمِيا 25:23; إرْمِيا 49:36"> فتُنهَبُ جِمالُها نَهبًا وتُسلَبُ مَواشيها الكثيرةُ. وأُبَدِّدُ معَ كُلِّ رِيحٍ أولَئِكَ المَقصوصي الشَّعرِ، وأُنزِلُ بهِمِ النَّكبَةَ مِنْ كُلِّ جِهَةٍ يقولُ الرّبُّ.</span>
<span className="v" id="v33" data-v="33" data-xref="إرْمِيا 10:22; إرْمِيا 9:11; إرْمِيا 51:37"> فتَصيرُ حاصورُ مأوًى لِبناتِ آوى، قَفْرًا إلى الأبدِ، لا يَسكُنُ فيها إنسانٌ ولا يَتَغرَّبُ بشَرٌ“.</span>
</div>
</div>

## على عيلام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="التَّكوين 10:22; دانيال 8:2; إرْمِيا 25:25"> وقالَ الرّبُّ لإرميا النَّبيِّ على عيلامَ، في بَدءِ عَهدِ صِدقيَّا مَلِكِ يَهوذا:</span>
<span className="v" id="v35" data-v="35" data-xref="إشَعْياء 22:6; إرْمِيا 51:56; المَزامِير 46:9"> ”سأَكسِرُ قَوسَ عيلامَ، مَصدَرَ قُوَّتِها،</span>
<span className="v" id="v36" data-v="36" data-xref="إرْمِيا 49:32; رُؤيا يُوحَنّا 7:1; حِزْقِيال 5:10"> وأجلِبُ علَيهِمِ الرِّياحَ الأربَعَ مِنْ أطرافِ السَّماءِ الأربعَةِ وأُبَدِّدُهُم معَ الرِّياحِ كُلِّها، ولا تَبقى أُمَّةٌ لا يَلجأُ إلَيها مُهَجَّروهُم.</span>
<span className="v" id="v37" data-v="37" data-xref="إرْمِيا 9:16; إرْمِيا 48:2; إرْمِيا 49:22"> وأُنزِلُ بِسُكَّانِ عيلامَ الذُّعْرَ أمامَ أعدائِهِم وأمامَ الّذينَ يَطلُبونَ حياتَهُم، وأجلِبُ علَيهِمِ الشَّرَّ ونارَ غضَبي. وأتبَعُهُم بِالسَّيفِ حتّى أَفنيَهُم.</span>
<span className="v" id="v38" data-v="38" data-xref="إرْمِيا 43:10; دانيال 7:9–14"> وأجعَلُ عرشي في عيلامَ وأُهلِكُ مَلِكَها ورُؤساءَها،</span>
<span className="v" id="v39" data-v="39" data-xref="إرْمِيا 48:47; هُوشَع 3:5; ميخا 4:1"> لكنِّي في الأيّامِ الآتيةِ أُعيدُ أمجادَها يقولُ الرّبُّ“.</span>
</div>
