## الخزاف وعمله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقال الرّبُّ لإرميا:</span>
<span className="v" id="v2" data-v="2" data-xref="العِبرانِيّين 1:1; اعمال الرسل 9:6; إرْمِيا 19:1–2"> ”قُمْ واَنزِلْ إلى بَيتِ الخَزَّافِ وهُناكَ أُسمِعُك كَلامي“.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 15:14; ﻳﻮﻧﺎﻥ 1:3; اعمال الرسل 26:19"> فنَزَلْتُ إلى بَيتِ الخَزَّافِ، فإذا هوَ يَعمَلُ عمَلًا على الدُّولابِ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﺭﻭﻣﻴﺔ 9:20–23; إرْمِيا 18:6; إشَعْياء 45:9"> فلمَّا لم يَصلُحِ الوِعاءُ الّذي كانَ يَصنَعُهُ مِنَ الطِّينِ في يَدِهِ، عادَ وصَنَعَهُ وِعاءً آخَرَ كما رأى أنْ يَصنَعَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5"> فقالَ ليَ الرّبُّ:</span>
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 64:8; ﺭﻭﻣﻴﺔ 9:20–21; إرْمِيا 18:4"> ”ألا أقدِرُ أنْ أصنَعَ بِكُم كهذا الخَزَّافِ يا بَيتَ إِسرائيلَ؟ أنتُم في يَدي مِثلُ الطِّينِ في يَدِ الخَزَّافِ.</span>
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 1:10; إرْمِيا 12:14–17; إرْمِيا 45:4"> كُلَّما تَكَلَّمتُ على أُمَّةٍ أو على مَملَكةٍ أنِّي سَأقلَعُها وأهدِمُها وأُهلِكُها</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 18:21; إرْمِيا 26:13; ﻳﻮﻧﺎﻥ 3:9–10"> فرَجَعَت تِلكَ الأمَّةُ أوِ المَملَكةُ عَنْ شَرِّها الّذي مِنْ أجلِهِ تَكَلَّمتُ علَيها، نَدِمْتُ على الشَّرِّ الّذي نَويتُ أنْ أُنزِلَهُ بِها.</span>
<span className="v" id="v9" data-v="9" data-xref="إرْمِيا 31:28; عامُوس 9:11–15; إرْمِيا 1:10"> وكُلَّما تكَلَّمْتُ على أُمَّةٍ أو على مَملَكةٍ أنِّي سَأبنيها وأغرِسُها،</span>
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 33:18; صَمُوئيلَ الأوَّلُ 2:30; حِزْقِيال 18:24"> ففَعَلَتِ الشَّرَّ في عيَنيَّ وما سَمِعَت لِصوتي، نَدِمْتُ على الخَيرِ الّذي نَوَيتُ أنْ أصنَعَهُ إليها.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 17:13; إشَعْياء 55:6–7; إرْمِيا 35:15"> ”فالآنَ قُلْ لِرجالِ يَهوذا وسُكَّانِ أُورُشليمَ: هذا ما قالَ الرّبُّ: أُهَيِّئُ علَيكُم شرًّا وأُدبِّرُ علَيكُم مَكيدةً، فاَرجِعوا كُلُّ واحدٍ مِنكُم عَنْ طَريقِهِ الشِّرِّيرِ، وأصلِحوا سُلوكَكُم وأعمالَكُم.</span>
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 2:25; إرْمِيا 7:24; التَّثْنِيَة 29:19"> فيقولونَ: لا نَقدِرُ. سنَتبَعُ أفكارَنا ونَسيرُ بِحسَبِ نيَّاتِ قُلوبِنا الشِّرِّيرةِ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 5:30; هُوشَع 6:10; إرْمِيا 23:14"> فقالَ الرّبُّ:<br />إِسألوا بَينَ الأمَمِ<br />منْ سَمِعَ بِمِثلِ هذا<br />بَنو إِسرائيلَ شعبي<br />عَمِلوا عمَلًا فَظيعًا.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 6:68"> هل تَخْلو جُرودُ لبنانَ مِنَ الثَّلجِ<br />أم تَنضَبُ ينابيعُهُ البارِدَةُ الجاريةُ؟</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 6:16; إرْمِيا 2:32; هُوشَع 11:2"> لكِنَّ شَعبي نَسَوني<br />وبَخَّروا للأصنامِ الباطِلةِ.<br />عثَروا في طُرُقِهِمِ القَويمةِ القديمةِ<br />وسَلَكوا في طَريقٍ غَيرِ مُمَهَّدٍ،</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="المَزامِير 22:7; إرْمِيا 25:9; إرْمِيا 19:8"> ليَجعَلوا أرضَهُم رُعْبًا<br />وعُرضَةً لِصَفيرِ الهُزْءِ.<br />فَكُلُّ مَنْ يَمُرُّ فيها<br />يَرتَعِبُ ويَهُزُّ رَأْسَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="إرْمِيا 13:24; إرْمِيا 2:27; أيُّوب 27:21"> كَريحٍ شرقيَّةٍ أُبَدِّدُهُم<br />أمامَ وجهِ العَدُوِّ،<br />وأُريهِم في يومِ الضِّيقِ<br />ظَهري لا وجهي.</span>
</div>
</div>

## مكيدة على إرميا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 52:2; إرْمِيا 11:19; مَلاخِي 2:7"> فقالوا: ”تَعالَوا نَكيدُ لإرميا مَكيدةً، فالشَّريعةُ لا تُحرَمُ مِنْ كاهنٍ سِواهُ، ولا حُسنُ المَشورةِ مِنْ حكيمٍ، ولا كَلِمةُ الوحي مِنْ نبِيٍّ. تعالَوا نَتَّهِمُهُ ولا نُصغي إلى كَلِمةٍ مِنْ كَلِماتِهِ“.</span>
</div>
<div className="bp">

</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 35:7; المَزامِير 57:6; المَزامِير 106:23"> أيُجازَى الخَيرُ بِالشَّرِّ؟<br />فَهُم حفَروا حُفرَةً لِهَلاكي.<br />أُذكُرْ وقوفي بَينَ يَدَيكَ<br />لأتَكَلَّمَ مِنْ أجلِهِم بِالخَيرِ<br />وأرُدَّ عَنهُم غضَبَكَ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 109:9–20; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 5:3; عامُوس 4:10"> سَلِّمْ بَنيهِم إلى الجُوعِ<br />واَدفَعْهُم إلى يَدِ السَّيفِ<br />لِتَكُنْ نِساؤُهُم ثَكالى وأرامِلَ،<br />وليَكُنْ رِجالُهُم قَتْلى الوباءِ<br />وشُبَّانُهُم صَرْعى السَّيفِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="إرْمِيا 6:26; المَزامِير 140:5; صَفَنْيا 1:10–11"> لِيُسمَعْ صُراخٌ مِنْ بُيوتِهِم<br />حينَ تُباغِتُهُم بِالغُزاةِ،<br />فهُم حفَرُوا هُوَّةً ليَأخُذوني<br />وأخفَوا لِرِجلَيَّ فِخاخًا.</span>
</div>

<span className="v" id="v23" data-v="23" data-xref="المَزامِير 35:4; المَزامِير 69:22–28; نَحَمْيا 4:4–5"> وأنتَ تَعلَمُ مَكايدَهُم لِقَتْلي،<br />فلا تَغفِرْ يا ربُّ إثْمَهُم.<br />لا تَمْحُ خطيئَتَهُم مِنْ أمامِكَ<br />وعاقِبْهُم في وقتِ غضَبِكَ.</span>
