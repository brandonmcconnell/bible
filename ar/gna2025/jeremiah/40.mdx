## إرميا يقيم عند جدليا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 31:15; اعمال الرسل 12:6–7; أفَسُس 6:20"> وكلَّمَ الرّبُّ إرميا بَعدَ أنْ أطلَقَهُ نَبوزَرادانُ رئيسُ الشُّرطَةِ مِنَ الرَّامَةِ حَيثُ كانَ أخذَهُ مُكبَّلًا بِالقُيودِ بَينَ جميعِ أسرى أُورُشليمَ ويَهوذا الّذينَ سُبُوا إلى بابِلَ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 29:24–28; إرْمِيا 22:8–9; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:15–17"> فاَنفَرَدَ رئيسُ الشُّرطَةِ بإرميا وقالَ لَه: ”الرّبُّ إلَهُكَ تكَلَّمَ بِهذا الشَّرِّ على هذا الموضِعِ،</span>
<span className="v" id="v3" data-v="3" data-xref="ﺭﻭﻣﻴﺔ 2:5; إرْمِيا 50:7; دانيال 9:11–12"> وهوَ الّذي جَلَبَهُ كما قالَ، خَطِئْتُم إليهِ وما سَمِعْتُم لَه، فتَمَّ علَيكُم هذا الأمرُ.</span>
<span className="v" id="v4" data-v="4" data-xref="إرْمِيا 39:11–12; التَّكوين 20:15; التَّكوين 13:9"> والآنَ أفلَتُّ يَدَكَ مِنَ القيودِ، فإنْ رَغِبتَ أنْ تَجيءَ معي إلى بابِلَ، فتعالَ، وأنا أعتَني بِأمرِكَ. وإنْ ساءَكَ أنْ تَجيءَ مَعي إلى بابِلَ فابْقَ. أُنظُرْ! فالأرضُ كُلُّها بَينَ يَدَيكَ، وأينَما طابَ لكَ ووافَقَ أنْ تَذهَبَ فاَذْهَبْ</span>
<span className="v" id="v5" data-v="5" data-xref="إرْمِيا 39:14; المُلُوكِ الثّانِي 8:7–9; إرْمِيا 41:2"> أمَّا إذا بَقيتَ، فَعُدْ إلى جَدَلْيا بنِ أحيقامَ بنِ شافانَ الّذي ولَّاهُ مَلِكُ بابِلَ على مُدُنِ يَهوذا، وأقِمْ معَهُ بَينَ الشَّعبِ، أو فاَذهَبْ أينَما تيَسَّرَ لكَ“. وأعطاهُ زادًا وهديَّةً وصَرَفَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إرْمِيا 39:14; القُضاة 20:1; القُضاة 21:1"> فجاءَ إرميا إلى جَدليا بنِ أحيقامَ في مِصفاةَ وأقامَ معَهُ بَينَ الشَّعبِ الباقي في البِلادِ.</span>
</div>
</div>

## جدليا حاكم يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 39:10; المُلُوكِ الثّانِي 25:4; إرْمِيا 39:4"> ولمَّا سَمِعَ القادَةُ والجُنودُ الّذينَ لَجأُوا إلى الصَّحراءِ أنَّ مَلِكَ بابِلَ ولَّى جَدَليا بنَ أحيقامَ على أرضِ يَهوذا، ووكَّلَ إليهِ الفُقَراءَ مِنَ الرِّجالِ والنِّساءِ والأطفالِ الّذينَ بَقوا في البِلادِ ولم يَسْبِهِمِ البابِليُّونَ إلى بابِلَ،</span>
<span className="v" id="v8" data-v="8" data-xref="إرْمِيا 42:1; التَّثْنِيَة 3:14; إرْمِيا 42:8"> جاؤوا إلى جَدَليا في مِصفاةَ، وهُم إسماعيلُ بنُ نَثَنيا ويوحانانُ ويوناثانُ اَبنا قارِيعَ وسَرايا بنُ تَنحومَثَ، وبَنو عيفايَ النَّطوفاتيِّ، ويَزَنْيا بنُ المَعكيِّ، هُم ورِجالُهُم.</span>
<span className="v" id="v9" data-v="9" data-xref="إرْمِيا 27:11; المُلُوكِ الثّانِي 25:24; صَمُوئيلَ الأوَّلُ 20:16–17"> فطَمأنَهُم جَدَليا بِقولِهِ لهُم: ”لا تخافوا مِنْ عُبوديَّةِ البابِليِّينَ. أُسكُنوا في هذِهِ الأرضِ واَخدُموا مَلِكَ بابِلَ.</span>
<span className="v" id="v10" data-v="10" data-xref="إرْمِيا 40:12; التَّثْنِيَة 1:38; إشَعْياء 16:9"> وأنا مُقيمٌ في مِصفاةَ لِأستَقبِلَ البابِليِّينَ الّذينَ يأتونَ إِلينا، أمَّا أنتُم فاَجمَعوا الخمرَ والثِّمارَ والزَّيتَ واخزُنوها في أوعيَتِكُم واَسكُنوا في مُدُنِكُمُ الّتي تَحتَلُّونَها“.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 16:4; التَّكوين 36:8; إرْمِيا 24:9"> وكذلِكَ فعَلَ جميعُ اليهودِ الّذينَ في مُوآبَ وبَينَ بَني عَمُّونَ وفي أدومَ وسائِرِ الأراضي حينَ سَمِعوا أنَّ مَلِكَ بابِلَ ترَكَ بَقيَّةً مِنَ الشَّعبِ في يَهوذا ووَلَّى علَيهِم جَدَليا بنَ أحيقامَ بنِ شافانَ.</span>
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 43:5"> فرَجَعوا مِنْ جميعِ المواضعِ الّتي طُرِدوا إليها وجاؤوا إلى أرضِ يَهوذا، إلى جَدَليا في مِصفاةَ، وجَمَعوا مِنَ الخمرِ والثِّمارِ شيئًا كثيرًا جدًّا.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 42:1; إرْمِيا 40:8"> ثُمَّ إنَّ يوحانانَ بنَ قاريحَ وجميعَ القادَةِ الّذينَ لَجأوا إلى الصَّحراءِ جاؤوا إلى جَدَليا في مِصفاةَ</span>
<span className="v" id="v14" data-v="14" data-xref="إرْمِيا 41:10; صموئيلَ الثّانِي 10:1–6; إرْمِيا 25:21"> وقالوا لَه: ”هل عَلِمْتَ أنَّ بَعليسَ مَلِكَ بَني عَمُّونَ أرسَلَ إِسماعيلَ بنَ نَثَنيا لِيَقتُلَكَ؟“ فما صَدَّقَهُم جَدَليا.</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 26:8; صموئيلَ الثّانِي 21:17; إرْمِيا 42:2"> فقالَ يوحانانُ بنُ قاريحَ لِجَدَليا سِرًّا في مصفاةَ: ”دَعني أَذهَبُ فأقتُلَ إسماعيلَ بنَ نَثَنيا دونَ أنْ يَعلَمَ أحدٌ. لِماذا يَقتُلُكَ هوَ فيَتَبَدَّد جميعُ اليهودِ المُجتمِعينَ إليكَ وتَهلِك بَقيَّةُ يَهوذا“.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 10:16–17; ﺭﻭﻣﻴﺔ 3:8; إرْمِيا 41:2"> فأجابَهُ جَدَليا: ”لا تفعَلْ، فما تقولُهُ على إسماعيلَ اَفتِراءٌ“.</span>
</div>
