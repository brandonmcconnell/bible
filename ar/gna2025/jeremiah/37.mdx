## صدقيا يناشد إرميا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 22:24; المُلُوكِ الثّانِي 24:17; المُلُوكِ الثّانِي 24:12"> وأمرَ نبوخذنَصَّرُ مَلِكُ بابِلَ، فمَلَّكَ صِدْقيَّا بنَ يوشيَّا مكانَ كُنْيا بنِ يوياقيمَ على أرضِ يَهوذا.</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الثّانِي 24:19–20; الأمثال 29:12; أخبارِ الأيّامِ الثّانِي 36:12–16"> فلم يسمَعْ هوَ ولا رِجالُهُ ولا شعبُ تِلكَ الأرضِ لِكلامِ الرّبِّ على لِسانِ إرميا النَّبيِّ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="إرْمِيا 21:1–2; إرْمِيا 29:25; إرْمِيا 42:20"> وأرسَلَ المَلِكُ صِدْقيَّا يوخَلَ بنَ شَلَمْيا وصَفَنْيا بنَ مَعسِيَّا الكاهنَ إلى إرميا النَّبيِّ يقولُ لَه: ”صَلِّ مِنْ أجلِنا إلى الرّبِّ إلهِنا“.</span>
<span className="v" id="v4" data-v="4" data-xref="إرْمِيا 37:15; إرْمِيا 32:2–3"> وكانَ إرميا يَروحُ ويَجيءُ بَينَ الشَّعبِ، قَبلَ أنْ يُلقَى في السِّجنِ.</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 17:15; إرْمِيا 37:11; إرْمِيا 34:21"> وكانَ البابِليُّونَ يُحاصِرونَ أُورُشليمَ، فلمَّا سَمِعوا بأنَّ جيشَ فِرعونَ خرَجَ إليهِم مِنْ مِصْرَ اَرتَدُّوا عَنْ أُورُشليمَ.</span>
<span className="v" id="v6" data-v="6"> فقالَ الرّبُّ لإرميا النَّبيِّ:</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:17; المُلُوكِ الثّانِي 22:18; حِزْقِيال 17:17"> ”قُلْ لِمَلِكِ يَهوذا الّذي أرسلَكَ لِتَستَشيرَني: سيَرجِعُ جيشُ فِرعَونَ الّذي خرَجَ لِمُساعَدَتِكُم إلى أرضِ مِصْرَ،</span>
<span className="v" id="v8" data-v="8" data-xref="إرْمِيا 39:2–8; إرْمِيا 38:23; إرْمِيا 34:21–22"> ويَرجِعُ البابليُّونَ ويُحارِبونَ هذِهِ المدينةَ ويَأخُذونَها ويَحرقونَها بِالنَّارِ.</span>
<span className="v" id="v9" data-v="9" data-xref="عُوبَدْيا 1:3; إرْمِيا 29:8; غَلاطِيَّة 6:3"> فلا تَخدَعوا أنفُسَكُم فتقولوا: سيذهَبُ البابليُّونَ عنَّا لأنَّهُم لا يَذهبونَ.</span>
<span className="v" id="v10" data-v="10" data-xref="إرْمِيا 50:45; إشَعْياء 30:17; يُوئيل 2:11"> بل لو قتَلْتُم جميعَ جيشِ البابليِّينَ الّذينَ يُحارِبونَكُم وبَقيَ مِنهُم عدَدٌ مِنَ الجَرحى في عَسكرِهِم، فهؤلاءِ يقومونَ ويحرِقونَ هذِهِ المدينةَ بِالنَّارِ“.</span>
</div>
</div>

## إرميا في السجن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 37:5"> ولمَّا رجَعَ جيشُ البابليِّينَ عَنْ أُورُشليمَ عِندَ اقترابِ جيشِ فِرعونَ،</span>
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 1:1; نَحَمْيا 6:11; المُلُوكِ الأوَّلُ 19:3"> خرَجَ إرميا مِنْ أُورُشليمَ لِيَذهبَ إلى أرضِ بنيامينَ ويأخُذَ حِصَّتَهُ مِنْ إِرثِ عائِلَتِهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="زَكَريّا 14:10; إرْمِيا 38:7; لُوقا 23:2"> فلمَّا وصَلَ إلى بابِ بنيامينَ قبَضَ علَيهِ رئيسُ الحَرسِ واسمُهُ يَرئِيَّا بنُ شَلَمْيا بنِ حَنَنِيَّا وقالَ لَه: ”أنتَ هاربٌ إلى البابليِّينَ“.</span>
<span className="v" id="v14" data-v="14" data-xref="المَزامِير 52:1–2; إرْمِيا 40:4–6; المَزامِير 27:12"> فقالَ لَه إرميا: ”هذا كَذِبٌ. ما أنا هارِبٌ إلى البابليِّين“. فلم يَسمَعْ لَه، بل قبَضَ علَيهِ وجاءَ بهِ إلى الرُّؤساءِ.</span>
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 38:26; إرْمِيا 20:1–3; إرْمِيا 26:16"> فثارَ غضَبُ الرُّؤساءِ على إرميا وضرَبوهُ وألقوهُ في بَيتِ يوناثانَ الكاتِبِ، لأنَّهُم جعَلوا ذلِكَ البَيتَ سِجنًا.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إرْمِيا 38:6; إرْمِيا 38:10–13; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:53"> فدخَلَ إرميا إلى سرَاديبِ السِّجنِ وأقامَ هُناكَ أيّامًا كثيرةً.</span>
<span className="v" id="v17" data-v="17" data-xref="إرْمِيا 21:7; إرْمِيا 24:8; المُلُوكِ الأوَّلُ 14:1–4"> ثُمَّ أرسَلَ المَلِكُ صِدْقيَّا وأخذَهُ وسألَهُ في قصرِهِ سِرًّا: ”هل كَلَّمَكَ الرّبُّ حَقًّا؟“ فقالَ إرميا: ”نعم، وقالَ لي سَتُسَلَّمُ إلى يَدِ مَلِكِ بابِلَ“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 25:8; يُوحَنّا 10:32; دانيال 6:22"> وقالَ إرميا للمَلِكِ صِدْقيَّا: ”بِماذا خَطِئْتُ إليكَ وإلى رِجالِكَ وإلى هذا الشَّعبِ حتّى ألقَيتُموني في السِّجنِ؟</span>
<span className="v" id="v19" data-v="19" data-xref="إرْمِيا 2:28; إرْمِيا 28:10–17; إرْمِيا 6:14"> وأينَ أنبياؤُكُمُ الّذينَ تنَبَّأوا لكُم قائِلينَ إنَّ مَلِكَ بابِلَ لا يأتي علَيكُم ولا على هذِهِ الأرضِ؟</span>
<span className="v" id="v20" data-v="20" data-xref="إرْمِيا 36:7; إرْمِيا 38:6–9; اعمال الرسل 28:18–19"> ”والآنَ إِسمَعْ يا سيِّدي المَلِكُ واَقبَلْ تَضَرُّعي إليكَ ولا تَرُدَّني إلى بَيتِ يوناثانَ الكاتِبِ لِئلَّا أموتَ هُناكَ“.</span>
<span className="v" id="v21" data-v="21" data-xref="إرْمِيا 38:13; إرْمِيا 38:28; إرْمِيا 38:9"> فأمرَ المَلِكُ صِدْقيَّا أن يودَعَ إرميا في سِجنِ القصرِ، وأنْ يُعطَى لَه رغيفٌ مِنَ الخبزِ كُلَّ يومٍ مِنْ سُوقِ الخَبَّازينَ إلى أنْ يَنفُدَ الخبزُ كُلُّهُ مِنَ المدينةِ. فأقامَ إرميا في سِجنِ القصرِ.</span>
</div>
