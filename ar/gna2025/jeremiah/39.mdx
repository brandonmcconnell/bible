## سقوط أُورُشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="حِزْقِيال 24:1–2; إرْمِيا 52:4–7; المُلُوكِ الثّانِي 25:1–12"> في الشَّهرِ العاشِرِ من السَّنةِ التَّاسِعةِ لِصدْقيَّا مَلِكِ يَهوذا جاءَ نبوخذنَصَّرُ مَلِكُ بابِلَ وجميعُ جيشِهِ إلى أُورُشليمَ وحاصَرَها.</span>
<span className="v" id="v2" data-v="2" data-xref="إرْمِيا 52:6–7; ميخا 2:12–13; المُلُوكِ الثّانِي 25:3–4"> وفي الشَّهرِ الرَّابِعِ مِنَ السَّنةِ الحاديَةَ عَشْرَةَ لِصِدْقيَّا، اَختَرَقَ العدُوُّ أسوارَ المدينةِ.</span>
<span className="v" id="v3" data-v="3" data-xref="إرْمِيا 21:4; إرْمِيا 38:17; إرْمِيا 39:13"> ودخَلَ كُلُّ رُؤساءِ مَلِكِ بابِلَ وجَلَسوا بِالبابِ الأوسَطِ، وهُم: نَرجَلُ شراصَرُ رئيسُ المَجوسِ، وسَمْجَرُ نَبو وسرسَخيمُ رئيسُ الخِصيانِ وسواهُم.</span>
<span className="v" id="v4" data-v="4" data-xref="عامُوس 2:14; أخبارِ الأيّامِ الثّانِي 32:5; المُلُوكِ الثّانِي 25:4–7"> فلمَّا رآهُم صِدْقيَّا مَلِكُ يَهوذا وجميعُ جُنودِهِ هَرَبوا وخرَجوا مِنَ المدينةِ لَيلًا مِنْ طريقِ بستانِ المَلِكِ مِنَ البابِ الّذي بَينَ السُّورَينِ، وذَهَبوا في طريقِ غَورِ الأردُنِّ.</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الثّانِي 23:33; يَشُوع 5:10; إرْمِيا 52:26–27"> فلَحِقَ بِهِم جيشُ البابِليِّينَ في سَهلِ أريحا، فقَبَضوا على صِدْقيَّا وأخَذوهُ إلى نبوخذنَصَّرَ مَلِكِ بابِلَ في رَبلَةَ بِأرضِ حَماةَ، فحكَمَ علَيهِ.</span>
<span className="v" id="v6" data-v="6" data-xref="إرْمِيا 52:10; إرْمِيا 34:19–21; إرْمِيا 24:8–10"> وذبَحَ مَلِكُ بابِلَ بَني صِدْقيَّا في رَبلَةَ أمامَ عَينَيهِ، كما ذبَحَ جميعَ أشرافِ يَهوذا.</span>
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 12:13; المُلُوكِ الثّانِي 25:7; إرْمِيا 52:11"> وفقَأَ عَينَي صِدْقيَّا وقَيَّدَهُ بِسِلسِلَتينِ مِنْ نُحاسٍ لِيَجيءَ بهِ إلى بابِلَ.</span>
<span className="v" id="v8" data-v="8" data-xref="نَحَمْيا 1:3; إرْمِيا 38:18; إرْمِيا 21:10"> وأحرَقَ البابِليُّونَ قصرَ المَلِكِ وبُيوتَ الشَّعبِ بِالنَّارِ وهَدَموا أسوارَ أُورُشليمَ.</span>
<span className="v" id="v9" data-v="9" data-xref="إرْمِيا 40:1; التَّكوين 37:36; إرْمِيا 10:18"> وسَبى نَبُوزَرادانُ، رئيسُ الشُّرطَةِ إلى بابِلَ بَقيَّةَ الّذينَ اَعتَصَموا في المدينةِ، والهارِبينَ الّذينَ لَجأُوا إليهِ، وسائِرَ الصُنَّاع.</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الثّانِي 25:12; إرْمِيا 40:7; إرْمِيا 52:16"> أمَّا الفُقراءُ مِنَ الشَّعبِ الّذينَ لا يَملِكونَ شيئًا، فتَركَهُم نَبوزَرادانُ في أرضِ يَهوذا وأعطاهُم كُرومًا وحُقولًا في ذلِكَ الوقتِ.</span>
</div>
</div>

## إرميا يخرج من السجن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 24:23; إرْمِيا 15:21; أيُّوب 5:19"> قالَ نبوخذنَصَّرُ مَلِكُ بابِلَ لِنَبوزَرادانَ، رئيسِ الشُّرطَةِ في شأنِ إرميا:</span>
<span className="v" id="v12" data-v="12" data-xref="الأمثال 16:7; إرْمِيا 40:4; الأمثال 21:1"> ”خُذْهُ وأكرِمْهُ ولا تُسِئْ إليهِ في شيءٍ، بل اعمَلْ لَه كما يقولُ لكَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 39:3; إرْمِيا 39:9"> فأرسَلَ نَبوزَرادانُ رئيسُ الشُّرطَةِ، ونَبوشَزبانُ رئيسُ الخِصيانِ، ونَرجَلُ شَراصَرُ رئيسُ المَجوسِ، وجميعُ رِجالِ مَلِكِ بابِلَ،</span>
<span className="v" id="v14" data-v="14" data-xref="إرْمِيا 38:28; المُلُوكِ الثّانِي 22:12; إرْمِيا 26:24"> أرسَلوا وأخَذوا إرميا مِنَ السِّجنِ ليُسلِموهُ إلى جَدَليا بنِ أحيقامَ بنِ شافانَ، فيَجِيءَ بهِ إلى بَيتِهِ، ليُقيمَ بَينَ الشَّعبِ.</span>
</div>
</div>

## وعد الرّبّ لعبد ملك

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:9; إرْمِيا 32:1–2; إرْمِيا 36:1–5"> وقالَ الرّبُّ لإرميا عِندَما كانَ مَحبوسًا في السِّجنِ:</span>
<span className="v" id="v16" data-v="16" data-xref="دانيال 9:12; زَكَريّا 1:6; يَشُوع 23:14–15"> ”إِذهَبْ وقُل لِعَبدَ مَلِكَ الكُوشيِّ: يقولُ الرّبُّ القديرُ إلهُ إِسرائيلَ: سَأُتِمُّ كَلامي على هذِهِ المدينةِ لِلشَّرِّ لا لِلخَيرِ، ويكونُ ذلِكَ أمامَكَ يومَ يَتِمُّ.</span>
<span className="v" id="v17" data-v="17" data-xref="المَزامِير 41:1–2; المَزامِير 91:14–15; المَزامِير 50:15"> لَكنِّي أُنقِذُكَ في ذلِكَ اليومِ، فلا تَقَعُ في أيدي الّذينَ تخافُ مِنهُم.</span>
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 38:2; المَزامِير 34:22; إرْمِيا 21:9"> وأُنَجِّيكَ فلا تَسقُطُ بِالسَّيفِ، بل تَنجو بِحياتِكَ لأنَّكَ تَوَكَّلْتَ عليَّ، يقولُ الرّبُّ“.</span>
