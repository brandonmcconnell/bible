## داود يمسح ملكًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المُلُوكِ الثّانِي 9:1; صَمُوئيلَ الأوَّلُ 15:23; المَزامِير 78:68–71"> وقالَ الرّبُّ لِصَموئيلَ: ”إلى متى تَتأسَّفُ على شاوُلَ وأنا رفَضتُه مَلِكًا على إِسرائيلَ؟ فاملِأ القَرْنَ بالزَّيتِ وتَعالَ أُرسِلُكَ إلى يَسَّى في بيتَ لَحمَ، لأنِّي اَختْرتُ مِنْ بَنيهِ مَلِكًا“.</span>
<span className="v" id="v2" data-v="2" data-xref="صَمُوئيلَ الأوَّلُ 9:12; صَمُوئيلَ الأوَّلُ 20:29; ﻣﺘﻰ 10:16"> فقالَ صَموئيلُ: ”كيفَ أذهَبُ؟ إنْ عرَفَ شاوُلُ يَقتُلُني“. فقالَ لَه الرّبُّ: ”خُذْ معَكَ عِجْلةً مِنَ البقَرِ وقُلْ: جِئتُ لأُقَدِّمَ ذبيحةً لِلرّبِّ،</span>
<span className="v" id="v3" data-v="3" data-xref="صَمُوئيلَ الأوَّلُ 9:16; صَمُوئيلَ الأوَّلُ 16:12–13; صموئيلَ الثّانِي 15:11"> واَدعُ يَسَّى إلى الذَّبيحةِ، وأنا أُشيرُ علَيكَ بِما تفعَلُ، واَمسَحْ مَلِكًا مَنْ أُسَمِّيهِ لكَ“.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الأوَّلُ 2:13; أخبارِ الأيّامِ الأوّلُ 12:17–18; صَمُوئيلَ الأوَّلُ 21:1"> ففعَلَ صَموئيلُ كما أمرَهُ الرّبُّ وجاءَ إلى بَيتَ لَحمَ. فاَضطرَبَ شُيوخُ المدينةِ عِندَ لِقائِهِ وقالوا لَه: ”ألِخَيرٍ مَجيئُكَ؟“</span>
<span className="v" id="v5" data-v="5" data-xref="يَشُوع 3:5; اللّاويِّين 20:7–8; الخُرُوج 19:10"> فقالَ ”نعم. جِئتُ لأقَدِّمَ ذبيحةً، فتَطهَّروا وتَعالوا معي إلى الذَّبيحةِ“. وأمرَ يَسَّى وبَنيهِ أنْ يتَطهَّروا أيضًا، ودَعاهُم إلى الذَّبيحةِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="صَمُوئيلَ الأوَّلُ 17:13; أخبارِ الأيّامِ الأوّلُ 2:13; صَمُوئيلَ الأوَّلُ 17:22"> فلمّا حضَروا نظَرَ صَموئيلُ إلى أليآبَ بنِ يَسَّى، فقالَ في نفْسِهِ: ”هذا هوَ المَلِكُ الّذي اختارَهُ الرّبُّ“.</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 7:24; إرْمِيا 17:10; لُوقا 16:15"> فقالَ الرّبُّ لِصموئيلَ: ”لا تَلتَفِتْ إلى مَنظرِهِ وطُولِ قامَتِهِ، فأنا رَفَضتهُ لأنَّ الرّبَّ لا ينظُرُ كما ينظُرُ الإنسانُ. فالإنسانُ ينظُرُ إلى المَظهَرِ، وأمَّا الرّبُّ فينظُرُ إلى القلبِ“.</span>
<span className="v" id="v8" data-v="8" data-xref="صَمُوئيلَ الأوَّلُ 17:13; أخبارِ الأيّامِ الأوّلُ 2:13"> ثمَّ دَعا يَسَّى اَبنَهُ أبينادابَ لِلمُرورِ أمامَ صَموئيلَ، فقالَ: ”وهذا أيضًا لَم يَختَرْهُ الرّبُّ“.</span>
<span className="v" id="v9" data-v="9" data-xref="صَمُوئيلَ الأوَّلُ 17:13; أخبارِ الأيّامِ الأوّلُ 2:13; صموئيلَ الثّانِي 13:3"> ثُمَّ طلَبَ يَسَّى مِنِ اَبنِهِ شَمَّةَ، أنْ يَمرَّ بهِ، فقالَ صَموئيلُ: ”وهذا أيضًا لم يَختَرْهُ الرّبُّ“.</span>
<span className="v" id="v10" data-v="10" data-xref="أخبارِ الأيّامِ الأوّلُ 2:13–15"> وهكذا حتّى مرَّ سَبْعَةٌ مِنْ بَنيهِ أمامَ صَموئيلَ، فقالَ صَموئيلُ لِيسَّى: ”لم يَختَرِ الرّبُّ أحدًا مِنْ هؤلاءِ“.</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 78:70–71; صموئيلَ الثّانِي 7:8; صموئيلَ الثّانِي 13:3"> ثمَّ قالَ لَه: ”أهولاءِ جميعُ بَنيكَ؟“ فأجابَهُ: ”بَقيَ الصَّغيرُ وهوَ يَرعى الغنَمَ“. فقالَ لَه: ”أرسِلْ فَجِئنا بهِ، لأنَّنا لا نأكُلُ حتّى يأتيَ إلى هُنا“.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:7; المَزامِير 89:19–20; نَشِيدِ الأنشاد 5:10"> فأرسلَ وجاءَ بهِ، وكانَ حسَنَ العينَينِ، وسيمَ المنظَرِ. فقالَ الرّبُّ لِصَموئيلَ: ”قُمِ اَمسَحْهُ مَلِكًا، فهذا هوَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="صَمُوئيلَ الأوَّلُ 10:6; العِبرانِيّين 1:9; يُوحَنّا 3:34"> فأخذَ صَموئيلُ قرنَ الزَّيتِ ومسحَهُ مَلِكًا مِنْ بَينِ إخوَتِهِ، فحلَّ روحُ الرّبِّ على داوُدَ مِنْ ذلِكَ اليومِ فصاعِدًا. ونهضَ صَموئيلُ وعادَ إلى الرَّامةِ.</span>
</div>
</div>

## داود في قصر شاول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="المَزامِير 51:11; صَمُوئيلَ الأوَّلُ 18:12; القُضاة 9:23"> وفارَقَ روحُ الرّبِّ شاوُلَ، وأزعَجَهُ روحٌ شرِّيرٌ مِنْ عِندِهِ.</span>
<span className="v" id="v15" data-v="15"> فقالَ لَه خَدَمُهُ: ”ها روحٌ شرِّيرٌ مِنْ عِندِ الرّبِّ يُزعِجُكَ،</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الثّانِي 3:15; المُلُوكِ الأوَّلُ 10:8; صَمُوئيلَ الأوَّلُ 19:9"> فَمُرنا يا سيِّدي أنْ نبحَثَ عَنْ رَجُلٍ يُحسِنُ الضَّربَ بالعودِ، حتّى إذا اَعتراكَ الرُّوحُ الشِّرِّيرُ مِنْ عِندِ اللهِ يضرِبُ بِالعودِ فتَنتَعِشُ“.</span>
<span className="v" id="v17" data-v="17"> فقالَ لهُم شاوُلُ: ”فَتِّشوا عَنْ رَجُلٍ يُحسِنُ الضَّربَ بِالعودِ وجيئوني بهِ“.</span>
<span className="v" id="v18" data-v="18" data-xref="صَمُوئيلَ الأوَّلُ 3:19; التَّكوين 39:23; التَّكوين 39:2"> فقالَ أحدُ خَدمِهِ: ”رأيتُ اَبنًا لِيسَّى مِنْ بَيتَ لحمَ يُحسِنُ الضَّربَ بِالعودِ، وهوَ شُجاعٌ وأهلٌ لِلحربِ، وفصيحٌ حسَنُ المَنظَرِ، والرّبُّ معَهُ“.</span>
<span className="v" id="v19" data-v="19" data-xref="صَمُوئيلَ الأوَّلُ 16:11; صَمُوئيلَ الأوَّلُ 17:15; المَزامِير 78:70–72"> فأرسَلَ شاولُ إلى يَسَّى يقولُ لَه: ”أرسِلْ إليَّ داودَ اَبنَكَ الّذي يَرعى الغنَمَ“.</span>
<span className="v" id="v20" data-v="20" data-xref="صَمُوئيلَ الأوَّلُ 10:27; الأمثال 18:16; صَمُوئيلَ الأوَّلُ 17:18"> فأخذَ يَسَّى حمارًا علَيهِ خُبزٌ وزِقُّ خمرٍ وجَديٌ مِنَ المعَزِ، وأرسلَ ذلِكَ كُلَّهُ معَ داوُدَ اَبنِهِ إلى شاوُلَ.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 41:46; الأمثال 22:29; المَزامِير 146:3"> فجاءَ داوُدُ إلى شاوُلَ وراحَ يَخدمُهُ، فأحَبَّهُ جِدًّا وجَعلَهُ حامِلَ سِلاحِهِ.</span>
<span className="v" id="v22" data-v="22"> وأرسلَ شاوُلُ إلى يَسَّى يقولُ لَه: ”دعْ داوُدَ هُنا في خِدمَتي، لأنَّهُ يَنعَمُ بِرضاي“.</span>
<span className="v" id="v23" data-v="23" data-xref="صَمُوئيلَ الأوَّلُ 16:14–16; صَمُوئيلَ الأوَّلُ 18:10–11; لُوقا 11:24–26"> وكانَ إذا اَعتَرى شاوُلَ الرُّوحُ الشِّرِّيرُ مِنْ عِندِ اللهِ يأخُذُ داوُدُ العودَ ويَضرِبُ بهِ، فيَستَريحُ شاوُلُ ويَنتعشُ، ويَنصرِفُ الرُّوحُ الشِّرِّيرُ عَنهُ.</span>
</div>
