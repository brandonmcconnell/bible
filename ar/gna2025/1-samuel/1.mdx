## ألقانة وأهل بيته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أخبارِ الأيّامِ الأوّلُ 6:34; أخبارِ الأيّامِ الأوّلُ 6:25–27; صَمُوئيلَ الأوَّلُ 1:19"> كانَ رَجُلٌ مِنَ الرَّامةِ مِنْ جبَلِ أفرايمَ اَسمُهُ ألقانَةُ بنُ يَروحامَ بنِ أَلِيهُوَ بنِ توحوَ بنِ صُوفٍ الأفرايميِّ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 16:1–2; القُضاة 13:2; التَّكوين 29:23–29"> وكانَت لَه زَوجتانِ، إحداهُما حَنَّةُ والأخرى فِنِنَّةُ. فرُزِقَت فِنِنَّةُ بَنينَ، وأمَّا حَنَّةُ فما كانَ لها بَنونَ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="لُوقا 2:41; يَشُوع 18:1; التَّثْنِيَة 12:5–7"> وكانَ ألقانَةُ يَصعدُ مِنْ مدينتِهِ كُلَّ سنَةٍ لِيسجُدَ لِلرّبِّ القديرِ ويُقدِّمَ لَه الذَّبائحَ في شيلوهَ، حَيثُ كانَ حِفني وفِنحاسُ، اَبنا عالي، كاهنَينِ لِلرّبِّ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 12:17–18; التَّثْنِيَة 16:11; التَّثْنِيَة 12:5–7"> وكانَ إذا قدَّمَ ألقانَةُ الذَّبيحةَ لِلرّبِّ أعطى فِنِنَّةَ زَوجَتَهُ وبَنيها وبَناتِها حِصَصًا مِنَ الذَّبيحةِ</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 30:2; التَّكوين 29:30–31; التَّكوين 45:22"> وأمَّا حَنَّةُ فيُعطيها حِصَّةً واحدةً معَ أنَّهُ كانَ يُحِبُّها، لأنَّ الرّبَّ جعَلَها عاقِرًا.</span>
<span className="v" id="v6" data-v="6" data-xref="أيُّوب 6:14; أيُّوب 24:21; اللّاويِّين 18:18"> وكانَت فِنِنَّةُ ضَرَّتُها تُغضِبُها وتُهينُها بِسبَبِ ذلِكَ.</span>
<span className="v" id="v7" data-v="7" data-xref="صَمُوئيلَ الأوَّلُ 2:19"> وهكذا كانَ يَحدُثُ كُلَّ سنَةٍ عِندَ صُعودِهِم إلى بَيتِ الرّبِّ، فكانَت فِنِنَّةُ تُغضِبُها فتبكي حَنَّةُ ولا تأكُلُ.</span>
<span className="v" id="v8" data-v="8" data-xref="راعُوث 4:15; أيُّوب 6:14; إشَعْياء 54:6"> فقالَ لها ألقانَةُ زَوجُها: ”يا حَنَّةُ، لماذا تبكينَ ولا تأكُلينَ؟ ولماذا يكتئِبُ قلبُكِ؟ أما أنا خَيرٌ لكِ مِنْ عشَرَةِ بَنينَ؟“</span>
</div>
</div>

## حنة وعالي الكاهن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="صَمُوئيلَ الأوَّلُ 3:3; المَزامِير 27:4; المَزامِير 5:7"> بَعدَ أنْ أكلوا وشرِبوا في شيلوهَ، قامَت حَنَّةُ، ووقَفَت أمامَ الرّبِّ تُصلِّي. وكانَ عالي الكاهنُ جالِسًا على كُرسيٍّ عِندَ مدخَلِ هيكَلِ الرّبِّ.</span>
<span className="v" id="v10" data-v="10" data-xref="أيُّوب 7:11; لُوقا 22:44; أيُّوب 10:1"> فصلَّت إلى الرّبِّ بِمَرارةٍ وبَكَت</span>
<span className="v" id="v11" data-v="11" data-xref="القُضاة 13:5; العَدَد 6:5; التَّكوين 29:32"> ونذَرَت نَذْرًا وقالَت: ”أيُّها الرّبُّ القديرُ، إذا نظَرْتَ إلى شَقاءِ أمَتِكَ وذَكرْتَني وما نَسيتَني، بل رزَقتَني مولودًا ذَكَرًا، فأنا أُكَرِّسُهُ لكَ كُلَّ حياتِهِ، دونَ أنْ يَقُصَّ شَعرَ رأسِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:17; يَعقُوب 5:16; لُوقا 18:1"> وأطالَت حَنَّةُ صلاتَها لِلرّبِّ، وكانَ عالي الكاهنُ يُراقِبُ فَمَها.</span>
<span className="v" id="v13" data-v="13" data-xref="ﺭﻭﻣﻴﺔ 8:26; اعمال الرسل 2:13; التَّكوين 24:42–45"> كانَت تُصلِّي في قلبِها، وشَفتاها تتَحرَّكانِ ولا تُخرِجانِ صوتًا، فظنَّهَا عالي سَكْرى.</span>
<span className="v" id="v14" data-v="14" data-xref="أفَسُس 4:25; أيُّوب 8:2; الأمثال 6:9"> فقالَ لها: ”إلى متى أنتِ سَكْرى؟ أفيقي مِنْ خمرِكِ“.</span>
<span className="v" id="v15" data-v="15" data-xref="المَزامِير 62:8; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:19; المَزامِير 42:4"> فأجابَت: ”لا يا سيِّدي. أنا اَمرأةٌ حزينةُ النَّفْسِ لم أشربْ خمرًا ولا مُسكِرًا، بل أكشِفُ نفْسي أمامَ الرّبِّ.</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 2:12; أيُّوب 6:2–3; ﻣﺘﻰ 12:34–35"> فلا تَحسِبْ أمَتَكَ مِنْ بَناتِ السُّوءِ، فأنا أطَلتُ الصَّلاةَ مِنْ شِدَّةِ الحُزنِ والغَمِّ“.</span>
<span className="v" id="v17" data-v="17" data-xref="مَرْقُس 5:34; القُضاة 18:6; المَزامِير 20:3–5"> فأجابَها عالي: ”إذهَبي بِسلامٍ، وإلهُ إِسرائيلَ يُعطيكِ ما طلَبْتِ مِنهُ“.</span>
<span className="v" id="v18" data-v="18" data-xref="راعُوث 2:13; ﺭﻭﻣﻴﺔ 15:13; الجامِعَة 9:7"> فقالَت: ”أرجو يا سيِّدي أنْ أحظى بِعَطفِكَ“. ومَضَت في طريقِها، وأكلَت، وزالَ الحُزنُ عَنْ وجهِها.</span>
</div>
</div>

## مولد صموئيل وتكريسه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 21:1; التَّكوين 30:22; صَمُوئيلَ الأوَّلُ 1:11"> وبكَّرَ ألقانَةُ وأهلُ بَيتِهِ في الصَّباحِ وسجَدوا لِلرّبِّ، ثُمَّ رجَعوا إلى منزِلِهِم في الرَّامةِ. ونامَ ألقانَةُ معَ حَنَّةَ زَوجَتِه واَستَجابَ الرّبُّ صلاتَها،</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 41:51–52; ﻣﺘﻰ 1:21; الخُرُوج 2:10"> فحَبِلَت في تِلكَ السَّنةِ ووَلَدَتِ اَبنًا ودَعتْهُ صَموئيلَ لأنَّها قالَت: ”مِنَ الرّبِّ طَلَبتُهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="صَمُوئيلَ الأوَّلُ 1:3; التَّثْنِيَة 12:11; يَشُوع 24:15"> وبَعدَ ذلِكَ، صَعِدَ ألقانَةُ معَ جميعِ أهلِ بَيتِهِ إلى شيلوهَ لِيُقدِّمَ لِلرّبِّ الذَّبيحةَ السَّنويَّةَ ويوفيَ نَذْرَهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="صَمُوئيلَ الأوَّلُ 1:28; صَمُوئيلَ الأوَّلُ 1:11; لُوقا 2:22"> أمَّا حَنَّةُ فلم تَصعَدْ وقالَت لِزَوجِها: ”متى فطَمْتُ الصَّبيَّ آخُذُهُ إلى شيلوهَ لِيَحضَرَ أمامَ الرّبِّ ويُقيمَ هُناكَ كُلَّ حياتِهِ“.</span>
<span className="v" id="v23" data-v="23" data-xref="صموئيلَ الثّانِي 7:25; لُوقا 11:27; التَّكوين 21:7–8"> فقالَ لها ألقانَةُ: ”إفعلي ما يَحسُنُ في عَينيكِ واَنتَظِري حتّى تَفطِميهِ، ونرجو أنْ يُحَقِّقَ الرّبُّ وعدَهُ“. فَلزمَت حَنَّةُ البيتَ تُرضِعُ اَبنَها حتّى فطَمَتْهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="العَدَد 15:9–10; يَشُوع 18:1; صَمُوئيلَ الأوَّلُ 4:3–4"> فلمَّا فَطَمتْهُ صَعِدت بِهِ إلى الرّبِّ في شيلوهَ ومعَها عِجْلٌ اَبنُ ثَلاثِ سِنينَ وقُفَّةٌ مِن دقيقٍ وزِقُّ خمرٍ، وكانَ الصَّبيُّ طِفلًا.</span>
<span className="v" id="v25" data-v="25" data-xref="لُوقا 18:15–16; لُوقا 2:22"> فذَبَحوا العِجْلَ وقدَّموا الصَّبيَّ إلى عالي الكاهنِ.</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الثّانِي 4:30; المُلُوكِ الثّانِي 2:4; المُلُوكِ الثّانِي 2:6"> وقالت لَه حَنَّةُ: ”أتَذكُرُ يا سيِّدي، أنا المرأةُ الّتي وقَفَت أمامَكَ هُنا تُصلِّي إلى الرّبِّ.</span>
<span className="v" id="v27" data-v="27" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:15; المَزامِير 118:5; ﻣﺘﻰ 7:7"> طَلبْتُ مِنهُ هذا الصَّبيَّ، فأعطاني ما طَلَبْتُ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 24:26; التَّكوين 24:52; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:15"> فكَرَّسْتُهُ لِلرّبِّ كُلَّ حياتِهِ. فهوَ مُكَرَّسٌ لهُ“. وسجَدوا هُناكَ لِلرّبِّ.</span>
</div>
