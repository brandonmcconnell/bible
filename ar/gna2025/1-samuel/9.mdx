## شاول وصموئيل يتلاقيان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صَمُوئيلَ الأوَّلُ 14:51; اعمال الرسل 13:21; أخبارِ الأيّامِ الأوّلُ 9:36–39"> وكانَ رَجُلٌ جبَّارٌ وغنيٌّ مِن قبيلَةِ بنيامينَ اَسمُهُ قَيشُ بنُ أبييلَ بنِ صَرورَ بنِ بَكورَةَ بنِ أفيحَ،</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 6:2; العَدَد 13:33; صَمُوئيلَ الأوَّلُ 16:7"> وكانَ لَه اَبنٌ حسَنُ الطَّلعةِ في زَهوَةِ العُمرِ اَسمُهُ شاوُلُ. ولم يَكُنْ في بَني إِسرائيلَ رَجُلٌ أبهى مِنهُ، وكانَ يَزيدُ طُولًا على جميعِ الشَّعبِ مِنْ كَتفِهِ وما فوقُ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="القُضاة 5:10; صَمُوئيلَ الأوَّلُ 10:2; القُضاة 10:4"> فحدَثَ أنْ ضَلَّت حميرٌ لِقيسَ أبي شاوُلَ، فقالَ قَيسُ لِشاوُلَ اَبنِهِ: ”خُذْ معَكَ واحدًا مِنَ الخَدَمِ، واَذهَبْ في طَلَبِ الحميرِ“.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الثّانِي 4:42; يَشُوع 24:33; يُوحَنّا 3:23"> فقَطَعا جبَلَ أفرايمَ، ولم يَجِداها، بَحَثَا عَنها في أرضِ شَليشَةَ ولَكِنْ دونَ نتيجةٍ. ثُمَّ في أرضِ شَعليمَ وبَعدَها في أرضِ بنيامينَ، فلم يَعثُرا على أثَرٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="صَمُوئيلَ الأوَّلُ 10:2; صَمُوئيلَ الأوَّلُ 1:1; لُوقا 12:11"> فلمَّا دخَلا أرضَ صُوفَ قالَ شاوُلُ لِخادِمِه الّذي معَهُ: ”تَعالَ نَرجِعُ لِئلَّا يتَحوَّلَ قلقُ أبي على الحميرِ إلى قلقٍ أكبرَ علَينا“.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 13:1; التَّثْنِيَة 33:1; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:11"> فقالَ لَه خادِمُهُ: ”هُناكَ رَجُلٌ مِنْ رِجالِ اللهِ في هذِهِ المدينةِ، وهوَ مُحتَرمٌ، وكُلُّ ما يقولهُ يَتِمُّ. فتعالَ نذهَبُ إليهِ عَلَّهُ يَدُلُّنا إلى الإتِّجاهِ الصَّحيحِ“.</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الأوَّلُ 14:3; المُلُوكِ الثّانِي 8:8; المُلُوكِ الثّانِي 4:42"> فقالَ لَه شاوُلُ: ”إذا ذَهَبنا إليهِ، فماذا نُقَدِّمُ لَه؟ الخبزُ نفَدَ وأكياسُنا فرَغَت، ولا شيءَ لَدَينا فماذا نُعطيهِ؟“</span>
<span className="v" id="v8" data-v="8"> فأجابَهُ الخادِمُ: ”معي رُبعُ مِثقالِ فِضَّةٍ أقَدِّمُهُ لَه فيهدِينا إلى طريقِنا“.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 30:10; المُلُوكِ الثّانِي 17:13; عامُوس 7:12"> وكانَ فيما سبَقَ إذا أرادَ أحدٌ مِنْ بَني إِسرائيلَ أنْ يذهَبَ لِيَستشيرَ اللهَ يقولُ: ”تَعالَوا نذهَبُ إلى الرَّائي“. لأنَّ الّذي يُقالُ لَه اليومَ ”نبيٌّ“ كانَ يُقالُ لَه مِنْ قَبلُ راءٍ.</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الثّانِي 5:13–14"> فقالَ شاوُلُ لِخادِمِه: ”حسَنًا نذهَبُ إليهِ“. وقصَدا المدينةَ الّتي فيها رَجُلُ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 24:11; الخُرُوج 2:16; التَّكوين 24:18–20"> وبينَما هُما يَصعَدانِ إلى المدينةِ، صادَفا فتَياتٍ خارِجاتٍ لِيَستَقينَ ماءً، فقالا لهُنَّ: ”هلِ الرَّائي هُنا؟“</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 31:54; صَمُوئيلَ الأوَّلُ 16:2; صَمُوئيلَ الأوَّلُ 10:5"> فأجَبنَ: ”نعَم، إنَّهُ في المدينةِ جاءَها مِنْ وقتٍ قصيرٍ لأنَّ الشَّعبَ سَيُقَدِّمُ اليومَ ذبيحةً على التَّلَّةِ فأسرِعا إليهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 6:11; ﻣﺘﻰ 26:26; لُوقا 24:30"> فحالَما تَدخُلانِ المدينَةَ تَجِدانِه، قَبلَ أنْ يَصعَدَ إلى التَّلَّةِ لِيَأكُلَ، لأنَّ المَدعوِّينَ مِنَ الشَّعبِ لا يأكلونَ حتّى يَجيءَ ويُبارِكَ الذَّبيحةَ. فاَصعَدا الآنَ تَجِدانِه في الحالِ“.</span>
<span className="v" id="v14" data-v="14"> فصَعِدا إلى المدينةِ وفيما هُما يَدخلانِها كانَ صَموئيلُ يَخرُجُ مِنها لِيَصعَدَ إلى التَّلَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="اعمال الرسل 13:21; صَمُوئيلَ الأوَّلُ 15:1; أيُّوب 33:16"> وكانَ الرّبُّ أوحى إلى صَموئيلَ، قَبلَ أنْ يأتيَهُ شاوُلُ بِيومٍ، وقالَ لَه:</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 10:1; المَزامِير 106:44; الخُرُوج 3:7–9"> ”غدًا، في مِثلِ هذِهِ السَّاعةِ، أُرسِلُ إليكَ رَجُلًا مِنْ أرضِ بنيامّينَ، فاَمسَحْهُ رئيسًا على شعبي إِسرائيلَ، فيُخَلِّصَ شعبي مِن أيدي الفلِسطيِّينَ، لأنِّي نَظَرتُ إلى آلامِ شعبي وسَمِعتُ صُراخَهُم“.</span>
<span className="v" id="v17" data-v="17" data-xref="هُوشَع 13:11; صَمُوئيلَ الأوَّلُ 16:6–12; نَحَمْيا 13:25"> فلمَّا رأى صَموئيلُ شاولَ قالَ لَه الرّبُّ: ”ها هوَ الرَّجُلُ الّذي كلَّمتُكَ عَنهُ. هذا يَحكُمُ شعبي“.</span>
<span className="v" id="v18" data-v="18"> فَدَنا شاوُلُ مِنْ صَموئيلَ، وهوَ في بابِ المدينةِ، وقالَ لَه ”أخبِرني أينَ بَيتُ الرَّائي؟“</span>
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:25; يُوحَنّا 4:29"> فأجابَهُ صَموئيلُ: ”أنا هوَ الرَّائي، فاَصعَدا أمامي إلى التَّلَّةِ وكُلا اليومَ معي، وفي الغَدِ أخبِرُكُما عَنْ كُلِّ ما في قلبَيكُما، وأصرِفُكما.</span>
<span className="v" id="v20" data-v="20" data-xref="صَمُوئيلَ الأوَّلُ 8:5; صَمُوئيلَ الأوَّلُ 12:13; صَمُوئيلَ الأوَّلُ 9:3"> أمَّا الحميرُ الّتي ضَلَّت لكَ مُنذُ ثلاثَةِ أيّامٍ، فلا يَنشَغِلْ بالُكَ علَيها لأنَّها وُجِدَت. والآنَ لِمَنْ ستكونُ كُلُّ ثَروةٍ في إِسرائيلَ إنْ لم تَكُنْ لكَ ولِكُلِّ بَيتِ أبيكَ؟“</span>
<span className="v" id="v21" data-v="21" data-xref="صَمُوئيلَ الأوَّلُ 15:17; القُضاة 20:46–48; المَزامِير 68:27"> فأجابَهُ شاوُلُ: ”أنا بنيامينيٌّ مِنْ أصغَرِ أسباطِ بَني إِسرائيلَ، وقبيلَتي أصغَرُ قَبائِلِها، فكيفَ تقولُ لي مِثلَ هذا الكلامِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 43:32; لُوقا 14:10"> فأخَذَ صَموئيلُ شاولَ وخادِمَهُ ودخَلَ بِهِما المَجلِسَ، وأجلَسَهُما في صَدْرِ المَدعوِّينَ، وهم نَحوَ ثَلاثينَ رَجُلًا،</span>
<span className="v" id="v23" data-v="23" data-xref="صَمُوئيلَ الأوَّلُ 1:5; التَّكوين 43:34"> وقالَ لِلطَّباخِ: ”أعطِني الحِصَّةَ الّتي سَلَّمتُها إليكَ وقُلتُ لكَ: إحفَظْها عِندَكَ“.</span>
<span className="v" id="v24" data-v="24" data-xref="حِزْقِيال 24:4; اللّاويِّين 7:32–33; الخُرُوج 29:22"> فأخذَ الطبَّاخُ السَّاقَ والإليَةَ ووَضَعَهُما أمامَ شاولَ، فقالَ صَموئيلُ لِشاوُلَ: ”ضَعْ أمامَكَ هذا الّذي بقيَ وكُلْ، فأنا حَفِظْتُه لكَ إلى هذا الوقتِ لِتأكُلَهُ معَ الّذينَ دَعوتُهم مِنَ الشَّعبِ“. فأكَلَ شاوُلُ معَ صَموئيلَ في ذلِكَ اليومِ.</span>
<span className="v" id="v25" data-v="25" data-xref="اعمال الرسل 10:9; التَّثْنِيَة 22:8; صموئيلَ الثّانِي 11:2"> ثُمَّ نَزَلا مِن التَّلَّةِ إلى المدينةِ، وتَحدَّثا طَويلًا على سَطحِ البَيتِ، حيثُ فَرَشوا لِشاولَ ونامَ.</span>
</div>
</div>

## صموئيل يمسح شاول ملكًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 44:4; يَشُوع 7:13; القُضاة 19:28"> وبَكَّرَ صَموئيلُ عِندَ طُلوعِ الفَجرِ، فَدَعا شاوُلَ عَنِ السَّطحِ وقالَ لَه: ”إنهَضْ، سأرافِقُكَ مَسافةً قَصيرةً“. فقامَ شاوُلُ وخرَجَ معَ صَموئيلَ.</span>
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 15:14–15; صَمُوئيلَ الأوَّلُ 20:38–39; المُلُوكِ الثّانِي 9:5–6"> وبَينَما هُما نازِلانِ عِندَ طَرَفِ المدينةِ، قالَ لَه صَموئيلُ: ”قُلْ لِخادِمِكَ أنْ يَسبُقَنا، وقِفْ أنتَ قليلًا فأُسمِعَكَ كلامَ اللهِ“. فسبَقَهُما الخادِمُ.</span>
