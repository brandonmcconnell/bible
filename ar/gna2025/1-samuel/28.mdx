<span className="v" id="v2" data-v="2" data-xref="ﺭﻭﻣﻴﺔ 12:9; صموئيلَ الثّانِي 16:16–19; صَمُوئيلَ الأوَّلُ 27:10"> فأجابَهُ داوُدُ: ”وأنتَ ستعلمُ ما يفعَلُ عبدُكَ“. فقالَ أخيشُ: ”إذًا، أجعلُكَ حاملَ سلاحي طُولَ الأيَّامِ“.</span>

## شاول عند العرافة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="اللّاويِّين 19:31; صَمُوئيلَ الأوَّلُ 25:1; اللّاويِّين 20:27"> وماتَ صَموئيلُ وناحَ علَيهِ جميعُ بَني إِسرائيلَ، ودفنوهُ في الرَّامةِ مدينتِهِ. وكانَ شاوُلُ نفى العرَّافينَ والسَّحرَةَ مِنَ البلادِ.</span>
<span className="v" id="v4" data-v="4" data-xref="صَمُوئيلَ الأوَّلُ 31:1; يَشُوع 19:18; المُلُوكِ الثّانِي 4:8"> فاَجتمعَ الفلِسطيُّونَ وعسكَروا في شُونَم. وجمَعَ شاوُلُ رجالَ إِسرائيلَ جميعَهُم وعسكَروا في جَلبوعَ.</span>
<span className="v" id="v5" data-v="5" data-xref="إشَعْياء 57:20–21; دانيال 5:6; إشَعْياء 21:3–4"> فلمَّا رأى شاوُلُ مُعسكَرَ الفلِسطيِّينَ خافَ واَضطربَ قلبُهُ جِدًّا.</span>
<span className="v" id="v6" data-v="6" data-xref="العَدَد 12:6; صَمُوئيلَ الأوَّلُ 14:37; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:9"> فسألَ الرّبَّ ماذا يفعَلُ، فلم يُجبْهُ الرّبُّ لا بالأحلامِ ولا بالعرافةِ ولا بالأنبياءِ.</span>
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 16:16; يَشُوع 17:11; أخبارِ الأيّامِ الأوّلُ 10:13"> فقالَ لمُعاونيهِ: ”فتِّشوا لي عَنْ عرَّافةٍ حتّى أذهبَ إليها وأستشيرَها“. فقالوا لَه: ”في عينَ دورٍ عرَّافةٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 8:19; أخبارِ الأيّامِ الأوّلُ 10:13; المُلُوكِ الأوَّلُ 22:30"> فتنكَّرَ شاوُلُ، ولبسَ ثيابًا غيرَ ثيابِهِ، وسارَ هوَ ورجلانِ معَهُ، وجاؤوا العرَّافةَ ليلًا. فقالَ لها: ”تكهَّني لي بالأرواحِ وأطْلِعي لي مَنْ أسمِّيهِ لكِ“.</span>
<span className="v" id="v9" data-v="9" data-xref="صَمُوئيلَ الأوَّلُ 28:3; المُلُوكِ الثّانِي 5:7; صموئيلَ الثّانِي 18:13"> فقالَت لَه: ”أما علمتَ ما فعَلَ شاوُلُ، كيفَ قطعَ العَرَّافينَ والسَّحَرةَ مِنَ البِلادِ؟ فلِماذا تنصِبُ لي شرَكًا لِتُهلكَني؟“</span>
<span className="v" id="v10" data-v="10" data-xref="صموئيلَ الثّانِي 14:11; صَمُوئيلَ الأوَّلُ 19:6; التَّثْنِيَة 18:10–12"> فحلَفَ لها شاوُلُ: ”حَيٌّ هوَ الرّبُّ لن يقَعَ علَيكِ ضَررٌ في هذا الأمرِ“.</span>
<span className="v" id="v11" data-v="11"> فسألَتْهُ العرَّافةُ: ”مَنْ أُطلِعُ لكَ؟“ قالَ: ”أطْلعي لي صَموئيلَ“.</span>
<span className="v" id="v12" data-v="12" data-xref="صَمُوئيلَ الأوَّلُ 28:3; المُلُوكِ الأوَّلُ 14:5"> فلمَّا رأتِ العرَّافةُ صَموئيلَ، صرَخت صرخَةً عاليةً وقالَت لِشاوُلَ: ”لماذا خدَعْتَني أنتَ شاوُلُ؟“</span>
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 82:6–7; يُوحَنّا 10:34–35; الخُرُوج 22:28"> فقالَ لها: ”لا تخافي. ماذا رأيتِ؟“ فقالَت: ”رأيتُ رُوحًا يَطلُعُ مِنَ الأرضِ“.</span>
<span className="v" id="v14" data-v="14" data-xref="صَمُوئيلَ الأوَّلُ 15:27; المُلُوكِ الثّانِي 2:8; صَمُوئيلَ الأوَّلُ 24:8"> فقالَ لها: ”ما هيئَتُهُ؟“ قالَت: ”رَجلٌ شيخٌ يرتدي جُبَّةً“. فعرَفَ شاوُلُ أنَّهُ صَموئيلُ، فسجَدَ على وجهِهِ إلى الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="لُوقا 16:23–26; صَمُوئيلَ الأوَّلُ 28:6; صَمُوئيلَ الأوَّلُ 18:12"> فقالَ صَموئيلُ لشاوُلَ: ”لماذا أزْعجتَني وأطلَعتَني؟“ فأجابَهُ شاوُلُ: ”أنا في ضيقٍ شديدٍ، لأنَّ الفلِسطيِّينَ يُحاربونَني واللهُ أهمَلَني وما عادَ يُجيبُني لا بالأنبياءِ ولا بالأحلامِ، فدَعوتُكَ لتُخبِرَني بما أفعَلُ</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الثّانِي 6:27; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:5; المَزامِير 68:1–3"> فقالَ صَموئيلُ: ”لماذا تسألُني، والرّبُّ أهملَكَ وصارَ عدوَّكَ</span>
<span className="v" id="v17" data-v="17" data-xref="صَمُوئيلَ الأوَّلُ 15:27–29; صَمُوئيلَ الأوَّلُ 16:13; صَمُوئيلَ الأوَّلُ 13:13–14"> وهوَ فعَلَ بكَ كما قالَ على لساني، واَنتزعَ المَملكةَ مِنْ يَدِكَ وأعطاها لصاحبِكَ داوُدَ.</span>
<span className="v" id="v18" data-v="18" data-xref="صَمُوئيلَ الأوَّلُ 15:9; المُلُوكِ الأوَّلُ 20:42; المَزامِير 50:21–22"> فأنتَ ما أطَعتَ أمرَ الرّبِّ وما شَفيتَ غليلَ غضَبِهِ في عماليقَ، لذلِكَ فعَلَ بكَ اليومَ ما فعَلَ.</span>
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 5:5; إرْمِيا 28:16–17; المُلُوكِ الأوَّلُ 22:20"> كذلِكَ سيُسلِّمُ الرّبُّ بَني إِسرائيلَ معَكَ إلى أيدي الفلِسطيِّينَ، وغَدًا تكونونَ معي أنتَ وبَنوكَ، وأيضًا جيشُ إِسرائيلَ يُسلِّمُهُ الرّبُّ إلى أيدي الفلِسطيِّينَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="صَمُوئيلَ الأوَّلُ 28:5; أيُّوب 26:2; أيُّوب 15:20–24"> فسقطَ شاوُلُ في الحالِ بطولِهِ على الأرضِ واَرتعبَ مِنْ كلامِ صَموئيلَ، وما عادت بهِ قوَّةٌ لأنَّهُ لم يَذُقْ طَعامًا كُلَّ يومِهِ وليلتِهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="القُضاة 12:3; أيُّوب 13:14; صَمُوئيلَ الأوَّلُ 19:5"> فتَقدَّمَت إليهِ العرَّافةُ ورأتْهُ في خوفٍ شديدٍ، فقالَت لَه: ”أنا يا سيِّدي أطعْتُ أمرَكَ، وخاطرْتُ بنفْسي، وسَمعتُ لِكلامِكَ الّذي كلَّمْتني بهِ.</span>
<span className="v" id="v22" data-v="22"> فاَسمعْ أنتَ الآنَ لكلامي، فأُقدِّمَ لكَ كِسرَةَ خُبزٍ وتأكلَ، فتَقوى على السَّيرِ في الطَّريقِ“.</span>
<span className="v" id="v23" data-v="23" data-xref="المُلُوكِ الأوَّلُ 21:4; لُوقا 24:29; لُوقا 14:23"> فرفضَ وقالَ: ”لا آكلُ“. فألحَّ علَيهِ مُعاوِناهُ والعَرَّافةُ أيضًا، فسمِعَ لهُم وقامَ عنِ الأرضِ وجلسَ على السَّريرِ.</span>
<span className="v" id="v24" data-v="24" data-xref="لُوقا 15:23; التَّكوين 18:7–8"> وكانَ للعرَّافةِ في البيتِ عِجْلٌ مُسمَّنٌ، فأسرَعَت وذبحَتهُ وأخَذَت دقيقًا وعجَنتهُ وخَبزتهُ فطيرًا</span>
<span className="v" id="v25" data-v="25"> وقدَّمت إلى شاوُلَ ومُعاونيهِ فأكلوا، ثُمَّ قاموا واَنصرَفوا في تلكَ اللَّيلةِ.</span>
