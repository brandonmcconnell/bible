## يوآحاز ملك إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المُلُوكِ الثّانِي 11:21; المُلُوكِ الثّانِي 8:26; المُلُوكِ الثّانِي 11:4"> في السَّنةِ الثَّالثةِ والعِشرينَ ليوآشَ مَلِكِ يَهوذا، ملَكَ يوآحازُ بنُ ياهو على إِسرائيلَ بِالسَّامِرةِ سَبعَ عَشْرةَ سنَةً.</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الأوَّلُ 12:26–33; المُلُوكِ الأوَّلُ 14:16; المُلُوكِ الثّانِي 10:29"> وفعَلَ الشَّرَّ في نظَرِ الرّبِّ، وما حادَ عَنْ خطايا يَرُبعامَ بنِ ناَباطَ الّذي جعَلَ شعبَ إِسرائيلَ يَخطَأُ،</span>
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الأوَّلُ 19:17; المُلُوكِ الثّانِي 13:24–25; القُضاة 2:14"> فاَشتَدَّ غضَبُ الرّبِّ على شعبِ إِسرائيلَ وأسلَمَهُم إلى يَدِ حَزائيلَ مَلِكِ أرامَ وبَنهَدَدَ اَبنِهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الثّانِي 14:26; الخُرُوج 3:7; المَزامِير 78:34"> فتَضَرَّعَ يوآحازُ إلى الرّبِّ، فسَمِعَ لَه لأنَّهُ رأى كم جارَ مَلِكُ أرامَ على مَملكَةِ إِسرائيلَ.</span>
<span className="v" id="v5" data-v="5" data-xref="نَحَمْيا 9:27; المُلُوكِ الثّانِي 14:25; المُلُوكِ الثّانِي 14:27"> وأعطى الرّبُّ بَني إِسرائيلَ مُنقِذًا فحَرَّرَهُم مِنْ سيطرةِ الأراميِّينَ، وأقاموا في ديارِهِم بِسلامٍ كما كانَ مِنْ قَبلُ.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 16:33; المُلُوكِ الثّانِي 13:2; التَّثْنِيَة 7:5"> إلَّا أنَّهُم ما حادوا عَنْ خطايا بَيتِ يَرُبعامَ الّذي خَطِئَ وجعَلَ شعبَ إِسرائيلَ يَخطَأُ، بل ثابَروا علَيها وبَقيَ تِمثالُ أشيرةَ قائِمًا في السَّامِرةِ.</span>
<span className="v" id="v7" data-v="7" data-xref="عامُوس 1:3; المُلُوكِ الثّانِي 10:32; المُلُوكِ الأوَّلُ 20:15"> ولم يَبقَ ليوآحازَ سِوى خمسينَ فارِسًا وعَشْرِ مَركباتٍ وعشَرَةِ آلافِ رَجُلٍ، لأنَّ مَلِكَ أرامَ أبادَ كُلَّ شيءٍ آخَرَ وجعَلَهُ مِثلَ التُّرابِ الّذي يُداسُ.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الأوَّلُ 11:4; المُلُوكِ الأوَّلُ 14:29; المُلُوكِ الأوَّلُ 14:31"> وما بَقيَ مِنْ أخبارِ يوآحازُ وبَسالَتِه وأعمالِهِ مُدوَّنٌ في سِفْرِ أخبارِ الأيّامِ لِمُلوكِ إِسرائيلَ.</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الثّانِي 10:35; المُلُوكِ الثّانِي 13:13; المُلُوكِ الثّانِي 14:8"> وماتَ يوآحازَ ودُفِنَ معَ آبائِهِ في السَّامِرةِ، وملَكَ يوآشُ اَبنُهُ مكانَهُ.</span>
</div>

## يوآش يخلف يوآحاز

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10"> في السَّنةِ السَّابِعةِ والثَّلاثينَ ليوآشَ مَلِكِ يَهوذا، ملَكَ يوآشُ بنُ يوآحازَ على إِسرائيلَ بِالسَّامِرةِ سِتَّ عَشْرَةَ سنَةً.</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 13:2; المُلُوكِ الثّانِي 10:29; المُلُوكِ الثّانِي 3:3"> وفعَلَ الشَّرَّ في نظَرِ الرّبِّ، وما حادَ عَنْ خطايا يَرُبعامَ بنِ ناَباطَ الّذي خَطِئَ وجعَلَ شعبَ إِسرائيلَ يَخطَأُ.</span>
<span className="v" id="v12" data-v="12" data-xref="المُلُوكِ الثّانِي 13:14–25; المُلُوكِ الثّانِي 14:8–16; أخبارِ الأيّامِ الثّانِي 25:17–24"> وما بَقيَ مِنْ أخبارِ يوآشَ وبَسالَتِه وأعمالِه، وحُروبِهِ ضدَّ أمَصْيا مَلِكِ يَهوذا، مُدوَّنٌ في سِفْرِ أخبارِ الأيّامِ لِمُلوكِ إِسرائيلَ.</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الأوَّلُ 1:21; المُلُوكِ الثّانِي 14:28–29; المُلُوكِ الأوَّلُ 2:10"> وماتَ يوآشُ ودُفِنَ في السَّامِرةِ معَ مُلوكِ إِسرائيلَ، وجلَسَ يَرُبعامُ اَبنُهُ مكانَهُ.</span>
</div>
</div>

## موت النبي أليشع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الثّانِي 2:12; فِيلِبِّي 2:26; التَّكوين 48:1"> وسقَطَ أليشَعُ في مرضٍ أدَّى إلى موتِهِ فيما بَعدُ، فقدِمَ إليهِ يوآشُ مَلِكُ إِسرائيلَ، وقالَ: ”يا أبي يا أبي، يا حامي حِمى إِسرائيلَ“.</span>
<span className="v" id="v15" data-v="15"> فقالَ لَه أليشَعُ: ”خُذْ قوسًا وسِهامًا“، فأخذَ قوسًا وسِهامًا.</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الثّانِي 4:34; التَّكوين 49:24; المَزامِير 144:1"> ثُمَّ قالَ لَه: ”ضَعْ يدَكَ على القوسِ“، فوضَعَ يدَهُ. ووضَعَ أليشَعُ يدَهُ على يَدِ المَلِكِ</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الأوَّلُ 20:26; المُلُوكِ الثّانِي 5:10–14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:18"> وقالَ لَه: ”إفتَحِ النَّافِذَةَ مِنْ جِهَةِ الشَّرقِ“، ففَتَحَها. ثُمَّ قالَ: ”إرمِ السَّهْمَ“، فرَماهُ. فقالَ أليشَعُ: ”هذا سَهمُ نَصْرٍ مِنَ الرّبِّ سَهمُ نَصْرٍ على أرامَ، ستَضرِبُ الأراميِّينَ في أفيقَ حتّى تُبيدَهُم“.</span>
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الثّانِي 4:6; الخُرُوج 17:11; إشَعْياء 20:2–4"> ثُمَّ قالَ لِلمَلِكِ: ”خُذِ السِّهامَ“. فأخذَها. وقالَ لَه: ”إرمِ بِها إلى الأرضِ“، فرمى ثَلاثَةَ سِهامٍ وتوقَّفَ.</span>
<span className="v" id="v19" data-v="19" data-xref="المُلُوكِ الثّانِي 13:25; العَدَد 16:15; المُلُوكِ الثّانِي 4:40"> فغَضِبَ علَيهِ رَجُلُ اللهِ وقالَ لَه: ”لو رميتَ خمسةَ سِهامٍ أو ستَّةً، لكُنتَ ضربْتَ الأراميِّينَ حتّى أبَدتَهُم. لكِنِ الآنَ تَهزِمُهُم فقط ثَلاثَ مرَّاتٍ“.</span>
<span className="v" id="v20" data-v="20" data-xref="المُلُوكِ الثّانِي 24:2; المُلُوكِ الثّانِي 3:7; المُلُوكِ الثّانِي 6:23"> ثُمَّ ماتَ أليشَعُ ودُفِنَ. وهجَمَ غُزاةٌ مِنَ الموآبيِّينَ على أرضِ إِسرائيلَ كعادَتِهم أوَّلَ الرَّبيعِ</span>
<span className="v" id="v21" data-v="21" data-xref="حِزْقِيال 37:1–10; اعمال الرسل 19:12; يُوحَنّا 5:28–29"> وفي ذلِكَ الحينِ كانَ بَعضُ الإسرائيليِّينَ يَقبِرونَ مَيْتًا، فلمَّا رأوا الغُزاةَ رمَوا المَيْتَ في قبرِ أليشَعَ وهرَبوا. فلمَّا مَسَّ عِظامَ أليشَعَ عاشَ وقامَ على قدَمَيهِ.</span>
</div>
</div>

## موت حزائيل ملك أرام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="المُلُوكِ الثّانِي 8:12; المُلُوكِ الثّانِي 13:3–7; المَزامِير 106:40–42"> واَضطَهَدَ حَزائيلُ مَلِكُ أرامَ شعبَ إِسرائيلَ طُولَ عَهدِ المَلِكِ يوآحازَ</span>
<span className="v" id="v23" data-v="23" data-xref="المَزامِير 105:8; المُلُوكِ الثّانِي 14:27; ميخا 7:18–20"> فتَحَنَّنَ الرّبُّ علَيهِم ورَحِمَهُم وعطَفَ علَيهِم لِعَهدِهِ الّذي قطَعَهُ معَ إبراهيمَ وإسحَقَ ويعقوبَ، ولم يشَأْ أنْ يُبيدَهُم فأبقاهُم إلى الآنَ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="المَزامِير 125:3; لُوقا 18:7"> ثُمَّ ماتَ حَزائيلُ مَلِكُ أرامَ، وجلَسَ بَنهَدَدُ اَبنُهُ مكانَهُ،</span>
<span className="v" id="v25" data-v="25" data-xref="المُلُوكِ الثّانِي 13:18–19; المُلُوكِ الثّانِي 10:32; المُلُوكِ الثّانِي 14:25"> فعادَ يوآشُ بنُ يوآحازَ وهزَمَ بَنهَدَدَ ثَلاثَ مرَّاتٍ واَستَرَدَّ مِنهُ مُدُنَ إِسرائيلَ الّتي أخذَها مِنْ يوآحازَ أبيهِ في الحربِ.</span>
</div>
