## مقتل بيت أخاب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="القُضاة 8:30; المُلُوكِ الأوَّلُ 13:32; المُلُوكِ الأوَّلُ 21:8–14"> وكانَ لأخابَ سَبعونَ اَبنًا في السَّامِرةِ. فكتَبَ ياهو إلى حُكَّامِ المدينةِ وأعيانِها، وإلى مُتَعَهِّدي تربيةِ أبناءِ أخابَ وأحفادِهِ يقولُ:</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الثّانِي 5:6"> ”عِندَكُمُ الآنَ بَنو سيِّدِكُم والمَراكِبُ والخَيلُ والمُدُنُ المُحَصَّنَةُ والسِّلاحُ،</span>
<span className="v" id="v3" data-v="3" data-xref="صموئيلَ الثّانِي 2:8–9; المُلُوكِ الأوَّلُ 12:20–21; صموئيلَ الثّانِي 2:12–17"> فاَنتَخِبوا عِندَ وصولِ رِسالتي هذِهِ الأفضَلَ والأصلَحَ مِنهُم وأجلِسوهُ على عرشِ أبيه، واَستَعِدُّوا لِلقِتالِ دِفاعًا عَنْ بَيتِ أخابَ“.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الثّانِي 9:24; المُلُوكِ الثّانِي 9:27; إشَعْياء 27:4"> فاَسْتَولى علَيهِم الخوفُ وقالوا: ”مَلِكانِ لم يَصمُدا أمامَهُ، فكيفَ نحنُ؟“</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الأوَّلُ 20:32; المُلُوكِ الأوَّلُ 20:4; يَشُوع 9:11"> فأرسَلوا وكيلَ القصرِ وحاكِمَ المدينةِ والشُّيوخَ والمُرَبِّينَ إلى ياهو يقولونَ: ”نحنُ عبيدُكَ، وكُلُّ ما تَأْمُرُنا بهِ نعمَلُهُ. لا نُقيمُ أحدًا مَلِكًا، وما تراهُ حسَنًا فاَعمَلْهُ“.</span>
<span className="v" id="v6" data-v="6" data-xref="رُؤيا يُوحَنّا 2:20–23; التَّثْنِيَة 5:9; العَدَد 25:4"> فكتَبَ إليهِم ثانيةً يقولُ: ”إنْ كُنتُم لي ومِنَ المُطيعينَ لأمري، فاَقْطَعوا رُؤوسَ بَني مَلِكِكُم واَحمِلوها إليَّ في مِثلِ هذِهِ السَّاعةِ مِنْ غدٍ إلى يَزرَعيلَ“. وكانَ بَنو المَلِكِ سَبعينَ رَجُلًا عِندَ أعيانِ المدينةِ الّذينَ رَبَّوهُم.</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الأوَّلُ 21:21; المُلُوكِ الثّانِي 11:1; المُلُوكِ الثّانِي 10:9"> فلمَّا وصَلَت رِسالةُ ياهو إليهِم أخذوا بَني المَلِكِ السَّبعينَ وذبَحوهُم ووضَعوا رُؤُوسَهُم في سِلالٍ وأرسَلوها إليهِ في يَزرَعيلَ</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 21:23; صموئيلَ الثّانِي 11:18–21; مَرْقُس 6:28"> فجاءَ الرَّسولُ إليهِ وقالَ: ”جاؤوا بِرُؤوسِ بَني المَلِكِ أخابَ“. فقالَ: ”إجعَلوها كومَتَينِ عِندَ مَدخَلِ المدينةِ إلى الصَّباحِ“.</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الثّانِي 9:14–24; إشَعْياء 5:3; هُوشَع 1:4"> فلمَّا جاءَ الصَّباحُ خرَجَ الشَّعبُ وقالَ: ”أنتُم أبرياءُ، أنا الّذي ثارَ على المَلِكِ يورامَ وقتلَهُ، لكِنْ مَنِ الّذي قتَلَ هؤلاءِ</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الأوَّلُ 21:29; المُلُوكِ الأوَّلُ 21:19; المُلُوكِ الثّانِي 9:7–10"> فاَعلَموا الآنَ أنْ لا شَيءَ مِنْ كلامِ الرّبِّ على بَيتِ أخابَ إلَّا ويَصيرُ. وهوَ إنَّما فعَلَ ما تكَلَّمَ بهِ على لِسانِ عَبدِهِ إيليَّا“.</span>
<span className="v" id="v11" data-v="11" data-xref="أيُّوب 18:19; المُلُوكِ الأوَّلُ 18:19; المُلُوكِ الأوَّلُ 22:6"> ثُمَّ قتَلَ ياهو جميعَ الباقينَ مِنْ بَيتِ أخابَ في يَزرَعيلَ وجميعَ رِجالِهِ ومَعارِفَهُ وكهنَتَهُ، ولم يُبقِ مِنهُم أحدًا.</span>
</div>
</div>

## مقتل أنسباء أخزيا وأخاب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12"> ثُمَّ ذهَبَ ياهو إلى السَّامِرةِ. فلمَّا وصَلَ إلى مكانٍِ يُدعى مَحلَّةَ الرُّعاةِ،</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الثّانِي 8:29; المُلُوكِ الثّانِي 8:24; أخبارِ الأيّامِ الثّانِي 21:17"> صادَفَ جماعةً مِنْ أنسباءِ أخَزْيا مَلِكِ يَهوذا، فسألَهُم: ”مَنْ أنتُم؟“ فأجابوا ”نحنُ مِنْ أنسباءِ أخَزْيا، جِئْنا لِنُسَلِّمَ على بَني المَلِكِ يورامَ وبَني المَلِكَةِ إيزابَلَ“.</span>
<span className="v" id="v14" data-v="14" data-xref="أخبارِ الأيّامِ الثّانِي 22:10; المُلُوكِ الثّانِي 10:6; المُلُوكِ الثّانِي 10:10–11"> فقالَ ياهو لِرِجالِهِ: ”إقبِضوا علَيهِم“ فقبَضوا علَيهِم وذبَحوهُم على حافَّةِ بِئرٍ هُناكَ، وكانوا اَثنَينِ وأربَعينَ رَجُلًا.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 35:14–19; عَزْرا 10:19; حِزْقِيال 17:18"> وسارَ ياهو مِنْ هُناكَ، فلَقيَ يونادابَ بنَ ركابٍ قادِمًا لاَستِقبالِهِ، فبادَرَهُ ياهو بِالسَّلامِ وسألَهُ: ”هل أنتَ مُخلِصٌ لي إخلاصي لكَ؟“ فأجابَهُ: ”نعم“. فقالَ ياهو: ”هاتِ يَدَكَ“. فناوَلَهُ يَدَهُ. فأصعَدَهُ على مَركبتِهِ</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الأوَّلُ 19:10; ﻣﺘﻰ 6:5; المُلُوكِ الثّانِي 10:31"> وقالَ لَه: ”تَعالَ معي وتُشاهِدُ تَعَلُّقي الشَّديدَ بِالرّبِّ“. وأركبَهُ معَهُ في مَركبتِهِ.</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الثّانِي 9:8; أخبارِ الأيّامِ الثّانِي 22:8; مَلاخِي 4:1"> وجاءَ السَّامِرةَ، فقتَلَ جميعَ مَنْ بَقيَ لأخابَ هُناكَ وأبادَهُم، كما قالَ الرّبُّ لإيليَّا.</span>
</div>
</div>

## مقتل عباد البعل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الأوَّلُ 16:31–32; المُلُوكِ الثّانِي 3:2; ﺭﻭﻣﻴﺔ 3:8"> ثُمَّ جمَعَ ياهو الشَّعبَ وقالَ لهُم: ”عبَدَ أخابُ البَعلَ قليلًا، لكِنْ ياهو سيَعبُدُهُ كثيرًا.</span>
<span className="v" id="v19" data-v="19" data-xref="المُلُوكِ الأوَّلُ 22:6; المُلُوكِ الأوَّلُ 18:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3"> فاَدعوا إليَّ جميعَ أنبياءِ البَعلِ وعُبَّادِهِ وكهَنتِهِ، دونَ أنْ يتَخَلَّفَ مِنهُم أحدٌ، لأنَّ لي ذبيحةً عظيمةً لِلبَعلِ، وكُلُّ مَنْ يتَخَلَّفُ يُقتَلُ“. وكانَ ذلِكَ مكيدةً مِنْ ياهو لِيُهلِكَ عُبَّادَ البَعلِ.</span>
<span className="v" id="v20" data-v="20" data-xref="يُوئيل 1:14; الخُرُوج 32:5; المُلُوكِ الأوَّلُ 18:19–20"> ثُمَّ قالَ ياهو: ”نادوا بِاَجتِماعٍ مُقَدَّسٍ لِلبَعلِ“. فنادوا بهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="المُلُوكِ الأوَّلُ 16:32; المُلُوكِ الثّانِي 11:18; يُوئيل 3:2"> وأذاعَ ياهو النِّداءَ في كُلِّ أرضِ إِسرائيلَ فأقبلَ جميعُ العُبَّادِ ودخلوا بَيتَ البَعلِ، فاَمتَلأَ بِهِم.</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 28:2; ﻣﺘﻰ 22:11–12"> فقالَ ياهو لِلقَيِّمِ على ثِيابِ العِبادةِ: ”أخرِجْ ملابِسَ لِجميعِ عُبَّادِ البَعلِ“، ففَعلَ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 13:30; ﻣﺘﻰ 25:32–33; ﻣﺘﻰ 13:41"> ودخَلَ ياهو معَ يهونادابَ بنِ رَكابٍ إلى بَيتِ البَعلِ وقالَ لِلعُبَّادِ: ”أُنظُروا إنْ كانَ أحدٌ مِنْ عِبادِ الرّبِّ بَينكُم هُنا فهُنا لا يجوزُ أنْ يكونَ إلَّا عُبَّادُ البَعلِ“.</span>
<span className="v" id="v24" data-v="24" data-xref="المُلُوكِ الأوَّلُ 20:30–42"> ثُمَّ دخَلَ ليُقَدِّمَ ذبائِحَ ومُحرَقاتٍ، وبَعدَ أنْ أقامَ لَه خارِجًا ثَمانينَ رَجُلًا، قالَ لهُم: ”إذا نجا أحدٌ مِنْ هؤُلاءِ الّذينَ جِئتُ بِهِم إلى هُنا تؤْخَذُ أنفُسُكُم بَدَلًا مِنهُ“.</span>
<span className="v" id="v25" data-v="25" data-xref="رُؤيا يُوحَنّا 16:6–7; حِزْقِيال 22:21–22; المُلُوكِ الأوَّلُ 18:40"> ثُمَّ دخَلَ لِيُقَدِّمَ ذبائِحَ ومُحرَقاتٍ فلمَّا فرَغَ مِنْ تَقديمِ المُحرَقةِ، قالَ لِلحرَسِ والضُبَّاطِ: ”أُدخُلوا واَقتُلوهُم ولا تَدَعوا أحدًا يَفلِتُ“. فضَرَبوهُم بِحَدِّ السَّيفِ وطَرَحوهُم خارجًا، ثُمَّ دخَلوا إلى مِحرابِ بَيتِ البَعلِ</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الأوَّلُ 14:23; المُلُوكِ الثّانِي 3:2; المُلُوكِ الثّانِي 19:18"> وأخرَجوا التَّماثيلَ وأحرَقوها.</span>
<span className="v" id="v27" data-v="27" data-xref="دانيال 3:29; عَزْرا 6:11; دانيال 2:5"> وكسَّروا تِمثالَ البَعلِ، وهدَموا بَيتَهُ وجعَلوهُ مَزبَلَةً إلى هذا اليومِ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28"> وهكذا قضى ياهو على عِبادةِ البَعلِ مِنْ إِسرائيلَ،</span>
<span className="v" id="v29" data-v="29" data-xref="المُلُوكِ الأوَّلُ 12:28–30; المُلُوكِ الأوَّلُ 14:16; المُلُوكِ الأوَّلُ 13:33–34"> إلَّا أنَّهُ ما حادَ عَنْ خطايا يَرُبعامَ بنِ ناَباطَ الّذي جعَلَ شعبَ إِسرائيلَ يَخطَأُ، فأبقى على عِجْلَي الذَّهَبِ اللَّذينِ في بَيتَ إيلَ وفي دانَ.</span>
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الثّانِي 10:35; المُلُوكِ الثّانِي 13:1; المُلُوكِ الثّانِي 13:10"> فقالَ الرّبُّ لِياهو: ”عَمِلتَ ما هوَ قويمٌ في نظَري فأحسَنتَ، وكُلُّ ما نَوَيتُهُ في قلبي فعَلْتَهُ بِبَيتِ أخابَ، لِذلِكَ سَيجلِسُ مِنْ بَنيكَ إلى الجيلِ الرَّابِعِ على عرشِ إِسرائيلَ“.</span>
<span className="v" id="v31" data-v="31" data-xref="المُلُوكِ الثّانِي 10:29; الأمثال 4:23; العِبرانِيّين 2:1"> ولم يَتَمَسَّكْ ياهو بِشريعةِ الرّبِّ إلهِ إِسرائيلَ بِكُلِّ قلبِهِ، ولم يَحِدْ عَنْ خَطايا يَرُبعامَ الّذي جعَلَ شعبَ إِسرائيلَ يَخطَأُ.</span>
</div>
</div>

## موت ياهو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="المُلُوكِ الثّانِي 8:12; المُلُوكِ الأوَّلُ 19:17; المُلُوكِ الثّانِي 13:22"> وفي تِلكَ الأيّامِ بدأَ الرّبُّ يَقتَطِعُ مِنْ أرضِ إِسرائيلَ، فاَحتَلَّ حَزائيلُ مَلِكُ أرامَ جميعَ المَناطِقِ</span>
<span className="v" id="v33" data-v="33" data-xref="عامُوس 1:3–4; التَّثْنِيَة 2:36; يَشُوع 13:9–12"> شرقيَّ الأردُنِّ إلى مدينةِ عَروعيرَ ونهرِ أرنونَ جَنوبًا، ومِنها جميعُ أراضي جلعادَ وباشانَ الّتي سكنَها بَنو جادَ ورَأوبينَ ومنَسَّى.</span>
<span className="v" id="v34" data-v="34" data-xref="المُلُوكِ الأوَّلُ 11:41; المُلُوكِ الأوَّلُ 14:29; المُلُوكِ الأوَّلُ 14:19"> وما بَقيَ مِنْ أخبارِ ياهو وأعمالِهِ وبَسالتِهِ مُدوَّنٌ في سِفْرِ أخبارِ الأيّامِ لِمُلوكِ إِسرائيلَ.</span>
<span className="v" id="v35" data-v="35" data-xref="المُلُوكِ الأوَّلُ 14:20; المُلُوكِ الأوَّلُ 14:31; المُلُوكِ الأوَّلُ 1:21"> وماتَ ياهو ودُفِنَ معَ آبائِهِ في السَّامِرةِ، وملَكَ يَهوآحازَ اَبنُهُ مكانَهُ.</span>
<span className="v" id="v36" data-v="36"> وكانَت أيّامُ مُلْكِهِ بِالسَّامِرةِ ثَمانيَ وعشرينَ سنَةً.</span>
