## مرض حزقيّا وشفاؤه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 38:1–22; أخبارِ الأيّامِ الثّانِي 32:24–26; فِيلِبِّي 2:27"> وفي تِلكَ الأيّامِ مَرِضَ حَزْقِيَّا وأشرَفَ على الموتِ، فجاءَهُ إشعيا بنُ آموصَ النَّبيُّ وقالَ لَه: ”هذا ما يقولُ الرّبُّ: ضَعْ وَصيَّتَكَ لأهلِ بَيتِكَ لأنَّكَ ستموتُ“.</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 50:15; ﻣﺘﻰ 6:6; إشَعْياء 38:2–3"> فأدارَ حَزْقِيَّا وجهَهُ إلى الحائِطِ وصلَّى إلى الرّبِّ</span>
<span className="v" id="v3" data-v="3" data-xref="نَحَمْيا 13:14; نَحَمْيا 13:22; نَحَمْيا 5:19"> قالَ: ”أُذكُرْ يا ربُّ أنِّي أمامَكَ سِرْتُ بِالحَقِّ وبِسلامةِ القلبِ، وأنِّي عَمِلْتُ الخيرَ أمامَكَ“. وبكى حَزْقِيَّا بُكاءً مُرًّا.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الثّانِي 22:14; المُلُوكِ الأوَّلُ 7:8"> وما إنْ خرَجَ إشعيا مِنْ الدَّارِ الوُسطى حتّى قالَ لَه الرّبُّ:</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 66:19–20; إشَعْياء 38:5; صَمُوئيلَ الأوَّلُ 9:16"> ”إرجِعْ إلى حَزْقِيَّا رئيسِ شعبي وقُلْ لَه: هذا ما يقولُ لكَ الرّبُّ إلهُ داوُدَ جَدِّكَ: سَمِعْتُ صلاتَكَ ورَأيتُ دُموعَكَ، وها أنا أشفيكَ، وبَعدَ غدٍ تَصعَدُ إلى هَيكلِ الرّبِّ.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الثّانِي 19:34; اعمال الرسل 27:24; إشَعْياء 10:24"> وسأُطيلُ عُمرَكَ خمسَ عَشْرَةَ سنَةً، وأُنقِذُكَ مِنْ يَدِ مَلِكِ أشُّورَ وأحمي هذِهِ المدينةَ مِنْ أجلي ومِنْ أجلِ داوُدَ عبدي“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 38:21; المُلُوكِ الثّانِي 2:20–22; المُلُوكِ الثّانِي 4:41"> فعادَ إشعيا إلى المَلِكِ وقالَ: ”خُذوا قُرْصَ تينٍ وضَعوهُ على القُرْحِ ليَشفى“.</span>
<span className="v" id="v8" data-v="8" data-xref="القُضاة 6:17; إشَعْياء 7:11; المُلُوكِ الثّانِي 19:29"> وقالَ حَزْقِيَّا لإشعيا: ”ما العلامةُ أنَّ الرّبَّ سيَشْفيني، فأصعَدُ بَعدَ غدٍ إلى هَيكلِ الرّبِّ؟“</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 38:7–8; مَرْقُس 8:11–12; ﻣﺘﻰ 16:1–4"> فقالَ إشعيا: ”هذِهِ هيَ العلامةُ أنَّ الرّبَّ يُحَقِّقُ ما قالهُ: ما تختارُ أنْ يتَقَدَّمَ الظِلُّ عَشْرَ دَرَجاتٍ أم يَرجِعُ عَشْرَ دَرَجاتٍ؟“</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الثّانِي 3:18; المُلُوكِ الثّانِي 2:10; يُوحَنّا 14:12"> فأجابَ حَزْقِيَّا: ”مِنَ السَّهلِ أنْ يتَقَدَّمَ الظِلُّ عَشْرَ دَرَجاتٍ، فدَعْهُ يَرجِعُ عَشْرَ دَرَجاتٍ“.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 38:8; يَشُوع 10:12–14; أخبارِ الأيّامِ الثّانِي 32:24"> فصلَّى إشعيا إلى الرّبِّ فتَراجَعَ الظِلُّ إلى الوراءِ عَشْرَ دَرجاتٍ على الدَّرَجِ الّذي بَناهُ المَلِكُ آحازُ.</span>
</div>
</div>

## رسل ملك بابل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 39:1–8; أخبارِ الأيّامِ الثّانِي 32:31; إشَعْياء 13:1"> وحينَ سَمِعَ مَرودَخُ بَلادانُ بنُ بَلادانَ مَلِكُ بابِلَ بِمَرَضِ حَزْقِيَّا، بعَثَ إليهِ برِسَائِلَ وهَدايا.</span>
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 39:2; أخبارِ الأيّامِ الثّانِي 32:25–27; المُلُوكِ الأوَّلُ 10:15"> ففَرِحَ حَزْقِيَّا بِها وأرى الوَفدَ الّذي حمَلَها إليهِ خزائِنَ نَفائِسِه مِنْ فضَّةٍ وذهَبٍ وأطيابٍ وزيتِ طيبٍ وأسلِحةٍ وما إلى ذلِكَ مِمَّا في مُستَودَعِهِ. لم يَترُكْ شيئًا إلَّا أراهُم إيَّاهُ في قصرِهِ وفي سائِرِ مَملكَتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 13:5; صموئيلَ الثّانِي 12:7–15; المُلُوكِ الثّانِي 5:25–26"> فأقبَلَ إشعيا على المَلِكِ حَزْقِيَّا وسألَهُ: ”ماذا قالَ لكَ هَؤلاءِ القَومُ ومِنْ أينَ جاؤوا؟“ فأجابَ حَزْقِيَّا: ”جاؤوا مِنْ أرضٍ بعيدةٍ، مِنْ بابِلَ“.</span>
<span className="v" id="v15" data-v="15" data-xref="أيُّوب 31:33; يَشُوع 7:19; المُلُوكِ الثّانِي 20:13"> فسألَه إشعيا: ”ماذا رَأوا في قصرِكَ؟“ فأجابَ: ”كُلَّ شيءٍ، لا شيءَ في قصري وخزائني إلَّا أرَيتُهُم إيَّاهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الأوَّلُ 22:19; المُلُوكِ الثّانِي 7:1; إشَعْياء 1:10"> فقالَ إشعيا لِحَزْقِيَّا: ”إسمَع قولَ الرّبِّ:</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الثّانِي 24:13; أخبارِ الأيّامِ الثّانِي 36:10; إرْمِيا 52:17–19"> ستَأتي أيّامٌ يُؤخَذُ فيها إلى بابِلَ كُلُّ ما في قصرِكَ مِمَّا اَدَّخَرَهُ آباؤُكَ إلى هذا اليومِ ولا يَبقى شيءٌ.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الثّانِي 33:11; المُلُوكِ الثّانِي 24:12; دانيال 1:3"> ومِنْ بَنيكَ يُؤخَذُ الّذينَ تَلِدُهُم مِنْ صُلْبِكَ. فيكونونَ خِصيانًا في قصرِ مَلِكِ بابِلَ“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="صَمُوئيلَ الأوَّلُ 3:18; زَكَريّا 8:19; لُوقا 2:10"> فقالَ حَزْقِيَّا لإشعيا: ”أقبَلُ حُكْمَ الرّبِّ الّذي أعلَنْتَ“. وأضافَ: ”ما دامَ لي سلامٌ وأمْنٌ في أيّامي“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="نَحَمْيا 3:16; أخبارِ الأيّامِ الثّانِي 32:32; أخبارِ الأيّامِ الثّانِي 32:4"> وما بَقيَ مِنْ أخبارِ حَزْقِيَّا وبَسالَتِهِ، وإنشائِهِ البِرَكةَ والقناةَ، وإدخالِهِ الماءَ إلى المدينةِ، مُدوَّنٌ في سِفرِ أخبارِ الأيّامِ لِمُلوكِ يَهوذا.</span>
<span className="v" id="v21" data-v="21" data-xref="أخبارِ الأيّامِ الثّانِي 32:33; المُلُوكِ الأوَّلُ 14:31; أخبارِ الأيّامِ الثّانِي 26:23"> وماتَ حَزْقِيَّا، ومَلَكَ منَسَّى اَبنُهُ مكانَهُ.</span>
</div>
