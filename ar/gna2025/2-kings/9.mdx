## ياهو يُمسح ملكًا على إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صَمُوئيلَ الأوَّلُ 10:1; المُلُوكِ الثّانِي 4:29; المُلُوكِ الأوَّلُ 1:39"> ودَعا أليشَعُ النَّبيُّ أحدَ جماعةِ الأنبياءِ وقالَ لَه: ”خُذْ قارورَةَ الزَّيتِ هذِهِ واَسرِعْ إلى راموثَ جلعادَ.</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الثّانِي 9:5; المُلُوكِ الأوَّلُ 19:16–17; المُلُوكِ الثّانِي 9:11"> وحينَ تَصِلُها تَجدُ ياهو بنَ يوشافاطَ بنِ نَمشي، فاَدْعُه مِنْ بَينِ رِفاقِهِ واَنفَرِدْ بهِ في غُرفَةٍ داخليَّةٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الأوَّلُ 19:16; الخُرُوج 29:7; صَمُوئيلَ الأوَّلُ 16:2"> وخُذْ قارورةَ الزَّيتِ هذِهِ وصُبَّ مِنها على رأْسِهِ وقُلْ: قالَ الرّبَّ: مَسَحْتُكَ ملِكًا على إِسرائيلَ. ثُمَّ اَفتَحِ البابَ واَهرُبْ بِسُرعةٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4"> فمَضى النَّبيُّ الشَّابُّ إلى راموثَ جلعادَ،</span>
<span className="v" id="v5" data-v="5" data-xref="القُضاة 3:19"> فلمَّا دخَلَها وجدَ قادةَ الجيشِ جالِسينَ مَعًا. فقالَ: ”معي رِسالةٌ إليك أيُّها القائدُ“. فسألَ ياهو: ”إلى مَنْ مِنَّا؟“ فأجابَ: ”إليكَ أيُّها القائدُ“.</span>
<span className="v" id="v6" data-v="6" data-xref="أخبارِ الأيّامِ الثّانِي 22:7; المُلُوكِ الأوَّلُ 19:16; المُلُوكِ الثّانِي 9:3"> فقامَ ودخَلَ البَيتَ، فصَبَّ الزَّيتَ على رأسِهِ وقالَ لَه: ”قالَ الرّبُّ إلهُ إِسرائيلَ: مَسَحْتُكَ مَلِكًا على شعبي إِسرائيلَ،</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الأوَّلُ 21:15; المُلُوكِ الأوَّلُ 18:4; التَّثْنِيَة 32:43"> فاَضرِبْ بَيتَ أخابَ مَولاكَ اَنتِقامًا لي مِنْ إيزابَلَ لِدِماءِ جميعِ أنبيائي وسائِرِ عبيدي</span>
<span className="v" id="v8" data-v="8" data-xref="صَمُوئيلَ الأوَّلُ 25:22; التَّثْنِيَة 32:36; المُلُوكِ الثّانِي 14:26"> حتّى يَبيدوا جميعًا. وسَأقطَعُ لأخابَ كُلَّ ذكَرٍ، مِنَ الصَّغيرِ إلى الكبيرِ، في إِسرائيلَ.</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الأوَّلُ 15:29; المُلُوكِ الأوَّلُ 21:22; المُلُوكِ الأوَّلُ 14:10–11"> وأجعَلُ بَيتَ أخابَ كبَيتِ يَرُبعامَ بنِ ناَباطَ، وكبَيتِ بَعشا بنِ أخيَّا.</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الأوَّلُ 21:23; المُلُوكِ الثّانِي 9:35–36; المُلُوكِ الثّانِي 9:3"> وأمَّا إيزابَلُ، فتأْكُلُها الكِلابُ في حقلِ يَزرَعيلَ ولا يَدفِنُها أحدٌ“. وفتَحَ البابَ وهربَ.</span>
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 29:26; اعمال الرسل 26:24; هُوشَع 9:7"> فخَرَجَ ياهو إلى قادةِ الجيشِ رِفاقِهِ، فسألوهُ: ”ما الخبَرُ؟ لِماذا جاءَكَ هذا المجنونُ؟“ فأجابَهُم: ”أنتُم تَعرِفونَ هذا الرَّجُلَ وكلامَهُ“.</span>
<span className="v" id="v12" data-v="12"> فقالوا: ”لا نَعرِفُ، فأخبِرْنا“. فقالَ لهُم: ”كلّمَني بأنَّ الرّبَّ قالَ إنَّه مسَحَني مَلِكًا على إِسرائيلَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 11:7–8; المُلُوكِ الأوَّلُ 1:39; المُلُوكِ الأوَّلُ 1:34"> فأسرَعَ كُلُّ واحدٍ مِنهُم وخلَعَ عباءَتَهُ عَنهُ وفرَشَها تَحتَ قدَمَي ياهو عِندَ أعلى الدَّرَجِ، ونفَخوا جميعًا في البوقِ ونادوا: ”يَحيا المَلِكُ ياهو“.</span>
</div>
</div>

## مقتل ملك إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الثّانِي 8:28; المُلُوكِ الأوَّلُ 22:3; المُلُوكِ الأوَّلُ 16:16"> وثارَ ياهو بنُ يوشافاطَ بنُ نَمشي على يورامَ المَلِكِ. وكانَ يورامُ يُحامي معَ جيشِ إِسرائيلَ كُلِّهِ عَنْ راموثَ جلعادَ ضِدَّ حَزائيلَ مَلِكِ أرامَ.</span>
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الثّانِي 8:29; صَمُوئيلَ الأوَّلُ 27:9–11; أخبارِ الأيّامِ الثّانِي 22:6"> لكِنَّهُ كانَ رجَعَ لِيَتَعالَجَ في يَزرَعيلَ مِنَ الجُروحِ الّتي أصابَهُ بِها الأراميُّونَ في ساحةِ القِتالِ. فقالَ ياهو لِرِفاقِهِ قادةِ الجيشِ: ”إنْ كُنتُم معي، فلا تَدَعوا أحدًا يخرُجُ مِنَ المدينةِ لِيَنقُلَ الخبرَ إلى يَزرَعيلَ“.</span>
<span className="v" id="v16" data-v="16" data-xref="أخبارِ الأيّامِ الثّانِي 22:6–7; المُلُوكِ الثّانِي 8:28–29"> وركِبَ ياهو وسارَ إلى يَزرَعيلَ، حَيثُ كانَ يورامُ على فِراشِ المرضِ، وأخَزْيا مَلِكُ يَهوذا يَعودُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="صَمُوئيلَ الأوَّلُ 16:4; حِزْقِيال 33:2–9; المُلُوكِ الثّانِي 7:14"> وكانَ الحارِسُ واقِفًا على البُرجِ في يَزرَعيلَ، فرأى جماعةَ ياهو مُقبِلَةً فصاحَ: ”أرى جماعةً“. فقالَ يورامُ: ”خُذْ فارسًا واَرسِلْهُ لِلِقائِهِم، وليَقُل لهُم: ”أسلامٌ؟“</span>
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الثّانِي 9:19; المُلُوكِ الثّانِي 9:22; إرْمِيا 16:5"> فذهَبَ الفارِسُ واَستَقبَلَهُم وقالَ: ”يقولُ المَلِكُ: ”أسلامٌ؟“ فأجابَهُ ياهو: ”ما لكَ ولِلسَّلامِ؟ إلتَحِقْ بي“. فنَقَلَ الحارسُ الخبرَ وقالَ: ”وصلَ الرَّسولُ إليهِم ولم يَرجِعْ“.</span>
<span className="v" id="v19" data-v="19"> فأرسلَ المَلِكُ فارِسًا آخَرَ، فجاءَهُم وقالَ لهُم: ”يقولُ المَلِكُ: أسلامٌ؟“ فأجابَهُ ياهو: ”ما لكَ ولِلسَّلامِ؟ إلتَحِقْ بي“.</span>
<span className="v" id="v20" data-v="20" data-xref="صموئيلَ الثّانِي 18:27; المُلُوكِ الأوَّلُ 19:17; الجامِعَة 9:10"> فنَقَلَ الحارسُ الخبرَ وقالَ: ”وصَلَ إليهِم ولم يَرجِعْ، وسائِقُ المَركبةِ يَسوقُ بِعُنْفٍ كياهو بنِ نَمشي“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="المُلُوكِ الأوَّلُ 21:1–7; أخبارِ الأيّامِ الثّانِي 22:7; ميخا 1:13"> فقالَ يورامُ: ”جَهِّزوا مَركبتي“. فجَهَّزوا مَركبتَهُ وخرَجَ معَ أخَزْيا مَلِكِ يَهوذا، كُلُّ واحدٍ في مَركبتِهِ، لاَستِقبالِ ياهو. فصادَفاهُ عِندَ حقلِ نابوتَ اليَزرَعيليّ.</span>
<span className="v" id="v22" data-v="22" data-xref="أخبارِ الأيّامِ الثّانِي 21:13; ناحُوم 3:4; المُلُوكِ الأوَّلُ 16:30–33"> فلمَّا رآهُ يورامُ قالَ لَه: ”أسلامٌ يا ياهو؟“ فأجابَ: ”أيُّ سلامٍ هذا ما دامَت أمُّكَ إيزابَلُ تَعبُدُ آلِهةً غريبةً وتُمارِسُ السِّحرَ“.</span>
<span className="v" id="v23" data-v="23" data-xref="المُلُوكِ الثّانِي 11:14; أخبارِ الأيّامِ الثّانِي 23:13"> فقالَ يورامُ لأخَزْيا: ”خيانَةٌ يا أخَزْيا“. واَرتَدَّ هارِبًا.</span>
<span className="v" id="v24" data-v="24" data-xref="المُلُوكِ الأوَّلُ 22:34; الأمثال 21:30; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:3"> فقَبَضَ ياهو بِيَدِهِ على القَوسِ ورَمى يورامَ بَينَ كتِفَيهِ، فنفَذَ السَّهمُ مِنْ قلبِه فسَقَطَ في مَركبتِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="المُلُوكِ الأوَّلُ 21:19; المُلُوكِ الأوَّلُ 21:24–29; إشَعْياء 13:1"> فقالَ ياهو لِبِدْقَرَ مُرافِقِهِ: ”خُذْهُ واَرْمِهِ في حقلِ نابوتَ اليَزرَعيليّ، واَذْكُرْ يومَ كُنّا مَعًا راكِبَينِ وراءَ أخابَ أبيهِ كيفَ تكَلَّمَ الرّبُّ علَيهِ وقالَ:</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الأوَّلُ 21:19; التَّثْنِيَة 5:9; أخبارِ الأيّامِ الثّانِي 25:4"> رأيتُ مَقتَلَ نابوتَ وبَنيهِ بِالأمسِ، وهُنا في هذِهِ الحقلَةِ ذاتِها أُجازيكَ أقولُ أنا الرّبُّ. فاَرفَعْهُ واَرْمِهِ في الحقلَةِ على ما قالَ الرّبُّ“.</span>
</div>
</div>

## مقتل ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="القُضاة 1:27; يَشُوع 17:11; المُلُوكِ الثّانِي 8:29"> ولمَّا رأى المَلِكُ أخَزْيا ما جَرى هربَ في طريقِ بَيتِ جَنْ، فأسرَعَ ياهو إلى اللَّحاقِ بهِ وقالَ: ”إرْموهُ“. فرَمَوهُ أيضًا في المَركبةِ عِندَ مُنحَدَرِ غَورٍ، بِالقُربِ مِنْ مدينةِ يِبلَعامَ فأُصيبَ. لكِنَّهُ هربَ، حتّى وصلَ إلى مَجِدُّو وفيها ماتَ.</span>
<span className="v" id="v28" data-v="28" data-xref="أخبارِ الأيّامِ الثّانِي 35:24; المُلُوكِ الثّانِي 23:30; المُلُوكِ الثّانِي 12:21"> فحَمَلَهُ رِجالُهُ في المَركبةِ إلى أورُشليمَ، ودَفَنوهُ معَ آبائِهِ في مدينةِ داوُدَ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="أخبارِ الأيّامِ الثّانِي 21:18–19; أخبارِ الأيّامِ الثّانِي 22:1–2; المُلُوكِ الثّانِي 8:24–25"> وكانَ أخَزْيا توَلَّى المُلْكَ على يَهوذا في السَّنةِ الحاديةَ عَشْرَةَ لِيورامَ بنِ أخابَ مَلِكِ إِسرائيلَ.</span>
</div>
</div>

## مقتل الملكة إيزابل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="إرْمِيا 4:30; حِزْقِيال 23:40; حِزْقِيال 24:17"> ثُمَّ دخَلَ ياهو إلى يَزرَعيلَ، فلمَّا سَمِعَت إيزابَلُ بِدُخولِهِ، كحَّلَت عَينَيها وزَيَّنَت رأسَها وأطَلَّت مِنْ نافِذةٍ في القصرِ.</span>
<span className="v" id="v31" data-v="31" data-xref="المُلُوكِ الأوَّلُ 16:9–20"> فلمَّا دخَلَ ياهو مِنَ البابِ قالَت: ”أهلًا وسَهلًا بِزِمْري الجديدِ قاتِلِ سيِّدِهِ“.</span>
<span className="v" id="v32" data-v="32" data-xref="أسْتِير 2:15; الخُرُوج 32:26; أخبارِ الأيّامِ الثّانِي 11:12"> فرَفَعَ ياهو نظَرَهُ نحوَ النَّافِذةِ وقالَ: ”مَنْ معي؟“ فأطَلَّ علَيهِ اَثْنانِ أو ثَلاثَةٌ مِنَ الخدَمِ،</span>
<span className="v" id="v33" data-v="33" data-xref="ميخا 7:10; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:15; إشَعْياء 25:10"> فقالَ لهُم: ”إقْذِفوها مِنَ النَّافِذةِ“ ففَعلوا فتَبَعثَر دَمُها على الحائِطِ وعلى الخَيلِ الّتي داسَتها.</span>
<span className="v" id="v34" data-v="34" data-xref="المُلُوكِ الأوَّلُ 16:31; المُلُوكِ الأوَّلُ 21:25; إشَعْياء 65:15"> ودخَلَ ياهو القصرَ وأكَلَ وشَرِبَ. ثُمَّ قالَ: ”خُذوا هذِهِ الملعونَةَ واَدفنوها لأنَّها بِنتُ مَلِكٍ“.</span>
<span className="v" id="v35" data-v="35" data-xref="اعمال الرسل 12:23; إشَعْياء 14:18–20; أيُّوب 31:3"> فذَهَبوا لِيَدْفنوها فلم يجِدوا مِنها إلَّا جُمجُمَتَها ورِجلَيها ويَدَيها.</span>
<span className="v" id="v36" data-v="36" data-xref="المُلُوكِ الأوَّلُ 21:23; اللّاويِّين 8:36; المُلُوكِ الثّانِي 14:25"> فعادوا وأخبَروا ياهو، فقالَ: ”تَمَّ ما قالَ الرّبُّ على لِسانِ إيليَّا التَّشْبيِّ: في حقلِ يَزرَعيلَ تأْكُلُ الكِلابُ لَحمَ إيزابَلَ،</span>
<span className="v" id="v37" data-v="37" data-xref="إرْمِيا 8:2; المَزامِير 83:10; إرْمِيا 16:4"> وتكونُ بَقايا جُثَّتِها هُناكَ كالزِّبْلِ، حتّى لا يَقدِرَ أحدٌ أنْ يَعرِفَها ويقولَ: هذِهِ إيزابَلُ“.</span>
</div>
