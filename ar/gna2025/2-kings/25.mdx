## سقوط أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 52:4–11; أخبارِ الأيّامِ الأوّلُ 6:15; إرْمِيا 34:1–6"> وفي السَّنةِ التَّاسِعةِ لِلمَلِكِ صِدقيَّا، في العاشِرِ مِنَ الشَّهرِ العاشِرِ، زحَفَ نَبوخَذنَصَّرُ معَ جميعِ جُيوشِهِ على أورُشليمَ وحاصَروها وبَنَوا حَولَها المَتاريسَ.</span>
<span className="v" id="v2" data-v="2"> فدَخَلَتِ المدينةُ تَحتَ الحِصارِ إلى السَّنةِ الحاديةَ عَشْرَةَ لِلمَلِكِ صِدقيَّا.</span>
<span className="v" id="v3" data-v="3" data-xref="إرْمِيا 52:6; حِزْقِيال 5:12; إرْمِيا 38:2"> وفي اليومِ التَّاسِعِ مِنَ الشَّهرِ الرَّابعِ اَشتَدَّ الجوعُ،</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 33:21; المُلُوكِ الثّانِي 25:5; إرْمِيا 39:2–7"> ففَتَحَ الشَّعبُ ثُغرَةً في سُورِ المدينةِ وهربَ المَلِكُ صِدقيَّا وجميعُ الجُنودِ ليلًا مِنْ طريقِ البابِ الّذي بَينَ السُّورَينِ، بِالقُربِ مِنْ بُستانِ المَلِكِ، والبابِليُّونَ مُحيطونَ بِالمدينةِ، ذهَبوا في طريقِ غَورِ الأُردُنِّ.</span>
<span className="v" id="v5" data-v="5" data-xref="إشَعْياء 30:16; إرْمِيا 39:5; عامُوس 2:14–16"> فتَبِعَ جيشُ البابِليِّينَ المَلِكَ حتّى لَحِقوا بهِ في سَهلِ أريحا، حَيثُ تفَرَّقَ جيشُهُ عَنهُ.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الثّانِي 23:33; إرْمِيا 34:21–22; إرْمِيا 21:7"> فأخَذوهُ إلى مَلِكِ بابِلَ في رَبلَةَ بِأرضِ حماةَ وحكَموا علَيهِ:</span>
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 32:4–5; القُضاة 16:21; إرْمِيا 39:6–7"> ذبَحوا بَنيهِ أمامَ عينَيهِ، ثُمَّ فقَأوا عينَيهِ وأوثَقوهُ بِسِلسِلَتَينِ مِنْ نُحاسٍ وجاؤوا بهِ إلى بابِلَ.</span>
</div>
</div>

## خراب الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إرْمِيا 52:12–16; إرْمِيا 40:1–4; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:12"> وفي اليومِ السَّابعِ مِنَ الشَّهرِ الخامِسِ في السَّنةِ التَّاسِعةَ عَشْرَةَ لِلمَلِكِ نَبوخذنَصَّرَ مَلِكِ بابِلَ، جاءَ نَبوزرادانُ قائِدُ حرَسِه وكبيرُ حاشيتِهِ إلى أورُشليمَ</span>
<span className="v" id="v9" data-v="9" data-xref="عامُوس 2:5; المَزامِير 74:3–7; المُلُوكِ الأوَّلُ 9:8"> وأحرقَ فيها هَيكلَ الرّبِّ وقصرَ المَلِكِ وجميعَ بُيوتِ الأشرافِ</span>
<span className="v" id="v10" data-v="10" data-xref="نَحَمْيا 1:3; إرْمِيا 39:8; إرْمِيا 52:14–23"> وهدَمَ جُنودُهُ كُلَّ أسوارِها.</span>
<span className="v" id="v11" data-v="11" data-xref="حِزْقِيال 5:2; إرْمِيا 15:1–2; إرْمِيا 39:9"> وسبَى نَبوزرادانُ سائِرَ الّذينَ بَقَوا في المدينةِ، والهارِبينَ اللَّاجِئينَ إلى مَلِكِ بابِلَ، وسائِرَ المَهَرَةِ مِنَ الصُنَّاعِ.</span>
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 40:7; المُلُوكِ الثّانِي 24:14; حِزْقِيال 33:24"> لكِنَّهُ ترَكَ مِنْ مَساكينِ الشَّعبِ بَعضَ الكرَّامينَ والفلَّاحينَ.</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الثّانِي 36:18; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:10; المُلُوكِ الثّانِي 20:17"> وحَطَّمَ البابِليُّونَ أعمِدَةَ النُّحاسِ والقواعِدَ والحَوضَ الكبيرَ في الهَيكلِ، وحمَلوا كُلَّ نُحاسِها إلى بابِلَ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 27:3; المُلُوكِ الأوَّلُ 7:47–50; الخُرُوج 38:3"> وأخَذوا القُدورَ والرُّفوشَ والسَّكاكينَ والأوانيَ والصُّحونَ وجميعَ الأدواتِ الّتي تُستَعمَلُ لِلعِبادةِ في الهَيكلِ</span>
<span className="v" id="v15" data-v="15" data-xref="دانيال 5:2–3; العَدَد 7:13–14; المُلُوكِ الأوَّلُ 7:48–51"> كذلِكَ أخَذَ نَبوزرادانُ، قائِدُ الحرَسِ، الطُّسوتَ والمَجامِرَ والسَّكاكينَ والقُدورَ والمَناراتِ والصُّحونَ والأقداحَ وكُلَّ مَصنوعاتِ الذَّهبِ والفِضَّةِ.</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الأوَّلُ 7:47"> كما أخذَ مَصنوعاتِ النُّحاسِ الّتي صَنَعَها المَلِكُ سُليمانُ لِلهَيكلِ وهيَ العَمودانِ والحَوضُ الكبيرُ وقواعِدُهُ الاثنا عشَرَ، وكانَت ثقيلَةً لِكثافَةِ نُحاسِها.</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الأوَّلُ 7:15–22; إرْمِيا 52:21–23"> وكانَ طُولُ كُلٍّ مِنَ العَمودَينِ ثَمانيَ عَشْرَةَ ذِراعًا، وعلَيهِ تاجٌ مِنْ نُحاسٍ، واَرتِفاعُ التَّاجِ ثَلاثُ أذرُعٍ، وعلى التَّاجِ حَبيكَةٌ ورُمَّانٌ مِنْ حَولِها، والكُلُّ مِنْ نُحاسٍ.</span>
</div>
</div>

## شعب يهوذا يسبى إلى بابل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 29:25; عَزْرا 7:1; إرْمِيا 21:1"> وأخذَ قائِدُ الحرَسِ سَرايا الكاهنَ الأوَّلَ وصَفَنيا الكاهنَ الثَّانيَ وحُرَّاسَ أبوابِ الهَيكلِ الثَّلاثَةَ.</span>
<span className="v" id="v19" data-v="19" data-xref="أسْتِير 1:14"> وأخذَ مِنَ المدينةِ وكيلَ الجُندِ، وخمسةَ رِجالٍ مِنْ حاشيةِ المَلِكِ، ومُعاوِنَ قائِدِ الجُندِ الّذي كانَ يَحفَظُ سِجِلَّ المُجَنَّدينَ، وسِتِّينَ رَجُلًا آخَرينَ.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:16; إرْمِيا 52:26–27; المُلُوكِ الثّانِي 23:33"> هؤلاءِ أخَذَهُم نَبوزرادانُ قائِدُ الحرَسِ وساقَهُم إلى مَلِكِ بابِلَ في رِبلَةَ،</span>
<span className="v" id="v21" data-v="21" data-xref="التَّثْنِيَة 28:64; المُلُوكِ الثّانِي 23:27; التَّثْنِيَة 4:26"> فقَتَلَهُمُ المَلِكُ في أرضِ حماةَ. وهكذا سبَى سُكَّانَ يَهوذا مِنْ أرضِهِم.</span>
</div>
</div>

## جدليا يحكم يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="إرْمِيا 39:14; المُلُوكِ الثّانِي 22:12; إرْمِيا 41:2"> وأمَّا الّذينَ أبقاهُم نَبوخذنَصَّرُ في يَهوذا، فوَلَّى علَيهِم جَدَلْيا بنَ أخيقامَ بنِ شافانَ.</span>
<span className="v" id="v23" data-v="23" data-xref="إرْمِيا 40:7–9; إرْمِيا 40:11–12; يَشُوع 18:26"> فلمَّا سَمِعَ قادةُ الجُيوشِ وجُنودُهُم الّذينَ لم يَستَسلِموا أنَّ مَلِكَ بابِلَ ولَّى جَدَلْيا، أتَوا إليهِ في المِصفاةِ وهُم: إسمَعيلُ بنُ نَثَنيا ويوحَنانُ بنُ قاريحَ وسَرايا بنُ تَنحومَثَ النَّطوفاتيِّ وبازَنيا بنُ المَعكيِّ.</span>
<span className="v" id="v24" data-v="24" data-xref="إرْمِيا 40:9–10; إرْمِيا 43:6; صموئيلَ الثّانِي 19:23"> فحَلفَ جَدَلْيا لهُم وقالَ: ”لا تَخافوا مِنَ البابِليِّينَ فخَيرٌ لكُم أنْ تَظَلُّوا في أرضِكُم آمِنينَ وتَخضَعوا لِمَلِكِ بابِلَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="زَكَريّا 7:5; زَكَريّا 8:19; إرْمِيا 40:15"> لكِنْ في الشَّهرِ السَّابِعِ مِنْ هذِهِ السَّنةِ جاءَ إسمَعيلُ بنُ نَثَنيا بنِ أليشاماعَ مِنَ النَّسلِ المَلكيِّ، ومعَهُ عشَرَةُ رِجالٍ، وقتَلوا جَدَلْيا، والبابِليِّينَ الّذينَ معَهُ في المِصفاةِ.</span>
<span className="v" id="v26" data-v="26" data-xref="إرْمِيا 43:4–7; إرْمِيا 41:16–18; إرْمِيا 42:14–22"> فقامَ الشَّعبُ، مِنَ الصَّغيرِ إلى الكبيرِ، وعلى رأسِهِم قادةُ الجُيوشِ، ورحَلوا إلى مِصْرَ خَوفًا مِنَ البابِليِّينَ.</span>
</div>
</div>

## إطلاق يوياكين من السجن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="إرْمِيا 52:31–34; التَّكوين 40:20; التَّكوين 40:13"> وفي اليومِ السَّابِعِ والعشرينَ مِنَ الشَّهرِ الثَّاني عشَرَ، في السَّنةِ السَّابِعةِ والثَّلاثينَ لِسَبْيِ يوياكينَ مَلِكِ يَهوذا، صارَ أويلُ مَرودَخُ مَلِكًا على بابِلَ. وفي السَّنةِ الّتي ملَكَ فيها، عَفا عَنْ يوياكينَ مَلِكِ يَهوذا، فأطلَقَهُ مِنَ السِّجنِ</span>
<span className="v" id="v28" data-v="28" data-xref="دانيال 2:37; إرْمِيا 27:6–11; دانيال 5:18–19"> وكلَّمَهُ بِلُطفٍ، وجعَلَ عرشَهُ أعلى مِنْ عُروشِ الّذينَ معَهُ في بابِلَ.</span>
<span className="v" id="v29" data-v="29" data-xref="صموئيلَ الثّانِي 9:7; إشَعْياء 61:3; التَّكوين 41:42"> وغيَّرَ يوياكينُ ثيابَ سِجنِهِ، وبَقيَ يتَناوَلُ الطَّعامَ على مائِدةِ مَلِكِ بابِلَ كُلَّ أيّامِ حياتِهِ.</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 11:3; ﻣﺘﻰ 6:11; نَحَمْيا 11:23"> وكانَت لَه نَفَقَةٌ دائِمَةٌ مِنَ المَلِكِ تَفي بِحاجتِهِ اليوميَّةِ، إلى يومِ وفاتِهِ.</span>
</div>
