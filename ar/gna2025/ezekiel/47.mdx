## النهر الجاري من الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="رُؤيا يُوحَنّا 22:1; زَكَريّا 14:8; يُوئيل 3:18"> ورجعَ بيَ الرَّجلُ إلى المَدخلِ الشَّرقيِّ فَرأيتُ تَحتَ العتبةِ مياهًا تجري على طُولِ جانبِ الهَيكلِ الأمامي صَوبَ الجَنوبِ وعَبرَ المذبَحِ إلى الشَّرقِ.</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 44:4; حِزْقِيال 44:2"> وخرجَ بي مِنْ طريقِ بابِ الشِّمالِ ودارَ بي في الطَّريقِ الخارجيِّ إلى البابِ الخارجيِّ عِندَ الطَّريقِ المُتَّجهِ نحوَ الشَّرقِ، فإذا بالمياهِ تجري مِنَ الجانبِ الجَنوبيّ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 40:3; اعمال الرسل 10:45–46; اعمال الرسل 11:16–18"> ولمَّا خرجَ الرَّجلُ نحوَ الشَّرقِ كانَ بيدِهِ خيطٌ، فقاسَ ألفَ ذِراعٍ واَجتازَ بيَ المياهَ الّتي غمرَتني حتّى الكَعبَينِ.</span>
<span className="v" id="v4" data-v="4" data-xref="كُولُوسِي 1:6; ﺭﻭﻣﻴﺔ 15:19; اعمال الرسل 19:10–20"> ثُمَّ قاسَ ألفَ ذِراعٍ فغمرَتني المياهُ إلى الرُّكبتَينِ. ثُمَّ قاسَ ألفَ ذِراعٍ فغمرَتني المياهُ إلى الحَقوَينِ.</span>
<span className="v" id="v5" data-v="5" data-xref="إشَعْياء 11:9; حَبَقُّوق 2:14; ﻣﺘﻰ 13:31–32"> ثُمَّ قاسَ ألفَ ذِراعٍ، فإذا بنهرٍ تعذَّرَ عليَّ الاجتيازُ فيهِ لأنَّ المياهَ صارَت طاغيةً لا يُمكِنُ اجتيازُها إلَّا سباحةً.</span>
<span className="v" id="v6" data-v="6" data-xref="حِزْقِيال 44:5; حِزْقِيال 40:4; زَكَريّا 4:2"> فقالَ لي: ”أرأيتَ يا اَبنَ البشَرِ؟“<br />وذهبَ بي وأرجعَني إلى شاطئِ النَّهرِ.</span>
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 47:12; رُؤيا يُوحَنّا 22:2; المُلُوكِ الثّانِي 2:13"> ولمَّا رَجعتُ إذا على شاطئِ النَّهرِ أشجارٌ كثيرةٌ مِنْ هُنا ومِنْ هُناكَ.</span>
<span className="v" id="v8" data-v="8" data-xref="يَشُوع 3:16; التَّثْنِيَة 3:17; المُلُوكِ الثّانِي 2:19–22"> فقالَ لي:<br />”هذِهِ المياهُ تخرُجُ نحوَ الوادي الشَّرقيَّةِ وتنزِلُ إلى الأردُنِّ، وعِندَما تصبُّ في مياهِ البحرِ المَيْتِ تُحوِّلُها إلى مياهٍ عذبةٍ.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 7:37–38; يُوحَنّا 6:63; يُوحَنّا 4:14"> وكُلُّ نفسٍ حَيَّةٍ تزحفُ حَيثُ يَجري النَّهرُ تحيا، ويتكاثرُ السَّمكُ لأنَّ هذِهِ المياهَ تجري إلى هُناكَ حتّى تصيرَ مياهُ البحرِ عذبةً. فكُلُّ ما يبلُغُ إليهِ النَّهرُ يحيا.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 4:19; المَزامِير 104:25; العَدَد 34:6"> ويقفُ على شَطِّ هذا البحرِ صيَّادونَ مِنْ عينِ جَدي إلى عينِ عِجلايمَ، يكونُ لهُم هُناكَ مَنشَرٌ للشِّباكِ، ويكونُ سمَكُهُ على أصنافِهِ كسَمكِ البحرِ المُتوسِّطِ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّثْنِيَة 29:23; رُؤيا يُوحَنّا 22:11; إرْمِيا 17:6"> أمَّا مُستنقعاتُهُ وبُركُهُ فلا تصيرُ عذبةً، بل تَبقى مالِحةً.</span>
<span className="v" id="v12" data-v="12" data-xref="رُؤيا يُوحَنّا 22:2; المَزامِير 1:3; إرْمِيا 17:8"> وعلى شاطئِهِ مِنْ هُنا ومِنْ هُناكَ يَنبتُ كُلُّ شجرٍ يُؤكَلُ، ولا يَذبُلُ ورقُهُ ولا ينقطِعُ ثمرُهُ، بل كُلَّ شهرٍ يحمِلُ بواكيرَ لأنَّ مياهَهُ تخرُجُ مِنَ الهَيكلِ، فيكونُ ثمرُهُ للطَّعامِ وورقُهُ للشِّفاءِ“.</span>
</div>
</div>

## حدود أرض إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 48:4–6; إرْمِيا 3:18; العَدَد 34:2–12"> وقالَ السَّيِّدُ الرّبُّ: ”هذِهِ هيَ الحدودُ الّتي فيها ترِثونَ الأرضَ على حسَبِ أسباطِ إِسرائيلَ الاثني عشَرَ، وليوسُفَ سَهمانِ،</span>
<span className="v" id="v14" data-v="14" data-xref="حِزْقِيال 20:5–6; التَّكوين 12:7; حِزْقِيال 48:29"> يرِثُ كُلُّ واحدٍ مِثلَ أخيهِ مِنْ هذِهِ الأرضِ الّتي أقسَمتُ رافعًا يَدي أنِّي أُعطيها لآبائِكُم، فتكونُ لكُم ميراثًا.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="العَدَد 34:8; حِزْقِيال 48:1"> ”وهذِهِ حدودُ الأرضِ مِنْ جهةِ الشِّمالِ: مِنَ البحرِ المتوسِّطِ على طريقِ حِتلونَ إلى ليبو حماةَ، إلى صدَدَ</span>
<span className="v" id="v16" data-v="16" data-xref="حِزْقِيال 48:1; العَدَد 13:21; صموئيلَ الثّانِي 8:8"> وبيروتَه وسبرائيمَ الّتي بَينَ حدودِ دِمشقَ وحدودِ حماةَ، وإلى حصرَتيكونَ الّتي على حدودِ حورانَ.</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 48:1; العَدَد 34:9"> وشماليَّ هذِهِ الحدودِ بَينَ البحرِ المتوسِّطِ وحصرَنونَ تكونُ دِمشقُ وحماةُ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 13:10; التَّكوين 31:23; التَّكوين 31:47"> وتكونُ الحدودُ الشَّرقيَّةُ بَينَ حورانَ ودِمشقَ، وما بَينَ جلعادَ وأرضِ إِسرائيلَ عِندَ نهرِ الأردُنِّ، ثُمَّ إلى البحرِ المَيتِ مُنتهيةً إلى تامارَ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 48:28; التَّثْنِيَة 32:51; العَدَد 34:5"> ”وتكونُ الحدودُ الجَنوبيَّةُ مِنْ تامارَ إلى مياهِ مريبوثَ في قادِشَ، ومِنْ هُناكَ بمحاذاةِ وادي مِصْرَ إلى البحرِ المتوسِّطِ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="العَدَد 34:6; حِزْقِيال 48:1"> ”وتكونُ الحدودُ الغربيَّةُ مِنَ البحرِ المُتوسِّطِ إلى قبالةِ ليبو حماةَ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21"> ”فتَقسِمونَ هذِهِ الأرضَ بَينَكُم على حسَبِ أسباطِ إِسرائيلَ.</span>
<span className="v" id="v22" data-v="22" data-xref="كُولُوسِي 3:11; أفَسُس 3:6; اعمال الرسل 11:18"> تَقسِمونَها بالقُرعةِ ميراثًا لكُم وللغُرَباءِ الدُّخلاءِ فيما بَينكُمُ الّذينَ ولدوا بَنينَ، فيكونونَ لكُم كالأصيلِ في بَني إِسرائيلَ يُقاسِمونَكُمُ الميراثَ فيما بَينَ أسباطِ إِسرائيلَ.</span>
<span className="v" id="v23" data-v="23"> والسِّبطُ الّذي يكونُ فيهِ الغريبُ دخيلًا، هُناكَ تُعطونَهُ ميراثَهُ، يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
</div>
