## رثاء صور

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 19:1; إرْمِيا 9:10; حِزْقِيال 28:12"> ”وأنتَ يا اَبنَ البشَرِ، أنشِدْ رثاءً على صورَ.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 27:4; إشَعْياء 23:11; إشَعْياء 23:2–3"> قُلْ لِصورَ السَّاكنةِ عِندَ مَداخلِ البحرِ، المُتاجرةِ معَ شُعوبِ الجزُرِ البعيدةِ: هكذا قالَ السَّيِّدُ الرّبُّ:<br />يا صورُ أنتِ قُلتِ<br />أنا كامِلةُ الجَمالِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 26:5"> أنتِ سفينةٌ في قلبِ البحارِ<br />وبانوكِ كَمَّلوا جَمالَكِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 3:9; إشَعْياء 14:8; نَشِيدِ الأنشاد 4:8"> بسروٍ مِنْ سَنيرَ صنعوا ألواحَكِ،<br />ومِنْ أرزِ لبنانَ رفعوا علَيكِ السَّواريَ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 10:4; إشَعْياء 2:13; زَكَريّا 11:2"> مِنْ بلُّوطِ باشانَ صنعوا مَجاذيفَكِ،<br />ومِنْ عاجٍ مُرَصَّعٍ في الشَّربينِ،<br />مِنْ جزُرِ كِتِّيمَ، صنعوا مَقاعدَكِ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 10:9; الخُرُوج 25:4; التَّكوين 10:4"> الكتَّانُ الموشَّى مِنْ مِصْرَ نشرتِهِ شِراعًا ورايةً،<br />والقِرمزُ والأرجوانُ مِنْ جزُرِ أليشةَ<br />كانا غِطاءَ مَقاعدِكِ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 10:18; المُلُوكِ الأوَّلُ 9:27; حِزْقِيال 27:11"> سُكَّانُ صَيدونَ وأروادَ كانوا ملَّاحيكِ،<br />ورِجالُكِ السَّاهرونَ كانوا الرَّبابِنَةَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الأوَّلُ 5:18; يَشُوع 13:5; المَزامِير 83:7"> كِبارُ صُنَّاعِ جُبيلَ والماهرونَ فيها<br />كانوا نجَّارينَ عِندَكِ،<br />جميعُ سُفنِ البحرِ وملَّاحوها<br />عمِلوا لترويجِ تجارتِكِ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 38:5; حِزْقِيال 30:5; إشَعْياء 66:19"> جنودٌ مِنْ بلادِ فارِسَ ولودَ وفوطَ خدَموا في جيشِكِ وكانوا رِجالَكِ في الحربِ. علَّقوا التُّرسَ والخوذةَ في ثُكْناتِكِ وزادوكِ مَجدًا.</span>
<span className="v" id="v11" data-v="11" data-xref="حِزْقِيال 27:3–4"> أبناءُ أروادَ معَ جيشِكِ كانوا على أسوارِكِ منْ حَولِكِ، وأبطالُ جمَدَ كانوا في بروجِكِ وعلَّقوا تروسَهُم على أسوارِكِ مِنْ حولِكِ. هُم أكملوا جَمالَكِ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 10:4; إشَعْياء 23:10; حِزْقِيال 38:13"> جُزرُ ترشيشَ تاجرَت مَعكِ لِكثرةِ غِناكِ في كُلِّ شيءٍ، وبالفِضَّةِ والحديدِ والقصديرِ والرَّصاصِ بادَلتْكِ بَضائِعَكِ.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 10:2; إشَعْياء 66:19; حِزْقِيال 39:1"> ياوانُ وتوبالُ وماشِكُ تاجروا معَكِ، وبالعبيدِ وآنيةِ النُّحاسِ دفعوا ثمنَ سِلَعِكِ.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 10:3; حِزْقِيال 38:6; أخبارِ الأيّامِ الأوّلُ 1:6"> سُكَّانُ توجرمةَ بادلوكِ بضائِعَكِ بالخَيلِ والفُرسانِ والبِغالِ.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 10:7; رُؤيا يُوحَنّا 18:12; المُلُوكِ الأوَّلُ 10:22"> وبَنو رودسَ تاجروا معَكِ وجزرٌ كثيرةٌ كانَت سُوقًا لسِلَعِكِ وحملَت إليكِ قرونَ العاجِ والأبنوسِ ثمنًا لها.</span>
<span className="v" id="v16" data-v="16" data-xref="القُضاة 10:6; حِزْقِيال 28:13; صموئيلَ الثّانِي 10:6"> أدومُ تاجَرت معَكِ لكثرةِ صَنائِعِكِ، فبادَلتْكِ بضاعتَكِ بالقماشِ الأصفرِ والأرجوانِ والوشيِّ والكتَّانِ والمرجانِ والياقوتِ.</span>
<span className="v" id="v17" data-v="17" data-xref="القُضاة 11:33; المُلُوكِ الأوَّلُ 5:9; إرْمِيا 8:22"> بَيتُ يَهوذا وإِسرائيلَ تاجروا معَكِ، وبحنطةِ مِنِّيتَ والمِسكِ والعسَلِ والزَّيتِ والبلَسانِ دفعوا ثمنَ سِلَعِكِ.</span>
<span className="v" id="v18" data-v="18" data-xref="إشَعْياء 7:8; التَّكوين 15:2; حِزْقِيال 47:16–18"> دِمشقُ تاجَرت معَكِ لكثرةِ صَنائِعِكِ وغِناكِ في كُلِّ شيءٍ، فزوّدَتْكِ بخمرِ حَلبونَ وصوفٍ زهارٍ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 30:23–24; المَزامِير 45:8; نَشِيدِ الأنشاد 4:13–14"> قبائلُ ددانَ وياوانَ، مِنْ يوزالَ وما بَعدَها، تاجَروا معَكِ وكانَ في أسواقِكِ حديدُهُمُ المطروقُ وقصبُ الطِّيبِ.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 25:3; حِزْقِيال 27:15"> ددَانُ تاجَرت معَكِ بأقمشةِ السُّروجِ،</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 60:7; التَّكوين 25:13; المُلُوكِ الأوَّلُ 10:15"> والعربُ وجميعُ شُيوخِ قيدارَ كانوا تجَّارَكِ، بالحُملانِ والكِباشِ والتُّيوسِ.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 10:7; إشَعْياء 60:6; المَزامِير 72:10"> تُجَّارُ شَبا ورعمةَ كانوا تُجَّارَكِ وبأفضلِ أنواعِ الطيِّبِ والأحجارِ الكريمةِ والذَّهَبِ بادلوكِ بضائِعَكِ.</span>
<span className="v" id="v23" data-v="23" data-xref="المُلُوكِ الثّانِي 19:12; إشَعْياء 37:12; عامُوس 1:5"> حرّانُ وكِنَّةُ وعدنُ وشَبا وأشُّورُ وكلمدُ تاجَروا معَكِ.</span>
<span className="v" id="v24" data-v="24" data-xref="المُلُوكِ الثّانِي 2:8"> هُم تاجَروا معَكِ بالأنسجةِ الفاخرةِ، بالثِّيابِ الزَّرقاءِ المُطرَّزةِ، وبالنَّفائسِ مِنَ السَّجّادِ المُزركَشِ المَفتولةِ خيوطُهُ.</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 2:16; المَزامِير 48:7; إشَعْياء 23:14"> سفُنُ ترشيشَ نقلَت بضائِعَكِ، فاَغتنيتِ وتمجَّدتِ كُلَّ المَجدِ في قلبِ البحارِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="المَزامِير 48:7; حِزْقِيال 26:19; اعمال الرسل 27:14"> ملَّاحوكِ جاؤوا بكِ إلى مياهٍ غزيرةٍ، فحطَّمَتْكِ الرِّيحُ الشَّرقيَّةُ، في قلبِ البحارِ،</span>
<span className="v" id="v27" data-v="27" data-xref="الأمثال 11:4; حِزْقِيال 27:24; حِزْقِيال 27:12"> وسقطَ في البحرِ يومَ سُقوطِكِ غِناكِ وتِجارتُكِ وبِضاعتُكِ وملَّاحوكِ وربابنَتُكِ ونجَّاروكِ وتُجَّارُكِ وجميعُ جُنودِكِ وكُلُّ شعبِكِ.</span>
<span className="v" id="v28" data-v="28" data-xref="حِزْقِيال 26:10; الخُرُوج 15:14; حِزْقِيال 31:16"> مِنْ صوتِ صُراخِ ربابنَتِكِ تتزلزلُ السَّواحلُ،</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 26:16; حِزْقِيال 32:10; رُؤيا يُوحَنّا 18:17–24"> ويتركُ ماسكو المَقاذيفِ والملَّاحونَ وجميعُ ربابنَةِ البحرِ سُفنَهُم ويقفونَ على البَرِّ،</span>
<span className="v" id="v30" data-v="30" data-xref="إرْمِيا 6:26; صموئيلَ الثّانِي 1:2; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:10"> ويولولونَ علَيكِ ويصرخونَ بمرارةٍ، ويُذرُّونَ فَوقَ رُؤوسِهِمِ التُّرابَ ويتمرَّغونَ في الرَّمادِ.</span>
<span className="v" id="v31" data-v="31" data-xref="حِزْقِيال 7:18; إشَعْياء 22:12; إشَعْياء 15:2"> ويحلُقونَ شَعرَ رُؤوسِهِم علَيكِ ويتحزَّمونَ بالمُسوحِ، ويبكونَ علَيكِ بِمرارةٍ وينحُبونَ نحيبًا مُرًّا.</span>
<span className="v" id="v32" data-v="32" data-xref="حِزْقِيال 26:17; رُؤيا يُوحَنّا 18:18; حِزْقِيال 27:2"> وفي نُواحِهِم يُنشِدونَ علَيكِ هذا الرِّثاءَ:<br />مَنْ كصورَ تهدَّمت في وسَطِ البحرِ</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="رُؤيا يُوحَنّا 18:19; حِزْقِيال 27:3; حِزْقِيال 28:16"> بخُروجِ بضائِعِكِ مِنَ البحارِ<br />أشبعتِ شعوبًا كثيرينَ،<br />وبكثرةِ ثروتِكِ وتِجارتِكِ أغنيتِ مُلوكَ الأرضِ.</span>
</div>
