## الرب يجعل حزقيال رقيبًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 33:30; حِزْقِيال 33:17; حِزْقِيال 3:11"> ”يا اَبنَ البشَرِ، قُلْ لِبَني شعبِكَ: إذا جلَبتُ الحربَ على أرضٍ ما فاَختارَ الشَّعبُ رجلًا وجعَلوهُ رقيبًا لهُم.</span>
<span className="v" id="v3" data-v="3" data-xref="هُوشَع 8:1; نَحَمْيا 4:20; نَحَمْيا 4:18"> ثُمَّ رَأى الرَّجلُ جيشَ العدوِّ مُقبِلًا على البلادِ ونفخَ في البوقِ وأنذرَ الشَّعبَ.</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 18:13; اعمال الرسل 18:6; إرْمِيا 6:17"> فإنْ سمِعَ السَّامِعُ صوتَ البوقِ وما تنبَّهَ، ثُمَّ جاءَ الجيشُ وقتَلَهُ، فدمُهُ يكونُ على رأسِهِ،</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 9:19–21; العِبرانِيّين 11:7; يُوحَنّا 8:39"> لأنَّهُ سمِعَ صوتَ البوقِ وما تنبَّهَ لَه. فلو كانَ تنبَّهَ لَه لنجا بنفسِهِ.</span>
<span className="v" id="v6" data-v="6" data-xref="حِزْقِيال 3:18–20; حِزْقِيال 33:8–9; التَّكوين 42:22"> أمَّا إذا رأى الرَّقيبُ جيشَ العدوِّ مُقبِلًا وما نفخَ في البوقِ وما أنذرَ الشَّعبَ، فجاءَ الجيشُ وقتَلَ واحدًا مِنهُم، فهذا الواحدُ يكونُ قُتِلَ بخطيئةِ الرَّقيبِ ومِنْ يَدِ الرَّقيبِ أطلبُ دمَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 3:17–21; إرْمِيا 26:2; حِزْقِيال 2:7–8"> ”وأنتَ يا اَبنَ البشَرِ، جعَلتُكَ رقيبًا على بَيتِ إِسرائيلَ فتسمعُ الكَلِمةَ مِنْ فمي وتُنذِرُهُم عنِّي.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 20:26–27; حِزْقِيال 33:6; حِزْقِيال 33:14"> فإذا قُلتُ للشِّرِّيرِ: يا شرِّيرُ موتًا تموتُ، وقصَّرتَ أنتَ عَنْ إنذارِهِ، فهذا الشِّرِّيرُ يموتُ في إثمِهِ، لكنِّي مِنْ يَدِكَ أطلبُ دمَهُ.</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 3:19; اعمال الرسل 20:26; الأمثال 15:10"> أمَّا إذا أنذرتَ الشِّرِّيرَ ليتوبَ عَنْ طريقِهِ وما تابَ، فإنَّهُ يموتُ في إثْمِهِ، وتكونُ خلَّصتَ نفْسَكَ.</span>
</div>
</div>

## مسؤولية الفرد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 49:14; حِزْقِيال 37:11; حِزْقِيال 24:23"> ”وأنتَ يا اَبنَ البشَرِ، فقُلْ لشعبِ إِسرائيلَ: أنتُم قُلتُم: معاصينا وخطايانا علَينا ونحنُ نفنى بها، فكيفَ تريدُنا أنْ نحيا؟</span>
<span className="v" id="v11" data-v="11" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:4; حِزْقِيال 18:23; رسالَةُ بُطرُسَ الثّانيَة 3:9"> قُلْ لهُم: حَيٌّ أنا، يقولُ السَّيِّدُ الرّبُّ، لا أكونُ مسرورًا بموتِ الشِّرِّيرِ، لكنْ بتوبتِهِ عَنْ شرِّهِ فيحيا. فتوبوا، توبوا عَنْ طُرقِكُمُ الشِّرِّيرةِ، فلماذا تموتونَ يا بَيتَ إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الثّانِي 7:14; حِزْقِيال 33:18–19; ﺭﻭﻣﻴﺔ 3:25"> ”وأنتَ يا اَبنَ البشَرِ قُلْ لبَني شعبِكَ إنَّ بِرَّ البارِّ لا يُنقذُهُ إنْ هوَ اَرتكبَ معصيةً، وشَرُّ الشِّرِّيرِ لا يُهلِكُهُ إنْ هوَ تابَ عن شَرِّهِ، والبارُّ لا يقدِرُ أنْ يحيا في بِرِّهِ السَّابقِ إنْ هوَ خطِئَ.</span>
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 18:24; العِبرانِيّين 10:38; حِزْقِيال 18:4"> إذا قُلتَ للبارِّ إنَّكَ تحيا، فتوكَّلَ على بِرِّهِ وفعَلَ الإثْمَ، فجميعُ بِرِّه السَّابقِ لا يُذكَرُ وبسبَب إثْمِهِ الّذي فعلَهُ يموتُ.</span>
<span className="v" id="v14" data-v="14" data-xref="حِزْقِيال 18:27; إرْمِيا 18:7–8; هُوشَع 14:1"> وإذا قُلتَ للشِّرِّيرِ إنكَ تموتُ، فتابَ عَنْ خطيئتهِ وعمِلَ ما هوَ حَقٌّ وعَدلٌ،</span>
<span className="v" id="v15" data-v="15" data-xref="حِزْقِيال 20:11; لُوقا 19:8; الخُرُوج 22:1–4"> وردَّ الرَّهنَ وأدَّى ما اَختلسَهُ وسلَكَ في طريقِ الحياةِ مِنْ دونِ أنْ يفعلَ إثمًا، فهوَ يحيا ولا يموتُ.</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 43:25; حِزْقِيال 18:22; ميخا 7:18–19"> جميعُ خطاياهُ السَّابقةِ لا تُذكَرُ لَه لأنَّه عمِلَ ما هوَ حَقٌّ وعَدلٌ فيحيا.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 25:24–26; حِزْقِيال 18:25; حِزْقِيال 18:29"> ”وبَنو شعبِكَ يقولونَ: طريقُ السَّيِّدِ الرّبِّ غَيرُ مُستقيمٍ: بل طريقُهُم هوَ غَيرُ مُستقيمٍ.</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 33:12–13; رسالَةُ بُطرُسَ الثّانيَة 2:20–22; العِبرانِيّين 10:38"> إذا اَرتدَّ البارُّ عَنْ بِرِّهِ وفعَلَ الإثْمَ، فهوَ يموتُ بسبَبِهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 18:27–28; حِزْقِيال 33:14"> وإذا تابَ الشِّرِّيرُ عَنْ شَرِّهِ وعَمِل ما هوَ حَقٌ وعَدلٌ فهوَ يحيا بِهما.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 16:27; حِزْقِيال 18:25; حِزْقِيال 33:17"> وتقولونَ: طريقُ السَّيِّدِ الرّبِّ غَيرُ مُستقيمٍ! بل أنا أدينُكُم، كُلُّ واحدٍ على طُرقِهِ، يا بَيتَ إِسرائيلَ“.</span>
</div>
</div>

## سقوط أورشليم ودمار الأرض

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="حِزْقِيال 1:2; إرْمِيا 39:1–8; المُلُوكِ الثّانِي 25:4"> وفي الخامسِ مِنَ الشَّهرِ العاشرِ مِنَ السَّنةِ الثَّانيةَ عشْرةَ مِنْ سَبينا، جاءني أحدُ النَّاجينَ مِنْ أُورُشليمَ فقالَ: ”سقطَتِ المدينةُ“.</span>
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 1:3; حِزْقِيال 3:26–27; حِزْقِيال 24:26–27"> وفي المساءِ السَّابقِ لمجيئِهِ حلَّت عليَّ يدُ الرّبِّ واَنفتحَ فمي. فلمَّا جاءني في الصَّباحِ كانَ اَنفتحَ فمي وما عُدتُ أبكمَ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v24" data-v="24" data-xref="حِزْقِيال 33:27; إشَعْياء 51:2; اعمال الرسل 7:5"> ”يا اَبنَ البشَرِ، يقولُ سُكَّانُ تِلكَ الخَرائبِ في أرضِ إِسرائيلَ: كانَ إبراهيمُ واحدًا وورِثَ الأرضَ ونحنُ كثيرونَ فلنا أُعطيتِ الأرضُ ميراثًا.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 12:16; إرْمِيا 7:9–10; التَّكوين 9:4"> لذلِكَ قُلْ لهُم: أنتُم تأكلونَ لحمًا بدمِهِ وتعبُدونَ الأصنامَ وتسفُكونَ الدَّمَ، فهل ترثونَ الأرضَ؟</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 27:40; ميخا 2:1–2; صَفَنْيا 3:3"> أنتُم تعتمدونَ على سُيوفِكُم وتفعَلونَ الرِّجسَ ويُنجِّسُ كُلُّ واحدٍ مِنكُم اَمرأةَ قريبِهِ، فهل ترِثونَ الأرضَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="صَمُوئيلَ الأوَّلُ 13:6; إرْمِيا 42:22; حِزْقِيال 39:4"> واَنقلْ لهُم كلامي أنا السَّيِّدُ الرّبُّ: ”حَيٌّ أنا، مَنْ كانَ في الخرائبِ يسقطُ بالسَّيفِ، ومَنْ كانَ في البرِّيَّةِ تأكلُهُ الوحوشُ، ومَنْ كانَ في المَغاورِ والجِبالِ الحصينةِ يموتُ بالوَباءِ.</span>
<span className="v" id="v28" data-v="28" data-xref="حِزْقِيال 7:24; إرْمِيا 44:2; إرْمِيا 44:6"> وأجعَلُ الأرضَ خَرِبةً مُقفِرةً وأسحقُ كبرياءَها وجبروتَها، فتصيرُ جبالُ إِسرائيلَ مُقفِرةً لا عابرَ فيها.</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 6:7; حِزْقِيال 36:17–18; حِزْقِيال 22:2–15"> فيعلمونَ أنِّي أنا هوَ الرّبُّ حينَ أجعلُ الأرضَ خَرِبةً مُقفِرةً عِقابًا لهُم على جميعِ الأرجاسِ الّتي عمِلوها.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 22:16–17; إرْمِيا 23:35; إشَعْياء 29:13"> ”وأنتَ يا اَبنَ البشَرِ، فشعبُكَ يتكلَّمونَ علَيكَ بجانبِ الجُدرانِ وفي مَداخِل البيوتِ، ويقولُ الواحدُ مِنهُم للآخَرِ والرَّجلُ لأخيهِ: تعالَوا فاَسمعوا الكلامَ الآتي مِنْ عِندِ الرّبِّ.</span>
<span className="v" id="v31" data-v="31" data-xref="ﻣﺘﻰ 13:22; المَزامِير 78:36–37; يَعقُوب 1:22–24"> فشعبي يدخلونَ إليكَ مُتجمهرينَ ويَجلسونَ أمامَكَ ويسمعونَ كلامَكَ ولا يعمَلونَ بهِ، لأنَّهُم بأفواهِهِم يُظهِرونَ حبًّا، لكنْ قلوبُهُم تسعى إلى الكسْبِ الخسيسِ.</span>
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 6:20; مَرْقُس 4:16–17; يُوحَنّا 5:35"> وإنَّما أنتَ لهُم كأغنيةِ حُبٍّ مِنْ صاحبِ صوتٍ رخيمٍ يُحسِنُ العزفَ، فيسمعونَ كلامَكَ ولا يعمَلونَ بهِ.</span>
<span className="v" id="v33" data-v="33" data-xref="إرْمِيا 28:9; صَمُوئيلَ الأوَّلُ 3:19–20; حِزْقِيال 2:5"> لكنْ عِندَما يَصدُقُ كلامُكَ، وهوَ سيصدُقُ يعلمونَ أنَّ نبيًّا كانَ بينَهُم“.</span>
</div>
