## رعاة إِسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="إرْمِيا 23:1; إرْمِيا 3:15; إشَعْياء 40:11"> ”يا اَبنَ البشَرِ، تنبَّأْ على رُعاةِ إِسرائيلَ وقُلْ لهُم: هكذا قالَ السَّيِّدُ الرّبُّ للرُّعاةِ: ويلٌ لرعاةِ إِسرائيلَ الّذينَ كانوا يَرعَونَ أنفُسَهُم. أمَّا الرُّعاةُ الصَّالحونَ فهُمُ الّذينَ يَرعَونَ الغنَمَ</span>
<span className="v" id="v3" data-v="3" data-xref="زَكَريّا 11:16; إرْمِيا 2:30; حِزْقِيال 33:25–26"> أنتُم تأكلونَ اللَّبنَ وتلبَسونَ الصُّوفَ وتذبحونَ الخروفَ السَّمينَ ولكنَّكُم لا تَرعَونَ الغنَمَ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 18:12–13; ﻣﺘﻰ 9:36; حِزْقِيال 34:16"> الضِّعافُ مِنها لا تُقوُّونَها، والمريضةُ لا تداوونَها، والمكسورةُ لا تجبرونَها، والشَّاردةُ لا تردُّونَها، والمَفقودةُ لا تبحثونَ عَنها، وإنَّما تسلَّطتُم علَيها بقسوةٍ وعُنفٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 9:36; أخبارِ الأيّامِ الثّانِي 18:16; إرْمِيا 23:2"> فتبعثَرت مِنْ غَيرِ راعٍ وصارت مأكلًا لكُلِّ وُحوشِ البرِّيَّةِ.</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 142:4; رسالَةُ بُطرُسَ الأُولَى 2:25; يُوحَنّا 10:16"> تاهت غنَمي في جميعِ الجِبالِ وعلى كُلِّ تلَّةٍ عاليةٍ، وتشتَّتَت على وجهِ الأرضِ ولا مَنْ يسألُ ولا مَنْ يبحثُ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 22:2–3; ميخا 3:8–9; المَزامِير 82:1–7"> لذلِكَ أيُّها الرُّعاةُ اَسمعوا كَلِمةَ الرّبِّ:</span>
<span className="v" id="v8" data-v="8" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:13; اعمال الرسل 20:33; حِزْقِيال 34:5–6"> حَيٌّ أنا، يقولُ السَّيِّدُ الرّبُّ، بما أنَّ غنَمي صارت نهبًا ومأكلًا لكُلِّ وُحوشِ البرِّيَّةِ، وبما أنَّها مِنْ غيرِ راعٍ ولا يسألُ رُعاتي عَنْ غنَمي، بل يرعى الرُّعاةُ أنفُسَهُم وغنَمي لا يَرعَونَها،</span>
<span className="v" id="v9" data-v="9"> لذلِكَ أيُّها الرُّعاةُ اَسمعوا كلِمةَ الرّبِّ:</span>
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 34:2; زَكَريّا 10:3; حِزْقِيال 13:8"> أنا أقومُ ضِدَّ الرُّعاةِ فأسترِدُّ غنَمي مِنْ أيديهِم وأمنعُهُم عَنْ أنْ يرعَوا الغنَمَ، فلا يرعى الرُّعاةُ أنفُسَهُم مِنْ بَعدُ، وأنقِذُ غنمي مِنْ أفواهِهِم فلا تكونُ لهُم مأكلًا“.</span>
</div>
</div>

## الله راعي شعبه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 10:16; لُوقا 19:10; إرْمِيا 23:3"> وقالَ السَّيِّدُ الرّبُّ: ”سأسألُ عن غنَمي وأتفقَّدُها</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 40:11; يُوحَنّا 10:11–12; لُوقا 19:10"> كما يتفقَّدُ الرَّاعي قطيعَهُ يومَ يكونُ في وسَطِ غنَمِهِ المُنتشرةِ، وأنقِذُها مِنْ جميعِ المَواضِعِ الّتي تشتَّتَت فيها يومَ الغيمِ والضَّبابِ.</span>
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 28:25–26; حِزْقِيال 11:17; إشَعْياء 11:11–16"> وأُخرِجُها مِنْ بَينِ الشُّعوبِ وأجمعُها مِنَ البُلدانِ وأجيءُ بها إلى أرضِها، وأرعاها على جبالِ إِسرائيلَ وفي الأوديةِ وفي جميعِ مساكنِ أرضِ إِسرائيلَ.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 10:9; إرْمِيا 31:25; المَزامِير 31:8–10"> في مرعًى صالحٍ أرعاها، وفي جبالِ إِسرائيلَ العاليةِ تكونُ حظيرتُها. هُناكَ تربِضُ في حظيرةٍ صالحةٍ وترعى في مرعىً خصيبٍ على جِبالِ إِسرائيلَ.</span>
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 3:15; المَزامِير 23:1–2; يُوحَنّا 21:15"> أنا أرعى غنَمي. وأنا أُعيدُها إلى حظيرتِها</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 19:10; حِزْقِيال 34:4; حِزْقِيال 34:11"> فأبحثُ عنِ المفقودةِ وأردُّ الشَّاردةَ وأجبُرُ المكسورةَ وأقوِّي الضَّعيفةَ وأحفظُ السَّمينةَ والقويَّةَ وأرعاها كُلَّها بعَدلٍ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 34:20–22; ﻣﺘﻰ 25:32–33; زَكَريّا 10:3"> ”وأنتم يا غنَمي، سأحكُمُ بَينَ ماشيةٍ وماشيةٍ، بَينَ الكِباشِ والتُّيوسِ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 23:13; لُوقا 11:52; العَدَد 16:13"> أمَا كفى بعضكُم أنْ يرعى المرعى الصَّالحَ حتّى يدوسَ برجلَيهِ باقي مراعِيكُم، وأنْ يشرَبَ المياهَ الصَّافيةَ حتّى يُكدِّرَها برجلَيهِ؟</span>
<span className="v" id="v19" data-v="19"> فيكونُ على البَعضِ الآخَرِ مِنْ غنَمي أنْ يرعى ما اَنداسَ منَ المرعى ويشربَ ما تكدَّرَ مِنَ المياهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 25:31–46; حِزْقِيال 34:10; المَزامِير 22:12–16"> لذلِكَ قالَ السَّيِّدُ الرّبُّ لرعاةِ غنَمِهِ: ”سأحكُمُ بَينَ الماشيةِ السَّمينةِ والماشيةِ الهزيلةِ،</span>
<span className="v" id="v21" data-v="21" data-xref="زَكَريّا 11:5; التَّثْنِيَة 33:17; لُوقا 13:14–16"> لأنَّكُم دفعتُم ضِعافَ غنَمي بالجَنبِ والكَتِفِ، ونطَحتُموها بِقُرونِكُم إلى أنْ شتَّتوها إلى الخارجِ.</span>
<span className="v" id="v22" data-v="22" data-xref="إرْمِيا 23:2–3; المَزامِير 72:12–14; حِزْقِيال 34:10"> فأُخلِّصُ غنَمي ولا تكونُ مِنْ بَعدُ نهبًا، وأحكُمُ بَينَ ماشيةٍ وماشيةٍ،</span>
<span className="v" id="v23" data-v="23" data-xref="حِزْقِيال 37:24–25; إرْمِيا 30:9; إشَعْياء 40:11"> وأُقيمُ علَيها راعيًا واحدًا ليرعاها كعبدي داوُدَ، فهوَ يرعاها ويكونُ لها راعيًا صالحًا.</span>
<span className="v" id="v24" data-v="24" data-xref="إرْمِيا 31:33; حِزْقِيال 36:28; إرْمِيا 23:5–6"> وأنا الرّبُّ أكونُ لغنَمي إلهًا ويكونُ الرَّاعي الّذي كعبدي داوُدَ لها رئيسًا. أنا الرّبُّ تكلَّمتُ.</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 11:6–9; حِزْقِيال 37:26; اللّاويِّين 26:6"> وأُعاهِدُ غنَمي عَهدَ سلامٍ وأردُّ الوحشَ الضَّاريَ عَنِ الأرضِ، فتسكنُ في البرِّيَّةِ آمنةً وتنامُ في الغابِ.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 28:12; إشَعْياء 44:3; اللّاويِّين 26:4"> وأجعَلُها هيَ والأرضَ المحيطةَ بِجَبلي بركةً، وأُنزِلُ المطَرَ في أوانِهِ فيكونُ بهِ الخيرُ.</span>
<span className="v" id="v27" data-v="27" data-xref="إرْمِيا 30:8; اللّاويِّين 26:4; اللّاويِّين 26:13"> ويُعطي شجَرُ البرِّيَّةِ ثمرَهُ، والأرضُ غلَّتَها، وتكونُ غنَمي على أرضِها آمنةً، فتعلمُ أنِّي أنا الرّبُّ حينَ أَكسِرُ نيرَها وأُنقِذُها مِنْ أيدي الّذينَ اَستعبدوها.</span>
<span className="v" id="v28" data-v="28" data-xref="إرْمِيا 46:27; إرْمِيا 30:10; حِزْقِيال 39:26"> ولا تكونُ مِنْ بَعدُ نهبًا لِلأمَمِ، ووحوشُ الأرضِ لا تأكُلُها، فتسكنُ آمنةً ولا أحدَ يُخيفُها.</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 36:29; إشَعْياء 4:2; حِزْقِيال 34:26–27"> وأُقيمُ لها غرسًا ناميًا فلا تكونُ مِنْ بَعدُ فريسةً للجوعِ في الأرضِ ولا تحمِلُ مِنْ بَعدُ تعييرَ الأمَمِ.</span>
<span className="v" id="v30" data-v="30" data-xref="حِزْقِيال 37:27; حِزْقِيال 34:24; المَزامِير 46:11"> فيَعلمُ الجميعُ أنِّي أنا هوَ الرّبُّ إلهُهُم، معَهُم أكونُ. وأنَّ بَيتَ إِسرائيلَ هُم شعبي، يقولُ السَّيِّدُ الرّبُّ.</span>
<span className="v" id="v31" data-v="31" data-xref="المَزامِير 100:3; يُوحَنّا 10:16; المَزامِير 78:52"> وأنتُم يا غنَمي، يا غنَمَ مرعاي، بشَرٌ أنتُم وأنا إلهُكُم، يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
