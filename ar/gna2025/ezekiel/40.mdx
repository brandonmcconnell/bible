<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 1:1; دانيال 7:1; دانيال 7:7"> وفي رُؤيا مِنْ رُؤى اللهِ جاءَ بي إلى أرضِ إِسرائيلَ ووضعَني على جَبلٍ شامخٍ جدًّا، علَيهِ مِنْ جهةِ الجَنوبِ أبنيةٌ تُشبِهُ المدينةَ.</span>

## الدار الخارجية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 42:20; إشَعْياء 26:1; رُؤيا يُوحَنّا 21:12"> فإذا بسُورٍ يُحيطُ بالهَيكلِ، وفي يَدِ الرَّجلِ قصَبةُ القياسِ وطُولُها سِتُّ أذرعٍ، وكُلُّ ذِراعٍ ذراعٌ وشِبرٌ، فقاسَ السُّورَ فكانَ اَرتفاعُهُ قصبةً واحدةً وسماكتُهُ قصبةً واحدةً.</span>
<span className="v" id="v6" data-v="6" data-xref="حِزْقِيال 43:1; حِزْقِيال 8:16; حِزْقِيال 11:1"> وجاءَ إلى البابِ المُتَّجهِ نحوَ الشَّرقِ وصعِدَ في درجاتِهِ وقاسَ عتبةَ البابِ فكانَ عرضُها قصبةً واحدةً.</span>
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 40:36; حِزْقِيال 40:33; أخبارِ الأيّامِ الأوّلُ 23:28"> ثُمَّ قاسَ كُلَّ غُرفةٍ فكانَ طُولُها قصبةً وعرضُها قصبةً، وكانَ ما بَينَ الغُرفِ خمسُ أذرعٍ، وكانَ طُولُ عتبةِ البابِ بِجانبِ رواقِ البابِ الدَّاخليِّ قصبةً واحدةً.</span>
<span className="v" id="v8" data-v="8"> وقاسَ رِواقَ البابِ</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 45:19"> فإذا هوَ ثماني أذرعٍ وأُطُرُ أبوابِهِ ذِراعانِ. وكانَ رِواقُ البابِ مِنَ الدَّاخلِ قُبالةَ الهَيكلِ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10"> وقاسَ غُرفَ البابِ الّذي إلى الشَّرقِ وكانَت ثلاثًا مِنْ هُنا وثلاثًا مِنْ هُناكَ ولكُلٍّ مِنها قياسٌ واحدٌ، ولأطُرِها قياسٌ واحدٌ مِنْ هُنا ومِنْ هُناكَ.</span>
<span className="v" id="v11" data-v="11"> وقاسَ عرضَ مدخلِ البابِ وطُولَهُ فإذا عرضُهُ عشْرُ أذرعٍ. وطولُهُ ثلاثَ عشْرةَ ذِراعًا.</span>
<span className="v" id="v12" data-v="12"> وكانَ أمامَ الغُرفِ حافَّةٌ لها ذِراعٌ مِنْ هُنا وذِراعٌ مِنْ هُناكَ، ولكُلِّ غُرفةٍ سِتُّ أذرعٍ مِنْ هُنا وسِتُّ أذرعٍ مِنْ هُناكَ.</span>
<span className="v" id="v13" data-v="13"> وقاسَ البابَ مِنَ الجِدارِ الخَلفيِّ إلى الجِدارِ الخَلفيِّ في الغُرفتَينِ المُتقابلتَينِ فإذا هوَ خمسٌ وعشرونَ ذِراعًا.</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 27:9; إشَعْياء 62:9; حِزْقِيال 42:1"> وقاسَ الرِّواقَ فكانَ عرضُهُ عشرينَ ذِراعًا وفي آخرِ الرِّواقِ باحةُ الدَّارِ.</span>
<span className="v" id="v15" data-v="15"> ومِنْ أمامِ بابِ الدُّخولِ إلى أمامِ رواقِ البابِ الدَّاخليِّ خمسونَ ذِراعًا.</span>
<span className="v" id="v16" data-v="16" data-xref="أخبارِ الأيّامِ الثّانِي 3:5; المُلُوكِ الأوَّلُ 6:4; حِزْقِيال 41:26"> وللغُرفِ نوافذُ في جُدرانِها الخارجيَّةِ وهكذا في الجِدرانِ الدَّاخليَّةِ بَينَ الغُرفِ. فكانَ على كُلِّ مُحيطِ الدَّاخلِ نوافذُ وعلى الأطُرِ نَخيلٌ يُزيِّنُها.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="رُؤيا يُوحَنّا 11:2; حِزْقِيال 42:1; أخبارِ الأيّامِ الأوّلُ 23:28"> وجاءَ الرَّجلُ بي إلى الدَّارِ الخارجيَّةِ، فإذا برصيفٍ يُحيطُ بها وفَوقَ الرَّصيفِ ثلاثونَ غُرفةً.</span>
<span className="v" id="v18" data-v="18"> وهذا الرَّصيفُ المُمتَدُّ بِجانبِ الأبوابِ على طُولِها هوَ الرَّصيفُ الأسفلُ.</span>
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 40:23; حِزْقِيال 40:27; حِزْقِيال 46:1–2"> وقاسَ العَرضَ مِنْ مَدخلِ البابِ الأسفلِ إلى المَدخلِ الخارجيِّ مِنَ الدَّارِ الداخليَّةِ فكانَ مئةَ ذِراعٍ إلى الشَّرقِ ومئةَ ذِراعٍ إلى الشِّمالِ.</span>
<span className="v" id="v20" data-v="20" data-xref="حِزْقِيال 40:6"> وقاسَ طُولَ وعَرضَ المَدخلِ المُتَّجهِ نحوَ الشِّمالِ للدَّارِ الخارجيَّةِ فكانَ قياسُ</span>
<span className="v" id="v21" data-v="21" data-xref="حِزْقِيال 40:29–30; حِزْقِيال 40:25–26; حِزْقِيال 40:10–16"> غُرفِهِ الّتي هيَ ثلاثٌ مِنْ هُنا وثلاثٌ مِنْ هُناكَ، وقياسُ أطُرِهِ وأروِقتِهِ على قياسِ المَدخلِ الأوَّلِ، طُولُهُ خمسونَ ذِراعًا. وعرضُهُ خمسٌ وعشرونَ ذِراعًا.</span>
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 40:31; حِزْقِيال 40:37; حِزْقِيال 40:6"> وكانَ قياسُ نوافذِهِ وأروقتِهِ ونخيلِهِ على قياسِ المَدخلِ الّذي يتَّجهُ نحوَ الشَّرقِ ويُصعَدُ إليهِ في سبعِ درجاتٍ، وأروقتُهُ أمامَها.</span>
<span className="v" id="v23" data-v="23" data-xref="حِزْقِيال 40:19; حِزْقِيال 40:27–28; الخُرُوج 38:9–12"> وللدَّارِ الدَّاخليَّةِ مَدخلٌ قُبالةَ المَدخلِ الشَّماليِّ وآخرُ قُبالةَ المَدخلِ الشَّرقيِّ. وقاسَ المسافةَ مِنْ مَدخلٍ إلى مَدخلٍ فكانَت مئةَ ذِراعٍ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="حِزْقِيال 46:9; حِزْقِيال 40:6; حِزْقِيال 40:20–21"> وأخذَني نحوَ الجَنوبِ، فإذا بمَدخلٍ هُناكَ، فقاسَ أُطُرَهُ وأروقتَهُ فكانَت مُطابِقةً للبابَينِ الآخرَينِ.</span>
<span className="v" id="v25" data-v="25" data-xref="حِزْقِيال 40:33; حِزْقِيال 40:16; حِزْقِيال 40:21–22"> وكانَ للمَدخلِ نوافذُ ولأروقتِهِ على مُحيطِهِ كتِلكَ النَّوافذِ في الأروقَةِ الأولى، وكانَ طُولُه خمسينَ ذِراعًا وعَرضُهُ خمسًا وعشرينَ ذِراعًا.</span>
<span className="v" id="v26" data-v="26" data-xref="حِزْقِيال 40:22; حِزْقِيال 40:16; حِزْقِيال 40:6"> ودرجاتُهُ سبعَ درجاتٍ تُؤدِّي إلى أروقتِهِ، ولَه نخيلٌ، واحدةٌ مِنْ هُنا وواحدةٌ مِنْ هُناكَ على أُطُرِهِ.</span>
<span className="v" id="v27" data-v="27" data-xref="حِزْقِيال 40:23; حِزْقِيال 40:32; حِزْقِيال 40:19"> وللدَّارِ الدَّاخليَّةِ بابٌ نحوَ الجَنوبِ: وقاسَ المَسافةَ مِنْ بابٍ إلى بابٍ نحوَ الجَنوبِ فكانَت مئةَ ذِراعٍ.</span>
</div>
</div>

<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="حِزْقِيال 40:35; حِزْقِيال 40:32"> وجاءَ بي إلى الدَّارِ الدَّاخليَّةِ مِنَ المَدخلِ الجَنوبيِّ وقاسَ المَدخلَ فإذا قياسُهُ كالمَداخلِ السَّابقةِ،</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 40:25; حِزْقِيال 40:16; حِزْقِيال 40:7"> وغُرفُهُ وأُطُرهُ ورِواقُهُ كتِلكَ الّتي سبَقتْها، ولَه ولرِواقِهِ على مُحيطِهِ نوافذُ، وطُولُه خمسونَ ذِراعًا وعَرضُهُ خمسٌ وعشرونَ ذِراعًا.</span>
<span className="v" id="v30" data-v="30" data-xref="حِزْقِيال 40:21; حِزْقِيال 40:36; حِزْقِيال 40:25"> وعلى مُحيطِهِ أروِقةٌ طُولُها خمسٌ وعشرونَ ذِراعًا وعَرضُها خمسُ أذرعٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="حِزْقِيال 40:26; حِزْقِيال 40:34; حِزْقِيال 40:22"> ورِواقُهُ قُبالةَ الدَّارِ الخارجيَّةِ، وعلى أُطُرِهِ نخيلٌ ودرجاتُهُ ثماني درجاتٍ.</span>
</div>

<span className="v" id="v32" data-v="32" data-xref="حِزْقِيال 40:35; حِزْقِيال 40:28–31"> وجاءَ بي إلى الدَّارِ الدَّاخليَّةِ نحوَ الشَّرقِ وقاسَ المَدخلَ فإذا قياسُهُ كالأقيسةِ السَّابقةِ،</span>
<span className="v" id="v33" data-v="33" data-xref="حِزْقِيال 40:21; حِزْقِيال 40:36; حِزْقِيال 40:25"> وكذلِكَ غُرفُهُ وأُطُرُهُ ورِواقُهُ، ولَه نوافذُ على مُحيطِهِ، وطُولُهُ خمسونَ ذِراعًا وعَرضُهُ خَمسةٌ وعِشرونَ ذِراعًا.</span>
<span className="v" id="v34" data-v="34" data-xref="حِزْقِيال 40:22; حِزْقِيال 40:31; حِزْقِيال 40:37"> ورِواقُهُ قُبالةَ الدَّارِ الخارجيَّةِ، وعلى أُطُرِهِ نخيلٌ مِنْ هُنا ومِنْ هُناكَ ودرجاتُهُ ثماني درجاتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="حِزْقِيال 47:2; حِزْقِيال 40:32; حِزْقِيال 44:4"> وقادني إلى المَدخلِ الشِّماليِّ وقاسَهُ.</span>
<span className="v" id="v36" data-v="36" data-xref="حِزْقِيال 40:29; حِزْقِيال 40:21; حِزْقِيال 40:16"> فإذا قياسُ غُرفِهِ وأُطُرِهِ ورِواقِهِ كالقياساتِ السَّابقةِ. ولَه نوافذُ على مُحيطِهِ وطُولُه خمسونَ ذِراعًا وعَرضُهُ خمسٌ وعشرونَ ذِراعًا.</span>
<span className="v" id="v37" data-v="37" data-xref="حِزْقِيال 40:34; حِزْقِيال 40:31"> ورِواقُهُ قُبالةَ الدَّارِ الخارجيَّةِ وعلى أُطُرِهِ نخيلٌ مِنْ هُنا ومِنْ هُناكَ ودرجاتُهُ ثماني درجاتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="أخبارِ الأيّامِ الثّانِي 4:6; حِزْقِيال 40:17; اللّاويِّين 8:21"> وهُناكَ غُرفةٌ مَدخلُها مِنْ رِواقِ الأبوابِ وفيها تُغسَلُ المُحرَقةُ.</span>
<span className="v" id="v39" data-v="39" data-xref="حِزْقِيال 46:2; اللّاويِّين 1:3–17; اللّاويِّين 4:2–3"> وفي رِواقِ البابِ مائدتانِ مِنْ هُنا ومائدتانِ مِنْ هُناكَ لتُذبَحَ علَيها المُحرَقةُ وذبيحةُ الخطيئةِ وذبيحةُ الإثْمِ.</span>
<span className="v" id="v40" data-v="40"> وفي الجانبِ الخارجيِّ عِندَ الدَّرجاتِ إلى مَدخلِ بابِ الشِّمالِ مائدتانِ، وفي الجانبِ الآخرِ عِندَ رِواقِ البابِ مائدتانِ.</span>
<span className="v" id="v41" data-v="41" data-xref="حِزْقِيال 40:39–40"> أربعُ موائدَ مِنْ هُنا وأربعُ موائدَ مِنْ هُناكَ عِندَ جانبِ البابِ، ثماني موائدَ يُذبَحُ علَيها.</span>
<span className="v" id="v42" data-v="42" data-xref="حِزْقِيال 40:39; الخُرُوج 20:25"> وكانَت أيضًا أربعُ موائدَ للمُحرَقةِ مِنْ حجارةٍ مَنحوتةٍ طُولُ الواحدةِ ذِراعٌ ونِصفٌ، وعَرضُها ذِراعٌ ونِصفٌ، وعلوُّها ذِراعٌ، توضَعُ علَيها الأدواتُ الّتي تُذبَحُ بِها المُحرَقةُ والذَّبيحةُ.</span>
<span className="v" id="v43" data-v="43" data-xref="اللّاويِّين 1:8; اللّاويِّين 1:6; اللّاويِّين 8:20"> ولها إفريزٌ عَرضُهُ شِبرٌ على مُحيطِها، وعلى الموائدِ اللُّحومُ المُقدَّمةُ.</span>
</div>
