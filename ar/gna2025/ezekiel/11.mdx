## الحكم على أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="حِزْقِيال 3:14; حِزْقِيال 11:24; حِزْقِيال 11:13"> ثُمَّ رفعني الرُّوحُ وجاءَ بي إلى البابِ الشَّرقيِّ لهَيكلِ الرّبِّ، فإذا عِندَ مدخَلِ البابِ خمسةٌ وعشرونَ رجُلًا، ورأيتُ بَينَهُم يازنيا بنَ عزورَ وفلطيا بنَ بنايا،</span>
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 30:1; إرْمِيا 5:5; المَزامِير 2:1–2"> الّذينَ يُفكِّرونَ بالإثْمِ ويُقدِّمونَ المَشورةَ السَّيِّئةَ في هذِهِ المدينةِ،</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 12:27; حِزْقِيال 12:22; حِزْقِيال 11:7–11"> فيقولونَ: عمَّا قريبٍ نبني بيوتًا. المدينةُ هيَ القِدْرُ ونحنُ اللَّحمُ.</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 20:46–47; حِزْقِيال 21:2; حِزْقِيال 3:2–15"> لذلِكَ تنبَّأْ علَيهِم، أنذِرْهُم يا اَبنَ البشَرِ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="إرْمِيا 17:10; إرْمِيا 16:17; رُؤيا يُوحَنّا 2:23"> وحلَّ عليَّ روحُ الرّبِّ وقالَ لي: ”قُلْ: هكذا قالَ الرّبُّ. أعرِفُ ما تقولونَ يا بَيتَ إِسرائيلَ وما يخطُرُ ببالِكُم.</span>
<span className="v" id="v6" data-v="6" data-xref="حِزْقِيال 7:23; حِزْقِيال 22:27; حِزْقِيال 22:12"> أكثرتُم قَتلاكُم في هذِهِ المدينةِ ومَلأْتُم شوارِعَها.</span>
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 24:3–13; ميخا 3:2–3; المُلُوكِ الثّانِي 25:18–22"> فقَتلاكُم في وسَطِها هُمُ اللَّحمُ والمدينةُ هيَ القِدْرُ، وأنتُم سأُخرِجُكُم مِنها.</span>
<span className="v" id="v8" data-v="8" data-xref="الأمثال 10:24; إشَعْياء 24:17–18; أيُّوب 3:25"> فَزِعتُم مِنَ السَّيفِ فأنا أجلُبُ علَيكُمُ السَّيفَ، يقولُ السَّيِّدُ الرّبُّ!</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 5:8; المَزامِير 106:41; التَّثْنِيَة 28:36"> وأُخرِجُكُم مِنَ المدينةِ وأُسلِّمُكُم إلى أيدي الغُرباءِ وأُنفِّذُ حُكمي فيكُم.</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الثّانِي 14:25; إرْمِيا 52:9–10; المُلُوكِ الثّانِي 25:19–21"> بالسَّيفِ تسقطونَ وعِندَ حدودِ أرضِ إسرائيلَ أحكُمُ علَيكُم، فتعلمونَ أنِّي أنا هوَ الرّبُّ.</span>
<span className="v" id="v11" data-v="11" data-xref="حِزْقِيال 11:3; حِزْقِيال 11:7–10"> هذِهِ المدينةُ لا تكونُ لكم قِدْرًا، وأنتُم لا تكونونَ اللَّحمَ فيها، بل عِندَ حدودِ أرضِكُم أحكُمُ علَيكُم،</span>
<span className="v" id="v12" data-v="12" data-xref="حِزْقِيال 8:10; حِزْقِيال 8:16; حِزْقِيال 8:14"> فتعلمونَ أنِّي أنا الرّبُّ الّذي ما سلَكتُم في فرائضِهِ ولا عمِلتُم بأحكامِهِ، بل عمِلتُم بأحكامِ الأمَمِ الّذينَ حَولَكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 11:1; حِزْقِيال 9:8; اعمال الرسل 5:5"> وبَينَما أنا أتنبَّأُ ماتَ فلْطيا بنُ بنايا، فسقَطتُ على وجهي ساجدًا وصرَختُ بصوتٍ عظيمٍ: ”آهِ، أيُّها السَّيِّدُ الرّبُّ، أتُفني الباقينَ مِنْ بَني إِسرائيلَ؟“</span>
</div>
</div>

## وعد الرّبّ للمسبيين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14"> فقالَ ليَ الرّبُّ:</span>
<span className="v" id="v15" data-v="15" data-xref="حِزْقِيال 33:24; إشَعْياء 65:5; يُوحَنّا 16:2"> ”يا اَبنَ البشَرِ، يقولُ سُكَّانُ أُورُشليمَ لإخوتِكَ وأقربائِكَ وكُلِّ بَني إِسرائيلَ الّذينَ في السَّبْي: بما أنَّكُم اَبتَعدتُم عَنِ الرّبِّ، فلنا أُعطيت هذِهِ الأرضُ ميراثًا.</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 8:14; التَّثْنِيَة 30:3–4; المَزامِير 90:1"> فقُلْ لهُم: هكذا قالَ السَّيِّدُ الرّبُّ. إذا كنتُ أبعدتُهُم في الأمَمِ وشتَّتُّهُم في البُلدانِ، فأنا كُنتُ لهُم حَيثُ ذهبوا مَعبدًا يَحميهِم.</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 28:25; إرْمِيا 3:18; حِزْقِيال 34:13"> وقُلْ لهُم ما قالَ السَّيِّدُ الرّبُّ: سأجمَعُهُم مِنْ بَينِ الشُّعوبِ وأُعيدُهُم جميعًا مِنَ البُلدانِ الّتي شتَّتُّهُم فيها وأعطيهِم أرضَ إِسرائيلَ.</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 37:23; حِزْقِيال 5:11; حِزْقِيال 42:7–8"> فيجيئونَ إليها ويُزيلونَ جميعَ أصنامِها وأرجاسِها مِنها.</span>
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 36:26–27; إرْمِيا 31:33; المَزامِير 51:10"> وأُعطيهِم قلبًا جديدًا وروحًا جديدًا في أحشائِهِم، وأنزِعُ مِنهُم قلبَ الحجَرِ وأُعطيهِم قلبًا مِنْ لَحمٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="حِزْقِيال 14:11; المَزامِير 105:45; إرْمِيا 30:22"> فيَسلكون في فرائِضي ويحفظون أحكامي ويعمَلوا بِها، فيكونونَ لي شعبًا وأكونُ لهُم إلهًا.</span>
<span className="v" id="v21" data-v="21" data-xref="حِزْقِيال 9:10; الجامِعَة 11:9; إرْمِيا 1:16"> أمَّا الّذينَ قلوبُهُم تتبعُ الأصنامَ والأرجاسَ، فأجلُبُ على رؤوسِهِم عاقبةَ سُلوكِهِم، يقولُ السَّيِّد الرّبُ“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 10:19; حِزْقِيال 1:19–20"> ثُمَّ رفَعَ الكروبيمُ أجنحَتَهُم، واَرتفعتِ الدَّواليبُ معَهُم ومَجدُ الرّبِّ فَوقَ الكروبيمِ.</span>
<span className="v" id="v23" data-v="23" data-xref="زَكَريّا 14:4; حِزْقِيال 8:4; حِزْقِيال 10:4"> وصعِدَ مَجدُ الرّبِّ عنِ المدينةِ ووقفَ على الجبَلِ شرقيَّها</span>
<span className="v" id="v24" data-v="24" data-xref="حِزْقِيال 3:12; حِزْقِيال 8:3; حِزْقِيال 11:1"> ورفعني الرُّوحُ في الرُّؤيا وجاءَ بي إلى المَسبيِّينَ في أرضِ بابلَ، حَيثُ اَرتفَعت عنِّي الرُّؤيا الّتي رأيتُها.</span>
<span className="v" id="v25" data-v="25" data-xref="حِزْقِيال 3:4; حِزْقِيال 3:27; حِزْقِيال 3:17"> فأخبَرتُ المسبيِّينَ بكُلِّ ما أراني إيّاهُ الرّبُّ.</span>
</div>
</div>
