## سيف الرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 20:46; حِزْقِيال 4:7; حِزْقِيال 25:2"> ”يا اَبنَ البشَرِ، إِلتفِتْ نحوَ اليمينِ وتكلَّمْ على الجَنوب، وتنبَّأْ على غابةِ أرضِ النَّقَبِ.</span>
<span className="v" id="v3" data-v="3" data-xref="أيُّوب 9:22; إرْمِيا 21:13; حِزْقِيال 5:8"> قُلْ لغابةِ النَّقَبِ اَسمعي كَلِمةَ الرّبِّ: هكذا قالَ السيِّدُ الرّبُّ: سأُشعِلُ فيكِ نارًا، فتأكُلُ فيكِ كُلَّ شجَرةٍ خضراءَ وكُلَّ شجَرةٍ يابسةٍ، ولا ينطفئُ لهيبُها المُشتعِلُ وتحترقُ بها جميعُ البشَرِ مِنَ الجَنوبِ إلى الشِّمالِ.</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 20:47; حِزْقِيال 7:2; حِزْقِيال 6:11–14"> فيَرى كُلُّ البشَرِ أنِّي أنا الرَّبُّ أشعلتُها فلا تنطفئُ“.</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 21:30; ناحُوم 1:9; صَمُوئيلَ الأوَّلُ 3:12"> فقُلتُ: ”آهِ أيُّها السَّيِّدُ الرّبُّ! سيقولونَ لي: أما زِلتَ تستخدمُ الأمثالَ؟</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 22:4; حِزْقِيال 6:11; إرْمِيا 19:10"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 7:17; حِزْقِيال 7:26; إشَعْياء 13:7"> ”يا اَبنَ البشَرِ، إِلتفتْ نحوَ أُورُشليمَ وتكلَّمْ على معابِدِها وتنبَّأْ على أرضِ إِسرائيلَ،</span>
<span className="v" id="v8" data-v="8"> وقُلْ لأرضِ إِسرائيلَ: هكذا قالَ الرّبُّ: أنا خَصمُكِ، فأستلُّ سيفي مِنْ غِمدِهِ وأقطعُ مِنكِ الأبرارَ والأشرارَ جميعًا.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 34:5–6; حِزْقِيال 21:28; حِزْقِيال 21:3"> أستلُّهُ على كُلِّ بشَرٍ منَ الجَنوبِ إلى الشِّمالِ،</span>
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 110:5–6; حِزْقِيال 20:47; حِزْقِيال 21:25–27"> فيعرِفُ كُلُّ البشَرِ أنِّي أنا الرّبُّ سَللْتُ سيفي مِنْ غِمدِهِ ولا يعودُ إليهِ.</span>
<span className="v" id="v11" data-v="11" data-xref="حِزْقِيال 21:19; إرْمِيا 25:33; إرْمِيا 51:20–23"> وأنتَ يا اَبنَ البشَرِ تنهّدْ! تنهّدْ بقلبٍ مُنكَسرٍ وبِمرارةٍ أمامَ عُيونِهِم.</span>
<span className="v" id="v12" data-v="12" data-xref="حِزْقِيال 21:6; يُوئيل 1:13; حِزْقِيال 21:14"> فإذا سألوكَ: لماذا تتنهَّدُ؟ قُلْ لهُم: لأنَّ خَبَرًا يَجيءُ فيذوبُ لَه كُلُّ قلبٍ، وترتخي كُلُّ يَدٍ، وتخورُ كُلُّ نفسٍ، وترتجفُ كُلُّ رُكبةٍ. بل جاءَ الخبرُ وسيتحقَّقُ، يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 21:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:2; حِزْقِيال 21:25"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الثّانِي 24:1; العَدَد 24:10; اللّاويِّين 26:24"> ”يا اَبنَ البشَرِ تنبَّأْ، وقُلْ:<br />سَيفٌ، سَيفٌ مَسنونٌ مَصقولٌ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="حِزْقِيال 21:10; إرْمِيا 17:27; حِزْقِيال 21:22"> مَسنونٌ للذَّبحِ.<br />مَصقولٌ ليلمَعَ كالبَرقِ.<br />أم نحنُ نلهو؟<br />احتقرتُ العصا يا اَبني<br />وكُلُّ شيءٍ مِنَ الخشَبِ،</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="حِزْقِيال 21:20; حِزْقِيال 21:4; حِزْقِيال 14:17"> لذلِكَ أُعطي السَّيفَ للصَّقلِ<br />حتّى يُقبَضَ علَيهِ بالكفِّ.<br />وهوَ الآن مَسنونٌ مَصقولٌ<br />ليُجعَلَ في يَدِ القاتلِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 5:13; حِزْقِيال 21:14; زَكَريّا 6:8"> فاَصرُخْ ووَلوِلْ يا اَبنَ البشَرِ،<br />لأنَّ السَّيفَ يكونُ على شعبي<br />وعلى رؤساءِ إِسرائيلَ جميعًا،<br />فمصيرُهُم إلى السَّيفِ كمصيرِ شعبي.<br />لذلِكَ دُقَّ على صَدرِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18"> هوَ اَمتحانٌ لشعبي،<br />فإنْ رفضوا أنْ يتوبوا<br />جرى لهُم هذا كُلُّه.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 4:1–3; إرْمِيا 1:10; حِزْقِيال 5:1–17"> ”وأنتَ يا اَبنَ البشَرِ<br />تنبَّأْ واَضرِبْ كفًّا على كفٍّ.<br />دَعِ السَّيفَ يهوي مرَّتَينِ،<br />دَعْهُ يهوي ثلاثَ مرَّاتٍ.<br />هوَ سَيفٌ فتَّاكٌ،<br />سَيفُ المقتَلةِ العُظمى،</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="عامُوس 1:14; إرْمِيا 49:2; حِزْقِيال 25:5"> فتذوبُ القلوبُ هلَعًا<br />ويتكاثرُ القتلى على الأبوابِ،<br />سَيفي سَيفٌ برَّاقٌ.<br />برَّاقٌ مَصقولٌ للذَّبحِ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="القُضاة 18:20; الأمثال 16:33; القُضاة 17:5"> إِضرِبْ يا سَيفُ يَمينًا ويَسارًا!<br />إِضرِبْ كيفما اَتَّجهتَ!</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 4:2; حِزْقِيال 26:9; إرْمِيا 51:14"> وأنا سأَضرِبُ كفًّا على كفٍّ وأشفي غَيظي.<br />أنا الرّبُّ تكلَّمتُ“.</span>
</div>
</div>

## سيف ملك بابل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="العَدَد 5:15; حِزْقِيال 21:24; حِزْقِيال 29:16"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v24" data-v="24" data-xref="هُوشَع 4:2; ميخا 3:10–12; إرْمِيا 3:2"> ”وأنتَ يا اَبنَ البشَرِ إِجعَلْ لكَ طريقَينِ لمجيءِ سَيفِ ملِكِ بابلَ. ليخرُجَ الطَّريقانِ مِنْ نُقطةٍ واحدةٍ، وضَعْ إشارةَ سَيرٍ في رأسِ طريقِ المدينةِ،</span>
<span className="v" id="v25" data-v="25" data-xref="حِزْقِيال 35:5; حِزْقِيال 21:29; حِزْقِيال 17:19"> واَرسُمْ طريقًا لِمجيءِ السَّيفِ إلى ربَّةَ، مدينةِ بَني عَمُّونَ، وإلى يَهوذا وأُورُشليمَ الحصينةِ.</span>
<span className="v" id="v26" data-v="26" data-xref="حِزْقِيال 17:24; المَزامِير 75:7; إرْمِيا 13:18"> فملِكُ بابلَ سيقفُ عِندَ مُفتَرقِ الطَّريقَينِ ليستدِلَّ بالعرافةِ على طريقٍ يَسلُكُه. فيَهزُّ السِّهامَ ويستشيرُ الأصنامَ وينظرُ في كَبدِ الذَّبيحةِ،</span>
<span className="v" id="v27" data-v="27" data-xref="ميخا 5:2; إرْمِيا 23:5–6; زَكَريّا 9:9"> فتلتقطُ يمينُهُ السَّهمَ الّذي يُشيرُ إلى طريقِ أُورُشليمَ، لينصُبَ المجانيقَ ويأمرَ بالقتلِ، ويرفعَ الصَّوتَ بهُتافِ الحربِ، ويهجُمَ على الأبوابِ بالأكباشِ ويَبنيَ المتاريسَ ويحفُرَ الخنادِقَ.</span>
<span className="v" id="v28" data-v="28" data-xref="حِزْقِيال 21:20; حِزْقِيال 21:9–10; صَفَنْيا 2:8–10"> ومعَ أنَّ أهلَ أُورُشليمَ يزعمونَ أنَّ هذِهِ العرافةَ باطلةٌ بسبَبِ القَسَمِ الّذي بَينَهُم وبَينَ ملِكِ بابلَ. فسيذكرونَ إثمَهُم عِندَما يُساقونَ إلى السَّبْي.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 21:25; حِزْقِيال 22:28; المَزامِير 37:13"> ”لذلِكَ هكذا قالَ السَّيِّدُ الرّبُّ: بما أنَّكُم تذكَّرتُم إثْمَكُم وجاهَرتُم بمَعاصيكُم وأظهَرتُم خطاياكُم في جميعِ أعمالِكُم، فستُؤخَذونَ بيدِ الأعداءِ.</span>
<span className="v" id="v30" data-v="30" data-xref="حِزْقِيال 16:38; التَّكوين 15:14; إرْمِيا 47:6–7"> وأنتَ أيُّها النَّجِسُ الشِّرِّيرُ يا رئيسَ بَني إِسرائيلَ جاءَ يومُكَ وبلَغَ إثْمُكَ مُنتهاهُ،</span>
<span className="v" id="v31" data-v="31" data-xref="ناحُوم 1:6; إرْمِيا 6:22–23; حِزْقِيال 14:19"> يقولُ السَّيِّدُ الرّبُّ. إنزعِ العِمامةَ واَرمِ التَّاجَ عَنْ رأسِكَ. هذِهِ الحالُ لا تَبقى، بل أرفعُ المُنخفِضَ وأحُطُّ الرَّفيعَ،</span>
<span className="v" id="v32" data-v="32" data-xref="حِزْقِيال 25:10; مَلاخِي 4:1; حِزْقِيال 20:47–48"> وأجلبُ خرابًا على خرابٍ. وهذا لن يكونَ حتّى يَجيءَ الّذي اَختَرتُهُ للحُكْمِ علَيكُم.</span>
</div>

## عقاب بني عمون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v33" data-v="33"> ”وأنتَ يا اَبنَ البشَرِ تنبَّأْ على بَني عَمُّونَ الّذينَ يُعيِّرونَ بَني إِسرائيلَ وقُلْ لهُم: السَّيفُ، السَّيفُ مَسلولٌ للذَّبحِ جاهزٌ للقتلِ يلمَعُ كالبَرقِ.</span>
<span className="v" id="v34" data-v="34"> على الأشرارِ المُنافقينَ لأنَّ يومَهُم جاءَ وإثْمَهُم بلَغَ مُنتهاهُ. رُؤاهُم باطلةٌ وعرافتُهُم كاذِبةٌ حينَ يقولونَ:</span>
<span className="v" id="v35" data-v="35"> وأنتَ أيُّها المَلِكُ أرجعِ السَّيفَ إلى غِمدِهِ. ولكنْ في المَوضِعِ الّذي خُلِقتَ فيهِ أَدينُكَ.</span>
<span className="v" id="v36" data-v="36"> فأصبُّ علَيكَ غَيظي وأنفخُ علَيكَ نارَ غضبي وأُسلِّمُكَ إلى أيدي قَتلةٍ مُحترفينَ.</span>
<span className="v" id="v37" data-v="37"> فتكونَ مأكلًا للنَّارِ، ويَسيلُ دمُكَ في وسَطِ الأرضِ، ولا تُذكَرُ مِنْ بَعدُ. أنا الرّبُّ تكلَّمتُ“.</span>
