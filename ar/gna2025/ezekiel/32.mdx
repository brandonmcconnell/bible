## تشبيه فرعون بالتمساح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="حِزْقِيال 30:20; حِزْقِيال 32:17; حِزْقِيال 31:1"> وفي الأوَّلِ مِنَ الشَّهرِ الثَّاني عشَرَ مِنَ السَّنةِ الثَّانيةَ عشْرةَ بَعدَ السَّبْيِ قالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 29:3; ناحُوم 2:11–13; حِزْقِيال 27:2"> ”يا اَبنَ البشَرِ، أنشِدْ رِثاءً على فِرعَونَ ملِكِ مِصْرَ وقُلْ لَه: تحسِبُ نفسَكَ أسدًا بَينَ الأمَمِ، وأنتَ مِثلُ تمساحٍ في المياهِ، تخبُطُ أنهارَكَ وتُكدِّرُ مياهَها برجليكَ وتُعكِّرُ سواقيَها.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 12:13; إرْمِيا 16:16; الجامِعَة 9:12"> سأبسُطُ علَيكَ شبكتي بأيدي جَمعٍ كثيرٍ مِنَ الشُّعوبِ، فيُصعِدونَكَ إلى الشَّاطئِ بخُيوطِها،</span>
<span className="v" id="v4" data-v="4" data-xref="إشَعْياء 18:6; حِزْقِيال 29:5; حِزْقِيال 31:12–13"> وأطرحُكَ على الأرضِ وأتركُكَ على وجهِ البَرِّ وأجلُبُ علَيكَ جميعَ طُيورِ السَّماءِ وأُشبِعُ مِنكَ كُلَّ وحوشِ الأرضِ،</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 31:12"> وأُلقي لَحمَكَ على الجِبالِ وأملأُ الأوديةَ مِنْ جيفَتِكَ.</span>
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 34:3; إشَعْياء 34:7; رُؤيا يُوحَنّا 14:20"> وأروي الأرضَ إلى رؤوسِ الجِبالِ بسيلٍ مِنْ دَمِكَ، وتمتلئُ مِنه السَّواقي.</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 24:29; يُوئيل 2:31; إشَعْياء 13:10"> وعِندَما أمحوكَ أغطِّي السَّماواتِ وأُظلِمُ نُجومَها وأحجُبُ الشَّمسَ بسحابٍ، والقمرُ لا يُنيرُ بنورِهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 1:14; الأمثال 20:20"> وأُطفِئُ كُلَّ أنوارِ السَّماءِ مِنْ فَوقُ وأنشرُ الظُّلمةَ على أرضِكَ، يقولُ السَّيِّدُ الرّبُّ.</span>
<span className="v" id="v9" data-v="9" data-xref="رُؤيا يُوحَنّا 18:10–15; حِزْقِيال 30:26; إرْمِيا 25:15–25"> وأُدخِلُ الذُّهولَ إلى قُلوبِ شعوبٍ كثيرينَ حينَ أُذيعُ خبرَ هلاكِكَ في الأمَمِ حتّى في بُلدانٍ لا تعرِفُها.</span>
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 26:16; حِزْقِيال 27:35; المُلُوكِ الأوَّلُ 9:8"> وأجعَلُ الرُّعبَ يستولي على شُعوبٍ كثيرينَ، ويرتعدُ علَيكَ مُلوكُهُم حينَ أشهَرُ سيفي أمامَهُم، ويرتجفُ كُلُّ واحدٍ مِنهُم على الدَّوامِ في يومِ سُقوطِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 43:10; حِزْقِيال 26:7; حِزْقِيال 30:22–25"> وقالَ السَّيِّدُ الرّبُّ: ”سيفُ ملِكِ بابلَ يَحلُّ بكَ،</span>
<span className="v" id="v12" data-v="12" data-xref="حِزْقِيال 28:7; حِزْقِيال 29:19; حِزْقِيال 31:11"> فأُسقِطُ شعبَكَ بسيوفِ جبابرةٍ منَ الأمَمِ القُساةِ، فيُدمِّرونَ كبرياءَ مِصْرَ ويُهلِكونَ جيشَهُ كُلَّهُ.</span>
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 29:11; حِزْقِيال 29:8; حِزْقِيال 30:12"> وأُبيدُ جميعَ بَهائِمِها عَنِ المياهِ الغزيرةِ فلا تُكدِّرُها قدَمُ بشَرٍ مِنْ بَعدُ ولا تُعكِّرُها أظلافُ البَهائمِ.</span>
<span className="v" id="v14" data-v="14"> وفي ذلِكَ الحينِ أجعَلُ مياهَ مِصْرَ صافيةً وأنهارَها تجري كالزَّيتِ، يقولُ السَّيِّدُ الرّبُّ.</span>
<span className="v" id="v15" data-v="15" data-xref="حِزْقِيال 6:7; الخُرُوج 7:5; الخُرُوج 14:4"> وعِندَما أجعَلُ أرضَ مِصْرَ خَرِبةً خاليةً مِنْ كُلِّ ما فيها، وأضرِبُ جميعَ سُكَّانِها، يَعلمونَ أنِّي أنا هوَ الرّبُّ.</span>
<span className="v" id="v16" data-v="16" data-xref="صموئيلَ الثّانِي 1:17; أخبارِ الأيّامِ الثّانِي 35:25; حِزْقِيال 26:17"> هذا رِثاءٌ تُنشِدهُ نِساءُ الأمَمِ كُلِّها على مِصْرَ وجيشِها، يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 32:1; حِزْقِيال 33:21; حِزْقِيال 1:2"> وفي الخامسَ عشَرَ مِن الشَّهرِ الثَّاني عشَرَ مِنَ السَّنةِ الثَّانيةَ عشْرةَ بَعدَ السَّبْيِ قالَ ليَ الرّبُّ:</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 31:14; حِزْقِيال 32:2; إرْمِيا 1:10"> ”يا اَبنَ البشَرِ، وَلوِلْ على شعبِ مِصْرَ وأهبِطْهُ هوَ وسائرُ الأُمَمِ العظيمةِ إلى الأرضِ السُّفلى معَ الهابطينَ في الهاويةِ.</span>
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 31:18; حِزْقِيال 32:29–30; إرْمِيا 9:25–26"> قُلْ للمِصْريِّينَ: أتفوقونَ أحدًا في الحُسنِ الآنَ؟ إِنزِلوا واَضطَّجِعوا معَ الّذينَ لا مَجدَ لهُم.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 28:3; حِزْقِيال 32:23–26; حِزْقِيال 32:29–30"> أنتُم تسقطونَ بَينَ القتلى بالسَّيفِ، والسَّيفُ مَسلولٌ لقتلِكُم جميعًا.</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 14:9–10; حِزْقِيال 32:27; حِزْقِيال 32:19"> أقوياءُ الجبابِرةِ والّذينَ حاربوا مَعكُم يُرحِّبونَ بكُم في القُبورِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 32:26; حِزْقِيال 32:29–30; حِزْقِيال 32:24"> ”هُناكَ أشُّورُ وجميعُ جنودِها مِنْ حَولِها في قُبورِهِم. كُلُّهُم قتلى سقطوا بالسَّيفِ،</span>
<span className="v" id="v23" data-v="23" data-xref="حِزْقِيال 26:20; حِزْقِيال 32:24–27; المَزامِير 27:13"> وقُبورُهُم في أسفلِ الهاويةِ حَولَ ملِكِهِم. كُلُّهُم قتلى سقطوا بالسَّيفِ بَعدَ أنْ ألقَوا الرُّعبَ في أرضِ الأحياءِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="حِزْقِيال 32:18; حِزْقِيال 26:20; إرْمِيا 49:34–39"> ”هُناكَ عيلامُ وكُلُّ جنودِها حَولَ قبرِها. كُلُّهُم قتلى سقطوا بالسَّيفِ وهبطوا بلا مَجدٍ إلى الأرضِ السُّفلى. ألقَوا الرُّعبَ في أرضِ الأحياءِ ثُمَّ حملوا عارَهُم مع الهابطينَ في الهاويةِ.</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 139:8; أخبارِ الأيّامِ الأوّلُ 10:4; صموئيلَ الثّانِي 1:20"> في وسَطِ القتلى اَضطَّجعَ ملِكُ عيلامَ وقبورُ جندِهِ حَولَ قبرِهِ. جميعُهُم بلا مَجدٍ قُتِلوا بالسَّيفِ لأنَّهُم ألقَوا رُعبَهُم في أرضِ الأحياءِ، ثُمَّ حملوا عارَهُم معَ الهابطينَ في الهاويةِ وجُعِلوا بَينَ القتلى.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="حِزْقِيال 27:13; التَّكوين 10:2; حِزْقِيال 38:2–3"> ”وهُناكَ ماشكُ وتوبالُ تُحيطُ بهِما قبورُ جنودِهما. كُلُّهُم مقتولونَ بلا مَجدٍ بالسَّيفِ لأنَّهُم ألقَوا رُعبَهُم في أرضِ الأحياءِ.</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 14:18–19; أيُّوب 3:13–15; حِزْقِيال 32:21"> لا يضطَّجعونَ معَ الجبابِرةِ القُدماءِ الّذينَ سقطوا وهبطوا بلا مَجدٍ إلى القُبورِ معَ أسلحتِهِم وجعَلوا سيوفَهُم تَحتَ رؤوسِهِم وتروسَهُم على عِظامِهِم، وهُمُ الّذينَ ألقَوا الرُّعبَ في أرضِ الأحياءِ،</span>
<span className="v" id="v28" data-v="28" data-xref="دانيال 2:34–35"> بل يتحطَّمونَ بَينَ الّذينَ لا مَجدَ لهُم ويضطَّجعونَ معَ القتلى بالسَّيفِ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="إرْمِيا 49:7–22; مَلاخِي 1:3–4; عُوبَدْيا 1:1–9"> ”وهُناكَ أدومُ، مُلوكُها وكُلُّ رؤسائِها والّذينَ جُعِلوا معَ جبابِرتِهِم بَينَ القتلى بالسَّيفِ. فيضطَّجعونَ بلا مَجدٍ معَ الهابطينَ في الهاويةِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="حِزْقِيال 38:6; حِزْقِيال 39:2; حِزْقِيال 28:21"> ”وهُناكَ أُمراءُ الشِّمالِ كُلُّهُم وجميعُ الصَّيدونيِّينَ الّذينَ هبطوا معَ القتلى. كانوا جبابِرةً مُرعِبين فحلَّ بهِمِ الخِزيُ واضطَّجعوا بلا مَجدٍ معَ القتلى بالسَّيفِ، وحملوا عارَهُم معَ الهابطينَ في الهاويةِ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="حِزْقِيال 31:16; حِزْقِيال 14:22; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:13"> وقالَ السَّيِّدُ الرّبُّ: ”ويراهُم فِرعَونُ جميعًا فيتعزَّى هوَ وجيوشُهُ الّذينَ قُتِلوا بالسَّيفِ،</span>
<span className="v" id="v32" data-v="32" data-xref="حِزْقِيال 32:27; التَّكوين 35:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:11"> فبَعدَ أنْ ألقى رُعبَهُ في أرضِ الأحياءِ رقدَ هوَ وجيوشُهُ بلا مَجدٍ معَ القتلى بالسَّيفِ، يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
