## نبوءة على بني عمون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="إرْمِيا 49:1–6; عامُوس 1:13–15; حِزْقِيال 6:2"> ”يا اَبنَ البشَرِ، إلتفتْ إلى بَني عَمُّونَ وتنبَّأْ علَيهِم.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 36:2; المَزامِير 70:2–3; الأمثال 17:5"> وقُلْ لهُم إِسمعوا كَلِمةَ السَّيِّدِ الرّبِّ: بما أنَّكُم فَرِحتُم عِندَما تَدنَّسَ هَيكلي وخَرِبت أرضُ إِسرائيلَ وذهبَ بَيتُ يَهوذا إلى السَّبْي،</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 28:33; القُضاة 6:33; المُلُوكِ الأوَّلُ 4:30"> لذلِكَ أجعَلُ أرضَكُم لأبناءِ المَشرِقِ ميراثًا. فيُقيمونَ حظائِرَهُم، ويَنصِبونَ خيامَهُم، ويأكلونَ ثمرَها ويشربونَ لبنَها.</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 21:20; صموئيلَ الثّانِي 12:26; إشَعْياء 17:2"> وأجعَلُ ربَّةَ مسرحًا للإبلِ وأراضي عَمُّونَ مربِضًا للغنَمِ، فتعلمونَ أنِّي أنا هوَ الرّبُّ.</span>
<span className="v" id="v6" data-v="6" data-xref="صَفَنْيا 2:8; عُوبَدْيا 1:12; صَفَنْيا 2:10"> وبما أنَّكُم صفَّقتُم بأيديكُم وخبطتُم بأرجلِكُم مِنَ الفرَحِ وشمتُّم بكُلِّ اَحتقارٍ ومِنْ كُلِّ قلوبِكُم بأرضِ إِسرائيلَ،</span>
<span className="v" id="v7" data-v="7" data-xref="صَفَنْيا 1:4; حِزْقِيال 25:16; حِزْقِيال 6:14"> لذلِكَ أمدُّ يدي علَيكُم وأجعَلُكُم نَهبًا للأمَمِ، وأقطَعُكُم مِنَ الشُّعوبِ وأبيدُ أرضَكُم مِنَ الأراضي وأُدمِّرُكُم، فتعلمونَ أنِّي أنا هوَ الرّبُّ“.</span>
</div>
</div>

## نبوءة على موآب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 15:1–9; إرْمِيا 25:21; إرْمِيا 48:1–47"> وقالَ السَّيِّدُ الرّبُّ: ”وبما أنَّ موآبَ وسعيرَ قالَت إنَّ بَيتَ يَهوذا مِثلُ جميعِ الأمَمِ،</span>
<span className="v" id="v9" data-v="9" data-xref="إرْمِيا 48:23; يَشُوع 13:17; يَشُوع 13:19–20"> لذلِكَ أفتحُ أعالي أرضِ موآبَ مِنْ جهةِ بَيتِ يشِموتَ وبَعلِ مَعونَ وقريتايم فأقضي بذلِكَ على مَجدِها</span>
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 21:32; المَزامِير 83:3–6; حِزْقِيال 25:4"> وأُسلِّمُ بَني عَمُّونَ ميراثًا لأبناءِ المَشرِقِ، فلا يُذكَرُ بَنو عَمُّونَ في الأمَمِ.</span>
<span className="v" id="v11" data-v="11" data-xref="حِزْقِيال 35:15; حِزْقِيال 30:19; المَزامِير 149:7"> وأُجري أحكامًا على موآبَ، فيعلموا أنِّي أنا هوَ الرّبُّ“.</span>
</div>
</div>

## نبوءة على آدوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 49:7–22; حِزْقِيال 25:8; المَزامِير 137:7"> وقالَ السَّيِّدُ الرّبُّ: ”وبما أنَّ أدومَ اَنتقمت مِنْ بَيتِ يَهوذا واَرتكبت إثْمًا باَنتقامِها مِنهُم،</span>
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 29:8; مَلاخِي 1:3–4; إرْمِيا 25:23"> لذلِكَ أمدُّ يدي على أدومَ وأقطعُ مِنها البشَرَ والبَهائمَ وأجعلُها خرابًا. مِنْ تيمانَ إلى ددَّانَ يسقطونَ بالسَّيفِ.</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 11:14; التَّكوين 27:29; العِبرانِيّين 10:30–31"> وأصبُّ اَنتقامي على أدومَ بأيدي شعبي إِسرائيلَ، فيفعَلونَ في أدومَ بمُقتَضى غضَبي وسَخطي، فيَعلمونَ اَنتقامي، يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
</div>

## نبوءة على الفلسطيّين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="حِزْقِيال 25:6; إشَعْياء 14:29–31; حِزْقِيال 25:12"> وقالَ السَّيِّدُ الرّبُّ: ”وبما أنَّ الفلِسطيِّينَ اَنتقموا مِنْ يَهوذا اَنتقامًا فظيعًا بقلوبٍ خبيثةٍ وأكثروا التَّدميرَ في أرضِ عدوٍّ قديمٍ،</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 30:14; صَفَنْيا 2:4–15; حِزْقِيال 25:13"> لذلِكَ أمدُّ يدي على الفلِسطيِّينَ وأقطعُ الكريتيِّينَ وأُبيدُ بقيَّةَ سكَّانِ ساحلِ البحرِ،</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 25:11; حِزْقِيال 25:14; حِزْقِيال 5:15"> وأُجري علَيهِمِ اَنتقامًا رهيبًا وأُنزِلُ بهِم عِقابًا شديدًا، فيَعلمونَ حينَ أحلُّ اَنتقامي علَيهِم أنِّي أنا هوَ الرّبُّ“.</span>
</div>
