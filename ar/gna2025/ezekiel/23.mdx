## الأختان الزانيتان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ ليَ الرّبُّ:</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 16:44–46; إرْمِيا 3:7–10"> ”يا اَبنَ البشَرِ، كانَتِ اَمرأتانِ، اَبنَتا أُمٍّ واحدةٍ،</span>
<span className="v" id="v3" data-v="3" data-xref="اللّاويِّين 17:7; حِزْقِيال 16:22; يَشُوع 24:14"> وزنَتا في صِباهما في مِصْرَ. هُناكَ دغدغوا ثَدْييهِما وداعبوا نُهودَ بكارتِهِما.</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 16:8; المَزامِير 76:2; إرْمِيا 2:2–3"> وكانَ إِسمُ الكُبرى أهولةَ واَسمُ أختِها أهوليبةَ. أهولةُ هيَ السَّامرةُ وأهوليبةُ هيَ أُورُشليمُ. تَزوَّجتُهُما فأنجبتا ليَ البنينَ والبَناتِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="هُوشَع 5:13; المُلُوكِ الثّانِي 17:3; المُلُوكِ الثّانِي 16:7"> ”فزنَت أهولةُ عليَّ وعَشِقت مُحبِّيها بَني أشُّورَ جيرانِها،</span>
<span className="v" id="v6" data-v="6" data-xref="حِزْقِيال 23:23"> مِنْ لابسي الأرجوانِ، والحُكَّامِ والوُلاةِ، وجميعِ الفتيانِ الأقوياءِ، والفُرسانِ راكبي الخَيلِ.</span>
<span className="v" id="v7" data-v="7" data-xref="هُوشَع 5:3; هُوشَع 6:10; حِزْقِيال 22:3–4"> وأغدقت فَواحشَها على جميعِ النُّخبةِ مِنْ بَني أشُّورَ وتنجَّست بأصنامِ جميعِ الّذينَ عَشِقَتْهُم.</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 23:19; حِزْقِيال 23:3; الخُرُوج 32:4"> وما أقلعت عَنْ فَواحشَ اَتَّخذتْها في مِصْرَ، حَيثُ ضاجعوها في صِباها وداعبوا نُهودَ بكارتِها وأفرغوا شهوتَهُم علَيها.</span>
<span className="v" id="v9" data-v="9" data-xref="هُوشَع 11:5; المُلُوكِ الثّانِي 15:29; المُلُوكِ الثّانِي 17:23"> لذلِكَ أسلَمتُها إلى أيدي مُحبِّيها بَني أشُّورَ الّذينَ عَشِقَتْهُم.</span>
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 23:29; هُوشَع 2:10; إرْمِيا 22:8–9"> هُم عَرَّوها بَعدَ أنْ أخذوا بَنيها وبَناتِها وقتلوها بالسَّيفِ، فصارَت عِبرةً للنِّساءِ بما أصابَها مِنَ العقابِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 3:8–11; حِزْقِيال 16:47–51; حِزْقِيال 23:4"> ”فرأت أختُها أهوليبةُ ما فعلَتْهُ أهولةُ، فزادت علَيها فسادًا في عِشقِها وتفوَّقَت علَيها في فَواحشِها،</span>
<span className="v" id="v12" data-v="12" data-xref="المُلُوكِ الثّانِي 16:7–15; حِزْقِيال 23:23; أخبارِ الأيّامِ الثّانِي 28:16–23"> فعشِقَت بَني أشُّورَ مِنَ الحُكَّامِ والوُلاةِ جيرانِها لابسي الثِّيابِ الفاخرةِ، والفُرسانِ راكبي الخَيلِ وجميعِ الفتيانِ الأقوياءِ.</span>
<span className="v" id="v13" data-v="13" data-xref="هُوشَع 12:1–2; المُلُوكِ الثّانِي 17:18–19; حِزْقِيال 23:31"> فرأيتُ أنَّها تنجَّست وأنَّ لها ولأختِها طريقًا واحدًا،</span>
<span className="v" id="v14" data-v="14" data-xref="حِزْقِيال 8:10; إرْمِيا 22:14; إرْمِيا 50:2"> لكنَّها زادت على فَواحشِها حينَ رأت صُوَرَ رجالٍ مِنَ البابليِّينَ منقوشةً بالأحمرِ على الحائطِ،</span>
<span className="v" id="v15" data-v="15" data-xref="القُضاة 8:18; صَمُوئيلَ الأوَّلُ 18:4; إشَعْياء 22:21"> حَولَ خُصورِهِم أحزِمةٌ وعلى رؤوسِهِم عَمائِمُ مُتهدِّلةٌ، ولجميعِهِم منظَرُ القادةِ الكبارِ مِنْ بَني بابلَ في أرضِ مولدِهِم.</span>
<span className="v" id="v16" data-v="16" data-xref="حِزْقِيال 16:29; ﻣﺘﻰ 5:28; أيُّوب 31:1"> فما إنْ لمَحتهُم عيناها حتّى عشِقَتهُم، وأرسلتْ إليهِم رُسُلًا إلى أرضِ البابليِّينَ.</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 23:28; حِزْقِيال 23:22; التَّكوين 10:10"> فجاءَها بَنو بابلَ واَرتكبوا معَها الفحشاءَ ونجَّسوها، فتنجَّست بهِم ثم عافتهُم نفسُها.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 78:59; المَزامِير 106:40; إرْمِيا 12:8"> وأظهرت فَواحشَها وتعرَّت، فعافتها نفْسي كما عافت نفْسي أختَها.</span>
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 23:3; حِزْقِيال 20:7; حِزْقِيال 16:51"> وأكثرت فَواحشَها لتتذكَّرَ أيّامَ صِباها الّتي زنت فيها في أرضِ مِصْرَ،</span>
<span className="v" id="v20" data-v="20" data-xref="حِزْقِيال 17:15; حِزْقِيال 16:26; إرْمِيا 5:8"> وعَشِقت رجالًا في شهوةِ الحميرِ والخَيل.</span>
<span className="v" id="v21" data-v="21"> هكذا يا أهوليبةُ اشتَقتِ إلى فُجورِ صِباكِ، حينَ داعبَ المصريُّونَ نهدَيكِ وثَدييكِ الفتيَّينِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 16:37; رُؤيا يُوحَنّا 17:16; إرْمِيا 6:22–23"> ”لذلِكَ يا أهوليبةُ، هكذا قالَ السَّيِّدُ الرّبُّ. سأُثيرُ علَيكِ عُشَّاقَكِ الّذين عافتْهُم نفسُكِ وأجيءُ بهِم لقتالِكِ مِنْ كُلِّ جهةٍ،</span>
<span className="v" id="v23" data-v="23" data-xref="إرْمِيا 50:21; المُلُوكِ الثّانِي 24:2; المُلُوكِ الثّانِي 20:14–17"> مِنْ بَني بابلَ، ومِنْ بَني فقودَ وشوعَ وقوعَ وجميعِ بَني أشُّورَ معَهُم، ومِنَ الفتيانِ والأقوياءِ والحُكَّامِ والوُلاةِ كُلِّهم، ومِنَ القادةِ الكِبارِ وراكبي الخَيلِ كُلِّهم.</span>
<span className="v" id="v24" data-v="24" data-xref="صموئيلَ الثّانِي 24:14; حِزْقِيال 26:10; حِزْقِيال 23:45"> فيُقبِلونَ علَيكِ منَ الشِّمالِ بالمركباتِ والعجَلاتِ وبحشدٍ منَ الشُّعوبِ وينقضُّونَ علَيكِ بالدِّرعِ والتُّرسِ والخُوذةِ منْ كُلِّ جهةٍ، وأُسلِّمُكِ إليهِم، فيحكمونَ علَيكِ بأحكامِهِم.</span>
<span className="v" id="v25" data-v="25" data-xref="حِزْقِيال 23:47; حِزْقِيال 8:1–18; حِزْقِيال 15:6–7"> وأُسلِّطُ غَيرتي علَيكِ، فيعاملونَكِ بسَخطٍ ويقطعونَ أنفَكِ وأُذنيكِ، وتسقطُ بقيّتُكِ بالسَّيفِ، ويأخذونَ بنيكِ وبَناتِكِ، مِنَ المدينةِ الّتي تأكلُها النَّارُ.</span>
<span className="v" id="v26" data-v="26" data-xref="حِزْقِيال 16:39; إرْمِيا 13:22; حِزْقِيال 23:29"> وينزِعونَ عنكِ ثيابَكِ ويأخذونَ أدواتِ زينتِكِ.</span>
<span className="v" id="v27" data-v="27" data-xref="حِزْقِيال 16:41; حِزْقِيال 23:3; حِزْقِيال 23:19"> وأُنهي فُجورَكِ عنكِ وزِناكِ الّذي بدأتِ بهِ في أرضِ مِصْرَ، فلا ترفعينَ عينَيكِ إلى المصريِّينَ ولا تذكُرينَهُم مِنْ بَعدُ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="حِزْقِيال 23:22; حِزْقِيال 23:17; إرْمِيا 34:20"> ”نعم، قالَ السَّيِّدُ الرّبُّ. سأُسلِّمُكِ إلى الّذينَ أبغضتِ وعافتْهُم نفسُكِ،</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 16:39; حِزْقِيال 23:25–26; حِزْقِيال 23:45–47"> فينتقمونَ مِنكِ، ويَسلبونَكِ كُلَّ ما جنيتِه بتعبِكِ، ويتركونَكِ عُريانةً متعرِّيةً، فتنكشِفُ عَورةُ زِناكِ. فُجورُكِ وفَواحشُكِ</span>
<span className="v" id="v30" data-v="30" data-xref="حِزْقِيال 6:9; المَزامِير 106:35–38; إرْمِيا 2:18–20"> جلبتِ علَيكِ هذا، لأنَّكِ زنيتِ مثلَ الأمَمِ وتنجَّستِ بأصنامِهِم.</span>
<span className="v" id="v31" data-v="31" data-xref="المُلُوكِ الثّانِي 21:13; إرْمِيا 7:14–15; حِزْقِيال 16:47–51"> في طريقِ أختِكِ سلكتِ، فأجعلُ كأسَها في يَدِكِ.</span>
<span className="v" id="v32" data-v="32" data-xref="إشَعْياء 51:17; المَزامِير 60:3; حِزْقِيال 22:4–5"> هكذا قالَ السَّيِّدُ الرّبُّ.<br />”تشربينَ كأسَ أختِكِ<br />وهيَ عميقةٌ واسعةٌ،<br />وتكونينَ للضَّحِكِ والسُّخريةِ<br />لأنَّ الكأسَ عظيمةُ الاتِّساعِ،</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 51:17; إرْمِيا 25:27; إرْمِيا 25:15–16"> فتمتلئينَ سُكرًا وحُزنًا<br />كأسُ رُعبٍ وخرابٍ<br />هيَ كأسُ أختِكِ السَّامرةِ،</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="المَزامِير 75:8; إشَعْياء 51:17; رُؤيا يُوحَنّا 18:7"> تشربينَها حتّى الثُّمالةِ<br />وبِشظايا خَزَفِها<br />تُمزِّقينَ ثَدييكِ،<br />لأنِّي أنا تكلَّمتُ،<br />يقولُ السَّيِّدُ الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="المُلُوكِ الأوَّلُ 14:9; إشَعْياء 17:10; إرْمِيا 32:33"> لذلِكَ قالَ السَّيِّدُ الرّبُّ: ”بما أنَّكِ نسيتني ونبذتني وراءَ ظهرِكِ، فتحمَّلي أنتِ عاقبةَ فُجورِكِ وفَواحشِكِ“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="حِزْقِيال 22:2; إشَعْياء 58:1; حِزْقِيال 16:2"> وقالَ ليَ الرّبُّ: ”يا اَبنَ البشَرِ ألا تدينُ أهولةَ وأهوليبةَ؟ فاخبِرْهُما بأرجاسِهِما.</span>
<span className="v" id="v37" data-v="37" data-xref="حِزْقِيال 16:36; إرْمِيا 32:35; إرْمِيا 7:31"> فكُلٌّ مِنهُما زنت ويداها مُلطَّختانِ بالدَّمِ. زنت بعبادةِ الأصنامِ، وبَنوها الّذينَ ولَدتْهُم لي قدَّمتْهُم طَعامًا إكرامًا لِلأصنامِ.</span>
<span className="v" id="v38" data-v="38" data-xref="حِزْقِيال 20:13; المُلُوكِ الثّانِي 21:7; حِزْقِيال 7:20"> وفعلَت بي هذا أيضًا: نجَّست بَيتيَ المُقدَّسَ في الوقتِ نفسِهِ ودنَّست يومَ السَّبتِ.</span>
<span className="v" id="v39" data-v="39" data-xref="المُلُوكِ الثّانِي 21:4; حِزْقِيال 44:7; إرْمِيا 23:11"> كانَت تذبحُ بَنيها لأصنامِها وتدخلُ بَيتيَ المُقدَّسَ في الوقتِ نفسِهِ لتُدنِّسَهُ. نعم، هكذا فعلَت في وسَطِ بيتي.</span>
<span className="v" id="v40" data-v="40" data-xref="المُلُوكِ الثّانِي 9:30; إرْمِيا 4:30; حِزْقِيال 16:13–16"> بل أرسلت إلى رجالٍ غُرَباءَ تدعوهُم مِنْ بعيدٍ. أرسلَت إليهِم رُسُلًا وما أسرعَ ما جاؤوا، فاَغتسلَت لأجلِهِم وكحَّلَت عينيها وتحلَّت بالحلي.</span>
<span className="v" id="v41" data-v="41" data-xref="أسْتِير 1:6; إرْمِيا 44:17; إشَعْياء 65:11"> وجلَست على سريرٍ فاخرٍ وهيَّأت أمامَهُم مائدةً وضعَت علَيها بَخوري وزيتي.</span>
<span className="v" id="v42" data-v="42" data-xref="حِزْقِيال 16:11–12; رُؤيا يُوحَنّا 12:3; الخُرُوج 32:18–19"> وكانَ صوتُ جماعةٍ مِنْ أهلِ المُجونِ يُسمَعُ عِندَها، جماعةٌ مِنْ رُعاعِ القومِ، سكارى جِيءَ بِهِم مِنَ البراري. جَعلوا في يدي كُلِّ واحدةٍ أساورَ وعلى رأسِ كُلٍّ مِنهما تاجًا.</span>
</div>
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="حِزْقِيال 23:3; إرْمِيا 13:23; عَزْرا 9:7"> فقُلتُ: ألا يَزنونَ الآنَ معَ اَمرأةٍ بُليَت بالعُهْرِ؟</span>
<span className="v" id="v44" data-v="44" data-xref="حِزْقِيال 23:9–13; حِزْقِيال 23:3"> فهُم دخَلوا علَيها كما يدخُلُ الرِّجالُ على عاهرةٍ. نعم، هكذا دخلوا على أهولةَ وعلى أهوليبةَ المرأتَينِ الفاجرتَينِ،</span>
<span className="v" id="v45" data-v="45" data-xref="هُوشَع 6:5; اللّاويِّين 20:10; حِزْقِيال 23:36–39"> لكنَّ الأبرارَ سيحكُمونَ علَيهِما حُكمَهُم على الزَّواني وعلى سفَّاكاتِ الدِّماءِ، لأنَّهُما زانيتانِ وأيديهِما مُلطَّخةٌ بالدَّمِ“.</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="حِزْقِيال 16:40; إرْمِيا 15:4; إرْمِيا 24:9"> وهذا ما قالَ السَّيِّدُ الرّبُّ: ”إصعَدْ إليهِما بجُموعِ الشَّعبِ وأسلِمْهُما للرُّعبِ والنَّهبِ.</span>
