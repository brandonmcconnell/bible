## يشوع يختن بني إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 30:6; التَّثْنِيَة 10:16; الخُرُوج 4:25"> في ذلِكَ الوقتِ قالَ الرّبُّ ليَشوعَ: ”إصنَعْ لكَ سكاكينَ مِنْ صَوَّانٍ واَختُنْ بَني إِسرائيلَ“.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 16:24; التَّكوين 17:23–27"> فصنَعَ يَشوعُ سكاكينَ مِنْ صَوَّانٍ وختَنَ بَني إِسرائيلَ عِندَ جَبعَةَ هاعَرلوتَ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="العَدَد 26:64–65; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:5; التَّثْنِيَة 2:16"> وسبَبُ خَتْنِ يَشوعَ لهُم كانَ: أنَّ كُلَّ ذَكَرٍ ومُحارِبٍ مِنَ الشَّعبِ الّذينَ خرجوا مِنْ مِصْرَ ماتَ في البرِّيَّةِ على الطَّريقِ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 12:8–9; هُوشَع 6:6–7; ﻣﺘﻰ 12:7"> وكانَ كُلُّ الشَّعبِ الّذينَ خرجوا مِنْ مِصْرَ اَختَتَنوا، أمَّا الّذينَ وُلِدوا في البرِّيَّةِ في الطَّريقِ فلم يَختَتِنوا</span>
<span className="v" id="v6" data-v="6" data-xref="العَدَد 14:23; التَّثْنِيَة 2:14; العِبرانِيّين 3:11"> لأنَّ بَني إِسرائيلَ ساروا أربعينَ سنَةً في البرِّيَّةِ إلى أنِ اَنقَرَضَ كُلُّ المُحاربينَ الخارِجينَ مِنْ مِصْرَ الّذينَ لم يُطيعوا أمرَ الرّبِّ فأقسَمَ أنْ لا يُريَهُمُ الأرضَ الّتي حلَفَ لآبائِهِم أنْ يُعطيَها لِبَنيهِم، أرضًا تَدُرُّ لَبنًا وعسَلًا.</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 14:31; التَّثْنِيَة 1:39"> أمَّا بَنوهُمُ الّذينَ خلَّفوهُم فهُمُ الّذينَ ختَنهُم يَشوعُ لأنَّهُم لم يَختَتِنوا في الطَّريقِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 34:25"> ولمَّا اَختَتَنَ جميعُ الشَّعبِ أقاموا في المحلَّةِ إلى أنْ بَرِئوا.</span>
<span className="v" id="v9" data-v="9" data-xref="يَشُوع 4:19; صَمُوئيلَ الأوَّلُ 17:26; التَّكوين 34:14"> فقالَ الرّبُّ ليَشوعَ: ”اليومَ نَزَعتُ عارَ المِصْريِّينَ عَنكُم“. فدُعيَ ذلِكَ المكانُ الجِلجالَ إلى هذا اليومِ.</span>
</div>
</div>

## الفصح الأول في كنعان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 12:6; العَدَد 9:1–5; حِزْقِيال 12:3"> ونزَلَ بَنو إِسرائيلَ بالجِلجالِ واَحتَفَلوا بِعيدِ الفِصحِ في مساءِ اليومِ الرَّابِعَ عشَرَ مِنَ الشَّهرِ في سهلِ أريحا،</span>
<span className="v" id="v11" data-v="11" data-xref="اللّاويِّين 23:14; الخُرُوج 13:6–7; اللّاويِّين 23:6"> وفي اليومِ الّذي تَلا الفِصحَ أكلوا مِن غلَّةِ الأرضِ فطيرًا وفريكًا.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 16:35; إشَعْياء 65:13–14; الأمثال 13:22"> ومُنذُ أنْ أكلوا مِنْ غلَّةِ الأرضِ اَنقطَعَ عَنهُمُ المَنُّ فعاشوا مِنْ غلَّةِ أرضِ كنعانَ في تِلكَ السَّنةِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 18:2; العَدَد 22:23; العَدَد 22:31"> ولمَّا كانَ يَشوعُ عِندَ أريحا رفَعَ عينَيهِ فرأى رَجُلًا واقِفًا قُبالتَهُ وفي يَدِهِ سَيفُهُ مَسلولٌ. فاَقترَبَ مِنهُ يَشوعُ وسألَهُ: ”أمَعَنا أنتَ أمْ معَ أعدائِنا؟“</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 55:4; التَّكوين 17:3; رُؤيا يُوحَنّا 19:10–14"> فأجابَ: ”كلَّا، بل أنا رئيسُ جُندِ الرّبِّ وها أنا الآنَ جئتُ لخِدمَتِكَ“. فاَنحَنى يَشوعُ حتّى الأرضِ وسجَدَ وقالَ: ”بماذا تأمُرُ عبدَكَ يا ربُّ؟“</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 3:5; اعمال الرسل 7:32–33; رسالَةُ بُطرُسَ الثّانيَة 1:18"> فقالَ لهُ رئيسُ جُندِ الرّبِّ: ”إخلَعْ نعلَيكَ مِنْ رِجلَيكَ لأنَّ الموضِعَ الّذي أنتَ فيهِ مُقدَّسٌ“. ففعَلَ يَشوعُ كذلِكَ.</span>
</div>
