## مواليد رأوبين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 35:22; التَّكوين 29:32; التَّكوين 48:15–22"> ولِرَأوبينَ، بِكْرِ يَعقوبَ، بَنونَ. كانَ هوَ البِكْرَ، إلَّا أنَّه ضاجَعَ جواريَ أبيهِ فخَسِرَ حَقَّ البُكوريَّةِ، فأُعطيَ الحَقُّ لِبني يوسفَ بنِ يَعقوبَ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 2:6; ميخا 5:2; العِبرانِيّين 7:14"> وقوِيَ سِبْطُ يَهوذا على سائِرِ أسباطِ إخوتِهِ، فخَرَجَ مِنهُ مَلِكٌ علَيهِم جميعًا.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 46:9; الخُرُوج 6:14; العَدَد 26:5–9"> وبَنو رَأوبينَ، بِكْرِ يَعقوبَ، أربعَةٌ: حَنوكُ وفَلُّو وحَصرونُ وكرمي.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4"> وبَنو يوئيلَ: شَمْعيا اَبنُهُ، وجوجُ بنُ شَمْعيا، وشَمعي بنُ جوجَ،</span>
<span className="v" id="v5" data-v="5"> وميخا بنُ شَمعي، ورآيا بنُ ميخا، وبَعلُ بنُ رآيا،</span>
<span className="v" id="v6" data-v="6" data-xref="أخبارِ الأيّامِ الأوّلُ 5:26; المُلُوكِ الثّانِي 15:29; أخبارِ الأيّامِ الثّانِي 28:20"> وبَثيرَةُ بنُ بَعلَ، وهوَ الّذي أسَرَهُ تَغلثْ فلاسَرُ مَلِكُ أشُّورَ، وهوَ رئيسُ الرَّأوبينيِّينَ.</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الأوّلُ 5:17"> وأنسباؤُهُ بِحسَبِ عشائِرِهِم ومَواليدِهِم: يَعيئيلُ وزكَريا</span>
<span className="v" id="v8" data-v="8" data-xref="أخبارِ الأيّامِ الأوّلُ 5:4; العَدَد 32:34; العَدَد 32:38"> وبالِعُ بنُ عازارَ بنِ شامِعَ بنِ يوئيلَ هؤلاءِ سكَنوا في عَروعيرَ واَنتَشَروا شمالًا إلى نَبو وبَعلَ مَعونَ،</span>
<span className="v" id="v9" data-v="9" data-xref="يَشُوع 22:8–9"> وشرقًا إلى مَدخَلِ الصَّحراءِ المُؤدِّيةِ إلى نهرِ الفُراتِ، لأنَّ ماشيتَهُم كثُرَت في أرضِ جلعاد.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 21:9; أخبارِ الأيّامِ الأوّلُ 5:18–21; المَزامِير 83:6"> وقاتَلوا الهاجريِّينَ في أيَّامَ شاوُلَ وأبادوهُم، فسكنوا في خيامِهِمِ المُنتَشِرةِ في القِسمِ الشَّرقي مِنْ جلعاد.</span>
</div>
</div>

## مواليد جاد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يَشُوع 13:11; يَشُوع 13:24–28; التَّثْنِيَة 3:10–17"> وسكَنَ بَنو جادَ مُقابِلَهُم في أرضِ باشانَ المُمتَدَّةِ إلى سَلْكَةَ.</span>
<span className="v" id="v12" data-v="12"> وكانَ يوئيلُ رئيسًا، ويليهِ شافامُ. وكانَ يَعنايُ وشافاطُ رئيسَينِ أيضًا في باشانَ.</span>
<span className="v" id="v13" data-v="13"> وأنسِباؤُهُم بِحسَبِ عيالِهِم سبعةٌ: ميخائيلُ ومُشلَّامُ وشَبَعُ ويُورايُ ويَعكانُ وزيعُ وعابِرُ.</span>
<span className="v" id="v14" data-v="14"> هؤلاءِ بَنو أبيجايِلَ بنِ حُوري بنِ ياروحَ بنِ جلعادَ بنِ ميخائيلَ بنِ يشيشايَ بنِ يَحْدو بنِ بُوزَ.</span>
<span className="v" id="v15" data-v="15"> وكانَ أخي بنُ عبديئيلَ بنِ جُوني رئيسَ بيتِ أبائِهِم.</span>
<span className="v" id="v16" data-v="16" data-xref="أخبارِ الأيّامِ الأوّلُ 27:29; نَشِيدِ الأنشاد 2:1; إشَعْياء 35:2"> وسكنوا في جلعادَ، في باشانَ وتوابِعِها، وفي جميعِ مراعي شارونَ.</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الثّانِي 14:16; المُلُوكِ الثّانِي 14:28; المُلُوكِ الثّانِي 15:32"> وجرى تَدوينُ اَنتِسابِهِم جميعًا في أيّامِ يوتامَ مَلِكِ يَهوذا ويَربعامَ مَلِكِ إسرائيلَ.</span>
</div>
</div>

## جيوش الأسباط الشرقيين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="العَدَد 1:3; يَشُوع 4:12–13"> وكانَ في بني رَأوبينَ وبَني جادَ وبَني نِصفِ سِبْطِ منَسَّى أَرْبَعَةٌ وَأَرْبَعُونَ أَلْفًا وَسَبْعُ مِئَةٍ وَسِتُّونَ مُقاتِلًا جبَّارًا مُدَرَّبًا على القِتالِ بالتُّرْسِ والسَّيفِ والقَوسِ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 25:15; أخبارِ الأيّامِ الأوّلُ 1:31; أخبارِ الأيّامِ الأوّلُ 5:10"> فقاتلوا الهاجريِّينَ وأعوانَهُم</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 22:4–5; المَزامِير 9:10; أخبارِ الأيّامِ الثّانِي 18:31"> وغلَبوهُم لأنَّهُم صرَخوا إلى اللهِ في القِتالِ فاَستجابَ لهُم لاتِّكالِهِم علَيهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="رُؤيا يُوحَنّا 18:13; العَدَد 31:35"> ونهَبوا خمسينَ ألفًا مِنَ الجِمالِ ومئَتَينِ وخمسينَ ألفًا مِنَ الغنَمِ، وألفَينِ مِنَ الحميرِ، وأسَروا مِنَ النَّاسِ مئَةَ ألفٍ.</span>
<span className="v" id="v22" data-v="22" data-xref="أخبارِ الأيّامِ الثّانِي 32:8; ﺭﻭﻣﻴﺔ 8:31; يَشُوع 23:10"> وسقطَ قَتلى كثيرونَ مِنَ الهاجريِّينَ لأنَّ القِتالَ ضِدَّهُم كانَ بِمَشيئةِ اللهِ، وسكَنَ بَنو رَأوبينَ في أماكِنِهم إلى وقتِ السَّبْي.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 3:8–9; نَشِيدِ الأنشاد 4:8; المَزامِير 133:3"> وسكنَ بَنو نِصفِ سِبْطِ منَسَّى في أرضِ باشانَ المُمتَدَّةِ إلى بَعلَ حَرمونَ وسَنيرَ وجبَلِ حَرمونَ، لأنَّهم تكاثَروا.</span>
<span className="v" id="v24" data-v="24" data-xref="أخبارِ الأيّامِ الأوّلُ 4:38"> وهؤلاءِ رُؤساءُ عائلاتِهِم: عافَرُ ويَشعي وأليئيلُ وعَزْرئيلُ ويَرْميا وهودَيا ويَحديئيلُ، وكُلُّهُم رِجالٌ أشِدَّاءُ ورؤساءُ مَشهورونَ لِعائلاتِهِم.</span>
<span className="v" id="v25" data-v="25" data-xref="رُؤيا يُوحَنّا 17:5; المَزامِير 106:34–39; الخُرُوج 34:15"> لكنَّ الشَّعبَ خانوا إلهَ آبائِهِم وخَطِئوا باَتِّباعِ آلهةِ شعوبٍ أُخرى طرَدَهُمُ الرّبُّ مِنْ أمامِهِم.</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الثّانِي 15:19; المُلُوكِ الثّانِي 17:6; المُلُوكِ الثّانِي 15:29"> فأثارَ الرّبُّ علَيهِم فولَ مَلِكَ أشُّورَ، وهوَ تَغلَثْ فلاسَرُ، فسَبى بَني رَأوبينَ وبَني جادَ وبَني نِصفِ منَسَّى، وجاءَ بِهِم إلى حَلَحَ وخابورَ وهارا ونهرِ جوزانَ وهُم فيها إلى هذا اليومِ.</span>
</div>
</div>

## مواليد لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27"> وبَنو لاويّ: جَرشونُ وقَهاتُ ومَراري.</span>
<span className="v" id="v28" data-v="28"> وبَنو قَهاتَ عَمرامُ ويِصْهارُ وحبرونُ وعُزِّيئيلُ.</span>
<span className="v" id="v29" data-v="29"> وبَنو عَمرامُ: هرونُ وموسى ومريمُ. وبَنو هرونَ: نادابُ وأبيهو وألعازارُ وإيثامارُ.</span>
<span className="v" id="v30" data-v="30"> وألعازارُ ولَدَ فينحاسَ، وفينحاسُ ولَدَ أبيشوعَ،</span>
<span className="v" id="v31" data-v="31"> وأبيشوعُ ولَدَ بُقِّيَ، وبُقِّي ولَدَ عُزِّيَ،</span>
<span className="v" id="v32" data-v="32"> وعُزِّي ولَدَ زَرَحْيا، وزَرَحْيا ولَدَ مَرايوثَ،</span>
<span className="v" id="v33" data-v="33"> ومرايوثُ ولَدَ أمَرْيا، وأمَرْيا ولَدَ أخيطوبَ،</span>
<span className="v" id="v34" data-v="34"> وأخيطوبُ ولَدَ صادوقَ، وصادوقُ ولَدَ أخيمعصَ،</span>
<span className="v" id="v35" data-v="35"> وأخيمعصُ ولَدَ عزَرْيا، وعزَرْيا ولَدَ يوحنانَ،</span>
<span className="v" id="v36" data-v="36"> ويوحنانُ ولَدَ عزَرْيا وهوَ الّذي خدمَ في بَيتِ الرّبِّ الّذي بناهُ سُليمانُ في أورُشليمَ،</span>
<span className="v" id="v37" data-v="37"> وعزَرْيا ولَدَ أمَرْيا وأمَرْيا ولَدَ أخيطوبَ،</span>
<span className="v" id="v38" data-v="38"> وأخيطوبُ ولَدَ صادوقَ، وصادوقُ ولَدَ شلُّومَ،</span>
<span className="v" id="v39" data-v="39"> وشلُّومُ ولَدَ حَلْقيا، وحَلْقيا ولَدَ عزَرْيا،</span>
<span className="v" id="v40" data-v="40"> وعزَرْيا ولَدَ سَرايا، وسَرايا ولَدَ يوصاداقَ،</span>
<span className="v" id="v41" data-v="41"> ويوصاداقُ سَباهُ الرّبُّ معَ سُكَّانِ يَهوذا، وأورُشليمَ على يَدِ نَبوخذنَصَّرَ.</span>
