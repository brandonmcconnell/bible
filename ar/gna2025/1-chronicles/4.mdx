## مواليد يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 46:12; التَّكوين 38:29; العَدَد 26:20–21"> وبَنو يَهوذا: فارَصُ وحصرونُ وكَرمي وحُورُ وشوبالُ.</span>
<span className="v" id="v2" data-v="2" data-xref="القُضاة 13:25; أخبارِ الأيّامِ الأوّلُ 2:52–54; يَشُوع 15:33"> واَبنُ شوبالَ: رآيا ولَدَ يَحَثَ، ويَحَثَ ولَدَ أخومايَ ولاهدَ. هذِهِ عشائِرُ الصَّرْعيِّينَ.</span>
<span className="v" id="v3" data-v="3" data-xref="القُضاة 15:11; أخبارِ الأيّامِ الثّانِي 11:6"> وهؤلاءِ بَنو عيطَمَ: يَزرَعيلُ ويَشْما ويَدْباشُ، واَسمُ أُختِهِم هَصَّلَلفوني.</span>
<span className="v" id="v4" data-v="4" data-xref="أخبارِ الأيّامِ الأوّلُ 4:18; أخبارِ الأيّامِ الأوّلُ 2:50; أخبارِ الأيّامِ الأوّلُ 4:39"> وفنوئيلُ بَنى مدينةَ جَدورَ، وعازَرُ بَنى حُوشَةَ، وهُما اَبنا حُورَ بِكْرِ أفراتَةَ الّذي بَنى بَيتَ لَحمَ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الأوّلُ 2:24"> وكانَ لأشحورَ باني تَقوعَ اَمرأتانِ هُما: حَلاةُ ونَعرَةُ.</span>
<span className="v" id="v6" data-v="6"> فولَدَت لَه نَعرَةُ أخُزَّامَ وحافَرَ وتَيمانيَّ وأخشَتاريّ.</span>
<span className="v" id="v7" data-v="7"> وولَدَت لَه حَلاةُ صَرَتَ وصوحَرَ وأثنانَ.</span>
<span className="v" id="v8" data-v="8"> وكانَ قوصُ أبا عانوبَ وهَصوبيبَةَ وعشائر أخَرحيلَ بنِ هارُمَ.</span>
<span className="v" id="v9" data-v="9" data-xref="صَمُوئيلَ الأوَّلُ 4:21; أخبارِ الأيّامِ الأوّلُ 7:23; التَّكوين 35:18"> وكانَ يعبيصُ أشهَرَ إخوَتِهِ، وسَمَّتْهُ أمُّهُ يعبيصَ لأنَّها قالَت في نفْسِها: ”ولَدْتُهُ بِحزنٍ“.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 7:7–11; المَزامِير 116:1–4; الأمثال 10:22"> وصلَّى يعبيصُ إلى إلهِ إِسرائيلَ: ”بارِكْني، ووَسِّعْ أرضي، لِتكُنْ يَدُكَ معي، ولَتحَفظْني مِنَ الشَّرِّ لِئلَّا أشقَى“. فاَسْتجابَ اللهُ لِصلاتِهِ.</span>
</div>
</div>

## عشائر أخرى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11"> وكَلوبُ أخو شوحَةَ ولَدَ مَحيرَ، وهوَ أبو أشتونَ.</span>
<span className="v" id="v12" data-v="12"> وأشتونُ ولَدَ بَيتَ رافا وفاسِحَ وتَحِنَّةَ باني مدينةِ ناحاشَ. هؤلاءِ أهلُ رَيكَةَ.</span>
<span className="v" id="v13" data-v="13" data-xref="يَشُوع 15:17; القُضاة 1:13; القُضاة 3:9–11"> وقَنازُ ولَدَ عُثنئيلَ وساراي، وعُثنئيلُ ولَدَ حثاتَ.</span>
<span className="v" id="v14" data-v="14" data-xref="نَحَمْيا 11:35; المُلُوكِ الثّانِي 24:14"> ومَعونوثايُ ولَدَ عَفرَةَ، وسارايُ ولَدَ يُوآبَ باني جيهاراشيمَ، أي وادي الصُّنَّاعِ، لأنَّهم كانوا صُنَّاعًا.</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 13:6; العَدَد 14:6–10; يَشُوع 15:13–20"> وبَنو كالَبَ بنِ يَفُنَّةَ: عيرُو وأيلَةُ وناعِمُ. وأيلَةُ ولَدَ قَنازَ.</span>
<span className="v" id="v16" data-v="16"> وبَنو يَهلَلئيلَ: زيفُ وزيفَةُ وثيريَّا وأسرَئيلُ.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الأوّلُ 4:19; أخبارِ الأيّامِ الأوّلُ 6:57; يَشُوع 15:50"> وبَنو عَزرَةَ: يَثُرُ ومَرَدُ وعافِرُ ويالونُ. وتزَوَّجَ مَرَدُ بِثيَةَ فولَدَت مِيريام وشَمَّايَ ويِشبَحَ باني مدينةِ أشتَموعَ.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الأوّلُ 2:42; أخبارِ الأيّامِ الأوّلُ 4:39; يَشُوع 15:58"> هؤلاءِ بَنو بِثيَةَ بِنتِ فِرعَونَ. وولَدَت طارَدُ، اَمرأتُه اليهوديَّةُ، يارَدَ باني مدينةِ جَدورَ، وحابَرَ باني سُوكُو، ويَقوتيئيلَ باني زانوحَ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يَشُوع 15:44; صَمُوئيلَ الأوَّلُ 23:1–13"> وتزَوَّجَ هُوديا أُختَ نَحَمَ، وإليهِما تَنتَسِبُ عَشيرَةُ جَرْمَ في قَعيلَةَ وعَشيرَةُ مَعكةَ في أشتَموعَ.</span>
<span className="v" id="v20" data-v="20"> وبَنو شيمونَ: أمنونُ ورِنَّةُ وبُنحانانُ وتيلونُ. واَبنا يِشْعي: زوحَيتُ وبَنزوحَيتُ.</span>
</div>
</div>

## مواليد شيلة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 38:5; أخبارِ الأيّامِ الأوّلُ 2:3; العَدَد 26:20"> وبَنو شيلَةَ بنِ يَهوذا: عيرُ باني مدينةِ لَيكةَ، ولَعدَةَ باني مَريشةَ، وعشائِرُ ناسجي الكتَّانِ مِنْ مدينةِ بَيتَ أشبيعَ،</span>
<span className="v" id="v22" data-v="22"> ويوقيمُ وأهلُ مدينةِ كزيبا، ويوآشُ وسارافُ الّذينَ تَزَوَّجُوا نِساءً مُوآبيَّاتٍ وسكَنوا في بَيتَ لَحمَ. وهذِهِ كُلُّها أحداثٌ قديمةٌ.</span>
<span className="v" id="v23" data-v="23" data-xref="المَزامِير 81:6; أخبارِ الأيّامِ الأوّلُ 4:14"> هؤلاءِ هُمُ الخزَّافونَ الّذينَ أقاموا معَ المَلِكِ في نَتاعيمَ وجديرَةَ وعَمِلوا في خِدمتِهِ.</span>
</div>
</div>

## مواليد شمعون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 46:10; الخُرُوج 6:15; العَدَد 26:12–14"> وبَنو شِمعونَ: نَموئيلُ ويامينُ ويريبُ وزارحُ وشاوُلُ.</span>
<span className="v" id="v25" data-v="25"> وشاوُلُ ولَدَ شَلُّومَ، وشَلُّومُ ولَدَ مِبسامَ، ومِبسامُ ولَدَ مِشْماعَ.</span>
<span className="v" id="v26" data-v="26"> ومِشماعُ ولَدَ حَمُوئيلَ وحَموئيلُ ولَدَ زَكُّورَ، وزَكُّورُ ولَدَ شِمعي.</span>
<span className="v" id="v27" data-v="27" data-xref="العَدَد 26:14; العَدَد 26:22; العَدَد 2:13"> وكانَ لِشِمعي ستَّةَ عشَرَ اَبنًا وستُّ بَناتٍ. ولم يكُنْ لإِخوتِهِ بَنونَ كثيرونَ، وعشائِرُهُم لم تكثُرْ كما كثُرَ بَنو يَهوذا.</span>
<span className="v" id="v28" data-v="28" data-xref="يَشُوع 19:2–3; يَشُوع 15:28–29; يَشُوع 19:9"> وكانوا يسكنونَ في بِئرَ سَبْعَ ومُولادَةَ وحَصَرِ شوعالَ</span>
<span className="v" id="v29" data-v="29" data-xref="يَشُوع 19:3–4"> وبَلْهَةَ وعاصِمَ وتُولادَ</span>
<span className="v" id="v30" data-v="30" data-xref="نَحَمْيا 11:28; يَشُوع 19:4–5; صَمُوئيلَ الأوَّلُ 30:1"> وبَتوئيلَ وحُرمَةَ وصِقلَغَ</span>
<span className="v" id="v31" data-v="31" data-xref="يَشُوع 19:5–6"> وبَيتَ مَركَبوتَ وحَصَرَ سوسيمَ. وبَيتَ بِرْئي وشَعَرايمَ: هذِهِ مُدُنُهم إلى عَهدِ المَلِكِ داوُدَ.</span>
<span className="v" id="v32" data-v="32" data-xref="يَشُوع 19:7"> وقُراهُم خَمْسٌ وهيَ: عَيطَمُ وعَينُ رمونَ وتُوكنُ وعاشانُ</span>
<span className="v" id="v33" data-v="33" data-xref="يَشُوع 19:8"> والمَزارِعُ الّتي حَولَها إلى بَلدةِ بَعالَةَ. هذِهِ مَساكِنُهُم وأنسابُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34"> وأمَّا مَشوبابُ ويَمليكُ ويوشا بنُ أمَصْيا،</span>
<span className="v" id="v35" data-v="35"> ويوئيلُ وياهو بنُ يُوشيبيا بنِ سَرايا بنِ عَسيئيلَ</span>
<span className="v" id="v36" data-v="36"> وأليوعينايُ ويَعقوبا ويَشوحايا وعَسايا وعَدِيئيلُ ويَسِمِئيلُ وبَنايا</span>
<span className="v" id="v37" data-v="37"> وزَيزا بنُ شِفعي بنِ ألُونَ بنِ بَدايا بنِ شِمْري بنِ شَمْعيا،</span>
<span className="v" id="v38" data-v="38" data-xref="أخبارِ الأيّامِ الأوّلُ 5:24; التَّكوين 6:4"> فهُم رؤساءُ عشائِرِهِم وعائِلاتِهِمِ الّذينَ تكاثَروا،</span>
<span className="v" id="v39" data-v="39" data-xref="أخبارِ الأيّامِ الأوّلُ 4:4; أخبارِ الأيّامِ الأوّلُ 4:18; يَشُوع 15:58"> فاَمتَدُّوا غربًا إلى مَدخَلِ جَدورَ سَعيًا وراءَ مرعًى لِماشيَتِهِم شرقيَّ الوادي،</span>
<span className="v" id="v40" data-v="40" data-xref="القُضاة 18:7–10; المَزامِير 105:23; المَزامِير 78:51"> فصادَفوا مرعًى خصيبًا صالحًا، وأرضًا وسيعَةً، هادئَةً وآمنَةً، سكَنَها بَنو حامَ قديمًا.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="إشَعْياء 14:28–32; المُلُوكِ الثّانِي 18:8–12; أخبارِ الأيّامِ الأوّلُ 4:34–38"> وفي أيّامِ حَزْقِيَّا مَلِكِ يَهوذا، جاءَ هؤلاءِ الشِّمعونيُّونَ المذكورَةُ أسماؤُهُم إلى جَدورَ. فهَدَموا خيامَ السَّاكنينَ هُناكَ مِنَ المَعُونيِّينَ وأبادوهُم وسكَنوا في أماكِنِهم إلى هذا اليومِ، لأنَّهُم وجَدوا مرعًى لِماشيتهِم.</span>
<span className="v" id="v42" data-v="42" data-xref="التَّكوين 36:8–9; التَّثْنِيَة 1:2"> وذهَبَ خَمْسُ مئَةِ رَجُلٍ مِنْ بَني شِمعونَ شرقًا إلى جبَلِ سَعيرَ وعلى رأسِهِم بَنو يِشعِي: فَلْطيا ونَعْريا ورفايا وعُزِّيئيلُ،</span>
<span className="v" id="v43" data-v="43" data-xref="صَمُوئيلَ الأوَّلُ 30:17; صَمُوئيلَ الأوَّلُ 15:7–8; صموئيلَ الثّانِي 8:12"> فقَتَلوا مَنْ نَجا مِنْ بَني عَماليقَ وسكَنوا هُناكَ إلى هذا اليومِ.</span>
</div>
