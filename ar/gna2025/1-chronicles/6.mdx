## بنو لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 46:11; الخُرُوج 6:16; العَدَد 26:57"> وبَنو لاويّ: جَرشونُ وقَهاتُ ومَراري.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 6:18; الخُرُوج 6:21–24; أخبارِ الأيّامِ الأوّلُ 23:12"> واَبنا جَرشومَ: لِبني وشِمْعي.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 6:20; اللّاويِّين 10:1; اللّاويِّين 10:12"> وبَنو قَهاتَ: عَمرامُ ويَصهارُ وحبرونُ وعُزِّيئيلُ.</span>
<span className="v" id="v4" data-v="4" data-xref="أخبارِ الأيّامِ الأوّلُ 6:50; الخُرُوج 6:25; عَزْرا 7:1–5"> واَبنا مَراري: مَحلي وموشي. هذِهِ عشائِرُ اللَّاويِّينَ بِحسَبِ أبائِهِم.</span>
<span className="v" id="v5" data-v="5"> واَبنُ لِبني بنِ جَرشومَ: يَحَثُ، واَبنُ يَحَثَ زِمَّةُ،</span>
<span className="v" id="v6" data-v="6"> واَبنُ زِمَّةَ يوآخُ، واَبنُ يوآخَ عِدُّو،</span>
<span className="v" id="v7" data-v="7"> واَبنُ عِدُّو زارحُ، واَبنُ زارحَ يأثَرايُ.</span>
<span className="v" id="v8" data-v="8" data-xref="صموئيلَ الثّانِي 8:17; صموئيلَ الثّانِي 15:27; المُلُوكِ الأوَّلُ 1:44"> واَبنُ عَمينادابَ بنِ قَهاتَ: قورحُ، واَبنُ قورحَ أسيرُ،</span>
<span className="v" id="v9" data-v="9"> واَبنُ أسيرَ تحثُ، واَبنُ تحثَ أوريئيلُ، واَبنُ أوريئيلَ عُزِّيا واَبنُ عُزِّيا شاولُ.</span>
<span className="v" id="v10" data-v="10" data-xref="أخبارِ الأيّامِ الثّانِي 3:4; أخبارِ الأيّامِ الثّانِي 3:1; المُلُوكِ الأوَّلُ 6:1–7"> واَبنا ألقانةَ: عماسايُ وأخيموتُ.</span>
<span className="v" id="v11" data-v="11" data-xref="عَزْرا 7:3"> وبَنو أخيموتَ ألقانةُ، وصوفايَ، ونحثُ،</span>
<span className="v" id="v12" data-v="12" data-xref="نَحَمْيا 11:11; أخبارِ الأيّامِ الأوّلُ 9:11"> وأليآبُ، ويروحامُ، وألقانةُ.</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الثّانِي 35:8; المُلُوكِ الثّانِي 22:12–14; المُلُوكِ الثّانِي 22:4"> واَبنا صموئيلَ يوئيلُ البكرُ: والثاني أبيَّا.</span>
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الثّانِي 25:18; نَحَمْيا 11:11; زَكَريّا 6:11"> واَبنُ مَحلي بنِ مراري: لِبني، واَبنُ لِبني: شِمْعي، واَبنُ شِمْعي: عُزَّةُ،</span>
<span className="v" id="v15" data-v="15" data-xref="حَجَّي 2:2; المُلُوكِ الثّانِي 25:18; حَجَّي 1:1"> واَبنُ عُزَّةَ: شِمعي، واَبنُ شِمْعي: حَجيا، واَبنُ حَجيا: عَسايا.</span>
</div>

## جوقة الغناء في بيت الرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 6:16; أخبارِ الأيّامِ الأوّلُ 6:1"> وهؤلاءِ هُمُ الّذينَ أوكلَ إليهِم داوُدُ إدارةَ جوقةِ الغِناءِ في بَيتِ الرّبِّ، بَعدَ أن اَستقرَّ فيهِ تابوتُ العَهدِ.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الأوّلُ 23:7; العَدَد 3:18; العَدَد 3:21"> وكانوا يَتناوبونَ على الخِدمةِ بالغِناءِ أمامَ خَيمةِ الاجتماعِ، إلى أنْ بَنى سُليمانُ بَيتَ الرّبِّ في أورُشليمَ.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الأوّلُ 23:12; أخبارِ الأيّامِ الأوّلُ 6:2–3"> وهؤلاءِ هُمُ الّذينَ تَوَلَّوا هذا المَنصِبَ، وبَنوهُم مِنْ بَعدِهِم:</span>
<span className="v" id="v19" data-v="19" data-xref="أخبارِ الأيّامِ الأوّلُ 23:21; أخبارِ الأيّامِ الأوّلُ 24:26; الخُرُوج 6:19"> ابنُ ألقانةَ بنِ يَروحامَ بنِ إيلئيلَ بنِ توحَ</span>
<span className="v" id="v20" data-v="20" data-xref="أخبارِ الأيّامِ الأوّلُ 6:43"> بنِ صوفَ بنِ ألقانةَ بنِ مَحَثَ بنِ عماسيَ</span>
<span className="v" id="v21" data-v="21"> بنِ ألقانةَ بنِ يوئيلَ بنِ عزَرْيا بن صَفَنيا</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 6:24; أخبارِ الأيّامِ الأوّلُ 6:18; أخبارِ الأيّامِ الأوّلُ 6:2"> بنِ تَحَثَ بنِ أسيرَ بنِ أبياسافَ بنِ قورحَ</span>
<span className="v" id="v23" data-v="23"> بنِ يَصهارَ بنِ قهاتَ بنِ لاويّ بنِ يَعقوبَ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24"> وكانَ أسافُ ثاني مُديرٍ للجوقةِ، وهوَ ابنُ برَخْيا بنِ شِمْعي</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 6:24; أخبارِ الأيّامِ الأوّلُ 6:35–36"> بنِ ميخائيلَ بنِ بعَسْيا بنِ مَلْكيا</span>
<span className="v" id="v26" data-v="26" data-xref="صَمُوئيلَ الأوَّلُ 1:1; أخبارِ الأيّامِ الأوّلُ 6:35"> بنِ أثنايَ بنِ زارحَ بنِ عَدايا</span>
<span className="v" id="v27" data-v="27" data-xref="صَمُوئيلَ الأوَّلُ 1:1; صَمُوئيلَ الأوَّلُ 1:19–20"> بنِ أيثانَ بنِ زِمَّةَ بنِ شِمْعي</span>
<span className="v" id="v28" data-v="28" data-xref="أخبارِ الأيّامِ الأوّلُ 6:33; صَمُوئيلَ الأوَّلُ 8:2"> بنِ يحثَ بنِ جرشومَ بنِ لاويّ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="أخبارِ الأيّامِ الأوّلُ 6:19; العَدَد 3:33"> وكانَ أيثانُ ثالثَ مُديرٍ للجوقةِ، وهوَ ابنُ قيشي بنِ عبدي بنِ ملُّوخَ</span>
<span className="v" id="v30" data-v="30"> بنِ حشبيا بنِ أمَصْيا بنِ حَلْقيا</span>
<span className="v" id="v31" data-v="31" data-xref="صموئيلَ الثّانِي 6:17; أخبارِ الأيّامِ الأوّلُ 15:25–1; أخبارِ الأيّامِ الأوّلُ 15:16–22"> بنِ أَمصي بنِ باني بنِ شامرَ</span>
<span className="v" id="v32" data-v="32" data-xref="المَزامِير 135:1–3; المُلُوكِ الأوَّلُ 8:6–13; عَزْرا 6:18"> بنِ مَحلي بنِ مُوشي بنِ مَراري بنِ لاويّ.</span>
<span className="v" id="v33" data-v="33" data-xref="أخبارِ الأيّامِ الأوّلُ 6:28; أخبارِ الأيّامِ الأوّلُ 15:17; أخبارِ الأيّامِ الأوّلُ 25:1–5"> وكانَ أنسباؤُهُم مِنَ اللَّاويِّينَ يتَولَّونَ سائرَ الخِدماتِ في بَيتِ اللهِ.</span>

## مواليد هرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34"> وكانَ هرونُ وبَنوهُ مِنْ بَعدِهِ يُقدِّمونَ القرابينَ على مذبَحِ المُحرقةِ ومذبَحِ البَخورِ، ويقومونَ بِكُلِّ ما يَختَصُّ بقُدسِ الأقداسِ وبالذَّبائحِ والتَّقدماتِ الّتي كانَ يُكفِّرُ بها بَنو إِسرائيلَ عنْ خطاياهُم، على حسَبِ ما أمرَ بهِ موسى عبدُ اللهِ.</span>
<span className="v" id="v35" data-v="35"> وهؤلاءِ هُم بَنو هرونَ: ألعازارُ اَبنُهُ، فينحاسُ بنُ ألعازارَ، وأبيشوعُ بنُ فينحاسَ،</span>
<span className="v" id="v36" data-v="36"> وبقيُّ بنُ أبيشوعَ وزَرَحْيا بنُ عُزِّي،</span>
<span className="v" id="v37" data-v="37" data-xref="المَزامِير 45:1; العَدَد 26:10–11; الخُرُوج 6:21–24"> ومَرايوثُ بنُ زَرَحْيا، وأمَرْيا بنُ مَرايوثَ، وأخيطوبُ بنُ أمَرْيا،</span>
<span className="v" id="v38" data-v="38" data-xref="العَدَد 3:19; أخبارِ الأيّامِ الأوّلُ 16:1"> وصادوقُ بنُ أخيطوبَ، وأخيمعصُ بنُ صادوقَ.</span>
</div>

## مساكن اللاويين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="نَحَمْيا 7:44; عَزْرا 2:41; المَزامِير 75:1"> وهذِهِ مَساكِنُ بَني هرونَ مِنْ عَشيرةِ القهاتيِّينَ معَ ضياعِهِم وتُخومِهِم، لأنَّ الحِصَّةَ الأولى كانَت لهُم:</span>
<span className="v" id="v40" data-v="40"> حبرونُ في أرضِ يَهوذا وما حَولَها مِنَ المراعي.</span>
<span className="v" id="v41" data-v="41"> وأمَّا حُقولُ المدينةِ وضِياعُها التَّابعةُ لها، فأُعطيت لكالَبَ بنِ يَفُنَّةَ.</span>
<span className="v" id="v42" data-v="42"> وأُعطيَتْ لِبني هرونَ أيضًا مُدنُ الملجأِ: حبرونُ ولِبنةُ ومَراعيهما. ويَتيرُ وأشتموعُ ومراعيهُما.</span>
<span className="v" id="v43" data-v="43" data-xref="الخُرُوج 6:16; أخبارِ الأيّامِ الأوّلُ 6:16–17; التَّكوين 46:11"> وحيلَينُ ودبيرُ ومراعيهُما،</span>
<span className="v" id="v44" data-v="44" data-xref="المَزامِير 89:1; أخبارِ الأيّامِ الأوّلُ 15:17; أخبارِ الأيّامِ الأوّلُ 25:1"> وعاشانُ وبيَت شَمسُ ومَراعيهما.</span>
<span className="v" id="v45" data-v="45"> وأُعطيَت لهُم مِنْ أرضِ سِبْطِ بِنيامينَ: جَبَعُ ومراعيها، وعَلْمَثُ ومراعيها، وعَناثوثُ ومراعيها. وجميعُ مدُنِهِم ثلاثَ عَشْرةَ مدينةً لِكُلِّ عشائِرِهِم.</span>
<span className="v" id="v46" data-v="46"> وللباقينَ مِنْ بَني قهاتَ أُعطيت عَشْرُ مُدُنٍ بالقرعَةِ مِنْ أرضِ أفرايمَ ودانَ ونِصفِ منَسَّى.</span>
</div>
<div className="bp">
<span className="v" id="v47" data-v="47" data-xref="يَشُوع 21:7; الخُرُوج 6:19; أخبارِ الأيّامِ الأوّلُ 23:21"> وأُعطيَت لِبَني جَرشومَ، بِحسَبِ عشائِرهِم، ثَلاثَ عَشْرَةَ مدينةً مِنْ أرضِ بَني يَسَّاكرَ وأشيرَ ونَفتالي ونِصفِ منَسَّى في باشانَ.</span>
<span className="v" id="v48" data-v="48" data-xref="أخبارِ الأيّامِ الأوّلُ 23:2; العَدَد 3:1–4; العَدَد 18:1–32"> ولِبَني مَراري أُعطيَت، بِحسَبِ عشائرهِم، اَثنتا عَشْرَةَ مدينةً بالقُرعةِ مِنْ أرضِ بَني رَأوبينَ وجادَ وزبولونَ.</span>
<span className="v" id="v49" data-v="49" data-xref="اللّاويِّين 4:20; الخُرُوج 27:1–8; الخُرُوج 30:1–7"> وهكذا أعطَى بَنو إِسرائيلَ للَّاويِّينَ مُدُنًا معَ مراعيها ليسكنوا فيها.</span>
<span className="v" id="v50" data-v="50" data-xref="اللّاويِّين 10:16; أخبارِ الأيّامِ الأوّلُ 24:1; العَدَد 20:26–28"> والمُدُنُ الّتي في أرضِ بَني يَهوذا وشمعونَ وبنيامينَ، والّتي سبَقَ ذِكرُها، أُعطيت لهُم بالقُرعةِ أيضًا.</span>
</div>
<div className="bp">
<span className="v" id="v51" data-v="51"> وبعضُ عشائِرِ بَني قَهاتَ نالوا مُدُنًا معَ مراعيها في أرضِ بَني أفرايمَ،</span>
<span className="v" id="v52" data-v="52"> وهيَ مُدُنُ الملجأِ: شكيمُ في جبَلِ أفرايمَ، وجازرُ،</span>
<span className="v" id="v53" data-v="53" data-xref="صموئيلَ الثّانِي 8:17; حِزْقِيال 44:15; صموئيلَ الثّانِي 15:35–36"> ويَقمَعامُ وبَيتُ حُورنَ</span>
<span className="v" id="v54" data-v="54" data-xref="التَّكوين 25:16; يَشُوع 21:10; يَشُوع 21:3–8"> وأيِّلونُ وجَتُّ رمُّونَ.</span>
<span className="v" id="v55" data-v="55" data-xref="يَشُوع 14:13; يَشُوع 21:11–13; يَشُوع 15:13"> وفي أرضِ سِبْطِ نِصفِ منَسَّى غربًا أُعطيَت للباقينَ مِنْ بَني قَهاتَ عانيرُ ومراعيها، وبِلْعامُ ومراعيها.</span>
</div>
<div className="bp">
<span className="v" id="v56" data-v="56" data-xref="يَشُوع 15:13; يَشُوع 14:13"> وأُعطيَت لِبَني جَرشومَ هذِهِ المُدُنُ معَ مراعيها: في أرضِ نِصفِ سِبْطِ منَسَّى شرقًا: جولانُ في باشانَ وعشتاروتُ.</span>
<span className="v" id="v57" data-v="57" data-xref="يَشُوع 21:13–14; يَشُوع 15:48; يَشُوع 10:29"> في أرضِ سِبْطِ يسَّاكرَ قادِشُ ودَبَرَةُ</span>
<span className="v" id="v58" data-v="58" data-xref="يَشُوع 10:3; يَشُوع 15:49; يَشُوع 12:13"> وراموثُ وعانيمُ.</span>
<span className="v" id="v59" data-v="59" data-xref="يَشُوع 21:16; يَشُوع 15:42; صَمُوئيلَ الأوَّلُ 6:12–19"> وفي أرضِ سِبْطِ أشيرَ: مَشآلُ وعَبدونُ،</span>
<span className="v" id="v60" data-v="60" data-xref="إرْمِيا 1:1; المُلُوكِ الأوَّلُ 2:26; إرْمِيا 11:23"> وحُقوقُ ورَحوبُ.</span>
<span className="v" id="v61" data-v="61" data-xref="يَشُوع 21:4–5; أخبارِ الأيّامِ الأوّلُ 6:18; يَشُوع 21:20–26"> وفي أرضِ سِبْطِ نَفتالي: قادِشُ في الجليلِ، وحمُّونُ وقريَتايمُ.</span>
</div>
<div className="bp">
<span className="v" id="v62" data-v="62" data-xref="يَشُوع 21:27–33; أخبارِ الأيّامِ الأوّلُ 6:71–76; الخُرُوج 2:22"> ولِلباقينَ مِنْ بَني مَراري أُعطيت هذِهِ المُدُنُ معَ مراعيها: في أرضِ سِبْطِ زَبولونَ: رمُّونو وتابورُ.</span>
<span className="v" id="v63" data-v="63" data-xref="يَشُوع 21:7; يَشُوع 21:34–40; العَدَد 3:20"> وفي أرضِ سِبْطِ رَأوبينَ شرقيَّ الأردنِّ قُبالَةَ أريحا: باصِرُ في السَّهلِ ويهصةُ</span>
