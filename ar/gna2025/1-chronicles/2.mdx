## مواليد يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 32:28; التَّكوين 35:22–26; رُؤيا يُوحَنّا 7:5–8"> وهؤلاءِ بَنو يَعقوبَ: رَأوبينُ وشِمعونُ ولاويُّ ويَهوذا ويسَّاكَرُ وزَبولونُ</span>
<span className="v" id="v2" data-v="2"> ودانُ ويوسُفُ وبنيامينُ ونَفتالي وجادُ وأشيرُ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 38:2–10; التَّكوين 46:12; العَدَد 26:19"> وبَنو يَهوذا: عيرُ وأونانُ وشيلَةُ الّذينَ ولَدَتْهُم لَه بِنتُ شُوعَ الكنعانيَّةُ. وكانَ عيرُ بِكْرُ يَهوذا شرِّيرًا في نظَرِ الرّبِّ فأماتَهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 1:3; التَّكوين 38:13–30; نَحَمْيا 11:4"> وولَدَت لَه تامارُ كَنَّتُهُ فارصَ وزارَحَ. فجميعُ بَني يَهوذا خمسةٌ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 46:12; راعُوث 4:18; العَدَد 26:21"> واَبنا فارَصَ: حصرونُ وحامولُ.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 4:31; يَشُوع 7:17–18; يَشُوع 7:1"> وبَنو زارَحَ: زِمري وأيثانُ وهَيمانُ وكلكُولُ ودارَعُ.</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 6:18; التَّثْنِيَة 7:26; يَشُوع 7:25"> واَبنُ كرمي: عاكارُ الّذي عكَّرَ صَفوَ بَني إِسرائيلَ لأنَّهُ اَحتَفَظَ لِنفْسِهِ بِغَنيمةٍ حَرامٍ.</span>
<span className="v" id="v8" data-v="8"> واَبنُ أيثانَ: عزَرْيا.</span>
</div>
</div>

<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 26:6; صموئيلَ الثّانِي 3:39; صموئيلَ الثّانِي 2:18–23"> وأُختاهُم صَرويَّةُ وأبيجايِلُ. وبَنو صَرويَّةَ: أبشايُ ويوآبُ وعَسائيلُ.</span>

## مواليد حصرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18"> وكالَبُ بنُ حصرونَ ولَدَ مِنْ عَزُوبَةَ اَمرأتِهِ اَبنَةً اَسمُها يريعوثُ وثَلاثَةَ بَنينَ هُم: ياشَرُ وشوبابُ وأردونُ.</span>
<span className="v" id="v19" data-v="19" data-xref="أخبارِ الأيّامِ الأوّلُ 2:50; أخبارِ الأيّامِ الأوّلُ 2:24; أخبارِ الأيّامِ الأوّلُ 4:4"> وماتَت عَزُوبَةُ فتَزَوَّجَ كالَبُ أفراتَةَ، فوَلَدَت لَه حُورًا،</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 31:2; الخُرُوج 36:1–2; الخُرُوج 37:1"> وحُورٌ ولَدَ أوري، وأوري ولَدَ بَصَلْئيلَ.</span>
<span className="v" id="v21" data-v="21" data-xref="العَدَد 27:1; التَّثْنِيَة 3:15; العَدَد 26:29"> ثُمَّ تزَوَّجَ حصرونُ بِنتَ ماكيرَ أبي جلعادَ، وهوَ اَبنُ سِتِّينَ سنَةً، فولَدَت لَه سَجوبَ.</span>
<span className="v" id="v22" data-v="22" data-xref="يَشُوع 13:30; التَّثْنِيَة 3:14; العَدَد 32:41"> وسَجوبُ ولَدَ يائيرَ. وكانَت لَه ثلاثٌ وعشرونَ مدينةً في أرضِ جلعادَ.</span>
<span className="v" id="v23" data-v="23" data-xref="يَشُوع 13:30; التَّثْنِيَة 3:14; العَدَد 32:41–42"> ثُمَّ أخَذَ مِنْ جَشُّورَ وآرامَ سِتِّينَ مدينةً، ومنها قُرى يائيرُ وقَناتُ وتوابِعُها. فهذا كُلُّه لِبَني ماكيرَ أبي جلعادَ.</span>
<span className="v" id="v24" data-v="24" data-xref="أخبارِ الأيّامِ الأوّلُ 4:5; أخبارِ الأيّامِ الأوّلُ 2:9; صَمُوئيلَ الأوَّلُ 30:14"> وبَعدَ وفاةِ حصرونَ تزَوَّجَ كالَبُ اَبنَةَ أفراتَةَ اَمرأةِ أبيهِ، فَولَدَت لَه أشحورَ أبا تَقُوعَ.</span>
</div>
</div>

## مواليد يرحمئيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25"> وبَنو يَرحَمْئيلَ بِكْرِ حصرونَ: رامُ بِكْرُه، وبونَةُ وأورَنُ وأوصَمُ وأخيَّا.</span>
<span className="v" id="v26" data-v="26"> وكانَ ليَرحَمْئيلَ اَمرأةٌ أُخرى اَسمُها عَطارَةُ، فولَدَت لَه أونامَ.</span>
<span className="v" id="v27" data-v="27"> وبَنو رامَ: مَعْصٌ ويَمينُ وعاقَرُ.</span>
<span className="v" id="v28" data-v="28"> واَبنا أونامَ: شَمَّايُ وياداعُ. واَبنا شمَّاي: نادابُ وأبيشورُ.</span>
<span className="v" id="v29" data-v="29"> واَسمُ اَمرأةِ أبيشورَ أبيجايلُ الّتي ولَدَت لَه أحبانَ ومُوليدَ.</span>
<span className="v" id="v30" data-v="30"> واَبنا نادابَ: سالَدٌ وأفَّايمُ. وماتَ سالَدُ بِلا بَنينَ.</span>
<span className="v" id="v31" data-v="31" data-xref="أخبارِ الأيّامِ الأوّلُ 2:34–35"> واَبنُ أفَّايمَ: يَشْعي، واَبنُ يَشْعي: شيشانُ، واَبنُ شيشانَ: أحلايُ.</span>
<span className="v" id="v32" data-v="32"> واَبنا ياداعَ، أخي شَمَّايَ: يَثَرُ ويوناثانُ. وماتَ يَثَرُ بِلا بَنينَ.</span>
<span className="v" id="v33" data-v="33"> واَبنا يوناثانَ: فالَتٌ وزازا. هؤلاءِ هُم بَنو يَرحَمئيلَ.</span>
<span className="v" id="v34" data-v="34"> ولم يكُنْ لِشيشانَ بَنونَ بل بَناتٌ. وكانَ لِشيشانَ عَبدٌ مِصْريٌّ اَسمُهُ يَرحَعُ،</span>
<span className="v" id="v35" data-v="35"> فزَوَّجَهُ اَبنَتَهُ، فولَدَت لَه عَتَّايَ.</span>
<span className="v" id="v36" data-v="36" data-xref="أخبارِ الأيّامِ الأوّلُ 11:41"> ومِنْ عَتَّايَ إلى أليشَمَعَ: ناثانُ</span>
<span className="v" id="v37" data-v="37"> وزابادُ وأفلالُ وعوبيدُ</span>
<span className="v" id="v38" data-v="38"> وياهو وعزَرْيا</span>
<span className="v" id="v39" data-v="39"> وحالَصُ وإلعاسَةُ</span>
<span className="v" id="v40" data-v="40"> وسِسْمايُ وشَلُّومُ</span>
<span className="v" id="v41" data-v="41"> ويَقَميَّةُ وأليشَمَعُ.</span>
</div>

## سائر مواليد كالب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="أخبارِ الأيّامِ الأوّلُ 2:23–24; العِبرانِيّين 12:23; ﺭﻭﻣﻴﺔ 8:29"> وبَنو كالَبَ، أخي يَرحَمْئيلَ: ميشاعُ بِكْرُهُ، وهوَ أبو زيفَ. واَبنُ ميشاعَ: مَريشَةُ، وهوَ أبو حبرونَ.</span>
<span className="v" id="v43" data-v="43"> وبَنو حبرونَ: قورَحُ وتَفُّوحُ وراقَمُ وشامَعُ.</span>
<span className="v" id="v44" data-v="44"> وشامَعُ ولَدَ راقَمَ، وهوَ أبو يَرُقْعامَ. وراقَمُ ولَدَ شَمَّايَ.</span>
<span className="v" id="v45" data-v="45" data-xref="يَشُوع 15:58"> وشَمَّايُ ولَدَ مَعُونَ، وهوَ أبو بَيتِ صورَ.</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="أخبارِ الأيّامِ الأوّلُ 2:18–19; أخبارِ الأيّامِ الأوّلُ 2:48"> وكانَ لِكالَبَ جاريَةٌ اَسمُها عيفَةُ، فولَدَت لَه حارانَ وموصا وجازيزَ. وحارانُ ولَدَ جازيزَ.</span>
<span className="v" id="v47" data-v="47"> وبَنو يَهدايَ: راجَمُ ويوثامُ وجيشانُ وفَلَطُ وعيفَةُ وشاعَفُ.</span>
<span className="v" id="v48" data-v="48" data-xref="التَّكوين 25:5–6; أخبارِ الأيّامِ الأوّلُ 2:46"> وأمَّا مَعكةُ جاريةُ كالَبَ الأخرَى، فولَدَت لَه شَبَرَ وتَرحَنَةَ.</span>
<span className="v" id="v49" data-v="49" data-xref="يَشُوع 15:31; أخبارِ الأيّامِ الأوّلُ 2:42; إشَعْياء 10:31"> ثُمَّ ولَدَت شاعَفَ، وهوَ أبو مَدمَنَّةَ، وشَوَا، وهوَ أبو مكْبينا وأبو جَبَعا. وكانَ لِكالَبَ أيضًا اَبنَةٌ اَسمُها عكسةُ.</span>
</div>
<div className="bp">
<span className="v" id="v50" data-v="50" data-xref="صَمُوئيلَ الأوَّلُ 7:1; يَشُوع 15:60; أخبارِ الأيّامِ الأوّلُ 2:53"> وهؤلاءِ أيضًا بَنو كالَبَ: حُورُ بِكْرُهُ مِنِ اَمرأتِهِ أفراتَةَ. وحُورُ ولَدَ شوبالَ باني قَريَةِ يَعاريمَ،</span>
<span className="v" id="v51" data-v="51" data-xref="أخبارِ الأيّامِ الأوّلُ 4:4; يُوحَنّا 7:42; ﻣﺘﻰ 2:6"> وسَلْما باني بَيتَ لَحمَ، وحاريفَ باني بَيتَ جاديرَ.</span>
<span className="v" id="v52" data-v="52" data-xref="أخبارِ الأيّامِ الأوّلُ 4:2"> وبَنو شوبالَ، باني قريَةِ يَعاريمَ: هَرُواهُ، أي نِصفَ سُكَّانِ مَنوحوتَ</span>
<span className="v" id="v53" data-v="53" data-xref="صموئيلَ الثّانِي 23:38; القُضاة 13:25; القُضاة 13:2"> وعشائِرُ قريةِ يَعاريمَ: اليَثريُّونَ والفُوتيُّونَ والشَّماتيُّونَ والمِشراعيُّونَ. ومِنْ هؤلاءِ خرَجَ الصَرْعيُّونَ والأشتاؤوليُّونَ.</span>
<span className="v" id="v54" data-v="54" data-xref="نَحَمْيا 7:26; عَزْرا 2:22; نَحَمْيا 12:28"> وبَنو سلْما، باني قريةِ بَيتَ لَحمَ: النَّطوفاتيُّونَ وعَطْروتُ بَيتِ يوآبَ والصَّرعيُّونَ الّذينَ هُم إحدى عَشيرَتَي مَنوحوتَ.</span>
<span className="v" id="v55" data-v="55" data-xref="المُلُوكِ الثّانِي 10:15; القُضاة 1:16; إرْمِيا 35:19"> وعشائِرُ الصَّفُّوريِّينَ سُكَّانِ يَعبيصَ: التَّرعيُّونَ والشَّمعيُّونَ والسُّوكيُّونَ، وهُم القَينيُّونَ الّذينَ جاؤُوا مِنْ حَمَّةَ وتَحَدَّرَ مِنهُم الرَّكابيُّونَ.</span>
