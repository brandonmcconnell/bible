## من آدم إلى إبراهيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 3:38; التَّكوين 4:25–32"> آدمُ ولَدَ شيتًا، وشيتُ ولَدَ أنوشَ.</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 3:37; التَّكوين 5:12–20; التَّكوين 5:9"> أنوشُ ولَدَ قينانَ، قِينانُ ولَدَ مَهلَلْئيلَ، ومَهلَلْئيلُ ولَدَ يارَدُ.</span>
<span className="v" id="v3" data-v="3" data-xref="يَهُوذا 1:14; التَّكوين 5:21–31; لُوقا 3:36–37"> يارَدُ ولَدَ أخنوخَ، وأخنوخُ ولَدَ مَتوشالَحَ، ومَتوشالَحَ ولَدَ لامَكَ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 5:32; التَّكوين 9:18; لُوقا 17:26"> لامَكُ ولَدَ نُوحًا، ونُوحٌ ولَدَ سامًا وحامًا ويافَثَ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 10:1–5; حِزْقِيال 38:6; حِزْقِيال 27:13"> بَنو يافثَ: جُومَرُ وماجُوحُ ومادايُ وياوانُ وتوبالُ وماشَكُ وتيراسُ.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 10:3"> وبَنو جومَرَ: أشْكَنازُ وريفاتُ وتوجَرْمَةُ.</span>
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 2:10; العَدَد 24:24; إشَعْياء 66:19"> وبَنو ياوانَ: أليشَةُ وتَرشيشُ وكِتِّيمُ ودُودانيمُ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 10:6–7"> بَنو حامَ: كوشٌ ومِصْرائيمُ وفوطُ وكنعانُ.</span>
<span className="v" id="v9" data-v="9"> وبَنو كُوشٍ: سَبا وحويلَةُ وسَبْتا ورَعْما وسَبْتكا. وبَنو رَعْما: شَبا ودادانُ.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 10:8–12; ميخا 5:6"> وكوشٌ ولَدَ نَمرُودَ وهوَ أوَّلُ قائدٍ في الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 10:13–18"> ومِصْرائيمُ ولَدَ لُوديمَ وعَنامَ ولَهابَ ونَفتُوحَ</span>
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 2:23; إرْمِيا 47:4; عامُوس 9:7"> وفَتروسَ وكَسلوحَ وأكريتَ الّذينَ خرَجَ مِنهُمُ الفِلسطيُّونَ.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 9:22; التَّكوين 23:20; التَّكوين 10:15–19"> وكنعانُ ولَدَ صيدونَ بِكْرَهُ وحِثَّا.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 15:21; يَشُوع 3:10; الخُرُوج 33:2"> وكنعانُ كانَ أيضًا أبا اليَبوسيِّينَ والأموريِّينَ والجِرجاشيِّينَ</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 3:8; المُلُوكِ الأوَّلُ 9:20; الخُرُوج 13:5"> والحُوِّيِّينَ والعَرقيِّينَ والسِّينيِّينَ</span>
<span className="v" id="v16" data-v="16" data-xref="العَدَد 34:8; المُلُوكِ الأوَّلُ 8:65"> والأرواديِّينَ والصُّمّاريِّينَ والحُّماثيِّينَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 10:22–32; عَزْرا 4:2; حِزْقِيال 32:24"> بنو سامٍ: عيلامُ وأشُّورُ وأرفَكْشادُ ولُودٌ وأرامُ وعُوصٌ وحُولٌ وجاثِرُ وماشِكُ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 11:12–15; التَّكوين 10:24"> وأرفَكْشادُ ولَدَ شالَحَ، وشالَحُ ولَدَ عابَرَ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 10:21; التَّكوين 11:16–17; التَّكوين 10:25"> ووُلِدَ لِعابَرَ اَبنانِ، أحدُهُما فالَجٌ لأنَّهُ في أيّامِهِ اَنقَسَمَتِ الأرضُ، والآخَرُ يَقطانُ</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 10:26–27"> ويَقطانُ ولَدَ ألمودادَ وشالَفَ وحَضرَموتَ ويارَحَ</span>
<span className="v" id="v21" data-v="21"> وهَدُورامَ وأوزالَ ودِقلَةَ</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 10:28"> وعيبالَ وأبيماييلَ وشَبا</span>
<span className="v" id="v23" data-v="23" data-xref="المُلُوكِ الأوَّلُ 10:11; التَّكوين 25:18; صَمُوئيلَ الأوَّلُ 15:7"> وأوفيرَ وحويلَةَ ويوبابَ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 11:10–26; لُوقا 3:34–36"> ومِنْ سامٍ إلى أبرامَ: أرفَكْشادُ وشالَحُ</span>
<span className="v" id="v25" data-v="25" data-xref="لُوقا 3:35"> وعابَرُ وفالَجُ ورَعُو</span>
<span className="v" id="v26" data-v="26" data-xref="لُوقا 3:34–35"> وسَروجُ وناحُورُ وتارَحُ</span>
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 11:27–32; نَحَمْيا 9:7; يَشُوع 24:2"> وأبرامُ وهوَ إبراهيمُ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 17:19–21; التَّكوين 21:2–5; التَّكوين 21:9–10"> واَبنا إبراهيمَ: إسحَقُ وإسماعيلُ.</span>
</div>
</div>

## مواليد إسماعيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 60:7; التَّكوين 25:12–16; نَشِيدِ الأنشاد 1:5"> وهذِهِ مواليدُ إسماعيلَ: نَبايُوتُ بِكرُهُ وقيدارُ وأدبَئيلُ ومِبْسامُ</span>
<span className="v" id="v30" data-v="30" data-xref="إشَعْياء 21:11; التَّكوين 25:15"> ومِشْماعُ ودُومَةُ ومَسَّا وحَدَدُ وتيماءُ</span>
<span className="v" id="v31" data-v="31"> ويَطورُ ونافيشُ وقِدمَةُ.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 25:1–4; إرْمِيا 25:23; القُضاة 6:1–6"> وأمَّا بَنو قطورةَ، جاريةِ إبراهيمَ، فهُم: زِمْرانُ ويُقشانُ ومَدانُ ومِديانُ ويِشباقُ وشوحٌ. واَبنا يقشانَ: شَبا وددانُ.</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 60:6"> وبَنو مِديانَ: عَيفَةُ وعِفْرٌ وحَنُوكُ وأبيداعُ وألدَعَةُ.</span>
</div>
</div>

## مواليد عيسو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="التَّكوين 32:28; اعمال الرسل 7:8; ﻣﺘﻰ 1:2"> وولَدَ إبراهيمُ إسحَقَ. واَبنا إسحَقَ: عيسو ويَعقوبُ وهوَ إِسرائيلُ.</span>
<span className="v" id="v35" data-v="35" data-xref="التَّكوين 36:9–10; التَّكوين 36:4–5"> وبَنو عيسو: أليفازُ ورَعوئِيلُ ويَعوشُ ويعلامُ وقورَحُ.</span>
<span className="v" id="v36" data-v="36" data-xref="إرْمِيا 49:20; عامُوس 1:12; عُوبَدْيا 1:9"> وبَنو أليفازَ: تَيمانُ وأومارُ وصَفيٌ وجَعثامُ وقَنازُ وتِمناعُ وعَماليقُ.</span>
<span className="v" id="v37" data-v="37"> وبَنو رَعوئيلَ: نَحْتُ وزارَحُ وشَمَّةُ ومِزَّةُ.</span>
</div>

## سكان أدوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="التَّكوين 36:20–30"> وبَنو سعيرَ: لوطانُ وشوبالُ وصِبعونُ وعانَةُ وديشونُ وإيصَرُ وديشانُ.</span>
<span className="v" id="v39" data-v="39" data-xref="التَّثْنِيَة 2:12; التَّكوين 36:22; التَّثْنِيَة 2:22"> واَبنا لوطانَ: حُوري وهومامُ. وأختُ لوطانَ: تِمناعُ.</span>
<span className="v" id="v40" data-v="40" data-xref="التَّكوين 36:23–24"> وبَنو شُوبالَ: عَليانُ ومناحَةُ وعيبالُ وشفيٌ وأونامُ. واَبنا صِبعونَ: أيَّةُ وعانَةُ.</span>
<span className="v" id="v41" data-v="41" data-xref="التَّكوين 36:25–26"> واَبنُ عانَةَ: ديشونُ. وبَنو ديشونَ: حَمرانُ وأشبانُ ويثرانُ وكَرانُ.</span>
<span className="v" id="v42" data-v="42" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:21; التَّكوين 36:27–28"> وبَنو إيصَرَ: بِلْهانُ وزَعوانُ ويَعقانُ. واَبنا ديشانَ: عُوصٌ وأرانُ.</span>
</div>
</div>

## ملوك أدوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="التَّكوين 36:31–43; العَدَد 24:17–19; التَّكوين 49:10"> هؤلاءِ المُلوكُ مَلَكُوا في أرضِ أدومَ على التَّوالي، قَبلَ أنْ يَملِكَ مَلِكٌ في بَني إِسرائيلَ: بالِعُ بنُ بَعورَ مِنْ دِنهابة،</span>
<span className="v" id="v44" data-v="44" data-xref="إشَعْياء 34:6; إشَعْياء 63:1; عامُوس 1:12"> ويوبابُ بنُ زارَحَ مِنْ بَصرَةَ،</span>
<span className="v" id="v45" data-v="45" data-xref="التَّكوين 36:11; أيُّوب 2:11"> وماتَ يوبابُ، وملَكَ بَعدَهُ حوشامُ مِنْ أرضِ التِّيمانيِّينَ،</span>
<span className="v" id="v46" data-v="46"> وهَدَدُ بنُ بَدَدَ مِنْ عَويتَ وهوَ الّذي هزَمَ المِديانيِّينَ في بلادِ موآبَ،</span>
<span className="v" id="v47" data-v="47" data-xref="التَّكوين 36:36"> وماتَ هَدَدُ، وملَكَ بَعدَهُ سَملَةُ مِنْ مَسريقَةَ،</span>
<span className="v" id="v48" data-v="48" data-xref="التَّكوين 36:37"> ومات سَملَةُ، فَمَلَكَ بَعدَهُ شاوُلُ مِنْ رَحبوتَ على الفُراتِ،</span>
<span className="v" id="v49" data-v="49"> وماتَ شاوُلُ فمَلَكَ بَعدَهُ بَعلَ حَنانُ بنُ عكْبورَ،</span>
<span className="v" id="v50" data-v="50" data-xref="التَّكوين 36:39"> وماتَ بَعلَ حَنانُ، فمَلَكَ بَعدَهُ هَدَدُ مِنْ فاعي، واَسمُ اَمرأتِهِ مَهيطبيلُ بِنتُ مَطرِدَ بِنتِ ميزَهَبَ.</span>
</div>
<div className="bp">
<span className="v" id="v51" data-v="51" data-xref="التَّكوين 36:40"> وأُمراءُ أدومَ: تِمناعُ وعَلوَةُ ويَتيتُ</span>
<span className="v" id="v52" data-v="52"> وأهوليبامَةُ وإيلَةُ فينونَ</span>
<span className="v" id="v53" data-v="53"> وقَنازُ وتيمانُ ومِبصارُ</span>
<span className="v" id="v54" data-v="54" data-xref="التَّكوين 36:41–43"> ومَجديئيلُ وعيرامُ.</span>
</div>
