<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 46:21; أخبارِ الأيّامِ الأوّلُ 7:6–12; العَدَد 26:38"> وبَنو بنيامينَ خمسةٌ وهُم، على حسَبِ أعمارِهِم: بالَعُ بِكْرُهُ، وأشبيلُ وأخرَخُ</span>
<span className="v" id="v2" data-v="2"> ونُوحَةُ ورافا.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 46:21; العَدَد 26:40"> وبَنو بالَعَ: أدارُ وجَيرا وأبيهُودُ</span>
<span className="v" id="v4" data-v="4"> وأبيشوعُ ونُعمانُ وأخُوخُ</span>
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الأوّلُ 7:12; العَدَد 26:39; القُضاة 3:15"> وحَيرا وشَفوفانُ وحُورامُ.</span>
<span className="v" id="v6" data-v="6" data-xref="أخبارِ الأيّامِ الأوّلُ 2:54; القُضاة 3:20–30; أخبارِ الأيّامِ الأوّلُ 7:10"> وبَنو أحودُ الّذينَ كانوا رؤساءَ عائِلاتِهِم وسُكَّانُ جَبْعَ، ثُمَّ نُقِلوا إلى مَناحَةَ،</span>
<span className="v" id="v7" data-v="7"> هُم: نُعمانُ وأخيا، جَيرا أبو عُزَّا وأخيحودَ، هوَ الّذي نقَلَهُم إلى مَناحَةَ،</span>
<span className="v" id="v8" data-v="8" data-xref="راعُوث 1:1; التَّكوين 25:6"> وكانَ لِشحرايمَ بَنونَ في أرضِ مُوآبَ، بَعدَ أنْ طَلَّقَ حُوشيمَ وبَعْرا اَمرأتَيهِ،</span>
<span className="v" id="v9" data-v="9"> فوَلَدَ مِنْ حودَشَ اَمرأتِهِ يوبابَ وظِبْيا ومَيشا ومَلكامَ</span>
<span className="v" id="v10" data-v="10"> ويَعوصَ وشَبيا ومُرمَةَ، وهؤلاءِ كانوا رؤساءَ عائلاتِهِم.</span>
<span className="v" id="v11" data-v="11"> وولَدَ أيضًا مِنْ حُوشيمَ اَمرَأتِهِ أبيطوبَ وألفَعَلَ.</span>
<span className="v" id="v12" data-v="12" data-xref="عَزْرا 2:33; نَحَمْيا 7:37; نَحَمْيا 6:2"> وبَنو ألفَعَلَ: عابِرُ ومِشعامُ وشامِرُ وهوَ الّذي بَنى أونو ولُودَ وتوابِعَها.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الأوّلُ 2:52; يَشُوع 19:42; أخبارِ الأيّامِ الأوّلُ 4:4"> وكانَ بَريعةُ وشَمَعُ رئيسَينِ لِعائِلاتِهِما سكَنا في أيِّلونَ وطَردا سُكَّانَ جَتَّ.</span>
<span className="v" id="v14" data-v="14"> وبَنو بريعةَ: أخيو وشاشَقُ ويَريمُوتُ</span>

</div>
<div className="bp">
<span className="v" id="v17" data-v="17"> وبَنو ألفَعَلَ: زَبَدْيا ومَشُلَّامُ وحَزقي وحابِرُ</span>
<span className="v" id="v18" data-v="18"> ويشمرايُ ويَزَلياهُ ويوبابُ.</span>

<span className="v" id="v20" data-v="20"> وأليعينايُ وصِلَّتايُ وإيليئيلُ</span>

<span className="v" id="v23" data-v="23"> وعَبدونُ وزِكري وحانانُ</span>
<span className="v" id="v24" data-v="24"> وحَننيا وعيلامُ وعَنثوثيا</span>
<span className="v" id="v25" data-v="25"> ويَفَدْيا وفَنوئيلُ.</span>
<span className="v" id="v26" data-v="26"> وبَنو يروحامَ: شِمشرايُ وشَحَرْيا وعثَلْيا</span>
<span className="v" id="v27" data-v="27"> ويَعرَشْيا وإيليَّا وزِكري.</span>
<span className="v" id="v28" data-v="28" data-xref="القُضاة 1:21; يَشُوع 15:63; نَحَمْيا 11:1"> هؤلاءِ رؤساءُ عائلاتِهِم، بِحسَبِ مواليدِهِم، وكانَت مَساكِنُهُم في أورُشليمَ.</span>

<span className="v" id="v30" data-v="30" data-xref="أخبارِ الأيّامِ الأوّلُ 9:36–37"> واَبنُهُ البِكْرُ عَبدونُ، ثُمَّ صُورُ وقيشُ وبَعلُ ونادابُ</span>
<span className="v" id="v31" data-v="31" data-xref="أخبارِ الأيّامِ الأوّلُ 9:37"> وجَدورُ وأخيو وزاكرُ.</span>
<span className="v" id="v32" data-v="32" data-xref="أخبارِ الأيّامِ الأوّلُ 9:38"> ومقْلوثُ ولَدَ شَماةَ. وسكنَ أبناؤُهُم في أورُشليمَ معَ سائِرِ أبناءِ عَشيرَتِهِم.</span>

## ذرية الملك شاول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="صَمُوئيلَ الأوَّلُ 9:1; صموئيلَ الثّانِي 2:8; صَمُوئيلَ الأوَّلُ 31:2"> ونيرُ ولَدَ قَيشًا، وقَيشٌ ولَدَ شاولَ. وبَنو شاولَ: يوناثانُ ومَلكيشوعُ وأبينادابُ وإشبَعْلُ.</span>
<span className="v" id="v34" data-v="34" data-xref="صموئيلَ الثّانِي 4:4; صموئيلَ الثّانِي 9:12; صموئيلَ الثّانِي 9:10"> ويوناثانُ ولَدَ مَريبَعَلَ، ومَريبَعَلُ ولَدَ ميخا.</span>
<span className="v" id="v35" data-v="35" data-xref="أخبارِ الأيّامِ الأوّلُ 9:41"> وبَنو ميخا: فيثُونُ ومالِكُ وتاريعُ وآحازُ.</span>
<span className="v" id="v36" data-v="36" data-xref="أخبارِ الأيّامِ الأوّلُ 9:42"> وآحازُ ولَدَ يُوعَدَّةَ ويُوعَدَّةُ ولَدَ عَلمَثَ وعَزمُوتَ وزِمْري. وزِمْري ولَدَ مُوصا.</span>
<span className="v" id="v37" data-v="37" data-xref="أخبارِ الأيّامِ الأوّلُ 9:43"> ومُوصا ولَدَ بِنْعا، وبِنْعا ولَدَ رافَةَ، ورافَةُ ولَدَ ألعاسَةَ، وألعاسَةُ ولَدَ أصيلَ.</span>
<span className="v" id="v38" data-v="38"> ولأصيلَ سِتَّةُ بَنينَ وهذِهِ أسماؤُهُم: عَزْريقامُ وبُكْرُو وإسماعيلُ وشَعَرْيا وعوبَدْيا وحانانُ.</span>
<span className="v" id="v39" data-v="39"> وبَنو عاشِقَ أخيهِ: أُولامُ بِكْرُهُ، ويَعُوشُ وأليفَلَطُ.</span>
<span className="v" id="v40" data-v="40" data-xref="المَزامِير 128:3–6; أخبارِ الأيّامِ الثّانِي 14:8; أخبارِ الأيّامِ الأوّلُ 12:2"> وكانَ بَنو أولامَ رِجالًا جَبابِرةً ورُماةَ سِهامٍ، وكانَ لهُم بَنونَ وبَنو بَنينَ كثيرونَ بلَغوا مئَةً وخمسينَ. كُلُّ هؤلاءِ مَرَّ ذِكرُهُم مِنْ بَني بنيامينَ.</span>
