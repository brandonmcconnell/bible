## الانتصار على الملك عوج

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العَدَد 21:33–35; التَّثْنِيَة 1:4; التَّثْنِيَة 29:7"> ثُمَّ تَحوَّلْنا فصَعِدْنا في طريقِ باشانَ، فخرَجَ علَينا عُوجُ مَلِكُ باشانَ بِجميعِ قومِهِ وحارَبَنا في إذرعي.</span>
<span className="v" id="v2" data-v="2" data-xref="رُؤيا يُوحَنّا 2:10; أخبارِ الأيّامِ الثّانِي 20:17; العَدَد 21:23–25"> فقالَ ليَ الرّبُّ: ”لا تَخَفْهُ، فإنِّي سلَّمتُهُ إلى يَدِكَ، هوَ وجميعَ قومِهِ وأرضِهِ، لِتَفعَلَ بهِ كما فعَلْتَ بسيحون مَلِكِ الأموريِّينَ الّذي كانَ مُقيمًا بحَشبونَ“.</span>
<span className="v" id="v3" data-v="3" data-xref="العَدَد 21:35; يَشُوع 13:12; يَشُوع 13:30"> فسَلَّمَ الرّبُّ إلهُنا إلى أيدينا عُوجَ مَلِكِ باشانَ أيضًا وجميعَ قومِهِ، فضربْناهُ حتّى لم يبقَ لَه باقٍ.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الأوَّلُ 4:13; يَشُوع 12:4; يَشُوع 13:30–31"> وفَتحنا جميعَ مُدُنِهِ في ذلِكَ الوقتِ. لم تَبقَ مدينةٌ لم نأخُذْها مِنهُم، وعدَدُها ستُّونَ مدينةً في كُلِّ إقليمِ أرجوبَ الّذي هوَ مَملَكةُ عُوجَ في باشانَ</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 1:28; العَدَد 13:28; العِبرانِيّين 11:30"> وهيَ كُلُّها مُدُنٌ مُحصَّنةٌ بأسوارٍ شامخةٍ وأبوابٍ ومزاليجَ، ما عدا مُدُنَ الفرِّزيِّينَ الكثيرةَ جِدًّا.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 2:34; المَزامِير 135:10–12; التَّثْنِيَة 2:24"> فحَلَّلْنا في كُلِّ مدينةٍ قَتْلَ جميعِ الرِّجالِ والنِّساءِ والأطفالِ، كما فعَلْنا في مُدُنِ سيحونَ مَلِكِ حَشبونَ.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 2:35; يَشُوع 8:27; يَشُوع 11:11–14"> وأمَّا البَهائمُ والمُدُنُ فغَنِمْناها لأنفُسِنا</span>
<span className="v" id="v8" data-v="8" data-xref="يَشُوع 13:9–12; يَشُوع 12:2–6; العَدَد 32:33–42"> وأخذْنا في ذلِكَ الوقتِ مِنْ أيدي مَلِكَي الأموريِّينَ الأرضَ الّتي في عَبرِ الأردنِّ شرقًا، مِنْ وادي أرنونَ إلى جبَلِ حرمونَ.</span>
<span className="v" id="v9" data-v="9" data-xref="أخبارِ الأيّامِ الأوّلُ 5:23; المَزامِير 29:6; حِزْقِيال 27:5"> وحرمونُ يُسمِّيهِ الصَّيدونيُّونَ سريونَ، والأموريُّونَ يُسمُّونَه سنيرَ.</span>
<span className="v" id="v10" data-v="10" data-xref="يَشُوع 13:31; يَشُوع 13:11–12; التَّثْنِيَة 4:49"> أخذْنا جميعَ مُدُنِ السَّهلِ وكُلَّ جلعادَ وكُلَّ باشانَ إلى سَلخةَ وإذرعي، مدينتَي عُوجَ في باشانَ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 14:5; إرْمِيا 49:2; صموئيلَ الثّانِي 12:26"> وعُوجُ هذا هوَ آخِرُ مَنْ بقيَ مِنَ الرَّفائيِّين، وعرشُهُ الّذي منْ حديدٍ لا يزالُ في مدينةِ ربَّةَ الّتي لبَني عمُّونَ، وطُولُه تِسْعُ أذرعٍ وعَرضُهُ أربعُ أذرعٍ.</span>
</div>
</div>

## الأسباط في عبر الأردن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 2:36; العَدَد 32:32–38; يَشُوع 13:8–28"> وهذِهِ الأرضُ مَلَكْناها في ذلِكَ الوقتِ، مِنْ عروعيرَ الّتي على وادي أرنونَ، وأعطيتُ نِصفَ جبَلِ جلعادَ بِمُدنِهِ لبَني رَأوبينَ وبَني جادَ،</span>
<span className="v" id="v13" data-v="13" data-xref="العَدَد 32:39–42; يَشُوع 13:29–32; أخبارِ الأيّامِ الأوّلُ 5:23–26"> وأعطيتُ نِصفَ سِبْطِ منَسَّى باقي جلعادَ وكُلَّ إقليمِ أرجوبَ الّذي هوَ جميعُ باشانَ مَملكة عُوجَ. وكانت كُلُّ أرضِ باشانَ تُسَمَّى أرضَ الرَّفائيِّينَ.</span>
<span className="v" id="v14" data-v="14" data-xref="العَدَد 32:41; يَشُوع 13:13; صموئيلَ الثّانِي 3:3"> فلمَّا أخذَ يائيرُ بنُ منَسَّى جميعَ إقليمِ أرجوبَ إلى حُدودِ أرضِ الجَشوريِّينَ والمَعكيِّينَ، سَمَّى باشانَ باَسْمِهِ وهي تُعرَفُ بحؤوثِ يائيرَ إلى يومِنا هذا.</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 32:39–40; يَشُوع 17:1–3; يَشُوع 22:7"> وأعطيتُ جلعادَ لِماكيرَ.</span>
<span className="v" id="v16" data-v="16" data-xref="العَدَد 21:24; التَّثْنِيَة 2:37; صموئيلَ الثّانِي 24:5"> وأعطيتُ بَني رَأوبينَ وبَني جادَ مِنْ جلعادَ إلى وادي أرنونَ، وجعَلْتُ وسطَ الوادي حُدودًا لأرضِهِم جَنوبًا، ووادي يَبُّوقَ حُدودًا لها شَمالًا وهيَ حدودُ أرضِ بَني عمُّونَ.</span>
<span className="v" id="v17" data-v="17" data-xref="يَشُوع 12:3; التَّكوين 14:3; يَشُوع 3:16"> وأعطيتُهُمُ العرَبةَ والأردُنَّ الّذي يُتاخِمُ أرضَهُم مِنْ كنَّارَةَ إلى بحرِ العرَبةِ، البحرِ المَيتِ، تَحتَ سُفوحِ جبَلِ فسْجةَ نحوَ الشَّرقِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="العَدَد 32:20–24; يَشُوع 4:12–13; يَشُوع 1:12–15"> وقُلتُ لكُم في ذلِكَ الوقتِ: ”إنَّ الرّبَّ إلهَكُم أعطاكُم هذِهِ الأرضَ لِتَرثوها، فيَعبُرُ إليها جميعُ الأقوياءِ مِنكُم بسلاحِهِم قُدَّامَ إخوتِكُم بَني إِسرائيلَ،</span>
<span className="v" id="v19" data-v="19"> إلَّا نِساءَكُم وأطفالَكُم وماشيتَكُم، فأنا أعلمُ أنَّ لكُم ماشيةً كثيرةً، فيُقيمونَ في مُدُنِكُمُ الّتي أعطيتُكُم إيَّاها</span>
<span className="v" id="v20" data-v="20" data-xref="يَشُوع 22:4; يَشُوع 22:8"> إلى أنْ يُريحَ الرّبُّ إخوتَكُم مِثلَكُم ويرثوا هُم أيضًا الأرضَ الّتي أعطاهُمُ الرّبُّ إلهُكُم في عَبرِ الأردنِّ غربًا، ثُمَّ ترجِعونَ، كُلٌّ إلى ميراثِهِ الّذي أعطَيتُهُ لكُم“.</span>
<span className="v" id="v21" data-v="21" data-xref="العَدَد 27:18–23; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:10; المَزامِير 9:10"> وقلتُ ليَشوعَ في ذلِكَ الوقتِ: ”رأت عيناكَ جميعَ ما فعَلَ الرّبُّ إلهُكُم بِهذَينِ المَلِكينِ، سيحونَ وعُوجَ، وكذلِكَ سيفعَلُ الرّبُّ بِجميعِ المَمالِكِ الّتي أنتَ ستَعبُرُ إليها،</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 14:14; التَّثْنِيَة 20:4; التَّثْنِيَة 1:30"> فلا تَخَفهُم، لأنَّ الرّبَّ إلهَكُم هوَ المُحاربُ عَنكُم“.</span>
</div>
</div>

## موسى يحرم من دخول كنعان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:8–9"> وقلتُ للرّبِّ في ذلِكَ الوقتِ مُستَعطِفًا:</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 15:11; التَّثْنِيَة 11:2; المَزامِير 71:19"> ”أيُّها الرّبُّ الإلهُ، بدأتَ تُريني أنا عَبدَكَ، عظَمَتَكَ ويَدَكَ القديرةَ. لا إلهَ في السَّماءِ والأرضِ يصنَعُ مِثلَ أعمالِكَ وجَبروتِكَ.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 4:21–22; حِزْقِيال 20:6; الخُرُوج 3:8"> دعْني أعبُرُ فأرى الأرضَ الطيِّبةَ الّتي في عَبرِ الأردُنِّ غربًا وذلِكَ الجبَلَ الجميلَ ولبنانَ“.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 1:37; التَّثْنِيَة 31:2; أخبارِ الأيّامِ الأوّلُ 22:7–9"> ولكنَّ الرّبَّ كانَ غاضِبًا عليَّ بسببِكُم ولم يَسمعْ لي، بل قالَ لي: ”كفى، لا تَزِدْ في الكلامِ معي في هذا الأمرِ.</span>
<span className="v" id="v27" data-v="27" data-xref="العَدَد 27:12; التَّثْنِيَة 34:1–4; التَّكوين 13:14–15"> لَكنِ اَصعَدْ إلى رأسِ جبَلِ فسْجةَ واَرفعْ عينيكَ غربًا وشَمالًا وجَنوبًا وشَرقًا واَنظُرْ، لأنَّكَ لا تَعبُرُ هذا الأردُنَّ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 31:3; التَّثْنِيَة 31:7; التَّثْنِيَة 1:38"> فاَذهبْ إلى يَشوعَ وشَدِّدْ عزيمَتَهُ وشجِّعْهُ لأنَّه هوَ الّذي يعبُرُ أمامَ هؤلاءِ الشَّعبِ ويُملِّكُهُمُ الأرضَ الّتي تراها“.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّثْنِيَة 34:6; التَّثْنِيَة 4:46; العَدَد 25:3"> ثُمَّ أقمْنا في الوادي قُبالةَ بيتَ فَغورَ.</span>
</div>
