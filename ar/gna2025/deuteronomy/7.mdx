## شعب الله الخاص

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّثْنِيَة 31:3; اعمال الرسل 13:19; التَّثْنِيَة 4:38"> وإذا أدخلَكُمُ الرّبُّ إلهُكُم الأرضَ الّتي أنتُم مُزمِعونَ أنْ تمتلِكوها، وطَردَ أُمَمًا كثيرةً مِنْ أمامِكُم كالحِثِّيِّينَ والجَرجاشيِّين والأموريِّينَ والكنعانيِّينَ والفَرزِّيِّينَ والحَوِّيِّينَ واليَبوسيِّينَ، وهُم سَبعةُ شُعوبٍ أعظمُ وأكثرُ مِنكُم،</span>
<span className="v" id="v2" data-v="2" data-xref="القُضاة 2:2; الخُرُوج 23:32–33; التَّثْنِيَة 20:16–17"> وأسلَمَهُم إلى أيديكُم وضَربتموهُم، فاَجْعَلوهُم مُحَرَّمينَ علَيكُم. لا تقطَعوا مَعهُم عَهدًا، ولا تتَحنَّنُوا علَيهِم،</span>
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الأوَّلُ 11:2; يَشُوع 23:12–13; الخُرُوج 34:15–16"> ولا تُصاهِروهُم، فتُعطُوا بَناتِكُم لبَنيهِم وتأخذوا بَناتِهِم لبَنيكُم</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 6:15; القُضاة 2:11; التَّثْنِيَة 4:26"> لأنَّهُم يَردُّونَ بَنيكُم عَنِ اَتِّباعِ الرّبِّ، فيَعبدونَ آلهةً أخرى. فيَشتدُّ غضَبُ الرّبِّ علَيكُم ويُبيدُكُم سريعًا.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 23:24; التَّثْنِيَة 12:2–3; الخُرُوج 34:13"> بل هذا ما تفعلونَ بِهِم: تهدِمونَ مذابِحَهُم، وتُحَطِّمونَ أصنامَهُمُ المَنصوبةَ، وتقطعونَ أوتادَ آلهتِهِم، وتحرِقونَ تماثيلَهُم بالنَّار.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 14:2; الخُرُوج 19:5–6; رسالَةُ بُطرُسَ الأُولَى 2:9"> فأنتُم شعبٌ مُقدَّسٌ للرّبِّ إلهِكُمُ الّذي اَختارَكُم لَه مِنْ بَينِ جميعِ الشُّعوبِ الّتي على وجهِ الأرضِ</span>
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 10:22; ﺭﻭﻣﻴﺔ 9:11–15; ﺭﻭﻣﻴﺔ 11:6"> لا لأنَّكُم أكثرُ مِنْ جميعِ الشُّعوبِ فأنتُم أقلُّها</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 20:2; الخُرُوج 32:13; أفَسُس 2:4–5"> بل لِمَحبَّتِهِ ومُحافَظتِهِ على اليمينِ الّتي حلَفَها لآبائِكُم، فأخرجَكُم بيدٍ قديرةٍ وفداكُم مِنْ دارِ العُبوديَّةِ، مِنْ قبضةِ فِرعَونَ مَلِكِ مِصْرَ.</span>
<span className="v" id="v9" data-v="9" data-xref="العِبرانِيّين 10:23; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:13; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:3"> فاَعلمُوا أنَّ الرّبَّ إلهَكُم هوَ اللهُ الإلهُ الأمينُ، يحفظُ العَهدَ والرَّحمةَ لمُحبِّيهِ والعامِلينَ بوصاياهُ إلى ألفِ جِيلٍ.</span>
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 59:18; ناحُوم 1:2; ﺭﻭﻣﻴﺔ 12:19"> ويُجازي مُبغضيهِ، في الحالِ فيُبيدُهُم ولا يُمهِلهُم.</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 14:15; التَّثْنِيَة 4:1; التَّثْنِيَة 5:32"> فاَعمَلوا بالوصايا والسُّنَنِ والأحكامِ الّتي آمُرُكُم اليومَ أنْ تعملوا بِها.</span>
</div>
</div>

## جزاء الطاعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 28:1–14; التَّثْنِيَة 7:9; اللّاويِّين 26:3–13"> فإذا سَمِعْتَ هذِهِ الأحكامَ يا شعبَ إِسرائيلَ وحفِظتَها وعمِلتَ بها، فجَزاؤُكَ أنْ يحفَظَ الرّبُّ إلهُكَ عَهدَهُ لكَ ورحمتَهُ الّتي أَقسمَ علَيها لآبائِكَ،</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 14:21; يُوحَنّا 15:10; التَّثْنِيَة 28:3–5"> فيُحِبُّكَ ويُباركُكَ ويُكثِّرُكَ، ويُباركُ ثمرَةَ أحشائِكَ وثمَرةَ تُرْبتِكَ مِنْ قمحٍ وعصيرٍ وزيتٍ ونِتاجِ بقَرٍ وغنَمٍ على وجهِ الأرضِ الّتي أَقسمَ لآبائِكَ أنْ يُعطيَها لكَ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 23:26–33; التَّثْنِيَة 28:4; المَزامِير 127:3"> وتكونُ مُباركًا فَوقَ جميعِ الشُّعوبِ، ولا يكونُ في نسلِكَ ولا في نسلِ بَهائمِكَ عقيمٌ ولا عاقرٌ.</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 15:26; المَزامِير 105:36–37; اللّاويِّين 26:3–4"> ويَردُّ الرّبُّ عَنكَ جميعَ الأمراضِ وكُلَّ داءٍ خبيثٍ عَرَفتَهُ في مِصْرَ لا يُصيبُكَ بهِ، بل يُصيبُ مُبغضيكَ.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 23:33; القُضاة 8:27; التَّثْنِيَة 13:8"> وتقضي على جميعِ الشُّعوبِ الّذينَ يُسلِّمُهُم إليكَ الرّبُّ إلهُكَ. لا تُشفِقْ علَيهِم ولا تعبُدْ آلهتَهُم، ففي ذلِكَ شَرَكٌ لهلاكِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="العَدَد 13:32; العَدَد 33:53; التَّثْنِيَة 18:21"> فلا تَقُلْ في نفْسِكَ، يا شعبَ إِسرائيلَ، إنَّ هؤلاءِ الأمَمَ أكثرُ مِنِّي، فكيفَ أقدِرُ أنْ أطرُدَهُم.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 105:5; التَّثْنِيَة 31:6; التَّثْنِيَة 1:29"> لا تخَفْهُم بل تذَكَّرْ ما أنزلَ الرّبُّ إلهُكَ بفِرعَونَ وبسائرِ المِصْريِّينَ</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 4:34; نَحَمْيا 9:10–11; إرْمِيا 32:20–21"> مِنَ النَّكَباتِ العظيمةِ الّتي رَأتْها عيناكَ واَذكُرِ المُعجِزاتِ والعجائبَ واليدَ القديرةَ والذِّراعَ المَرفوعةَ الّتي بِها أخرجَكَ الرّبُّ إلهُكَ. هكذا يفعَلُ الرّبُّ إلهُكَ بجميعِ الشُّعوبِ الّذينَ أنتَ خائِفٌ مِنهُم.</span>
<span className="v" id="v20" data-v="20" data-xref="يَشُوع 24:12; الخُرُوج 23:28–30"> ويُرسِلُ علَيهِمِ الرّبُّ إلهُكَ الذُّعرَ حتّى يُبيدَ الباقينَ والمُختبئينَ مِنْ وجهِكَ.</span>
<span className="v" id="v21" data-v="21" data-xref="نَحَمْيا 1:5; التَّثْنِيَة 10:17; نَحَمْيا 9:32"> فلا ترهَبْهُم لأنَّ الرّبَّ إلهَكَ فيما بَينَكَ إلهٌ عظيمٌ رهيبٌ</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 23:29–30; يَشُوع 15:63"> يُبيدُ أولئكَ الأُمَمَ مِنْ أمامِ وجهِكَ قليلًا قليلًا. فأنتَ لا تقدرُ أنْ تُفنيَهُم سريعًا لئلَّا يَكثُرَ علَيكَ وحشُ البَرِّيَّةِ.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 8:20; التَّثْنِيَة 2:15; إرْمِيا 17:18"> الرّبُّ إلهُكَ يُسلِّمُهُم إليكَ ويُوقِعُ علَيهِمِ الحَيرةَ الشَّديدةَ حتّى يفنَوا.</span>
<span className="v" id="v24" data-v="24" data-xref="يَشُوع 10:8; يَشُوع 23:9; يَشُوع 1:5"> ويدفعُ مُلوكَهُم إلى يَدِكَ فتَمحو أسماءَهُم مِنْ تَحتِ السَّماءِ. لا يقِفُ أحدٌ في وجهِكَ حتّى تُفنيَهُم.</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 32:20; أخبارِ الأيّامِ الأوّلُ 14:12; التَّثْنِيَة 17:1"> وتماثيلُ آلهتِهِم تحرُقُها بالنَّار. لا تشتَهِ ما علَيها مِنَ الفضَّةِ والذَّهبِ، ولا تأخذْهُ لكَ لِئلَّا يكونَ شرَكًا لهَلاكِكَ، لأنَّهُ رِجْسٌ عِندَ الرّبِّ إلهِكَ.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 13:17; هُوشَع 14:8; حِزْقِيال 11:18"> فلا تُدخِلْ بَيتَكَ رِجْسًا لئلَّا تكونَ مُحَرَّمًا مِثلَهُ. تَجنَّبْهُ واَمْقُتْهُ لأنَّهُ مُحَرَّمٌ.</span>
</div>
