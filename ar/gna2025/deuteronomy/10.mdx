## لوحان آخران للوصايا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 34:1–2; الخُرُوج 25:10–15; التَّثْنِيَة 10:3–4"> في ذلِكَ الوقتِ قالَ ليَ الرّبُّ: ”إنحَتْ لوحَينِ مِنْ حجَرٍ كاللَّوحَينِ الأوَّلَينِ واَصعَدْ إليَّ إلى الجبَلِ واَصنَعْ تابوتًا مِنْ خشَبٍ تضعُهُما فيهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الأوَّلُ 8:9; التَّثْنِيَة 10:5; التَّثْنِيَة 4:13"> فأكتبُ على اللَّوحَينِ الكلماتِ الّتي كتَبتُها على اللَّوحَينِ الأوَّلَينِ اللَّذَينِ كسَّرتَهُما“.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 34:4; الخُرُوج 37:1–9; الخُرُوج 25:10"> فصنَعْتُ تابوتًا مِنْ خشَبِ السِّنطِ، ونَحَتُّ لوحَينِ مِنْ حجَرٍ كالأوَّلَينِ، وصَعِدتُ الجبَلَ واللَّوحانِ في يَديَّ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 9:10; الخُرُوج 34:28; العِبرانِيّين 12:18–19"> فكتبَ الرّبُّ علَيهما كالكِتابةِ الأولى، أي العَشْرَ الوصايا الّتي كلَّمَكُمُ الرّبُّ بِها في الجبَلِ مِنْ وسَطِ النَّارِ في يومِ اجتماعِكُم هُناكَ، وسلَّمَهُما إليَّ.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 34:29; الخُرُوج 40:20; التَّثْنِيَة 10:2"> ثُمَّ رَجَعتُ ونزَلْتُ مِنَ الجبَلِ ووضَعتُ اللَّوحَينِ في التَّابوتِ الّذي صَنَعتُهُ، فكانا هُناكَ كما أمرَني الرّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="العَدَد 33:38; العَدَد 20:23–28; العَدَد 33:30–33"> ورحَلَ بَنو إِسرائيلَ مِنْ آبارِ بَني يعقانَ إلى مُوسيرَ. هُناكَ ماتَ هرونُ ودُفِنَ، فتولَّى خِدمةَ الكهَنوتِ مكانَهُ إلِعازارُ اَبنُهُ.</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 33:32–34"> ورحَلوا مِنْ هُناكَ إلى جَدجودَ، ومِنها إلى يُطْباتَ الّتي تَجري فيها أنهارُ ماءٍ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 21:5; التَّثْنِيَة 18:5; العَدَد 6:23–26"> في ذلِكَ الوقتِ خَصَّ الرّبُّ سِبْطَ لاوي بِحَملِ تابوتِ عَهدِ الرّبِّ والوقوفِ أمامَهُ ليخدِموه ويباركوا باَسمِهِ كما هيَ حالُهُم إلى هذا اليومِ.</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 44:28; العَدَد 18:20–24; التَّثْنِيَة 18:1–2"> لذلِكَ لم يكنْ للَّاويِّينَ نصيبٌ ومُلْكٌ معَ إخوتِهِم بَني قومِهِم، وإنَّما الرّبُّ مَصدَرُ كُلِّ مُلْكٍ لهُم، كما كلَّمَهُمُ الرّبُّ إلهُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 9:18–19; التَّثْنِيَة 9:25; الخُرُوج 34:28"> وقالَ موسى: ”وأنا أقمتُ بالجبَلِ، مِثلَ الأيّامِ الأُولى، أربعينَ نهارًا وأربعينَ ليلةً، فسمِعَ ليَ الرّبُّ هذِهِ المرَّةَ أيضًا ولم يُحِبَّ أنْ يُهلِكَهُم“.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 33:1; الخُرُوج 32:34"> ثُمَّ قالَ ليَ الرّبُّ: ”تَقومُ وتسيرُ مُرتَحِلًا أمامَ الشَّعبِ ليدخلوا ويرِثوا الأرضَ الّتي أقسمْتُ لآبائِهِم أنْ أُعطيَها لهُم“.</span>
</div>
</div>

## ما يطلبه الرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ميخا 6:8; التَّثْنِيَة 6:5; لُوقا 10:27"> والآنَ يا شعبَ إِسرائيلَ، ما الّذي يطلبُهُ مِنكَ الرّبُّ إلهُكَ إلَّا أنْ تخافَهُ، وتسلُكَ في كُلِّ طُرُقِهِ، وتُحِبَّهُ وتعبُدَهُ بِكُلِّ قلبِكَ وكُلِّ نفْسِكَ</span>
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 32:39; التَّثْنِيَة 6:24; يَعقُوب 1:25"> وتعمَلَ بوصاياهُ وسُنَنِهِ الّتي أنا آمُرُكَ بها اليومَ لِخَيرِكَ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 19:5; المَزامِير 24:1; المَزامِير 115:16"> للرّبِّ إلهِكَ السَّماواتُ وسماواتُ السَّماواتِ والأرضُ وكُلُّ ما فيها.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّثْنِيَة 4:37; التَّثْنِيَة 7:7–8; ﺭﻭﻣﻴﺔ 9:13–23"> لكنَّهُ تعلَّقَ قلبُهُ بآبائِكَ فأحبَّهُم واَختارَ ذُرِّيَّتهُم مِنْ بَعدِهِم، وأنتُم هيَ هذِهِ الذُّرِّيَّةُ الّتي اختارها مِنْ بَينِ الشُّعوبِ كما تَرونَ اليومَ.</span>
<span className="v" id="v16" data-v="16" data-xref="إرْمِيا 4:4; التَّثْنِيَة 30:6; اللّاويِّين 26:41"> فكرِّسوا قلوبَكُم للرّبِّ، يا بَني إِسرائيلَ، ولا تعانِدوا بَعدَ الآنَ</span>
<span className="v" id="v17" data-v="17" data-xref="اعمال الرسل 10:34; ﺭﻭﻣﻴﺔ 2:11; دانيال 2:47"> لأنَّ الرّبَّ إلهَكُم هوَ إلهُ الآلهةِ وربُّ الأربابِ، الإلهُ العظيمُ الجبَّارُ الرَّهيبُ الّذي لا يُحابي ولا يَرتشي.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 146:9; المَزامِير 68:5; إرْمِيا 49:11"> يحكمُ لليتيمِ والأرملةِ، ويُحبُّ الغريبَ ويُرزِقُهُ طَعامًا وكِسوةً.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 22:21; اللّاويِّين 19:33–34; لُوقا 17:18"> فأحِبُّوا الغريبَ لأنَّكم كُنتُم غُرباءَ في أرضِ مِصْرَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 6:13; التَّثْنِيَة 13:4; لُوقا 4:8"> إتَّقِ الرّبَّ إلهَكَ، يا شعبَ إِسرائيلَ، ولا تعبُدْ سِواهُ، وتمَسَّكْ بهِ ولا تحلِفْ إلَّا باَسمِهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="إرْمِيا 17:14; المَزامِير 106:21–22; الخُرُوج 15:2"> هوَ فخرُكَ، وهوَ إلهُكَ الّذي صَنعَ مَعكَ تِلكَ العَظائمَ والأهوالَ الّتي رَأتها عيناكَ.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 46:27; الخُرُوج 1:5; التَّكوين 15:5"> كانَ آباؤُكَ سَبعينَ نفْسًا حينَ نزَلوا إلى مِصْرَ، والآنَ صَيَّركَ الرّبُّ إلهُكَ في الكَثرةِ مِثلَ نُجومِ السَّماءِ.</span>
</div>
</div>
