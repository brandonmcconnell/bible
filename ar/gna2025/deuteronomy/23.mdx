## المرفضون من جماعة الربّ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="غَلاطِيَّة 3:28; إشَعْياء 56:3–4; اللّاويِّين 21:17–21"> لا يتَزوَّجْ رَجُلٌ زَوجةَ أبيهِ، فينتَهِكَ حُرمةَ أبيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="زَكَريّا 9:6; إشَعْياء 57:3; العِبرانِيّين 12:8"> لا يدخُلْ مَرضوضُ الخِصيتَينِ ولا مقطوعُ العُضْوِ التناسُليِّ جماعةَ المُؤمنينَ بالرّبِّ.</span>
<span className="v" id="v3" data-v="3" data-xref="نَحَمْيا 13:1–2; نَحَمْيا 13:23; راعُوث 4:6"> ولا يدخُلِ اَبْنُ زِنًى، ولا أحدٌ مِنْ نَسلِهِ، في جماعةِ المُؤمنينَ بالرّبِّ، ولو في الجيلِ العاشرِ.</span>
<span className="v" id="v4" data-v="4" data-xref="نَحَمْيا 13:2; رسالَةُ بُطرُسَ الثّانيَة 2:15; المُلُوكِ الأوَّلُ 18:4"> ولا يدخُلْ عَمُّونيٌّ ولا مُوآبيٌّ ولا أحدٌ مِنْ نسلِهِ في جماعةِ المُؤمنينَ بالرّبِّ، ولو في الجيلِ العاشِرِ وإلى الأبدِ</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 7:7–8; ﺭﻭﻣﻴﺔ 8:31; أفَسُس 2:4–5"> لأنَّهُم لم يستَقبِلوكُمْ بِالخُبزِ والماءِ في الطَّريقِ عِندَ خُروجِكُم مِنْ مِصْرَ، ولأنَّهُمُ اَستَأجَروا علَيكُم بلْعامَ بنَ بَعورَ مِنْ مدينةِ فَتورَ في آرامَ الّتي بَينَ النَّهرينِ لِيَلعَنَكُم.</span>
<span className="v" id="v6" data-v="6" data-xref="عَزْرا 9:12; صموئيلَ الثّانِي 12:31; نَحَمْيا 13:23–25"> فأبى الرّبُّ إلهُكُم أنْ يسمَعَ لِبَلْعام، فحَوَّلَ لكُمُ الرّبُّ إلهُكُمُ اللَّعنةَ بَركَةً لأنَّهُ أحبَّكُم.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 10:19; الخُرُوج 22:21; الخُرُوج 23:9"> لا تُسالِموهُم ولا تطلُبوا الخَيرَ لهُم طُولَ أيّامِكُم أبدًا.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 20:5–6; أفَسُس 2:12–13; التَّثْنِيَة 23:1–3"> لا تكرَهوا الأدوميّينَ لأنَّهُم إخوتُكُم مِنْ بني قومِكُم، ولا تكرَهوا المِصْريِّينَ لأنَّكُم كُنتُم نُزَلاءَ في أرضِهِم.</span>
<span className="v" id="v9" data-v="9" data-xref="يَشُوع 6:18; القُضاة 20:26; رُؤيا يُوحَنّا 19:11–14"> ومِنَ الجيلِ الثَّالثِ فصاعِدًا يَحُقُّ لِلبَنينَ الّذينَ يُولَدونَ لهُم ولِلأدومِيِّينَ أنْ يَدخُلوا في جماعةِ المُؤمنينَ بالرّبِّ.</span>
</div>
</div>

## نظافة المحلة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="اللّاويِّين 15:16; العَدَد 5:2–3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:11–13"> إذا خرَجْتُم لِمُحارَبةِ أعدائِكُم فتَجنَّبوا كُلَّ نجاسةٍ بِحسَبِ الشَّريعةِ.</span>
<span className="v" id="v11" data-v="11" data-xref="العِبرانِيّين 10:22; أفَسُس 5:26–27; اللّاويِّين 14:9"> إذا كانَ فيكُم رَجُلٌ غيرُ طاهرٍ مِنْ مَنامٍ لَه في اللَّيلِ، فليَخرُجْ إلى خارجِ المَحلَّةِ ولا يدخُلْ داخِلَها.</span>
<span className="v" id="v12" data-v="12"> وقبلَ غُروبِ الشَّمسِ يغتَسِلُ بِالماءِ، وعِندَ غُروبِها يدخُلُ داخلَ المَحلَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 24:6–8"> خَصِّصوا أمكِنةً خارجَ المَحلَّةِ تخرُجونَ إليها لِقَضاءِ حاجَتِكُم.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 17:1; اللّاويِّين 26:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:16"> وخُذوا معَكُم وَتَدًا معَ عِدَّتِكُم لِتَحفِروا بهِ حُفرةً عِندَما تجلِسونَ خارجًا، ثُمَّ تعودونَ وتطمُرونَها بَعدَ قضاءِ حاجَتِكُم.</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 30:15; عُوبَدْيا 1:14"> لأنَّ الرّبَّ إلهَكُم سائرٌ في وسَطِ مَحلَّتِكُم لِيُخَلِّصَكُم ويُسلِّمَ أعداءَكُم إلى أيديكُم، فلتكُنْ مَحلَّتُكُم غيرَ نَجِسةٍ لِئلَّا يَرى فيكُم عَيبًا فينصَرِفَ عَنكُم.</span>
</div>
</div>

## أحكام متفرقة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 22:21; زَكَريّا 7:10; إرْمِيا 7:6"> لا تُسلِّموا عبدًا لجأَ إليكُم مِنْ مَولاهُ،</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الثّانِي 23:7; المُلُوكِ الأوَّلُ 14:24; اللّاويِّين 19:29"> بل دَعوهُ يُقيمُ عِندَكُم في الموضِعِ الّذي يختارُهُ ويَطيبُ لَه في مُدُنِكُم. لا تُرهِقوهُ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="فِيلِبِّي 3:2; ﻣﺘﻰ 7:6; رُؤيا يُوحَنّا 22:15"> لا يكُنْ مِنْ بَناتِ بَني إِسرائيلَ ولا مِنْ بَنيهِ بَغيٌّ أو مأبونٌ في هياكلِ العِبادةِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 22:25; اللّاويِّين 25:35–37; المَزامِير 15:5"> لا تُدخِلوا إلى بَيتِ الرّبِّ إلهِكُم أُجرةَ بَغيٍّ ولا مأبونٍ وفاءً لِنَذْرٍ ما، لأنَّهما كِلَيهِما رِجسٌ لدى الرّبِّ إلهِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 15:10; التَّثْنِيَة 15:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:58"> لا تُقرِضوا إخوتَكُم مِنْ بَني قومِكُم بِربًى يدفَعونَهُ إليكُم فِضَّةً أو طَعامًا أو أيَّ شيءٍ آخَرَ مِمَّا يُقرَضُ بِالرِّبى،</span>
<span className="v" id="v21" data-v="21" data-xref="الجامِعَة 5:4–5; المَزامِير 66:13–14; المَزامِير 116:18"> بل أقرِضوا الغريبَ بِالرِّبى ولا تُقرِضوا إخوتَكُم مِنْ بَني قومِكُم، فيُبارِكَ الرّبُّ إلهُكُم جميعَ أعمالِ أيديكُم في الأرضِ الّتي أنتُم داخِلونَ لِتمتَلِكوها.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22"> إذا نَذَرتُم نَذْرًا للرّبِّ إلهِكُم فلا تُؤخِّروا وفاءَهُ، لأنَّ الرّبَّ إلهَكُم يُطالِبُكُم بهِ، فتكونُ علَيكُم خطيئةٌ.</span>
<span className="v" id="v23" data-v="23" data-xref="الأمثال 20:25; القُضاة 11:30–31; صَمُوئيلَ الأوَّلُ 1:11"> لكنِ اَقتَنَعتُم أنْ تَنذُروا نَذْرًا للرّبِّ فلا خطيئةَ علَيكُم.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:26; ﺭﻭﻣﻴﺔ 12:13; العِبرانِيّين 13:5"> ما وَعَدتُم بهِ فاَحفَظوهُ، واَعمَلوا كما نَذَرتُم للرّبِّ إلهِكُم تَطَوُّعًا.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 12:1–2; مَرْقُس 2:23; لُوقا 6:1–2"> إذا دَخَلتُم كَرْمَ أحدٍ، فكُلوا مِنَ العنبِ على قَدْرِ ما تشتَهونَ حتّى تشبَعوا، ولا تجعَلوا مِنهُ شيئًا في وِعائِكُم.</span>
<span className="v" id="v26" data-v="26"> وإذا دَخلتُم زَرعَ أحدٍ، فاَقطُفوا مِنهُ بأيديكُم فَرْكًا، ولا تُلقوا على السُّنبُلِ مَناجِلَكُم.</span>
</div>
