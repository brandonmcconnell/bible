## يشوع يخلف موسى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وتابعَ موسى كلامَهُ إلى جميعِ بَني إِسرائيلَ</span>
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 34:7; العَدَد 27:17; المُلُوكِ الأوَّلُ 3:7"> فقالَ لهُم: ”أنا اليومَ اَبنُ مئَةٍ وعِشرينَ سنَةً، لا أقدِرُ بَعدُ على الرَّواحِ والمَجيءِ. والرّبُّ قالَ لي: لن تعبُرَ هذا الأُردُنَّ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 9:3; التَّثْنِيَة 31:23; التَّثْنِيَة 3:28"> فالرّبُّ إلهُكُم يعبُرُ أمامَكُم، وهوَ يُزيلُ تِلكَ الأُمَمَ مِنْ أمامِكُم فتَرِثونَها. ويَشوعُ هوَ يَقودُكُم في عُبورِكُم كما قالَ الرّبُّ.</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 23:28–31; التَّثْنِيَة 2:33; التَّثْنِيَة 3:3–11"> ويفعَلُ الرّبُّ بِتلكَ الأُمَمِ كما فعَلَ بسيحونَ وعوجَ، مَلِكَيِ الأموريِّينَ، فدَمَّرَ أرضَهُما.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 7:2; التَّثْنِيَة 20:16–17; الخُرُوج 23:32–33"> فمتى أسلَمَهُمُ الرّبُّ إلى أيديكُم تفعَلونَ بِهم حسَبَ جميعِ الوصايا الّتي أمرتُكُم بِها.</span>
<span className="v" id="v6" data-v="6" data-xref="يَشُوع 1:9; أفَسُس 6:10; المَزامِير 27:14"> تشَجَّعوا وتقَوَّوا ولا تخافوا مِنهُم ولا ترهَبوهُم لأنَّ الرّبَّ إلهَكُم سائِرٌ معَكُم، لا يُهمِلُكُم ولا يَترُكُكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 3:28; التَّثْنِيَة 31:6; التَّثْنِيَة 1:38"> ثُمَّ دعا موسى يَشوعَ وقالَ لَه أمامَ عيونِ جميعِ بَني إِسرائيلَ: ”تشَجَّعْ وكُنْ قويًّا، فأنتَ تدخُلُ معَ هؤلاءِ الشَّعبِ إلى الأرضِ الّتي أقسَمَ الرّبُّ لآبائِكُم أنْ يُعطيَها لهُم، وأنتَ تُمَلِّكُهُم إيَّاها.</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 33:14; التَّثْنِيَة 31:6; يَشُوع 1:9"> والرّبُّ سائرٌ أمامَكَ، وهوَ يكونُ معَكَ، ولا يُهمِلُكَ ولا يَترُكُكَ، فلا تَخَفْ ولا ترهَبْ“.</span>
</div>
</div>

## حتى تقرأ الشريعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="العَدَد 4:15; يَشُوع 3:3; التَّثْنِيَة 17:18"> وكتَبَ موسى هذِهِ الشَّريعةَ وأعطاها لِكهَنةِ بَني لاوي، حامِلي تابوتِ عَهدِ الرّبِّ، وسائرِ شُيوخِ بَني إِسرائيلَ،</span>
<span className="v" id="v10" data-v="10" data-xref="اللّاويِّين 23:34–43; التَّثْنِيَة 15:1–2"> وقالَ لهُم: ”في نِهايةِ السَّبعِ السِّنينَ، في ميعادِ سنَةِ الإِعفاءِ مِنَ الدُّيونِ، في عيدِ المَظالِّ</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 23:2; يَشُوع 8:34–35; التَّثْنِيَة 16:16–17"> حينَما يأتي جميعُ بَني إِسرائيلَ ليَرَوا وجهَ الرّبِّ إلهِهِم في الموضِعِ الّذي يختارُهُ، تُقرَأُ هذِهِ الشَّريعةُ على مَسامِعِهِم جميعًا.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 4:10; المَزامِير 34:11–14; التَّثْنِيَة 6:6–7"> إجمَعوا الشَّعبَ، رِجالًا ونِساءً وأطفالًا، والغريبَ الّذي في مُدُنِكُم، لِيَسمَعوا ويتَعَلَّموا ويَتَّقوا الرّبَّ إلهَكُم، ويَحرِصوا على العمَلِ بجميعِ كلامِ تِلكَ الشَّريعةِ</span>
<span className="v" id="v13" data-v="13" data-xref="التَّثْنِيَة 11:2; التَّثْنِيَة 6:7; الأمثال 22:6"> وليَسمَعْ بَنوهُمُ الّذينَ لم يعلَموا ويتَعلَّموا مخافَةَ الرّبِّ إلهِكُم كُلَّ الأيّامِ الّتي تحيَونَها في الأرضِ الّتي أنتُم تعبُرونَ الأُردُنَّ إليها لتَمتَلِكوها“.</span>
</div>
</div>

## وصايا الرّبّ الأخيرة لموسى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="العَدَد 27:13; التَّثْنِيَة 31:23; أيُّوب 2:1"> وقالَ الرّبُّ لِموسى: ”قَرُبَ يومُ وَفاتِكَ، فاَدْعُ يَشوعَ وقِفْ معَهُ في خَيمَةِ الاجتماعِ لأتلوَ علَيهِ وصيَّتي“. فذَهبَ موسى ويَشوعُ ووَقَفا في خَيمَةِ الاجتماع.</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 40:38; المَزامِير 99:7; الخُرُوج 33:9–10"> فتَراءَى الرّبُّ في الخَيمَةِ، في عمودِ سحابٍ ووقَفَ على بابِ الخيمَةِ.</span>
<span className="v" id="v16" data-v="16" data-xref="القُضاة 10:6; القُضاة 10:13; القُضاة 2:12"> وقالَ الرّبُّ لِموسى: ”ستَرقُدُ معَ آبائِكَ، وهؤلاءِ الشَّعبُ سيَقومونَ عليَّ ويَفجُرونَ باَتِّباعِ آلِهةٍ غريبةٍ في الأرضِ الّتي هُم داخِلونَ إليها، ويترُكونَني ويَنقُضونَ عَهدي الّذي قطَعتُهُ معهُم.</span>
<span className="v" id="v17" data-v="17" data-xref="العَدَد 14:42; إشَعْياء 8:17; المَزامِير 104:29"> فيَشتَدُّ غضَبي علَيهِم في ذلِكَ الوقتِ، وأترُكُهُم وأحجُبُ وجهي عَنهُم، فيَصيرونَ مأكَلًا وتُصيبُهُم شُرورٌ وأضرارٌ كثيرةٌ، فيَقولونَ في ذلِكَ اليومِ: أصابَتنا هذِهِ الشُّرورُ لأنَّ إلهَنا لم يَعُدْ فيما بَينَنا.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّثْنِيَة 31:16–17"> وأنا أحجُبُ وجهي في ذلِكَ اليومِ بِسبَبِ جميعِ السَّيِّئاتِ الّتي فعَلوها حينَ مالوا إلى آلِهةٍ أُخرى.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 6:7; التَّثْنِيَة 31:21–22; التَّثْنِيَة 31:26"> فالآنَ اَكتُبْ هذا النَّشيدَ ولَقِّنْهُ بَني إِسرائيلَ لِيكون في ذاكِرَتِهِم شَهادةً لي علَيهِم</span>
<span className="v" id="v20" data-v="20" data-xref="هُوشَع 13:6; التَّثْنِيَة 6:10–12; نَحَمْيا 9:25–26"> حينَ أُدخِلُهُمُ الأرضَ الّتي أقسَمتُ لآبائِهِم علَيها، وهيَ أرضٌ تَدُرُّ لبَنًا وعسَلًا، فيأكُلونَ ويشبَعونَ ويسمَنونَ ويميلونَ إلى آلِهةٍ أُخرى ويعبُدونُها ويستَهينونَ بي وينقُضونَ عَهدي.</span>
<span className="v" id="v21" data-v="21" data-xref="هُوشَع 5:3; المَزامِير 139:2; يُوحَنّا 2:24–25"> فإذا أصابَتهُم شُرورٌ وأضرارٌ كثيرةٌ، يُنشِدونَ هذا النَّشيدَ أمامي شاهدًا علَيهِم، لأنَّهُ لن يُنسَى مِنْ ذاكِرةِ ذُرِّيَّتِهِم. فأنا أعلَمُ ما يَجولُ في خَواطِرِهِمِ اليومَ، مِنْ قَبلِ أنْ أُدخِلَهُمُ الأرضَ كما أقسَمتُ“.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 31:19; التَّثْنِيَة 31:9"> فكتبَ موسى النَّشيدَ في ذلِكَ اليومِ ولَقَّنَهُ بَني إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 31:7–8; التَّثْنِيَة 31:14; اعمال الرسل 7:45"> ثُمَّ قالَ موسى لِيشوعَ بنِ نُونٍ: ”تشَجَّعْ وكُنْ قَويًّا، فأنتَ تُدخِلُ بَني إِسرائيلَ الأرضَ الّتي أقسَمتُ لهُم علَيها، وأنا أكونُ معَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="التَّثْنِيَة 31:9; التَّثْنِيَة 17:18"> ولمَّا فرَغَ موسى مِنْ تدوينِ كلامِ هذِهِ الشَّريعةِ كُلِّها في سِفرٍ،</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 31:9"> قالَ لِلَّاويِّينَ، حامِلي تابوتِ عَهدِ الرّبِّ:</span>
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 31:19; المُلُوكِ الثّانِي 22:8–11; غَلاطِيَّة 2:19"> خُذوا سِفرَ الشَّريعةِ هذا وضَعوهُ إلى جانِبِ تابوتِ عَهدِ الرّبِّ إلهِكُم، فيكونَ هُناكَ علَيكُم شاهدًا.</span>
<span className="v" id="v27" data-v="27" data-xref="التَّثْنِيَة 32:20; التَّثْنِيَة 9:24; اعمال الرسل 7:51"> فأنا أعلَمُ تمَرُّدَكُم وعِنادَكُم لأنِّي، وأنا معكُم في الحياةِ، تمَرَّدتُم على الرّبِّ، فكيفَ بَعدَ موتي؟</span>
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 32:1; التَّثْنِيَة 4:26; التَّثْنِيَة 30:19"> إجمَعوا إليَّ شُيوخَ أسباطِكُم وقادَتَكُم حتّى أتلوَ على مَسامِعِهِم هذا الكلامَ، وأُشهِدَ علَيهِمِ السَّماءَ والأرضَ.</span>
<span className="v" id="v29" data-v="29" data-xref="القُضاة 2:19; التَّثْنِيَة 32:5; لُوقا 21:24"> فأنا أعلَمُ أنَّكُم بعدَ موتي ستَفسِدونَ وتحيدونَ عَنِ الطَّريقِ الّتي أوصَيتُكُم بِها، فيَنزِلُ بِكُمُ الشَّرُّ في آخِرِ الأيّامِ إذا فعَلتُم ما هوَ شَرٌّ في نظَرِ الرّبِّ حتّى تُكدِّروهُ بأعمالِ أيديكُم“.</span>
</div>
</div>
