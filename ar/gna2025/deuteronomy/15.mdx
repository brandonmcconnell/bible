## السنة السّابعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّثْنِيَة 31:10; الخُرُوج 23:10–11; اللّاويِّين 25:2–4"> في كُلِّ سبعِ سِنينَ تُجْرونَ إعفاءً مِنَ الدُّيونِ.</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 7:42; يَعقُوب 2:13; لُوقا 6:34–38"> وهذِهِ طريقةُ الإعفاءِ: كُلُّ صاحبِ دَينٍ مِنكُم يُعفي أخاهُ مِنْ بَني قومِهِ مِمَّا أقرضَهُ. لا يُطالِبُهُ لأنَّ الرّبَّ قالَ بإعفائِهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 23:20; ﻣﺘﻰ 17:25–26; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:6–7"> أمَّا الغريبُ فتُطالِبُهُ، وأمَّا ما يكونُ لكَ على أحدٍ مِنْ إخوَتِكَ بَني قومِكَ فَتُعفيهِ مِنهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="إشَعْياء 58:10–11; الأمثال 28:27; الأمثال 11:24–25"> لأنَّ الرّبَّ إلهَكُم يُبارِكُكُم في الأرضِ الّتي يُعطيكُم إيَّاها نَصيبًا تملِكونَهُ حتّى لا يكونَ فيما بَينَكُم مُحتاجٌ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 28:1–15; إشَعْياء 1:19–20; يَشُوع 1:7"> إنْ سَمِعتُم لِصوتِ الرّبِّ إلهِكُم وحَفِظتُم جميعَ هذِهِ الوصايا الّتي أنا آمُرُكُم بِها اليومَ وعَمِلتُم بِها</span>
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 28:12–13; لُوقا 6:35; نَحَمْيا 9:27"> يُبارِكُكُمُ الرّبُّ إلهُكُم كما وعَدَكُم، فيقتَرِضُ مِنكُم أُمَمٌ كثيرونَ وأنتُم لا تقتَرِضونَ، وتتَسَلَّطونَ على أُمَمٍ كثيرينَ وهُم علَيكُم لا يتَسَلَّطونَ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:16–17; التَّثْنِيَة 15:9; الأمثال 21:13"> إذا كانَ بَينَكُم مُحتاجونَ مِنْ إخوَتِكُم بَني قومِكُم في إحدى مُدُنِكُم، في الأرضِ الّتي يُعطيكُمُ الرّبُّ إلهُكُم، فلا تُقَسُّوا قُلوبَكُم ولا تَبخَلوا على إخوَتِكُم المُحتاجينَ فيما بَينكُم،</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 6:34; ﻣﺘﻰ 5:42"> بل اَفْتَحوا لهُم أيديَكُم وأقرِضوهُم مِقْدارَ ما يُعوِزُهُم،</span>
<span className="v" id="v9" data-v="9" data-xref="التَّثْنِيَة 24:15; ﻣﺘﻰ 20:15; إرْمِيا 17:10"> وتجَنَّبوا سُوءَ النِّيَّةِ في قلوبِكُم فتَقولونَ قَرُبَتِ السَّنةُ السَّابِعةُ، سنَةُ الإعفاءِ، فتَصرِفونَ نظَرَكُم عَنْ إخوتِكُمُ المُحتاجينَ ولا تُعطونَهُم، فيَصرُخونَ إلى الرّبِّ علَيكُم، فيَحسِب ذلِكَ علَيكُم خطيئةً.</span>
<span className="v" id="v10" data-v="10" data-xref="الأمثال 22:9; العِبرانِيّين 13:16; إشَعْياء 58:10"> بلِ اَعطوهُم ولا تتوَرَّعْ قلوبُكُم إذا أعطَيتُموهُم، وبِذلِكَ يُبارِكُكُمُ الرّبُّ إلهُكُم في كُلِّ أعمالِكُم وفي جميعِ ما في مُتَناوَلِ أيديكُم.</span>
<span className="v" id="v11" data-v="11" data-xref="مَرْقُس 14:7; ﻣﺘﻰ 26:11; يُوحَنّا 12:8"> فالأرضُ لا تخلو مِنْ مُحتاجٍ، ولِذلِكَ آمُرُكُمُ اليومَ أنْ تفتَحوا أيديَكُم لإخوتِكُمُ المَساكينِ المُحتاجينَ الّذينَ في أرضِكُم.</span>
</div>
</div>

## معاملة العبيد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 34:14; اللّاويِّين 25:39–41; الخُرُوج 21:2–6"> إذا باعَكَ عِبرانيٌّ أو عِبرانيَّةٌ نَفسَهُ، فيَخدُمُكَ سِتَّ سِنينَ وفي السَّنةِ السَّابِعةِ أَطْلِقْهُ مِنْ عِندِكَ حُرًّا.</span>
<span className="v" id="v13" data-v="13" data-xref="مَلاخِي 3:5; إرْمِيا 22:13; الأمثال 3:27–28"> وإذا أطلقتَهُ حُرًّا مِنْ عِندِكَ فلا تُطلِقْهُ فارِغًا لا شيءَ معَهُ</span>
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:2; الأمثال 10:22; المَزامِير 68:10"> بل زَوِّدْهُ مِنْ غنَمِكَ وبَيدَرِكَ ومَعصَرَتِكَ مِمَّا بارَكَكَ الرّبُّ إلهُكَ فيهِ.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّثْنِيَة 16:12; ﻣﺘﻰ 18:32–33; التَّثْنِيَة 5:14–15"> أُذكُرْ أنَّكَ كُنتَ عبدًا في أرضِ مِصْرَ وفداكَ الرّبُّ إلهُكَ. ولِذلِكَ أنا آمُرُكَ اليومَ بِهذِهِ الوصيَّةِ.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 21:5–6; المَزامِير 40:8; المَزامِير 40:6"> فإنْ قالَ لكَ: ”لا أخرُجُ مِنْ عِندِكَ لأنِّي أُحِبُّكَ وأُحِبُّ بَيتَكَ، وأجِدُ الإقامةَ عِندَكَ خَيرًا لي“،</span>
<span className="v" id="v17" data-v="17" data-xref="صَمُوئيلَ الأوَّلُ 1:22; اللّاويِّين 25:39–42"> فَخُذِ المِخرَزَ وضَعْهُ في أُذُنِهِ عِندَ بابِ بَيتِكَ، فيكونَ لكَ عبدًا طُولَ حياتِهِ، وجارِيَتُكَ أيضًا تفعَلُ بِها مِثلَ ذلِكَ.</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 17:7–8; إشَعْياء 16:14; إشَعْياء 21:16"> لا تأسَفْ أنْ تُطلِقَهُ حُرًّا مِنْ عِندِكَ، فهوَ كانَ يستَحِقُّ ضِعْفَ أُجرَةِ أجيرٍ لِخِدمَتِهِ لكَ سِتَّ سِنينَ، فيُبارِكُكَ الرّبُّ إلهُكَ في جميعِ ما تعمَلُهُ.</span>
</div>
</div>

## أبكار البهائم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 13:2; الخُرُوج 13:12; اللّاويِّين 27:26"> كُلُّ بِكْرٍ ذكَرٍ يُولَدُ لكَ في غنَمِكَ وبقَرِكَ تُقدِّسُهُ هِبةً لِلرّبِّ إلهِكَ. لا تستَعمِلِ البِكْرَ مِنْ بقَرِكَ ولا تَجُزَّ البِكْرَ مِنْ غنَمِكَ،</span>
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 12:17; التَّثْنِيَة 14:23; التَّثْنِيَة 12:5–7"> بل كُلْهُ أنتَ وأهلُ بَيتِكَ أمامَ الرّبِّ إلهِكَ، سنَةً بَعدَ سنَةٍ، في الموضِعِ الّذي يختارُهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّثْنِيَة 17:1; مَلاخِي 1:7–8; اللّاويِّين 22:19–25"> لكنْ إنْ كانَ بهِ عَيبٌ مِنْ عرَجٍ أو عمًى أو سائرِ العُيوبِ، فلا تذبَحْهُ لِلرّبِّ إلهِكَ.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 12:21–22; التَّثْنِيَة 12:15–16"> بل في بَيتِكَ تأكُلُهُ، سَواءٌ كُنتَ نَجِسًا أو طاهرًا بِحسَبِ الشَّريعةِ. تأكُلهُ كالغزالِ والوَعْلِ.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 12:16; اللّاويِّين 7:26; التَّثْنِيَة 12:23"> أمَّا دَمُهُ فلا تأكُلْهُ، بلِ اَسْفِكْهُ على الأرضِ كالماءِ.</span>
