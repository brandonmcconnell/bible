## وعد بالسعادة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:10; هُوشَع 2:23; زَكَريّا 13:9"> سيكونُ عددُ بَني إِسرائيلَ كرَملِ البحرِ الّذي لا يُقاسُ ولا يُعَدُّ، وفي المَوضِعِ الّذي قيلَ لهُم فيهِ: ما أنتم بِشعبي، يُقالُ لهُم: أنتُم أبناءُ اللهِ الحَيِّ.</span>
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 50:1; حِزْقِيال 16:25; حِزْقِيال 23:45"> ويجتَمِعُ بَيتُ يَهوذا وبَنو إِسرائيلَ جميعًا ويَختارونَ لهُم رئيسًا واحدًا، ويَنمَونَ في أرضِهِم، فيكونُ اليومُ الّذي يَزرَعُهُم فيهِ اللهُ يومًا عظيمًا.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 16:22; إشَعْياء 32:13–14; حِزْقِيال 19:13"> فقولوا لإِخوَتِكُم وأخَواتِكُم بَني إِسرائيلَ: ”أنتُم شعبي، وأنا أرحَمُكُم“.</span>
</div>
</div>

## إسرائيل الزانية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="إرْمِيا 13:14; هُوشَع 1:6; حِزْقِيال 8:18"> ”حاكِموا أمَّكُم، حاكِموها فما هيَ اَمرَأتي، ولا أنا رَجُلُها، لِتُزيحَ زِناها عَنْ وجهِها، وفِسقَها مِنْ بَينِ ثَديَيها</span>
<span className="v" id="v5" data-v="5" data-xref="إشَعْياء 1:21; هُوشَع 3:1; إرْمِيا 44:17–18"> لِئلَّا أفضحَ عُريَها وأرُدَّها إلى أصلِها كما كانَت يومَ ميلادِها، وأجعلَها كقَفرٍ وأقطعَ عَنها المطَرَ كأرضٍ قاحِلةٍ، وأُميتَها بالعطَشِ.</span>
<span className="v" id="v6" data-v="6" data-xref="أيُّوب 19:8; أيُّوب 3:23; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:7–9"> ولا أرحمُ بَنيها، فهُم بَنو زِنىً</span>
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 2:2; هُوشَع 5:13; حِزْقِيال 23:4"> لأنَّ أُمَّهُم زَنَت. تِلكَ الّتي حَبِلَت بِهِم عَمِلَت أعمالًا مُشينةً. قالَت: ”أتبَعُ عُشَّاقي الّذينَ يُعطونَني خُبزي ومائي وصوفي وكتاني وزَيتي وشرابي“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 1:3; هُوشَع 4:11; هُوشَع 13:2"> لِذلِكَ سأُسَيِّجُ طريقَها بِالشَّوكِ، وأُحَوِّطُهُ بِحائطٍ فَلا تَجِدُ سَبيلَها.</span>
<span className="v" id="v9" data-v="9" data-xref="هُوشَع 2:3; حَجَّي 1:6–11; مَلاخِي 1:4"> فتَتبعُ عُشَّاقَها ولا تَلحَقُ بِهِم، وتَطلُبُهُم فلا تَجدُهُم، فتقولُ: ”أذهَبُ وأرجِعُ إلى رَجُلي الأوَّلِ لأنَّ حاليَ معَهُ كانَت خيرًا مِما هيَ علَيهِ الآنَ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 23:29; إرْمِيا 13:26; هُوشَع 13:7–8"> هيَ لا تَعرِفُ أنِّي أنا أعطَيتُها القمحَ والخمرَ والزَّيتَ، وأكثَرْتُ لها الفِضَّةَ والذَّهبَ فصنعَتْ مِنهُما تِمثالًا لِبَعلٍ.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 1:13–14; عامُوس 5:21; إرْمِيا 7:34"> فلذلِكَ أستَعيدُ مِنها قمحي في وقتِه وخمري في مَوعِدِه، وأنزِعُ عَنها صوفي وكتَّاني اللَّذَينِ تكْسو بِهِما عَورَتَها.</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 7:23; إشَعْياء 5:5; هُوشَع 2:5"> فأكشِفُ جسَدَها كُلَّهُ أمامَ عيونِ عُشَّاقِها، وما مِنْ أحدٍ يُنقِذُها مِنْ يَدي،</span>
<span className="v" id="v13" data-v="13" data-xref="هُوشَع 13:6; القُضاة 3:7; هُوشَع 13:1"> وأُبطِلُ كُلَّ سُرورِها وأعيادِها ورؤُوسِ شُهورِها وسُبوتِها وكُلَّ اَحتِفالاتِها،</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 40:1–2; رُؤيا يُوحَنّا 12:6; حِزْقِيال 20:35–36"> وأُدَمِّرُ كَرمَها وتينَها ممَّا قالَت هوَ أُجرَتي الّتي نِلتُها مِنْ عُشَّاقي، وأُصَيِّرُهما وَعرًا فيَأكُلُهُما وَحشُ البرِّيَّةِ.</span>
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 2:2; حِزْقِيال 28:26; هُوشَع 11:1"> وأُعاقِبُها على الأيّامِ الّتي كانَت تُبَخِّرُ فيها لِلبَعلِ وتتَزَيَّنُ بِأساورِها وحُليها وتَتبَعُ عُشَّاقَها وتَنساني. هكذا قالَ الرّبُّ.</span>
</div>
</div>

## محبة الرّبّ لشعبه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 54:5; هُوشَع 2:7; إرْمِيا 3:14"> لِذلِكَ سأفتِنُها وأجيءُ بِها إلى البرِّيَّةِ وأخاطِبُ قلبَها.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 23:13; يَشُوع 23:7; المَزامِير 16:4"> وهُناكَ أُعيدُ إليها كُرومَها، مِنْ وادي عَكورَ إلى مَدخَلِ تَقْوةَ، فتَخضَعُ لي هُناكَ كما في صِباها وفي يومِ صُعودِها مِنْ أرضِ مِصْرَ.</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 34:25; حِزْقِيال 39:9–10; المَزامِير 46:9"> في ذلِكَ اليومِ أقولُ أنا الرّبُّ، تَدعوني زَوجي، ولا تَدعوني بَعلي مِنْ بَعدُ،</span>
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:2; رُؤيا يُوحَنّا 21:2; إشَعْياء 54:8–10"> لأنِّي سأُزيلُ اَسمَ البَعلِ مِنْ فَمِها، فلا تَذكُرُهُ مِنْ بَعدُ بِاَسْمِهِ.</span>
<span className="v" id="v20" data-v="20" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:20; يُوحَنّا 17:3; إرْمِيا 24:7"> وأقطَعُ لها عَهدًا في ذلِكَ اليومِ معَ وَحشِ البرِّيَّةِ وطُيورِ السَّماءِ وزَحَّافاتِ الأرضِ، وأكسِرُ القَوسَ والسَّيفَ وأدَواتِ الحربِ مِنَ الأرضِ وأجعلُها تَنامُ في أمانٍ.</span>
<span className="v" id="v21" data-v="21" data-xref="زَكَريّا 8:12; إشَعْياء 65:24; ﺭﻭﻣﻴﺔ 8:32"> وأتزَوَّجُكِ إلى الأبدِ. أتزَوَّجُكِ بِالصِّدقِ والعَدلِ والرَّأفَةِ والرَّحمةِ،</span>
<span className="v" id="v22" data-v="22" data-xref="يُوئيل 2:19; هُوشَع 1:4; هُوشَع 1:11"> أتزَوَّجُكِ بِكُلِّ أمانةٍ، فتَعرِفينَ أنِّي أنا الرّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﺭﻭﻣﻴﺔ 9:25–26; رسالَةُ بُطرُسَ الأُولَى 2:9–10; زَكَريّا 13:9"> وفي ذلِكَ اليومِ أستَجيبُ، يقولُ الرّبُّ، لِلسَّماواتِ، والسَّماواتُ تستَجيبُ لِلأرضِ،</span>
<span className="v" id="v24" data-v="24"> والأرضُ تستَجيبُ لِلقمحِ والخمرِ والزَّيتِ، وهذِهِ كُلُّها تستَجيبُ ليَزرَعيلَ.</span>
<span className="v" id="v25" data-v="25"> وأزرَعُ شعبي في الأرضِ وأرحمُ ”لا رَحمةَ“ وأقولُ لِ”لا شعبي“، أنتَ شعبي وهوَ يقولُ لي: أنتَ إلَهي.</span>
</div>
