<div className="bp">

</div>
<div className="bp">

<span className="v" id="v6" data-v="6" data-xref="هُوشَع 10:8; إشَعْياء 5:6; إشَعْياء 7:23"> ها أنتُم مُشَرَّدونَ مِنْ وجهِ الشِّدةِ، فمِصْرُ تَجمَعُكُم وموفُ تَدفِنُكُم. القُرَّاصُ يَرِثُ كُنوزَ فِضَّتِكُم، والعَوسَجُ يَنبُتُ في خيامِكُم.</span>
</div>
<div className="bp">

</div>
