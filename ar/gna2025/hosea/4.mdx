## خصومة الرّبّ مع إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 4:22; ميخا 6:2; إشَعْياء 3:13–14"> إسمَعوا كلِمةَ الرّبِّ يا بَني إِسرائيلَ. لِلرّبِّ خُصومةٌ معَ سُكَّانِ الأرضِ. فما في الأرضِ أمانٌ ولا رَحمةٌ ولا معرِفَةُ اللهِ،</span>
<span className="v" id="v2" data-v="2" data-xref="هُوشَع 6:9; إرْمِيا 6:7; هُوشَع 7:1"> بلِ اللَّعنةُ والغَدرُ والقَتلُ والسَّرِقةُ والفِسْقُ. هذِهِ كُلُّها تَجاوَزَت كُلَّ حَدٍّ، والدِّماءُ تَلحَقُ بِالدِّماءِ.</span>
<span className="v" id="v3" data-v="3" data-xref="صَفَنْيا 1:3; إشَعْياء 24:4–12; ناحُوم 1:4"> لِذلِكَ تنوحُ الأرضُ ويَذبُلُ كُلُّ ساكِنٍِ فيها معَ وَحشِ البرِّيَّةِ وطَيرِ السَّماءِ، بل سَمَكُ البحرِ أيضًا يَهلِكُ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 17:12; هُوشَع 4:17; عامُوس 5:13"> ولكنْ لا يُخاصِمْ أحدٌ شعبي ولا يُوَبِّخْهُ، فخُصومَتي معكُم أنتُم أيُّها الكهَنةُ.</span>
<span className="v" id="v5" data-v="5" data-xref="إرْمِيا 15:8; هُوشَع 2:2; غَلاطِيَّة 4:26"> تَسقُطونَ في النَّهارِ وفي اللَّيلِ، ويسقُطُ الأنبياءُ أيضًا معكُم، فأنتُم عِلَّةُ دَمارِ شعبِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 5:13; الأمثال 19:2; هُوشَع 6:6"> لَحِقَ الدَّمارُ بِشعبي لأنَّهُم لا يَعرفونَني. وبِما أنَّكُم رفَضتُم أنْ تَعرِفوني. فأنا أرفُضُكُم فلا تكونونَ لي كهَنةً، وبما أنَّكُم نَسيتُم شريعةَ إلهِكُم فأنا أيضًا أنسى بَنيكُم.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="هُوشَع 13:6; فِيلِبِّي 3:19; صَمُوئيلَ الأوَّلُ 2:30"> على قَدْرِ ما تكثُرونَ أيُّها الكهَنةُ تكثُرُ خطاياكُم، فسَأُبدِلُ مَجدَكُم هَوانًا.</span>
<span className="v" id="v8" data-v="8" data-xref="ميخا 3:11; إشَعْياء 56:11; تِيطُس 1:11"> تأكُلونَ مِنْ ذبائِحِ التَّكفيرِ عَنْ خطايا شعبي، فتَحمِلونَهُم على الإِكثارِ مِنَ الإِثْمِ.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 24:2; إرْمِيا 5:31; ﻣﺘﻰ 15:14"> فكما أُعاقِبُ الشَّعبَ، فكذلِكَ أُعاقِبُكُم أيُّها الكهَنةُ على طُرُقِكُم وأُجازيكم شَرًّا على أعمالِكُم.</span>
<span className="v" id="v10" data-v="10" data-xref="ميخا 6:14; اللّاويِّين 26:26; حَجَّي 1:6"> فتَأكُلونَ ولا تَشبَعونَ، وتَزنونَ ولا تكثُرونَ لأنَّكُم تَرَكتُمُ الرّبَّ وزَنَيتُم وراءَ آلهةٍ أُخَرَ.</span>
</div>
</div>

## التنديد بعبادة الآلهة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="الأمثال 20:1; لُوقا 21:34; إشَعْياء 28:7"> الخمرُ الجديدةُ والمُعَتَّقةُ تُعَطِّلانِ الفَهمَ.</span>
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 2:27; هُوشَع 5:4; حَبَقُّوق 2:19"> شعبي يستَشيرونَ الإلهَ الخشَبةَ ويستَخبِرونَ الإلهَ الوتَدَ. روحُ الزِّنى أضَلَّهُم، فزَنَوا في الخِفْيةِ عَنِّي.</span>
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 1:29; إرْمِيا 3:6; حِزْقِيال 6:13"> يذبَحونَ الذَّبائِحَ على رُؤوسِ الجِبالِ، ويُبَخِّرونَ تَحتَ أشجارِ البَلُّوطِ والحَورِ والبُطمِ لأنَّ ظِلَّها حسَنٌ.<br />بناتُكُم يَزْنينَ وكنَّاتُكُم يَفسُقْنَ،</span>
<span className="v" id="v14" data-v="14" data-xref="هُوشَع 4:1; يُوحَنّا 8:43; أفَسُس 4:18"> فلا أعاقِبُ بَناتِكُم على زِناهُنَّ ولا كنَّاتِكُم على فِسقِهِنَّ. الرِّجالُ أنفُسُهُمُ اَنْفَرَدوا بالزَّواني، وذَبَحوا الذَّبائِحَ معَ بَغايا المَعابِدِ. فالشَّعبُ الّذي لا يتَبَيَّنُ الحَقَّ يتَهَوَّرُ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="هُوشَع 12:11; هُوشَع 9:15; عامُوس 4:4"> إنْ كُنتُم أنتُم تَزْنُونَ يا بَني إسرائيلَ، وراءَ آلهةٍ أُخرى، فلا تَجعَلوا بَيتَ يَهوذا يَرتكِبونَ هذا الإثْمَ.<br />لا تذهَبوا إلى الجِلجالِ ولا تَصعَدوا إلى بَيتِ آونَ يا بَني إِسرائيلَ، ولا تَحلِفوا بِحياةِ الرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إرْمِيا 7:24; إرْمِيا 8:5; إرْمِيا 3:11"> جمَحتُم يا بَني إِسرائيلَ جَماحَ عِجْلَةٍ، فكيفَ يَرعاكُمُ الرّبُّ كخَروفٍ في مَرْجٍ رحيبٍ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="المَزامِير 81:12; رُؤيا يُوحَنّا 22:11; هُوشَع 13:2"> بَيتُ أفرايمَ شعبٌ مَصرورٌ بِالأوثانِ فاَتْرُكوهُم.</span>
<span className="v" id="v18" data-v="18" data-xref="ميخا 3:11; عامُوس 5:12; المُلُوكِ الثّانِي 17:7–17"> هُمُ اَنْصَرَفوا إِلى السُّكْرِ وأسلَموا أنفُسَهُم لِلزِّنى. أحَبُّوا الهَوانَ حُبَّهُم لِلمُجونِ.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 1:29; إرْمِيا 4:11–12; هُوشَع 13:15"> فلتَحمِلْهُمُ الرِّيحُ في أجنِحَتِها، وليَخْجَلوا مِنْ ذبائِحِهِم.</span>
</div>
