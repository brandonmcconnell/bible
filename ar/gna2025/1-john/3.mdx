## أبناء الله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 1:12; غَلاطِيَّة 3:26; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:18"> أنظُروا كم أحَبَّنا الآبُ<br />حتّى نُدعى أبناءَ اللهِ،<br />ونحنُ بِالحقيقَةِ أبناؤُهُ.<br />إذا كانَ العالَمُ لا يَعرِفُنا،<br />فلأنَّهُ لا يَعرِفُ اللهَ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="فِيلِبِّي 3:21; كُولُوسِي 3:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:12"> يا أحبّائي، نَحنُ الآنَ أبناءُ اللهِ.<br />وما اَنكشَفَ لنا بَعدُ ماذا سنَكونُ.<br />نَحنُ نَعرِفُ أنَّ المَسيحَ متى ظهَرَ نكونُ مِثلَهُ<br />لأنَّنا سنَراهُ كما هوَ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1; رسالَةُ يُوحَنّا الأُولَى 2:6; رسالَةُ بُطرُسَ الثّانيَة 3:14"> ومَنْ كانَ لَه هذا الرَّجاءُ في المَسيحِ<br />طهَّرَ نَفسَهُ كما أنَّ المَسيحَ طاهِرٌ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:17; ﺭﻭﻣﻴﺔ 3:20; ﺭﻭﻣﻴﺔ 4:15"> مَنْ خَطِئَ عَمِل شَرًّا،<br />لأنَّ الخَطيئَةَ شَرٌّ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:21; العِبرانِيّين 9:28; رسالَةُ بُطرُسَ الأُولَى 3:18"> تَعرِفونَ أنَّ المَسيحَ ظهَرَ لِيُزيلَ الخَطايا<br />وهوَ الّذي لا خَطيئَةَ لَه.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="رسالَةُ يُوحَنّا الثّالثَة 1:11; رسالَةُ يُوحَنّا الأُولَى 2:4; رسالَةُ يُوحَنّا الأُولَى 3:9"> مَنْ ثَبَتَ فيهِ لا يَخطأُ،<br />ومَنْ يَخطَأُ لا يكونُ رآهُ ولا عَرَفَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:29; رسالَةُ بُطرُسَ الأُولَى 1:15–16; ﺭﻭﻣﻴﺔ 2:13"> يا أبنائي لا يُضَلِّلْكُم أحَدٌ:<br />مَنْ عَمِلَ البِرَّ كانَ بارًّا<br />كما أنَّ المَسيحَ بارٌّ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="العِبرانِيّين 2:14; يُوحَنّا 8:44; كُولُوسِي 2:15"> ومَنْ عمِلَ الخَطيئةَ كانَ مِنْ إبليسَ،<br />لأنَّ إبليسَ خاطِئٌ مِنَ البَدءِ.<br />وإنَّما ظهَرَ اَبنُ اللهِ ليَهدِمَ أعمالَ إبليسَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:18; رسالَةُ بُطرُسَ الأُولَى 1:23; يُوحَنّا 1:13"> كُلُّ مَولودٍ مِنَ اللهِ لا يَعمَلُ الخَطيئَةَ<br />لأنَّ زَرعَ اللهِ ثابِتٌ فيهِ:<br />لا يَقدِرُ أنْ يَعمَلَ الخَطيئَةَ وهُوَ مِنَ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:8; رسالَةُ يُوحَنّا الثّالثَة 1:11; لُوقا 6:35"> بِهذا يتَبَيَّنُ أبناءُ اللهِ وأبناءُ إبليسَ.<br />ومَنْ لا يَعمَلُ البِرَّ لا يكونُ مِنَ اللهِ،<br />ولا يكونُ مِنَ اللهِ مَنْ لا يُحِبُّ أخاهُ.</span>
</div>
</div>

## أحبّوا بعضكم بعضًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 15:12; رسالَةُ يُوحَنّا الثّانيَة 1:5; رسالَةُ يُوحَنّا الأُولَى 4:7"> فالوَصيَّةُ الّتي سَمِعتُموها مِنَ البَدءِ<br />هِيَ أنْ يُحِبَّ بَعضُنا بَعضًا،</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="العِبرانِيّين 11:4; التَّكوين 4:4–15; الأمثال 29:27"> لا أنْ نكونَ مِثلَ قايينَ<br />الّذي كانَ مِنَ الشِّرِّيرِ فقَتَلَ أخاهُ.<br />ولماذا قتَلَهُ؟<br />لأنَّ أعمالَهُ كانَت شِرِّيرَةً وأعمالَ أخيهِ كانَت صالِحَةً بارَّةً.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 15:18–19; يُوحَنّا 17:14; لُوقا 21:17"> فلا تَتَعجَّبوا، أيُّها الإخوَةُ<br />إذا أبغَضَكُمُ العالَمُ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 13:35; رسالَةُ يُوحَنّا الأُولَى 5:2; رسالَةُ يُوحَنّا الأُولَى 4:20–21"> نَحنُ نَعرِفُ أنَّنا اَنتَقَلنا مِنَ الموتِ إلى الحياةِ<br />لأنَّنا نُحِبُّ إخوَتَنا.<br />مَنْ لا يُحِبُّ بَقِيَ في الموتِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 5:21–22; الأمثال 26:24–26; رُؤيا يُوحَنّا 21:8"> مَنْ أبغَضَ أخاهُ فهوَ قاتِلٌ<br />وأنتُم تَعرِفونَ أنَّ القاتِلَ<br />لا تَثبُتُ الحياةُ الأبدِيَّةُ فيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:9–11; أفَسُس 5:2; يُوحَنّا 13:34"> ونَحنُ عَرَفنا المَحبَّةَ حينَ ضَحَّى المَسيحُ بِنَفسِهِ لأجلِنا، فعَلَينا نَحنُ أنْ نُضَحِّيَ بِنُفوسِنا لأجلِ إخوَتِنا.</span>
<span className="v" id="v17" data-v="17" data-xref="العِبرانِيّين 13:16; إشَعْياء 58:7–10; التَّثْنِيَة 15:7–11"> مَنْ كانَتْ لَه خَيراتُ العالَمِ ورأى أخاهُ مُحتاجًا فأغلَقَ قلبَهُ عَنهُ، فكيفَ تَثبُتُ مَحبَّةُ اللهِ فيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="يَعقُوب 2:15–16; ﺭﻭﻣﻴﺔ 12:9; رسالَةُ بُطرُسَ الأُولَى 1:22"> يا أبنائي، لا تكُنْ مَحَبَّتُنا بِالكلامِ أو بِاللِّسانِ بَلْ بِالعَمَلِ والحَقِّ.</span>
</div>
</div>

## الثقة أمام الله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="العِبرانِيّين 10:22; رسالَةُ يُوحَنّا الأُولَى 3:21; العِبرانِيّين 6:10–11"> بِهذا نَعرِفُ أنَّنا مِنَ الحَقِّ<br />فتَطمَئِنُّ قُلوبُنا أمامَ اللهِ</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 139:1–4; إرْمِيا 17:10; العِبرانِيّين 4:13"> إذا وبَّخَتْنا قُلوبُنا،<br />لأنَّ اللهَ أعظَمُ مِنْ قُلوبِنا<br />وهوَ يَعلَمُ كُلَّ شيءٍ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:28; العِبرانِيّين 4:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:12"> إذا كانَت قُلوبُنا لا تُوَبِّخُنا أيُّها الأحِبّاءُ،<br />فلَنا ثِقَةٌ عِندَ اللهِ</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 15:7; يُوحَنّا 9:31; رسالَةُ يُوحَنّا الأُولَى 5:14"> أنْ نَنالَ مِنهُ كُلَّ ما نَطلُبُ<br />لأنَّنا نَحفَظُ وصاياهُ ونَعمَلُ بِما يُرضيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 6:29; يُوحَنّا 13:34; رسالَةُ بُطرُسَ الأُولَى 1:22"> ووَصِيَّتُهُ هِيَ أنْ نُؤمِنَ باَسمِ اَبنِهِ يَسوعَ المَسيحِ،<br />وأنْ يُحِبَّ بَعضُنا بَعضًا كما أوصانا.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:16; رسالَةُ يُوحَنّا الأُولَى 4:15–16; رسالَةُ يُوحَنّا الأُولَى 4:12–13"> ومَنْ عَمِلَ بِوَصايا اللهِ ثبَتَ في اللهِ وثبَتَ اللهُ فيهِ.<br />وإنَّما نَعرِفُ أنَّ اللهَ ثابِتٌ فينا<br />مِنَ الرُّوحِ الّذي وَهبَهُ لنا.</span>
</div>
