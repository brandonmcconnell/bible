<div className="bp">

<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 13:34–35; رسالَةُ يُوحَنّا الأُولَى 2:5; رسالَةُ يُوحَنّا الأُولَى 3:22–24"> ونحنُ نعرِفُ أنَّنا نُحِبُّ أبناءَ اللهِ إذا كُنّا نُحِبُّ اللهَ ونَعمَلُ بِوَصاياهُ،</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 14:15; يُوحَنّا 15:10; يُوحَنّا 14:21–24"> لأنَّ مَحبَّةَ اللهِ هِيَ في أنْ نَعمَلَ بِوَصاياه.<br />وما وَصاياهُ ثَقيلَةٌ.</span>

## الشهادة ليسوع المسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 15:26; العِبرانِيّين 9:14; يُوحَنّا 16:13"> هذا الّذي جاءَ هوَ يَسوعُ المَسيحُ،<br />جاءَ بِماءٍ ودَمٍ،<br />جاءَ لا بِالماءِ وحدَهُ، بَلْ بِالماءِ والدَّمِ.<br />والرُّوحُ هوَ الّذي يَشهَدُ،<br />لأنَّ الرُّوحَ هوَ الحَقُّ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 28:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:14; اعمال الرسل 5:32"> والّذينَ يَشهَدونَ هُم ثلاثةٌ.</span>
<span className="v" id="v8" data-v="8" data-xref="رسالَةُ يُوحَنّا الأُولَى 5:6–7; يُوحَنّا 15:26; رسالَةُ بُطرُسَ الأُولَى 3:21"> الرُّوحُ والماءُ والدَّمُ، وهَؤُلاءِ الثَّلاثَةُ هُم في الواحدِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 3:16–17; ﻣﺘﻰ 17:5; اعمال الرسل 17:31"> إذا كُنّا نَقبَلُ شَهادَةَ النَّاسِ،<br />فشَهادَةُ اللهِ أعظَمُ.<br />وهذِهِ هِيَ شهادَةُ اللهِ الّتي شَهِدَها لاَبنِهِ:</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ﺭﻭﻣﻴﺔ 8:16; غَلاطِيَّة 4:6; يُوحَنّا 3:33"> مَن يُؤمِنُ باَبنِ اللهِ، فَلهُ تِلكَ الشَّهادَةُ. ومَنْ لا يُصَدِّقُ اللهَ جعَلَهُ كاذِبًا،<br />لأنَّهُ لا يُؤمِنُ بالشَّهادَةِ الّتي شَهِدَها لاَبنِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﺭﻭﻣﻴﺔ 6:23; رسالَةُ يُوحَنّا الأُولَى 2:25; يُوحَنّا 3:36"> وهذِهِ الشَّهادَةُ هِيَ أنَّ اللهَ أعطانا الحياةَ الأبدِيَّةَ،<br />وأنَّ هذِهِ الحياةَ هِيَ في اَبنِهِ</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 3:36; يُوحَنّا 5:24; يُوحَنّا 1:12"> مَن يكونُ لَه الاَبنُ فلَهُ الحياةُ.<br />مَنْ لا يكونُ لَه اَبنُ اللهِ، فلا تكونُ لَه الحياةُ.</span>
</div>
</div>

## الحياة الأبديَّة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 20:31; يُوحَنّا 1:12; رسالَةُ بُطرُسَ الثّانيَة 1:10–11"> أكتُبُ إلَيكُم بِهذا لِتَعرِفوا أنَّ الحياةَ الأبدِيَّةَ لكُم، أنتُمُ الّذينَ تُؤمِنونَ باَسمِ اَبنِ اللهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 14:13; إرْمِيا 33:3; إرْمِيا 29:12–13"> والثِّقَةُ الّتي لنا عِندَ اللهِ هِيَ أنَّنا إذا طَلَبنا شيئًا مُوافِقًا لِمَشيئتِهِ اَستَجابَ لنا.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 11:9–10; مَرْقُس 11:24; الأمثال 15:29"> وإذا كُنّا نَعرِفُ أنَّهُ يَستَجيبُ لنا في كُلِّ ما نَطلُبُهُ مِنهُ، فنَحنُ نَعرِفُ أنَّنا نَنالُ كُلَّ ما نَطلبُهُ مِنهُ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="العِبرانِيّين 6:4–6; صَمُوئيلَ الأوَّلُ 2:25; إرْمِيا 11:14"> وإذا رأى أحَدٌ أخاهُ يَرتَكِبُ خَطيئَةً لا تُؤَدّي إلى الموتِ، فعَلَيهِ أنْ يَدعُوَ إلى اللهِ فيَمنَحَ أخاهُ الحياةَ. هذا يَصدُقُ على الّذينَ لا تُؤَدّي خَطاياهُم إلى الموتِ. ولكِنْ هُناكَ خَطايا تُؤَدّي إلى الموتِ، فلا أطلُبُ الصَّلاةَ لأجلِها.</span>
<span className="v" id="v17" data-v="17" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:4; رسالَةُ يُوحَنّا الأُولَى 5:16; يَعقُوب 4:7–10"> كُلُّ مَعصِيَةٍ خَطيئَةٌ، ولكِنْ هُناكَ مِنَ الخَطايا ما لا يُؤَدّي إلى الموتِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:9; رسالَةُ يُوحَنّا الأُولَى 5:4; رسالَةُ يُوحَنّا الأُولَى 5:1"> نَعرِفُ أنَّ كُلَّ مَنْ ولِدَ مِنَ اللهِ لا يَخطأُ، لأنَّ المولودَ مِنَ اللهِ يَصونُهُ فلا يَمَسُّهُ الشِّرِّيرُ.</span>
</div>
</div>
