## محبة الرّبّ لشعبه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="حَبَقُّوق 1:1; إشَعْياء 13:1; حَجَّي 1:1"> هذِهِ كَلِمةُ الرّبِّ إلى إِسرائيلَ كما أوحى بها على لِسانِ ملاخي:</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 41:8–9; إرْمِيا 31:3; التَّثْنِيَة 10:15"> قال الرّبُّ: ”أحبَبْتُكُم“. فقُلتُم: ”كيفَ أحبَبْتَنا؟“ فقالَ: ”أمَا عيسو أخًا ليَعقوبَ؟ فأحبَبْتُ يَعقوبَ</span>
<span className="v" id="v3" data-v="3" data-xref="عُوبَدْيا 1:10; حِزْقِيال 35:3–4; إرْمِيا 49:16–18"> وأبغَضتُ عيسو وجعَلتُ جِبالَهُ خرابًا ومِيراثَهُ لذِئابِ البرِّيَّةِ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 25:14; حِزْقِيال 35:9; حِزْقِيال 11:10"> قالَ الأدوميُّونَ بَنو عيسو: ”تشَتَّتَ شَمْلُنا، لكنَّنا سنعودُ ونبني الخَرائِبَ: ولكنْ يقولُ الرّبُّ القديرُ: ”هُم يَبنونَ وأنا أهدِمُ، فيُقالُ لهُم أهلُ الشَّرِّ والشَّعبُ الّذي غَضِبَ الرّبُّ علَيه إلى الأبدِ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 11:7; المَزامِير 58:10–11; حِزْقِيال 38:16"> فتَرى عُيونُكُم وتقولونَ: ”الرّبُّ عظيمٌ حتّى فيما وراءَ تُخومِ إِسرائيلَ“.</span>
</div>
</div>

## توبيخ الرّبّ لكهنته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="لُوقا 6:46; إشَعْياء 1:2; يُوحَنّا 13:13–17"> قالَ الرّبُّ القديرُ: ”الابنُ يُكرِمُ أباهُ، والعَبدُ يُكرِمُ سيِّدَهُ. فإنْ كُنتُ أنا أبًا، فأينَ كرامتي؟ وإنْ كُنتُ سيِّدًا، فأينَ مَهابَتي أيُّها الكهَنةُ الّذينَ تَحتَقِرونَ اَسمي. وتقولونَ: كيفَ اَحتَقَرْنا اَسمَكَ؟</span>
<span className="v" id="v7" data-v="7" data-xref="مَلاخِي 1:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:27–32; مَلاخِي 1:8"> إحتَقَرتُموهُ بِأنَّكُم تُقَرِّبونَ على مذبَحي خُبزًا نَجِسًا. وتقولونَ: كيفَ نجَّسناهُ؟ نَجَّستُموهُ بِقولِكُم: مائدةُ الرّبِّ مُحتقَرةٌ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 15:21; اللّاويِّين 22:19–25; هُوشَع 8:13"> إذا قَرَّبْتُمُ الأعمَى أوِ الأعرجَ أوِ السَّقيمَ ذبيحةً لي، أفلا يكونُ ذلِكَ شَرًّا؟ إنْ قَرَّبتُموهُ لِحاكِمِكُم أفيرضَى عَنكُم أو يَرفَعُ شأنَكُم؟“ هكذا قالَ الرّبُّ القديرُ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 32:11; إرْمِيا 27:18; رسالَةُ بُطرُسَ الأُولَى 1:17"> فالآنَ اَسْتَعطِفُوا أيُّها الكهنَةُ وجهَ اللهِ ليَحِنَّ علَينا. مِنْ أيديكُم وبِسبَبِكُم ما جَرى لنا. فكيفَ يرفعُ الرّبُّ القديرُ شأنَكُم؟</span>
<span className="v" id="v10" data-v="10" data-xref="ميخا 3:11; عامُوس 5:21–24; إرْمِيا 6:20"> يقولُ: ”ليتَ فيكُم مَنْ يُغلِقُ أبوابَ هَيكلي حتّى لا تُوقِدوا نارَ مذبحي عبَثًا. لا مسَرَّةَ لي بِكُم، ولا أرضى تَقدِمةً مِنْ أيديكُم.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 45:6; إشَعْياء 54:5; إشَعْياء 66:19–20"> فمِنْ مَشرِقِ الشَّمسِ إلى مَغرِبِها اَسمي عظيمٌ في الأُمَمِ، وفي كُلِّ مكانٍ يُحرَقُ لاَسمي البَخورُ وتُقرَّبُ تقدِمةٌ طاهرةٌ، لأنَّ اَسمي عظيمٌ في الأُمَمِ، أنا الرّبُّ القديرُ.</span>
<span className="v" id="v12" data-v="12" data-xref="مَلاخِي 1:6–8; مَلاخِي 2:8; عامُوس 2:7"> أمَّا أنتُم فدَنَّستُموهُ بِقولِكُم: مائِدةُ الرّبِّ مُنجَّسةٌ وثمَرَتُها طَعامٌ مَنبُوذٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="ميخا 6:3; اللّاويِّين 22:19–23; مَرْقُس 14:37–38"> وقالَ الرّبُّ القديرُ: ”تقولونَ: تَعِبْنا مِنْ هذا كُلِّهِ، وتتأفَّفونَ عليَّ. تَجيئُونَ بالمُغتَصَبِ والأعرَجِ والسَّقيمِ وتُقرِّبونَهُ تَقدِمةً لي. أفأرضى بهذا مِنْ أيديكم. أنا الرّبُّ؟</span>
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12; اعمال الرسل 5:1–10; الجامِعَة 5:4–5"> ملعُونٌ كُلُّ ماكرٍ يَنذُرُ ذكَرًا سليمًا في قطيعِهِ للرّبِّ ويذبحُ لَه ما يكونُ فيه عَيبٌ. فأنا الرّبُّ القديرُ مَلِكٌ عظيمٌ، واَسمي مَهيبٌ بَينَ الأُمَمِ“.</span>
</div>
</div>
