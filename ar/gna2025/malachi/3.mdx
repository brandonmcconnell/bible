## اقتراب يوم القضاء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَلاخِي 4:5; لُوقا 1:76; مَرْقُس 1:2–3"> وقالَ الرّبُّ القديرُ: ”ها أنا أُرسِلُ رسولي فيُهيِّئُ الطَّريقَ أمامي، وسُرعانَ ما يأتي إلى هَيكلِه الرّبُّ الّذي تَطلبونَهُ ورسولُ العَهدِ الّذي بهِ تُسَرُّونَ. هَا هوَ آتٍ.</span>
<span className="v" id="v2" data-v="2" data-xref="زَكَريّا 13:9; إشَعْياء 4:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:13–15"> فمَنْ تُرى يَحتَمِلُ يومَ مجيئِهِ. ومَنْ يَثبُتُ عِندَ ظُهورِهِ؟ فهوَ مثلُ نارِ المُمَحِّصِ وكصابُونِ القَصَّارِ.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 1:25; الأمثال 17:3; زَكَريّا 13:9"> ويَجلِسُ كمَنْ يُمحِّصُ الفِضَّةَ ويُنَقِّيها، فيُنَقِّي بَني لاوي ويُصفِّيهم كالذَّهبِ والفِضَّةِ ليُقرِّبوا التَّقدِمةَ للرّبِّ صادقينَ.</span>
<span className="v" id="v4" data-v="4" data-xref="أخبارِ الأيّامِ الثّانِي 7:1–3; إشَعْياء 56:7; زَكَريّا 8:3"> فتكونُ تَقدِمَةُ يَهوذا وأُورُشليمَ عُربُونًا للرّبِّ، كالأيّامِ السَّالفةِ والسِّنينَ القديمةِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="يَعقُوب 5:8–9; العِبرانِيّين 13:4; التَّثْنِيَة 5:11"> ”وأقتَرِبُ مِنكُم لأقاضِيَكُم وأكونَ شاهدًا عليمًا على العرَّافينَ والفاسقينَ والحالفينَ زُورًا، وعلى الّذينَ يظلُمونَ الأجيرَ في أُجرتِهِ والأرملةَ واليَتيمَ، والّذينَ يَصُدُّونَ الغريبَ ولا يخافُونَني، أنا الرّبُّ القديرُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="العَدَد 23:19; العِبرانِيّين 13:8; يَعقُوب 1:17"> ”فأنا الرّبُّ لا أتغيَّرُ، ولا أنتُم يا بَني يَعقوبَ تكلُّونَ.</span>
<span className="v" id="v7" data-v="7" data-xref="زَكَريّا 1:3; ﺭﻭﻣﻴﺔ 10:21; ﺭﻭﻣﻴﺔ 10:3"> مِنْ أيّامِ آبائِكُم زُغْتُم عَنْ فرائِضي وما عمِلتُم بها. إرجِعوا إليَّ أرجعْ إليكُم، أنا الرّبُّ القديرُ. وتقولونَ: كيفَ نرجِعُ؟</span>
<span className="v" id="v8" data-v="8" data-xref="الأمثال 3:9–10; ﻣﺘﻰ 22:21; مَرْقُس 12:17"> فأسألُكُم: أيَسلُبُ البشَرُ اللهَ؟ فأنتُم سَلَبتُموني، وتقولونَ: ماذا سلبْناكَ؟ سَلبتُموني العُشورَ والتَّقدماتِ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّثْنِيَة 28:15–19; حَجَّي 1:6–11; حَجَّي 2:14–17"> اللّعنةُ علَيكم، على الأمَّةِ كُلِّها. لأنَّكُم تسلُبونَني.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:6–8; ﻣﺘﻰ 6:33; الأمثال 3:9–10"> هاتوا جميعَ العُشورِ إلى بَيتِ مالِ الهَيكلِ، وليكُنْ في بيتي طعامٌ. جرِّبوني بذلِكَ، أنا الرّبُّ القديرُ، ترَوا إنْ كُنتُ لا أفتَحُ لكُم نوافذَ السَّماءِ وأفيضُ علَيكُم بَرَكةً لا حَصْرَ لها.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّثْنِيَة 11:14; يُوئيل 2:22; زَكَريّا 8:12"> وأمنَعُ عَنكُمُ الآفةَ، فلا تُفسِدُ ثمَرَ أرضِكُم ولا يكونُ لكُمُ الكرمُ عقيمًا في الحقلِ.</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 61:9; إشَعْياء 62:4; زَكَريّا 8:23"> فتُهنئُكُم جميعُ الأُمَمِ، لأنَّ أرضَكُم تكونُ أرضَ مسَرَّةٍ“. هكذا قالَ الرّبُّ القديرُ.</span>
</div>
</div>

## وعد الرّبّ بالرحمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="مَلاخِي 2:17; إشَعْياء 37:23; ﺭﻭﻣﻴﺔ 9:20"> وقالَ الرّبُّ: ”جُرتُم عليَّ بكلامِكُم وتقولونَ: ماذا تكلَّمنا علَيكَ؟</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 58:3; صَفَنْيا 1:12; المَزامِير 73:8–13"> تكلَّمتُم قائِلينَ: عِبادةُ اللهِ باطلةٌ، وما المَنفعةُ في حِفْظِنا شعائِرَهُ وفي سَيرِنا بلا لومٍ أمامَ الرّبِّ القديرِ؟</span>
<span className="v" id="v15" data-v="15" data-xref="مَلاخِي 4:1; إرْمِيا 12:1–2; مَلاخِي 2:17"> ونحنُ نرى أنَّ المُتَجبِّرينَ هُمُ السُّعَداءُ، وفاعِلي الشَّرِّ هُمُ الّذينَ يَنجَحونَ. يُجرِّبونَ اللهَ ويَنجونَ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="رُؤيا يُوحَنّا 20:12; المَزامِير 56:8; مَلاخِي 4:2"> وهُنا تكَلَّمَ خائفُو الرّبِّ، الواحدُ معَ الآخَرِ، وأصغى الرّبُّ وسمِعَ كلامَهُم. وكتَبَ الرّبُّ أمامَهُ كتابًا يُذكِّرُهُ بخائِفيهِ والّذينَ يُقدِّرونَ اَسمَهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 19:5; التَّثْنِيَة 7:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:18"> وقالَ الرّبُّ القديرُ: ”هؤلاءِ يكونونَ لي يومَ تَسجيلِ الأُمَمِ، وأُشفِقُ علَيهِم كما يُشفِقُ الإنسانُ على اَبنهِ الّذي يَخدِمُهُ،</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 25:46; المَزامِير 58:10–11; التَّكوين 18:25"> ويَعودونَ يُمَيِّزونَ بَينَ الصِّدِّيقِ والشِّرِّيرِ، بَينَ الّذي يعبدُ اللهَ والّذي لا يعبُدُهُ“.</span>
</div>
</div>

## مجيء يوم الرّبّ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19"> وقالَ الرّبُّ القديرُ: ”سيأتي يومٌ يَحترقُ فيهِ جميعُ المُتجبِّرينَ وفاعِلي الشَّرِّ كالقَشِّ في التَّنُّورِ المُتَّقِدِ. في ذلِكَ اليومِ يَحترقونَ، حتّى لا يَبقى لهُم أصلٌ ولا فَرْعٌ.</span>
<span className="v" id="v20" data-v="20"> وتُشرِقُ لكُم أيُّها المُتَّقونَ لاَسمي شمسُ البِرِّ والشِّفاءُ في أجنِحتِها، فتسرَحونَ وتمرَحونَ كالعُجُولِ المَعلوفةِ،</span>
<span className="v" id="v21" data-v="21"> وتدوسونَ الأشرارَ وهُم رمادٌ تَحتَ أخامِصِ أقدامِكُم، يومَ أعمَلُ عملي، أنا الرّبُّ القديرُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22"> ”أُذكُروا شريعةَ موسى عبدي الّتي أوصَيتُهُ بها في حوريبَ إلى جميعِ بَني إِسرائيلَ، لتكونَ فرائضَ وأحكامًا.</span>
<span className="v" id="v23" data-v="23"> ”ها أنا أُرسِلُ إليكُم إيليَّا النَّبيَّ، قَبلَ أنْ يجيءَ يومُ الرّبِّ العظيمِ الرَّهيبِ.</span>
<span className="v" id="v24" data-v="24"> فيُصالِحُ الآباءَ معَ البنينَ، والبنينَ معَ الآباءِ، لِئلَّا أجيءَ وأضربَ الأرضَ بالحِرمَانِ“.</span>
</div>
