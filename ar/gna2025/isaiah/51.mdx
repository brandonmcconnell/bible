## الرب يعزي شعبه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 51:7; المَزامِير 94:15; العِبرانِيّين 12:14"> ”إسمعوا يا مَنْ يُريدُون الحَقّ<br />ويَطلُبونَ عَدالةَ الرّبِّ!<br />أُنظُروا إلى الصَّخرِ الّذي نُحِتُّم مِنهُ.<br />وإلى المَقلَعِ الّذي مِنهُ أقتُلِعتُم!</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 33:24; غَلاطِيَّة 3:9–14; ﺭﻭﻣﻴﺔ 4:1–5"> أُنظُروا إلى إبراهيمَ أبيكُم.<br />وإلى سارةَ الّتي ولَدَتْكُم.<br />دَعاهُ الرّبُّ وهوَ رجُلٌ واحدٌ.<br />وبارَكَهُ وكَثَّرَ نَسلَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 49:13; يُوئيل 2:3; إشَعْياء 52:9"> الرّبُّ يُعزِّي صِهيَونَ.<br />يُعَزِّيها على كُلِّ خرَابِها<br />ويَجعَلُ قِفارَها كعَدنٍ<br />وصحراءَها كجَنَّةِ الرّبِّ.<br />فيَعودُ إليها الفرَحُ والسُّرورُ<br />وتَنطَلِقُ بالحَمدِ وصوتِ النَّشيدِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:9; إشَعْياء 2:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:21"> أصغُوا إليَّ يا شعبي.<br />يا أُمَّتي اَستَمِعوا لي:<br />الشَّريعةُ تَخرُجُ مِنْ عِندي.<br />وأحكامي نورٌ لِلشُّعوبِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="ﺭﻭﻣﻴﺔ 10:17–18; إشَعْياء 46:13; إشَعْياء 42:4"> عَدلي قريبٌ وخلاصي آتٍ<br />وذِراعايَ تَحكُمانِ الشُّعوبَ.<br />حتّى الجُزُرُ تَنتَظِرُني.<br />وعلى ذِراعَيَّ تعتَمِدُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 24:35; المَزامِير 102:25–26; إشَعْياء 34:4"> إرفَعوا إلى السَّماءِ عُيونَكُم<br />واَنظُروا إلى الأرضِ مِنْ تَحتُ:<br />السَّماواتُ كالدُّخانِ تضمَحِلُّ<br />والأرضُ كالثَّوبِ تَبلَى،<br />وسُكَّانُها كالبَعوضِ يَموتونَ.<br />أمَّا خلاصي فمدَى الدَّهرِ<br />وعَدلي لا يَسقُطُ أبدًا.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 5:11; المَزامِير 37:31; حِزْقِيال 2:6"> إسمَعوا يا مَنْ يَعرِفونَ العَدلَ،<br />أيُّها الّذينَ في قُلوبِهِم شريعتي:<br />لا تَخافوا تَعييرَ النَّاسِ<br />ومِنْ شَتائِمِهم لا تَرتَعِبوا.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 51:6; إشَعْياء 50:9; هُوشَع 5:12"> فَهُم كالثَّوبِ يأكُلُهمُ العُثُّ<br />وكالصُّوفِ يأكُلُهمُ السُّوسُ.<br />أمَّا عَدلي فيكونُ إلى الأبدِ<br />وخلاصي إلى مدَى الأجيالِ“.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 74:13–14; إشَعْياء 52:1; المَزامِير 89:10"> إستَفيقي يا ذِراعَ الرّبِّ،<br />إستَفيقي واَلبَسي الجَبروتَ<br />إستَفيقي كما في القديمِ،<br />كما في غابِرِ الأجيالِ.<br />أنتِ الّتي قَطَعتِ رهَبَ<br />وطَعَنتِ التِّنِّينَ طَعنًا.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 43:16; إشَعْياء 63:11–12; إشَعْياء 50:2"> وجَفَّفتِ مياهَ البحرِ،<br />مياهَ الغَمرِ العظيمِ،<br />فجَعَلتِ أعماقَهُ طريقًا<br />ليَعبُرَ فيهِ المُفتَدونَ،</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 25:8; رُؤيا يُوحَنّا 21:4; رُؤيا يُوحَنّا 7:17"> مُفتَدو الرّبِّ الرَّاجِعونَ<br />الآتونَ إلى صِهيَونَ مُرَنِّمينَ،<br />وعلى رُؤُوسِهِم فرَحٌ أبديٌّ.<br />يتبَعُهُمُ الفرَحُ والسُّرورُ<br />وتهرُبُ الحَسرَةُ والنُّواحُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 118:6; رسالَةُ بُطرُسَ الأُولَى 1:24; إشَعْياء 2:22"> وقالَ الرّبُّ:<br />”أنا الرّبُّ أرأفُ بكِ<br />فكيفَ أنتِ يا صِهيَونُ<br />تخافينَ مِنْ إنسانٍ يموتُ،<br />مِنْ بشَرٍ يزولُ كالعُشبِ،</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 45:12; إشَعْياء 40:22; المَزامِير 104:2"> وتَنسَينَ الرّبَّ الّذي صَنَعكِ<br />وبسَطَ السَّماواتِ وأسَّسَ الأرضَ.<br />تَفزَعينَ كُلَّ يومٍ بلا اَنقِطاعٍ<br />مِنْ غضَبِ الّذينَ يَظلِمونَكِ<br />كأنَّهُم يتَأهَّبونَ لتَدميرِكِ.<br />ولكِنْ أينَ غضَبُ الظَّالِمينَ؟</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 48:20; زَكَريّا 9:11; إشَعْياء 52:2"> عمَّا قريبٍ يُطلَقُ الأذِلَّاءُ،<br />فلا يموتونَ في الحبسِ مِنْ بَعدُ<br />ولا يَنقُصُ خبزُهُم أبدًا.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 31:35; إشَعْياء 54:5; عامُوس 9:5–6"> أنا هوَ الرّبُّ إلهُكِ!<br />أُثيرُ البحرَ فتَعجُّ أمواجُهُ،<br />والرّبُّ القديرُ اَسمي.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 59:21; إشَعْياء 49:2; يُوحَنّا 17:8"> علَّمتُكَ ماذا تتكلَّمُ،<br />وبظِلِّ يَدي سَتَرتُكَ،<br />أنا الّذي غرسَ السَّماواتِ<br />وأسَّسَ الأرضَ في مكانِها،<br />لتقولَ لِصهيَونَ: أنتِ شعبي“.</span>
</div>
</div>

## خاتمة آلام أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 52:1; رُؤيا يُوحَنّا 14:10; المَزامِير 60:3"> أَفيقي واَنهَضي يا أُورُشليمُ!<br />أَفيقي أيَّتُها الّتي شَرِبَت<br />مِنْ يدِ الرّبِّ كأسَ غضَبِهِ،<br />شرِبَت وجَرَعَت حتّى الثُّمالةَ</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إشَعْياء 49:21; المَزامِير 88:18; المَزامِير 142:4"> وما مِنْ أحدٍ يقودُها<br />مِنْ كُلِّ البَنينَ الّذينَ وَلَدْتَهُم،<br />ولا مَنْ يأخُذُ بيَدِها<br />مِنْ كُلِّ البنينَ الّذينَ رَبَّتْهُم.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="عامُوس 7:2; إشَعْياء 47:9; حِزْقِيال 14:21"> أصابَكِ اَثنانِ: البُؤسُ والدَّمارُ،<br />فمَنْ يا تُرَى يُشفِقُ علَيكِ؟<br />واَثنانِ: الجُوعُ والسَّيفُ،<br />فمَنْ يا تُرَى يُعزِّيكِ؟</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:11–12; رُؤيا يُوحَنّا 16:9–11; حِزْقِيال 39:19"> بنوكِ خاروا واَنبَطَحوا<br />في رأسِ كُلِّ شارعٍ،<br />كالغُزلانِ وقَعَت في الحَبائِلِ،<br />غَمَرهُم غضَبُ الرّبِّ<br />ونَقمَةُ الرّبِّ إلهِكِ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 29:9; إشَعْياء 54:11; إشَعْياء 51:17"> فاَسمَعي أيَّتُها البائِسةُ،<br />أيَّتُها السَّكْرى لا مِنَ الخمرِ،</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الأمثال 22:23; إشَعْياء 51:17; إرْمِيا 50:34"> ما يقولُ الرّبُّ سيِّدُكِ،<br />إلهُكِ المُنتَصِرُ لِشعبِهِ:<br />”ها أنا أخَذتُ مِنْ يَدِكِ<br />كأسَ ثُمالَةِ غضَبي،<br />فلا تَعودينَ تَشربينَهُ مِنْ بَعدُ،</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يَشُوع 10:24; زَكَريّا 12:2; إشَعْياء 49:25–26"> وجَعَلتُهُ في يَدِ مُعذِّبيكِ،<br />يَدِ الّذينَ قالوا لكِ:<br />إنحَني حتّى نعبُرَ.<br />فخَفَضْتِ كالأرضِ ظَهرَكِ،<br />وكالطَّريقِ يدوسُها العابِرونَ“.</span>
</div>
