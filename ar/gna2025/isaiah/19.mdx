## عقاب مصر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 12:12; رُؤيا يُوحَنّا 1:7; يَشُوع 2:11"> وحيٌ على مِصرَ: ها هوَ الرّبُّ راكِبٌ على سحابةٍ سريعةٍ وقادِمٌ إلى مِصْرَ. فتَرتَجِفُ أوثانُها مِنْ وجهِهِ، وتَذوبُ قُلوبُ المِصْريِّينَ في صُدورِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="القُضاة 7:22; صَمُوئيلَ الأوَّلُ 14:20; ﻣﺘﻰ 10:36"> ويُثيرُ الرّبُّ المِصْريِّينَ على المِصريِّينَ، فيُقاتِلُ الإنسانُ أخاهُ والرَّجلُ صديقَهُ، وتقومُ مدينةٌ على مدينةٍ ومملَكةٌ على مملكةٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 8:19; دانيال 2:2; أخبارِ الأيّامِ الأوّلُ 10:13"> ويُفرِغُ مِصْرَ مِنَ الرُّوحِ في داخِلِها ويُبَلبِلُ أفكارَهُم، فيَستشيرونَ الأوثانَ والسَّحرةَ والمُشَعوذينَ والعَرَّافينَ.</span>
<span className="v" id="v4" data-v="4" data-xref="إشَعْياء 20:4; إرْمِيا 46:26; حِزْقِيال 29:19"> ويُسلِّمُ المِصْريِّينَ إلى يَدِ سيِّدٍ قاسٍ ومَلكٍ طاغيةٍ يتسَلَّطُ علَيهِم. هكذا قالَ السيِّدُ الرّبُّ القديرُ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="إرْمِيا 51:36; حِزْقِيال 30:12; زَكَريّا 14:18"> وينشَفُ البحرُ مِنَ المياهِ ويجِفُّ النَّهرُ ويَيبَسُ.</span>
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 37:25; الخُرُوج 7:18; المُلُوكِ الثّانِي 19:24"> وتُنتِنُ الأنهارُ لقلَّةِ مائِها وتجِفُّ سَواقي مِصْرَ، فيَتلَفُ القَصَبُ والبَرديُّ</span>
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 32:20; يُوئيل 1:17–18; إشَعْياء 23:3"> ويتَبَدَّدُ كُلُّ نَباتٍ على ضِفافِ النِّيلِ ويَيبَسُ ولا يَبقى شيءٌ.</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 47:10; حَبَقُّوق 1:15; العَدَد 11:5"> فيَنتَحِبُ الصَّيَّادونَ ويَنوحُ كُلُّ مَنْ يُلقي الصِّنَّارةَ في النِّيلِ، ويتَحسَّرُ الّذينَ يمُدُّونَ الشَّبكةَ على المياهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 27:7; الأمثال 7:16; المُلُوكِ الأوَّلُ 10:28"> ويَبتَئسُ صُنَّاعُ الكتَّانِ المُمشَّطِ وحائِكو الأنسِجةِ البيضاءِ</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 11:10; الخُرُوج 7:19; الخُرُوج 8:5"> وتَنهارُ أعمِدَةُ الصِّناعةِ، وتكتَئِبُ نفوسُ العامِلينَ بالأُجرةِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="العَدَد 13:22; اعمال الرسل 7:22; المُلُوكِ الأوَّلُ 4:30"> ما أغبَى أُمراءَ صُوعَنَ، وما أسخَفَ مَشورةَ حُكماءِ فِرعَونَ! كيفَ يجرُؤونَ أنْ يقولوا لفِرعَونَ: ”نحنُ بنو الحُكماءِ وبَنو المُلوكِ الأقدَمينَ!“</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:20; إشَعْياء 14:24; القُضاة 9:38"> فأينَ حُكماؤُكَ يا فِرعَونُ؟ دَعهُم يُخبِرونَكَ ويكشِفونَ لكَ بماذا قضَى الرّبُّ القديرُ على مِصْرَ.</span>
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 30:13; إرْمِيا 2:16; إرْمِيا 46:14"> أُمراءُ صوعَنَ أغبياءُ، وأُمراءُ نوفَ مخدوعونَ، ومِصْرُ يُضَلِّلُها زُعماءُ شعبِها.</span>
<span className="v" id="v14" data-v="14" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:11; المُلُوكِ الأوَّلُ 22:20–23; حِزْقِيال 14:7–9"> مزجَ الرّبُّ في داخِلِهم روحَ دُوارٍ، فجَعَلوا مِصْرَ تتَرنَّحُ في كُلِّ أعمالِها كما يترَنَّحُ السَّكرانُ في قَيئِهِ.</span>
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 9:14–15; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:11–12; الأمثال 14:23"> فلا يبقَى لمِصْرَ مَنْ يقدِرُ أنْ يعمَلَ شيئًا، سواءٌ في ذلِكَ الرَّأسُ أوِ الذَّنبُ، والرَّفيعُ أوِ الوضيعُ.</span>
</div>
</div>

## إهتداء مصر وأشور

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إرْمِيا 51:30; ناحُوم 3:13; إشَعْياء 11:15"> في ذلِكَ اليومِ يكونُ المِصْريُّونَ كالنِّساءِ، فيَرتَعدونَ ويَرتجِفونَ مِنْ يَدِ الرّبِّ المَرفوعةِ علَيهِم.</span>
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 14:24; دانيال 4:35; حِزْقِيال 29:6–7"> وتكونُ أرضُ يَهوذا مَصدرَ رُعبٍ لمِصْرَ، كُلَّما تَذكَّرتِ المصيرَ الّذي قضَى بهِ علَيها الرّبُّ القديرُ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="صَفَنْيا 3:9; زَكَريّا 2:11; إشَعْياء 19:21"> وفي ذلِكَ اليومِ يكونُ في أرضِ مِصْرَ خمسُ مُدُنٍ تتكلَّمُ بلُغةِ كنعانَ وتَحلِفُ باَسمِ الرّبِّ القديرِ، يُقالُ لإحداها مدينةُ الشَّمسِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 28:18; يَشُوع 22:10; يَشُوع 22:26"> وفي ذلِكَ اليومِ يكونُ مذبَحٌ للرّبِّ في داخِلِ أرضِ مِصْرَ، ونُصبٌ مرفوعٌ للرّبِّ قُربَ حُدودِها عَلامةً وشهادةً للرّبِّ القديرِ في أرضِ مِصْرَ.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 50:15; تِيطُس 2:13; إشَعْياء 45:21–22"> فإذا ما صرَخَ المِصْريُّونَ إلى الرّبِّ في ضيقِهِم، أرسَلَ لهُم مُخلِّصًا ومُحاميًا فيُنقِذُهُم.</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 11:9; مَلاخِي 1:11; يُوحَنّا 17:3"> ويُعلِنُ الرّبُّ عَنْ نفسِهِ لهُم، فيَعرفونَ الرّبَّ في ذلِكَ اليومِ، ويَعبُدونَهُ بالذَّبيحةِ والتَّقدمةِ، ويَنذُرونَ لَه نُذورًا ويُوفونَ بها.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 32:39; العِبرانِيّين 12:11; هُوشَع 14:1"> ومعَ أنَّ الرّبَّ ضربَهُم بقساوةٍ، فإنَّهُ يَشفيهِم حينَ يَرجِعونَ إليهِ ويَستجيبُ لهُم.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 11:16; أفَسُس 2:18–22; إشَعْياء 35:8–10"> وفي ذلِكَ اليومِ يكونُ طريقٌ مِنْ مِصرَ إلى أشُّورَ، فتَجيءُ أشُّورُ إلى مِصْرَ ومِصْرُ إلى أشُّورَ، تعبُدُ مِصْرُ الرّبَّ معَ أشُّورَ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 65:8; التَّكوين 12:2; غَلاطِيَّة 3:14"> وفي ذلِكَ اليومِ تكونُ إِسرائيلُ ثالِثًا لمِصْرَ وأشُّورَ، وهذا بَركةٌ في وسَطِ الأرضِ.</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 29:23; هُوشَع 2:23; ﺭﻭﻣﻴﺔ 9:24–25"> ويمنَحُ الرّبُّ القديرُ بَركتَهُ قائِلًا: ”مُبارَكٌ شعبي مِصْرُ، وصَنعَةُ يَدي أشُّورُ وبنو إِسرائيلَ الّذينَ اَختَرتُهُم“.</span>
</div>
