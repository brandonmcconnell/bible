## حِزْقيا يستشير إشعيا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المُلُوكِ الثّانِي 19:1–37; عَزْرا 9:5; إشَعْياء 36:22–38"> فلمَّا سمِعَ الملِكُ حِزْقيَّا، مزَّقَ ثيابَهُ ولبِسَ مِسْحًا ودخَلَ بَيتَ الرّبِّ</span>
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 1:1; المُلُوكِ الثّانِي 18:18; إشَعْياء 36:3"> وأرسَلَ ألياقيمَ رئيسَ الدِّيوانِ المَلَكيِّ، وشَبْنَةَ أمينَ السِّرِّ، وشُيوخَ الكهَنةِ لابِسينَ المُسوحَ، إلى إشَعيا النَّبيِّ اَبنِ آموصَ.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 66:9; هُوشَع 13:13; إشَعْياء 26:17–18"> فقالوا لَه: يقولُ لكَ حِزْقيَّا: اليومُ يومُ الضِّيقِ والتَّأديبِ والمَهانةِ، فيَحينُ مولِدُ الجَنينِ ولا قُدرَةَ على الوِلادةِ.</span>
<span className="v" id="v4" data-v="4" data-xref="إشَعْياء 1:9; إشَعْياء 36:20; إشَعْياء 10:22"> لَيتَ الرّبَّ إلهَكَ يسمَعُ كلامَ رَبشاقا، رسولِ سيِّدِهِ ملِكِ أشُّورَ، فيُعاقِبَهُ على ما جاءَ في كَلامِهِ مِنْ تعييرٍ للإلهِ الحيِّ. فأقِمْ صلاةً مِنْ أجلِ البقيَّةِ الباقيةِ مِنَ الشَّعبِ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5"> فلمَّا سمِعَ إشَعيا ما نَقَلهُ إليهِ رجالُ المَلِكِ حِزْقيَّا،</span>
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 7:4; إشَعْياء 35:4; مَرْقُس 5:36"> قالَ لهُم: ”قُولوا لسيِّدِكُم: يوصيكَ الرّبُّ إلهُنا: لا تَخَفْ مِمَّا سَمِعتَهُ مِنْ تَجديفٍ عليَّ مِنْ صَعاليكِ ملِكِ أشُّورَ.</span>
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 31:8–9; أخبارِ الأيّامِ الثّانِي 32:21; إشَعْياء 33:10–12"> سأجعَلُهُ يسمَعُ خبَرًا، فيَرجِعُ إلى بلادِهِ حيثُ أُسقِطُهُ بالسَّيفِ“.</span>
</div>
</div>

## أشور يهدد مرة ثانية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="يَشُوع 10:29; العَدَد 33:20–21; المُلُوكِ الثّانِي 19:8–9"> ورجَعَ رَبشاقا، فسَمِعَ أنَّ مَلِكَ أشُّورَ رحَلَ مِنْ لخيشَ لمُحارَبةِ مدينةِ لِبنَةَ فذهَبَ إليهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 18:1; إشَعْياء 20:5; إشَعْياء 37:7"> ثُمَّ قيلَ للمَلِكِ إنَّ تِرهاقةَ ملِكَ كوشَ خرَجَ لمُقاتَلَتِهِ فأرسَلَ وفدًا إلى حِزْقيَّا يقولُ:</span>
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 36:15; المُلُوكِ الثّانِي 18:5; ﻣﺘﻰ 27:43"> ”لا تدَعْ إلهَكَ الّذي تتَّكِلُ علَيهِ يخدَعُكَ بقولِهِ لكَ: لنْ تسقُطَ أورُشليمُ في يَدِ مَلِكِ أشُّورَ.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 36:18–20; المُلُوكِ الثّانِي 18:33–35; إشَعْياء 14:17"> فأنتَ ولا شَكَّ سَمِعتَ بما فعَلَ مُلوكُ أشُّورَ بجميعِ البُلدانِ وكيفَ دَمَّروها، فهل تنجو أنتَ؟</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 11:31; المُلُوكِ الثّانِي 17:6; المُلُوكِ الثّانِي 18:11"> والأمَمُ الّتي أهلكَها آبائي هل أنقَذَتْها آلِهَتُها، كجوزانَ وحارانَ وراصَفَ وأبناءِ عَدنَ الّذينَ في تَلسارَ؟</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الثّانِي 19:13; المُلُوكِ الثّانِي 17:24; إشَعْياء 36:19"> أينَ مُلوكُ حماةَ وأرفادَ ومدينةِ سَفروايِمَ وهَينَعَ وعُوَّةَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الثّانِي 19:14; المَزامِير 123:1–4; يُوئيل 2:17–20"> فأخَذَ حِزْقيَّا الرَّسائِلَ مِنْ يَدِ رُسُلِ مَلِكِ أشُّورَ وقَرَأها وصَعِدَ إلى بَيتِ الرّبِّ وفتَحَ الرَّسائِلَ أمامَ الرّبِّ،</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 7:8–9; دانيال 9:3–4; يَعقُوب 5:13"> وصلَّى فقالَ:</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 25:22; المَزامِير 99:1; المَزامِير 86:10"> ”يا ربَّنا القديرَ إلهَ إسرائيلَ الجالِسَ على الكروبيمِ! أنتَ وحدَكَ إلهُ ممالِكِ الأرضِ. أنتَ صَنَعتَ السَّماواتِ والأرضَ.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الثّانِي 6:40; المَزامِير 17:6; إشَعْياء 37:4"> أمِلْ أذُنَيكَ يا ربُّ واَسمَعْ. إفتَحْ عينَيكَ واَنظُرْ واَستَمِعْ إلى أقوالِ سنحاريبَ الّتي أرسَلَها ليُهينَكَ بها أنتَ اللهُ الحيُّ.</span>
<span className="v" id="v18" data-v="18" data-xref="ناحُوم 2:11–12; المُلُوكِ الثّانِي 15:29; المُلُوكِ الثّانِي 17:6"> نعم يا ربُّ إنَّ مُلوكَ أشُّورَ خَرَّبوا جميعَ الأمَمِ وبُلدانَها</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 44:9–10; إرْمِيا 10:11; هُوشَع 8:6"> وألقَوا آلِهتَها في النَّارِ. فما هيَ بآلِهةٍ، بل خشَبٌ وحجارةٌ صَنَعَتْها أيدي النَّاسِ.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 46:10; المُلُوكِ الأوَّلُ 18:36–37; المَزامِير 83:17–18"> والآنَ أيُّها الرّبُّ إلهُنا، خَلِّصْنا مِنْ يَدَيهِ لتَعلَمَ مَمالِكُ الأرضِ كُلِّها أنَّكَ أنتَ الرّبُّ وحدَكَ“.</span>
</div>
</div>

## رسالة إشعيا إلى حِزْقيا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 58:9; إشَعْياء 38:3–6; اعمال الرسل 4:31"> فأرسَلَ إشَعيا بنِ آموصَ إلى حِزْقيَّا المَلِكِ يقولُ لهُ: ”إِستَجابَ الرّبُّ إلهُ إسرائيلَ لِصَلاتِكَ“،</span>
<span className="v" id="v22" data-v="22" data-xref="أيُّوب 16:4; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:13; إشَعْياء 1:8"> وهذا ما قالَهُ على سنحاريبَ مَلِكِ أشُّورَ: ”تَزدَريكَ وتسخَرُ مِنكَ البِكرُ اَبنَةُ صِهيَونَ، تَميلُ برَأسِها عَنكَ اَبنَةُ أورُشليمَ.</span>
<span className="v" id="v23" data-v="23" data-xref="حِزْقِيال 39:7; دانيال 5:20–23; إشَعْياء 17:7"> مَنْ عَيَّرْتَ وعلى مَنْ جَدَّفتَ؟ على مَنْ رفَعتَ صوتَكَ وإلى مَنْ تَطَلَّعتَ شامِخًا بعَينَيكَ؟ أنا قُدُّوسُ إسرائيلَ.</span>
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 14:8; إشَعْياء 10:18; الخُرُوج 15:9"> أنا الرّبُّ الّذي عَيَّرتَهُ على لِسانِ رِجالِكَ وقلتَ مُتَفاخرًا: بكلِّ مَراكبي صَعِدتُ رُؤوسَ جِبالِ لبنانَ، وقطَعتُ أطوَلَ أرزِهِ، وأفضَلَ سَرْوِهِ، ووَصَلتُ إلى أقصى أعاليهِ وإلى مَجاهِلِ غابِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="المُلُوكِ الأوَّلُ 20:10; إشَعْياء 36:12; المُلُوكِ الثّانِي 19:23–24"> وحَفَرتُ آبارًا في ديارٍ غريبةٍ، وجَفَّفتُ ببَطنِ قدَمي جميعَ أنهارِ مِصْرَ.</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 4:27–28; اعمال الرسل 2:23; رسالَةُ بُطرُسَ الأُولَى 2:8"> والآنَ فاَسمَعْ: أنا هوَ الّذي تصوَّرَ هذا كُلَّهُ ودبَّرَهُ مِنْ قديمِ الزَّمانِ وأنجَزَهُ على يَدِكَ في الزَّمنِ الحاضِرِ لهَدْمِ المُدُنِ المُحصَّنَةِ حتّى تصيرَ تِلالًا مِنَ الحجارةِ</span>
<span className="v" id="v27" data-v="27" data-xref="المَزامِير 129:6; المَزامِير 37:2; رسالَةُ بُطرُسَ الأُولَى 1:24"> فيما سُكّانُها أيديهِم قصيرةٌ، عاجِزونَ مُخزَونَ كعُشبِ الحقلِ وكالنَّباتِ الأخضَرِ وحشيشِ السُّطوحِ الّذي يلفَحُهُ الرِّيحُ قبلَ نُمُوِّه.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="رُؤيا يُوحَنّا 2:13; المَزامِير 139:1–11; إرْمِيا 23:23–24"> ”ولكنِّي عالِمٌ بقيامِكَ وقُعودِكَ، وخُروجِكَ ودُخولِكَ، وهَيجانِكَ عليَّ</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 38:4; إشَعْياء 30:28; إشَعْياء 37:34"> وعَجرَفَتِكَ الّتي وصلَ خبَرُها عاليًا إلى أذُنَيَّ. ولذلِكَ أضَعُ حَلقَةً في أنفِكَ ولِجامًا بَينَ شَفَتَيكَ، وأرُدُّكَ في الطَّريقِ الّتي جِئتَ مِنها“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الثّانِي 19:29; اللّاويِّين 25:4–5; المُلُوكِ الأوَّلُ 13:3–5"> ثُمَّ قالَ إشَعيا لحِزْقيَّا: ”وهذِهِ عَلامةٌ لكَ: تأكُلُ في هذِهِ السَّنةِ حِنطَةً برِّيَّةً، وفي السَّنةِ الثَّانيةِ ما يخلُفُ مِنها، وفي السَّنةِ الثَّالثةِ تَزرَعونَ وتَحصُدونَ وتَغرِسونَ كُرومًا وتأكلونَ ثِمارَها.</span>
<span className="v" id="v31" data-v="31" data-xref="إشَعْياء 27:6; المُلُوكِ الثّانِي 19:30–31; المَزامِير 80:9"> ويعودُ النَّاجونَ مِنْ ذُرِّيَّةِ يَهوذا يَغرِسونَ جُذورَهُم في الأرضِ ويُخرِجونَ ثمَرَهُم مِنْ فَوقُ.</span>
<span className="v" id="v32" data-v="32" data-xref="إشَعْياء 9:7; المُلُوكِ الثّانِي 19:31; إشَعْياء 59:17"> فمِنْ أورُشليمَ تخرُجُ البَقيَّةُ ومِنْ جبَلِ صِهيَونَ يخرُجُ النَّاجونَ. غيرَةُ الرّبِّ القديرِ تفعَلُ هذا.</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 17:12; إشَعْياء 10:32–34; إشَعْياء 17:14"> لذلِكَ يقولُ الرّبُّ على مَلِكِ أشُّورَ: لن يدخُلَ هذِهِ المدينةَ، ولا يرمي إليها سَهمًا، ولا يتقدَّمُ علَيها بتُرسٍ، ولا ينصِبُ علَيها مِترَسةً.</span>
<span className="v" id="v34" data-v="34" data-xref="إشَعْياء 37:29; الأمثال 21:30"> لكنْ في الطَّريقِ الّتي جاءَ مِنها يرجِعُ، وإلى هذِهِ المدينةِ لا يدخُلُ.</span>
<span className="v" id="v35" data-v="35" data-xref="المُلُوكِ الثّانِي 20:6; إشَعْياء 38:6; إشَعْياء 31:5"> فأحمي هذِهِ المدينةِ وأُخلِّصُها مِنْ أجلي ومِنْ أجلِ داوُدَ عبدي“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="إشَعْياء 10:12; المُلُوكِ الثّانِي 19:35; إشَعْياء 10:33–34"> وخرَجَ ملاكُ الرّبِّ وقتَلَ مِنْ جيشِ أشُّورَ مئَةَ ألفٍ وخمسةً وثمانينَ ألفًا. فلمَّا طلَعَ الصَّباحُ كانوا جميعًا جُثَثًا هامِدةً.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻳﻮﻧﺎﻥ 1:2; ﻳﻮﻧﺎﻥ 3:3; التَّكوين 10:11–12"> فاَنصرَفَ سنحاريبُ ملِكُ أشُّورَ راجِعًا إلى عاصِمَتِهِ نينوى.</span>
<span className="v" id="v38" data-v="38" data-xref="التَّكوين 8:4; إرْمِيا 51:27; عَزْرا 4:2"> وفيما هوَ ساجدٌ في معبَدِ نِسروخَ إلهِهِ، قتَلَهُ أدرَمَّلِكُ وشَرآصَرُ اَبناهُ بالسَّيفِ وهرَبا إلى أرضِ أراراطَ. وملَكَ آسَرحَدُّونُ اَبنُهُ مكانَهُ.</span>
</div>
