## الشيطان يجرب أيّوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يَعقُوب 5:11; حِزْقِيال 14:14; حِزْقِيال 14:20"> كانَ في أرضِ عوصَ رَجلٌ اَسمُهُ أيُّوبُ، وكانَ هذا الرَّجلُ نَزيهًا مُستقيمًا يَخافُ اللهَ ويَحيدُ عَنِ الشَّرِّ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 127:3–5; أيُّوب 42:13; المَزامِير 128:3"> ووُلِدَ لأيُّوبَ سبعةُ بَنينَ وثَلاثُ بَناتٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="أيُّوب 29:25; أيُّوب 42:12; القُضاة 6:3"> وكانَت مَواشيهِ سبعةَ آلافٍ مِنَ الغنَمِ، وثَلاثةَ آلافِ جمَلٍ، وخمْسَ مئَةِ فَدَّانِ بقَرٍ، وخَمْسَ مئَةِ أتانٍ، ولَه عبيدٌ كثيرونَ. فكانَ ذلِكَ الرَّجلُ أعظمَ أبناءِ المشرِق جميعًا.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="العِبرانِيّين 13:1; المَزامِير 133:1"> وكانَ بَنوهُ يَذهبونَ ويُقيمونَ وليمةً في بَيتِ كُلِّ واحدٍ مِنهُم بِدَورِهِ، ويَستَدعونَ أخواتِهِم الثَّلاثَ ليَأكُلْنَ ويَشرَبْنَ معَهُم.</span>
<span className="v" id="v5" data-v="5" data-xref="أيُّوب 42:8; التَّكوين 8:20; صَمُوئيلَ الأوَّلُ 16:5"> فإذا تمَّت أيّامُ كُلِّ وليمَةٍ كانَ أيّوبُ يُبَكِّرُ في الغدِ فيُصعِدُ مُحرقاتٍ عَنْ كُلِّ واحدٍ مِنهُم فيُطَهِّرُهُم. وكانَ يقولُ: ”لعَلَّ بَنيَّ خطِئوا وجَدَّفوا على اللهِ في قُلوبِهِم“. هكذا كانَ أيُّوبُ يفعَلُ كُلَّ الأيّامِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="أيُّوب 2:1; أيُّوب 38:7; زَكَريّا 3:1"> وجاءَ الملائِكةُ يومًا للمُثولِ أمامَ الرّبِّ، وجاءَ الشَّيطانُ أيضًا بَينَهُم.</span>
<span className="v" id="v7" data-v="7" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:8; رُؤيا يُوحَنّا 12:12–17; ﻣﺘﻰ 12:43"> فقالَ الرّبُّ لِلشَّيطانِ: ”مِنْ أينَ جِئْتَ؟“ فأجابَ الشَّيطانُ: ”مِنَ التَّجَوُّلِ في الأرضِ والسَّيرِ فيها“.</span>
<span className="v" id="v8" data-v="8" data-xref="أيُّوب 2:3; أيُّوب 1:1; العَدَد 12:7–8"> فقالَ لَه الرّبُّ: ”هل اَستَرعى اَنتِباهَكَ عبدي أيُّوبُ؟ فهوَ لا مَثيلَ لَهُ في الأرضِ لأنَّه رَجلٌ نَزيهٌ مُستَقيمٌ يخافُ اللهَ ويَحيدُ عَنِ الشّرِّ“.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 16:26; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 4:8; أيُّوب 2:10"> فأجابَ الشَّيطانُ: ”أيخافُ أيُّوبُ الرّبَّ مَجَّانًا؟</span>
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 34:7; زَكَريّا 2:5; رسالَةُ بُطرُسَ الأُولَى 1:5"> أما سيَّجْتَ حَولَهُ وحَولَ بَيتِهِ وحَولَ كُلِّ شيءٍ لَه مِنْ كُلِّ جِهَةٍ؟ أما بارَكْتَ أعمالَهُ فاَنتَشَرَت مَواشيهِ في الأرضِ؟</span>
<span className="v" id="v11" data-v="11" data-xref="أيُّوب 2:5; أيُّوب 19:21; رُؤيا يُوحَنّا 16:11"> ولكِنْ مُدَّ يدَكَ الآنَ ومُسَّ كُلَّ شيءٍ لَه، فتَرى كَيفَ يُجَدِّفُ علَيكَ في وجهِكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="لُوقا 22:31–32; يُوحَنّا 3:35–36; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:7"> فقالَ الرّبُّ لِلشَّيطانِ: ”ها أنا أجعَلُ كُلَّ شيءٍ لَه في قَبضَةِ يدِكَ، ولكِنْ إليهِ لا تمُدَّ يدَكَ“. وخرَجَ الشَّيطانُ مِنْ أمامِ وجهِ الرّبِّ.</span>
</div>
</div>

## أيُّوب يخسر كل شيء له

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="لُوقا 21:34; الأمثال 27:1; أيُّوب 1:4"> وحدَثَ يومًا أنَّ بني أيُّوبَ وبَناتِهِ كانوا يأكلونَ ويشرَبونَ خمرًا في بَيتِ أخيهِمِ الأكبرِ،</span>
<span className="v" id="v14" data-v="14" data-xref="صموئيلَ الثّانِي 15:13; إرْمِيا 51:31; صَمُوئيلَ الأوَّلُ 4:17"> فأقبَلَ رَسولٌ إلى أيُّوبَ وقالَ: ”كانَتِ البقَرُ تحرُثُ والحميرُ تَرعى بِجانِبِها،</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 10:7; أيُّوب 6:19; إشَعْياء 45:14"> فاَنقَضَّ علَيها بَنو سَبأٍ وأخَذوها وقتَلوا الرُّعيانَ بِحدِّ السَّيفِ، ونَجوتُ أنا وَحدي لأُخبِرَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الثّانِي 1:12; التَّكوين 19:24; المُلُوكِ الأوَّلُ 18:38"> وفيما هوَ يتكلَّمُ، أقبَلَ آخَرُ وقالَ: ”سقَطَت نارُ اللهِ مِنَ السَّماءِ وأحرقَتِ الغنَمَ والرُّعيانَ وأكلَتهُم، ونَجَوتُ أنا وَحدي لأُخبِرَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 11:28; حَبَقُّوق 1:6; التَّكوين 11:31"> وفيما هوَ يتكلَّمُ، أقبلَ آخَرُ وقالَ: ”هجَمَت ثَلاثُ عِصاباتٍ مِنَ الكَلدانيِّينَ على الجِمالِ وأخَذوها وقتَلوا الرُّعيانَ بِحَدِّ السَّيفِ، ونَجَوتُ أنا وحدي لأُخبِرَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="أيُّوب 1:4; أيُّوب 1:13; أيُّوب 27:14"> وفيما هوَ يتكلَّمُ، أقبلَ آخَرُ وقالَ: ”كانَ بَنوكَ وبَناتُكَ يأكُلونَ ويشرَبونَ خمرًا في بَيتِ أخيهِمِ الأكبرِ،</span>
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 42:36; لُوقا 13:1–5; القُضاة 16:30"> فهَبَّت ريحٌ شديدةٌ طلَعَت مِنْ عَبرِ القَفْرِ وصدمَت زَوايا البَيتِ الأربَعِ، فسقَطَت علَيهِم فماتوا، ونَجَوتُ أنا وحدي لأُخبِرَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:6; صموئيلَ الثّانِي 12:16–20; التَّكوين 37:34"> فقامَ أيُّوبُ وشَقَّ ثوبَهُ وجَزَّ شَعرَ رأسِهِ ووقَعَ على الأرضِ ساجدًا</span>
<span className="v" id="v21" data-v="21" data-xref="الجامِعَة 5:15; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:7; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:18"> وقالَ: ”عُريانًا خرَجْتُ مِنْ بَطنِ أُمِّي وعُريانًا أعودُ إلى هُناكَ الرّبُّ أعطى والرّبُّ أخذَ تبارَكَ اَسمُ الرّبِّ“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يَعقُوب 1:12; أيُّوب 2:10; رسالَةُ بُطرُسَ الأُولَى 1:7"> وما خَطِئَ أيُّوبُ معَ هذا كُلِّهِ، ولا عَتِبَ على اللهِ.</span>
</div>
</div>
