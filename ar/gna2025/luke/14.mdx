## يسوع يشفي مريضًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 3:2; لُوقا 20:20; المَزامِير 62:4"> ودخَلَ يومَ السَّبتِ بَيتَ أحدِ كِبارِ الفَرِّيسيِّينَ ليتَناوَلَ الطَّعامَ، وكانوا يُراقِبونَهُ.</span>
<span className="v" id="v2" data-v="2"> وإذا أمامَهُ رَجُلٌ تَورَّمَ جِلدُهُ بالاستسقاءِ،</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 12:10; مَرْقُس 3:4; لُوقا 13:14–16"> فقَالَ يَسوعُ لِعُلماءِ الشَّريعَةِ والفَرِّيسيِّينَ: ”أيَحِلُّ الشِّفاءُ في السَّبتِ أم لا؟“</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 22:46; ﻣﺘﻰ 21:25–27"> فسكَتوا. فأخَذَ يَسوعُ الرَّجُلَ بِيَدِهِ وشَفاهُ وصرَفَهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="لُوقا 13:15; الخُرُوج 23:4–5; ﻣﺘﻰ 12:11–12"> ثُمَّ قالَ لهُم: ”مَنْ مِنكُم يقَعُ اَبنُهُ أو ثَورُهُ في بِئرٍ يومَ السَّبتِ ولا يَنشُلُهُ مِنها في الحالِ؟“</span>
<span className="v" id="v6" data-v="6" data-xref="لُوقا 20:40; لُوقا 21:15; اعمال الرسل 6:10"> فما قَدِروا أن يُجاوِبوهُ.</span>
</div>
</div>

## الضيافة والتواضع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="فِيلِبِّي 2:3; لُوقا 11:43; ﻣﺘﻰ 23:6"> ولاحَظَ أنَّ بَعضَ المدعُوِّينَ يَختارونَ المَقاعِدَ الأُولى، فقَالَ هذا المَثَلَ:</span>
<span className="v" id="v8" data-v="8" data-xref="الأمثال 25:6–7"> ”إذا دَعاكَ أحَدٌ إلى وَليمةِ عُرسٍ، فلا تَجلِسْ في المَقعَدِ الأوّلِ. فلرُبَّما كانَ في المَدعُوِّينَ مَنْ هوَ أهَمُّ مِنكَ،</span>
<span className="v" id="v9" data-v="9" data-xref="الأمثال 16:18; الأمثال 11:2; الأمثال 3:35"> فيَجيءُ الّذي دَعاكُما ويَقولُ لكَ: أعطِهِ مكانَكَ! فتَخجَلَ، وتَقومَ إلى آخِرِ مَقعَدٍ.</span>
<span className="v" id="v10" data-v="10" data-xref="الأمثال 25:6–7; الأمثال 15:33; صَمُوئيلَ الأوَّلُ 15:17"> ولكنْ إذا دُعيتَ فاَجلِسْ في آخِرِ مَقعَدٍ، حتّى إذا جاءَ صاحِبُ الدَّعوةِ قالَ لكَ: قُم إلى فَوقُ، يا صَديقي. فيكبُرُ قَدرُكَ في نَظَرِ جميعِ المَدعُوِّينَ،</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 18:14; ﻣﺘﻰ 23:12; الأمثال 29:23"> لأنَّ مَنْ يَرفَعُ نَفسَهُ يَنخَفِضُ، ومَنْ يَخفِضُ نَفسَهُ يَرتَفِعُ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 5:46; يَعقُوب 2:1–6; لُوقا 6:32–36"> وقالَ يَسوعُ لِصاحِبِ الدَّعوَةِ: ”إذا أقَمتَ وَليمَةَ غَداءٍ أو عَشاءٍ، فلا تَدْعُ إلَيها أصدِقاءَكَ ولا إخوانَكَ ولا أقرباءَكَ ولا جيرانَكَ الأغنياءَ، لِئَلَّا يُبادِلوكَ الدَّعوةَ، فتَنالَ المُكافأةَ على عَمَلِكَ.</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 14:21; إشَعْياء 58:10; الأمثال 14:31"> بل إذا أقَمتَ وَليمةً، فاَدْعُ الفُقَراءَ والمُشَوَّهينَ والعُرْجَ والعُميانَ.</span>
<span className="v" id="v14" data-v="14" data-xref="الأمثال 19:17; اعمال الرسل 24:15; يُوحَنّا 5:29"> وهَنيئًا لكَ إذا فَعَلتَ لأنَّهُم لا يَقدِرونَ أنْ يُكافِئوكَ، فتُكافَأَ في قيامَةِ الأبرارِ“.</span>
</div>
</div>

## مثل الوليمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="رُؤيا يُوحَنّا 19:9; لُوقا 22:30; لُوقا 13:29"> فلمَّا سمِعَ أحَدُ المَدعُوِّينَ هذا الكلامَ قالَ لِيسوعَ: ”هَنيئًا لِمَن يَجلِسُ إلى المائِدَةِ في مَلكوتِ اللهِ!“</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 22:2–14; رُؤيا يُوحَنّا 22:17; إشَعْياء 55:1–7"> فأجابَهُ: ”أقامَ رَجُلٌ وَليمةً كبيرةً، ودَعا إلَيها كثيرًا مِنَ النَّاسِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 11:27–29; ﻣﺘﻰ 10:1–4; ﻣﺘﻰ 22:3–4"> ثُمَّ أرسَلَ خادِمَهُ ساعَةَ الوَليمةِ يَقولُ لِلمدعُوِّينَ: تَعالَوا، فكُلُّ شيءٍ مُهيَّأٌ!</span>
<span className="v" id="v18" data-v="18" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9–10; لُوقا 17:26–31; لُوقا 18:24"> فاَعتذَرُوا كُلُّهُم. قالَ لَهُ الأوَّلُ: إِشتَريَتُ حَقلًا ويَجِبُ أنْ أذهَبَ لأراهُ، أرجو مِنكَ أنْ تَعذُرَني.</span>
<span className="v" id="v19" data-v="19"> وقالَ آخَرُ: إِشتَرَيتُ خَمسةَ فَدادينَ، وأنا الآنَ ذاهِبٌ لأُجرِّبَها، أرجو مِنكَ أنْ تَعذُرَني.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 24:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:33; لُوقا 14:26–28"> وقالَ آخرُ: تَزوَّجْتُ اَمرأةً، فلا أقدِرُ أنْ أجيءَ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="لُوقا 14:13; ﻣﺘﻰ 11:5; لُوقا 14:24"> فرجَعَ الخادِمُ إلى سَيِّدهِ وأخبَرَهُ بِما جَرى، فغَضِبَ رَبُّ البَيتِ وقالَ لِخادِمِه: أُخرُجْ مُسرِعًا إلى شَوارِعِ المدينةِ وأزِقَّتِها وأدخِلِ الفُقَراءَ والمُشوَّهينَ والعُرجَ والعُميانَ إلى هُنا.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 14:2; أفَسُس 3:8; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:5–6"> فقَالَ الخادِمُ: جَرى ما أمَرْتَ بِه يا سَيِّدي، وبَقِيَت مَقاعِدُ فارِغَةٌ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 22:9–10; كُولُوسِي 1:28; اعمال الرسل 13:47–48"> فأجابَهُ السيِّدُ: أُخرُجْ إلى الطُّرُقاتِ والدُّروبِ وألزِمِ النَّاسَ بالدُّخولِ حتّى يَمتَلئَ بَيتي.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 21:43; اعمال الرسل 13:46; يُوحَنّا 8:24"> أقولُ لكُم: لَن يَذوقَ عَشائي أحَدٌ مِنْ أولئِكَ المَدعُوِّينَ!“</span>
</div>
</div>

## المطلوب من أتباع يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 6:24–27; لُوقا 12:1"> وكانَت جُموعٌ كَبيرةٌ تُرافِقُ يَسوعَ، فاَلتَفتَ وقالَ لهُم:</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 10:37; يُوحَنّا 12:25; فِيلِبِّي 3:8"> ”مَنْ جاءَ إليَّ وما أحبَّني أكثرَ مِنْ حُبِّهِ لأبيهِ وأُمِّهِ واَمرأتِهِ وأولادِهِ وإخوتِهِ وأخواتِهِ، بل أكثَرَ مِنْ حُبِّهِ لِنَفسِهِ، لا يَقدِرُ أنْ يكونَ تِلميذًا لي.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 10:38; لُوقا 9:23–25; مَرْقُس 8:34–37"> ومَنْ لا يَحمِلُ صَليبَهُ ويَتبعُني لا يَقدِرُ أنْ يكونَ تِلميذًا لي.</span>
<span className="v" id="v28" data-v="28" data-xref="الأمثال 24:27; ﻣﺘﻰ 10:22; يَشُوع 24:19–24"> فمَنْ مِنكُم، إذا أرادَ أنْ يَبنيَ بُرجًا، لا يَجلِسُ أوّلًا ويَحسِبُ النَّفقَةَ ليَرى هل يَقدِرُ أنْ يُكمِّلَهُ،</span>
<span className="v" id="v29" data-v="29"> لِئَلَّا يضَعَ الأساسَ ولا يَقدِرَ أنْ يُكمِّلَ، فيَستَهزئَ بِه النّاظِرونَ إلَيهِ كُلُّهُم</span>
<span className="v" id="v30" data-v="30" data-xref="العِبرانِيّين 6:11; رسالَةُ يُوحَنّا الثّانيَة 1:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:11–14"> ويَقولونَ: هذا الرَّجُلُ بَدأَ يَبني وما قَدِرَ أنْ يُكَمِّلَ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الأمثال 20:18; الأمثال 25:8; المُلُوكِ الثّانِي 18:20–22"> وأيُّ مَلِكٍ يَخرُجُ إلى مُحارَبَةِ مَلِكٍ آخَرَ، قَبْلَ أنْ يَجلِسَ ويُشاورَ نَفسَهُ هل يَقدِرُ أنْ يُواجِهَ بعَشرَةِ آلافٍ مَنْ يَزحَفُ إلَيهِ بِعشرينَ ألفًا؟</span>
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 12:20; يَعقُوب 4:6–10; لُوقا 12:58"> وإلَّا أرسَلَ إلَيهِ وفدًا، ما دامَ بَعيدًا عَنهُ، يَلتَمِسُ مِنهُ الصُّلحَ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="فِيلِبِّي 3:7–8; رسالَةُ يُوحَنّا الأُولَى 2:15–16; لُوقا 5:28"> وهكذا لا يَقدِرُ أحَدٌ مِنكُم أنْ يكونَ تِلميذًا لي، إلَّا إذا تَخَلَّى عَنْ كُلِّ شيءٍ لَهُ.</span>
</div>
</div>
