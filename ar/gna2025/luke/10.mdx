## يسوع يرسل الإثنين والسبعين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 6:7–13; لُوقا 9:52; العَدَد 11:24–26"> وبَعدَ ذلِكَ اَختارَ الرَّبُّ يَسوعُ اَثنينِ وسَبعينَ آخَرينَ، وأرسَلَهُم اَثنَينِ اَثنَينِ يَتَقدَّمونَهُ إلى كُلِّ مدينةٍ أو مَوضِعٍ عَزَمَ أنْ يذهَبَ إلَيهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 9:36–38; يُوحَنّا 4:35–38; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:6–9"> وقالَ لهُم: ”الحَصادُ كثيرٌ، ولكِنَّ العُمّالَ قَليلونَ. فاَطلُبوا مِنْ رَبِّ الحَصادِ أنْ يُرسِلَ عُمَّالًا إلى حصادِهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 10:16; اعمال الرسل 20:29; حِزْقِيال 2:3–6"> إِذهَبوا، ها أنا أُرسِلُكُم مِثلَ الخِرافِ بَينَ الذِّئابِ.</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 22:35; المُلُوكِ الثّانِي 4:29; لُوقا 10:4–12"> لا تَحمِلوا مِحفَظَةً، ولا كِيسًا، ولا حِذاءً، ولا تُسلِّموا على أحَدٍ في الطَّريقِ.</span>
<span className="v" id="v5" data-v="5" data-xref="صَمُوئيلَ الأوَّلُ 25:6; ﻣﺘﻰ 10:12–13; إشَعْياء 57:19"> وأيَّ بَيتٍ دَخَلْتُم، فَقولوا أوّلًا: السَّلامُ على هذا البَيتِ.</span>
<span className="v" id="v6" data-v="6" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:15–16; أفَسُس 5:6"> فإنْ كانَ فيهِ مَنْ يُحبُّ السَّلامَ، فسلامُكُم يَحِلُّ بِهِ، وإلَّا رجَعَ إلَيكُم.</span>
<span className="v" id="v7" data-v="7" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:17–18; ﻣﺘﻰ 10:10–11; غَلاطِيَّة 6:6"> وأقيموا في ذلِكَ البَيتِ، تأكُلونَ وتَشربونَ مِمَّا عِندَهُم، لأنَّ العامِلَ يَستَحِقُّ أُجرتَهُ، ولا تَنتَقِلوا مِنْ بَيتٍ إلى بَيتٍ.</span>
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:27; ﻣﺘﻰ 10:40; لُوقا 10:10"> وأيَّةَ مدينةٍ دَخَلتُم وقَبِلَكُم أهلُها، فكُلُوا مِمَّا يُقدِّمونَهُ لكُم.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 3:2; يُوحَنّا 3:5; لُوقا 10:11"> واَشفُوا مَرضاهُم وقولوا: مَلكوتُ اللهِ اَقتَرَبَ مِنكُم.</span>
<span className="v" id="v10" data-v="10" data-xref="اعمال الرسل 18:6; ﻣﺘﻰ 10:14; اعمال الرسل 13:51"> وأيَّةَ مدينةٍ دَخَلتُم وما قَبِلكُم أهلُها، فاَخرُجوا إلى شوارِعِها وقولوا:</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 10:9; اعمال الرسل 13:51; اعمال الرسل 13:26"> حتّى الغُبارُ العالِقُ بأقدامِنا مِنْ مدينتِكُم نَنفُضُهُ لكُم. ولكِنِ اَعلَموا أنَّ مَلكوتَ اللهِ اَقتَرَبَ.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 10:15; ﻣﺘﻰ 11:24; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:6"> أقولُ لكُم: سيكونُ مَصيرُ سَدومَ في يومِ الحِسابِ أكثرَ اَحتِمالًا مِنْ مَصيرِ تِلكَ المدينةِ.</span>
</div>
</div>

## المدن الكافرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 23:1–18; يُوئيل 3:4–8; حِزْقِيال 26:1–26"> ”الوَيلُ لكِ يا كورَزينُ! الوَيلُ لكِ يا بَيتَ صيدا! فلَو كانَتِ المُعجِزاتُ الّتي جَرَت فيكُما جرَت في صورَ وصيدا، لتابَ أهلُها مِنْ زمنٍ بعيدٍ ولَبِسوا المُسوحَ وقَعَدُوا على الرَّمادِ.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 15:22–25; ﺭﻭﻣﻴﺔ 2:1; لُوقا 12:47–48"> ولكِنَّ مصيرَ صورَ وصيدا في يومِ الحِسابِ سَيكونُ أكثرَ اَحتِمالًا مِنْ مَصيرِكُما.</span>
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 14:13–15; ﻣﺘﻰ 4:13; لُوقا 13:28"> وأنتِ يا كَفْرَناحومُ! أتَرتَفِعينَ إلى السَّماءِ؟ لا، إلى الجَحيمِ سَتَهبُطينَ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 13:20; ﻣﺘﻰ 10:40; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:8"> وقالَ يَسوعُ لتلاميذِه: ”مَنْ سَمِع إلَيكُم سَمِعَ إليَّ. ومَنْ رفَضكُم رفَضَني، ومَنْ رفَضَني رفَضَ الّذي أرسَلَني“.</span>
</div>
</div>

## رجوع الإثنين والسبعين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 16:20; مَرْقُس 16:17; لُوقا 9:1"> ورجَعَ الاثنانِ والسَّبعونَ رَسولًا فَرِحينَ وقالوا ليَسوعَ: ”يا رَبُّ، حتّى الشَّياطينُ تَخضَعُ لنا باَسمِكَ“.</span>
<span className="v" id="v18" data-v="18" data-xref="رُؤيا يُوحَنّا 12:7–9; إشَعْياء 14:12; رُؤيا يُوحَنّا 9:1"> فقَالَ لهُم: ”رأيتُ الشَّيطانَ يَسقُطُ مِنَ السَّماءِ مِثلَ البَرقِ.</span>
<span className="v" id="v19" data-v="19" data-xref="المَزامِير 91:13; مَرْقُس 16:18; ﺭﻭﻣﻴﺔ 16:20"> وها أنا أُعطيكُم سُلطانًا تَدوسونَ بِه الأفاعي والعقارِبَ وكُلَّ قُوَّةٍ للعَدُوِّ، ولا يضُرُّكُم شيءٌ.</span>
<span className="v" id="v20" data-v="20" data-xref="رُؤيا يُوحَنّا 20:15; رُؤيا يُوحَنّا 20:12; العِبرانِيّين 12:23"> ولكِنْ لا تَفرَحوا بأنَّ الأرواحَ تَخضَعُ لكُم، بلِ اَفرَحوا بأنَّ أسماءَكُم مكتوبَةٌ في السَّماواتِ“.</span>
</div>
</div>

## يسوع يبتهج

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:3; ﻣﺘﻰ 13:11–16; ﻣﺘﻰ 11:25–27"> وفي تلكَ السّاعَةِ اَبتَهَجَ يَسوعُ بِالرُّوحِ القُدُس، فقَالَ: ”أحمَدُكَ أيُّها الآبُ، يا رَبَّ السَّماءِ والأرضِ، لأنَّكَ أظهَرتَ للبُسَطاءِ ما أخفَيتَهُ عَنِ الحُكَماءِ والفُهَماءِ. نعم، أيُّها الآبُ، هكذا كانَت مَشيئَتُكَ.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 11:27; يُوحَنّا 1:18; يُوحَنّا 6:44–46"> أبي أعطاني كُلَّ شيءٍ. ما مِنْ أحَدٍ يَعرِفُ مَنْ هوَ الاَبنُ إلَّا الآبُ، ولا مَنْ هوَ الآبُ إلَّا الابنُ ومَنْ أرادَ الابنُ أنْ يُظهرَهُ لَه“.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 13:16–17"> واَلتَفَتَ إلى تلاميذِهِ، فقَالَ لهُم على اَنفِرادٍ: ”هَنيئًا لِمَن يَرى ما أنتُم تَرَونَ!</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 8:56; العِبرانِيّين 11:13; رسالَةُ بُطرُسَ الأُولَى 1:10–12"> أقولُ لكُم: كثيرٌ مِنَ الأنبياءِ والمُلوكِ تَمَنَّوا أنْ يَرَوا ما أنتُم تَرَونَ فما رَأوا، وأنْ يَسمَعوا ما أنتُم تَسمَعونَ فما سَمِعوا“.</span>
</div>
</div>

## مثل السامري الصالح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 22:34–39; لُوقا 18:18; ﻣﺘﻰ 19:16–19"> وقامَ أحَدُ عُلَماءِ الشَّريعةِ، فقَالَ لَهُ ليُحرِجَهُ: ”يا مُعَلِّمُ، ماذا أعمَلُ حتّى أرِثَ الحياةَ الأبدِيَّةَ؟“</span>
<span className="v" id="v26" data-v="26" data-xref="ﺭﻭﻣﻴﺔ 10:5; إشَعْياء 8:20; غَلاطِيَّة 3:21–22"> فأجابَهُ يَسوعُ: ”ماذا تَقولُ الشَّريعةُ؟ وكيفَ تُفسِّرُهُ؟“</span>
<span className="v" id="v27" data-v="27" data-xref="مَرْقُس 12:30–31; ﻣﺘﻰ 22:37–40; التَّثْنِيَة 6:5"> فقَالَ الرَّجُلُ: ”أحِبَّ الرَّبَّ إلهَكَ بِكُلِّ قَلبِكَ، وبِكُلِّ نَفسِكَ، وبِكُلِّ قُوَّتِكَ، وبِكُلِّ فِكرِكَ، وأحِبَّ قَريبَكَ مِثلَما تُحِبُّ نَفسَكَ“.</span>
<span className="v" id="v28" data-v="28" data-xref="حِزْقِيال 20:11; اللّاويِّين 18:5; ﻣﺘﻰ 19:17"> فقَالَ لَهُ يَسوعُ: ”بالصَّوابِ أجبتَ. إِعمَلْ هذا فتَحيا“.</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 16:15; ﺭﻭﻣﻴﺔ 4:2; اللّاويِّين 19:34"> فأرادَ مُعلِّمُ الشَّريعَةِ أنْ يُبرِّرَ نَفسَهُ، فقَالَ لِيَسوعَ: ”ومَنْ هوَ قَريبي؟“</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 88:4; لُوقا 18:31; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:12"> فأجابَهُ يَسوعُ: ”كانَ رَجُلٌ نازِلًا مِنْ أُورُشليمَ إلى أريحا، فوقَعَ بأيدي اللُّصوصِ، فعَرَّوهُ وضَرَبوهُ، ثُمَّ تَرَكوهُ بَينَ حيٍّ ومَيْتٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:16–18; المَزامِير 69:20; إرْمِيا 5:31"> واَتَّفَقَ أنَّ كاهِنًا نزَلَ في تلِكَ الطَّريقِ، فلمَّا رآهُ مالَ عَنهُ ومَشى في طريقِهِ.</span>
<span className="v" id="v32" data-v="32" data-xref="الأمثال 27:10; اعمال الرسل 18:17; المَزامِير 109:25"> وكذلِكَ أحَدُ اللّاويّينَ، جاءَ المكانَ فرَآهُ فمالَ عَنهُ ومَشى في طريقِهِ.</span>
<span className="v" id="v33" data-v="33" data-xref="المُلُوكِ الأوَّلُ 8:50; إرْمِيا 38:7–13; لُوقا 7:13"> ولكِنَّ سامِريًّا مُسافِرًا مَرَّ بِهِ، فلمَّا رَآهُ أشْفَقَ علَيهِ.</span>
<span className="v" id="v34" data-v="34" data-xref="ﺭﻭﻣﻴﺔ 12:20; المَزامِير 147:3; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:15"> فدَنا مِنهُ وسكَبَ زَيتًا وخَمرًا على جِراحِهِ وضَمَّدَها، ثُمَّ حَمَلهُ على دابَّتِهِ وجاءَ بِهِ إلى فُندُقٍ واَعتَنى بأمرِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="الأمثال 19:17; لُوقا 14:13; ﺭﻭﻣﻴﺔ 16:23"> وفي الغَدِ أخرَجَ السامِريُّ دينارَينِ، ودَفعَهُما إلى صاحِبِ الفُندُقِ وقالَ لَهُ: إِعتَنِ بأمرِهِ، ومَهما أنفَقْتَ زيادَةً على ذلِكَ أُوفيكَ عِندَ عودَتي.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="لُوقا 7:42; ﻣﺘﻰ 21:28–31; ﻣﺘﻰ 17:25"> فأيُّ واحدٍ مِنْ هَؤلاءِ الثَّلاثةِ كانَ في رأيِكَ قريبَ الّذي وقَعَ بأيدي اللُّصوصِ؟“</span>
<span className="v" id="v37" data-v="37" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:16–18; ميخا 6:8; رسالَةُ بُطرُسَ الأُولَى 2:21"> فأجابَهُ مُعَلِّمُ الشَّريعةِ: ”الّذي عامَلَهُ بالرَحمَةِ“. فقَالَ لَهُ يَسوعُ: ”إِذهَبْ أنتَ واَعمَلْ مِثلَهُ“.</span>
</div>
</div>

## يسوع عند مرتا ومريم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 11:19–20; يُوحَنّا 11:1–5; يُوحَنّا 12:1–3"> وبَينَما هُم سائِرونَ، دخَلَ يَسوعُ قَريةً، فرَحَّبَت بِهِ اَمرأةٌ اَسمُها مَرتا في بَيتِها.</span>
<span className="v" id="v39" data-v="39" data-xref="الأمثال 8:34; لُوقا 8:35; يُوحَنّا 12:3"> وكانَ لها أُختٌ اَسمُها مَريَمُ، جَلسَت عِندَ قَدمَي الرَّبِّ يَسوعَ تَستَمِعُ إلى كلامِهِ.</span>
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 6:27; لُوقا 12:29; ﻣﺘﻰ 14:15"> وكانَت مَرتا مُنَهمِكَةً في كثيرٍ من أمورِ الضِّيافَةِ، جاءَت وقالَت لِيَسوعَ: ”يا رَبُّ، أما تُبالي أن تَتْرُكَني أُختي أخدُمُ وحدي؟ قُلْ لها أن تُساعِدَني!“</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="فِيلِبِّي 4:6; لُوقا 21:34; ﻣﺘﻰ 6:25–34"> فأجابَها الرَّبُّ: ”مَرتا، مَرتا، أنتِ تقلَقينَ وتَهتمِّينَ بأُمورٍ كثيرةٍ،</span>
<span className="v" id="v42" data-v="42" data-xref="المَزامِير 27:4; المَزامِير 16:5–6; يُوحَنّا 10:27–28"> معَ أنَّ الحاجَةَ إلى شيءٍ واحدٍ. فمَريَمُ اَختارَتِ النَّصيبَ الأفضَلَ، ولن يَنزعَهُ أحَدٌ مِنها“.</span>
</div>
