## الشيطان يجرب يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 4:1–11; اعمال الرسل 10:38; مَرْقُس 1:12–13"> ورجَعَ يَسوعُ مِنْ نَهرِ الأُردُنِّ، وهوَ مُمتَلئٌ مِنَ الرُّوحِ القُدُسِ، فاَقتادَهُ الرُّوحُ في البرِّيَّةِ</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 34:28; العِبرانِيّين 4:15; التَّثْنِيَة 9:9"> أربعينَ يومًا، وإبليسُ يُجرِّبُهُ، وما أكَلَ شيئًا في تِلكَ الأيّامِ حتّى اَنقَضَت فَجاعَ.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 4:3; لُوقا 3:22"> فقَالَ لَه إبليسُ: ”إنْ كُنتَ اَبنَ اللهِ، فقُلْ لهذا الحجَرِ أنْ يَصيرَ خُبزًا“.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 8:3; ﻣﺘﻰ 4:4; ﻣﺘﻰ 6:31"> فأجابَهُ يَسوعُ: ”يَقولُ الكتابُ: ما بِالخُبزِ وحدَهُ يَحيا الإنسانُ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:15–16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:52; ﻣﺘﻰ 4:8"> وأصعَدَهُ إبليسُ إلى جبَلٍ مُرتَفِعٍ وأراهُ في لَحظَةٍ مِنَ الزَّمَنِ جميعَ ممالِكِ العالَمِ،</span>
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 12:31; رسالَةُ يُوحَنّا الأُولَى 5:19; يُوحَنّا 14:30"> وقالَ لَهُ: ”أُعطيكَ هذا السُّلطانَ كُلَّهُ ومَجدَ هذِهِ الممالِكِ، لأنَّني أملكُهُ وأنا أُعطيهِ لِمَنْ أشاءُ.</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 8:28; المَزامِير 72:11; إشَعْياء 45:14"> فإنْ سَجَدتَ لي يكونُ كُلُّهُ لكَ“.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 6:13; التَّثْنِيَة 10:20; ﻣﺘﻰ 4:10"> فأجابَهُ يَسوعُ: ”يَقولُ الكِتابُ: لِلرَّبِّ إلهِكَ تَسجُدُ، وإيَّاهُ وحدَهُ تَعبُدُ“.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 4:5–6; لُوقا 4:3; ﻣﺘﻰ 8:29"> وأخذَهُ إبليسُ إلى أُورُشليمَ، وأوقَفَهُ على شُرفَةِ الهَيكَلِ وقالَ لَهُ: ”إنْ كُنتَ اَبنَ اللهِ، فألقِ بِنَفْسِكَ مِنْ هُنا إلى الأسفَلِ،</span>
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 91:11–12; العِبرانِيّين 1:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:14"> لأنَّ الكِتابَ يَقولُ: يُوصي اللهُ مَلائكَتَهُ بِكَ لِيحفَظوكَ.</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 91:12"> وهُم يَحمِلونَكَ على أيديهِم لِئلَّا تَصدِمَ رِجلَكَ بحَجَرٍ“.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 6:16; العِبرانِيّين 3:8–9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:9"> فأجابَهُ يَسوعُ: ”ولكِنْ قِيلَ: لا تُجرِّبِ الرَّبَّ إلهَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="العِبرانِيّين 4:15; يَعقُوب 4:7; ﻣﺘﻰ 4:11"> وبَعدَما جرَّبَهُ إبليسُ بِكُلِّ تَجرِبَةٍ، فارَقهُ إلى حِينٍ.</span>
</div>
</div>

## الناصرة ترفض يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 4:23–25; اعمال الرسل 10:37; ﻣﺘﻰ 4:12"> ورجَعَ يَسوعُ إلى الجَليلِ، وهوَ مُمتلِئٌ بِقوَّةِ الرُّوحِ القُدُسِ، فَذاعَ صِيتُهُ في جميعِ تِلك الأنحاءِ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 4:23; ﻣﺘﻰ 9:8; ﻣﺘﻰ 9:35"> وكانَ يُعَلِّمُ في مَجامِعِهِم، فيُمجِّدونَهُ كُلُّهُم.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 17:2; اعمال الرسل 13:14–16; ﻣﺘﻰ 2:23"> وجاءَ يَسوعُ إلى النَّاصِرَةِ حَيثُ نَشأَ، ودخَلَ المجمَعَ يومَ السَّبتِ على عادَتِهِ، وقامَ ليَقرَأَ.</span>
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 61:1–3; اعمال الرسل 13:15; اعمال الرسل 7:42"> فناوَلوهُ كِتابَ النَّبيِّ إشَعيا، فلمَّا فتَحَ الكِتابَ وجَدَ المكانَ الّذي ورَدَ فيهِ:</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إشَعْياء 61:1; ﻣﺘﻰ 11:5; المَزامِير 146:7–8"> ”رُوحُ الرَّبِّ علَيَّ<br />لأنَّهُ مَسحني لأُبَشِّرَ المساكينَ،<br />أرسَلَني لأُناديَ لِلأسرى بالحُرِّيَّةِ،<br />وللعُميانِ بِعَودَةِ البصَرِ إليهِم،<br />لأُحَرِّرَ المَظلومينَ</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 61:2; إشَعْياء 63:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:1"> وأُعلِنَ الوَقتَ الّذي فيهِ<br />يَقبَلُ الرَّبُّ شعبَهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="لُوقا 4:17; يُوحَنّا 8:2; لُوقا 19:48"> وأغلَقَ يَسوعُ الكِتابَ وأعادَهُ إلى خادِمِ المَجمَعِ وجَلَسَ. وكانَت عُيونُ الحاضِرينَ كُلِّهِم شاخِصَةً إليهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 4:25–26; اعمال الرسل 3:18; اعمال الرسل 2:16–18"> فأخَذَ يقولُ لهُم: ”اليومَ تَمَّت هذِهِ الكَلِماتُ الّتي تَلَوْتُها على مَسامِعِكُم“.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 13:54–56; المَزامِير 45:2; اعمال الرسل 6:10"> فشَهِدوا لَه كُلُّهُم، وتَعَجَّبوا مِنْ كلامِ النِّعمَةِ الّذي يَخرُجُ مِنْ فَمِهِ، وقالوا: ”أما هوَ اَبنُ يوسُفَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 4:46–53; ﻣﺘﻰ 11:23–24; مَرْقُس 2:1–12"> فقَالَ لهُم يَسوعُ: ”ستَقولونَ لي هذا المثَلَ: يا طبيبُ إِشفِ نفسَكَ: فاَعمَلْ هُنا في وطنِكَ ما سَمِعنا أنَّكَ عمِلتَهُ في كَفْرناحومَ“.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 13:57; يُوحَنّا 4:44; مَرْقُس 6:4–5"> وقالَ لهُم: ”الحقَّ أقولُ لكُم: لا يُقبَلُ نَبِيٌّ في وَطَنِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="يَعقُوب 5:17; المُلُوكِ الأوَّلُ 17:1; ﻣﺘﻰ 20:15"> وبِحَقٍّ أقولُ لكُم: نَعَم، كانَ في إِسرائيلَ كثيرٌ مِنَ الأرامِلِ في زَمَنِ إيليَّا، حينَ تَوقَّفَ المطَرُ ثلاثَ سنَواتٍ وسِتَّةَ أشهُرٍ، فحَدثَتْ مَجاعَةٌ شَديدَةٌ في البِلادِ كُلِّها،</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الأوَّلُ 17:9–24; ﻣﺘﻰ 11:21; عُوبَدْيا 1:20"> وما أرسَلَ اللهُ إيليَّا إلى واحِدَةٍ مِنهُنَّ، بل أرسلَهُ إلى أرمَلَةٍ في صَرَفَةِ صَيدا.</span>
<span className="v" id="v27" data-v="27" data-xref="المُلُوكِ الثّانِي 5:1–27; يُوحَنّا 17:12; ﻣﺘﻰ 12:4"> وكانَ في إِسرائيلَ كثيرٌ مِنَ البُرصِ في زمَنِ النَّبي أليشَعَ، فما طهَّرَ اللهُ أحَدًا مِنهُم إلَّا نُعْمانَ السوريَّ“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="اعمال الرسل 5:33; أخبارِ الأيّامِ الثّانِي 24:20–21; إرْمِيا 38:6"> فلمَّا سَمِعَ الحاضِرونَ في المَجْمَعِ هذا الكلامَ غَضِبوا كثيرًا.</span>
<span className="v" id="v29" data-v="29" data-xref="يُوحَنّا 8:59; اعمال الرسل 21:28–32; العَدَد 15:35"> فقامُوا، وأخرَجُوهُ إلى خارِجِ المدينةِ، وجاؤُوا بِه إلى حافَّةِ الجبَلِ الّذي كانَت مدينتُهُم مَبنيّةً علَيهِ ليُلقُوهُ مِنها.</span>
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 10:39; يُوحَنّا 8:59; يُوحَنّا 18:6–7"> لكِنَّه مَرَّ مِنْ بَينِهِم ومَضى.</span>
</div>
</div>

## شفاء رجل فيه روح نجس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="مَرْقُس 1:21–28; ﻣﺘﻰ 4:13; اعمال الرسل 13:50–2"> ونَزَلَ إلى كَفْرَناحومَ، وهيَ مدينةٌ في الجَليلِ، وأخَذَ يُعَلِّمُ النَّاسَ في السَّبتِ.</span>
<span className="v" id="v32" data-v="32" data-xref="لُوقا 4:36; ﻣﺘﻰ 7:28–29; يُوحَنّا 6:63"> فتعجَّبوا كُلُّهُم مِنْ تَعليمِهِ، لأنَّهُ كانَ يتكَلَّمُ بِسُلطانٍ.</span>
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 1:23"> وكانَ في المَجمعِ رَجُلٌ فيهِ رُوحُ شَيطانٍ نَجِسٍ، فصاحَ بأعلى صوتِهِ:</span>
<span className="v" id="v34" data-v="34" data-xref="مَرْقُس 1:24; يَعقُوب 2:19; ﻣﺘﻰ 8:29"> ”آه، ما لكَ ولنا، يا يَسوعُ النّاصِريُّ؟ أجئْتَ لِتُهلِكنا؟ أنا أعرِفُ مَنْ أنتَ: أنتَ قُدُّوسُ اللهِ!“</span>
<span className="v" id="v35" data-v="35" data-xref="لُوقا 4:41; لُوقا 4:39; ﻣﺘﻰ 8:26"> فاَنتَهَرَهُ يَسوعُ، قالَ: ”إِخرَسْ واَخْرُجْ مِنَ الرَّجُلِ!“ فصَرَعَ الشَّيطانُ الرَّجُلَ في وَسْطِ المَجمَعِ وخرَجَ مِنهُ، مِنْ غَيرِ أنْ يُصيبَهُ بأذى.</span>
<span className="v" id="v36" data-v="36" data-xref="لُوقا 4:32; رسالَةُ بُطرُسَ الأُولَى 3:22; ﻣﺘﻰ 12:22–23"> فاَستَغرَبوا كُلُّهُم، وقالَ بَعضُهُم لِبَعضٍ: ما هذا الكلامُ؟ بِسُلطانٍ وقُوَّةٍ يأمُرُ الأرواحَ النَّجِسَةَ فتَخرُجُ“.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="مَرْقُس 1:45; لُوقا 4:14; إشَعْياء 52:13"> وذاعَ صيتُهُ في تِلكَ الأنحاءِ كُلِّها.</span>
</div>
</div>

## شفاء حماة بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 8:14–15; مَرْقُس 1:29–31; يُوحَنّا 11:22"> ثُمَّ ترَكَ المَجمعَ ودخَلَ بَيتَ سِمعانَ. وكانَت حَماةُ سِمعانَ مُصابَةً بِحُمَّى شَديدةٍ، فتَوسَّلوا إلَيهِ مِنْ أجلِها.</span>
<span className="v" id="v39" data-v="39" data-xref="لُوقا 4:41; لُوقا 4:35; لُوقا 8:24"> فدَنا مِنها، واَنتَهَرَ الحُمَّى فتركَتْها، فقامَت في الحالِ وأخذَت تَخدُمُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="مَرْقُس 1:32–34; ﻣﺘﻰ 8:16–17; مَرْقُس 5:23"> وعِندَ غُروبِ الشَّمسِ، جاءَ النَّاسُ بِمَرضاهُم إلى يَسوعَ، وكانوا مُصابينَ بِعِلَلٍ مُختَلِفَةٍ، فوضَعَ يَدَيهِ على كُلِّ واحدٍ مِنهُم وشفاهُ.</span>
<span className="v" id="v41" data-v="41" data-xref="مَرْقُس 1:34; يَعقُوب 2:19; لُوقا 4:34–35"> وخرَجَتِ الشَّياطينُ مِنْ مَرضى كثيرينَ وهيَ تَصرُخُ: ”أنتَ اَبنُ اللهِ!“ فكانَ يَسوعُ يَنتهِرُها ويَمنَعُها مِنَ الكلامِ، لأنَّها عرَفَت أنَّهُ المَسيحُ.</span>
</div>
</div>

## يسوع يُعلِّم في المجامع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="لُوقا 6:12; مَرْقُس 1:35–38; يُوحَنّا 4:34"> وخرَجَ يَسوعُ في الصَّباحِ، فذَهبَ إلى مكانٍ مُقفرٍ. وبَحثَ عَنهُ النَّاسُ، فلمَّا وجَدوهُ تَمسّكوا بِه لِئلَّا يَرحَلَ عَنهُم.</span>
<span className="v" id="v43" data-v="43" data-xref="مَرْقُس 1:38–39; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:2; يُوحَنّا 9:4"> فقَالَ لهُم: ”يَجِبُ علَيَّ أن أُبَشِّرَ سائِرَ المُدُنِ بِمَلكوتِ اللهِ، لأنِّي لِهذا أُرسِلتُ“.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 4:23; مَرْقُس 1:39; لُوقا 4:15"> ومَضى يُبَشِّرُ في مجامعِ اليهوديَّةِ.</span>
</div>
