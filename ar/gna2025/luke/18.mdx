## مثل الأرملة والقاضي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="كُولُوسِي 4:2; ﺭﻭﻣﻴﺔ 12:12; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:17"> وكلَّمَهُم بِمَثَلٍ على وُجوبِ المُداومَة على الصلاةِ مِنْ غَيرِ مَلَلٍ،</span>
<span className="v" id="v2" data-v="2" data-xref="الأمثال 29:7; المَزامِير 8:1–4; لُوقا 18:4"> قالَ: ”كانَ في إحدى المُدُنِ قاضٍ لا يَخافُ اللهَ ولا يَهابُ النَّاسَ.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 1:17; إرْمِيا 5:28; التَّثْنِيَة 27:19"> وكانَ في تِلكَ المدينةِ أرملةٌ تتردَّدُ إليهِ وتقولُ لَهُ: أنصِفْني مِنْ خَصْمي!</span>
<span className="v" id="v4" data-v="4" data-xref="العِبرانِيّين 4:12–13; لُوقا 16:3; لُوقا 12:17"> فكانَ يَرفُضُ طَلَبها، ولكنَّهُ بَعدَ مُدَّةٍ طويلةٍ قالَ في نَفسِهِ: معَ أنِّي لا أخافُ اللهَ ولا أهابُ النَّاسَ،</span>
<span className="v" id="v5" data-v="5" data-xref="لُوقا 11:8; لُوقا 18:39; القُضاة 16:16"> فسأُنصِفُ هذِهِ الأرمَلةَ لأنَّها تُزْعِجُني، وإلَّا ظَلَّت تَجيءُ وتُضايقُني“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="لُوقا 7:13"> وقالَ الرَّبُّ يَسوعُ: ”إِسمَعوا جيِّدًا ما قالَ هذا القاضي الظَّالمُ،</span>
<span className="v" id="v7" data-v="7" data-xref="العِبرانِيّين 10:35–37; رسالَةُ بُطرُسَ الثّانيَة 3:9; لُوقا 11:13"> فكيفَ لا يُنصِفُ اللهُ مُختاريهِ الضَّارعينَ إلَيهِ ليلَ نهارَ؟ وهل يُبطئُ في الاستجابَةِ لهُم؟</span>
<span className="v" id="v8" data-v="8" data-xref="العِبرانِيّين 10:23–26; رسالَةُ بُطرُسَ الثّانيَة 3:8–9; ﻣﺘﻰ 24:9–13"> أقولُ لكُم: إنَّهُ يُسرعُ إلى إنصافِهِم. ولكِنْ، أيَجِدُ اَبنُ الإنسانِ إيمانًا على الأرضِ يومَ يَجيءُ؟“</span>
</div>
</div>

## مثل الفريسي وجابي الضرائب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="لُوقا 16:15; الأمثال 30:12; إشَعْياء 65:5"> وقالَ هذا المثَلَ لِقومٍ كانوا على ثِقَةٍ بأنَّهُم أبرارٌ، ويَحتَقِرونَ الآخرينَ:</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 21:31–32; المُلُوكِ الأوَّلُ 8:30; المُلُوكِ الثّانِي 20:5"> ”صَعِدَ رَجُلانِ إلى الهَيكَلِ ليُصَلِّيا، واحدٌ فَرِّيسيٌّ والآخرُ مِنْ جُباةِ الضَّرائبِ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 6:5; إشَعْياء 65:5; رُؤيا يُوحَنّا 3:17"> فوقَفَ الفَرِّيسيُّ يُصلِّي في نَفسِهِ فيقولُ: شُكرًا لكَ يا اللهُ، فما أنا مِثلُ سائِرِ النَّاسِ الطَّامعينَ الظَّالِمينَ الزُّناةِ، ولا مِثلُ هذا الجابي!</span>
<span className="v" id="v12" data-v="12" data-xref="لُوقا 11:42; ﻣﺘﻰ 23:23–24; ﻣﺘﻰ 9:14"> فأنا أصومُ في الأُسبوعِ مَرَّتَينِ، وأوفي عُشْرَ دَخلي كُلِّهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="عَزْرا 9:6; رسالَةُ يُوحَنّا الأُولَى 1:8–10; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:15"> وأمَّا الجابي، فوقَفَ بَعيدًا لا يَجْرُؤُ أنْ يَرفَعَ عَينَيهِ نحوَ السَّماءِ، بل كانَ يَدُقُّ على صَدْرِهِ ويقولُ: إِرحَمْني يا اللهُ، أنا الخاطئُ!</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="لُوقا 14:11; لُوقا 16:15; يَعقُوب 4:10"> أقولُ لكُم: هذا الجابي، لا ذاكَ الفَرِّيسيُّ، نزَلَ إلى بَيتِهِ مَقبولًا عِندَ اللهِ. فمَنْ يرفَعْ نَفسَهُ يَنخَفِضْ، ومَنْ يخْفِضْ نَفسَهُ يَرتَفِعْ“.</span>
</div>
</div>

## يسوع يبارك الأطفال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="مَرْقُس 10:13–16; ﻣﺘﻰ 19:13–15; صَمُوئيلَ الأوَّلُ 1:24"> وجاءَ إلَيهِ بَعضُ النَّاسِ بأطفالٍ ليضَعَ يدَيهِ علَيهِم. فلمَّا رآهُمُ التَّلاميذُ اَنتَهرُوهُم.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 18:3–4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:14"> ولكِنْ يَسوعُ دعا الأطفالَ إلَيهِ وقالَ: ”دَعُوا الأطفالَ يَأتونَ إليَّ ولا تَمنَعوهُم، لأنَّ لأمثالِ هَؤلاءِ مَلكوتَ اللهِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 18:3; مَرْقُس 10:15; المَزامِير 131:1–2"> الحَقَّ أقولُ لكُم: مَنْ لا يَقبلُ مَلكوتَ اللهِ كأنَّهُ طِفْلٌ لا يَدخُلُهُ“.</span>
</div>
</div>

## الرجل الغني

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="مَرْقُس 10:17–31; اعمال الرسل 16:30; ﻣﺘﻰ 19:16–30"> وسألَهُ أحَدُ الوُجَهاءِ: ”أيُّها المُعَلِّمُ الصَّالحُ، ماذا أعمَلُ لأرِثَ الحياةَ الأبدِيَّةَ؟“</span>
<span className="v" id="v19" data-v="19" data-xref="أيُّوب 15:14–16; يَعقُوب 1:17; العِبرانِيّين 7:26"> فأجابَهُ يَسوعُ: ”لماذا تَدعوني صالِحًا؟ لا صالحٌ إلَّا اللهُ وحدَهُ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 20:12–17; التَّثْنِيَة 5:16–21; ﺭﻭﻣﻴﺔ 13:9"> أنتَ تَعرِفُ الوصايا: لا تَزْنِ، لا تَقْتُلْ، لا تَسرِقْ، لا تَشهَدْ بالزّورِ، أكرِمْ أباكَ وأُمَّكَ“.</span>
<span className="v" id="v21" data-v="21" data-xref="فِيلِبِّي 3:6; ﺭﻭﻣﻴﺔ 10:2–3; مَرْقُس 10:20–21"> فقَالَ الرَّجُلُ: ”مِنْ أيّامِ صِبايَ عمِلتُ بِهذِهِ الوصايا كُلِّها“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="لُوقا 12:33; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:18–19; ﻣﺘﻰ 19:21"> فلمَّا سمِعَ يَسوعُ كلامَهُ هذا، قالَ لَهُ: ”يُعْوِزُكَ شيءٌ واحدٌ، بِعْ كُلَّ ما تَملِكُ ووَزِّعْ ثَمنَهُ على الفُقراءِ، فيكونَ لكَ كَنزٌ في السَّماواتِ، وتَعالَ اَتبَعْني“.</span>
<span className="v" id="v23" data-v="23" data-xref="حِزْقِيال 33:31; رسالَةُ يُوحَنّا الأُولَى 2:15; أيُّوب 31:24–25"> فحَزِنَ الرَّجُلُ عِندَما سمِعَ هذا الكلامَ، لأنَّهُ كانَ غَنيًّا جدًّا.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الأمثال 30:9; يَعقُوب 2:5–7; مَرْقُس 10:23–27"> ورأى يَسوعُ أنَّهُ حزِنَ، فقَالَ: ”ما أصعَبَ دُخولَ الأغنياءِ إلى مَلكوتِ اللهِ!</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 23:24"> فمُرورُ الجمَلِ في ثُقبِ الإبرَةِ أسهَلُ مِنْ دُخولِ الغَنيِّ إلى مَلكوتِ اللهِ“.</span>
<span className="v" id="v26" data-v="26" data-xref="لُوقا 13:23"> فقَالَ السّامِعونَ: ”مَنْ يُمكِنُهُ أنْ يَخلُصَ، إذًا؟“</span>
<span className="v" id="v27" data-v="27" data-xref="إرْمِيا 32:17; ﻣﺘﻰ 19:26; لُوقا 1:37"> فأجابَ يَسوعُ: ”ما لا يُمكِنُ عِندَ النَّاسِ، مُمْكنٌ عِندَ اللهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="مَرْقُس 10:28; فِيلِبِّي 3:7; ﻣﺘﻰ 19:27"> فقَالَ لَهُ بُطرُسُ: ”ها نَحنُ تَركنا ما لنا وتَبِعناكَ!“</span>
<span className="v" id="v29" data-v="29" data-xref="مَرْقُس 10:29–31; ﻣﺘﻰ 10:37–39; لُوقا 14:26–28"> فأجابَ يَسوعُ: ”الحقَّ أقولُ لكُم: ما مِنْ أحَدٍ ترَكَ بَيتًا أوِ اَمرأةً أو إخوَةً أو أبًا أو أُمًّا أو أولادًا مِنْ أجلِ مَلكوتِ اللهِ،</span>
<span className="v" id="v30" data-v="30" data-xref="العِبرانِيّين 13:5–6; أيُّوب 42:10; رُؤيا يُوحَنّا 2:17"> إلَّا نالَ في هذِهِ الدُّنيا أضعافَ ما ترَكَ، ونالَ في الآخِرَةِ الحياةَ الأبدِيَّةَ“.</span>
</div>
</div>

## يسوع يُنبئ مرة ثالثة بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="ﻣﺘﻰ 20:17–19; إشَعْياء 53:1–12; المَزامِير 22:1–31"> وأخَذَ التَّلاميذَ الاثنَي عشَرَ على اَنفِرادٍ وقالَ لهُم: ”ها نَحنُ صاعِدونَ إلى أُورُشليمَ، فيَتِمُّ كُلُّ ما كتَبَهُ الأنبياءُ في اَبنِ الإنسانِ،</span>
<span className="v" id="v32" data-v="32" data-xref="لُوقا 23:11; إشَعْياء 50:6; إشَعْياء 53:3"> فسَيُسَلَّمُ إلى الوثَنيِّينَ، فيَستَهزِئونَ بِه ويَشتُمونَهُ ويَبصُقونَ علَيهِ،</span>
<span className="v" id="v33" data-v="33" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:3–4; لُوقا 24:7; ﻣﺘﻰ 27:63"> ثُمَّ يَجلِدونَهُ ويَقتُلونَهُ، وفي اليومِ الثَّالثِ يَقومُ“.</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 9:45; مَرْقُس 9:32; لُوقا 24:25"> فما فَهِمَ التَّلاميذُ شيئًا مِنْ ذلِكَ، وكانَ هذا الكلامُ مُغلَقًا علَيهِم، فما أدركوا مَعناهُ.</span>
</div>

## يسوع يشفي أعمى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 20:29–34; مَرْقُس 10:46–52; لُوقا 16:20–21"> واَقتَرَبَ يَسوعُ مِنْ أريحا، وكانَ رَجُلٌ أعمى جالِسًا على جانِبِ الطَّريقِ يَستَعطي.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 21:10–11; لُوقا 15:26"> فلمَّا أحسَّ بِمُرورِ الجُموعِ سألَ: ”ما هذا؟“</span>
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 2:23; يُوحَنّا 19:19; يُوحَنّا 12:35–36"> فأخبروهُ أنَّ يَسوعَ النّاصِريَّ يَمُرُّ مِنْ هُناكَ.</span>
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 9:27; لُوقا 18:39; ﻣﺘﻰ 15:22"> فصاحَ الأعمى: ”يا يَسوعُ اَبنَ داودَ، اَرحَمني!“</span>
<span className="v" id="v39" data-v="39" data-xref="التَّكوين 32:26–28; المَزامِير 141:1; لُوقا 18:15"> فاَنتَهَرَهُ السّائِرونَ في المُقدِّمَةِ ليَسكُتَ. لكِنَّهُ صاحَ بِصوتٍ أعلى: ”يا اَبنَ داودَ، اَرحَمني!“</span>
<span className="v" id="v40" data-v="40" data-xref="ﻣﺘﻰ 20:31–34; مَرْقُس 10:48–52"> فوقَفَ يَسوعُ وأمرَ بأنْ يُقَدِّموهُ إلَيهِ. فلمَّا اَقتَرَبَ سألَهُ:</span>
<span className="v" id="v41" data-v="41" data-xref="المُلُوكِ الأوَّلُ 3:5–15; ﻣﺘﻰ 20:21–22; فِيلِبِّي 4:6"> ”ماذا تُريدُ أنْ أعمَلَ لكَ؟“ فأجابَهُ: ”أنْ أُبصِرَ، يا سيِّدُ!“</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="ﻣﺘﻰ 8:3; لُوقا 8:48; لُوقا 17:19"> فقَالَ لَه يَسوعُ: ”أبصِرْ، إيمانُكَ شَفاكَ!“</span>
<span className="v" id="v43" data-v="43" data-xref="المَزامِير 146:8; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 1:10–12; إشَعْياء 43:21"> فأبصَرَ في الحالِ وتبِعَ يَسوعَ وهوَ يَحمَدُ اللهَ. ولمَّا رأى الشَّعبُ ما جرى، مَجَّدوا اللهَ كُلُّهُم.</span>
</div>
