## يسوع عند بيلاطس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 27:1–2; ﻣﺘﻰ 27:11–14; لُوقا 22:66"> وقامَ الحُضورُ كُلُّهُم وجاؤُوا بِه إلى بيلاطُسَ،</span>
<span className="v" id="v2" data-v="2" data-xref="اعمال الرسل 17:6–7; يُوحَنّا 19:12; لُوقا 23:14"> وأخذوا يَتَّهِمونَهُ فيقولونَ: ”وجَدْنا هذا الرَّجُلَ يثيرُ الفِتنَةَ في شَعبِنا، ويَمنَعُهُ أنْ يدفَعَ الجِزيَةَ إلى القَيصَرِ، ويَدَّعي أنَّهُ المسيحُ المَلِكُ“.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 1:49; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:13; لُوقا 22:70"> فسألَهُ بِيلاطُسُ: ”أأنتَ مَلِكُ اليَهودِ؟“ فأجابَهُ: ”أنتَ قُلتَ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 18:38; لُوقا 23:14–15; لُوقا 23:22"> فقَالَ بِيلاطُسُ لِرُؤساءِ الكَهنَةِ والجُموعِ: ”لا أجِدُ جُرمًا على هذا الرَّجُلِ!“</span>
<span className="v" id="v5" data-v="5" data-xref="مَرْقُس 1:14; يُوحَنّا 1:43; يُوحَنّا 2:11"> ولكنَّهُم أصَرُّوا على قولِهِم: ”إنَّهُ يُثيرُ الشَّعبَ بتَعليمِهِ في اليَهوديَّةِ كُلِّها، مِنَ الجَليلِ إلى هُنا“.</span>
</div>
</div>

## يسوع عند هيرودس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 5:37; لُوقا 13:1"> فسألَ بِيلاطُسُ عِندَما سَمِعَ هذا الكلامَ: ”هَلِ الرَّجُلُ مِنَ الجليلِ؟“</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 3:1; لُوقا 13:31; ﻣﺘﻰ 14:1"> فلمَّا عَرفَ أنَّهُ مِنْ ولايَةِ هِيرودُسَ أرسَلَهُ إلى هِيرودُسَ، وهوَ في ذلِكَ الوَقتِ نازِلٌ في أُورُشليمَ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="لُوقا 9:7–9; ﻣﺘﻰ 14:1; مَرْقُس 6:14"> فلمَّا رأى هِيرودُسُ يَسوعَ فَرِحَ كثيرًا، لأنَّهُ كانَ يَرغَبُ مِنْ زمانٍ بَعيدٍ أنْ يَراهُ لِكَثرَةِ ما سَمِعَ عَنهُ، ويَرجو أنْ يَشهَدَ آيَةً تَتِمُّ على يدِهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 27:14; ﻣﺘﻰ 7:6; رسالَةُ بُطرُسَ الأُولَى 2:23"> فسألَهُ مسائِلَ كثيرةً، فما أجابَهُ عَنْ شيءٍ.</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 11:53; لُوقا 23:14–15; لُوقا 23:5"> وقامَ رُؤساءُ الكَهنَةِ ومُعَلِّمو الشَّريعةِ يَتَّهِمونَهُ ويُشَدِّدونَ علَيهِ التُّهَم.</span>
<span className="v" id="v11" data-v="11" data-xref="مَرْقُس 15:16–20; ﻣﺘﻰ 27:27–30; المَزامِير 69:19–20"> فأهانَهُ هيرودُسُ وجُنودُهُ. واَستَهزَأَ بِه، فألبَسهُ ثَوبًا برّاقًا ورَدَّه إلى بِيلاطُسَ.</span>
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 4:27; رُؤيا يُوحَنّا 17:13–14; المَزامِير 83:4–6"> وتصَالَحَ هِيرودُسُ وبِيلاطُسُ في ذلِكَ اليومِ، وكانا مِنْ قَبلُ مُتَخاصِمَينِ.</span>
</div>
</div>

## الحكم على يسوع بالموت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 15:14; ﻣﺘﻰ 27:21–23; يُوحَنّا 18:38"> فدَعا بِيلاطُسُ رُؤساءَ الكَهنَةِ والزُّعَماءَ والشَّعبَ</span>
<span className="v" id="v14" data-v="14" data-xref="دانيال 6:4; العِبرانِيّين 7:26; لُوقا 23:1–2"> وقالَ لهُم: ”جِئتُم إليَّ بِهذا الرَّجُلِ وقُلتُم إنَّهُ يُضلِّلُ الشَّعبَ. فَفَحَصتُهُ أمامَكُم، فما وجَدْتُ أنَّهُ اَرتكَبَ شَيئًا ممَّا تتَّهِمونَهُ بِه،</span>
<span className="v" id="v15" data-v="15"> ولا هِيرودُسُ وجَدَ أيضًا، لأنَّهُ رَدَّهُ إِلينا. فلا شيءَ إذًا فعَلَهُ هذا الرَّجُلُ يَستوجِبُ بِه الموتَ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 27:26; لُوقا 23:22; مَرْقُس 15:15"> فسأجلِدُهُ وأُخلي سَبيلَهُ“.</span>
<span className="v" id="v17" data-v="17"> &#91;وكانَ على بِيلاطُسَ أنْ يُطلِقَ لهُم في كُلِّ عيدٍ واحدًا مِنَ السُّجناءِ&#93;.</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 21:36; اعمال الرسل 3:14; اعمال الرسل 22:22"> فَصاحوا بِصوتٍ واحدٍ: ”أقتُلْ هذا الرَّجُلَ وأطلِقْ لنا باراباسَ!“</span>
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 3:14; لُوقا 23:5; لُوقا 23:2"> وكانَ باراباسُ في السِّجنِ لاَشتِراكِهِ في فِتنةٍ وقَعَت في المدينةِ، ولاَرْتكابِهِ جَريمةَ قَتلٍ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 15:15; ﻣﺘﻰ 14:8–9; ﻣﺘﻰ 27:19"> فخاطبَهُم بِيلاطُسُ ثانيةً لأنَّهُ كانَ يُريدُ أنْ يُخلِيَ سَبيلَ يَسوعَ،</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 27:22–25; مَرْقُس 15:13; يُوحَنّا 19:15"> فصاحوا: ”إصلِبْهُ! إصلِبْهُ!“</span>
<span className="v" id="v22" data-v="22" data-xref="لُوقا 23:14; لُوقا 23:20; رسالَةُ بُطرُسَ الأُولَى 3:18"> فقَالَ لهُم ثالثةً: ”أيَّ شَرٍّ فعَلَ هذا الرَّجُلُ؟ لا أجِدُ علَيهِ ما يَستَوجِبُ بِه الموتَ. فسأجلِدُهُ وأُخلي سبيلَهُ!“</span>
<span className="v" id="v23" data-v="23" data-xref="زَكَريّا 11:8; المَزامِير 57:4; المَزامِير 22:12–13"> فألحّوا علَيهِ بأعلى أصواتِهِم طالِبينَ صَلبَهُ، واَشتَدَّ صِياحُهُم،</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 27:26; الخُرُوج 23:2; الأمثال 17:15"> فحكَمَ بِيلاطُسُ أنْ يُجابَ طلَبُهُم،</span>
<span className="v" id="v25" data-v="25" data-xref="اعمال الرسل 3:14; مَرْقُس 15:6–7; مَرْقُس 15:15"> فأطلَقَ الرَّجُلَ الّذي طَلَبوهُ، وكانَ في السِّجنِ لِجريمَةِ قَتلٍ وإثارَةِ فِتنةٍ، وأسلَمَ يَسوعَ إلى مَشيئَتِهِم.</span>
</div>
</div>

## يسوع على الصليب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 27:32–44; مَرْقُس 15:21–32; اعمال الرسل 6:9"> وبَينَما هُم ذاهِبونَ بِه، أمسَكوا سِمعانَ، وهوَ رَجُلٌ قِيرينيٌّ كانَ راجِعًا مِنَ الحَقلِ، فألقَوا علَيهِ الصَّليبَ ليَحمِلَهُ خَلفَ يَسوعَ.</span>
<span className="v" id="v27" data-v="27" data-xref="لُوقا 8:52; لُوقا 8:2; مَرْقُس 15:40"> وتَبِعَهُ جُمهورٌ كبيرٌ مِنَ الشَّعبِ، ومِنْ نِساءٍ كُنَّ يَلطِمْنَ صُدورَهُنَّ ويَنُحْنَ علَيهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="نَشِيدِ الأنشاد 8:4; نَشِيدِ الأنشاد 5:8; نَشِيدِ الأنشاد 1:5"> فاَلتَفَتَ يَسوعُ إلَيهِنَّ وقالَ: ”لا تَبكِينَ علَيَّ يا بناتِ أُورُشليمَ، بَلِ اَبكِينَ على أنفُسِكُنَّ وعلى أولادِكُنَّ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 24:19; لُوقا 21:23–24; مَرْقُس 13:17–19"> سَتَجيءُ أيَّامٌ يُقالُ فيها: هَنيئًا لِلواتي ما حَبِلْنَ ولا ولَدْنَ ولا أرضَعْنَ.</span>
<span className="v" id="v30" data-v="30" data-xref="رُؤيا يُوحَنّا 6:16; هُوشَع 10:8; إشَعْياء 2:19"> ويُقالُ لِلجِبالِ اَسقُطي علَينا، ولِلتِلالِ غطِّينا.</span>
<span className="v" id="v31" data-v="31" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:17–18; يُوحَنّا 15:6; الأمثال 11:31"> فإذا كانوا هكذا يَفعَلونَ بالغُصنِ الأخضَرِ، فكيفَ تكونُ حالُ الغُصنِ اليابسِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 15:27; إشَعْياء 53:12; ﻣﺘﻰ 27:38"> وساقوا مَعَهُ إلى القَتلِ اَثنَينِ مِنَ المُجرِمينَ.</span>
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 15:22–23; يُوحَنّا 19:17–18; يُوحَنّا 18:32"> ولمَّا وصَلوا إلى المكانِ المُسمَّى بالجُمجُمَةِ، صَلَبوهُ هُناكَ معَ المُجرِمَينِ، واحدًا عَنْ يَمينِهِ والآخَرَ عَنْ شِمالِهِ.</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 5:44; اعمال الرسل 7:60; رسالَةُ بُطرُسَ الأُولَى 3:9"> فقَالَ يَسوعُ: ”إِغفِرْ لهُم يا أبي، لأنَّهُم لا يَعرِفونَ ما يَعمَلونَ“. واَقتَسَموا ثيابَهُ مُقتَرعينَ علَيها.</span>
<span className="v" id="v35" data-v="35" data-xref="المَزامِير 22:17; إشَعْياء 42:1; رسالَةُ بُطرُسَ الأُولَى 2:4"> ووقَفَ الشَّعبُ هُناكَ يَنظُرونَ، ورُؤساؤُهُم يَقولونَ مُتَهكِّمينَ: ”خَلَّصَ غَيرَهُ فَليُخَلِّصْ نَفسَهُ، إنْ كانَ مَسيحَ اللهِ المُختارَ!“</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="المَزامِير 69:21; ﻣﺘﻰ 27:48; ﻣﺘﻰ 27:34"> واَستَهزَأَ بِهِ الجُنودُ أيضًا، وهُم يَقتَرِبونَ ويُناوِلونَهُ خَلًا</span>
<span className="v" id="v37" data-v="37"> ويَقولونَ: ”خَلِّصْ نَفسَكَ، إنْ كُنتَ مَلِكَ اليَهودِ!“</span>
<span className="v" id="v38" data-v="38" data-xref="مَرْقُس 15:26; ﻣﺘﻰ 27:37; يُوحَنّا 19:19–22"> وكانَ فَوقَ رأسِهِ لَوحَةٌ مكتوبٌ فيها: ”هذا مَلِكُ اليَهودِ!“</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="ﻣﺘﻰ 27:44; مَرْقُس 15:32; لُوقا 23:35"> وأخَذَ أحَدُ المُجرِمَينِ المُعلَّقَينِ على الصَّليبِ يَشتُمُهُ ويَقولُ لَه: ”أما أنتَ المَسيحُ؟ فخَلِّصْ نَفسَكَ وخَلِّصْنا!“</span>
<span className="v" id="v40" data-v="40" data-xref="أفَسُس 5:11; لُوقا 12:5; رُؤيا يُوحَنّا 15:4"> فاَنتَهَرَهُ المُجرِمُ الآخَرُ قالَ: ”أما تَخافُ اللهَ وأنتَ تَتحَمَّلُ العِقابَ نَفسَهُ؟</span>
<span className="v" id="v41" data-v="41" data-xref="اللّاويِّين 26:40–41; عَزْرا 9:13; رسالَةُ بُطرُسَ الأُولَى 1:19"> نَحنُ عِقابُنا عَدلٌ، نِلناهُ جَزاءَ أعمالِنا، أمَّا هوَ، فما عَمِلَ سُوءًا“.</span>
<span className="v" id="v42" data-v="42" data-xref="ﺭﻭﻣﻴﺔ 10:9–14; اعمال الرسل 16:31; رسالَةُ يُوحَنّا الأُولَى 5:11–13"> وقالَ: ”أُذكُرْني يا يسوعُ، متى جِئتَ في مَلكوتِكَ“.</span>
<span className="v" id="v43" data-v="43" data-xref="رُؤيا يُوحَنّا 2:7; المَزامِير 50:15; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:15–16"> فأجابَ يَسوعُ: ”الحقَّ أقولُ لكَ: سَتكونُ اليومَ مَعي في الفِردَوسِ“.</span>
</div>
</div>

## موت يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 27:45; مَرْقُس 15:33–34; عامُوس 8:9"> وعِندَ الظُّهرِ خَيَّمَ الظَّلامُ على الأرضِ كُلِّها حتّى السَّاعةِ الثَّالِثةِ.</span>
<span className="v" id="v45" data-v="45" data-xref="العِبرانِيّين 10:19–22; ﻣﺘﻰ 27:51; العِبرانِيّين 9:3–8"> واَحتَجَبَتِ الشَّمسُ واَنشَقَّ حِجابُ الهَيكَلِ مِنَ الوَسَطِ.</span>
<span className="v" id="v46" data-v="46" data-xref="المَزامِير 31:5; اعمال الرسل 7:59; يُوحَنّا 19:30"> وصرَخَ يَسوعُ صَرخةً قويَّةً: ”يا أبي، في يَدَيكَ أستَودِعُ رُوحي“. قالَ هذا وأسلَمَ الرُّوحَ.</span>
</div>
<div className="bp">
<span className="v" id="v47" data-v="47" data-xref="ﻣﺘﻰ 27:54; مَرْقُس 15:39; لُوقا 23:41"> فلمَّا رأى قائِدُ الحرَسِ ما جرى، مَجَّدَ اللهَ وقالَ: ”بالحَقيقَةِ، هذا الرَّجُلُ كانَ بارًّا“.</span>
<span className="v" id="v48" data-v="48" data-xref="لُوقا 18:13; اعمال الرسل 2:37; إرْمِيا 31:19"> والجُموعُ الّتي حَضرَت ذلِكَ المَشهَدَ، فرَأَت ما جَرى، رجَعَت وهيَ تَلطِمُ الصُّدورَ.</span>
<span className="v" id="v49" data-v="49" data-xref="لُوقا 8:2; المَزامِير 38:11; لُوقا 23:55"> وكانَ جميعُ أصدقاءِ يَسوعَ، والنِّساءُ اللَّواتي تَبِعْنَهُ مِنَ الجَليلِ، يُشاهِدونَ هذِهِ الأحداثَ عَنْ بُعدٍ.</span>
</div>
</div>

## دفن يسوع

<div className="bt" dir="rtl">
<div className="bp">
