## السَّبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّثْنِيَة 23:25; مَرْقُس 2:23–28; ﻣﺘﻰ 12:1–8"> ومَرَّ يَسوعُ بَينَ الحُقولِ في السَّبتِ، فأخَذَ تلاميذُهُ يَقطِفونَ السَّنابِلَ ويَفرُكونَها بأيديهِم ويأكُلونَ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 12:2; يُوحَنّا 5:16; يُوحَنّا 5:9–11"> فقَالَ لهُم بَعضُ الفَرِّيسيِّينَ: ”لِماذا تَعمَلونَ ما لا يَحِلُّ في السَّبتِ؟“</span>
<span className="v" id="v3" data-v="3" data-xref="صَمُوئيلَ الأوَّلُ 21:3–6; مَرْقُس 2:25–26; ﻣﺘﻰ 12:3–5"> فأجابَهُم يَسوعُ: ”أما قرَأتُم ما عَمِلَ داودُ عِندَما جاعَ هوَ ورِجالُهُ؟</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 24:5–9"> كيفَ دخَلَ بَيتَ اللهِ وتَناوَلَ خُبزَ القُربانِ وأكَلَ وأعطى مِنهُ رِجالَهُ، معَ أنَّ أكلَهُ لا يَحلُّ إلَّا لِلكَهنَةِ وحدَهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="مَرْقُس 2:27; رُؤيا يُوحَنّا 1:10; مَرْقُس 9:7"> وقالَ لهُم يَسوعُ: ”إِبنُ الإنسانِ هوَ ربُّ السَّبتِ“.</span>
</div>
</div>

## يسوع يشفي في السَّبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 3:1–6; ﻣﺘﻰ 12:9–14; لُوقا 4:31"> وفي سَبتٍ آخَرَ، دخَلَ المَجمعَ وأخذَ يُعَلِّمُ. وكانَ هُناكَ رجُلٌ يَدُهُ اليُمنى يابِسَةٌ.</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 20:20; المَزامِير 38:12; لُوقا 13:14"> فراقَبَ مُعَلِّمو الشَّريعَةِ والفَرِّيسيُّونَ يَسوعَ ليَرَوا هل يَشفي في السَّبتِ، فيَجِدوا ما يتَّهمونَهُ بِه.</span>
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 42:4; ﻣﺘﻰ 9:4; لُوقا 5:22"> وعرَفَ يَسوعُ أفكارَهُم، فقَالَ لِلرَّجُلِ الّذي يَدُهُ يابِسةٌ: ”قُمْ وقِفْ في وسْطِ المَجمعِ“. فقامَ الرَّجُلُ ووقَفَ هُناكَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="مَرْقُس 3:4; ﻣﺘﻰ 12:12–13; لُوقا 14:3"> فقَالَ لهُم يَسوعُ: ”أسألُكُم: أيحِلُّ في السَّبتِ عمَلُ الخَيرِ أم عمَلُ الشَّرِّ؟ إنقاذُ نَفسٍ أم إهلاكُها؟“</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 4:6–7; مَرْقُس 3:5; يُوحَنّا 5:8"> وأجالَ نَظرَهُ فيهِم جميعًا وقالَ لِلرَّجُلِ: ”مُدَّ يَدَكَ!“ فمَدَّها، فعادَت يدُهُ صَحيحةً.</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 2:1–2; اعمال الرسل 5:33; الجامِعَة 9:3"> فملأهُمُ الغضَبُ وتشاوروا كيفَ يَفعلونَ بيَسوعَ.</span>
</div>
</div>

## يسوع يختار الرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="العِبرانِيّين 5:7; مَرْقُس 6:46; مَرْقُس 1:35"> وفي تِلكَ الأيّامِ صَعِدَ إلى الجبَلِ ليُصلِّيَ، فقَضى اللَّيلَ كُلَّهُ في الصَّلاةِ للهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 3:13–19; ﻣﺘﻰ 9:36–4; رُؤيا يُوحَنّا 21:14"> ولمَّا طلَعَ الصُّبحُ، دعا تلاميذَهُ واَختارَ مِنهُم اَثني عشَرَ سمَّاهُم رُسُلًا، وهُم:</span>
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 1:13; ﻣﺘﻰ 4:21; ﻣﺘﻰ 4:18"> سِمعانُ الّذي سَمَّاهُ بُطرُسَ، وأنْدَراوُسُ أخوهُ، ويَعقوبُ ويوحنَّا، وفيلُبُّسُ وبَرتولَماوُسُ،</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 9:9; مَرْقُس 3:18; اعمال الرسل 1:13"> ومتَّى وتوما، ويَعقوبُ بنُ حَلْفى وسِمعانُ المُلقَّبُ بالوَطنيِّ الغيورِ،</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 14:22; يُوحَنّا 6:70–71; مَرْقُس 3:18"> ويَهوذا بنُ يَعقوبَ ويَهوذا أسخَريوطُ الّذي صارَ خائِنًا.</span>
</div>
</div>

## يسوع يعلم ويشفي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 11:21; ﻣﺘﻰ 4:23–25; المَزامِير 107:17–20"> ثُمَّ نزَلَ يَسوعُ معَهُم فوقَفَ في مكانٍ سهلٍ، وهُناكَ جُمهورٌ مِنْ تلاميذِهِ وجَمعٌ كبيرٌ مِنَ النَّاسِ مِنْ جميعِ اليَهوديَّةِ وأُورُشليمَ وساحِلِ صورَ وصيدا،</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 17:15; ﻣﺘﻰ 15:22; اعمال الرسل 5:16"> جاؤُوا لِيَسمَعوهُ وليَشفيَهُم مِنْ أمراضِهِم. وكانَ الّذينَ تُعذِّبُهُمُ الأرواحُ النَّجِسَةُ يَنالونَ الشِّفاءَ أيضًا.</span>
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 14:36; مَرْقُس 3:10; اعمال الرسل 5:15–16"> وحاوَلَ جميعُ النَّاسِ أنْ يلمَسوهُ، لأنَّ قُوَّةً كانَت تَخرُجُ مِنهُ وتَشفيهِم كُلَّهُم.</span>
</div>
</div>

<div className="bp">

<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 5:10–12; ﻣﺘﻰ 10:22; رسالَةُ بُطرُسَ الأُولَى 4:12–16"> هَنيئًا لكُم إذا أبغَضَكُمُ النَّاسُ وطَردوكُم وعَيَّروكُم ونَبذوكُم نَبْذَ الأشرارِ مِنْ أجلِ إبنِ الإنسانِ.</span>

</div>
<div className="bp">

## محبة الأعداء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:15; لُوقا 6:35; ﺭﻭﻣﻴﺔ 12:17–21"> ”ولكنِّي أقولُ لكُم أيُّها السّامِعونَ: أحِبُّوا أعداءَكُم، وأحسِنوا إلى مُبغِضيكُم،</span>
<span className="v" id="v28" data-v="28" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:12; ﺭﻭﻣﻴﺔ 12:14"> وبارِكوا لاعِنيكُم، وصَلُّوا لأجلِ المُسيئينَ إلَيكُم.</span>
<span className="v" id="v29" data-v="29" data-xref="العِبرانِيّين 10:34; ﻣﺘﻰ 5:39–42; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:20"> مَنْ ضَربَكَ على خَدِّكَ، فحَوِّلْ لَهُ الآخرَ. ومَنْ أخَذَ رِداءَكَ، فلا تَمْنَعْ عَنهُ ثَوبَكَ.</span>
<span className="v" id="v30" data-v="30" data-xref="التَّثْنِيَة 15:7–10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:6–14; ﻣﺘﻰ 6:12"> ومَنْ طلَبَ مِنكَ شَيئًا فأعطِهِ، ومَنْ أخَذَ ما هوَ لكَ فلا تُطالِبْهُ بِه.</span>
<span className="v" id="v31" data-v="31" data-xref="ﻣﺘﻰ 7:12; غَلاطِيَّة 5:14; ﻣﺘﻰ 22:39"> وعامِلوا النَّاسَ مِثلَما تُريدونَ أنْ يُعامِلوكُم.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 5:46–47; رسالَةُ بُطرُسَ الأُولَى 2:19–20"> فإنْ أحبَبْتُم مَنْ يُحبُّونكُم، فأيُّ فَضْلٍ لكُم؟ لأنَّ الخاطِئينَ أنفُسَهُم يُحبُّونَ مَنْ يُحبُّونَهُم.</span>
<span className="v" id="v33" data-v="33"> وإنْ أحسَنتُم إلى المُحسِنينَ إلَيكُم فأيُّ فَضْلٍ لكُم؟ لأنَّ الخاطِئينَ أنفُسَهُم يَعمَلونَ هذا.</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 5:42; لُوقا 14:12–14; لُوقا 6:35"> وإنْ أقرَضْتُم مَنْ تَرجُونَ أنْ تَستَرِدُّوا مِنهُم قَرضَكُم، فأيُّ فَضلٍ لكُم؟ لأنَّ الخاطِئينَ أنفُسَهُم يُقرِضونَ الخاطِئينَ ليَستَرِدُّوا قَرضَهُم.</span>
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 5:44–45; يُوحَنّا 13:35; الأمثال 19:17"> ولكن أحِبُّوا أعداءَكُم، أحسِنوا وأقرِضوا غَيرَ راجِينَ شيئًا، فيكونَ أجرُكُم عظيمًا، وتكونوا أبناءَ اللهِ العليِّ، لأنَّهُ يُنعِمُ على ناكِرِي الجميلِ والأشرارِ.</span>
<span className="v" id="v36" data-v="36" data-xref="أفَسُس 5:1–2; ﻣﺘﻰ 5:48; رسالَةُ بُطرُسَ الأُولَى 1:15–16"> كونوا رُحماءَ كما أنَّ اللهَ أباكُم رحيمٌ.</span>
</div>
</div>

<div className="bp">
