## يسوع يدعو التَّلاميذ الأولين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 1:16–20; ﻣﺘﻰ 4:18–22; مَرْقُس 3:9"> وكانَ يَسوعُ على شاطئِ بُحَيرَةِ جَنِّيسارَتَ، فاَزدَحَمَ النَّاسُ علَيهِ ليَسمَعوا كلامَ اللهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="مَرْقُس 1:19; ﻣﺘﻰ 4:21"> ورأى قارِبَينِ راسِيَينِ عِندَ الشَّاطئِ، خرَجَ مِنهُما الصيّادونَ ليَغسِلوا شباكَهُم.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 4:18; ﻣﺘﻰ 13:1–2; يُوحَنّا 1:41–42"> فصَعِدَ إلى واحدٍ مِنهُما، وكانَ لسِمعانَ، وطلَبَ مِنهُ أنْ يَبتَعدَ قليلًا عَنِ البرِّ. وجلَسَ يَسوعُ في القارِبِ يُعَلِّمُ الجُموعَ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 21:6; ﻣﺘﻰ 17:27"> ولمَّا ختَمَ كلامَهُ، قالَ لسِمعانَ: ”سِرْ إلى العُمقِ وألقوا شِباكَكُم لِلصَّيدِ“.</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 2:5; يُوحَنّا 15:14; لُوقا 6:46–48"> فأجابَهُ سِمعانُ: ”تَعِبنا اللَّيلَ كُلَّهُ، يا معلِّمُ، وما اَصطَدنا شيئًا. ولكنِّي أُلقي الشِّباكَ إجابَةً لِطَلَبِكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:58; يُوحَنّا 21:6–11; الجامِعَة 11:6"> وفَعلوا ذلِكَ فأمسكوا سَمكًا كثيرًا، وكادَت شِباكُهُم تَتَمزَّقُ.</span>
<span className="v" id="v7" data-v="7" data-xref="غَلاطِيَّة 6:2; فِيلِبِّي 4:3; اعمال الرسل 11:25"> فأشاروا إلى شُركائِهِم في القارِبِ الآخرِ أنْ يَجيئوا ويُساعِدوهُم. فَجاؤُوا وملأوا القارِبَينِ حتّى كادا يَغرَقانِ.</span>
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 6:5; أيُّوب 42:5–6; رُؤيا يُوحَنّا 1:17"> فلمَّا رأى سِمعانُ ما جرى وقَعَ على رُكبَتَي يَسوعَ وقالَ: ”إِبتعِدْ عنِّي، يا ربُّ! أنا رَجُلٌ خاطِئٌ“.</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 4:32; المَزامِير 8:8; مَرْقُس 9:6"> وكانَ في دَهشةٍ هوَ ورِفاقُهُ كُلُّهُم لِكثرَةِ السَّمكِ الّذي اَصطادوهُ.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 4:19; ﻣﺘﻰ 13:47; حِزْقِيال 47:9–10"> ومِثلُهُم يَعقوبُ ويوحنَّا، اَبنا زَبدي وشَريكا سِمعانَ. فقَالَ يَسوعُ لسِمعانَ: ”لا تخَفْ! ستكونُ بَعدَ اليومِ صيّادَ بشَرٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 4:20; لُوقا 5:28; ﻣﺘﻰ 19:27"> ولمَّا رَجَعوا بالقارِبَينِ إلى البَرِّ، تَركوا كُلَّ شيءٍ وتَبِعوا يَسوعَ.</span>
</div>
</div>

## يسوع يشفي أبرص

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 50:15; لُوقا 17:16; ﻣﺘﻰ 8:2–4"> وبَينَما هوَ في إحدى المُدُنِ، إذا برَجُلٍ غطَّى جسَدَهُ البَرَصُ، فلمَّا رأى يَسوعَ اَرتَمى على وجهِهِ وتَوسَّلَ إليهِ بِقولِهِ: ”يا سيِّدي، إنْ أرَدتَ فأنتَ قادِرٌ أنْ تُطَهِّرَني!“</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 9:29–30; يُوحَنّا 4:50–53; هُوشَع 14:4"> فمَدَّ يَسوعُ يدَهُ ولمَسَهُ وقالَ لَهُ: ”أريدُ، فاَطْهُرْ!“ فزالَ عَنهُ البَرَصُ في الحالِ.</span>
<span className="v" id="v14" data-v="14" data-xref="اللّاويِّين 14:2–32; لُوقا 17:14; مَرْقُس 1:44"> فأوصاهُ يَسوعُ أنْ لا يُخبِرَ أحدًا، وقالَ لَهُ: ”لكنِ إِذهَبْ إلى الكاهنِ وأرِهِ نَفسَكَ، ثُمَّ قَدِّمْ عَنْ شِفائِك ما أمرَ بِه موسى، شَهادةً عِندَهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 15:30–31; لُوقا 12:1; مَرْقُس 1:45–2"> وزادَ صيتُ يَسوعَ اَنتِشارًا، فأقبلَتْ إلَيهِ جُموعٌ كبيرةٌ لِتَسمَعَهُ وتُشفى مِنْ أمراضِها.</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 6:12; مَرْقُس 1:35–36; ﻣﺘﻰ 14:23"> ولكِنَّهُ كانَ يَعتَزِلُ في البراري لِيُصلِّي.</span>
</div>
</div>

## يسوع يشفي كسيحًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="لُوقا 6:19; لُوقا 8:46; ﻣﺘﻰ 11:5"> وكانَ في أحَدِ الأيّامِ يُعَلِّمُ، وبَينَ الحُضورِ بَعضُ الفَرِّيسيِّينَ ومُعَلِّمي الشَّريعةِ جاؤُوا مِنْ جَميعِ قُرى الجَليلِ واليهوديَّةِ ومِنْ أُورُشليمَ. وكانَت قُدرةُ الرَّبِّ تَشفي المَرضى على يَدِهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="مَرْقُس 2:3–12; ﻣﺘﻰ 9:2–8; يُوحَنّا 5:5–6"> وجاءَ بَعضُ النَّاسِ يَحمِلونَ كسيحًا على سَريرٍ، وحاولوا أنْ يَدخُلوا بِهِ ليَضعوهُ أمامَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 2:4; ﻣﺘﻰ 10:27; ﻣﺘﻰ 24:17"> فلمَّا عَجَزوا عَنِ الدُّخولِ لِكَثرةِ الزِّحامِ، صَعِدوا بهِ إلى السَّطحِ وكشَفُوا مكانًا فيه ودَلَّوهُ معَ فِراشهِ (من بين القرميد)، إلى وَسْطِ المَجلِسِ قُدَّامَ يَسوعَ.</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 14:9; ﻣﺘﻰ 9:2; مَرْقُس 2:5"> فلمَّا رأى يَسوعُ إيمانَهُم، قالَ لِلكَسيحِ: ”يا رَجُلُ، مَغفورَةٌ لكَ خَطاياكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 43:25; مَرْقُس 2:6–7; لُوقا 7:49"> فأخَذَ مُعَلِّمو الشَّريعةِ والفَرِّيسيُّونَ يَقولونَ في أنفُسِهم: ”مَنْ هذا الّذي يَنطِقُ بالتَّجديفِ؟ مَنْ يَقدِرُ أنْ يَغفِرَ الخطايا إلَّا اللهُ وَحدَهُ!“</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="لُوقا 24:38; إشَعْياء 66:18; المَزامِير 139:2"> فعَرَفَ يَسوعُ أفكارَهُم، فأجابَهُم: ”ما هذِهِ الأفكارُ في قلوبِكُم؟</span>
<span className="v" id="v23" data-v="23" data-xref="مَرْقُس 2:9; ﻣﺘﻰ 9:5"> أيُّما أسهَلُ؟ أنْ يُقالَ: مَغفورةٌ لكَ خَطاياكَ، أمْ أنْ يُقالَ: قُمْ واَمْشِ.</span>
<span className="v" id="v24" data-v="24" data-xref="رُؤيا يُوحَنّا 1:13; إشَعْياء 53:11; لُوقا 8:54"> سأُريكُم أنَّ اَبنَ الإنسانِ لَهُ سُلطانٌ على الأرضِ ليغفِرَ الخطايا“. وقالَ لِلكسيحِ: ”أقولُ لكَ: قُمْ واَحمِلْ فِراشَكَ واَذهَبْ إلى بَيتِكَ!“</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 9:24; لُوقا 5:13; المَزامِير 107:20–22"> فقامَ الرَّجُلُ في الحالِ بِمَشهَدٍ مِنَ الحاضِرينَ، وحمَلَ فِراشَهُ وذهَبَ إلى بَيتِهِ وهوَ يَحمَدُ اللهَ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 9:8; ﻣﺘﻰ 12:23; لُوقا 7:16"> فاَستولَتِ الحَيْرَةُ علَيهِم كُلِّهِم، فمَجَّدوا اللهَ. وملأهُمُ الخوفُ، فقَالوا: ”اليومَ رأينا عَجائِبَ!“</span>
</div>
</div>

## يسوع يدعو لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="مَرْقُس 2:13–22; ﻣﺘﻰ 9:9–17; يُوحَنّا 1:43"> وخرَجَ يَسوعُ بَعدَ ذلِكَ، فرأى جابيًا لِلضرائِبِ إِسمُهُ لاوي، جالِسًا في بَيتِ الجِبايَةِ، فقَالَ لَه يَسوعُ: ”إِتبَعْني!“</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 5:11; ﻣﺘﻰ 19:22–27; المُلُوكِ الأوَّلُ 19:19–21"> فقامَ وترَكَ كُلَّ شيءٍ وتَبِعَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="مَرْقُس 2:15; لُوقا 15:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:27"> وأقامَ لَهُ لاوي وَليمةً كَبيرةً في بيتِهِ، حضَرَها عدَدٌ كبيرٌ مِنْ جُباةِ الضَّرائبِ وغيرِهِم.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 21:28–32; لُوقا 5:17; مَرْقُس 7:3"> فقَالَ الفَرِّيسيُّونَ ومُعَلِّمو الشَّريعةِ مِنْ أتباعِهِم لِتلاميذِهِ مُتذَمِّرينَ: ”لماذا تأكُلونَ وتَشربونَ معَ جُباةِ الضَّرائِبِ والخاطِئينَ؟“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="مَرْقُس 2:17; ﻣﺘﻰ 9:12–13; إرْمِيا 8:22"> فأجابَ يَسوعُ: ”لا يَحتاجُ الأصِحَّاءُ إلى طبيبٍ، بَلِ المَرضى.</span>
<span className="v" id="v32" data-v="32" data-xref="لُوقا 19:10; لُوقا 15:10; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:15–16"> ما جِئتُ لأدعُوَ الأَبرارَ إلى التَّوبَةِ، بَلِ الخاطِئينَ“.</span>
</div>
</div>

## الصوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="ﻣﺘﻰ 9:14–17; مَرْقُس 2:18–22; يُوحَنّا 3:25"> وقالَ بَعضُهُم لِيَسوعَ: ”تلاميذُ يوحنَّا يَصومُونَ ويُصلُّونَ كثيرًا، ومثلُهُم تلاميذُ الفَرِّيسيِّينَ، أمَّا تلاميذُكَ فيأكُلونَ ويَشرَبونَ!“</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="يُوحَنّا 3:29; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:2; إشَعْياء 54:5"> فأجابَهُم يَسوعُ: ”أتقدِرونَ أنْ تَجعَلوا أهلَ العَريسِ يَصومونَ، والعَريسُ بَينهُم؟</span>
<span className="v" id="v35" data-v="35" data-xref="لُوقا 17:22; اعمال الرسل 14:23; اعمال الرسل 3:21"> ولكِنْ يَجيءُ وَقتٌ يُرفَعُ فيهِ العريسُ مِنْ بَينِهِم، وفي ذلِكَ الوقتِ يَصومُونَ“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 9:16–17; مَرْقُس 2:21–22; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:16"> وقالَ لهُم هذا المثَلَ: ”ما مِنْ أحدٍ يَنتَزعُ قِطعَةً مِنْ ثوبٍ جديدٍ لِترقيعِ ثوبٍ عتيقٍ، لِئَلَّا يُشَقَّ الثَّوبُ الجديدُ وتكونَ الرِقعةُ الّتي اَنتزَعها مِنهُ لا تُلائِمُ الثَّوبَ العتيقَ.</span>
<span className="v" id="v37" data-v="37" data-xref="يَشُوع 9:13; المَزامِير 119:83; يَشُوع 9:4"> وما مِنْ أحدٍ يضَعُ خَمرًا جَديدةً في أوعِيَةِ جِلدٍ عَتيقَةٍ، لِئَلَّا تَشُقَّ الخَمرُ الجديدةُ هذِهِ الأوعِيَةَ، فتَسيلَ الخَمرُ وتُتلَف الأوعِيَةُ.</span>
<span className="v" id="v38" data-v="38" data-xref="العِبرانِيّين 8:8–13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:17; حِزْقِيال 36:26"> بَلْ تُوضَعُ الخَمرُ الجديدةُ في أوعِيَةٍ جديدةٍ،</span>
<span className="v" id="v39" data-v="39" data-xref="العِبرانِيّين 11:39; مَرْقُس 7:7–13; إرْمِيا 6:16"> وما مِنْ أحَدٍ يشرَبُ خَمرًا مُعتَّقَةً ثُمَّ يَرغَبُ في الخَمرِ الجديدةِ، لأنَّهُ يقولُ: ”الخَمرُ المعَتَّقةُ طيِّبةٌ!“</span>
</div>
