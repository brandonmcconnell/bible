## المؤامرة وخيانة يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 12:6–23; ﻣﺘﻰ 26:2–5; يُوحَنّا 11:55–57"> وقَرُبَ عيدُ الفَطيرِ الّذي يُقالُ لَهُ الفِصحُ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 11:47–53; ﻣﺘﻰ 12:14; لُوقا 20:19"> وكانَ رُؤساءُ الكَهنَةِ ومُعَلِّمو الشَّريعةِ يَبحَثونَ عَنْ طريقةٍ يَقتُلونَ بِها يَسوعَ، لأنَّهُم كانوا يخافُونَ مِنَ الشَّعبِ.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 13:2; مَرْقُس 14:10–11; يُوحَنّا 13:26–27"> فدخَلَ الشَّيطانُ في يَهوذا المُلقَّبِ بالإسخَريوطيِّ، وهوَ مِنَ التَّلاميذِ الاثنَي عشَرَ،</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 4:1; اعمال الرسل 5:24; لُوقا 22:52"> فذهَبَ وفاوَضَ رُؤساءَ الكَهنَةِ وقادَةَ حرَسِ الهَيكَلِ كيفَ يُسَلِّمُهُ إليهِم.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 26:15–16; اعمال الرسل 1:18; ﻣﺘﻰ 27:3–5"> ففَرِحوا واَتَّفَقُوا أنْ يُعطوهُ شَيئًا مِنَ المالِ.</span>
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 14:2; ﻣﺘﻰ 26:5"> فرَضِيَ وأخَذَ يترَقَّبُ الفُرصةَ ليُسلِّمَهُ إليهِم بالخِفيَةِ عَنِ الشَّعبِ.</span>
</div>

## الاستعداد لعشاء الفصح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 12:18; مَرْقُس 14:12–16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:7"> وجاءَ يومُ الفَطيرِ، وفيهِ تُذبَحُ الخِرافُ لِعَشاءِ الفِصحِ.</span>
<span className="v" id="v8" data-v="8" data-xref="مَرْقُس 14:13–16; اعمال الرسل 4:19; اعمال الرسل 4:13"> فأرسَلَ يَسوعُ بُطرُسَ ويوحنَّا وقالَ لهُما: ”إِذهَبا هَيِّئا لنا عَشاءَ الفِصحِ لنأكُلَهُ“.</span>
<span className="v" id="v9" data-v="9"> فقَالا لَهُ: ”أينَ تُريدُ أنْ نُهيِّئَهُ؟“</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 16:4; ﻣﺘﻰ 26:18–19; اعمال الرسل 8:26–29"> فأجابَهُما: ”عِندَما تَدخُلانِ المدينةَ يُلاقيكُما رَجُلٌ يَحمِلُ جَرَّةَ ماءٍ، فاَتبَعاهُ إلى البَيتِ الّذي يَدخُلُهُ،</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 11:28; رُؤيا يُوحَنّا 3:20; لُوقا 19:34"> وقولا لِرَبِّ البَيتِ: يقولُ لكَ المُعَلِّمُ: أينَ الغُرفةُ الّتي سأتناوَلُ فيها عَشاءَ الفِصحِ معَ تلاميذي؟</span>
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 16:14–15; اعمال الرسل 1:13; يُوحَنّا 2:25"> فيُريكُما في أعلى البَيتِ غُرفَةً واسِعةً مَفروشَةً، وهُناكَ تُهيِّئانِهِ“.</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 21:33; لُوقا 19:32; يُوحَنّا 2:5"> فذَهبا ووَجَدا مِثلَما قالَ لهُما، فقاما بتَهيئَةِ عَشاءِ الفِصحِ.</span>
</div>
</div>

## عشاء الرّبّ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 26:20; مَرْقُس 14:17; التَّثْنِيَة 16:6–7"> ولمَّا جاءَ الوقتُ، جلَسَ يَسوعُ معَ الرُّسُلِ لِلطَعامِ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 12:50; يُوحَنّا 4:34; يُوحَنّا 17:1"> فقَالَ لهُم: ”كَمِ اَشتَهَيتُ أنْ أتناوَلَ عَشاءَ هذا الفِصحِ معكُم قَبلَ أنْ أتألَّمَ.</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 14:15; رُؤيا يُوحَنّا 19:9; لُوقا 22:30"> أقولُ لكُم: لا أتناوَلُهُ بَعدَ اليومِ حتّى يتِمَّ في مَلكوتِ اللهِ“.</span>
<span className="v" id="v17" data-v="17" data-xref="لُوقا 22:19; المَزامِير 116:13; التَّثْنِيَة 8:10"> وأخَذَ يَسوعُ كأسًا وشكَرَ وقالَ: ”خُذوا هذِهِ الكأسَ واَقتَسِموها بَينَكُم.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 26:29; لُوقا 22:16; مَرْقُس 14:25"> أقولُ لكُم: لا أشرَبُ بَعدَ اليومِ مِنْ عَصيرِ الكَرمَةِ حتّى يَجيءَ مَلكوتُ اللهِ“.</span>
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:23–29; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:16; مَرْقُس 14:22–24"> وأخَذَ خُبزًا وشكَرَ وكسَرَهُ وناوَلَهُم وقالَ: ”هذا هوَ جَسَدي الّذي يُبذَلُ مِنْ أجلِكُم. إِعمَلوا هذا لِذِكري“.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 26:28; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:25; زَكَريّا 9:11"> وكذلِكَ الكأسُ أيضًا بَعدَ العَشاءِ، فقَالَ: ”هذِهِ الكأسُ هيَ العَهدُ الجديدُ بِدَمي الّذي يُسفَكُ مِنْ أجلِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 41:9; يُوحَنّا 13:26; يُوحَنّا 13:21–22"> لكِنْ ها هيَ يدُ الّذي يُسلِّمُني على المائِدَةِ مَعي.</span>
<span className="v" id="v22" data-v="22" data-xref="اعمال الرسل 2:23; المَزامِير 109:6–15; المَزامِير 55:12–15"> فاَبنُ الإنسانِ سيَموتُ كما هوَ مكتوبٌ لَهُ، ولكِنَّ الوَيلَ لِمَنْ يُسَلِّمُهُ!“</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 13:22–25; مَرْقُس 14:19; ﻣﺘﻰ 26:22"> فأخَذَ التَّلاميذُ يَتَساءلونَ مَنْ مِنهُم سيَفعلُ هذا.</span>
</div>
</div>

## من هو الأكبر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="لُوقا 9:46; مَرْقُس 9:34; مَرْقُس 10:37–41"> ووقَعَ بَينَهُم جِدالٌ في مَنْ يكونُ أكبرَهُم،</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 20:25–28; مَرْقُس 10:41–45"> فقَالَ لهُم يَسوعُ: ”مُلوكُ الأمَمِ يَسودونَها، وأصحابُ السُّلطَةِ فيها يُريدونَ أنْ يَدعُوَهُمُ النَّاسُ مُحسِنينَ.</span>
<span className="v" id="v26" data-v="26" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:5; لُوقا 9:48; ﻣﺘﻰ 18:3–5"> أمَّا أنتُم، فما هكذا حالُكُم، بل ليَكُنِ الأكبَرُ فيكُم كالأصغَرِ، والرَّئيسُ كالخادِمِ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 20:28; لُوقا 12:37; يُوحَنّا 13:5–16"> فمَنْ هوَ الأكبَرُ: الجالِسُ لِلطَّعامِ أمِ الّذي يَخدُمُ؟ أما هوَ الجالِسُ لِلطَّعام؟ وأنا بَينَكُم مِثلُ الّذي يَخدُمُ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="العِبرانِيّين 2:18; العِبرانِيّين 4:15; اعمال الرسل 1:25"> وأنتُم ثَبَتُّم مَعي في مِحنَتي،</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 25:34; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:12; يَعقُوب 2:5"> وأنا أُعطيكُم مَلكوتًا كما أعطاني أبي،</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 19:28; ﻣﺘﻰ 8:11; رُؤيا يُوحَنّا 2:26–27"> فتأكُلونَ وتَشرَبونَ على مائِدَتي في مَلكوتي، وتَجلِسونَ على عُروشٍ لتَدينوا عشائِرَ بَني إِسرائيلَ الاثنَي عشَرَ“.</span>
</div>
</div>

## يسوع يُنبئ بإنكار بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:8; زَكَريّا 3:1; أيُّوب 1:6–12"> وقالَ الرَّبُّ يَسوعُ: ”سِمعانُ، سِمعانُ! ها هوَ الشَّيطانُ يَطلُبُ أنْ يُغَربِلَكُم مِثلَما يُغَربِلُ الزّارِعُ القَمحَ.</span>
<span className="v" id="v32" data-v="32" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:8–10; يُوحَنّا 21:15–17; يُوحَنّا 17:9–11"> ولكنّي طَلبَتُ لكَ أنْ لا تَفقِدَ إيمانَكَ. وأنتَ متى رَجَعتَ، ثَبِّتْ إخوانَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 14:31; ﻣﺘﻰ 26:33–35; مَرْقُس 14:29"> فأجابَهُ سِمعانُ: ”يا رَبُّ، أنا مُستَعِدٌّ أنْ أذهَبَ معَكَ إلى السِّجنِ وإلى الموتِ“.</span>
<span className="v" id="v34" data-v="34" data-xref="يُوحَنّا 13:38; ﻣﺘﻰ 26:34; مَرْقُس 14:30"> فقَالَ لَهُ يَسوعُ: ”أقولُ لكَ يا بُطرُسُ: لا يَصيحُ الدّيكُ اليومَ حتّى تُنكِرَني ثلاثَ مَرَّاتٍ“.</span>
</div>
</div>

## الاستعداد للمحنة الكبرى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 10:9–10; لُوقا 10:4; المَزامِير 23:1"> ثُمَّ قالَ لِتلاميذِهِ: ”عِندَما أرسَلتُكُم بِلا مالٍ ولا كِيسٍ ولا حِذاءٍ هل اَحتَجْتُم إلى شيءٍ؟“ قالوا: ”لا“.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 10:22–25; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 3:4; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:14–15"> فقَالَ لهُم: ”أمَّا الآنَ، فمَنْ عِندَهُ مالٌ فَلْيأخُذْهُ، أو كِيسٌ فَلْيَحمِلْهُ. ومَنْ لا سيفَ عِندَهُ، فَلْيبِعْ ثوبَهُ ويَشتَرِ سَيفًا.</span>
<span className="v" id="v37" data-v="37" data-xref="إشَعْياء 53:12; لُوقا 23:32; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:21"> أقولُ لكُم: يَجبُ أنْ تَتِمَّ فيَّ هذِهِ الآيةُ: وأحصَوهُ معَ المُجرِمينَ. وما جاءَ عنِّي لا بُدَّ أنْ يَتِمَّ“.</span>

</div>
</div>

## يسوع يصلي في جبل الزيتون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="لُوقا 21:37; يُوحَنّا 18:1–2; ﻣﺘﻰ 21:1"> ثُمَّ خرَجَ وذهَبَ كَعادَتِهِ إلى جبَلِ الزَّيتونِ يَتبَعُهُ تلاميذُهُ.</span>
<span className="v" id="v40" data-v="40" data-xref="ﻣﺘﻰ 6:13; ﻣﺘﻰ 26:36–46; لُوقا 22:46"> ولمَّا وصَلَ إلى المكانِ قالَ لهُم: ”صَلُّوا لِئَلَّا تَقَعُوا في التَّجرِبَةِ“.</span>
<span className="v" id="v41" data-v="41" data-xref="ﻣﺘﻰ 26:39; مَرْقُس 14:35; لُوقا 18:11"> واَبتعَدَ عَنهُم مَسافةَ رَميةِ حجَرٍ وركَعَ وصَلَّى،</span>
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 12:27–28; ﻣﺘﻰ 26:39; ﻣﺘﻰ 26:42"> فقَالَ: ”يا أبي، إنْ شِئْتَ، فأَبْعِدْ عنِّي هذِهِ الكأسَ! ولكِنْ لِتكُنْ إرادتُكَ لا إرادتي“.</span>
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 4:11; ﻣﺘﻰ 26:53; المَزامِير 91:11–12"> وظهَرَ لَه مَلاكٌ مِنَ السَّماءِ يقوِّيهِ.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:12; العِبرانِيّين 5:7; المَزامِير 22:12–21"> ووقَعَ في ضِيقٍ، فأجهَدَ نَفسَهُ في الصَّلاةِ، وكانَ عَرَقُهُ مِثلَ قَطَراتِ دَمٍ تتَساقَطُ على الأرض.</span>
<span className="v" id="v45" data-v="45" data-xref="مَرْقُس 14:40–41; ﻣﺘﻰ 26:43; ﻣﺘﻰ 26:40"> وقامَ عَنِ الصَّلاةِ ورجَعَ إلى التَّلاميذِ، فوجَدَهُم نِيامًا مِنَ الحُزنِ.</span>
<span className="v" id="v46" data-v="46" data-xref="لُوقا 22:40; ﻳﻮﻧﺎﻥ 1:6; لُوقا 21:34–36"> فقَالَ لهُم: ”ما بالُكُم نائِمينَ؟ قوموا وصَلّوا لِئَلَّا تَقَعوا في التَّجرِبَةِ“.</span>
</div>

## اعتقال يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v47" data-v="47" data-xref="مَرْقُس 14:41–50; يُوحَنّا 18:2–11; لُوقا 22:3–6"> وبَينَما هوَ يتكَلَّمُ، ظهَرَت عِصابَةٌ يَقودُها المدعوُّ يَهوذا، أحَدُ التَّلاميذِ الاثنَي عشَرَ، فدَنا مِنْ يَسوعَ ليُقبِّلَهُ.</span>
<span className="v" id="v48" data-v="48" data-xref="مَرْقُس 14:44–46; صموئيلَ الثّانِي 20:9–10; الأمثال 27:6"> فقَالَ لَهُ يَسوعُ: ”أبِقُبلَةٍ، يا يَهوذا، تُسَلِّمُ اَبنَ الإنسانِ؟“</span>
</div>
<div className="bp">

## بطرس ينكر يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v55" data-v="55" data-xref="المَزامِير 26:4–5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:15–17; مَرْقُس 14:66–72"> وأوقَدَ الحرَسُ نارًا في ساحَةِ الدَّارِ وقَعَدُوا حَولَها، وقعَدَ بُطرُسُ بَينَهُم.</span>

<span className="v" id="v57" data-v="57" data-xref="رسالَةُ يُوحَنّا الأُولَى 1:9; يُوحَنّا 18:25; ﻣﺘﻰ 26:70"> فأنكَرَ قالَ: ”أنا لا أعرِفُهُ، يا اَمرأةُ!“</span>

<span className="v" id="v58" data-v="58" data-xref="يُوحَنّا 18:25; ﻣﺘﻰ 26:71–72; مَرْقُس 14:69–70"> وبَعدَ قليلٍ رَآهُ رَجُلٌ فقَالَ: ”وأنتَ مِنهُم!“ فأجابَهُ بُطرُسُ: ”كلّا، يا رَجُلُ!“</span>

<span className="v" id="v62" data-v="62" data-xref="ﻣﺘﻰ 26:75; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:12; مَرْقُس 14:72"> فخرَجَ وبكى بُكاءً مُرًّا.</span>
