## مثل الخروف الضائع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:15; ﻣﺘﻰ 9:10–13; لُوقا 5:29–32"> وكانَ جُباةُ الضَّرائبِ والخاطِئونَ يَدنُونَ مِنْ يَسوعَ ليَسمَعوهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 9:11; لُوقا 19:7; لُوقا 7:39"> فقَالَ الفَرِّيسيُّونَ ومُعَلِّمو الشَّريعةِ مُتذمِّرينَ: ”هذا الرَّجُلُ يُرَحِّبُ بالخاطِئينَ ويأكُلُ مَعَهُم!“</span>
<span className="v" id="v3" data-v="3"> فكَلَّمَهُم بِهذا المَثَلِ:</span>
<span className="v" id="v4" data-v="4" data-xref="حِزْقِيال 34:16; حِزْقِيال 34:11–12; ﻣﺘﻰ 18:12–14"> ”مَنْ مِنكُم إذا كانَ لَهُ مئةُ خروفٍ، فأضاعَ واحدًا مِنها، لا يَترُكُ التِّسعَةَ والتِّسعينَ في البرِّيَّةِ ليَبحَثَ عَنِ الخَروفِ الضَّائِعِ حتّى يَجدَهُ؟</span>
<span className="v" id="v5" data-v="5" data-xref="العِبرانِيّين 12:2; لُوقا 19:9; إشَعْياء 46:3–4"> فإذا وجَدَهُ حَمَلهُ على كَتِفَيهِ فَرِحًا،</span>
<span className="v" id="v6" data-v="6" data-xref="لُوقا 15:10; لُوقا 15:24; رسالَةُ بُطرُسَ الأُولَى 2:25"> ورجَعَ إلى البَيتِ ودَعا أصدِقاءَهُ وجيرانَهُ وقالَ لهُم: إفرَحوا مَعي، لأنِّي وجَدتُ خَروفي الضائِعَ!</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 5:32; لُوقا 15:10; لُوقا 15:32"> أقولُ لكُم: هكذا يكونُ الفَرَحُ في السَّماءِ بِخاطئٍ واحِدٍ يَتوبُ أكثَرَ مِنَ الفَرحِ بِتسعَةٍ وتِسعينٍ مِنَ الأبرارِ لا يَحتاجونَ إلى التَّوبَةِ.</span>
</div>
</div>

## مثل الابن الضال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 21:23–31"> وقالَ يَسوعُ: ”كانَ لِرَجلٍ اَبنانِ،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 21:16–17; مَرْقُس 12:44; لُوقا 15:30"> فقَالَ لَهُ الأصغَرُ: يا أبي أعطِني حِصَّتي مِنَ الأملاكِ. فقَسَم لهُما أملاكَهُ.</span>
<span className="v" id="v13" data-v="13" data-xref="الأمثال 29:3; إرْمِيا 2:5; الأمثال 21:17"> وبَعدَ أيّامٍ قَليلةٍ، جمَعَ الابنُ الأصغَرُ كُلَّ ما يَملِكُ، وسافَرَ إلى بِلادٍ بَعيدَةٍ، وهُناكَ بَدَّدَ مالَهُ في العَيشِ بِلا حِسابٍ.</span>
<span className="v" id="v14" data-v="14" data-xref="أخبارِ الأيّامِ الثّانِي 33:11; عامُوس 8:9–12; هُوشَع 2:9–14"> فلمَّا أنفَقَ كُلَّ شيءٍ، أصابَت تِلكَ البلادَ مَجاعةٌ قاسِيَةٌ، فوقَعَ في ضيقٍ.</span>
<span className="v" id="v15" data-v="15" data-xref="تِيطُس 3:3; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:25–26; إشَعْياء 1:5"> فلَجأ إلى العَمَلِ عِندَ رَجُلٍ مِنْ أهلِ تِلكَ البِلادِ، فأرسَلَهُ إلى حُقولِهِ ليَرعى الخنازيرَ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 6:19–21; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:5; هُوشَع 12:1"> وكانَ يَشتَهي أنْ يَشبَعَ مِنَ الخُرنُوبِ الّذي كانَتِ الخنازيرُ تأكُلُهُ، فلا يُعطيهِ أحدٌ.</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 18:28; أفَسُس 2:4–5; لُوقا 15:18–19"> فرَجَعَ إلى نَفسِهِ وقالَ: كم أَجيرٍ عِندَ أبي يَفضُلُ عَنهُ الطَّعامُ، وأنا هُنا أموتُ مِنَ الجوعِ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:40; رسالَةُ يُوحَنّا الأُولَى 1:8–10; لُوقا 15:21"> سأقومُ وأرجِعُ إلى أبي وأقولُ لَه: يا أبي، أخطَأتُ إلى السماءِ وإلَيكَ،</span>
<span className="v" id="v19" data-v="19" data-xref="أيُّوب 42:6; لُوقا 5:8; المَزامِير 84:10"> ولا أستحِقُّ بَعدُ أنْ أُدعى لكَ اَبنًا، فعامِلْني كأَجيرٍ عِندَكَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 103:10–13; المَزامِير 86:5; المَزامِير 86:15"> فقامَ ورجَعَ إلى أبيهِ. فَرآهُ أبوهُ قادِمًا مِنْ بَعيدٍ، فأشفَقَ علَيهِ وأسرَعَ إلَيهِ يُعانِقُهُ ويُقَبِّلُهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 51:4; لُوقا 15:18–19; إرْمِيا 3:13"> فقَالَ لَهُ الابنُ: يا أبي، أخطَأْتُ إلى السَّماءِ وإلَيكَ، ولا أستَحِقُّ بَعدُ أنْ أُدعى لكَ اَبنًا.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 41:42; أسْتِير 8:2; رُؤيا يُوحَنّا 6:11"> فقَالَ الأبُ لخَدَمِهِ: أسرِعوا! هاتُوا أفخَرَ ثوبٍ وأَلْبِسوهُ، وضَعُوا خاتَمًا في إصبَعِهِ وحِذاءً في رِجلَيهِ.</span>
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 25:6; ﻣﺘﻰ 22:2–14; المَزامِير 63:5"> وقَدِّموا العِجلَ المُسمَّنَ واَذبَحوهُ، فنَأْكُلَ ونَفرَحَ،</span>
<span className="v" id="v24" data-v="24" data-xref="أفَسُس 2:5; حِزْقِيال 34:16; لُوقا 19:10"> لأنَّ اَبني هذا كانَ مَيْتًا فعاشَ، وكانَ ضالًا فَوُجِدَ. فأخذوا يَفرَحونَ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 150:4; إرْمِيا 31:4; لُوقا 15:11–12"> وكانَ الابنُ الأكبرُ في الحَقلِ، فلمَّا رجَعَ واَقتَرَبَ مِنَ البَيتِ، سَمِعَ صَوتَ الغِناءِ والرَّقصِ.</span>
<span className="v" id="v26" data-v="26"> فدَعا أحَدَ الخَدَمِ وسألَهُ: ما الخَبرُ؟</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 9:17; فِلِيْمُون 1:16; اعمال الرسل 22:13"> فأجابَهُ: رجَعَ أخوكَ سالِمًا، فذبَحَ أبوكَ العِجْلَ المُسَمَّنَ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻳﻮﻧﺎﻥ 4:9; صَمُوئيلَ الأوَّلُ 18:8; ﻳﻮﻧﺎﻥ 4:1–4"> فغَضِبَ ورَفَضَ أنْ يَدخُلَ. فخَرَجَ إلَيهِ أبوهُ يَرجو مِنهُ أنْ يَدخُلَ،</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 18:11–12; زَكَريّا 7:3; إشَعْياء 65:5"> فقَالَ لأبيهِ: خَدَمْتُكَ كُلَّ هذِهِ السِّنينَ وما عَصَيتُ لكَ أمرًا، فما أعطَيتَني جَدْيًا واحدًا لأفرَحَ بِهِ معَ أصحابي.</span>
<span className="v" id="v30" data-v="30" data-xref="الأمثال 29:3; لُوقا 15:12–13; لُوقا 15:32"> ولكنْ لمَّا رجَعَ اَبنُكَ هذا، بَعدَما أكَلَ مالَكَ معَ البَغايا، ذَبَحتَ العِجلَ المُسَمَّنَ!</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="مَرْقُس 7:27–28; ﺭﻭﻣﻴﺔ 9:4; ﺭﻭﻣﻴﺔ 11:1"> فأجابَهُ أبوهُ: يا اَبني، أنتَ مَعي في كُلِّ حِينٍ، وكُلُّ ما هوَ لي فهوَ لكَ.</span>
<span className="v" id="v32" data-v="32" data-xref="أفَسُس 2:1–10; لُوقا 15:24; إشَعْياء 35:10"> ولكِنْ كانَ علَينا أنْ نَفرَحَ ونَمرَحَ، لأنَّ أخاكَ هذا كانَ مَيْتًا فَعاشَ، وكانَ ضالًا فَوُجِدَ“.</span>
</div>
