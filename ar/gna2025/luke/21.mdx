## درهم الأرملة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 12:41–44; يُوحَنّا 8:20; ﻣﺘﻰ 27:6"> وتطَلَّعَ يَسوعُ فرَأى الأغنياءَ يُلقُونَ تَبَرُّعاتِهِم في صُندوقِ الهَيكَلِ.</span>
<span className="v" id="v2" data-v="2" data-xref="مَرْقُس 12:42"> ورأى أيضًا أرمَلةً مِسكينةً تُلقي فيهِ دِرهَمينِ،</span>
<span className="v" id="v3" data-v="3" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12; اعمال الرسل 10:34; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:2–3"> فقَالَ: ”الحقَّ أقولُ لكُم: هذِهِ الأرمَلةُ الفَقيرةُ ألقَت أكثَرَ مِمَّا ألقاهُ الآخَرونَ كُلُّهُم.</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 2:44–45; لُوقا 15:12; اعمال الرسل 4:34"> فَهُم ألقَوا في الصُّندوقِ مِنَ الفائِضِ عَنْ حاجاتِهِم. وأمَّا هيَ، فمِنْ حاجَتِها ألقَت كُلَّ ما تَملِكُ لمَعيشَتِها“.</span>
</div>

## خراب الهيكل ونهاية العالم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 2:20; ﻣﺘﻰ 24:1–51; مَرْقُس 13:1–37"> وتَحدَّثَ بَعضُهُم كيفَ أنَّ الهَيكَلَ مُزَيَّنٌ بِالحِجارَةِ البَديعةِ وتُحفِ النُّذورِ، فقَالَ يَسوعُ:</span>
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 24:2; مَرْقُس 13:2; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 5:18"> ”ستَجيءُ أيّامٌ لَنْ يُتركَ فيها مِمَّا تُشاهِدونَهُ حجَرٌ على حجَرٍ، بل يُهدَمُ كُلُّهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 1:6–7; مَرْقُس 13:14; مَرْقُس 13:3–4"> فسألوهُ: ”متى يَحدُثُ هذا، يا مُعَلِّمُ؟ وما هيَ العلامَةُ الّتي تَدُلُّ على قُربِ حُدوثِهِ؟“</span>
<span className="v" id="v8" data-v="8" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:1; مَرْقُس 13:5–6; أفَسُس 5:6"> فأجابَ: ”إِنتَبِهوا لِئَلَّا يُضَلِّلَكُم أحَدٌ! سيَجيءُ كثيرٌ مِنَ النَّاسِ مُنتَحِلينَ اَسمي، فَيقولونَ: أنا هوَ! وحانَ الوقتُ! فلا تَتبَعوهُم.</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 21:28; ﻣﺘﻰ 24:6–8; مَرْقُس 13:7–8"> وإذا سَمِعْتُم بأخبارِ الحُروبِ والثَّوراتِ فلا تَفزَعوا، لأنَّ هذا لا بُدَّ أنْ يَحدُثَ في أوّلِ الأمرِ ولكِنْ لا تكونُ الآخِرَةُ بَعدُ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="اعمال الرسل 2:19–20; مَرْقُس 13:8; أخبارِ الأيّامِ الثّانِي 15:5–6"> وقالَ يَسوعُ: ”سَتقومُ أُمَّةٌ على أُمَّةٍ ومَمْلكَةٌ على مَمْلكَةٍ،</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 24:29–30; لُوقا 21:25–27"> وتقَعُ زَلازِلُ شَديدةٌ، وتَحدُثُ أَوبئَةٌ ومجاعاتٌ في أماكِنَ كثيرةٍ، وتَجري أحداثٌ مُخيفَةٌ، وتَظهَرُ علاماتٌ هائِلةٌ في السَّماءِ.</span>
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:12–14; اعمال الرسل 5:40; اعمال الرسل 22:30"> وقَبْلَ هذا كُلِّهِ، يَعتَقِلُكُمُ النَّاسُ ويَضطَهِدونَكُم ويُسْلِمونَكُم إلى المجامِعِ والسُّجونِ ويَسوقونَكُم إلى المُلوكِ والحُكّامِ مِنْ أجلِ اَسمي.</span>
<span className="v" id="v13" data-v="13" data-xref="فِيلِبِّي 1:12; فِيلِبِّي 1:28; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 1:5"> ويكونُ هذا فُرصَةً لكُم، تَشهَدونَ فيها لِلبِشارةِ.</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 12:11–12; مَرْقُس 13:11; ﻣﺘﻰ 10:19–20"> ولا تَهتَمُّوا كيفَ تُدافِعونَ عَنْ أنفُسِكُم،</span>
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 1:9; أفَسُس 6:19; لُوقا 12:12"> لأنِّي سأُعطيكُم مِنَ الكلامِ والحِكمةِ ما يَعجِزُ جميعُ خُصومِكُم عَنْ رَدِّهِ أو نَقضِهِ.</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 12:53; رُؤيا يُوحَنّا 6:9; ميخا 7:5–6"> وسيُسَلِّمُكُم والدوكُم وإخوتُكُم وأقرباؤُكُم وأصدقاؤُكُم أنفسُهُم إلى الحُكَّامِ، ويَقتُلونَ مِنكُم،</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 15:19; لُوقا 6:22; يُوحَنّا 15:21"> ويُبغِضُكُم جميعُ النَّاسِ مِنْ أجلِ اَسمي.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 10:30; صَمُوئيلَ الأوَّلُ 14:45; لُوقا 12:7"> ولكِنْ شَعْرةٌ واحدةٌ مِنْ رؤُوسِكُم لا تقَعُ.</span>
<span className="v" id="v19" data-v="19" data-xref="العِبرانِيّين 10:36; ﻣﺘﻰ 10:22; يَعقُوب 1:3"> وبِثَباتِكُم تَخلُصونَ.</span>
</div>
</div>

## خراب أورشليم ومجيء ابن الإنسان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="لُوقا 19:43; دانيال 9:27; مَرْقُس 13:14"> ”فإذا رأيتُم أُورُشليمَ تُحاصِرُها الجُيوشُ، فاَعلَموا أنَّ خَرابَها قَريبٌ.</span>
<span className="v" id="v21" data-v="21" data-xref="إرْمِيا 6:1; التَّكوين 19:17; لُوقا 17:31–33"> فمَنْ كانَ في اليهودِيَّةِ فلْيَهرُبْ إلى الجبالِ، ومَنْ كانَ في المدينةِ فلْيَخرُجْ مِنها، ومَنْ كانَ في الحُقولِ فلا يَدخُلْ إلى المدينةِ،</span>
<span className="v" id="v22" data-v="22" data-xref="هُوشَع 9:7; إشَعْياء 65:12–16; إشَعْياء 63:4"> لأنَّ هذِهِ أيّامُ نَقمَةٍ يَتِمُّ فيها كُلُّ ما جاءَ في الكُتُبِ المُقدَّسَةِ.</span>
<span className="v" id="v23" data-v="23" data-xref="مَرْقُس 13:17; ﻣﺘﻰ 24:19; لُوقا 23:29"> الوَيلُ لِلحَبالى والمُرضِعاتِ في تِلكَ الأيّامِ! فستَنزِلُ النَّكبةُ على هذا البَلَدِ وغَضَبُ اللهِ على هذا الشَّعبِ،</span>
<span className="v" id="v24" data-v="24" data-xref="ﺭﻭﻣﻴﺔ 11:25; رُؤيا يُوحَنّا 11:2; إشَعْياء 63:18"> فيَسقُطونَ بِحَدِّ السَّيفِ، ويُؤخَذونَ أسرى في جَميعِ الأُمَمِ، ويَدوسُ الوَثَنيُّونَ أُورُشليمَ إلى أنْ يَتِمَّ زَمانُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 13:10; يُوئيل 2:30–31; رُؤيا يُوحَنّا 6:12–14"> وستَظهَرُ عَلاماتٌ في الشَّمسِ والقَمَرِ والنُّجومِ، ويُصيبُ الأُمَمَ في الأرضِ قَلَقٌ شَديدٌ ورُعبٌ مِنْ ضَجيجِ البحرِ واَضطِرابِ الأمواجِ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 24:29; التَّثْنِيَة 28:65–67; رسالَةُ بُطرُسَ الثّانيَة 3:10–12"> ويَسقُطُ النَّاسُ مِنَ الخَوفِ ومِنِ اَنتظارِ ما سَيحِلُّ بالعالَمِ، لأنَّ قوّاتِ السَّماءِ تَتزَعزَعُ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 24:30; رُؤيا يُوحَنّا 1:7; مَرْقُس 13:26"> وفي ذلِكَ الحِينِ يَرى النَّاسُ اَبنَ الإنسانِ آتيًا في سحابةٍ بِكُلِّ عِزَّةٍ وجَلالٍ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﺭﻭﻣﻴﺔ 8:23; إشَعْياء 60:1–2; أفَسُس 4:30"> وإذا بَدأَتْ تَتِمُّ هذِهِ الحوادِثُ، قِفوا واَرْفَعوا رُؤُوسَكُم لأنَّ خلاصَكُم قَريبٌ“.</span>
</div>
</div>

<span className="v" id="v30" data-v="30" data-xref="لُوقا 12:57"> فإذا أوْرَقَت، عرَفتُم أنَّ الصَّيفَ قَريبٌ.</span>

<span className="v" id="v35" data-v="35" data-xref="رُؤيا يُوحَنّا 16:15; المَزامِير 11:6; إرْمِيا 48:43–44"> لأنَّهُ كالفَخِّ يُطبِقُ على سكّانِ الأرضِ كُلِّهِم.</span>

</div>
