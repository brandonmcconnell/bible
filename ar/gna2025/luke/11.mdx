## الصلاة الرّبية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المَزامِير 19:14; لُوقا 22:39–45; ﺭﻭﻣﻴﺔ 8:26–27"> وكانَ يَسوعُ يُصلِّي في أحدِ الأماكنِ، فلمَّا أتَمَّ الصلاةَ قالَ لَهُ واحِدٌ مِنْ تلاميذِه: ”يا رَبُّ، عَلِّمْنا أنْ نُصلِّيَ، كما عَلَّمَ يوحنَّا تلاميذَهُ“.</span>

<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 6:12–15; ﻣﺘﻰ 26:41; إشَعْياء 43:25–26"> واَغفِرْ لنا خطايانا،<br />لأنَّنا نَغفِرُ لِكُلِّ مَنْ يُذنِبُ إلينا.<br />ولا تُدخِلْنا في التَّجربَةِ“،</span>

<div className="bp">

<span className="v" id="v6" data-v="6"> لأنَّ لي صَديقًا جاءَني مِنْ سَفرٍ ولا خُبزَ عِندي أُقَدِّمُ لَهُ،</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 13:25; غَلاطِيَّة 6:17; لُوقا 7:6"> فيُجيبُ صَديقُهُ مِنْ داخلِ البَيتِ: لا تُزعِجْني! البابُ مُقفَلٌ، وأولادي مَعي في الفِراشِ، فلا أقدِرُ أنْ أقومَ لأُعطِيَك.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 32:26; كُولُوسِي 4:12; ﺭﻭﻣﻴﺔ 15:30"> أقولُ لكُم: إنْ كانَ لا يقومُ ويُعطيهِ لأنَّهُ صَديقُهُ، فهوَ يَقومُ ويُعطيهِ كلَّ ما يَحتاجُ إلَيهِ لأنَّهُ لَجَّ في طَلَبِهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="مَرْقُس 11:24; يُوحَنّا 16:23–24; ﻣﺘﻰ 21:22"> لذَلِكَ أقولُ لكُم: إسألوا تنالوا، أطلُبوا تَجِدوا، دُقّوا البابَ يُفْتَحْ لكُم.</span>
<span className="v" id="v10" data-v="10" data-xref="يَعقُوب 4:3; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:54–58; المَزامِير 31:22"> فمَنْ يَسألْ يَنَلْ، ومَنْ يَطلُبْ يَجِدْ، ومَنْ يَدُقّ البابَ يُفتَحْ لَهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 7:9; إشَعْياء 49:15"> فأيُّ أبٍ مِنكُم إذا طَلبَ مِنهُ اَبنُهُ سَمكَةً أعطاهُ بدَلَ السَّمكَةِ حَيَّةً؟</span>
<span className="v" id="v12" data-v="12" data-xref="رُؤيا يُوحَنّا 9:10; لُوقا 10:19; حِزْقِيال 2:6"> أو طلَبَ مِنهُ بَيضَةً أعطاهُ عقرَبًا؟</span>
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 7:11; إشَعْياء 44:3–4; ﺭﻭﻣﻴﺔ 8:32"> فإذا كُنتُم أنتُمُ الأشرار تَعرِفونَ كيفَ تُحسِنونَ العَطاءَ لأبنائِكُم، فما أولى أباكُمُ السَّماوِيُّ بأنْ يهَبَ الرُّوحَ القُدُسَ لِلَّذينَ يَسألونَهُ؟“</span>
</div>
</div>

## يسوع وبعلزبول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 12:22–24; ﻣﺘﻰ 9:32–34; مَرْقُس 7:32–37"> وكانَ يَطرُدُ شَيطانًا أخرَسَ. فلمَّا خرَجَ الشَّيطانُ تكَلَّمَ الرَّجُلُ، فتعَجَّبَ الجُموعُ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 9:34; ﻣﺘﻰ 12:24–30; يُوحَنّا 8:48"> لكِنَّ بَعضَهُم قالَ: ”هوَ يَطرُدُ الشَّياطينَ ببعلَزبولَ رَئيسِ الشَّياطينِ“.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 6:30; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:22; مَرْقُس 8:11–12"> وطلَبَ آخَرونَ آيةً مِنَ السَّماءِ ليُجَرِّبوهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="مَرْقُس 3:23–27; ﻣﺘﻰ 12:25–29; إشَعْياء 19:2–3"> فعَرَفَ أفكارَهُم، فقَالَ لهُم: ”كُلُّ مَملكَةٍ تَنقَسِمُ تَخرَبُ وتَنهارُ بُيوتُها بَيتًا على بَيتٍ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 12:26; ﻣﺘﻰ 12:31–34; لُوقا 11:15"> وإذا اَنقَسَمَ الشَّيطانُ، فكيفَ تَثبتُ مَملكتُهُ؟ تَقولونَ إنِّي ببعلَزَبولَ أطرُدُ الشَّياطينَ.</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 9:49; أيُّوب 15:6; ﺭﻭﻣﻴﺔ 3:19"> فإنْ كُنتُ بِبَعلَزَبولَ أطرُدُ الشَّياطينَ، فَبِمَن يطرُدُهُ أتباعُكُم؟ لذلِكَ هُم يَحكُمونَ علَيكُم.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 8:19; ﻣﺘﻰ 12:28; لُوقا 10:9"> وأمَّا إذا كُنتُ بإصبعِ اللهِ أطرُدُ الشَّياطينَ، فمَلكوتُ اللهِ أقبلَ علَيكُم.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 3:27; ﻣﺘﻰ 12:29"> عِندَما يَحرُسُ الرَّجُلُ القَويُّ المُتَسَلِّحُ بَيتَهُ تكونُ أموالُهُ في أمانٍ.</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 49:24–25; إشَعْياء 53:12; رسالَةُ يُوحَنّا الأُولَى 3:8"> ولكِنْ إذا هاجَمَهُ رَجُلٌ أقوى مِنهُ وغَلبَهُ، يَنتَزِعُ مِنهُ كُلَّ سلاحِهِ الّذي كانَ يَعتَمِدُ علَيهِ ويُوزِّعُ ما سَلَبهُ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 12:30; لُوقا 9:50; مَرْقُس 9:40"> مَنْ لا يكونُ مَعي فهوَ علَيَّ، ومَنْ لا يجمَعُ مَعي فهوَ يُبَدِّدُ.</span>
</div>
</div>

## الجموع يطلبون آية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="لُوقا 11:16; مَرْقُس 8:11–12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:22"> واَزدَحَمَتِ الجُموعُ، فأخَذَ يَقولُ: ”هذا الجِيلُ جِيلٌ فاسِدٌ يَطلُبُ آيةً، ولَنْ يُعطى لَهُ سِوى آيةِ يُونانَ النَّبيِّ.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 12:40–42; ﻳﻮﻧﺎﻥ 1:17; ﻳﻮﻧﺎﻥ 3:2–10"> فكما كانَ النَّبيُّ يُونانُ آيةً لأهلِ نينوى، فكذلِكَ يكونُ اَبنُ الإنسانِ آيةً لِهذا الجِيلِ.</span>
<span className="v" id="v31" data-v="31" data-xref="أخبارِ الأيّامِ الثّانِي 9:1; المُلُوكِ الأوَّلُ 10:1–13; ﺭﻭﻣﻴﺔ 2:27"> مَلِكَةُ الجُنوبِ تَقومُ يومَ الحِسابِ معَ هذا الجِيلِ وتَحكُمُ علَيهِ، لأنَّها جاءَت مِنْ أقاصي الأرضِ لِتسمَعَ حِكمَةَ سُليمانَ، وهُنا الآنَ أعظَمُ مِنْ سُليمانَ.</span>
<span className="v" id="v32" data-v="32" data-xref="العِبرانِيّين 7:26; لُوقا 11:31; ﻳﻮﻧﺎﻥ 3:5–4"> وأهلُ نينوى يَقومونَ يَومَ الحِسابِ معَ هذا الجِيلِ ويَحكُمونَ علَيهِ، لأنَّهُم تابوا عِندَما سَمِعوا إنذارَ يُونانَ، وهُنا الآنَ أعظَمُ مِنْ يُونانَ!</span>
</div>
</div>

## نور الجسد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="لُوقا 8:16–17; مَرْقُس 4:21–22; ﻣﺘﻰ 5:15–16"> ”ما مِنْ أحَدٍ يُوقِدُ سِراجًا ويضَعُهُ في مَخبإٍ أو تَحتَ المِكيالِ، بل في مكانٍ مُرتَفِعٍ ليَستَنيرَ بِهِ الدّاخِلونَ.</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 6:22–23; اعمال الرسل 26:18; إرْمِيا 5:21"> سِراجُ الجسدِ هوَ العَينُ. فإنْ كانَت عَينُكَ سَليمَةً، كانَ جَسَدُكَ كُلُّهُ مُنيرًا. وإنْ كانَت عَينُكَ مَريضةً، كانَ جَسَدُكَ كُلُّهُ مُظلِمًا.</span>
<span className="v" id="v35" data-v="35" data-xref="الأمثال 16:25; رسالَةُ بُطرُسَ الثّانيَة 1:9; يَعقُوب 3:13–17"> فاَنْتَبِه، لِئَلَّا يَصيرَ النّورُ الّذي فيكَ ظَلامًا.</span>
<span className="v" id="v36" data-v="36" data-xref="الأمثال 20:27; الأمثال 6:23; الأمثال 4:18–19"> فإنْ كانَ جسَدُكَ كُلُّهُ مُنيرًا، ولا أثَرَ للظَّلامِ فيهِ، أنارَ بأكمَلِهِ كما لو أنارَ لكَ السِّراجُ بِضوئِهِ“.</span>
</div>
</div>

## يسوع يوبخ الفريسيين ومعلمي الشريعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="لُوقا 7:36; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:19–23; لُوقا 14:1"> وبَينَما هوَ يَتكلَّمُ، دَعاهُ أحَدُ الفَرِّيسيِّينَ إلى الغَداءِ عِندَهُ. فدخَلَ بَيتَهُ وجلَسَ لِلطَّعامِ.</span>
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 15:2–3; مَرْقُس 7:2–5; يُوحَنّا 3:25"> فتَعَجَّبَ الفَرِّيسيُّ لمَّا رأى يَسوعَ يَجلِسُ ولا يَغسِلُ يَديهِ قَبلَ الغَداءِ.</span>
<span className="v" id="v39" data-v="39" data-xref="تِيطُس 1:15; الأمثال 30:12; التَّكوين 6:5"> فقَالَ لَهُ الرَّبُّ يَسوعُ: ”أنتُم أيُّها الفَرِّيسيُّونَ تُطَهِّرونَ ظاهِرَ الكأسِ والصَّحنِ، وباطِنُكُم كُلُّهُ طمعٌ وخُبثٌ.</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 12:20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:36; ﻣﺘﻰ 23:26"> يا أغبياءُ، هذا الّذي صنَعَ الظاهِرَ، أمَا صنَعَ الباطِنَ أيضًا؟</span>
<span className="v" id="v41" data-v="41" data-xref="لُوقا 12:33; يَعقُوب 1:27; تِيطُس 1:15"> أعطُوا الفُقراءَ مِمَّا في داخِلِ كُؤوسِكُم وصُحونِكُم، يكُنْ كُلُّ شيءٍ لكُم طاهِرًا.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="ﻣﺘﻰ 23:23; لُوقا 18:12; الأمثال 21:3"> ولكِنِ الوَيلُ لكُم، أيُّها الفَرِّيسيُّونَ! تُعطونَ العُشرَ مِنَ النَّعنَعِ والصَّعتَرِ وسائِرِ البُقولِ، وتُهمِلونَ العَدلَ ومَحبَّةَ اللهِ. فهذا كانَ يَجبُ أنْ تَعمَلوا بِهِ مِنْ دونِ أنْ تُهمِلوا ذاكَ.</span>
</div>
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="لُوقا 20:46; مَرْقُس 12:38–39; لُوقا 14:7–11"> الوَيلُ لكُم، أيُّها الفَرِّيسيُّونَ! تُحبّونَ مكانَ الصَّدارَةِ في المَجامِعِ والتَّحيّاتِ في السّاحاتِ.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 23:27–28; المَزامِير 5:9; اعمال الرسل 23:3"> الوَيلُ لكُم أنتُم مِثْلُ القُبورِ المَجهولَةِ، يَمشي النَّاسُ علَيها وهُم لا يَعرِفونَ“.</span>
</div>
