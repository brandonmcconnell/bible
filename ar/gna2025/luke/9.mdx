## يسوع يرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 14:12; لُوقا 10:19; ﻣﺘﻰ 16:19"> ودَعا يَسوعُ تلاميذَهُ الاثنَي عشَرَ وأعطاهُم سُلطانًا على جَميعِ الشَّياطينِ وقُدرَةً على شِفاءِ الأمراضِ،</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 10:7–8; لُوقا 10:9; ﻣﺘﻰ 3:2"> ثُمَّ أرسَلَهُم ليُبشِّروا بِمَلكوتِ اللهِ ويَشفوا المَرضى.</span>
<span className="v" id="v3" data-v="3" data-xref="لُوقا 22:35; المَزامِير 37:3; لُوقا 10:4–12"> وقالَ لهُم: ”لا تَحمِلوا لِلطَّريقِ شَيئًا: لا عصًا، ولا كيسًا، ولا خُبزًا، ولا مالًا، ولا يكُنْ لأحدٍ مِنكُم ثَوبانِ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 10:11; مَرْقُس 6:10; لُوقا 10:5–8"> وأيَّ بَيتٍ دَخلتُم، فَفيهِ أقيمُوا ومِنهُ اَرحَلوا.</span>
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 13:51; اعمال الرسل 18:6; مَرْقُس 6:11"> وكُلُّ مدينةٍ لا يَقبَلُكُم أهلُها، فاَخرُجوا مِنها واَنفُضُوا الغُبارَ عَنْ أقدامِكُم نَذيرًا لهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 4:30; مَرْقُس 16:20; اعمال الرسل 5:15"> فخرَجَ التَّلاميذُ وساروا في القُرى، يُبَشِّرونَ ويَشفُونَ المَرضى في كُلِّ مكانٍ.</span>
</div>
</div>

## حيرة هيرودس

<div className="bt" dir="rtl">
<div className="bp">

<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 1:21; لُوقا 9:19; مَرْقُس 6:15"> وبَعضَهُم: ”إيليَّا ظهَرَ!“ وآخَرينَ: ”نَبيٌّ مِنَ القُدَماءِ قامَ“.</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 23:8; لُوقا 13:31–32"> ولكِنَّ هِيرودُسَ قالَ: ”يوحنَّا أنا قَطعْتُ رأسَهُ. فمَن هذا الّذي أسمَعُ عَنهُ مِثْلَ هذِهِ الأخبارِ؟“ وكانَ يَطلُبُ أنْ يَراهُ.</span>
</div>
</div>

## يسوع يطعم خمسة آلاف رجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 11:21; العِبرانِيّين 13:17; ﻣﺘﻰ 14:13–21"> ولمَّا رجَعَ الرُّسُلُ أخبَروا يَسوعَ بِكُلِّ ما عَمِلوهُ، فأخَذَهُم واَعتَزَلَ بِهِم عِندَ مدينةٍ اَسمُها بَيتُ صيدا.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 9:2; لُوقا 5:31; العِبرانِيّين 4:16"> وعرَفَ النَّاسُ فتَبِعوهُ، فاَستَقبَلَهُم وكَلَّمَهُم على مَلكوتِ اللهِ، وشَفى المُحتاجينَ مِنهُم إلى الشِّفاءِ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 6:1; يُوحَنّا 6:5–15; هُوشَع 13:5"> وأخَذَ النَّهارُ يَميلُ، فدَنا إليهِ تلاميذُهُ الاثنا عشَرَ وقالوا لَهُ: ”إِصرِفْ هذا الجمَعَ ليَذهَبوا إلى القُرى والمَزارِعِ المُجاوِرَةِ، فيَبيتوا فيها ويَجِدوا لهُم طعامًا، لأنَّنا هُنا في مكانٍ مُقفِرٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الأمثال 11:24–25; يُوحَنّا 6:5–9; المُلُوكِ الثّانِي 4:42–43"> فقَالَ لهُم يَسوعُ: ”أعطوهُم أنتُم ليأكُلوا“. فقَالوا: ”كُلُّ ما عِندَنا خَمسةُ أرغِفَةٍ وسَمكتانِ، إلَّا إذا ذَهَبْنا واَشتَرَينا طعامًا لِكُلِّ هؤُلاءِ النَّاسِ“.</span>
<span className="v" id="v14" data-v="14" data-xref="مَرْقُس 8:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:40; مَرْقُس 6:39–40"> وكانوا نحوَ خَمسةِ آلافِ رَجُلٍ. فقَالَ لِتلاميذِهِ: ”أقْعِدوهُم جَماعةً جَماعةً، في كُلِّ واحدةٍ مِنها خَمسونَ“.</span>
<span className="v" id="v15" data-v="15"> فأقعَدُوهُم كُلَّهُم كما أمرَهُم.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 14:19; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 4:4–5; ﻣﺘﻰ 15:36"> فأخَذَ الأرغِفَةَ الخَمسةَ والسَّمَكتَينِ ورفَعَ عَينَيهِ نَحوَ السَّماءِ وبارَكَ وكسَرَ وأعطى تلاميذَهُ ليُوزِّعوها على الجَمعِ.</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الثّانِي 4:44; ﻣﺘﻰ 15:37–38; يُوحَنّا 6:11–13"> فأكلوا كُلُّهُم حتّى شَبِعوا، ورفَعَ التَّلاميذُ اَثنتَي عَشْرَةَ قُفَّةً مِنَ الكِسَرِ الّتي فَضَلَتْ.</span>
</div>
</div>

<span className="v" id="v19" data-v="19" data-xref="مَلاخِي 4:5; لُوقا 9:7–8; ﻣﺘﻰ 14:2"> فأجابوا: ”يوحنَّا المَعمَدانُ“. وبَعضُهُم يقولُ: ”إيليَّا“. وآخَرونَ: ”نَبيٌّ مِنَ القُدَماءِ قامَ“.</span>

## يسوع يُنبئ بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 53:1–12; لُوقا 24:7; لُوقا 18:31–34"> وقالَ لِتلاميذِهِ: ”يَجِبُ على اَبنِ الإنسانِ أنْ يتألَّمَ كثيرًا، وأنْ يَرفُضَهُ الشُّيوخُ ورُؤساءُ الكَهنَةِ ومُعَلِّمو الشَّريعةِ، وأنْ يُقتَلَ وفي اليومِ الثَّالثِ يَقومُ مِنْ بَينِ الأمواتِ“.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 10:38–39; لُوقا 14:26–27; ﺭﻭﻣﻴﺔ 8:13"> وقالَ لِلجُموعِ كُلِّهِم: ”مَنْ أرادَ أنْ يَتبَعَني، فلْيُنكِرْ نَفسَهُ ويَحمِلْ صَليبَهُ كُلَّ يومٍ ويَتبَعْني.</span>
<span className="v" id="v24" data-v="24" data-xref="لُوقا 17:33; يُوحَنّا 12:25; اعمال الرسل 20:23–24"> مَنْ أرادَ أنْ يُخَلِّصَ حَياتَهُ يَخسَرُها، ومَنْ خَسِرَ حياتَهُ في سَبيلي يُخَلِّصُها.</span>
<span className="v" id="v25" data-v="25" data-xref="مَرْقُس 8:36; ﻣﺘﻰ 16:26; المَزامِير 49:6–8"> فماذا يَنفَعُ الإنسانَ لو رَبِحَ العالَمَ كُلَّهُ وخَسِرَ نَفسَهُ أو أهلَكَها.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 10:32–33; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:12; لُوقا 12:8–9"> مَن اَستَحى بي وبِكلامي، يَستَحي بِهِ اَبنُ الإنسانِ متى جاءَ في مَجدِهِ ومَجدِ الآبِ والملائِكَةِ الأطهارِ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 16:28; مَرْقُس 9:1; يُوحَنّا 21:22–23"> الحَقَّ أقولُ لكُم: في الحاضِرينَ هُنا مَنْ لا يَذوقونَ الموتَ حتّى يُشاهِدوا مَلكوتَ اللهِ“.</span>
</div>
</div>

## التجلي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 17:1–13; ﻣﺘﻰ 26:37–39; مَرْقُس 14:33–36"> وبَعدَ هذا الكلامِ بنَحوِ ثَمانيةِ أيّامٍ، أخذَ يَسوعُ بُطرُسَ ويوحنَّا ويَعقوبَ وصَعِدَ إلى الجبَلِ ليُصلِّي.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 17:2; مَرْقُس 9:2–3; الخُرُوج 34:29–35"> وبَينَما هوَ يُصلِّي، تَغيَّرَت هَيئَةُ وَجهِهِ وصارَت ثيابُهُ بَيضاءَ لامِعةً.</span>
<span className="v" id="v30" data-v="30" data-xref="العِبرانِيّين 3:3–6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:7–11; لُوقا 1:17"> وإذا رَجُلانِ يُكلِّمانِ يَسوعَ، وهُما موسى وإيليَّا،</span>
<span className="v" id="v31" data-v="31" data-xref="فِيلِبِّي 3:21; رسالَةُ بُطرُسَ الأُولَى 1:11–12; رسالَةُ بُطرُسَ الأُولَى 5:10"> ظَهَرا في مَجدٍ سَماويٍّ وأخَذا يتَحَدَّثانِ عَنْ موتِهِ الّذي كانَ علَيهِ أنْ يُتمِّمَهُ في أُورُشليمَ.</span>
<span className="v" id="v32" data-v="32" data-xref="دانيال 8:18; يُوحَنّا 1:14; دانيال 10:9"> وغلَبَ النُّعاسُ بُطرُسَ ورَفيقَيهِ، ولكِنَّهُم أفاقوا وشاهَدوا مَجدَهُ والرَّجُلينِ الواقفَينِ معَهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 9:5–6; مَرْقُس 10:38; المَزامِير 4:6–7"> وبَينَما هُما يُفارِقانِهِ قالَ لَه بُطرُسُ: ”يا مُعَلِّمُ، ما أجمَلَ أنْ نكونَ هُنا. فَلْنَنصُبْ ثلاثَ مَظالَّ، واحدةً لكَ وواحدةً لموسى وواحدةً لإيليَّا“. وكانَ لا يُدْرِكُ ما يَقولُ.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="الخُرُوج 40:34–38; ﻣﺘﻰ 17:5–7; القُضاة 13:22"> وبَينَما بُطرُسُ يَتكَلَّمُ، جاءَت سَحابَةٌ فظَلَّلَتْهُم، فخافَ التَّلاميذُ عِندَما دَخلوا في السَّحابَةِ.</span>
<span className="v" id="v35" data-v="35" data-xref="لُوقا 3:22; اعمال الرسل 3:22–23; ﻣﺘﻰ 3:17"> وقالَ صَوتٌ مِنَ السَّحابَةِ: ”هذا هوَ اَبني الّذي اَختَرْتُهُ، فلَهُ اَسمَعُوا!“</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 17:9; مَرْقُس 9:9–10; مَرْقُس 9:6"> وبَعدَما تكَلَّمَ الصَوتُ، رَأَوا يَسوعَ وحدَهُ. فسكتوا وما أخبَروا أحدًا في تِلكَ الأيّامِ بشيءٍ مِمَّا شاهدوهُ.</span>
</div>
</div>

## يسوع يشفي صبيًّا فيه روح نجس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 17:14–20; مَرْقُس 9:14–29"> وفي الغَدِ نَزَلُوا مِنَ الجبَلِ، فاَستَقبَلَهُ جَمعٌ كبيرٌ.</span>
<span className="v" id="v38" data-v="38" data-xref="زَكَريّا 12:10; يُوحَنّا 4:47; لُوقا 7:12"> وصاحَ رَجُلٌ مِنَ الجَمعِ: ”يا مُعَلِّمُ، أطلُبُ إليكَ أنْ تَنظُرَ إلى اَبني، فهوَ وَلَدي الأوحَدُ،</span>
<span className="v" id="v39" data-v="39" data-xref="مَرْقُس 9:20; لُوقا 8:29; لُوقا 4:35"> يُباغِتُهُ رُوحٌ نَجِسٌ بِصَرخَةٍ عالِيةٍ ويَخْبُطُهُ حتّى يُزبدَ، ولا يَترُكُهُ إلَّا بَعدَما يُنْهِكُهُ ويُرَضِّضُهُ.</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 9:1; اعمال الرسل 19:13–16; لُوقا 10:17–19"> وطَلَبتُ مِنْ تلاميذِكَ أنْ يَطْرُدوهُ فَما قَدِروا“.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="التَّثْنِيَة 32:5; ﺭﻭﻣﻴﺔ 2:4; يُوحَنّا 20:27"> فأجابَ يَسوعُ: ”يا لكُم مِنْ جِيلٍ غَيرِ مُؤمِنٍ فاسِدٍ! إلى متى أبقى مَعكُم وأحتَمِلُكُم؟“ وقالَ لِلرَّجُلِ: ”قَدِّمِ اَبنَكَ إلى هُنا!“</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="مَرْقُس 9:26–27; لُوقا 7:15; رُؤيا يُوحَنّا 12:12"> وبَينَما الصَّبيُّ يَدنو مِنْ يَسوعَ، صرَعَهُ الشَّيطانُ وخبَطَهُ. فاَنتَهرَ يَسوعُ الرُّوحَ النَّجِسَ، وشَفى الصَّبيَّ وسَلَّمَهُ إلى أبيهِ.</span>
<span className="v" id="v43" data-v="43" data-xref="لُوقا 9:43–45; لُوقا 5:26; ﻣﺘﻰ 17:22"> فتَعَجَّبَ الحاضِرونَ كُلُّهُم مِنْ قُدرَةِ اللهِ العظيمةِ. يسوع يُنبئ مرة ثانية بموته<br />وبَينَما هُم جميعًا مُتعجِّبونَ مِنْ كُلِّ ما عَمِلَ يَسوعُ، قالَ لِتلاميذِهِ:</span>
