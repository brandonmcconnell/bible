## ميلاد يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 3:1; ﻣﺘﻰ 24:14; اعمال الرسل 25:21"> وفي تِلكَ الأيّامِ أمَرَ القيصَرُ أوغُسطُسُ بإحصاءِ سكّانِ الإمبراطورِيَّةِ.</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 3:1; ﻣﺘﻰ 4:24; اعمال الرسل 26:30"> وجرى هذا الإحصاءُ الأوَّلُ عِندَما كانَ كِيرينِيوسُ حاكِمًا في سوريَّةَ.</span>
<span className="v" id="v3" data-v="3"> فذهَبَ كُلُّ واحدٍ إلى مَدينتِهِ ليكتَتِبَ فيها.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="ميخا 5:2; يُوحَنّا 7:42; صَمُوئيلَ الأوَّلُ 16:1"> وصَعِدَ يوسُفُ مِنَ الجَليلِ مِنْ مدينةِ النّاصِرَةِ إلى اليهوديَّةِ إلى بَيتَ لَحمَ مدينةِ داودَ، لأنَّهُ كانَ مِنْ بَيتِ داودَ وعشيرتِهِ،</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 22:22–27; ﻣﺘﻰ 1:18–19"> ليكتَتِبَ معَ مَريمَ خَطيبَتِهِ، وكانَت حُبلى.</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 33:11; رُؤيا يُوحَنّا 12:1–5; ميخا 5:2"> وبَينَما هُما في بَيتَ لَحمَ، جاءَ وَقتُها لِتَلِدَ،</span>
<span className="v" id="v7" data-v="7" data-xref="غَلاطِيَّة 4:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:9; إشَعْياء 53:2–3"> فولَدَتِ اَبنَها البِكرَ وقَمَّطَتْهُ وأضجَعَتهُ في مِذْودٍ، لأنَّهُ كانَ لا مَحَلَّ لهُما في الفُندُقِ.</span>
</div>
</div>

## الرعاة والملائكة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 31:39–40; المَزامِير 78:70–71; صَمُوئيلَ الأوَّلُ 17:34–35"> وكانَ في تِلكَ النّاحيةِ رُعاةٌ يَبيتونَ في البرِّيَّةِ، يتناوَبونَ السَّهَرَ في اللَّيلِ على رعِيَّتِهِم.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 60:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:18; المُلُوكِ الأوَّلُ 8:11"> فظهَرَ مَلاكُ الرَّبِّ لهُم، وأضاءَ مجَدُ الرَّبِّ حَولَهُم فَخافوا خَوفًا شَديدًا.</span>
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 40:9; إشَعْياء 61:1; إشَعْياء 52:7"> فقَالَ لهُمُ المَلاكُ: ”لا تَخافوا! ها أنا أُبَشِّرُكُم بِخَبرٍ عظيمٍ يَفرَحُ لَه جميعُ الشَّعبِ:</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 9:6; ﻣﺘﻰ 1:21; اعمال الرسل 2:36"> وُلِدَ لكُمُ اليومَ في مدينةِ داودَ مُخلِّصٌ هوَ المَسيحُ الرَّبُّ.</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 7:14; إشَعْياء 53:1–2; إشَعْياء 7:11"> وإلَيكُم هذِهِ العلامَةَ: تَجِدونَ طِفلًا مُقَمَّطًا مُضْجَعًا في مِذْودٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 103:20–21; رُؤيا يُوحَنّا 5:11; المَزامِير 148:2"> وظهَرَ معَ المَلاكِ بَغتةً جُمهورٌ مِنْ جُندِ السَّماءِ، يُسبِّحونَ اللهَ ويقولونَ:</span>
<span className="v" id="v14" data-v="14" data-xref="ﺭﻭﻣﻴﺔ 5:1; لُوقا 19:38; كُولُوسِي 1:20"> ”المَجدُ للهِ في العُلى، وفي الأرضِ السَّلامُ لِلحائزينَ رِضاهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:22; ﻣﺘﻰ 2:1–2; ﻣﺘﻰ 2:9–11"> ولمَّا اَنصَرَفَ المَلائِكةُ عَنهُم إلى السَّماءِ، قالَ الرُّعاةُ بَعضُهُم لِبَعضٍ: ”تَعالَوا نَذهَبُ إلى بَيتَ لحمَ لِنرى هذا الحَدَثَ الّذي أخبَرَنا بِه الرَّبُّ“.</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 19:32; لُوقا 2:12; لُوقا 2:7"> وجاؤُوا مُسرعينَ، فوَجَدوا مَريَمَ ويوسُفَ والطِّفلَ مُضجَعًا في المِذودِ.</span>
<span className="v" id="v17" data-v="17" data-xref="المَزامِير 66:16; لُوقا 8:39; المَزامِير 71:17–18"> فلمَّا رأوهُ أخبَروا بِما حَدَّثهُمُ المَلاكُ عَنهُ،</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 2:47; لُوقا 4:36; لُوقا 1:65–66"> فكانَ كُلُّ مَنْ سَمِعَ يتَعَجَّبُ مِنْ كلامِهِم.</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 2:51; لُوقا 1:66; الأمثال 4:4"> وحَفِظَت مَريمُ هذا كُلَّهُ وتأمَّلتْهُ في قَلبِها.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 106:48; إشَعْياء 29:19; اعمال الرسل 11:18"> ورَجَعَ الرُّعاةُ وهُمْ يُمَجِّدونَ اللهَ ويُسبِّحونَهُ على كُلِّ ما سَمِعوا ورَأوا كما أخبرهُمُ المَلاكُ.</span>
</div>
</div>

## ختان يسوع وتقديمه للرّب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="لُوقا 1:31; التَّكوين 17:12; غَلاطِيَّة 4:4–5"> ولمَّا بلَغَ الطِّفلُ يومَهُ الثّامنَ، وهوَ يومُ خِتانِهِ، سُمِّي يَسوعَ، كما سمَّاهُ المَلاكُ قَبلَما حَبِلَت بِه مَريَمُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اللّاويِّين 12:1–8"> ولمَّا حانَ يومُ طَهورِهِما بِحَسَبِ شَريعةِ موسى، صَعِدا بالطِّفلِ يَسوعَ إلى أُورُشليمَ ليُقَدِّماهُ لِلرَّبِّ،</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 13:2; الخُرُوج 22:29; العَدَد 3:13"> كما هوَ مكتوبٌ في شريعةِ الرَّبِّ: ”كُلُّ بِكرٍ فاتحِ رَحِمٍ هوَ قُدّوسٌ للرَّبِّ“،</span>
<span className="v" id="v24" data-v="24" data-xref="اللّاويِّين 12:6–8; اللّاويِّين 12:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:9"> وليُقدِّما الذَّبيحةَ الّتي تَفرِضُها شريعةُ الرَّبِّ: زَوْجَي يَمامٍ أو فَرخَي حَمامٍ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 25:9; لُوقا 2:38; لُوقا 1:6"> وكانَ في أُورُشليمَ رَجُلٌ بارٌّ تَقيٌّ إِسمُهُ سِمعانُ، يَنتَظِرُ العَزاءَ لإِسرائيلَ، والرُّوحُ القُدُسُ كانَ علَيهِ.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 8:51; العِبرانِيّين 11:5; لُوقا 9:27"> وكانَ الرُّوحُ القُدُسُ أوحَى إلَيهِ أنَّهُ لا يَذوقُ الموتَ قَبلَ أنْ يرى مَسيحَ الرَّبِّ</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 4:1; لُوقا 2:41; لُوقا 2:51"> فجاءَ إلى الهَيكَلِ بِوَحيٍ مِنَ الرُّوحِ. ولمَّا دخَلَ الوالدانِ ومعَهُما الطِّفلُ يَسوعُ ليُؤدِّيا عَنهُ ما تَفرِضُهُ الشَّريعةُ،</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 1:64; لُوقا 2:20; لُوقا 2:13–14"> حَمَلهُ سِمعانُ على ذِراعَيهِ وبارَكَ اللهَ وقالَ:</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="رُؤيا يُوحَنّا 14:13; التَّكوين 15:15; فِيلِبِّي 1:23"> ”يا ربُّ، تَمَّمْتَ الآنَ وَعدَكَ لي<br />فأطلِقْ عَبدَكَ بِسلامٍ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="إشَعْياء 49:6; لُوقا 2:10–11; لُوقا 3:6"> عَينايَ رأتا الخَلاصَ</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="إشَعْياء 62:1–2; إشَعْياء 42:1–4; المَزامِير 98:2–3"> الّذي هيَّأْتَهُ لِلشُعوبِ كُلِّها</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="إشَعْياء 60:1–3; إشَعْياء 49:6; اعمال الرسل 26:23"> نورًا لِهدايةِ الأُممِ<br />ومَجدًا لشَعبِكَ إِسرائيلَ“.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="لُوقا 1:65–66; إشَعْياء 8:18; لُوقا 2:48"> فتعَجَّبَ أبوهُ وأُمُّهُ ممَّا قالَهُ سِمعانُ فيهِ.</span>
<span className="v" id="v34" data-v="34" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:23; رسالَةُ بُطرُسَ الأُولَى 2:7–8; إشَعْياء 8:14–15"> وباركَهُما سِمعانُ وقالَ لمَريَمَ أُمِّهِ: ”هذا الطِّفلُ اَختارَهُ اللهُ لِسُقوطِ كثيرٍ مِنَ النَّاسِ وقِيامِ كثيرٍ مِنهُم في إِسرائيلَ. وهوَ علامَةٌ مِنَ اللهِ يُقاوِمونَها،</span>
<span className="v" id="v35" data-v="35" data-xref="التَّثْنِيَة 8:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:19; ﻣﺘﻰ 12:24–35"> لتَنكَشِفَ خفايا أفكارِهِم. وأمَّا أنتِ، فسَيفُ الأحزانِ سَينفُذُ في قلبِكِ“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="الخُرُوج 15:20; التَّكوين 30:13; اعمال الرسل 21:9"> وكانَت هُناكَ نَبِيَّةٌ كَبيرةٌ في السِّنِّ اَسمُها حَنَّةُ اَبنَةُ فَنُوئيلَ، مِنْ عَشيرَةِ آشيرَ، تَزوَّجت وهيَ بِكرٌ وعاشَت معَ زَوجِها سَبعَ سَنواتٍ،</span>
<span className="v" id="v37" data-v="37" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:5; المَزامِير 23:6; اعمال الرسل 14:23"> ثُمَّ بَقِيَت أرمَلةً فبَلغَتِ الرّابِعَةَ والثَّمانينَ مِنْ عُمرِها، لا تُفارِقُ الهَيكَلَ مُتَعَبِّدَةً بِالصّومِ والصّلاةِ ليلَ نَهارَ.</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 2:25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:15; مَرْقُس 15:43"> فحَضَرت في تِلكَ السّاعَةِ وحَمدَتِ الرّبّ وتَحدَّثت عَنِ الطِّفلِ يَسوعَ معَ كُلِّ مَنْ كانَ يَنتَظِرُ مِنَ اللهِ أنْ يَفديَ أورُشليمَ.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="لُوقا 2:4; غَلاطِيَّة 4:4–5; التَّثْنِيَة 12:32"> ولمَّا تمَّمَ يوسُفُ ومَريَمُ كُلَّ ما تَفرِضُهُ شريعةُ الرَّبِّ، رَجَعوا إلى الجليلِ، إلى مَدينتِهمِ النّاصِرَةِ.</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 2:52; لُوقا 1:80; إشَعْياء 11:1–5"> وكانَ الطِّفلُ يَسوعُ يَنمو ويَتقوَّى ويَمتَلئُ بالحِكمَةِ، وكانَت نِعمةُ اللهِ علَيهِ.</span>
</div>
</div>

## الصبي يسوع في الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="الخُرُوج 12:14; العَدَد 28:16; التَّثْنِيَة 12:18"> وكانَ والدا يَسوعَ يَذهبانِ كُلَّ سنَةٍ إلى أُورُشليمَ في عيدِ الفِصحِ.</span>
<span className="v" id="v42" data-v="42"> فلمَّا بلَغَ يَسوعُ الثَّانيَةَ عَشْرَةَ مِنْ عُمرِهِ، صَعِدوا إلى أُورُشليمَ كَعادَتِهِم في العيدِ.</span>
<span className="v" id="v43" data-v="43" data-xref="أخبارِ الأيّامِ الثّانِي 30:21–23; الخُرُوج 12:15; أخبارِ الأيّامِ الثّانِي 25:17"> وبَعدَما اَنقَضَت أيّامُ العيدِ وأخذوا طريقَ العودَةِ، بَقِيَ الصَّبيُّ يَسوعُ في أُورُشليمَ، ووالِداهُ لا يَعلَمانِ،</span>
<span className="v" id="v44" data-v="44" data-xref="المَزامِير 122:1–4; إشَعْياء 2:3; المَزامِير 42:4"> بل كانا يَظُنّانِ أنَّهُ معَ المُسافِرينَ. وبَعدَ مَسيرةِ يومٍ أخَذا يَبحثانِ عَنهُ عِندَ الأقاربِ والمعارِفِ،</span>
<span className="v" id="v45" data-v="45"> فما وَجداهُ. فرَجَعا إلى أُورُشليمَ يَبحَثانِ عَنهُ،</span>
<span className="v" id="v46" data-v="46" data-xref="لُوقا 5:17; إشَعْياء 49:1–2; إشَعْياء 50:4"> فوَجَداهُ بَعدَ ثلاثةِ أيّامٍ في الهَيكَلِ، جالِسًا معَ مُعَلِّمي الشَّريعَةِ، يستَمِعُ إليهِم ويَسألُهُم.</span>
<span className="v" id="v47" data-v="47" data-xref="ﻣﺘﻰ 7:28; لُوقا 4:32; المَزامِير 119:99"> وكانَ جميعُ سامِعيهِ في حَيرَةٍ مِنْ فَهمِهِ وأجوبَتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v48" data-v="48" data-xref="لُوقا 2:49; ﻣﺘﻰ 12:46; لُوقا 3:23"> ولمَّا رآهُ والداهُ تَعَجَّبا. وقالَت لَه أُمُّهُ: ”يا اَبني، لماذا فَعَلتَ بِنا هكذا؟ فأبوكَ وأنا تَعَذَّبْنا كثيرًا ونَحْنُ نَبحَثُ عَنكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v49" data-v="49" data-xref="يُوحَنّا 6:38; يُوحَنّا 8:29; مَلاخِي 3:1"> فأجابَهُما: ”ولماذا بَحَثتُما عنِّي؟ أما تَعرفانِ أنَّهُ يَجبُ أنْ أكونَ في بَيتِ أبي؟“</span>
<span className="v" id="v50" data-v="50" data-xref="لُوقا 18:34; لُوقا 9:45; مَرْقُس 9:32"> فما فَهِما معنى كلامِهِ.</span>
