## شفاء خادم أحد الضباط

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 8:5–13; لُوقا 7:1–10; ﻣﺘﻰ 7:28–29"> وبَعدَما ألقى يَسوعُ الأقوالَ في مسامِعِ النَّاسِ، دخَلَ كَفْرَناحُومَ.</span>
<span className="v" id="v2" data-v="2" data-xref="المُلُوكِ الثّانِي 5:2–3; التَّكوين 24:27; يُوحَنّا 4:46–47"> وكانَ لأحدِ الضُّبّاطِ خادِمٌ مريضٌ أشرَفَ على الموتِ، وكانَ عزيزًا علَيهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 8:5; يُوحَنّا 4:47; لُوقا 9:38"> فلمَّا سمِعَ بيَسوعَ، أرسَلَ إليهِ بَعضَ شُيوخِ اليَهودِ يسألُهُ أنْ يَجيءَ ليَشفِيَ خادِمَهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="رُؤيا يُوحَنّا 3:4; ﻣﺘﻰ 10:11; لُوقا 7:6–7"> فأقبلوا إلى يَسوعَ، وألحُّوا علَيهِ في القَولِ: ”هذا الرَّجُلُ يَستَحِقُّ أنْ تُساعِدَهُ،</span>
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الثّانِي 2:11–12; أخبارِ الأيّامِ الأوّلُ 29:3–9; المُلُوكِ الأوَّلُ 5:1"> لأنَّهُ يُحبُّ شَعبَنا، وهوَ الّذي بَنى لنا المَجمعَ“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 5:24; التَّكوين 32:10; يَعقُوب 4:6"> فذهَبَ يَسوعُ معهُم. ولمَّا اَقترَبَ مِنَ البَيتِ أرسَلَ إليهِ الضَّابِطُ بَعضَ أصحابِهِ يقولُ لَهُ: ”يا سيِّدُ، لا تُزعِجْ نَفسَكَ. أنا لا أستَحِقُّ أنْ تَدخُلَ تَحتَ سَقفِ بَيتي،</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 107:20; المَزامِير 33:9; لُوقا 4:36"> ولا أحسِبُ نَفسي أهلًا لأنْ أجيءَ إليكَ، ولكِنْ قُلْ كَلِمَةً فيُشفى خادِمي.</span>
<span className="v" id="v8" data-v="8" data-xref="كُولُوسِي 3:22; اعمال الرسل 23:17; اعمال الرسل 22:25–26"> فأنا مَرؤوسٌ ولي جُنودٌ تَحتَ أمري، أقولُ لِهذا: إِذهَبْ! فيذهَبُ، وللآخَرِ: تَعالَ! فيَجيءُ، ولِخادِمي: إعمَلْ هذا، فيَعمَلُ“.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 15:28; ﺭﻭﻣﻴﺔ 3:1–3; ﻣﺘﻰ 8:10"> فلمَّا سمِعَ يَسوعُ هذا الكلامَ، تعَجَبَّ مِنهُ واَلتَفتَ إلى الّذينَ يَتبَعونَهُ وقالَ: ”أقولُ لكُم: ما وجَدتُ مِثلَ هذا الإيمانِ حتّى في إِسرائيلَ!“</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 4:50–53; ﻣﺘﻰ 8:13; ﻣﺘﻰ 15:28"> ورجَعَ رُسُلُ الضَّابِطِ إلى البَيتِ، فوَجدوا أنَّ الخادِمَ تَعافَى.</span>
</div>
</div>

## إحياء ابن أرملة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 10:38"> وفي الغَدِ ذهَبَ يَسوعُ إلى مدينةٍ اَسمُها نايينُ، ومعَهُ تلاميذُهُ وجُمهورٌ كَبيرٌ.</span>
<span className="v" id="v12" data-v="12" data-xref="لُوقا 8:42; لُوقا 8:52; المُلُوكِ الأوَّلُ 17:9"> فلمَّا وصَلَ إلى بابِ المدينةِ، لَقِيَ مَيتًا مَحمولًا، وهوَ الإِبنُ الأوحَدُ لأُمِّهِ وهيَ أرمَلةٌ. وكانَ يُرافِقُها جمعٌ كبيرٌ مِنْ أهالي المدينةِ.</span>
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 103:13; إشَعْياء 63:9; المَزامِير 86:15"> فلمَّا رآها الرَّبُّ أشفَقَ علَيها وقالَ لها: ”لا تَبكي!“</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 5:25; يُوحَنّا 11:43–44; اعمال الرسل 9:40–41"> ودَنا مِنَ النَّعشِ ولَمَسهُ، فوقَفَ حامِلوهُ. فقَالَ: ”أيُّها الشَّابُّ، أقولُ لكَ: قُمْ!“</span>
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الثّانِي 13:21; المُلُوكِ الثّانِي 4:32–37; المُلُوكِ الأوَّلُ 17:23–24"> فجلَسَ الميتُ وأخَذَ يتكَلَّمُ، فسلَّمَهُ إلى أُمِّهِ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 9:8; لُوقا 7:39; لُوقا 1:68"> فسَيطَرَ الخَوفُ على الجَميعِ، وقالوا وهُم يُمَجِّدونَ اللهَ: ”ظهَرَ فينا نَبِيٌّ عظيمٌ، وتَفَقَّدَ اللهُ شَعبَهُ!“</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 4:24; مَرْقُس 6:14; ﻣﺘﻰ 9:26"> واَنتَشرَ هذا الخَبرُ عَنْ يَسوعَ في اليهودِيَّةِ كُلِّها وفي جميعِ النَّواحي المجاوِرَةِ لها.</span>
</div>
</div>

## يسوع ويوحنا المعمدان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 3:26; ﻣﺘﻰ 11:2–19"> وعرَفَ يوحنَّا مِنْ تلاميذِهِ كُلَّ هذِهِ الأُمورِ،</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 9:6–7; يُوحَنّا 4:25; التَّكوين 22:18"> فدَعا إِثنَينِ مِنهُم وأرسَلَهُما إلى الرَّبِّ ليَسألاهُ: ”هل أنتَ هوَ الآتي، أو نَنتَظرُ آخَرَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="لُوقا 7:19"> فجاءَ الرَّجلانِ إلى يَسوعَ وقالا لَه: ”أرسَلَنا يوحنَّا المَعمَدانُ لنَسأَلكَ: هل أنتَ هوَ الآتي، أو نَنتَظرُ آخَرَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 3:10; المَزامِير 90:7–9; العِبرانِيّين 12:6"> فشَفى يَسوعُ في تِلكَ السَّاعةِ كثيرًا مِنَ المُصابينَ بالأمراضِ والعاهاتِ والّذينَ فيهِم أرواحٌ شِرِّيرةٌ، وأعادَ البصَرَ إلى كثيرينَ مِنَ العُميانِ،</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 35:5–6; إشَعْياء 61:1–3; لُوقا 4:18"> ثُمَّ قالَ للرَّسولَينِ: ”إِرجِعا وأخبِرا يوحنَّا بِما رأيتُما وسَمِعتُما: العُميانُ يُبصِرونَ، والعُرجُ يَمشونَ، والبُرصُ يُطهَّرونَ، والصُّمُّ يَسمَعونَ، والموتى يقومونَ، والمساكينُ يَتلقَّونَ البِشارةَ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 11:6; رسالَةُ بُطرُسَ الأُولَى 2:7–8; ﺭﻭﻣﻴﺔ 9:32–33"> وهَنيئًا لِمَنْ لا يَفقِدُ إيمانَهُ بي“.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="أفَسُس 4:14; يُوحَنّا 1:23; لُوقا 1:80"> ولمَّا اَنصرَفَ رَسولا يوحنَّا، تَحدَّثَ يَسوعُ لِلجُموعِ عَنْ يوحنَّا، فقَالَ: ”ماذا خَرَجتُم إلى البرِّيَّةِ تَنظُرونَ؟ أَقَصَبَةً تَهُزُّها الرّيحُ؟</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 59:17; أسْتِير 8:15; أسْتِير 1:3"> بل ماذا خَرَجتُم لتَرَوا؟ أَرَجُلًا يَلبَسُ الثّيابَ النّاعِمَةَ؟ ولكِنَّ الّذينَ يَلبَسونَ الثّيابَ الفاخِرَةَ وأهلَ التَّرفِ هُم في قُصورِ المُلوكِ!</span>
<span className="v" id="v26" data-v="26" data-xref="لُوقا 1:76; يُوحَنّا 5:35; لُوقا 20:6"> قولوا لي: ماذا خَرجتُم تَنظُرونَ؟ أَنبيًّا؟ أقولُ لكُم: نعم، بل أفضلُ مِنْ نَبيٍّ.</span>
<span className="v" id="v27" data-v="27" data-xref="مَلاخِي 3:1; لُوقا 1:76; مَرْقُس 1:2"> فهوَ الّذي يقولُ فيهِ الكِتابُ: ها أنا أُرسِلُ رَسولي قُدَّامَكَ ليُهيِّئَ الطَّريقَ أمامَكَ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 13:16–17; ﻣﺘﻰ 11:11; كُولُوسِي 1:25–27"> أقولُ لكُم: ما وَلَدتِ النِّساءُ أعظمَ مِنْ يوحنَّا، ولكِنْ أصغَرُ الّذينَ في مَلكوتِ اللهِ أعظَمُ مِنهُ.</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 3:12; لُوقا 7:35; اعمال الرسل 18:25"> فجَميعُ الّذينَ سَمِعوا يوحنَّا حتّى جُباةُ الضَّرائِبِ أنفُسُهُم، أقَرُّوا بِبِرِّ اللهِ، فقَبِلوا مَعموديَّةَ يوحنَّا.</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 13:34; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:1; ﻣﺘﻰ 22:35"> وأمَّا الفَرِّيسيُّونَ وعُلماءُ الشَّريعَةِ، فرَفَضوا ما أرادَهُ اللهُ لهُم، فما تَعَمَّدوا على يَدِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="ﻣﺘﻰ 11:16–19; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:13; مَرْقُس 4:30"> وقالَ الرَّبُّ يَسوعُ: ”بِمَنْ أُشبِّهُ أبناءَ هذا الجيلِ؟ وماذا يُشبِهونَ؟</span>
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 11:16–19; إرْمِيا 5:3–5; الأمثال 17:16"> يُشبِهونَ أولادًا قاعِدينَ في السَّاحَةِ، يَصيحُ بَعضُهُم لِبَعضٍ: زَمَّرْنا لكُم فما رَقَصْتُم، ونَدَبنا لكُم فما بَكيتُم.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="لُوقا 1:15; مَرْقُس 1:6; يُوحَنّا 8:52"> جاءَ يوحنَّا المَعمَدانُ لا يأكُلُ الخُبزَ ولا يَشرَبُ الخَمرَ، فقُلتُم: فيهِ شَيطانٌ!</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 15:2; لُوقا 7:36; يُوحَنّا 12:2"> وجاءَ اَبنُ الإنسانِ يأكُلُ ويَشرَبُ، فقُلتُم: هذا رَجُلٌ أكولٌ وسِكِّيرٌ، وصَديقٌ لِجُباةِ الضَّرائِبِ والخاطِئينَ.</span>
<span className="v" id="v35" data-v="35" data-xref="الأمثال 8:32–36; ﻣﺘﻰ 11:19; هُوشَع 14:9"> لكِنَّ الحِكمَةَ يُبرِّرُها جميعُ أبنائِها“.</span>
</div>
</div>

## يسوع يغفر لامرأة خاطئة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="مَرْقُس 14:3–9; ﻣﺘﻰ 26:5–6; لُوقا 14:1"> ودعاهُ أحدُ الفَرِّيسيِّينَ إلى الطَّعامِ عِندَهُ، فدخَلَ بَيتَ الفَرِّيسيِّ وجلَسَ إلى المائِدَةِ.</span>
<span className="v" id="v37" data-v="37" data-xref="مَرْقُس 14:3–9; ﻣﺘﻰ 26:6–13; ﻣﺘﻰ 21:31"> وكانَ في المدينةِ اَمرَأةٌ خاطِئَةٌ، فعَلِمَت أنَّ يَسوعَ يأكُلُ في بَيتِ الفَرِّيسيِّ، فجاءَت ومعَها قارورَةُ طِيبٍ،</span>
<span className="v" id="v38" data-v="38" data-xref="المَزامِير 126:5–6; المَزامِير 51:17; التَّكوين 18:4"> ووقَفَت مِنْ خَلفٍ عِندَ قدَمَيْهِ وهيَ تَبكي، وأخذَت تَبُلُّ قَدَمَيهِ بِدُموعِها، وتَمسَحُهُما بشَعرِها، وتُقَبِّلُهُما، وتَدهَنُهُما بالطِّيبِ.</span>
<span className="v" id="v39" data-v="39" data-xref="لُوقا 7:16; لُوقا 15:2; مَرْقُس 7:21"> فلمَّا رأى الفَرِّيسيُّ صاحِبُ الدَّعوةِ ما جَرى، قالَ في نَفْسِهِ: ”لو كانَ هذا الرَّجُلُ نَبيًّا، لَعَرَفَ مَنْ هيَ هذِهِ المَرأةُ الّتي تَلمُسُهُ وما حالُها. فهيَ خاطِئَةٌ!“</span>
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 16:19; لُوقا 5:31; حِزْقِيال 33:31"> فقَالَ لَهُ يَسوعُ: ”يا سِمعانُ، عِندي ما أقولُهُ لكَ“. فقَالَ سِمعانُ: ”قُلْ، يا مُعَلِّمُ“.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="لُوقا 7:47; ﺭﻭﻣﻴﺔ 5:20; ﻣﺘﻰ 18:28"> فقَالَ يَسوعُ: ”كانَ لمُداينٍ دَينٌ على رَجُلينِ: خمسُ مئةِ دينارٍ على أحَدِهِما. وخَمسونَ على الآخَرِ.</span>
<span className="v" id="v42" data-v="42" data-xref="دانيال 9:18–19; كُولُوسِي 3:13; ﺭﻭﻣﻴﺔ 3:24"> وعجِزَ الرَّجُلانِ عَنْ إيفائِهِ دَينَهُ، فأعفاهُما مِنهُ. فأيُّهُما يكونُ أكثرَ حُبًّا لَهُ؟“</span>
</div>
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="المَزامِير 116:16–18; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:13–16; لُوقا 7:47"> فأجابَهُ سِمعانُ: ”أظُنُّ الّذي أعفاهُ مِنَ الأكثرِ“. فقَالَ لَهُ يَسوعُ: ”أصَبْتَ“.</span>
<span className="v" id="v44" data-v="44" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:10; القُضاة 19:21; التَّكوين 18:4"> واَلتَفَتَ إلى المَرأةِ وقالَ لسِمعانَ: ”أترى هذِهِ المَرأةَ؟ أنا دَخَلتُ بَيتَكَ، فما سكَبْتَ على قَدمَيَّ ماءً، وأمَّا هيَ فغَسَلتهُما بِدُموعِها ومسَحَتْهُما بشَعرِها.</span>
