## مثل الوكيل الخائن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:10; ﻣﺘﻰ 25:14–30; يَعقُوب 4:3"> وقالَ أيضًا لِتلاميذِهِ: ”كانَ رَجُلٌ غَنيٌّ وكانَ لَهُ وكِيلٌ، فجاءَ مَنْ أخبرَهُ بأنَّهُ يُبَدِّدُ أموالَهُ،</span>
<span className="v" id="v2" data-v="2" data-xref="الجامِعَة 11:9–10; رسالَةُ بُطرُسَ الأُولَى 4:5; رُؤيا يُوحَنّا 20:12"> فدَعاهُ وقالَ لَهُ: ما هذا الّذي أسمَعُ عَنكَ؟ أعطِني حِسابَ وكالَتِكَ، فأنتَ لا تَصلُحُ بَعدَ اليومِ لأنْ تكونَ وكيلًا لي.</span>
<span className="v" id="v3" data-v="3" data-xref="إرْمِيا 5:31; إشَعْياء 10:3; الأمثال 18:9"> فقَالَ الوكيلُ في نَفسِهِ: سَيَستَرِدُّ سيِّدي الوكالَةَ مِنِّي، فماذا أعمَلُ؟ لا أقوى على الفِلاحَةِ، وأستَحِي أنْ أستَعطِي.</span>
<span className="v" id="v4" data-v="4" data-xref="الأمثال 30:9; إرْمِيا 4:22; يَعقُوب 3:15"> ثُمَّ قالَ: عَرَفتُ ماذا أعمَلُ، حتّى إذا عزَلَني سيِّدي عَنِ الوِكالةِ، يَقبَلُني النَّاسُ في بُيوتِهِم.</span>
<span className="v" id="v5" data-v="5" data-xref="لُوقا 7:41–42; ﻣﺘﻰ 18:24"> فدَعا جميعَ الّذينَ علَيهِم دَينٌ لسَيِّدِهِ، وقالَ لأحدِهِم: كَم علَيكَ لسيِّدي؟</span>
<span className="v" id="v6" data-v="6" data-xref="تِيطُس 2:10; لُوقا 16:12; لُوقا 16:9"> أجابَهُ: مئةُ كَيلٍ مِنَ الزَّيتِ. فقَالَ لَهُ الوكيلُ: خُذْ صكوكَكَ واَجلِسَ في الحالِ واَكتُبْ خَمسينَ!</span>
<span className="v" id="v7" data-v="7" data-xref="نَشِيدِ الأنشاد 8:11–12; لُوقا 20:9; لُوقا 20:12"> وقالَ لآخرَ: وأنتَ، كَم علَيكَ لسَيِّدي؟ أجابَهُ: مِئةُ كَيلٍ مِنَ القَمحِ. فقَالَ لَهُ الوكيلُ: خُذْ صكوكَكَ واَكتُبْ ثَمانينَ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:5; أفَسُس 5:8; الأمثال 6:6–8"> فمَدَحَ السيِّدُ وكيلَهُ الخائِنَ على فِطْنَتِه، لأنَّ أبناءَ هذا العالَمِ أكثَرُ فِطنَةً مِنْ أبناءِ النّورِ في مُعامَلَةِ أمثالِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 19:21; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:17–19; دانيال 4:27"> وأنا أقولُ لكُم: إجعلوا لكُم أصدِقاءَ بالمالِ الباطِلِ، حتّى إذا نفَدَ قَبِلوكُم في المَساكِنِ الأبدِيَّةِ.</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 16:11–12; ﻣﺘﻰ 25:21; لُوقا 19:17"> مَنْ كانَ أمينًا على القَليلِ، كانَ أمينًا على الكَثيرِ. ومَنْ أساءَ الأمانَةَ في القَليلِ، أساءَ الأمانَةَ في الكَثيرِ.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 12:33; يَعقُوب 2:5; الأمثال 8:18–19"> وإذا كُنتُم غَيرَ أُمناءَ في المالِ الباطِلِ، فمَنْ يأتَمنُكُم في الغِنى الحَقِّ؟</span>
<span className="v" id="v12" data-v="12" data-xref="لُوقا 19:13–26; ﻣﺘﻰ 25:14–29; أيُّوب 1:21"> وإنْ كُنتُم غَيرَ أُمناءَ في ما هوَ لِغَيرِكُم، فمَنْ يُعطيكُم ما هوَ لكُم؟</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:15–16; ﻣﺘﻰ 6:24; يَعقُوب 4:4"> لا يَقدِرُ أحدٌ أنْ يَخدُمَ سيِّدينِ، لأنَّهُ إمَّا أنْ يُبغضَ أحدَهُما ويُحِبَّ الآخَرَ، وإمَّا أن يُوالي أحدَهُما ويَنبِذَ الآخَرَ. فأنتُم لا تَقدِرونَ أنْ تَخدُموا اللهَ والمالَ“.</span>
</div>
</div>

## الشريعة والطلاق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="لُوقا 23:35; لُوقا 20:47; إرْمِيا 6:13"> وكانَ الفَرِّيسيُّونَ، وهُم مِمَّنْ يُحبُّونَ المالَ، يَسمَعونَ هذا كُلَّهُ ويَهزأُونَ بِهِ.</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 16:7; الأمثال 21:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:5"> فقَالَ لهُم يَسوعُ: ”أنتُم تُبرِّرونَ أنفُسَكُم عِندَ النَّاسِ، لكِنَّ اللهَ يَعرِفُ ما في قُلوبِكُم. ورَفيعُ القَدْرِ عِندَ النَّاسِ رَجِسٌ عِندَ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 11:9–14; ﻣﺘﻰ 21:32; اعمال الرسل 3:24–25"> بَقِيَتِ الشَّريعةُ وتَعاليمُ الأنبياءِ إلى أنْ جاءَ يوحنَّا، ثُمَّ بَدأَتِ البِشارَةُ بِمَلكوتِ اللهِ، فأخَذَ كُلُّ إنسانٍ يُجاهِدُ ليَدْخُلَهُ قَسرًا.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 5:18; لُوقا 21:33; رسالَةُ بُطرُسَ الأُولَى 1:25"> ولكِنَّ زَوالَ السَّماءِ والأرضِ أسهَلُ مِنْ أنْ تَسقُطَ نُقطَةٌ واحدةٌ مِنَ الشَّريعةِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 19:9; مَرْقُس 10:11–12; ﻣﺘﻰ 5:31–32"> مَنْ طَلَّقَ اَمرأتَهُ وتَزوَّجَ غَيرَها زَنى، ومَنْ تزَوَّجَ اَمرأةً طَلَّقَها زَوجُها زَنى“.</span>
</div>
</div>

## مثل الغني ولعازر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="حِزْقِيال 16:49; عامُوس 6:4–6; لُوقا 18:24–25"> وقالَ يَسوعُ: ”كانَ رجُلٌ غنيٌّ يَلبَسُ الأُرجُوانَ والثِّيابَ الفاخِرَةَ ويُقيمُ الوَلائِمَ كُلَّ يومٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 3:2; صَمُوئيلَ الأوَّلُ 2:8; يَعقُوب 2:5"> وكانَ رَجُلٌ فَقيرٌ اَسمُهُ لِعازَرُ، تُغطِّي جِسمَهُ القُروحُ. وكانَ ينطَرِحُ عِندَ بابِ الرَّجُلِ الغَنيِّ،</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 15:27; مَرْقُس 7:28; يُوحَنّا 6:12"> ويَشتَهي أنْ يَشبَعَ مِنْ فَضَلاتِ مائِدَتِهِ، وكانَت الكِلابُ نَفسُها تَجيءُ وتَلحَسُ قُروحَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:24; المَزامِير 49:16–19; يَعقُوب 1:11"> وماتَ الفَقيرُ فحَمَلَتْهُ المَلائِكَةُ إلى جِوارِ إبراهيمَ. وماتَ الغَنيُّ ودُفِنَ.</span>
<span className="v" id="v23" data-v="23" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:4; لُوقا 16:28; ﻣﺘﻰ 18:9"> ورفَعَ الغَنيُّ عَينَيهِ وهوَ في الجَحيمِ يُقاسي العَذابَ، فَرأى إبراهيمَ عَنْ بُعدٍ ولِعازَرَ بِجانِبِه.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 4:10; رُؤيا يُوحَنّا 22:1; رُؤيا يُوحَنّا 7:16–17"> فنادى: إرحَمْني، يا أبي إبراهيمُ، وأرسِلْ لِعازَرَ ليَبُلَّ طرَفَ إصبَعِهِ في الماءِ ويُبَرِّدَ لِساني، لأنِّي أتَعذَّبُ كثيرًا في هذا اللَّهيبِ.</span>
<span className="v" id="v25" data-v="25" data-xref="لُوقا 6:24; رسالَةُ يُوحَنّا الأُولَى 2:15; المَزامِير 17:14"> فقَالَ لَهُ إبراهيمُ: تذَكَّرْ، يا اَبني، أنَّكَ نِلتَ نَصيبَكَ مِنَ الخَيراتِ في حياتِكَ، ونالَ لِعازَرُ نَصيبَهُ مِنَ البَلايا. وها هوَ الآنَ يَتَعزّى هُنا، وأنتَ تَتعذَّبُ هُناكَ.</span>
<span className="v" id="v26" data-v="26" data-xref="يُوحَنّا 3:36; المَزامِير 49:14; المَزامِير 50:22"> وفَوقَ كُلِّ هذا، فَبينَنا وبَينَكُم هُوّةٌ عَميقةٌ لا يَقدِرُ أحدٌ أن يجتازَها مِنْ عِندِنا إلَيكُم ولا مِنْ عِندِكُم إلينا.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27"> فقَالَ الغَنيُّ: أرجو مِنكَ، إذًا، يا أبي إبراهيمُ، أنْ تُرسِلَ لِعازرَ إلى بَيتِ أبي،</span>
<span className="v" id="v28" data-v="28" data-xref="اعمال الرسل 2:40; المَزامِير 49:12–13; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:6"> ليُنذِرَ إخوتي الخمسَةَ هُناكَ لِئَلَّا يَصيروا هُم أيضًا إلى مكانِ العذابِ هذا.</span>
<span className="v" id="v29" data-v="29" data-xref="اعمال الرسل 15:21; لُوقا 24:27; لُوقا 16:16"> فقَالَ لَه إبراهيمُ: عندَهُم موسى والأنبياءُ، فليَستَمِعوا إلَيهِم.</span>
<span className="v" id="v30" data-v="30" data-xref="رُؤيا يُوحَنّا 16:9–11; لُوقا 16:24; لُوقا 13:5"> فأجابَهُ الغَنيُّ: لا، يا أبي إبراهيمُ! ولكِنْ إذا قامَ واحدٌ مِنَ الأمواتِ وذهَبَ إلَيهِم يَتُوبونَ.</span>
<span className="v" id="v31" data-v="31" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:3; اعمال الرسل 19:8; يُوحَنّا 11:43–53"> فقَالَ لَهُ إبراهيمُ: إنْ كانوا لا يَستَمِعونَ إلى موسى والأنبياءِ، فهُم لا يَقتَنِعونَ ولو قامَ واحدٌ مِنَ الأمواتِ“.</span>
</div>
