## افتراق أبرام ولوط

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 12:9–20; يَشُوع 10:40; صموئيلَ الثّانِي 24:7"> فصَعِدَ أبرامُ مِنْ مِصْرَ إلى صحراءِ النَّقبِ، هوَ واَمْرَأتُهُ ولُوطٌ وكُلُّ ما يَملِكُ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 24:35; المَزامِير 112:1–3; التَّكوين 26:12–13"> وكانَ أبرامُ غنيًّا جِدًّا بِالمَاشيةِ والفِضَّةِ والذَّهَبِ،</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 12:8–9; التَّكوين 12:6"> وأخَذَ يَرتَحلُ مِنَ النَّقَبِ إلى بَيتِ إيلَ، حَيثُ نصَبَ خَيمَةً لَه مِنْ قَبلُ، بَينَ بَيتِ إيلَ وعايَ.</span>
<span className="v" id="v4" data-v="4" data-xref="إرْمِيا 29:12; التَّكوين 12:7–8; إشَعْياء 58:9"> وحَيثُ بَنى مذبَحًا. وهُناكَ دعا أبرامُ باَسمِ الرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="إرْمِيا 49:29; التَّكوين 25:27; التَّكوين 4:20"> وكانَ أيضًا للُوطٍ الّذي رافقَ أبرامَ غنَمٌ وبقَرٌ وخيامٌ،</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 36:6–7; الجامِعَة 5:10–11; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9"> فضاقَتِ الأرضُ بِسُكناهُما مَعًا لأنَّ أملاكَهُما كانَت كثيرةً.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 26:20; التَّكوين 12:6; يَعقُوب 3:16"> ووقَعَت خُصومَةٌ بَينَ رُعاةِ ماشيةِ أبرامَ ورُعاةِ ماشيةِ لُوطٍ، فيما الكنعانيُّونَ والفَرِّزِّيُّونَ مُقيمونَ في الأرضِ،</span>
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:6–7; اعمال الرسل 7:26; المَزامِير 133:1"> فقالَ أبرامُ لِلُوطٍ: ”لا تَكُنْ خُصومَةٌ بَيني وبَينَكَ، ولا بَينَ رُعاتي ورُعاتِكَ، فنَحنُ رَجُلانِ أخوَانِ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 20:15; التَّكوين 34:10; رسالَةُ بُطرُسَ الأُولَى 3:8–12"> الأرضُ كُلُّها بَينَ يَدَيكَ، فاَنفَصِلْ عنِّي. تذهَبُ إلى الشِّمالِ فأذهبُ إلى اليمينِ، أو إلى اليمينِ فأذهبُ إلى الشِّمالِ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 34:3; إشَعْياء 51:3; التَّكوين 14:8"> فرفَعَ لُوطٌ عَينيه فرأى وادي الأردنِّ باَتجاهِ صُوغَرَ رَيَّانَ كُلَّه، كَجَنَّةِ الرّبِّ، كأرضِ مِصْرَ، قَبلَ أنْ دَمَّرَ الرّبُّ سدومَ وعَمورةَ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 13:9; رسالَةُ بُطرُسَ الأُولَى 2:17; العِبرانِيّين 10:25"> فاَختارَ لُوطٌ لِنفْسِه وادي الأردنِّ ورَحَلَ إلى المَشرقِ. واَنفَصَلَ الواحدُ عَنِ الآخَرِ.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 19:29; التَّكوين 14:12; التَّكوين 19:1"> فأقامَ أبرامُ في أرضِ كنعانَ، وأقَامَ لُوطٌ في مُدُنِ الوادي، ونَقَل خيامَه إلى جِوارِ سدومَ.</span>
<span className="v" id="v13" data-v="13" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:10; رسالَةُ بُطرُسَ الثّانيَة 2:6–8; يَهُوذا 1:7"> وكانَ أهلُ سدومَ أشرارًا خاطِئين جِدًّا أمامَ الرّبِّ.</span>
</div>
</div>

## أبرام في حبرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 49:18; التَّثْنِيَة 3:27; التَّكوين 28:14"> وقالَ الرّبُّ لأبرامَ بَعدَما فارَقَهُ لُوطٌ: ”اَرفَعْ عينَكَ واَنظُرْ مِنَ المَوضِعِ الّذي أنتَ فيهِ شِمالًا وجَنوبًا وشَرقًا وغَربًا،</span>
<span className="v" id="v15" data-v="15" data-xref="أخبارِ الأيّامِ الثّانِي 20:7; التَّكوين 35:12; التَّكوين 15:18"> فهذِهِ الأرضُ كُلُّها أهَبُها لكَ ولِنسلِكَ إلى الأبدِ،</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 28:14; العَدَد 23:10; التَّكوين 32:12"> وأجعَلُ نسلَكَ كتُرابِ الأرضِ، فإنْ أمكنَ لأحدٍ أنْ يُحصيَهُ، فنَسلُكَ أيضًا يُحصَى.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 13:15; العَدَد 13:17–24"> قُمِ اَمشِ في الأرضِ طُولًا وعَرضًا، لأنِّي لكَ أهَبُها“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 18:1; التَّكوين 8:20; التَّكوين 35:27"> فاَنتقلَ أبرامُ بِخيامهِ وأتى إلى بلُّوطِ مَمْرا في حَبرونَ، فأقامَ هُناكَ وبَنى مذبَحًا للرّبِّ.</span>
</div>
</div>
