## ظهور الله في ممرا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 14:13; التَّكوين 48:3; التَّكوين 13:18"> وتَراءى الرّبُّ لإبراهيمَ عِندَ بَلُّوطِ مَمْرا، وهوَ جالسٌ بِبابِ الخَيمَةِ في حَرِّ النَّهارِ.</span>
<span className="v" id="v2" data-v="2" data-xref="العِبرانِيّين 13:2; التَّكوين 18:22; التَّكوين 19:1"> فرفَعَ عينيهِ ونظَرَ فرأى ثَلاثةَ رِجالٍ واقِفينَ أمامَهُ، فأسرعَ إلى لِقائِهِم مِنْ بابِ الخَيمةِ وسجدَ إلى الأرضِ</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 32:5"> وقالَ: ”إنْ كنتَ راضِيًا عليَّ يا سيّدي، فلا تَمُرَّ مُرورًا بِعَبدِكَ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 24:32; التَّكوين 19:2; التَّكوين 43:24"> دَعني أُقدِّمُ لكُم قليلًا مِنَ الماءِ، فتَغسِلونَ أرجلَكُم وتَستريحونَ تحتَ الشَّجَرَةِ.</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 104:15; القُضاة 13:15; القُضاة 6:18"> وأقدِّمُ لكُم كِسرةَ خُبزٍ، فتَسنِدونَ بها قُلوبَكُم ثُمَّ تَستَأنِفونَ سَفَرَكُم. وإلَّا لماذا مرَرْتُم بِعَبدِكُم؟“ فقالوا لهُ: ”إفعَلْ كما قُلتَ“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 13:33; العِبرانِيّين 13:2; اعمال الرسل 16:15"> فأسرعَ إبراهيمُ إلى سارةَ في الخَيمَةِ وقالَ لها: ”إعجِني في الحالِ ثَلاثةَ أكيالٍ مِنَ الدَّقيقِ الأبيضِ واَخبِزيها أرغِفةً“</span>
<span className="v" id="v7" data-v="7" data-xref="مَلاخِي 1:14; لُوقا 15:27; لُوقا 15:30"> واَندفعَ إبراهيمُ نحوَ البقَرِ فأخَذَ عِجْلًا رَخْصًا مُسَمَّنًا إلى الخادِم فأسرعَ في تَهيئَتِهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 19:3; القُضاة 13:15; رُؤيا يُوحَنّا 3:20"> ثُمَّ أخذَ زبدَةً ولَبَنًا والعِجْلَ الّذي هيَّأهُ، فوضَعَ هذا كُلَّه أمامَهُم. فأكلوا وهوَ واقفٌ أمامَهُم تَحتَ الشَّجرَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="تِيطُس 2:5; التَّكوين 24:67; التَّكوين 4:9"> ثُمَّ قالوا: ”أينَ سارةُ اَمرأتُكَ؟“ قالَ: ”هيَ في الخَيمةِ“.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 21:2; التَّكوين 17:21; ﺭﻭﻣﻴﺔ 9:8–9"> فقالَ أحدُهُم: ”سأرجِعُ إليكَ في مِثلِ هذا الوقتِ مِنَ السَّنَةِ المُقبِلَةِ ويكونُ لسارةَ اَمرأتِكَ اَبنٌ“. وكانَت سارةُ تسمَعُ عِندَ بابِ الخَيمةِ وراءَهُ</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 17:17; العِبرانِيّين 11:11–12; لُوقا 1:36"> وكانَ إبراهيمُ وسارةُ شَيخَينِ مُتَقَدِّمَينِ في السِّنِّ، واَمتنعَ أنْ يكونَ لسارةَ عادةٌ كما لِلنِّساءِ،</span>
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:6; التَّكوين 17:17; لُوقا 1:18–20"> فضَحِكت سارةُ في نفْسِها وقالت: ”أبَعدَما عَجَزْتُ وشاخَ زوجي تكونُ لي هذِهِ المِتعَةُ؟“</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 2:25"> فقالَ الرّبُّ لإبراهيمَ: ”ما بالُ سارةَ ضَحِكَت وقالت: ’أحقًّا ألِدُ وأنا الآنَ في شيخوخَتي؟‘</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 1:37; ﻣﺘﻰ 19:26; مَرْقُس 10:27"> أيصْعُبُ على الرّبِّ شيءٌ؟ في مِثلِ هذا الوقتِ مِنَ السَّنةِ المُقبِلَةِ أعودُ إليكَ ويكونُ لِسارةَ اَبنٌ“.</span>
<span className="v" id="v15" data-v="15" data-xref="رسالَةُ يُوحَنّا الأُولَى 1:8; الأمثال 28:13; أفَسُس 4:23"> فأنكَرَت سارَةُ وقالت: ”ما ضَحِكْتُ“، لأنَّها خافَت. فقالَ: ”لا، بل ضَحِكْتِ“.</span>
</div>
</div>

## وساطة إبراهيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="اعمال الرسل 15:3; اعمال الرسل 21:5; اعمال الرسل 20:38"> وقامَ الرِّجالُ مِنْ هُناكَ وتوَجَّهوا نحوَ سدومَ، وسارَ إبراهيمُ مَعَهُم لِيُشَيِّعَهُم.</span>
<span className="v" id="v17" data-v="17" data-xref="عامُوس 3:7; المَزامِير 25:14; يُوحَنّا 15:15"> فقالَ الرّبُّ في نفْسِه: ”هل أكتُمُ عَن إبراهيمَ ما أنوي أنْ أفعلَهُ،</span>
<span className="v" id="v18" data-v="18" data-xref="غَلاطِيَّة 3:8; التَّكوين 26:4; غَلاطِيَّة 3:14"> وإبراهيمُ سيكونُ أُمَّةً كبيرةً قويَّةً ويتَبارَكُ بِهِ جميعُ أُمَمِ الأرضِ؟</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 6:6–7; التَّثْنِيَة 4:9–10; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 3:4–5"> أنا اَختَرْتُه لِيُوصيَ بَنيه وأهلَ بَيتِه مِنْ بَعدِه بأنْ يسلُكوا في طُرُقي ويعمَلوا بالعَدلِ والإنصافِ، حتى أفيَ بما وَعَدْتُهُ بهِ“.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 19:13; حِزْقِيال 16:49–50; إشَعْياء 3:9"> وقالَ الرّبُّ لإبراهيمَ: ”كَثُرَتِ الشَّكوى على أهلِ سدومَ وعَمورةَ وعَظُمَت خطيئَتُهُم جدًّا</span>
<span className="v" id="v21" data-v="21" data-xref="الخُرُوج 3:8; التَّكوين 11:5; صَفَنْيا 1:12"> أنزِلُ وأرى هل فعَلوا ما يستوجِبُ الشَّكوَى الّتي بلَغَت إليَّ؟ أُريدُ أن أعلَمَ“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 19:1; اعمال الرسل 7:55; حِزْقِيال 22:30"> وتوَجَّهَ الرَّجلانِ مِنْ هُناكَ إلى سدومَ، وبقيَ إبراهيمُ واقفًا أمامَ الرّبِّ.</span>
<span className="v" id="v23" data-v="23" data-xref="العَدَد 16:22; صموئيلَ الثّانِي 24:17; التَّكوين 20:4"> فاَقتربَ إبراهيمُ وقالَ: ”أتُهلِكُ الصِّدِّيقَ معَ الشِّرِّيرِ؟</span>
<span className="v" id="v24" data-v="24" data-xref="إرْمِيا 5:1; إشَعْياء 1:9; ﻣﺘﻰ 7:13–14"> رُبَّما كانَ في المدينةِ خَمسونَ صِدِّيقًا، أتُهلِكُها كُلَّها ولا تصفَحُ عَنها مِنْ أجلِ الخَمسينَ صِدِّيقًا فيها؟</span>
<span className="v" id="v25" data-v="25" data-xref="أيُّوب 8:3; إشَعْياء 3:10–11; أيُّوب 8:20"> حَرامٌ علَيكَ أنْ تفعَلَ مِثلَ هذا الأمرِ، فتُهلِكُ الصِّدِّيقَ معَ الشِّرِّيرِ، فيَتساويانِ. حَرامٌ علَيكَ! أديَّانُ كُلِّ الأرضِ لا يَدينُ بالعَدلِ؟“</span>
<span className="v" id="v26" data-v="26" data-xref="إرْمِيا 5:1; ﻣﺘﻰ 24:22; إشَعْياء 65:8"> فقالَ الرّبُّ: ”إنْ وجَدْتُ خَمسينَ صِدِّيقًا في سدومَ صَفَحْتُ عَنِ المكانِ كلِّه إكرامًا لهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 3:19; إشَعْياء 6:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:47–48"> فأجابَ إبراهيمُ: ”ما بالي أُكَلِّمُ سيّدي هذا الكلامَ وأنا تُرابٌ ورمادٌ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 18:29; المُلُوكِ الأوَّلُ 20:32–33; التَّكوين 18:26"> رُبَّما نقَصَ الخَمسونَ بَريئًا خَمْسةً، أتُهلِكُ كُلَّ المدينةِ بِالخَمْسةِ؟“ فقالَ: ”لا أُهلِكُها إنْ وجَدتُ هناكَ خَمْسةً وأربعينَ“.</span>
<span className="v" id="v29" data-v="29" data-xref="أفَسُس 6:18; العِبرانِيّين 4:16"> وتابَعَ إبراهيمُ كلامَهُ فقالَ: ”وإنْ وجَدْتَ هُناكَ أربَعينَ؟“ فأجابَ: ”لا أفعَلُ إكرامًا للأربعينَ“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 89:7; المَزامِير 9:12; أسْتِير 4:11–16"> فقالَ إبراهيمُ: ”لا يغضَبْ سيّدي فأتَكلَّمَ: وإنْ وجَدْتَ هُناكَ ثَلاثينَ“. فأجابَ: ”لا أفعَلُ إنْ وجَدْتُ هُناكَ ثَلاثينَ“.</span>
<span className="v" id="v31" data-v="31" data-xref="لُوقا 11:8; التَّكوين 18:27; لُوقا 18:1"> فقالَ إبراهيمُ: ”ما بالي أُكْثِرُ الكلامَ أمامَ سيّدي: وإنْ وجَدْتَ هُناكَ عِشْرينَ؟“ قالَ: ”لا أُزيلُ المدينةَ إكرامًا لِلعشْرينَ“.</span>
<span className="v" id="v32" data-v="32" data-xref="يَعقُوب 5:15–17; القُضاة 6:39; ﻣﺘﻰ 7:7"> فقالَ إبراهيمُ: ”لا يغضَبْ سيّدي فأتكلَّمَ لآخرِ مرَّةٍ: وإنْ وجَدْتَ هُناكَ عشَرَةً؟“ قالَ: ”لا أُزِيلُ المدينةَ إكرامًا لِلعشَرَةِ“.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 18:22; التَّكوين 32:26; التَّكوين 31:55"> ومضَى الرّبُّ عِندَما فرَغَ مِنَ الكلامِ معَ إبراهيمَ، ورجَعَ إبراهيمُ إلى حَيثُ يُقيمُ.</span>
</div>
