## نسل قطورة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أخبارِ الأيّامِ الأوّلُ 1:32–33; التَّكوين 23:1–2; التَّكوين 28:1"> وعادَ إبراهيمُ فأخَذَ زوجةً اَسمُها قطورةُ،</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الأوّلُ 1:32–33; العَدَد 25:17–18; أيُّوب 2:11"> فولدَت لَه زِمرانَ ويَقشانَ ومَدانَ ومِديانَ ويِشباقَ وشُوحًا.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 25:13; إرْمِيا 49:8; حِزْقِيال 27:20"> وولَدَ يقشانُ شَبا وددانَ، وبنو ددانَ هُم أشُّوريمُ ولطوشيمُ ولأُمِّيمُ،</span>
<span className="v" id="v4" data-v="4" data-xref="إشَعْياء 60:6"> وبَنو مديانَ هُم عيفةُ وعِفرُ وحنوكُ وأبيداعُ وألْدعةُ. جميعُ هؤلاءِ بنو قطورةَ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 24:36; العِبرانِيّين 1:2; غَلاطِيَّة 3:29"> ووَهبَ إبراهيمُ لإسحَقَ جميعَ ما يَملِكُهُ،</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 21:14; القُضاة 6:3; اعمال الرسل 14:17"> وأمَّا بَنو سراريهِ فأعطاهُم عطايا وصرفَهُم، وهوَ بَعدُ حيٌّ، عَنْ إسحَقَ اَبنهِ إلى أرضِ المَشرِقِ.</span>
</div>

## موت إبراهيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 12:4"> وكانَ عددُ السِّنينَ الّتي عاشَها إبراهيمُ مئةً وخمْسًا وسبْعينَ سنَةً.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 35:28–29; التَّكوين 15:15; أخبارِ الأيّامِ الأوّلُ 29:28"> وفاضت روحُ إبراهيمَ وماتَ بشيبةٍ صالحةٍ، شيخًا شبِعَ مِنَ الحياةِ، واَنضمَّ إلى آبائِهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 50:13; التَّكوين 35:29; التَّكوين 49:29–30"> فدفنَه إسحَقُ وإسماعيلُ اَبناهُ في مغارةِ المكفيلةِ تُجاهَ مَمْرا، في حقلِ عَفرونَ بنِ صوحرَ الحثِّيِّ،</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 23:16; التَّكوين 49:31"> وهوَ الحقلُ الّذي اَشتراهُ إبراهيمُ مِنْ بَني حِثٍّ. هُناكَ دُفِنَ إبراهيمُ واَمرأتُهُ سارةُ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 22:17; التَّكوين 16:14; التَّكوين 24:62"> وبَعدَ موتِ إبراهيمَ باركَ اللهُ إسحَقَ اَبنَهُ. وأقامَ إسحَقُ عندَ بئرِ الحيِّ الرَّائي.</span>
</div>
</div>

## مواليد إسماعيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 83:6; التَّكوين 17:20; التَّكوين 21:13"> أمَّا بَنو إسماعيلَ بنِ إبراهيمَ الّذي ولَدتْهُ هاجرُ المِصْريَّةُ جاريةُ سارةَ لإبراهيمَ،</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الأوّلُ 1:29–31; إشَعْياء 60:7; نَشِيدِ الأنشاد 1:5"> فهذِهِ أسماؤُهُم بحسَبِ ولادتِهِم، على التوالي: نبايوتُ بِكرُ إسماعيلَ، وقيدارُ وأدَبْئيلُ ومبسامُ</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 21:11; إشَعْياء 21:16"> ومِشماعُ ودومَةُ ومسَّا</span>
<span className="v" id="v15" data-v="15" data-xref="أخبارِ الأيّامِ الأوّلُ 5:19; أيُّوب 2:11"> وحدارُ وتيما ويطورُ ونافيشُ وقِدْمَةُ.</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 17:20; التَّكوين 17:23"> هؤلاءِ هُم بَنو إسماعيلَ وهذِهِ أسماؤُهُم بِحَسبِ ديارِهِم وحُصونِهِم، وهُمُ اَثنا عشَرَ رئيسًا لِقبائِلِهم.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 15:15; التَّكوين 25:7–8"> وكانَ عدَدُ السِّنينَ الّتي عاشَها إسماعيلُ مئةً وسبْعًا وثلاثينَ سنَةً، ثُمَّ فاضت روحُهُ وماتَ واَنضمَّ إلى آبائِهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 16:12; التَّكوين 20:1; المُلُوكِ الثّانِي 23:29"> وكانَت مَساكِنُ بَني إسماعيلَ مِنْ حويلةَ إلى شورَ، شرقيَّ مِصْرَ في الطريقِ إلى أشورَ وهكذا نزَلَ إسماعيلُ بمواليِدِه قُبالةَ إخوتِهِ.</span>
</div>
</div>

## ولادة عيسو ويعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 1:2; لُوقا 3:34; اعمال الرسل 7:8"> وهذِهِ مواليدُ إسحَقَ بنِ إبراهيمَ: إبراهيمُ ولَدَ إسحَقَ.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 24:29; التَّكوين 24:67; التَّكوين 22:23"> وكانَ إسحقُ اَبنَ أربعينَ سنةً حينَ تزوَّجَ رفقةَ بنتَ بتوئيلَ الأراميِّ، أُختَ لابانَ، مِنْ سَهلِ أرامَ.</span>
<span className="v" id="v21" data-v="21" data-xref="لُوقا 1:13; إشَعْياء 58:9; عَزْرا 8:23"> وصلَّى إسحَقُ إلى الرّبِّ لأجلِ رفقةَ اَمرأتِهِ لأنَّها كانت عاقرًا، فاَستجابَ لَه الرّبُّ. وحبِلت رفقةُ</span>
<span className="v" id="v22" data-v="22" data-xref="صَمُوئيلَ الأوَّلُ 9:9; صَمُوئيلَ الأوَّلُ 28:6; صَمُوئيلَ الأوَّلُ 30:8"> واَزدحَمَ الولَدانِ في بطنِها، فقالت: ”إنْ كانَ الأمرُ هكذا فلماذا الحياةُ“. ومضَت لتسألَ الرّبَّ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﺭﻭﻣﻴﺔ 9:10–13; التَّكوين 27:40; التَّكوين 27:29"> فقالَ لها الرّبُّ: ”في بطنِكِ أمَّتانِ، ومِنْ أحشائِكِ يتفرَّعُ شعبانِ: شعبٌ يسودُ شعبًا، وكبيرٌ يَستعبِدُه صغيرٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24"> فلمَّا اَكتملت أيّامُ حَبلِها تَبيَّنَ أنَّ في بطنها توأمَينِ.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 27:11; التَّكوين 27:16; التَّكوين 27:23"> فخرَجَ الأوَّلُ أسمَرَ اللَّونِ كُلُّه كَفَروةِ شَعْرٍ فسمَّوهُ عيسو.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 27:36; هُوشَع 12:3; التَّكوين 38:28–30"> ثُمَّ خرَجَ أخوهُ ويدُهُ قابضةٌ على عَقِبِ عيسو، فسمَّوهُ يعقوبَ. وكانَ إسحَقُ اَبنَ ستِّينَ سنَةً حينَ ولدتْهُما رفقةُ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="العِبرانِيّين 11:9; التَّكوين 21:20; التَّكوين 10:9"> وكبُرَ الصَّبيَّانِ، فكانَ عيسو صيَّادًا ماهرًا ورجلًا يُحبُّ البرِّيَّةَ، ويعقوبُ رجلًا مُسالِمًا يلزمُ الخيامَ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 27:19; التَّكوين 27:31; التَّكوين 27:4"> فأحبَّ إسحَقُ عيسو لأنَّهُ اَستطابَ صيدَهُ، وأمَّا رفقةُ فأحبَّت يعقوبَ.</span>
</div>
</div>

## عيسو يتنازل عن بكوريته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="الأمثال 13:25; إشَعْياء 40:30–31; صَمُوئيلَ الأوَّلُ 14:28"> وطَبخَ يعقوبُ طبيخًا، فلمَّا عادَ عيسو مِنَ الحقلِ وهوَ خائِرٌ مِنَ الجُوعِ</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 36:1; الخُرُوج 15:15; التَّكوين 36:9"> قالَ ليعقوبَ: ”أطعِمني مِنْ هذا الأدامِ لأنِّي خائرٌ مِنَ الجُّوعِ“. لذلِكَ قيلَ لَه أدومُ.</span>
<span className="v" id="v31" data-v="31"> فقالَ لَه يعقوبُ: ”بِعْني اليومَ بَكُوريَّتَكَ“.</span>
<span className="v" id="v32" data-v="32" data-xref="مَلاخِي 3:14; أيُّوب 21:15; أيُّوب 22:17"> فأجابَ عيسو: ”أنا صائِرٌ إلى الموتِ، فما لي والبَكوريَّةُ“.</span>
<span className="v" id="v33" data-v="33" data-xref="العِبرانِيّين 12:16; التَّكوين 27:36; العِبرانِيّين 6:16"> فقالَ يعقوبُ: ”إحلِفْ ليَ اليومَ“. فحَلَفَ لَه وباعَ بَكُوريَّتَه لِيعقوبَ.</span>
<span className="v" id="v34" data-v="34" data-xref="فِيلِبِّي 3:18–19; العِبرانِيّين 12:16–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:32"> فأعطى يعقوبُ عيسو خُبزًا وطبيخًا مِنَ العدَسِ، فأكلَ وشربَ وقامَ ومضَى. واَستخفَّ عيسو بالبَكُوريَّةِ.</span>
</div>
