## سكّان الأرض

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 2:4; التَّكوين 9:7; التَّكوين 5:1"> وهؤُلاءِ مَواليدُ بَني نُوحٍ، سامٌ وحامٌ ويافَثُ، ومَنْ وُلِدَ لهُم مِنَ البَنينَ بَعدَ الطُّوفانِ:</span>
<span className="v" id="v2" data-v="2" data-xref="حِزْقِيال 38:2; حِزْقِيال 38:6; رُؤيا يُوحَنّا 20:8"> بَنو يافَثَ: جومَرُ وماجوجُ ومادايُ وياوانُ وتُوبالُ وماشَكُ وتِيراسُ.</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 27:14; إرْمِيا 51:27"> وبَنو جُومَرَ: أشْكَنَازُ ورِيفاثُ وتُوجَرْمَةُ.</span>
<span className="v" id="v4" data-v="4" data-xref="دانيال 11:30; إشَعْياء 23:12; العَدَد 24:24"> وبَنو ياوانَ: أليشَةُ وتَرشيشُ وكِتِّيمُ ودودَانيمُ.</span>
<span className="v" id="v5" data-v="5" data-xref="صَفَنْيا 2:11; إشَعْياء 42:4; إرْمِيا 2:10"> مِنْ هؤلاءِ تَفرَّقَت أُمَمُ البحرِ بِبُلدانِهِم وعَشائِرِهِم، كُلُّ جماعةٍ بِحَسَبِ لُغتِها.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="أخبارِ الأيّامِ الأوّلُ 1:8–16; حِزْقِيال 27:10; المَزامِير 105:23"> وبَنو حامٍ: كوشُ ومِصرايمُ وفُوطُ وكنعانُ.</span>
<span className="v" id="v7" data-v="7" data-xref="حِزْقِيال 27:22; حِزْقِيال 27:15; إشَعْياء 21:13"> وبَنو كُوشَ سَبا وحَويلَةُ وسَبْتَةُ ورَعْمَةُ وسَبْتَكا. وبَنو رَعْمَةَ: شَبا ودَدانُ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="ميخا 5:6"> وكُوشُ وَلَدَ نِمْرودَ، أوَّلَ جبَّارٍ في الأرضِ.</span>
<span className="v" id="v9" data-v="9" data-xref="أخبارِ الأيّامِ الثّانِي 28:22; التَّكوين 6:11; التَّكوين 25:27"> وكانَ صَيَّادًا جبَّارًا أمامَ الرّبِّ، ولِذلِكَ يُقالُ: كَنِمْرودَ، صَيَّادٌ جبَّارٌ أمامَ الرّبِّ.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 11:9; التَّكوين 11:2; التَّكوين 14:1"> واَبْتَدَأت مَملَكَتُه بابلُ وآرَكُ وآكَدُ، وكُلُّها في أَرضِ شِنْعارَ.</span>
<span className="v" id="v11" data-v="11" data-xref="ميخا 5:6; المَزامِير 83:8; ناحُوم 1:1"> مِنْ تِلكَ الأرضِ خرَجَ إلى أشُّورَ وبَنى نينوى ورَحوبوتَ عَيرَ وكالَحَ</span>
<span className="v" id="v12" data-v="12"> ورَسَنَ الّتي بَينَ نينوى وكالَحَ، وهيَ المدينةُ العظيمةُ.</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الأوّلُ 1:11–12; إرْمِيا 46:9; حِزْقِيال 30:5"> ومِصَرايمُ وَلَدَ بَني لُودَ وعَنامَ ولَهابَ ونَفتُوحَ</span>
<span className="v" id="v14" data-v="14" data-xref="إرْمِيا 47:4; أخبارِ الأيّامِ الأوّلُ 1:12; التَّثْنِيَة 2:23"> وفَتْروسَ وكَسْلوجَ، كما وَلَدَ كَفتورَ الّذينَ خَرَجَ مِنهُم الفلِسطيُّونَ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="العَدَد 34:2–15; الخُرُوج 34:11; التَّكوين 23:3"> وكنعانُ وَلَدَ صيدونَ بِكرَهُ، وحِثًا</span>
<span className="v" id="v16" data-v="16" data-xref="صموئيلَ الثّانِي 24:18; القُضاة 1:21; زَكَريّا 9:7"> واليَبُوسيِّينَ والأموريِّينَ والجِرْجاشيِّينَ</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 34:2"> والحِوِّيِّينَ والعَرْقيِّينَ والسِّينيِّينَ</span>
<span className="v" id="v18" data-v="18" data-xref="يَشُوع 18:22; صموئيلَ الثّانِي 8:9; حِزْقِيال 27:8"> والإرواديِّينَ والصَّماريِّينَ والحماتِيِّينَ. وبَعدَ ذلِك تَفرَّقَت عَشائِرُ الكنعانيِّينَ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 14:2; التَّكوين 19:24–25; يَشُوع 12:7–8"> وكانت أرضُ الكنعانيِّينَ مِنْ صيدونَ في اَتجاهِ جَرارَ إلى غَزَّةَ، ثُمَّ في اَتْجاهِ سَدومَ وعَمورَةَ وأدْمَةَ وَصَبُويِيمَ إلى لاشَعَ.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 10:6; التَّكوين 11:1–9"> هؤلاءِ بَنو حامٍ بِبُلدانِهِم وعَشائِرِهِم كُلُّ جماعَةٍ بِحَسَبِ لُغَتِها.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="العَدَد 24:24; التَّكوين 11:10–26"> ووُلِدَ لِسامٍ أيضًا بَنونَ، وهو أبو جميعِ بَني عابِرَ وأخو يافَثَ الأكبرِ.</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 66:19; إرْمِيا 25:25; التَّكوين 14:1–9"> وبَنو سامٍ: عيلامُ وأشُّورُ وأرفَكْشاد ولُودُ وأرامُ.</span>
<span className="v" id="v23" data-v="23" data-xref="أيُّوب 1:1; إرْمِيا 25:20"> وبَنو أرَامَ: عوصُ وحُولُ وجاثِرُ وماشُ.</span>
<span className="v" id="v24" data-v="24" data-xref="لُوقا 3:35; التَّكوين 11:12–15"> وأرفَكْشادُ وَلَدَ شالَحَ، وشالَحُ وَلَدَ عابِرَ.</span>
<span className="v" id="v25" data-v="25" data-xref="أخبارِ الأيّامِ الأوّلُ 1:19; التَّكوين 10:32; التَّثْنِيَة 32:8"> وَوُلِدَ لِعابِرَ اَبنانِ اَسمُ أحدِهما فالِجُ. لأنَّ في أيّامِهِ اَنقسمتِ الأرضُ، واَسمُ أخيهِ يقطانُ.</span>
<span className="v" id="v26" data-v="26" data-xref="أخبارِ الأيّامِ الأوّلُ 1:20–28"> ويقطانُ وَلَدَ ألْمُودادَ وشالَفَ وحضَرمَوتَ ويارَحَ</span>
<span className="v" id="v27" data-v="27" data-xref="أخبارِ الأيّامِ الأوّلُ 1:20–28"> وهَدورامَ وأُوزَالَ ودِقلَةَ</span>
<span className="v" id="v28" data-v="28" data-xref="أخبارِ الأيّامِ الأوّلُ 1:20–28; المُلُوكِ الأوَّلُ 10:1; التَّكوين 25:3"> وعُوبالَ وأبِيمايلَ وشَبا</span>
<span className="v" id="v29" data-v="29" data-xref="المُلُوكِ الأوَّلُ 9:28; أخبارِ الأيّامِ الأوّلُ 9:13; التَّكوين 2:11"> وأُوفيرَ وحَوِيلَةَ ويُوبَابَ. هؤلاءِ جميعُهُم بَنو يَقطانَ.</span>
<span className="v" id="v30" data-v="30" data-xref="العَدَد 23:7"> وكانَ مَسكِنُهُم مِنْ مِيشا في اَتْجاهِ سَفارَ إلى جبَلِ المَشرِقِ.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 10:5; التَّكوين 10:20; اعمال الرسل 17:26"> هؤلاءِ بَنو سامٍ بِبُلدانِهِم وعَشائِرِهم كُلُّ جَماعةٍ بِحسَبِ لُغَتِها.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 9:19; اعمال الرسل 17:26; التَّكوين 10:1"> هؤلاءِ عَشائرُ بَني نُوحٍ، بِمواليدِهِم وأُمَمِهِم، ومِنهُم تَفرَّقتِ الأُمَمُ في الأرضِ بَعدَ الطُّوفانِ.</span>
</div>
