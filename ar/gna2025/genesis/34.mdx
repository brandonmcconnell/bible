## اغتصاب دينة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 30:21; إرْمِيا 2:36; التَّكوين 30:13"> وخرَجت دينَةُ بِنتُ يعقوبَ مِنْ لَيئةَ اَمرأتِه لِتشاهِدَ بَناتِ تِلكَ الأرضِ،</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 5:28; أيُّوب 31:1; أيُّوب 31:9"> فرآها شكيمُ بنُ حَمُورَ الحِوِّيِّ، أميرِ تِلكَ الأرضِ، فأخذَها وضاجَعَها وأذَلَّها.</span>
<span className="v" id="v3" data-v="3" data-xref="هُوشَع 2:14; إشَعْياء 40:2; أخبارِ الأيّامِ الثّانِي 30:22"> وتعَلَّقَ قلبُه بِها فأحَبَّها ولاطَفَها</span>
<span className="v" id="v4" data-v="4" data-xref="القُضاة 14:2; صموئيلَ الثّانِي 13:13; التَّكوين 21:21"> وقالَ لأبيه حَمُورَ: ”خُذْ هذِهِ الفتاةَ زوجةً لي“.</span>
<span className="v" id="v5" data-v="5" data-xref="صموئيلَ الثّانِي 13:22; اللّاويِّين 10:3; صَمُوئيلَ الأوَّلُ 10:27"> وسَمِعَ يعقوبُ أنَّه دَنَّسَ دينةَ اَبنتَه، فسَكَتَ حتّى جاءَ بَنوهُ الّذينَ كانوا معَ ماشِيتِهِ في البرِّيَّةِ.</span>
</div>
</div>

## مصاهرة بني شكيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6"> فخرَجَ حَمورُ أبو شكيمَ إلى يعقوبَ لِيُكَلِّمَه،</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 7:15; القُضاة 20:6; التَّثْنِيَة 22:21"> وكان بنو يعقوبَ راجِعينَ مِنَ البرِّيَّةِ، فلمَّا سَمِعوا بِما جرى غضِبوا واَغتاظوا جِدًّا لأنَّ شكيمَ اَرتَكَبَ فِعْلًا شنيعًا في إِسرائيلَ بِمُضاجَعَةِ اَبنةِ يعقوبَ، وهذا عمَلٌ لا يُعمَلُ.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الأوَّلُ 11:2; المَزامِير 119:20; المَزامِير 84:2"> فقالَ لهُم حَمورُ: ”وقَعَ قلبُ شكيمَ اَبني في غرامِ اَبنتِكُم فأعطوها زوجةً لَه</span>
<span className="v" id="v9" data-v="9" data-xref="التَّثْنِيَة 7:3; التَّكوين 26:34–35; التَّكوين 19:14"> وصاهِرونا، أعطُونا بَناتِكُم وخُذُوا بَناتِنا</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 20:15; التَّكوين 47:27; التَّكوين 42:34"> وأقيموا معَنا. هذِهِ الأرضُ بَينَ أيديكُم فاَسكنوا فيها وجُولوا وتَمَلَّكوا“.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 33:15; التَّكوين 18:3"> وقالَ شكيمُ لأبي دينَةَ وإخوتِها: ”إرضَوا عليَّ، وما تطلُبُونَه مني أُعطِيه.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 22:16–17; التَّثْنِيَة 22:28–29; صَمُوئيلَ الأوَّلُ 18:25–27"> أَكثِروا عليَّ المَهْرَ والهَدايا فأُعطيَكُم قدْرَ ما تطلُبونَ، وأعطُوني الفتاةَ زوجةً لي“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 25:27–34; إشَعْياء 59:13; أيُّوب 13:4"> فأجابَ بَنو يعقوبَ شكيمَ وحَمورَ أَباهُ بِمَكْرٍ وكَيدٍ لأنَّ شكيمَ دنَّسَ دينةَ أُختَهُم،</span>
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 23:1–39; صموئيلَ الثّانِي 15:7; التَّكوين 17:11"> فقالوا لهُما: ”لا نقدِرُ أنْ نفعَلَ هذا، فنُعطي أختَنا لِرَجُلٍ غيرِ مَختونٍ، لأنَّه عارٌ عِندَنا.</span>
<span className="v" id="v15" data-v="15" data-xref="غَلاطِيَّة 4:12"> ولكِنَّنا نُوافقُكُم في حالٍ واحدةٍ: أنْ تصيروا مِثلَنا بأنْ يَختَتِنَ كُلُّ ذَكَرٍ مِنكُم،</span>
<span className="v" id="v16" data-v="16"> فنُعطِيكمُ بَناتِنا ونأخذُ لنا بَناتِكم ونُقيمُ معَكُم ونَصيرُ شعبًا واحدًا.</span>
<span className="v" id="v17" data-v="17"> وإِنْ لم تسمَعوا لنا وتَختَتِنوا نأخُذُ اَبنتَنَا ونمضي“.</span>
<span className="v" id="v18" data-v="18"> فحَسُنَ كلامُهُم عِندَ حَمورَ وشكيمَ اَبنِه.</span>
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 17:12; اعمال الرسل 13:50; صَمُوئيلَ الأوَّلُ 22:14"> ولم يتأخَّرِ الفتى عَنْ أنْ يفعَلَ ذلِكَ لِشَغَفِهِ باَبنةِ يعقوبَ، وكانَ أكرَمَ جمِيعِ أهلِ بَيتِ أبيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="راعُوث 4:1; الأمثال 31:23; عامُوس 5:12"> فلمَّا دخَلَ حَمورُ وشكيمُ اَبنُه بابَ مدينتِهِما قالا لأهلِها:</span>
<span className="v" id="v21" data-v="21"> ”هؤلاءِ القومُ مسالِمونَ لنا، فليُقِيموا في هذِهِ الأرضِ ويتَجوَّلوا فيها والأرضُ واسعةٌ جِدًّا أمَامَهم، فنتَزَوَّجُ بَناتِهم ونُزَوِّجُهُم بَناتِنا.</span>
<span className="v" id="v22" data-v="22"> ولَكِنَّهم لا يُوافِقُونَنا على أن يُقِيموا معَنا ونصِيرَ شعبًا واحدًا إلَّا إذا اَختَتَنَ كُلُّ ذَكَرٍ مِنّا مِثلَما هُم مُختَتنُونَ.</span>
<span className="v" id="v23" data-v="23" data-xref="الأمثال 1:12–13; يُوحَنّا 6:26–27; يُوحَنّا 2:16"> أفلا تَصيرُ مَواشيهِم ومُقتَنَياتُهُم وجميعُ بَهائِمِهِم لنا، إنْ نحنُ وافَقناهُم على هذا فأقاموا معَنا؟“</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 23:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:19; إشَعْياء 1:10–16"> فسَمِعَ لِحَمورَ وشكيمَ اَبنهِ جميعُ أهلِ الرَّأيِ في المدينةِ، واَختَتَنَ كُلُّ ذَكَرٍ في المدينةِ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 49:5–7; التَّكوين 29:33–34; العَدَد 31:7"> وفي اليومِ الثَّالثِ، وهُم بَعدُ مُتَوَجِّعونَ، أخَذَ كُلٌّ مِن اَبنَي يعقوبَ، شِمعونُ ولاوي أَخوي دينةَ، سَيفَه ودخَلا المدينة آمِنَينِ. فقتَلا كُلَّ ذَكَرٍ،</span>
<span className="v" id="v26" data-v="26" data-xref="صموئيلَ الثّانِي 2:26; التَّثْنِيَة 32:42; إشَعْياء 31:8"> ومِنهُم حَمورُ وشكيمُ اَبنُه، وأخذا دينَةَ مِنْ بَيتِ شكيمَ وخَرَجا.</span>
<span className="v" id="v27" data-v="27" data-xref="يَشُوع 7:13; يَشُوع 7:21; الخُرُوج 2:14"> ثُمَّ دخَلَ بَنو يعقوبَ كُلُّهم على القتلى ونهَبوا ما في المدينةِ اَنتِقامًا لِتدنيسِ أُختِهِم،</span>
<span className="v" id="v28" data-v="28" data-xref="أيُّوب 1:15–16; التَّثْنِيَة 8:17–18; أيُّوب 20:5"> وأخذوا غنَمَهُم وبقَرَهُم وحَميرَهُم وكُلَّ ما في المدينةِ وما في الحُقولِ.</span>
<span className="v" id="v29" data-v="29"> وسَبَوا وغَنِموا جميعَ ثروتِهِم وكُلَّ أطفالِهِم ونِسائِهِم وسائِرَ ما في البُيوتِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 105:12; يَشُوع 7:25; الخُرُوج 5:21"> فقالَ يعقوبُ لِشِمعونَ ولاوي: ”كَدَّرتُماني وجَلَبْتُما عليَّ البُؤسَ عِندَ أهلِ هذِهِ الأرضِ، وعِندَ الكنعانِيِّينَ والفَرزِّيّينَ، ونحنُ قليلٌ عَديدُنا. فإنِ اَجتمعوا عليَّ وهاجَموني هَلِكْتُ أنا وأهلُ بيتي“.</span>
<span className="v" id="v31" data-v="31" data-xref="الأمثال 6:34; التَّكوين 34:13; التَّكوين 49:7"> فأجابا: ”ولَكِنْ، أمِثلَ زانيةٍ يُعامِلُ أُختَنا؟“</span>
</div>
</div>
