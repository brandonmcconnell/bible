## السقوط

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:14; رُؤيا يُوحَنّا 12:9"> وكانَتِ الحَيَّةُ أحيلَ جميعِ حيواناتِ البرِّيَّةِ الّتي خلَقَها الرّبُّ الإلهُ. فقالت لِلمَرأةِ: ”أحقًّا قالَ اللهُ: لا تأكُلاَ مِنْ جميعِ شجَرِ الجَنَّةِ؟“</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 58:4"> فقالتِ المَرأةُ لِلحَيَّةِ: ”مِنْ ثَمَرِ شجَرِ الجَنَّةِ نأكُلُ،</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 2:16–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:17; الخُرُوج 19:12–13"> وأمَّا ثَمَرُ الشَّجَرَةِ الّتي في وسَطِ الجَنَّةِ فقالَ اللهُ: لا تأكُلا مِنهُ ولا تَمَسَّاهُ لئلّا تَموتا“.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 8:44; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:14"> فقالتِ الحَيَّةُ لِلمَرأةِ: ”لن تموتا،</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 28:2; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:13–15"> ولَكِنَّ اللهَ يعرِفُ أنكُما يومَ تأكُلانِ مِنْ ثَمَرِ تِلكَ الشَّجَرَةِ تنفَتِحُ أعينُكُما وتَصيرانِ مِثلَ اللهِ تعرفانِ الخيرَ والشَّرَّ“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:16; يَعقُوب 1:14–15; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:14"> ورأتِ المَرأةُ أنَّ الشَّجَرةَ طيِّبةٌ لِلمَأكلِ وشَهيّةٌ لِلعَينِ، وأنَّها باعِثَةٌ لِلفَهْمِ، فأخذَت مِنْ ثَمَرِها وأكَلَت وأعطَت زوجَها أيضًا، وكانَ مَعَها فأكَلَ.</span>
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 59:6; التَّكوين 2:25; التَّكوين 3:5"> فاَنْفَتَحت أعيُنُهما فعَرفا أنَّهُما عُريانَانِ، فخاطا مِنْ وَرَقِ التِّينِ وصَنَعا لهُما مآزِرَ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إرْمِيا 23:24; الأمثال 15:3; العِبرانِيّين 4:13"> وسَمِعَ آدمُ واَمرأتُه صوتَ الرّبِّ الإلهِ وهوَ يتمشَّى في الجَنَّةِ عِندَ المساءِ، فاَخْتبأَا مِنْ وَجهِ الرّبِّ الإِلهِ بَينَ شجَرِ الجَنَّةِ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 4:9; التَّكوين 16:8; التَّكوين 18:20–21"> فنادَى الرّبُّ الإلهُ آدمَ وقالَ لَه: ”أينَ أنتَ؟“</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 2:25; المَزامِير 119:120; إشَعْياء 47:3"> فأجابَ: ”سَمِعتُ صوتَكَ في الجَنَّةِ، فَخِفتُ ولأنِّي عُريانٌ اَختَبأتُ“.</span>
<span className="v" id="v11" data-v="11" data-xref="ﺭﻭﻣﻴﺔ 3:20; التَّكوين 4:10; المَزامِير 50:21"> فقالَ الرّبُّ الإلهُ: ”مَنْ عَرَّفَكَ أنَّكَ عُريانٌ؟ هل أكلتَ مِنَ الشَّجَرَةِ الّتي أوصَيتُكَ أنْ لا تأكُلَ مِنها؟“</span>
<span className="v" id="v12" data-v="12" data-xref="الأمثال 28:13; يَعقُوب 1:13–15; أيُّوب 31:33"> فقالَ آدمُ: ”المرأةُ الّتي أعطَيتني لِتَكونَ مَعي هيَ أعطتني مِنَ الشَّجرَةِ فأكَلْتُ“.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:14; التَّكوين 4:10–12"> فقالَ الرّبُّ الإلهُ لِلمرأةِ: ”لِماذا فَعَلتِ هذا؟“ فأجابَتِ المَرأةُ: ”الحَيَّةُ أغوتْني فأكلْتُ“.</span>
<span className="v" id="v14" data-v="14" data-xref="ميخا 7:17; إشَعْياء 65:25; المَزامِير 72:9"> فقالَ الرّبُّ الإلهُ لِلحَيَّةِ: ”لأنَّكِ فعَلْتِ هذا فأنتِ مَلعونَةٌ مِنْ بَينِ جميعِ البَهائِمِ وجميعِ وُحوشِ البَرِّ. على بَطنِك تَزحفينَ وتُرابًا تأكُلينَ طُولَ أيّامِ حياتِكِ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 16:20; رسالَةُ يُوحَنّا الأُولَى 3:8; رُؤيا يُوحَنّا 12:17"> بَينَكِ وبَينَ المَرأةِ أُقيمُ عَداوةً وبَينَ نسلِكِ ونسلِها فهوَ يتَرقَّبُ مِنكِ الرّأسَ وأنتِ تتَرقَّبينَ مِنْه العَقِبَ“.</span>
<span className="v" id="v16" data-v="16" data-xref="تِيطُس 2:5; كُولُوسِي 3:18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:3"> وقالَ لِلمَرأةِ: ”أزيدُ تعَبَكِ حينَ تَحبَلينَ، وبالأوجاعِ تَلِدينَ البَنينَ. إلى زَوجِكِ يكونُ اَشتياقُكِ، وهوَ علَيكِ يسودُ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 8:20–22; التَّكوين 5:29; إشَعْياء 24:5–6"> وقالَ لآدمَ: ”لأنَّكَ سَمِعتَ كلامَ اَمْرأتِكَ، فأكَلْتَ مِنَ الشَّجَرَةِ الّتي أوصَيتُكَ أنْ لا تأكُلَ مِنها تكونُ الأرضُ مَلعونَةً بِسبَبِكَ. بِكَدِّكَ تأكُلُ طَعامَكَ مِنها طولَ أيّامِ حياتِكَ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 13:7; أيُّوب 5:5; إشَعْياء 7:23"> شَوكًا وعَوسجًا تنبِتُ لكَ، ومِنْ عُشْبِ الحقلِ تقتاتُ.</span>
<span className="v" id="v19" data-v="19" data-xref="الجامِعَة 12:7; أيُّوب 34:15; المَزامِير 104:29"> بِعَرَقِ جبينِكَ تأكلُ خُبزَكَ حتّى تَعودَ إلى الأرضِ لأنَّكَ مِنها أُخِذْتَ. فأنتَ تُرابٌ، وإلى التُّرابِ تعُودُ“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 2:23; ﻣﺘﻰ 1:23; التَّكوين 5:29"> وسمَّى آدمُ اَمرأتَهُ حَوَّاءَ، لأنَّها أُمُّ كُلِّ حيٍّ.</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 61:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:21; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:2–3"> وصنَعَ الرّبُّ الإلهُ لِآدمَ واَمرأتِهِ ثيابًا مِنْ جِلْدٍ وكساهُما.</span>
<span className="v" id="v22" data-v="22" data-xref="رُؤيا يُوحَنّا 2:7; رُؤيا يُوحَنّا 22:14; الأمثال 3:18"> وقالَ الرّبُّ الإلهُ: ”صارَ آدمُ كواحدٍ مِنَّا يعرِفُ الخيرَ والشَّرَّ. والآنَ لعلَّهُ يَمُدُّ يَدَهُ إلى شجَرةِ الحياةِ أيضًا فيأخُذَ مِنها ويأكُلَ، فيحيا إلى الأبدِ“.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 4:2; التَّكوين 3:19; التَّكوين 4:12"> فأخرَجَ الرّبُّ الإلهُ آدمَ مِنْ جَنَّةِ عَدْنٍ لِيفلَحَ الأرضَ التي أُخِذَ مِنها.</span>
<span className="v" id="v24" data-v="24" data-xref="العِبرانِيّين 1:7; المَزامِير 104:4; الخُرُوج 25:18–22"> فطَرَدَ آدمَ وأقامَ الكروبِيمَ شَرقيَّ جَنَّةِ عَدْنٍ، وسَيفًا مُشتَعِلًا مُتَقَلِّبًا لِحِراسةِ الطَّريقِ إلى شجَرةِ الحياةِ.</span>
</div>
