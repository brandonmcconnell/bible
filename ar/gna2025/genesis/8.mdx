## نهاية الطوفان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 30:22; التَّكوين 19:29; المَزامِير 136:23"> وتذكَّرَ اللهُ نُوحًا وجميعَ الوحوشِ والبَهائِمِ الّتي معَهُ في السَّفينةِ، فأرسلَ ريحًا على الأرضِ فتَناقَصَتِ المياهُ</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 7:11; ﻣﺘﻰ 8:26–27; الأمثال 8:28"> واَنْسَدَّت ينابيعُ الغَمْرِ ونوافِذُ السَّماءِ، فتَوقَّفَ سُقوطُ المَطَرِ مِنَ السَّماءِ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 7:24; التَّكوين 7:11"> وتراجَعَتِ المياهُ عَنِ الأرضِ شيئًا فشيئًا حتى نَقَصَت بَعدَ مئَةٍ وخَمْسينَ يومًا.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الثّانِي 19:37; إرْمِيا 51:27; إشَعْياء 37:38"> فاَستَقَرَّتِ السَّفينةُ في الشَّهرِ السَّابعِ، في اليومِ السَّابعَ عشَرَ مِنهُ، على جِبالِ أراراطَ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 7:11"> وأخذَتِ المياهُ تتَناقصُ إلى الشَّهرِ العاشِرِ حتى ظَهَرت رؤوسُ الجِبالِ في أوَّلِ يومٍ مِنهُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 6:16; دانيال 6:10"> وكانَ بَعدَ أربَعينَ يومًا أنْ فتَحَ نُوحٌ النافذَةَ التي صنَعَها في السَّفينةِ</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 147:9; المُلُوكِ الأوَّلُ 17:6; المُلُوكِ الأوَّلُ 17:4"> وأرسَلَ الغُرابَ. فخرَجَ الغُرابُ وأخذَ يَروحُ ويَجيءُ إلى أنْ جَفَّتِ المياهُ عَنِ الأرضِ.</span>
<span className="v" id="v8" data-v="8" data-xref="نَشِيدِ الأنشاد 2:14; نَشِيدِ الأنشاد 1:15; ﻣﺘﻰ 10:16"> ثُمَّ أرسَلَ الحمامَةَ مِنْ عِندِه لِيرَى هل قلَّتِ المياهُ على وجهِ الأرضِ،</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 11:28; المَزامِير 116:7; التَّثْنِيَة 28:65"> فلم تَجِدِ الحمامَةُ مُستَقَرًّا لِرِجْلِها، فرَجَعَت إلى نُوحٍ في السَّفينةِ، لأنَّ المياهَ كانَت على وجهِ الأرضِ كُلِّها. ومَدَّ نُوحٌ يَدَهُ فأمسكَها وأدخلَهَا إلى السَّفينةِ معَهُ.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 7:4; التَّكوين 7:10; التَّكوين 8:12"> وَاَنْتَظَرَ أيضًا سَبْعَةَ أيّامٍ أُخَرَ وعادَ فأرسَلَ الحمامَةَ مِنَ السَّفينةِ،</span>
<span className="v" id="v11" data-v="11" data-xref="نَحَمْيا 8:15; زَكَريّا 4:12–14; ﺭﻭﻣﻴﺔ 10:15"> فرجَعَتِ الحمامَةُ إليهِ في المساءِ تحمِلُ في فَمِها ورَقَةَ زيتونٍ خضراءَ، فعَلِمَ نُوحٌ أنَّ المياهَ قَلَّت على الأَرضِ.</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 25:9; إشَعْياء 30:18; التَّكوين 8:10"> واَنْتَظَرَ أيضًا سَبْعةَ أيّامٍ أُخَرَ وأرسَلَ الحمامَةَ، فلم ترجعْ إليهِ هذِهِ المَرَّةَ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 7:11"> وفي السَّنةِ الواحدَةِ والسِّتِّ مئةٍ مِنْ عُمر نُوحٍ، في اليومِ الأوَّلِ مِنَ الشَّهرِ الأوَّلِ، جَفَّتِ المياهُ عَنِ الأرضِ. فرفَعَ نُوحٌ غِطاءَ السَّفينةِ ونظَرَ فرأى وجهَ الأرضِ ناشِفًا.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 7:13–14; التَّكوين 7:11"> وفي الشَّهرِ الثَّاني، في اليومِ السّابعِ والعِشرينَ مِنهُ، يَبِسَتِ الأرضُ.</span>
</div>
</div>

## الخروج من السفينة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15"> فخاطَبَ اللهُ نُوحًا قالَ:</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 7:13; المَزامِير 121:8; المَزامِير 91:11"> ”اَخْرُجْ مِنَ السَّفينةِ، أنتَ واَمرأتُكَ وبَنوكَ وَنِسوةُ بَنيكَ معَكَ،</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 1:22; التَّكوين 9:1; إرْمِيا 31:27–28"> وأخرِجْ كُلَّ حيوانٍ معَكَ مِنَ الطُّيورِ والبَهائِمِ وسائرِ ما يَدِبُّ على الأرضِ، فتَتَوالدَ في الأرضِ وتَنموَ وتكثُرَ علَيها“.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 121:8"> فخَرجَ نُوحٌ وبَنوهُ واَمرأتُه ونِسوَةُ بَنيهِ معَه.</span>
<span className="v" id="v19" data-v="19"> وكُلُّ حيوانٍ مِنَ البَهائِمِ والطُّيورِ وسائِرِ ما يَدِبُّ على الأرضِ. خرَجَت بِأصنافِها مِنَ السَّفينةِ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="العِبرانِيّين 13:15–16; ﺭﻭﻣﻴﺔ 12:1; التَّكوين 22:9"> وبنى نُوحٌ مذبَحًا للرّبِّ وأخذَ مِنْ جميعِ البَهائِمِ والطُّيورِ الطَّاهرَةِ بِحسَبِ الشَّريعةِ، فأصعَدَ مُحرَقاتٍ على المذبَحِ.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:15; أفَسُس 5:2; إشَعْياء 54:9–10"> وتَنَسَّمَ الرّبُّ رائِحَةَ الرِّضَى فقالَ في قلبهِ: ”لن ألعنَ الأرضَ مرَّةً أُخرى بِسَبَبِ الإنسانِ، فهوَ يتَصَوَّرُ الشَّرَّ في قلبِه مُنذُ حداثتهِ، ولن أُهلِكَ كُلَّ حيٍّ كما فعَلْتُ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يَعقُوب 5:7; المَزامِير 74:16–17; إرْمِيا 5:24"> وما دامتِ الأرضُ باقيةً،<br />فالزَّرعُ والحَصادُ،<br />والبَردُ والحَرُّ،<br />والصَّيفُ والشِّتاءُ،<br />واللَّيلُ والنَّهارُ،<br />لا تبطُلُ أبدًا“.</span>
</div>
