## لقاء يعقوب براحيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="القُضاة 6:3; القُضاة 6:33; هُوشَع 12:12"> ونهضَ يعقوبُ ومضَى إلى أرضِ المشرِقِ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 4:14; إشَعْياء 49:10; يُوحَنّا 4:6"> ونظَرَ فرأى في البرِّيَّةِ بئرًا وثلاثةَ قُطعانٍ مِنَ الغنَمِ رابضةً عِندَها، لأنَّ الرُّعاةَ كانوا يَسقونَ القُطعانَ مِنْ تلكَ البئرِ، والحجَرُ على فَمِ البئرِ كانَ كبيرًا.</span>
<span className="v" id="v3" data-v="3"> فكانوا يَجمعونَ القُطعانَ كُلَّها هُناكَ ويُدحرِجونَ الحجَرَ عنْ فَمِ البئرِ، فيَسقونَ الغنَمَ ثمَّ يردُّونَ الحجَرَ إلى موضِعهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 28:10; التَّكوين 27:43; اعمال الرسل 7:2"> فقالَ لهُم يعقوبُ: ”مِنْ أينَ أنتُم يا إخوتي؟“ قالوا: ”مِنْ حارانَ“.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 24:24; التَّكوين 24:29; التَّكوين 31:53"> فقالَ لَهم: ”أتعرفونَ لابانَ بنَ ناحورَ؟“ فقالوا: ”نَعرِفُهُ“.</span>
<span className="v" id="v6" data-v="6" data-xref="صموئيلَ الثّانِي 20:9; التَّكوين 37:14; الخُرُوج 18:7"> فقالَ لهُم: ”هل هوَ بخيرٍ؟“ قالوا: ”بِخيرٍ، وها راحيلُ اَبنتُهُ مُقبِلَةً معَ الغنَمِ“.</span>
<span className="v" id="v7" data-v="7" data-xref="أفَسُس 5:16; غَلاطِيَّة 6:9–10"> فقالَ لهُم: ”النَّهارُ طويلٌ بَعدُ، وما الآنَ وقتُ تجميعِ المواشي، فاَسقوا الغنَمَ وخُذوها إلى المرعَى“.</span>
<span className="v" id="v8" data-v="8" data-xref="مَرْقُس 16:3; لُوقا 24:2; التَّكوين 34:14"> قالوا: ”لا نقدِرُ حتّى تجتمِعَ القُطعانُ كُلُّها، ونُدحرِج الحجَرَ عن فَمِ البِئرِ، فنسقي الغنَمَ“.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 2:21; نَشِيدِ الأنشاد 1:7–8; الخُرُوج 2:15–16"> وبينَما هوَ يُحادِثُهُم جاءت راحيلُ معَ غنَمِ لابانَ أبيها، لأنَّها كانَت راعيةً.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 2:17"> فلمَّا رآها يعقوبُ معَ غنَمِ لابانَ خالِه، تقدَّمَ ودحرَجَ الحجَرَ عَنْ فَمِ البئرِ وسقى الغنَمَ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 33:4; التَّكوين 45:14–15; التَّكوين 45:2"> وقبَّلَ يعقوبُ راحيلَ ورفعَ صوتَه وبكى،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 24:28; التَّكوين 13:8; التَّكوين 14:14–16"> وأخبرَها أنَّهُ نسيبُ أبيها واَبنُ رِفقةَ، فأسرعت وأخبرت أباها.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 24:29; اعمال الرسل 20:37; ﺭﻭﻣﻴﺔ 16:16"> فلمَّا سَمِعَ لابانُ خبرَ قدومِ يعقوبَ اَبنِ أختِهِ هَبَّ للقائِه فعانقَهُ وقَبَّلَه وجاءَ به إلى بَيتِه. وأخبرَ يعقوبُ لابانَ بكلِّ ما جرى،</span>
<span className="v" id="v14" data-v="14" data-xref="صموئيلَ الثّانِي 19:12–13; التَّكوين 2:23; القُضاة 9:2"> فقالَ لَه لابانُ: ”أنتَ حَقًّا مِنْ عظمي ولَحمي“. وأقامَ يعقوبُ عِندَه شهرًا.</span>
</div>
</div>

## زواج يعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 30:28; التَّكوين 31:7"> ثُمَّ قالَ لابانُ ليعقوبَ: ”ألأنَّكَ نسيبي تخدُمُني مَجَّانًا؟ أخبرْني ما أُجرتُكَ؟“</span>
<span className="v" id="v16" data-v="16" data-xref="راعُوث 4:11; التَّكوين 49:31; التَّكوين 31:4"> وكانَ للابانَ اَبنتانِ، اَسمُ الكُبرى لَيئةُ واَسمُ الصُّغرى راحيلُ.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 12:11; التَّكوين 46:19–22; التَّكوين 24:16"> وكانَت لَيئةُ ضعيفةَ العينينِ، وراحيلُ حسنَةَ الهَيئةِ جميلةَ المنظَرِ.</span>
<span className="v" id="v18" data-v="18" data-xref="هُوشَع 12:12; الخُرُوج 22:16–17; التَّكوين 34:12"> فأحبَّ يعقوبُ راحيلَ، فقالَ للابانَ: ”أخدُمُكَ سَبْعَ سِنينَ وتُعطيني راحيلَ زوجةً لي“.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 6:11; إشَعْياء 6:5; المَزامِير 12:2"> فقالَ لابانُ: ”أنْ تأخُذَها أنتَ خيرٌ مِنْ أنْ أُعطيَها لِغيرِكَ. فأقمْ عِندي“.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:7; أفَسُس 5:2; هُوشَع 12:12"> فخَدمَهُ يعقوبُ سَبْعَ سِنينَ ليأخُذَ راحيلَ، وكانَت هذِهِ المُدَّةُ قصيرةً في نظرهِ لأنَّه كانَ يُحبُّها.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="القُضاة 15:1; التَّكوين 29:20; التَّكوين 4:1"> وقالَ يعقوبُ للابانَ: ”إكتملتِ المُدَّةُ فأعطِني اَمرأتي لأتزوَّجَها“.</span>
<span className="v" id="v22" data-v="22" data-xref="رُؤيا يُوحَنّا 19:9; راعُوث 4:10–13; القُضاة 14:10–18"> فجَمعَ لابانُ كُلَّ أهلِ حارانَ وصنعَ لهُم وليمةً،</span>
<span className="v" id="v23" data-v="23" data-xref="ميخا 7:5; التَّكوين 38:14–15; التَّكوين 24:65"> وعِندَ الغُروبِ أخذَ لَيئةَ بَدلَ راحيلَ وجاءَ بها إلى يعقوبَ فدخلَ علَيها.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 16:1; التَّكوين 30:9–12; التَّكوين 46:18"> ووهبَ لابانُ جاريتَه زِلْفةَ لابنتهِ لَيئةَ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 7:2; الأمثال 11:31; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:13"> فلمَّا طَلعَ الصَّباحُ عرَفَ يعقوبُ أنَّها لَيئةُ، فقالَ للابانَ: ”ماذا فعلْتَ بي؟ أما خدمتُكَ لآخذَ راحيلَ؟ فلماذا خدعْتني؟“</span>
<span className="v" id="v26" data-v="26"> فأجابَ لابانُ: ”في بلادِنا لا تتزوَّجُ الصُّغرى قَبلَ الكُبرى.</span>
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 18:18; القُضاة 14:12; التَّكوين 2:2–3"> أكمِلْ أسبوعَ زواجِكَ مِنْ لَيئةَ، فأُعطيَكَ راحيلَ أيضًا بَدلَ سَبْعِ سِنينَ أُخرى مِنَ الخدمةِ عندي“.</span>
<span className="v" id="v28" data-v="28"> فوافقَ يعقوبُ وأكمَلَ أُسبوعَ زواجهِ مِنْ لَيئةَ، فأعطاهُ لابانُ راحيلَ اَمرأةً لَه.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 35:22; التَّكوين 29:24; التَّكوين 30:3–8"> ووَهبَ لابانُ جاريتَه بِلْهةَ جاريةً لراحيلَ.</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 31:41; التَّكوين 29:20; التَّكوين 31:15"> فدخَلَ يعقوبُ على راحيلَ أيضًا وأحبَّها أكثرَ مِنْ لَيئةَ. وعادَ فخدمَ عِندَ لابانَ سَبْعَ سِنينَ أُخرى.</span>
</div>
</div>

## بنو يعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="صَمُوئيلَ الأوَّلُ 2:21; ﻣﺘﻰ 10:37; التَّثْنِيَة 21:15"> ورأى الرّبُّ أنَّ لَيئةَ مكروهَةٌ فجعَلَها وَلودًا وأمَّا راحيلُ فكانَت عاقرًا.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّثْنِيَة 26:7; الخُرُوج 4:31; الخُرُوج 3:7"> فحَبِلت لَيئةُ وولدتِ اَبنًا وسمَّتْهُ رَأوبينَ، لأنَّها قالت: ”رأى الرّبُّ عَنائي، والآنَ يُحبُّني زوجي“</span>
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 34:25; التَّكوين 49:5–6; التَّكوين 34:30"> وحبِلت أيضًا وولدتِ اَبنًا فقالت: ”سَمِعَ الربُّ أنِّي مكروهةٌ، فرزقَني هذا أيضًا“، وسَمَّتْهُ شِمعونَ.</span>
<span className="v" id="v34" data-v="34" data-xref="الخُرُوج 32:26–29; التَّكوين 49:5–7; التَّكوين 46:11"> وحَبِلت أيضًا وولدتِ اَبنًا فقالت: ”الآنَ يلوي عليَّ قلبُ زوجي، لأنِّي وَلَدتُ ثلاثةَ بَنينَ“. وسمَّتْهُ لاوي.</span>
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 1:2; أخبارِ الأيّامِ الأوّلُ 5:2; التَّكوين 49:8–12"> وحَبِلت أيضًا وولدتِ اَبنًا وقالت: ”الآنَ أُقِرُّ بفضلِ الرّبِّ“. فسَمَّتْهُ يَهوذا، وتوقَّفَت عنِ الوِلادةِ.</span>
</div>
