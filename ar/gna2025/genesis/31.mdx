## يعقوب يلجأ إلى الهرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الجامِعَة 4:4; المَزامِير 49:16–17; أيُّوب 31:31"> وسَمِعَ يعقوبُ أنَّ بني لابانَ يقولونَ: ”أخذَ يعقوبُ كُلَّ ما كانَ لأبينا، ومِمَّا لأبينا جمَعَ كُلَّ هذِهِ الثَّروةِ“.</span>
<span className="v" id="v2" data-v="2" data-xref="دانيال 3:19; صَمُوئيلَ الأوَّلُ 18:9–11; التَّكوين 4:5"> ونظَرَ يعقوبُ إلى لابانَ، فرآهُ تغيَّرَ نحوَهُ عمَّا كانَ علَيه مِنْ قَبلُ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 28:15; التَّكوين 32:9; العِبرانِيّين 13:5"> فقالَ الرّبُّ لِيعقوبَ: ”إرجِعْ إلى أرضِ آبائِكَ وعَشيرتِكَ وأنا أكونُ معَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4"> فأرسلَ يعقوبُ ودعا راحيلَ ولَيئةَ إلى البرِّيَّةِ حيثُ كانَت غنَمُهُ،</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 31:2–3; التَّكوين 31:53; التَّكوين 48:15"> وقالَ لهُما: ”أرى أباكُما تغَيَّرَ نحوي عمَّا كان علَيهِ مِنْ قَبلُ، ولكن إلهُ أبي كانَ معي.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 30:29; أفَسُس 6:5–8; تِيطُس 2:9–10"> وأنتما تعرِفانِ أنِّي خدمْتُ أباكُما بِكُلِّ قُدرَتي،</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 37:28; التَّكوين 31:29; العَدَد 14:22"> وأبوكُما غدَرَ بي وغيَّرَ معي في أُجرَتي عشْرَ مرَّاتٍ. ولَكنَّ اللهَ لم يدَعْهُ يُسيءُ إليَّ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 30:32"> فكُلَّما قالَ: ”الرُّقْطُ تكونُ أجرَتَكَ“ ولَدَت جميعُ الغنَمِ رُقْطًا، أو قالَ: ”المُخَطَّطَةُ تكونُ أجرَتَكَ“ وَلَدت جَميعُ الغنَمِ مُخَطَّطَةً.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 31:1; التَّكوين 31:16; الأمثال 13:22"> فأخَذَ اللهُ مَواشيَ أبيكُما وأعطانيها.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 13:1; التَّكوين 30:39; التَّكوين 20:6"> وحَدَث هذا وقتَ وِحامِ الغنَمَ، حينَ رَفَعْتُ عينيَّ ونَظَرْتُ في المَنامِ فرأيتُ التُّيوسَ الّتي تَشُبُّ على الغنَمِ مُخَطَّطَةً ورَقْطاءَ ونَمْراء.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 58:9; التَّكوين 31:13; صَمُوئيلَ الأوَّلُ 3:6"> فقالَ ليَ ملاكُ اللهِ في الحُلُمِ: ”يا يعقوبُ! قلتُ نعم، ها أنا.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 3:7; التَّثْنِيَة 24:15; التَّكوين 31:42"> قالَ: إرفعْ عينيكَ واَنظُرْ. جميعُ التُّيوسِ الّتي تشُبُّ على الغنَمِ مُخَطَّطَةٌ ورَقْطَاءُ ونَمْراءُ، لأنِّي رأيتُ كُلَّ ما يفعَلُهُ لابانُ بِكَ.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 31:3; التَّكوين 32:9; التَّكوين 28:12–22"> أنا إلهُ بَيتِ إيلَ حيثُ نصَبْتَ عَمودًا ومَسَحْتَه بالزَّيتِ لِتُكرِّسَهُ لي، ونَذَرْتَ لي نَذْرًا. والآنَ قُمِ اَخرُجْ مِنْ هذِهِ الأرضِ واَرجِعْ إلى أرضِ مَولِدِكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="راعُوث 4:11; التَّكوين 2:24; التَّكوين 29:24"> فأجابَتْه راحيلُ ولَيئةُ: ”هل بقيَ لنا نصيبٌ ومِيراثٌ في بَيتِ أبينا؟</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 29:15–20; التَّكوين 31:41; التَّكوين 29:27–30"> حُسِبْنا عِندَه غريبَتَينِ، فباعَنا وأكلَ ثمَنَنا؟</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 30:35–1; التَّكوين 31:9; المَزامِير 45:10"> كُلُّ الثَّروةِ الّتي أخَذَها اللهُ مِنْ أبينا وَأعطاكَ إيَّاها هيَ لنا ولِبَنينا، فاَعمَلْ بكلِّ ما قالَه اللهُ لكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="صَمُوئيلَ الأوَّلُ 30:17; التَّكوين 24:61; التَّكوين 24:10"> فقَامَ يعقوبُ وحَمَلَ بَنيهِ وزَوجاتِه على الجِمالِ،</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 27:1–2; التَّكوين 27:41; التَّكوين 35:27–29"> وساقَ كُلَّ ماشيتهِ وكُلَّ ما اَمتلَكَهُ واَقتناهُ في سَهلِ أرامَ. وقصَدَ إلى إسحَقَ أبيهِ في أرضِ كنعانَ.</span>
<span className="v" id="v19" data-v="19" data-xref="صَمُوئيلَ الأوَّلُ 19:13; التَّكوين 31:30; هُوشَع 3:4"> وكانَ لابانُ غائبًا يَجُزُّ غنَمَهُ، فسَرَقت راحيلُ أصنامَ أبيها.</span>
<span className="v" id="v20" data-v="20"> وخدَعَ يعقوبُ لابانَ الأراميَّ ولم يُخبرْهُ بِفرارِه.</span>
<span className="v" id="v21" data-v="21" data-xref="المُلُوكِ الثّانِي 12:17; التَّكوين 37:25; التَّكوين 46:28"> وهرَبَ بِجميعِ ما كانَ لَه، فعَبَرَ نهرَ الفُراتِ وتوجَّهَ إلى جبَلِ جِلعادَ.</span>
</div>
</div>

## لابان يلحق يعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="أيُّوب 5:12–13; التَّكوين 30:36; الخُرُوج 14:5–31"> وتلقَّى لابانُ بَعدَ ثلاثةِ أيّامٍ خَبرَ فَرارِ يعقوبَ،</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 13:8; التَّكوين 24:27; الخُرُوج 2:11"> فأخذَ رِجالَه معَهُ وسَعى وراءَهُ مَسيرةَ سَبْعةِ أيّامٍ حتّى لَحِقَ بِه في جبَلِ جِلعادَ،</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 24:50; التَّكوين 31:29; التَّكوين 20:3"> فجَاءَ اللهُ إلى لابانَ الأراميِّ في الحُلُمِ ليلًا وقالَ لَه: ”إيَّاكَ أنْ تُكَلِّمَ يعقوبَ بِخَيرٍ أو شَرٍّ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 12:8; التَّكوين 33:18; العِبرانِيّين 11:9"> وكانَ يعقوبُ نصَبَ خَيمَتَه في التِّلالِ حينَ لَحِقَ بهِ لابانُ، فخيَّمَ لابانُ معَ رِجالِه هُناكَ في جبَلِ جِلعادَ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="صَمُوئيلَ الأوَّلُ 30:2; التَّكوين 12:18; التَّكوين 20:9–10"> وقالَ لابانُ لِيعقوبَ: ”ماذا فعَلْتَ؟ أقلَقتَ بالي وسُقْتَ بنتَيَّ كما تُساقُ سَبايا الحربِ.</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 15:20; الأمثال 26:23–26; التَّكوين 31:3–5"> ولِماذا هرَبْتَ خِفيةً وأقلَقْتَني ولم تُخبِرْني، فأُشَيِّعَكَ بِفَرحٍ وغِناءٍ ودُفٍّ وكَنَّارةٍ؟</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 31:55; راعُوث 1:14; اعمال الرسل 20:37"> ولم يَدَعْني أُقبِّلُ حفَدَتي وبَناتي، فأنتَ بِغَباوةٍ فعَلْتَ.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 31:53; التَّكوين 31:42; التَّكوين 31:24"> والآنَ أنا قادِرٌ أَنْ أُعامِلَكُم بسُوءٍ لولا أنَّ إلهَ أبيكُم كلَّمَني البارحةَ فقالَ لي: إيَّاكَ أنْ تُكلِّمَ يعقوبَ بِخَيرٍ أو شَرٍّ.</span>
<span className="v" id="v30" data-v="30" data-xref="إرْمِيا 10:11; التَّكوين 31:19; القُضاة 18:24"> وأنتَ إنَّما اَنصَرَفْتَ مِنْ عِندي لأنَّكَ اَشتَقْتَ إلى بَيتِ أبيكَ، ولكنْ لماذا سَرَقْتَ آلِهَتي؟“</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الأمثال 29:25; التَّكوين 31:26–27; التَّكوين 20:11"> فأجابَه يعقوبُ: ”خِفتُ أنْ تغتَصِبَ بِنتَيكَ منّي.</span>
<span className="v" id="v32" data-v="32" data-xref="صَمُوئيلَ الأوَّلُ 12:3–5; التَّكوين 31:30; التَّكوين 31:19"> وأمَّا آلِهتُكَ، فإذا وجَدْتَها معَ أحدٍ مِنّا فلا يستَحِقُّ الحياةَ. أَثبِتْ ما هوَ لكَ معي أمامَ رِجالِنا وخُذْهُ“. وكانَ يعقوبُ لا يعرِفُ أنَّ راحيلَ سَرَقَت آلِهةَ لابانَ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 24:67; التَّكوين 24:28"> فدخَلَ لابانُ خَيمةَ يعقوبَ وخَيمةَ لَيئةَ وخَيمةَ الجاريتَينِ، فَما وجَدَ شيئًا. وخرَجَ مِنْ خَيمةِ لَيئةَ ودَخَلَ خَيمَةَ راحيلَ.</span>
<span className="v" id="v34" data-v="34" data-xref="التَّكوين 31:19; التَّكوين 31:17"> وكانَت راحيلُ أخَذَتِ الأصنامَ ووَضَعَتْها في رَحْلِ الجمَلِ وجَلست فوقَها. ففَتَّشَ لابانُ الخَيمةَ كُلَّها، فما وجَدَ شيئًا،</span>
<span className="v" id="v35" data-v="35" data-xref="اللّاويِّين 19:32; الخُرُوج 20:12; رسالَةُ بُطرُسَ الأُولَى 2:18"> وقالت راحيلُ لأبيها: ”لا يَغيظُكَ يا سيِّدي أنِّي لا أقدِرُ أنْ أقومَ أمَامَك لأنَّ عليَّ عادَةَ النِّساءِ“. فلم يَجِدْ لابانُ أصنامَهُ الّتي فتَّشَ عَنها.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="أفَسُس 4:26; التَّكوين 49:7; العَدَد 16:15"> فاَحتَدَّ يعقوبُ وخاصَمَ لابانَ وقالَ لَه: ”ما جَريمَتي وما خطيئَتي حتّى خرَجتَ مُسرِعًا ورائي؟</span>
<span className="v" id="v37" data-v="37" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:4–5; صَمُوئيلَ الأوَّلُ 12:3–4"> فَتَّشْتَ جميعَ أشيائي، فماذا وَجَدْتَ مِنْ جميعِ أشياءِ بَيتِكَ؟ إنْ وَجَدْتَ شيئًا، فضَعْهُ هُنا أمامَ رِجالي ورجالِكَ، لِيَحكُموا بَيني وبَينَكَ.</span>
<span className="v" id="v38" data-v="38" data-xref="التَّكوين 30:27; التَّثْنِيَة 28:4; حِزْقِيال 34:2–4"> لي عشرونَ سنَةً معَكَ، فلا نِعاجُكَ أسقطت مَواليدَها ولا عِنازُكَ، ولا أنا مِنْ كِباشِ غنَمِكَ أكلْتُ.</span>
<span className="v" id="v39" data-v="39" data-xref="الخُرُوج 22:31; الخُرُوج 22:10–13; اللّاويِّين 22:8"> ما كانَت تَفتَرِسُهُ الوحوشُ لم أُحضِرْ إليكَ بُرهانًا عَنهُ، وإنَّما كُنتُ أتحَمَّلُ خَسَارَتَه وحدي، وكُنتَ أنتَ تُطالبُني بهِ، سَواءٌ سُرِقَ في النَّهارِ أو سُرِقَ في اللَّيلِ.</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 2:8; الخُرُوج 2:19–22; رسالَةُ بُطرُسَ الأُولَى 5:2–4"> وكانَ يأكُلُني الحَرُّ في النَّهارِ، والصَّقيعُ في اللَّيلِ، ولَطَالما هرَبَ النَّومُ مِنْ عينَيَّ.</span>
<span className="v" id="v41" data-v="41" data-xref="التَّكوين 31:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:26"> لي عِشرونَ سنَةً في بَيتِكَ، خَدَمْتُكَ فيها أربَعَ عشْرَةَ سنَةً بَدلَ بِنتَيكَ وسِتَّ سنواتٍ بَدلَ غَنَمِكَ، وغَيَّرتَ معي في أُجرَتي عشْرَ مرَّاتٍ.</span>
<span className="v" id="v42" data-v="42" data-xref="المَزامِير 31:7; التَّكوين 31:53; إشَعْياء 8:13"> ولو لم يكُنْ إلهُ أبي، إلهُ إبراهيمَ ومَهابةُ إسحَقَ معي، لكُنتَ الآنَ صَرَفْتَني عَنكَ فارغَ اليدَينِ. ولكنَّ اللهَ نظَرَ إلى عَنائي وتَعَبِ يَديَّ، فَوَبَّخَكَ البارِحَةَ“.</span>
</div>
</div>

## العهد بين يعقوب ولابان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v43" data-v="43"> فأجابَ لابانُ يعقوبَ: ”البَناتُ بَناتي، وبَنوهُنَّ بَنيَّ، والغنَمُ غنَمي، وكُلُّ ما تراهُ هوَ لي، فماذا أقدِرُ الآنَ أنْ أفعلَ لأَستعِيدَ بَناتي والبَنينَ الّذينَ وَلَدْتَهم؟</span>
<span className="v" id="v44" data-v="44" data-xref="التَّكوين 31:52; يَشُوع 24:25–27; يَشُوع 22:27"> فتَعالَ نقطَعُ عَهدًا، أنا وأنتَ، ونُقيمُ شاهدًا بَيني وبينَكَ“.</span>
<span className="v" id="v45" data-v="45" data-xref="التَّكوين 28:18–22"> فأخذَ يعقوبُ حجَرًا ونصَبَه عَمودًا</span>
<span className="v" id="v46" data-v="46" data-xref="الجامِعَة 3:5; التَّكوين 31:23; التَّكوين 31:54"> وقالَ لِرجالهِ: ”إجْمَعُوا حِجارةً“. فجَمَعوا حِجارةً وكَوَّموها وأكلوا طَعامًا فوقَ الكُومةِ.</span>
<span className="v" id="v47" data-v="47" data-xref="العِبرانِيّين 12:1"> وسَمَّاها لابانُ يَجَرْسَهْدُوثا، وسَمَّاها يعقوبُ جَلعيدَ.</span>
<span className="v" id="v48" data-v="48" data-xref="يَشُوع 24:27; يَشُوع 13:8–9; التَّكوين 31:23"> وقالَ لابانُ: ”هذِهِ الكُومَةُ تكونُ شاهِدًا بَيني وبَينَكَ اليومَ“. ولِذلِكَ سُمِّيت جَلعِيدَ.</span>
<span className="v" id="v49" data-v="49" data-xref="القُضاة 11:29; القُضاة 10:17; صَمُوئيلَ الأوَّلُ 7:5–6"> وسُمِّيت أَيضًا المِصفَاةَ، لأنَّ لابانَ قالَ: ”يُصافي الرّبُّ بَيني وبَينَكَ حينَ يتوارَى واحدُنا عَنِ الآخرِ.</span>
<span className="v" id="v50" data-v="50" data-xref="إرْمِيا 42:5; إرْمِيا 29:23; ميخا 1:2"> إنْ أَذللتَ بِنتيَّ أو تَزَّوجتَ نِساءً علَيهما، فلا أحدَ مِنّا معَكَ لِيرى، ولكنَّ اللهَ شاهِدٌ بَيني وبَينَكَ“.</span>
<span className="v" id="v51" data-v="51"> وقالَ لابانُ لِيعقوبَ: ”هذِهِ هيَ الكُومَةُ، وها هوَ العَمودُ الّذي وضَعْتُ بَيني وبَينَكَ.</span>
