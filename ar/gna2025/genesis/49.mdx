## بركة يعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العَدَد 24:14; إرْمِيا 23:20; دانيال 10:14"> ثُمَّ دَعا يعقوبُ بَنيه وقالَ: ”إجتَمِعوا لأنبِئَكُم بِما يَحدُثُ لَكُم في الأيّامِ الآتيةِ:</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 34:11; الأمثال 7:1; الأمثال 23:22"> اَجتَمِعوا واَسمَعوا يا بَني يعقوبَ،<br />واَصغوا إلى إِسرائيلَ أبيكُم.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 21:17; المَزامِير 78:51; المَزامِير 105:36"> رَأوبينُ أنتَ بِكْري،<br />قُوَّتي وباكورةُ رُجُولَتي.<br />فاضلٌ في الرِّفعَةِ<br />فاضِلٌ في العِزِّ،</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 35:22; أخبارِ الأيّامِ الأوّلُ 5:1; التَّثْنِيَة 27:20"> هائجٌ كالسَّيلِ لا تُفضَّلُ<br />لأنَّك عَلَوتَ فِراشَ أبيكَ.<br />فَحَرَّمتَ جاريتي عليَّ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 34:25–31; التَّكوين 46:10–11; التَّكوين 29:33–34"> شِمعُونُ ولاوي أخوَانِ،<br />سُيوفُهُما سِلاحُ العُنفِ،</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 34:30; المَزامِير 26:9; المَزامِير 16:9"> مَجلِسُهُما لا أدخُلُه،<br />وفي صُحبَتِهِما لا أبتَهِجُ.<br />ففي غضَبِهما قتَلا بشَرًا،<br />وفي نقمَتِهما عرقَبا ثَورًا.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 19:1–9; أخبارِ الأيّامِ الأوّلُ 4:24–31; أخبارِ الأيّامِ الأوّلُ 6:65"> مَلعونٌ غضَبُهما فهوَ شديدٌ،<br />وغَيظُهما لأنَّه قاسٍ.<br />سأُفَرِّقُهُما بَينَ بَني يعقوبَ،<br />وأُشَتِّتُهُما في أرضِ إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="العِبرانِيّين 7:14; التَّكوين 27:29; أخبارِ الأيّامِ الأوّلُ 5:2"> يَهوذا يَحمَدُك إخوَتُكَ،<br />يَدُكَ على رِقابِ أعدائِكَ.<br />يسجُدُ لكَ بَنو أبيكَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="العَدَد 24:9; العَدَد 23:24; رُؤيا يُوحَنّا 5:5"> يَهوذا شِبْلُ أسَدٍ.<br />مِنَ الأطرافِ صَعِدْتَ يا اَبني،<br />كأسدٍ يَركعُ ويَربِضُ<br />وكَلَبوةٍ، فمَنْ يُقيمُه؟</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="لُوقا 1:32–33; العَدَد 24:17; إشَعْياء 9:6"> لا يزولُ الصَّولجانُ مِنْ يَهوذا<br />ولا عصا السُّلطانِ مِنْ صُلْبِهِ،<br />إلى أنْ يتَبوَّأَ في شيلُوه<br />مَنْ لَه طاعةُ الشُّعوبِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 63:1–3; يُوئيل 3:18; رُؤيا يُوحَنّا 7:14"> يَربُطُ بالكَرمَةِ جَحْشَه،<br />وبالدَّاليَةِ اَبنَ أتانِهِ.<br />يَغسِلُ بالخمرِ ثِيابَهُ،<br />وبِدَمِ العِنَبِ رِدَاءَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الأمثال 23:29"> تَحمرُّ مِنَ الخمرِ عيناهُ،<br />ومِنَ اللَّبَنِ تَبيَضُّ أسنانُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="التَّثْنِيَة 33:18–19; التَّكوين 30:20; يَشُوع 19:10–16"> زَبولونُ يسكُنُ ساحِلَ البحرِ<br />ويكونُ مَرفأً للسُّفُنِ،<br />وتُخُومُه عِندَ صيدونَ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="أخبارِ الأيّامِ الأوّلُ 12:32; التَّكوين 30:18; التَّثْنِيَة 33:18"> يسَّاكَرُ حِمارٌ ضَخمٌ<br />رابِضٌ بَينَ الحَظائِرِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="المَزامِير 81:6; ﻣﺘﻰ 23:4; يَشُوع 14:15"> رأى المَرَاحَ أنَّهُ هَنيءٌ<br />والأرضَ أنها نعيمٌ،<br />فأحنى كَتِفَه لِلحَمْلِ<br />وصارَ للسُّخرةِ عبدًا.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 30:6; التَّثْنِيَة 33:22; القُضاة 18:26–27"> دانُ يَدينُ شعبَهُ<br />كأحَدِ أسباطِ إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الأوّلُ 12:35; القُضاة 18:22–31; القُضاة 14:1–15"> يكونُ ثُعبانًا على الطَّريقِ،<br />أُفعوانًا على السَّبيلِ،<br />يَلسَعُ الفرَسَ في عَقِبِها<br />فيَقَعُ راكِبُها إلى الوراءِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 119:174; المَزامِير 119:166; ميخا 7:7"> أنتَظِرُ خلاصَكَ يا ربُّ!</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 30:11; أخبارِ الأيّامِ الأوّلُ 5:11–22; التَّثْنِيَة 33:20–21"> جادُ يَغزوهُ الغُزاةُ،<br />فيَغزُو لاحِقًا بهِم.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 33:24–25; التَّكوين 30:13; التَّكوين 46:17"> أشيرُ طَعامُه دَسِمٌ<br />يُثيرُ شَهيَّةَ المُلوكِ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التَّثْنِيَة 33:23; التَّكوين 30:8; يَشُوع 19:32–39"> نفتالي غزالةٌ شَريدةٌ<br />تَلِدُ ظِباءً جميلةً.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 41:52; المَزامِير 128:3; المَزامِير 1:1–3"> يوسُفُ غُصنٌ مُثمِرٌ،<br />غُصنٌ مُثمِرٌ على عينِ ماءٍ،<br />فُروعُه صَعِدَت على سُورٍ،</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 37:24; المَزامِير 64:3; المَزامِير 118:13"> هاجَمَه أصحابُ السِّهامِ،<br />وخاصَموه وضايَقوه كثيرًا.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="المَزامِير 132:2; إشَعْياء 28:16; المَزامِير 132:5"> ولكنَّ قَوسَه بَقيت ثابِتَةً،<br />وتَشَدَّدَت سَواعِدُ يَدَيهِ.<br />بِقُدْرَةِ الجبَّارِ إلهِ يعقوبَ،<br />بِاَسمِ الرَّاعي صَخرةِ إسرائيلَ،</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 28:13; التَّكوين 35:11; فِيلِبِّي 4:19"> بإلهِ أبيكَ الّذي يَنصُرُكَ،<br />بالقديرِ الّذي يُبارِكُكَ.<br />بَرَكاتُ السَّماءِ مِنْ فوقُ<br />وبَرَكاتُ الغَمْرِ الرَّاكِدِ في الأسفَلِ.<br />بَرَكاتُ الثَّدْيَينِ والرَّحِمِ،</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 33:15–16; حَبَقُّوق 3:6; إشَعْياء 66:5"> بَرَكاتُ أبيكَ الّتي تفوقُ<br />بركاتِ الجِبالِ الأزليَّةِ<br />وخَيراتُ الرَّوابي الأبديَّةِ<br />لِتَكُنْ على رأسِ يوسُفَ<br />على هامَةِ خيرَةِ إخوتِهِ.</span>

<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="القُضاة 20:25; التَّكوين 35:18; حِزْقِيال 22:27"> بنيامينُ ذئبٌ مُفتَرِسٌ،<br />في الصَّباحِ يلتَهِمُ فرِيسَتَهُ،<br />وعِندَ الغُروبِ يُقَسِّمُ غَنيمَتهُ“.</span>
</div>

## وصيَّة يعقوب وموته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="رُؤيا يُوحَنّا 7:4; أسْتِير 8:11; المُلُوكِ الأوَّلُ 18:31"> هؤلاءِ كُلُّهُم أسباطُ إِسرائيلَ الإثنا عشَرَ، وهذا ما قالَه لهُم أبوهُم حينَ باركَهُم كُلُّ واحدٍ وَفقَ ما يُناسِبُهُ مِنَ البَرَكَةِ.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 47:30; التَّكوين 50:13; التَّكوين 15:15"> وأوصاهُم قالَ: ”أموتُ الآنَ وأنضَمُّ إلى آبائي. اَدفِنوني معَ آبائي في المَغارةِ الّتي في حقلِ عِفرُونَ الحثِّيِّ.</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 23:20; التَّكوين 23:16–18; التَّكوين 23:8–9"> المَغارةُ الّتي في حقلِ المَكفيلَةِ، تُجاهَ مَمْرا في أرضِ كنعانَ، تِلكَ الّتي اَشْتراها إبراهيمُ معَ الحقلِ منْ عَفرونَ الحثِّيِّ لِتكونَ مُلْكًا لِقَبرٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 25:9; التَّكوين 35:29; التَّكوين 50:13"> هُناكَ دُفِنَ إبراهيمُ وسارةُ اَمرأتُه، وهُناكَ دُفِنَ إسحَقُ ورِفقَةُ اَمرأتُهُ، وهُناكَ دُفِنَت لَيئةُ.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 23:17–20"> شِراءُ الحقلِ والمَغارةِ الّتي فيهِ كانَ مِنْ بَني حِثٍّ“.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 49:29; التَّكوين 25:8; التَّكوين 35:29"> فلمَّا فرَغَ يعقوبُ مِنْ وَصيَّتِهِ لِبَنيهِ ضَمَّ رجليهِ على السَّريرِ وأسلَمَ الرُّوحَ واَنضمَّ إلى آبائهِ.</span>
</div>
