## يعقوب في بيت إيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 31:13; هُوشَع 12:4; المَزامِير 91:15"> وقالَ اللهُ لِيعقوبَ: ”قُمِ اَصعَدْ إلى بيتِ إيلَ، وأَقِمْ هُناكَ، واَبنِ مذبَحًا للهِ الّذي تراءَى لكَ حينَ هَرَبْتَ مِنْ وجهِ عيسو أخيكَ“.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 19:14; الخُرُوج 19:10; حِزْقِيال 18:31"> فقالَ يعقوبُ لأهلِ بَيتِه ولِكُلِّ مَنْ كانَ معَهُ: ”أَزيلوا الآلِهةَ الغريبَةَ الّتي بَينَكُم وتطهَّرُوا وبدِّلوا ثِيابَكُم،</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 43:2; التَّكوين 32:7; التَّكوين 31:3"> وقُوموا نصعَدْ إلى بَيتِ إيلَ، فأَبنيَ هُناكَ مذبَحًا للهِ الّذي أعانَني في يومِ ضيقي، وكانَ معي في الطَّريقِ الّتي سَلَكْتُها“.</span>
<span className="v" id="v4" data-v="4" data-xref="القُضاة 9:6; التَّثْنِيَة 7:25; الخُرُوج 32:20"> فأعطَوا يعقوبَ كُلَّ الآلهةِ الغريبةِ الّتي في أَيديهِم والحَلَقَ الّذي في آذانِهِم، فطَمَرَها يعقوبُ تحتَ البُطْمةِ التي عِندَ شكيمَ.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 23:27; أخبارِ الأيّامِ الثّانِي 17:10; يَشُوع 5:1"> ثُمَّ رحَلوا، فحَلَّ ذُعْرٌ مِنَ اللهِ على المُدُنِ الّتي حَولَهُم، فلم يَجرُؤْ أَهلُها على السَّعي ورَاءَ بَني يعقوبَ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 28:19; التَّكوين 28:22; التَّكوين 12:8"> وجاءَ يعقوبُ وجميعُ القومِ الّذين معَه إلى لُوزَ الّتي في أرضِ كنعانَ وهيَ بَيتُ إيلَ.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 35:3; التَّكوين 35:1; التَّكوين 28:13"> وبَنى هُناكَ مذبَحًا وسَمَّى المَوضِعَ إلهَ بَيتِ إيلَ، لأنَّ اللهَ تجَلَّى لَه هُناكَ حِينَ هرَبَ مِنْ وجهِ أخيهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 24:59; صَمُوئيلَ الأوَّلُ 31:13; القُضاة 2:5"> وماتَت دبُورةُ مُرضِعَةُ رِفقةَ فدُفِنَت تحتَ البَلُّوطةِ في أَسفلِ بَيتِ إيلَ، وسُمِّيَ المَوضِعُ بلُّوطَةَ البُكاءِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 35:1; التَّكوين 18:1; التَّكوين 26:2"> وتراءَى اللهُ لِيعقوبَ أَيضًا حينَ جاءَ مِنْ سَهلِ أرامَ وبارَكَهُ</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 17:5; التَّكوين 32:27–28; التَّكوين 17:15"> وقالَ لَه: ”إسمُكَ يعقوبُ. لا يُدعَى اَسمُكَ بَعدَ الآنَ يعقوبَ، بل إِسرائيلَ“. فسمَّاهُ إِسرائيلَ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 17:1; التَّكوين 9:1; التَّكوين 22:17"> وقالَ لَه اللهُ: ”أنا اللهُ القديرُ. اَنْمُ واَكْثُرْ. أُمَّةٌ ومجموعةُ أُمَمٍ تكونُ مِنكَ، وملوكٌ مِنْ صُلبِكَ يَخرُجونَ،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 12:7; التَّكوين 28:13; التَّكوين 26:3–4"> والأرضُ الّتي وهَبْتُها لإِبراهيمَ وإسحَقَ أهبُها لكَ ولِنَسلِكَ مِنْ بَعدِكَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 17:22; التَّكوين 18:33; القُضاة 6:21"> ثُمَّ اَرتفَعَ اللهُ عَنهُ في المَوضِعِ الّذي كلَّمَه فيهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 28:18–19; صَمُوئيلَ الأوَّلُ 7:12; الخُرُوج 17:15"> فنَصَبَ يعقوبُ هُناكَ عَمودًا مِنْ حجَرٍ، وسكبَ علَيه خمرًا وصبَّ زَيتًا لِيُكَرِّسَهُ لِلرّبِّ.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 28:19"> وسَمَّى ذلِكَ المَوضِعَ الّذي كَلَّمَه اللهُ فيهِ بَيتَ إِيلَ.</span>
</div>
</div>

## مولد بنيامين وموت راحيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ميخا 5:2; التَّكوين 48:7; المُلُوكِ الثّانِي 5:19"> ثُمَّ رحَلوا مِنْ بَيتِ إيلَ. وبَينما هُم بَعدُ على مَقرُبَةٍ مِنْ أفراتَةَ بدَأَت راحِيلُ تَلِدُ فتعَسَّرَت علَيها الوِلادةُ.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 30:24; صَمُوئيلَ الأوَّلُ 4:19–21"> ولمَّا تعَسَّرَت علَيها الوِلادةُ قالت لها القابِلَةُ: ”لا تخافي. هذا اَبنٌ آخَرُ لكِ“.</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 12:20; اعمال الرسل 7:59; لُوقا 23:46"> فسَمَّتْهُ راحِيلُ، قَبلَ أنْ تفيضَ رُوحُها، بَنْ أوني. وأمَّا أبوه فَسَمَّاهُ بِنيامينُ.</span>
<span className="v" id="v19" data-v="19" data-xref="ميخا 5:2; التَّكوين 48:7; راعُوث 4:11"> وماتَت راحيلُ ودُفِنت في طريقِ أفراتَةَ، وهيَ بَيتُ لَحمٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="صَمُوئيلَ الأوَّلُ 10:2; التَّكوين 35:14; صموئيلَ الثّانِي 18:17–18"> ونصبَ يعقوبُ عَمودًا على قبرِها، وهوَ عَمودُ قبرِ راحيلَ إلى اليومِ.</span>
<span className="v" id="v21" data-v="21" data-xref="ميخا 4:8; لُوقا 2:8"> ثُمَّ رحَلَ يعقوبُ مِنْ هُناكَ ونصبَ خَيمتَه على الجانبِ الآخَرِ مِنْ مَجْدَلِ عِدْرٍ.</span>
<span className="v" id="v22" data-v="22" data-xref="أخبارِ الأيّامِ الأوّلُ 5:1; اللّاويِّين 18:8; صموئيلَ الثّانِي 20:3"> وبَينَما هوَ ساكِنٌ في تِلكَ الأرضِ ذهَبَ رَأوبينُ فضاجَعَ بِلْهَةَ، مَحظِيَّةَ أبيهِ، فسَمِعَ بِذلِكَ يعقوبُ.<br />وكانَ بَنو يعقوبَ اَثنَي عشَرَ: ”</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 30:18–20; التَّكوين 46:8–15; التَّكوين 29:32–35"> بَنو لَيئةَ: رَأوبينُ، بِكْرُ يعقوبَ، وشِمعونُ ولاوي ويَهوذا ويسَّاكَرُ وزَبُولُونُ،</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 30:22–24; التَّكوين 46:19–22; التَّكوين 35:16–18"> واَبنا راحيلَ: يوسُفُ وبنيامينُ،</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 46:23–25; التَّكوين 37:2; التَّكوين 30:4–8"> واَبنا بِلْهَةَ، جاريةِ راحيلَ، دانُ ونفتالي،</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 30:9–13; التَّكوين 46:16–18; التَّكوين 25:20"> واَبنا زِلْفَةَ، جاريةِ لَيئَةَ، جادُ وأَشيرُ، هؤلاءِ بَنو يعقوبَ الذينَ وُلِدوا لَه في سَهلِ أَرامَ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 13:18; التَّكوين 23:19; التَّكوين 18:1"> وجاءَ يعقوبُ إلى إسحَقَ أبيه عِندَ مَمْرا بالقُربِ مِنْ قريةٍ أربَعَ، وهيَ حَبرونُ، حيثُ تغَرَّبَ إبراهيمُ وإِسحَقُ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 25:7; التَّكوين 50:26; التَّكوين 47:28"> وكانَ عُمرُ إِسحَقَ مئةً وثمانينَ سنَةً.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 15:15; التَّكوين 49:33; التَّكوين 49:31"> وفاضَت روحُ إِسحَقَ وماتَ واَنْضَمَّ إلى آبائِهِ شيخًا شَبِعَ مِنَ الحياةِ. ودَفَنَهُ عِيسو ويعقوبُ اَبناهُ.</span>
</div>
