## إسحق في جرار

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 12:10; التَّكوين 25:11; التَّكوين 21:22–32"> وكانَ في الأرضِ جُوعٌ غيرُ الجُوعِ الأوَّلِ الّذي كانَ في أيّامِ إبراهيمَ، فذهبَ إسحَقُ إلى أبيمالِكَ، مَلِكِ الفلِسطيِّينَ في جَرارَ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 12:7; المَزامِير 37:3; التَّكوين 17:1"> فتراءى لَه الرّبُّ وقالَ: ”لا تَنْزِلْ إلى مِصْرَ، بلِ اَسكُنْ في الأرضِ الّتي أدُلُّكَ علَيها.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 28:15; المَزامِير 105:9; التَّكوين 17:8"> تغَرَّبْ بهذهِ الأرضِ وأنا أكونُ معَكَ وأُبارِكُكَ، فأُعطيَ لكَ ولِنسلِكَ جميعَ هذِهِ البِلادِ، وأفي باليمينِ الّتي حلَفتُها لإبراهيمَ أبيكَ،</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 3:25; التَّكوين 15:5; التَّكوين 22:17–18"> وأُكثِّرُ نسلَكَ كنجومِ السَّماءِ وأُعطيهم جميعَ هذِهِ البلادِ، ويتباركُ فيهم جميعُ أُمَمِ الأرضِ،</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 7:24; التَّكوين 22:18; التَّكوين 22:16"> لأنَّ إبراهيمَ سَمِعَ كلامي، وحفِظَ أوامري ووصايايَ وفرائِضي وشرائعي“.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 20:1"> فأقامَ إسحَقُ في جَرارَ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الأمثال 29:25; التَّكوين 12:13; التَّكوين 20:2"> وسألَهُ أهلُ جَرارَ عَنِ اَمرأتِهِ فقالَ: ”هيَ أُختي“، لأنَّهُ خافَ أنْ يقولَ: ”هيَ اَمرأتي“، لِئلَّا يَقْتُلوهُ بسببِها وكانَت جميلةَ المَنظَرِ.</span>
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 62:5; الأمثال 5:18–19; الأمثال 7:6"> ولمَّا مضَى على إقامتِهِ هُناكَ وقتٌ طويلٌ حدَثَ أنَّ أبيمالِكَ، مَلِكَ الفلِسطيِّينَ أَطَلَّ مِنْ نافذةٍ لَه ونظرَ فرأى إسحَقَ يُداعِبُ رِفقةَ اَمرأتَهُ.</span>
<span className="v" id="v9" data-v="9"> فدَعاهُ وقالَ لَه: ”إذًا هيَ اَمرأتُكَ، فلماذا قلتَ إنَّها أُختُكَ؟“ فقالَ إسحَقُ: ”لأنِّي ظَننتُ أنَّني رُبَّما أهلِكُ بسببِها“.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 20:9–10; التَّكوين 12:18–19"> فقالَ أبيمالِكُ: ”ماذا فعَلتَ بِنا؟ لولا قليلٌ لضاجعَ أحدُ أبناءِ شعبِنا اَمرأتَكَ فنُذنبَ“.</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 105:15; التَّكوين 20:6; الأمثال 6:29"> وأوصى أبيمالِكُ جميعَ الشَّعبِ قالَ: ”مَنْ مسَّ هذا الرَّجلَ أو اَمرأتَهُ فموتًا يموتُ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="مَرْقُس 4:8; التَّكوين 24:1; التَّكوين 26:3"> وزرعَ إسحَقُ في تِلكَ الأرضِ، فحصدَ في تِلكَ السَّنةِ مئةَ ضِعفٍ، وباركَهُ الرّبُّ</span>
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 112:3; الأمثال 10:22; التَّكوين 24:35"> فعظُمَت مكانتُهُ وتزايدَت إلى أنْ صارَ رَجُلًا عظيمًا جدًّا.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 37:11; التَّكوين 13:2; الأمثال 27:4"> وكانَت لَه ماشيةُ غنَمٍ وماشيةُ بقَرٍ وعبيدٌ كثيرونَ، فحسَدَهُ الفلِسطيّونَ.</span>
</div>
</div>

## الآبار بين جرار وبئر سبع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 21:30; التَّكوين 21:25"> وردَمَ الفلِسطيّون جميعَ الآبارِ الّتي حفرَها عبيدُ أبيه إبراهيمَ في أيّامِهِ وملأُوها تُرابًا.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 1:9"> وقالَ أبيمالِكُ لإسحَقَ: ”أخرُجْ مِنْ عِندِنا لأنَّكَ صِرتَ أعظمَ مِنَّا كثيرًا“.</span>
<span className="v" id="v17" data-v="17"> فخرَجَ إسحَقُ مِنْ هُناكَ ونزَلَ في وادي جَرارَ وأقامَ فيهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="زَكَريّا 13:2; هُوشَع 2:17; العَدَد 32:38"> وعادَ إسحَقُ فحفرَ آبارَ الماءِ التي رَدمَها الفلِسطيّونَ بَعدَ موتِ إبراهيمَ أبيهِ، وكانَت حُفِرَت في أيّامِهِ. وسَمَّاها إسحَقُ بالأسماءِ الّتي كانَ سَمَّاها بها أبوهُ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 4:10–11; يُوحَنّا 7:38; نَشِيدِ الأنشاد 4:15"> وحفرَ عبيدُ إسحَقَ في الوادي فَوجدوا هُناكَ نبعَ ماءٍ،</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 21:25"> فتخاصمَ رُعاةُ جَرارَ معَ رعاةِ إسحَقَ قائِلينَ: ”هذا الماءُ لنا“. فسمَّى إسحَقُ البِئرَ عِسِقَ، لأنَّهم تنازعوا علَيها.</span>
<span className="v" id="v21" data-v="21" data-xref="عَزْرا 4:6"> وحَفروا بِئْرًا أُخرى فتخاصموا علَيها أيضًا، فسمَّاها سِطْنَةً</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 17:6; الخُرُوج 1:7; المَزامِير 118:5"> وحَفروا بِئرًا أخرى فما تخاصموا علَيها، فسمَّاها رَحوبوتَ، وقالَ: ”الآنَ رَحُبَ الرّبُّ لنا لنَكثُرَ في الأرضِ“.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="القُضاة 20:1; التَّكوين 21:31; التَّكوين 46:1"> وصَعِدَ إسحَقُ مِنْ هُناكَ إلى بئرَ سَبْعَ،</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 15:1; الخُرُوج 3:6; إشَعْياء 43:1–2"> فتراءى لَه الرّبُّ في تِلكَ اللَّيلةِ وقالَ لَه: ”أنا إلهُ إبراهيمَ أبيكَ. لا تَخفْ، فأنا مَعكَ وأبارِكُكَ وأُكثِّرُ نسلَكَ مِنْ أجلِ عبدي إبراهيمَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 13:18; المَزامِير 116:17; التَّكوين 12:7–8"> فبنى إسحَقُ هُناكَ مذبَحًا ودعا باَسمِ الرّبِّ. ونصَبَ هُناكَ خَيمَةً وحفرَ رِجالُه بِئرًا.</span>
</div>
</div>

## الحلف بين إسحق وأبيمالك

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 21:22–32; التَّكوين 20:3"> وجاءَ إليهِ أبيمالِكُ مِنْ جَرارَ ومَعهُ آحزّاتُ مُرافِقُه، وفيكولُ قائدُ جيشِهِ.</span>
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 26:16; القُضاة 11:7; اعمال الرسل 7:14"> فقالَ لهُم إسحَقُ: ”ما بالُكُم جِئتُم إليَّ وأنتُم أبغضتموني وصرفتموني مِنْ عِندِكُم؟“</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 21:22–23; إشَعْياء 45:14; إشَعْياء 61:9"> فقالوا: ”الآنَ رأينا أنَّ الرّبَّ معَكَ، فقُلنا: لِيكُنْ حِلفٌ بَينَنا وبَينَكَ، ولنقطَعْ معَكَ عَهدًا</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 24:31; التَّكوين 21:22; التَّكوين 22:17"> أنْ لا تُسيءَ إلينا، بل تُعاملُنا بالحُسنَى كما عامَلْناكَ وصرَفْناكَ بسلامٍ. أنتَ الآنَ مُبارَكٌ مِنَ الرّبِّ“.</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 19:3; ﺭﻭﻣﻴﺔ 12:18; العِبرانِيّين 12:14"> فأقامَ لهُم إسحَقُ وليمةً، فأكلوا وشبِعوا،</span>
<span className="v" id="v31" data-v="31" data-xref="صَمُوئيلَ الأوَّلُ 20:16–17; العِبرانِيّين 6:16; التَّكوين 25:33"> وبكّروا في الغدِ فتحالفوا. وصرفَهُم إسحَقُ، فمضَوا مِنْ عِندهِ بسلامٍ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="الأمثال 10:4; الأمثال 13:4; الأمثال 2:4–5"> وفي ذلِكَ اليومِ جاءَ عبيدُ إسحَقَ فأخبروهُ عنِ البئرِ الّتي حفروها، قالوا: ”وَجَدْنا ماءً“.</span>
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 21:31; التَّكوين 26:28"> فسمَّاها إسحَقُ شِبْعَةَ فصارَ اَسمُ المدينةِ بئرَ سَبْعَ إلى هذا اليومِ.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="التَّكوين 36:2; التَّكوين 36:13; الخُرُوج 34:16"> ولمَّا صارَ عيسو اَبنَ أربعينَ سنَةً تزوَّج يهوديتَ بنتَ بيري الحثِّيِّ وبَسمةَ بنتَ إيلونَ الحثِّيِّ،</span>
<span className="v" id="v35" data-v="35" data-xref="التَّكوين 27:46–2; التَّكوين 28:8; التَّكوين 6:2"> فكانتا لإسحَقَ ورِفقةَ خَيبةً مُرَّةً.</span>
</div>
</div>
