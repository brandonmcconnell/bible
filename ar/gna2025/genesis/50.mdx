## مأتم يعقوب ودفنه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 46:4; التَّثْنِيَة 6:7–8; اعمال الرسل 8:2"> فاَرتَمى يوسُفُ على وجهِ أبيهِ وبَكى علَيهِ وقبَّله.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 50:26; أخبارِ الأيّامِ الثّانِي 16:14; يُوحَنّا 19:39–40"> وأوصَى أطبَّاءَهُ أنْ يُحَنِّطوا أباهُ، فحنَّطَ الأطِبَّاءُ يعقوبَ.</span>
<span className="v" id="v3" data-v="3" data-xref="العَدَد 20:29; التَّثْنِيَة 34:8; التَّكوين 50:10"> واَسْتَغرَقَ تَحنِيطُه أربعينَ يومًا، وهيَ المُدَّةُ الّتي تكتَمِلُ فيها أيّامُ المُحنَّطينَ. وبَكى المِصْريُّونَ على يعقوبَ سَبعينَ يومًا،</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 18:3; أسْتِير 4:2"> ولمَّا اَنقَضَت أيّامُ البُكاءِ على يعقوبَ قالَ يوسُفُ لأهلِ بَيتِ فِرعَونَ: ”إنْ حَظيتُ برضاكُم، فقولوا على مَسامِعِ فِرعَونَ</span>
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الثّانِي 16:14; إشَعْياء 22:16; ﻣﺘﻰ 27:60"> إنَّ أبي اَسْتَحْلَفَني وقالَ لي: حانَت ساعةُ موتي، فاَدفِنِّي في قبري الّذي حَفَرْتُه لي في أرضِ كنعانَ. هُناكَ اَدفنِّي والآنَ أنا ذاهبٌ، فأدفِنُ أبي وأرجِعُ“.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 48:21"> فكانَ جوابُ فِرعَونَ: ”إذهَبْ واَدفِنْ أباكَ كما اَسْتَحلَفَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 14:16"> فصَعِدَ يوسُفُ لِيَدفِنَ أباهُ، وصَعِدَ معَهُ جميعُ رجالِ حاشيةِ فِرعَونَ وكبارِ أهلِ بَيتهِ، وجميعُ أعيانِ مِصْرَ،</span>
<span className="v" id="v8" data-v="8" data-xref="العَدَد 32:24–27; الخُرُوج 10:26; الخُرُوج 10:8–9"> وجميعُ أهلِ بَيتِ يوسُفَ وإخوتُه وأهلُ بيتِ أبيه بعدَما تَركوا عيالَهم وغنَمَهم وبقَرَهم في أرضِ جاسانَ.</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 14:28; المُلُوكِ الثّانِي 18:24; نَشِيدِ الأنشاد 1:9"> وصَعِدَتْ معَ يوسُفَ مَركباتٌ وفُرسانٌ، فكانَ المَوكِبُ عظيمًا جدًّا.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="أيُّوب 2:13; اعمال الرسل 8:2; صموئيلَ الثّانِي 1:17"> فلمَّا وصَلُوا إلى بَيدَرِ أطادَ الّذي في عَبْرِ الأردُنِّ نَدَبُوه هناكَ كثيرًا جدًّا، وأقامَ لَه يوسُفُ مناحَةً سَبْعَةَ أيّامٍ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 10:15–19; التَّكوين 24:6; التَّثْنِيَة 11:30"> ورأى سُكّانُ أرضِ كنعانَ المناحَةَ في بَيدَرِ أطادَ فقالوا: ”هذِهِ مناحَةٌ عظيمَةٌ لِلمِصْرِيِّينَ“. فسُمِّيَ المَوضِعُ بالعِبريَّةِ آبلَ مِصْرايمَ، وهوَ في عَبْرِ الأردُنِّ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 7:16; التَّكوين 49:29–32; أفَسُس 6:1"> وعَمِلَ لَه بَنُوه كَما أوصاهُم،</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 49:29–31; التَّكوين 23:16–18; اعمال الرسل 7:16"> فحَمَلُوهُ إلى أرضِ كنعانَ ودَفَنوه في مَغارَةِ حقلِ المَكفيلَةِ، تُجاهَ مَمْرا، وهيَ الّتي اَشتراها إبراهيمُ معَ الحقلِ مِنْ عِفرونَ الحثِّيِّ مُلْكًا لِقبرٍ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14"> ثُمَّ رجَعَ يوسُفُ إلى مِصْرَ بَعدَ أنْ دَفَنَ أباهُ، هوَ وإخوتُه وسائرُ مَنْ صَعِدَ معَهُ لِدَفنِ أبيهِ.</span>
</div>
</div>

## موت يوسف

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="الأمثال 28:1; التَّكوين 42:21–22; المَزامِير 53:5"> فلمَّا رأى إخوةُ يوسُفَ أنَّ أباهُم ماتَ قالوا: ”لعلَّ يوسُفَ حقَدَ علَينا فيُجازِيَنا الآنَ على الشَّرِّ الّذي فَعَلْناه بِهِ“.</span>
<span className="v" id="v16" data-v="16" data-xref="الأمثال 29:25"> فأرسلوا مَنْ يقولُ لِيوسُفَ: ”أوصانا أبوكَ قبلَ مَوتِه قالَ:</span>
<span className="v" id="v17" data-v="17" data-xref="أفَسُس 4:32; ﻣﺘﻰ 6:14–15; التَّكوين 45:8"> قولوا ليوسُفَ: اَغفِرْ لإخوتِكَ ذنبَهُم وخطيئَتَهُم فهُم حَقًّا أساؤُوا إليكَ، والآنَ نتَوَسَّلُ إليكَ أنْ تغفِرَ ذنبَ عبيدِ إلهِ أبِيكَ“. فبَكى يوسُفُ حينَ بلَغَهُ هذا الكلامُ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 37:7–11; التَّكوين 42:6; التَّكوين 45:3"> وجاءَ إخوتُه بِأنفُسِهِم فاَرتَمَوا بَينَ يَدَيهِ وقالوا: ”ها نحنُ عبيدٌ لكَ“.</span>
<span className="v" id="v19" data-v="19" data-xref="ﺭﻭﻣﻴﺔ 12:19; التَّكوين 45:5; التَّكوين 30:2"> فقالَ لهُم يوسُفُ: ”لا تخافوا. هل أنا مكان اللهِ؟</span>
<span className="v" id="v20" data-v="20" data-xref="ﺭﻭﻣﻴﺔ 8:28; المَزامِير 119:71; التَّكوين 45:5–8"> الشَّرُّ الّذي أرَدتُموه لي أرادَهُ اللهُ خيرًا كما ترَونَ، لِيُنقِذَ حياةَ كثيرٍ مِنَ النَّاسِ.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 47:12; رسالَةُ بُطرُسَ الأُولَى 3:9; التَّكوين 45:10–11"> والآنَ لا تخافوا. أنا أعولُكُم أنتُم وعيالُكُم“. وعَزَّاهُم وطيَّبَ قلوبَهُم.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22"> وأقامَ يوسُفُ في مِصْرَ، هوَ وأهلُ بيتِ أبيهِ، وعاشَ مئةً وعشْرَ سِنينَ.</span>
<span className="v" id="v23" data-v="23" data-xref="العَدَد 32:39; التَّكوين 30:3; أيُّوب 42:16"> ورأى يوسُفُ مِنْ بني أفرايمَ الجِيلَ الثَّالثَ، وأيضًا بَنو ماكيرَ بنِ منَسَّى وُلِدُوا على رُكبَتَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 3:16–17; التَّكوين 35:12; التَّكوين 26:3"> وقالَ يوسُفُ لإخوتِهِ: ”حانَت ساعةُ موتي، واللهُ سيَذكُرُكُم بالخَير ويُخرِجُكُم مِنْ هذِهِ الأرضِ إلى الأرضِ الَّتي أقسَمَ علَيها لإبراهيمَ وإسحَقَ ويعقوبَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 13:19; يَشُوع 24:32; التَّكوين 47:29–31"> واَستَحلَفَ يوسُفُ بَني إِسرائيلَ إخوتَه وقالَ: ”حينَ يَذكُرُكُمُ اللهُ بالخَيرِ خُذوا عِظامي معَكُم مِنْ هُنا“.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 50:2–3; التَّكوين 47:9; يَشُوع 24:29"> وماتَ يوسُفُ وهوَ اَبنُ مئةٍ وعشْرِ سنينَ، فحَنَّطُوهُ ووَضَعُوه في تابوتٍ بِمِصْرَ.</span>
</div>
</div>
