## إسحق يبارك يعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صَمُوئيلَ الأوَّلُ 3:2; التَّكوين 48:10; التَّكوين 25:23–25"> ولمَّا شاخَ إسحَقُ وكلَّت عيناهُ عَنِ النَّظَرِ دعا عيسو اَبنَهُ الأكبرَ وقالَ لَه: ”يا اَبني“، قالَ: ”نعم. ها أنا“.</span>
<span className="v" id="v2" data-v="2" data-xref="يَعقُوب 4:14; الجامِعَة 9:10; الأمثال 27:1"> فقالَ: ”صِرتُ شيخًا كما ترى ولا أعرِفُ متى أموتُ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 25:27–28; التَّكوين 10:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:12"> فخُذْ عُدَّتَكَ وجُعْبَتَكَ وقوسَكَ واَخرُجْ إلى البرِّيَّةِ وتصيَّدْ لي صَيدًا،</span>
<span className="v" id="v4" data-v="4" data-xref="العِبرانِيّين 11:20; التَّكوين 48:9; التَّكوين 49:28"> وهَيِّئْ ليَ الأطعمةَ الّتي أُحبُّ، وجِئْني بِها فآكلَ وأبارِكَكَ قَبلَ أنْ أموتَ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5"> وكانَت رِفقةُ سامعةً حينما كلَّمَ إسحَقُ عيسو اَبنَهُ. فلمَّا خرجَ عيسو إلى البرِّيَّةِ ليصطادَ صَيدًا ويجيءَ بهِ إلى أبيهِ،</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 25:28"> قالت رِفقةُ ليعقوبَ اَبنِها: ”سَمِعتُ أباكَ يقولُ لعيسو أخيكَ:</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 6:26; صَمُوئيلَ الأوَّلُ 24:19; التَّثْنِيَة 33:1"> جِئْني بصيدٍ وهَيِّئْ ليَ أطعمةً فآكلَ مِنها وأباركَكَ أمامَ الرّبِّ قَبلَ موتي.</span>
<span className="v" id="v8" data-v="8" data-xref="أفَسُس 6:1; التَّكوين 27:13; اعمال الرسل 5:29"> والآنَ يا اَبني، اَسمَعْ لكلامي واَعملْ بِما أُوصيكَ بِه.</span>
<span className="v" id="v9" data-v="9" data-xref="القُضاة 13:15; صَمُوئيلَ الأوَّلُ 16:20"> إذهبْ إلى الماشيةِ وخُذْ لي مِنها جَديَيْنِ مِنْ خِيرةِ المَعَزِ، فأُهيّئَهُما أطعمةً لأبيكَ كما يُحِبُّ.</span>
<span className="v" id="v10" data-v="10"> فتُحضِرُهما إلى أبيكَ، ويأكلُ لِيبارِكَكَ قَبْلَ موتِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 25:25"> فقالَ يعقوبُ لرِفقةَ أُمِّهِ: ”لكنَّ عيسو أخي رَجُلٌ أشعَرُ وأنا رجُلٌ أملَسُ.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 27:18; التَّكوين 25:27; مَلاخِي 1:14"> ماذا لو جَسَّني أبي فوجدَني مُخادِعًا؟ ألا أجلِبُ على نفْسي لعنةً لا برَكةً؟“</span>
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 27:25; صَمُوئيلَ الأوَّلُ 25:24; صموئيلَ الثّانِي 14:9"> فقالت لَه أمُّهُ: ”عليَّ لعنَتُكَ يا اَبني. ما علَيكَ إلَّا أنْ تسمَعَ لِكلامي وتذهبَ وتجيئَني بالجَديَينِ.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 27:9; الأمثال 23:2–3; التَّكوين 25:28"> فذهبَ وجاءَ بهما إلى أُمِّهِ، فهيَّأت أطعمةً على ما يُحبُّ أبوهُ.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 27:27; لُوقا 20:46; لُوقا 15:22"> وأخذت رِفقةُ ثيابَ عيسو اَبنِها الأكبرِ الفاخرةَ الّتي عِندَها في البَيتِ، فألبسَتْها يعقوبَ اَبنَها الأصغرَ</span>
<span className="v" id="v16" data-v="16"> وكست يَدَيهِ والجانبَ الأملَسَ مِنْ عُنُقِهِ بِجِلدِ المَعَزِ.</span>
<span className="v" id="v17" data-v="17"> وناولت رِفقةُ يعقوبَ ما هيَّأتْهُ مِنَ الأطعمةِ والخُبزِ،</span>
<span className="v" id="v18" data-v="18"> فدخلَ على أبيهِ وقالَ: ”يا أبي“، قالَ: ”نعم، مَنْ أنتَ يا اَبني؟“</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 28:15; ﻣﺘﻰ 26:70–74; زَكَريّا 13:3–4"> فقالَ لَه يعقوبُ: ”أنا عيسو بِكرُكَ. فعَلْتُ كما أمرْتَني. قُمِ اَجلِسْ، وكُلْ مِنْ صَيدي، واَمنَحْني برَكَتَكَ“.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 20:7; أيُّوب 13:7; التَّكوين 24:12"> فقالَ لَه إسحَقُ: ”ما أسرعَ ما وجَدْتَ صَيدًا يا اَبني!“ قالَ: ”الرّبُّ إلهُكَ وفَّقَني“.</span>
<span className="v" id="v21" data-v="21" data-xref="يَعقُوب 4:8; إشَعْياء 57:19; التَّكوين 27:12"> فقالَ: ”تعالَ لأجُسَّكَ يا اَبني، فأعرفَ هل أنتَ اَبني عيسو أم لا“.</span>
<span className="v" id="v22" data-v="22"> فتقدَّمَ يعقوبُ إلى إسحَقَ أبيهِ، فجَسَّهُ وقالَ: ”الصَّوتُ صوتُ يعقوبَ، ولكنَّ اليدَينِ يَدا عيسو“.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 27:16; العِبرانِيّين 11:20; ﺭﻭﻣﻴﺔ 9:11–12"> ولم يعرِفْهُ، لأنَّ يَدَيهِ كانتا مُشعِرتَينِ كيَدَيْ عيسو أخيهِ. فقبلَ أنْ يباركَهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="كُولُوسِي 3:9; أفَسُس 4:25; الأمثال 12:22"> قالَ: ”هل أنتَ حقًّا اَبني عيسو؟“ قالَ: ”أنا هوَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 27:4"> فقالَ: ”قدِّمْ لي مِنْ صَيدِكَ، يا اَبني، حتّى آكُلَ وأُبارِكَكَ“. فقَدَّمَ لَه فأكَلَ، وجاءَ بخمرٍ فشربَ.</span>
<span className="v" id="v26" data-v="26"> وقالَ لَه إسحَقُ: ”تقَدَّمْ وقبِّلْني يا اَبني“.</span>
<span className="v" id="v27" data-v="27" data-xref="العِبرانِيّين 6:7; المَزامِير 65:10; العِبرانِيّين 11:20"> فتَقَدَّمَ وقبَّلَه، فشمَّ رائحةَ ثيابِه وباركَه وقالَ:<br />”ها رائحةُ اَبني<br />كرائِحةِ حقلٍ بارَكَه الرّبُّ</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 33:28; التَّثْنِيَة 33:13; المَزامِير 133:3"> يُعطيكَ اللهُ مِنْ نَدى السَّماءِ<br />ومِنْ خُصوبَة الأرْضِ<br />فَيضًا مِنَ الحِنطةِ والخمرِ!</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 12:3; العَدَد 24:9; إشَعْياء 49:23"> وتخدُمُكَ الشُّعوبُ<br />وتسجُدُ لكَ الأُمَمُ!<br />سيِّدًا تكونُ لإخوَتِكَ،<br />وبَنو أُمِّكَ يَسجدونَ لكَ.<br />مَلعونٌ مَنْ يلعَنُكَ،<br />ومُبارَكٌ مَنْ يُبارِكُكَ!“</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30"> فما إنْ فرَغَ إسحَقُ مِنْ برَكتِه، وخرَجَ يعقوبُ مِنْ عِندِه حتّى رجَعَ عيسو أخوهُ مِنَ الصَّيدِ.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 27:4"> فهيَّأَ هوَ أيضًا أطعمةً وجاءَ بِها إلى أبيهِ وقالَ لَه: ”قُمْ يا أبي، وكُلْ مِنْ صَيدي، وبارِكْني“.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 27:18"> فقالَ لَه أبوهُ: ”مَنْ أنتَ؟“ قالَ: ”أنا اَبنُكَ البِكرُ عيسو“.</span>
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 28:3–4; ﺭﻭﻣﻴﺔ 11:29; أفَسُس 1:3"> فاَرتَعَشَ إسحَقُ اَرتعاشًا شديدًا وقالَ: ”فمَنْ هوَ الذي صادَ صَيدًا وجاءَني به، فأكَلْتُ مِنه كلِّه قَبلَ أنْ تجيءَ وباركْتُه؟ نعم، بارَكْتُه ومُبارَكًا يكونُ“.</span>
<span className="v" id="v34" data-v="34" data-xref="العِبرانِيّين 12:17; الأمثال 1:31; صَمُوئيلَ الأوَّلُ 30:4"> فلمَّا سَمِعَ عيسو كلامَ أبيهِ صرَخَ عاليًا بِمرارةٍ وقالَ لَه: ”بارِكْني أنا أيضًا يا أبي“.</span>
<span className="v" id="v35" data-v="35" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:6; أيُّوب 13:7; مَلاخِي 2:10"> فأجابَه: ”جاءَ أخوكَ بِمَكْرٍ وأخذَ برَكَتَكَ“.</span>
<span className="v" id="v36" data-v="36" data-xref="التَّكوين 25:26; يُوحَنّا 1:47; التَّكوين 32:28"> فقالَ عيسو: ”ألأنَّ اَسمَهُ يعقوبُ تعَقَّبني مرَّتَينِ؟ أخذَ بَكورِيَّتي، وها هوَ الآنَ يأخُذُ برَكتي“. وقالَ: ”أما أبقيتَ لي برَكَةً؟“</span>
<span className="v" id="v37" data-v="37" data-xref="صموئيلَ الثّانِي 8:14; التَّكوين 27:28–29; التَّكوين 25:23"> فأجابَه إسحَقُ: ”ها أنا جعَلتُهُ سيِّدًا لكَ، وأعطيتُه جميعَ إخوتِه عبيدًا، وزَوَّدْتُه بالحِنطةِ والخمرِ، فماذا أعمَلُ لكَ يا اَبني؟“</span>
<span className="v" id="v38" data-v="38" data-xref="العِبرانِيّين 12:17; التَّكوين 27:34; إشَعْياء 65:14"> فقالَ عيسو: ”أما لكَ غيرُ برَكةٍ واحدةٍ يا أبي؟ بارِكْني أنا أيضًا يا أبي“. ورفعَ عيسو صوتَه وبكى.</span>
<span className="v" id="v39" data-v="39" data-xref="التَّكوين 27:28; العِبرانِيّين 11:20; يَشُوع 24:4"> فأجابَه أبوُه:<br />”بعيدًا عن خُصوبَةِ الأرضِ<br />يكونُ مَسكِنُكَ،<br />وعَنْ نَدى السَّماءِ مِنْ فوقُ.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="المُلُوكِ الثّانِي 8:20–22; التَّكوين 25:23; أخبارِ الأيّامِ الثّانِي 25:11–12"> بِسَيفِكَ تعيشُ وأخاكَ تخدُمُ<br />فإذا قوِيتَ تكسِر عَنْ عُنُقِكَ نِيرَه“.</span>
</div>
</div>

## يعقوب يهرب من عيسو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:12–15; التَّكوين 37:4; التَّكوين 50:3–4"> وحقَد عيسو على يعقوبَ بِسبَبِ البرَكةِ الّتي باركَهُ بها أبوه. وقالَ عيسو في نفْسهِ: ”إقتربت أيّامُ الحِدادِ على أبي. فأقتلُ يعقوبَ أخي“.</span>
<span className="v" id="v42" data-v="42" data-xref="التَّكوين 37:18–20; صَمُوئيلَ الأوَّلُ 30:5; التَّكوين 42:21–22"> وجاءَ مَنْ أخبرَ رِفقةَ بكلامِ عيسو، فاَستدعت يعقوبَ وقالت لَه: ”أخوكَ ينوي أنْ يقتُلَكَ.</span>
<span className="v" id="v43" data-v="43" data-xref="التَّكوين 27:8; التَّكوين 11:31; التَّكوين 27:13"> والآنَ اَسمعْ لكلامي يا اَبني، فقُمْ اَهربْ إلى لابانَ أخي في حارانَ،</span>
<span className="v" id="v44" data-v="44" data-xref="التَّكوين 31:38; التَّكوين 31:41"> وأقِمْ عِندَه أيّامًا قليلةً حتّى يهدأ غضَبُ أخيكَ</span>
<span className="v" id="v45" data-v="45" data-xref="الأمثال 19:21; التَّكوين 4:8–16; التَّكوين 9:5–6"> فإذا هدأَ غضَبُ أخيكَ ونسِيَ ما فعَلْتَ بهِ أُرسِلُ وآخذُكَ مِنْ هُناكَ. لماذا أفقدُكُما في يومٍ واحدٍ؟“</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="التَّكوين 24:3; التَّكوين 26:34–35; التَّكوين 28:8"> وقالت رِفقةُ لإسحَقَ: ”سَئِمتُ حياتي مِن اَمرأتَي عيسو الحِثِّيَّتَينِ، فإنْ تزوَّجَ يعقوبُ بواحدةٍ مِنْ بَناتِ حِثٍّ مِثلَ هاتَينِ أو مِنْ بَناتِ سائرِ أهلِ هذهِ الأرضِ، فما نَفْعُ حياتي؟“</span>
</div>
</div>
