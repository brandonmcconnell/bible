## يوسف وإخوته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 17:8; التَّكوين 23:4; التَّكوين 36:7"> وأقامَ يعقوبُ في أَرضِ كنعانَ، حيثُ تغرَّبَ أَبوهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 35:25–26; التَّكوين 35:22; صَمُوئيلَ الأوَّلُ 2:22–24"> وهذِهِ سيرةُ بَني يعقوبَ: كانَ يوسُفُ، وهوَ فتىً في السَّابعَةَ عَشْرَةَ مِنْ عُمرِه، يرعَى الغنَمَ معَ إخوتِهِ بَني بِلْهَةَ وبَني زِلفَةَ زوجَتي أبيهِ، فأخبرَ أبَاه بِما يُشاعُ عَنْ مَساوِئِهِم.</span>
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 3:35; التَّكوين 37:32; صموئيلَ الثّانِي 13:18"> وكان يعقوبُ يُحِبُّ يوسُفَ أكثرَ مِنْ سائِرِ بَنيهِ لأنَّه اَبنُ شَيخوخَتِهِ. فصنَعَ لَه قميصًا مُلَوَّنًا.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 27:41; رسالَةُ يُوحَنّا الأُولَى 3:12; رسالَةُ يُوحَنّا الأُولَى 4:20"> ورأَى إخوتُهُ أنَّ أبَاهُم يُحِبُّه أكثرَ مِنهُم جميعًا. فأَبغَضوهُ حتى لم يقدِروا أنْ يُكَلِّموهُ بِمَوَدَّةٍ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 28:12; التَّكوين 42:9; يُوحَنّا 17:14"> ورأى يوسُفُ حُلُمًا، فلمَّا أخبَرَ بِهِ إخوتَه زادَ بُغضُهم لَه.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 44:18; القُضاة 9:7"> قالَ لهُم: ”إسْمعوا هذا الحُلُمَ الّذي رأيتُه.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 44:14; التَّكوين 42:6; التَّكوين 43:26"> رأيتُ كأنَّنا نحزِمُ حُزَمًا في الحقلِ، فوَقَفَت حُزمَتي بَغتَةً واَنتَصَبَت ثُمَّ أحاطَت بِها حُزمُكُم وسجَدَت لها“.</span>
<span className="v" id="v8" data-v="8" data-xref="صَمُوئيلَ الأوَّلُ 10:27; اعمال الرسل 7:35; العِبرانِيّين 10:29"> فقالَ لَه إخوَتُه: ”أتَظنُّ أنَّكَ تَملِكُ وتتَسَلَّطُ علَينا؟“ وزادَ بُغضُهُم لَه لأجلِ أحلامِهِ وكلامِهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 7:9–14; التَّكوين 37:7; دانيال 8:10"> ورأى حُلُمًا آخرَ، فَقَصَّهُ على إخوتِهِ قالَ: ”رأيتُ حُلُمًا آخرَ، كأنَّ الشَّمسَ ساجِدةٌ لي والقمرَ وأحدَ عَشَرَ كوكبًا“.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 27:29; فِيلِبِّي 2:10–11; إشَعْياء 60:14"> ولمَّا قَصَّهُ على أبيهِ وإخوتِهِ إنتَهرَه أبوهُ وقالَ لَه: ”ما هذا الحُلُمُ الّذي رأيتَه؟ أنَجيءُ أنا وأمُّكَ وإخوتُكَ فنسجُدَ لكَ إلى الأرضِ؟“</span>
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 7:9; لُوقا 2:19; لُوقا 2:51"> فحَسدَهُ إخوتُه. وأمَّا أبوه فحفِظَ هذا الكلامَ في قلبهِ.</span>
</div>
</div>

## إخوة يوسف يبيعونه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 37:1; التَّكوين 33:18; التَّكوين 34:25–31"> وذَهَبَ إخوةُ يوسُفَ ليرعَوا غنَمَ أبيهِم عِندَ شكيمَ،</span>
<span className="v" id="v13" data-v="13" data-xref="أفَسُس 6:1–3; صَمُوئيلَ الأوَّلُ 3:8; صَمُوئيلَ الأوَّلُ 3:16"> فقالَ يعقوبُ ليوسُفَ: ”إخوتُكَ يرعَونَ الغنَمَ عِندَ شكيمَ، فتَعالَ أُرسِلُكَ إليهِم“. قالَ: ”نعم، ها أنا“.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 35:27; لُوقا 19:42; المَزامِير 125:5"> فقالَ لَه: ”اَذهبْ واَنظُرْ كيفَ حالُ إخوتِكَ وحالُ الغنَمِ وجِئْني بالخبَرِ!“ وأَرسلَه مِنْ وادي حَبرونَ، فجاءَ إلى شكِيمَ.</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 4:27; القُضاة 4:22; التَّكوين 21:14"> وصادَفَه رَجُلٌ وهوَ تائِهٌ في البرِّيَّةِ، فسألَه: ”ماذا تطلُبُ؟“</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 19:10; نَشِيدِ الأنشاد 1:7"> فأجابَ: ”أَطلبُ إخوتي. أخبِرني أينَ يَرعَونَ“.</span>
<span className="v" id="v17" data-v="17" data-xref="المُلُوكِ الثّانِي 6:13"> فقالَ الرَّجُلُ: ”رَحلوا مِنْ هُنا، وسَمِعْتُهُم يقولونَ: نذهبُ إلى دُوثانَ“. فسعَى يوسُفُ وراءَ إخوتِه، فوَجَدَهُم في دُوثانَ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 23:12; المَزامِير 37:12; المَزامِير 37:32"> فلمَّا رأَوه عَنْ بُعدٍ وهوَ يقتربُ مِنهُم تآمَروا ليقتُلُوه.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 28:12; التَّكوين 37:11; التَّكوين 37:5"> فقالَ بعضُهم لِبعضٍ: ”ها صاحبُ الأحلامِ مُقبلٌ نحوَنا.</span>
<span className="v" id="v20" data-v="20" data-xref="تِيطُس 3:3; الأمثال 27:4; ﻣﺘﻰ 2:2–16"> تَعالوا نقتُلْه ونَطرَحْهُ في بِئرٍ ونقولُ: ”وَحشٌ شَرِسٌ أكَلَهُ، ونرى ماذا تنفَعُ أحلامُهُ“.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 42:22; التَّكوين 35:22; يَشُوع 10:28"> فسَمِعَ رَأوبينُ، فأنقَذهُ مِنْ أَيديهِم وقالَ: ”لا نقتُلُه!“</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 22:12; التَّثْنِيَة 13:9; التَّكوين 42:22"> ثُمَّ قالَ لهُم رَأُوبينُ: ”لا تسفُكُوا دَمًا. إطرَحوهُ في هذِهِ البِئرِ الّتي في البرِّيَّةِ ولا ترفَعوا أيدِيَكُم علَيه“. قالَ هذا لِينقِذَه مِنْ أيديهم ويَرُدَّهُ إلى أَبيهِ.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 37:3; ﻣﺘﻰ 27:28; التَّكوين 42:21"> فلمَّا وصَلَ يوسُفُ إلى إخوتِهِ نزَعُوا عَنه القميصَ المُلوَّنَ الّذي يَلبسُهُ</span>
<span className="v" id="v24" data-v="24" data-xref="إرْمِيا 38:6; المَزامِير 130:1–2; زَكَريّا 9:11"> وأخذُوه وطَرَحوهُ في البِئرِ، وكانَتِ البِئرُ فارِغَةً لا ماءَ فيها،</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 37:28; التَّكوين 43:11; إرْمِيا 8:22"> ثُمَّ جَلَسوا يأكُلون. ورَفَعُوا عُيونَهم فرأَوا قافِلَةً مِن الإسماعيليِّينَ مُقبِلَةً مِنْ جِلْعادَ، وجِمالُهُم مُحَمَّلَةٌ صُمْغًا وبَلْسَمًا ومُرًّا، وهُم في طريقِ نُزُولِهم إلى مِصْرَ.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 37:20; التَّكوين 4:10; التَّكوين 25:32"> فقالَ يَهوذا لإخوتِهِ: ”ما الفائِدَةُ مِنْ أنْ نقتُلَ أخانا ونُخفيَ دَمَهُ؟</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 16:26; التَّكوين 42:21; صَمُوئيلَ الأوَّلُ 18:17"> تعالَوا نبيعُه لِلإِسماعيليِّينَ ولا نرفَعُ أيديَنا علَيهِ، فهُوَ أخونا مِنْ لَحمِنا ودَمِنا“. فسَمِعَ لَه إخوَتُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="اعمال الرسل 7:9; المَزامِير 105:17; التَّكوين 45:4–5"> ومَرَّ تُجَّارٌ مديانيُّونَ فأمسَكُوا يوسُفَ وأصعدُوه مِنَ البِئرِ وباعُوه لِلإِسماعيليِّينَ بِعِشْرينَ مِنَ الفِضَّةِ، فجاؤوا بهِ إلى مِصْرَ</span>
<span className="v" id="v29" data-v="29" data-xref="أيُّوب 1:20; التَّكوين 37:34; التَّكوين 44:13"> ورَجَع رَأوبينُ إلى البئر، فلم يَجدْ يوسُفَ في البئرِ، فمزّقَ ثيابَهُ</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 42:13; إرْمِيا 31:15; التَّكوين 42:32"> ورجَعَ إلى إخوتِهِ وقالَ لهُم: ”الولَدُ غيرُ موجُودٍ، وأنا إلى أينَ أذهبُ؟“</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 37:23; الأمثال 28:13; التَّكوين 37:3"> فأخَذوا قميصَ يوسُفَ وذبَحوا تَيسًا مِنَ المَعَزِ وغَمَسُوا القميصَ في الدَّمِ.</span>
<span className="v" id="v32" data-v="32" data-xref="لُوقا 15:30; التَّكوين 37:3; التَّكوين 44:20–23"> وأرسلوا القميصَ المُلوَّنَ إلى أبيهِم وقالوا: ”وَجَدْنا هذا. فتَحَقَّقْ أقميصُ اَبنِكَ هوَ أم لا؟“</span>
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 37:20; التَّكوين 44:28; يُوحَنّا 13:7"> فتَحَقَّقَهُ وقالَ: ”قميصُ اَبني. وَحشٌ شَرِسٌ أكَلَهُ. مَزَّقَ يوسُفَ تَمزيقًا“.</span>
<span className="v" id="v34" data-v="34" data-xref="التَّكوين 37:29; صموئيلَ الثّانِي 3:31; المَزامِير 69:11"> وشَقَّ يعقوبُ ثيابَه ولَبِسَ المِسْحَ حِدادًا على اَبنهِ، وناحَ أيّامًا كثيرةً.</span>
<span className="v" id="v35" data-v="35" data-xref="التَّكوين 42:38; أيُّوب 2:11; صموئيلَ الثّانِي 12:17"> وقامَ جميعُ بَنيهِ وبَناتِه يُعَزُّونَه، فأبى أنْ يتَعَزَّى وقالَ: ”بل أنزِلُ إلى عالمِ الأمواتِ نائِحًا على اَبني“. وبَكى علَيه يعقوبُ.</span>
</div>
<div className="bp">
