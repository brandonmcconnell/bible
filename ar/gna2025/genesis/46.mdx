## رحيل يعقوب إلى مصر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 28:13; التَّكوين 26:22–25; التَّكوين 28:10"> فرَحَلَ يعقوبُ بِجميعِ ما كانَ لَه وجاءَ إلى بِئرِ سَبْعَ، فذبَحَ ذبائِحَ لإلهِ أبيهِ إسحَقَ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 15:1; أيُّوب 33:14–15; أيُّوب 4:13"> فقالَ لَه اللهُ ليلًا في رُؤيا: ”يعقوبُ، يعقوبُ!“ قالَ: ”نعم، ها أنا“.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 12:2; التَّثْنِيَة 26:5; إشَعْياء 43:1–2"> قالَ: ”أنا اللهُ إلهُ أبيكَ. لا تَخَفْ أنْ تنزِلَ إلى مِصْرَ، فسَأجعَلُكَ أُمَّةً عظيمةً هُناكَ</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 3:8; التَّكوين 28:15; التَّكوين 48:21"> أنا أنزِلُ معَكَ إلى مِصْرَ وأنا أُصعِدُكَ منها ويوسُفُ هوَ يُغمِضُ عينَيكَ ساعةَ تموتُ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 45:21; التَّكوين 45:19; التَّكوين 45:27"> فقامَ يعقوبُ مِنْ بِئرِ سَبْعَ. وحَمَلَهُ بَنوه، كما حَمَلوا أطفالَهُم ونِساءَهُم على المَركباتِ الّتي أرسَلها فِرعَونُ.</span>
<span className="v" id="v6" data-v="6" data-xref="يَشُوع 24:4; إشَعْياء 52:4; اعمال الرسل 7:15"> وأخَذوا ماشيَتَهم وكُلَّ ما اَقتَنَوه في أرضِ كنعانَ وجاؤوا إلى مِصْرَ. وكانَ معَ يعقوبَ جميعُ نسلِهِ:</span>
<span className="v" id="v7" data-v="7"> بَنوه وبَنو بَنيهِ وبَناتُه وبَناتُ بَنيهِ، عِندَما جاءَ إلى مِصْرَ.</span>
</div>
</div>

## بنو يعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 1:1–5; التَّثْنِيَة 33:6; أخبارِ الأيّامِ الأوّلُ 2:1–55"> وهذِهِ أسماءُ بَني يعقوبَ الّذينَ جاؤوا إلى مِصْرَ: رَأوبينُ بِكْرُ يعقوبَ،</span>
<span className="v" id="v9" data-v="9"> وبَنو رَأوبينَ: حَنوكُ وفَلّو وحصرونُ وكَرمي،</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 6:15; التَّكوين 29:33; التَّكوين 28:1"> بَنو شِمعونَ: يَمُوئِيلُ ويامينُ وأوهَدُ وياكينُ وصُوحَرُ وشأُولُ اَبنُ الكنعانِّيةِ،</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 29:34; أخبارِ الأيّامِ الأوّلُ 6:16; العَدَد 8:1–26"> وبَنو لاوي: جِرشُونُ وقَهاتُ ومَراري،</span>
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الأوّلُ 4:21; التَّكوين 38:10; التَّكوين 29:35"> وبَنو يَهوذا عِيرُ وأُونانُ اللَّذانِ ماتا في أرضِ كنعانَ وشِيلَةُ وفارَصُ وزارَحُ، واَبنا فارَصَ: حصرونُ وحامولُ،</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الأوّلُ 12:32; التَّكوين 35:23; التَّكوين 30:14–18"> وبَنو يسَّاكرَ: تولاعُ وفُوَّةُ ويُوبُ وشِمرُونُ،</span>
<span className="v" id="v14" data-v="14" data-xref="التَّثْنِيَة 33:18–19; العَدَد 26:26–27; العَدَد 1:30–31"> وبَنو زبولونَ: سارَدُ وإِيلونُ وياحَلْئيلُ.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 29:32–35; التَّكوين 30:17–21; التَّكوين 34:1–31"> هؤلاءِ بَنو لَيئةَ الّذينَ ولَدَتْهم لِيعقوبَ في سَهلِ أرامَ معَ دينَةَ اَبنَتِهِ، وجميعُهُم ثَلاثةٌ وثلاثونَ نفْسًا.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="العَدَد 26:15–17; التَّكوين 30:11; العَدَد 1:24–25"> وبَنو جادَ: صِفْيونُ وحَجِّي وشُوني وأصْبونُ وعيري وأرودي وأرئيلي،</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 49:20; التَّكوين 30:13; العَدَد 1:13"> وبَنو أشيرَ: يِمنَةُ ويِشوَةُ ويِشوي وبَريعَةُ وسارَحُ أُختُهم، واَبنا بَرِيعَةَ: حابَرُ ومَلْكِيئيلُ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 29:24; التَّكوين 35:26; التَّكوين 30:9–13"> هؤلاءِ بَنو زِلْفَةَ الّتي أَعطاها لابانُ لِليئةَ اَبنَتِهِ فوَلَدَتْهم لِيعقوبَ، وجميعُهم ستَّ عَشْرَة نفْسًا.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 44:27; التَّكوين 29:18; التَّكوين 35:24"> واَبنا راحيلَ اَمرأةِ يعقوبَ: يوسُفُ وبنيامينُ.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 41:50–52; التَّكوين 41:45; التَّثْنِيَة 33:13–17"> ووُلِدَ لِيوسُفَ في أرضِ مِصْرَ منَسَّى وأفرايمُ مِنْ زوجتِهِ أسناتَ بِنتِ فوطي فارَعَ كاهنِ مدينَةِ أُونٍ.</span>
<span className="v" id="v21" data-v="21" data-xref="أخبارِ الأيّامِ الأوّلُ 7:6–12; أخبارِ الأيّامِ الأوّلُ 8:1–7; العَدَد 26:38–40"> وبَنو بنيامينَ: بالَعُ وباكَرُ وأشبيلُ وجيرا ونعمانُ وإِيحي ورُوشُ ومُفِّيمُ وحُفِّيمُ وأرْدُ.</span>
<span className="v" id="v22" data-v="22"> هؤلاءِ بَنو راحيلَ الّذينَ وُلِدُوا لِيعقوبَ. وجميعُهُم أربَعَ عشْرَةَ نفْسًا.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="العَدَد 1:38–39; التَّكوين 49:16–17; العَدَد 1:12"> واَبنُ دانَ: حوشيمُ،</span>
<span className="v" id="v24" data-v="24" data-xref="أخبارِ الأيّامِ الأوّلُ 7:13; العَدَد 1:42–43; التَّكوين 35:25"> وبَنو نَفتالي: ياحَصْئيلُ وجوني ويِصْرُ وشِلِّيمُ.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 29:29; التَّكوين 35:25; التَّكوين 30:3–8"> هؤلاءِ بَنو بِلْهَةَ الّتي أعطاها لابانُ لراحيلَ اَبنتهِ فوَلَدَتْهم لِيعقوبَ، وجميعُهم سَبْعُ أنفُسٍ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 1:5; التَّكوين 35:11; القُضاة 8:30"> فجميعُ الّذينَ جاؤوا إلى مِصْرَ معَ يعقوبَ، وهُم مِنْ صُلْبِه، ستَّةٌ وستُّونَ نفْسًا ما عدا نساءَ بَنيه.</span>
<span className="v" id="v27" data-v="27" data-xref="التَّثْنِيَة 10:22; اعمال الرسل 7:14; الخُرُوج 1:5"> واَبنا يوسُفَ اللَّذانِ وُلِدَا لَه في مِصْرَ نَفسانِ، فيكونُ جميعُ الّذينَ دخَلوا مِصْرَ مِنْ بَيتِ يعقوبَ سَبْعينَ نفْسًا.</span>
</div>
</div>

## لقاء يوسف ويعقوب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 45:10; التَّكوين 44:16–34; التَّكوين 49:8"> وأرسَلَ يعقوبُ اَبنَه يَهوذا قُدَّامَه إلى يوسُفَ لِيُلاقيَهُ في أرضِ جاسانَ. ثُمَّ جاؤوا أرضَ جاسانَ،</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 15:20; التَّكوين 45:14–15; التَّكوين 33:4"> فأعَدَّ يوسُفُ مَركَبَته وصَعِدَ إلى هُناكَ لِيلاقيَ يعقوبَ أباه، فلمَّا رآهُ أقبلَ علَيه وعانَقَهُ وبَكى طويلًا.</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 2:29–30; التَّكوين 45:28"> فقالَ لَه يعقوبُ: ”دَعني أموتُ الآنَ بَعدَما رأيتُ وجهَكَ وعرَفْتُ أنَّكَ حَيٌّ بَعدُ“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 47:1–3; العِبرانِيّين 2:11; اعمال الرسل 18:3"> وقالَ يوسُفُ لإخوتِه ولأهلِ بَيتِ أبيهِ: ”أنا ذاهِبٌ إلى فِرعَونَ لأقولَ لَه: إخوتي وأهلُ بيتِ أبي الّذينَ كانوا في أرضِ كنعانَ جاؤوا إليَّ.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 47:3; المُلُوكِ الأوَّلُ 9:27; المُلُوكِ الأوَّلُ 18:5–6"> وهم رُعاةُ غنَمٍ وأصحابُ ماشيةٍ، وأتَوا بِغنَمِهِم وبقَرِهِمِ وجميعِ ما هوَ لهُم.</span>
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 46:32; التَّكوين 47:2–4; ﻳﻮﻧﺎﻥ 1:8"> فإذا اَسْتَدعاكُم فِرعَونُ وقالَ لَكُم: ما حِرْفَتُكُم؟</span>
<span className="v" id="v34" data-v="34" data-xref="الخُرُوج 8:26; التَّكوين 43:32; التَّكوين 46:32"> فقولوا: كُنَّا يا سيِّدي أصحابَ ماشيةٍ مِنْ صِغَرِنا إلى الآنَ، نحنُ وأباؤُنا جميعًا. وهكذا يُمكِنُكُم أنْ تُقيمُوا بأرضِ جاسانَ، لأنَّ كُلَّ راعي غنَمٍ مَمقوتٌ عِندَ المِصْريِّينَ“.</span>
</div>
