## كأس يوسف في عدل بنيامين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 42:25; التَّكوين 43:16; التَّكوين 24:2"> ثُمَّ قالَ يوسُفُ لِوكيلِ بَيتهِ: ”اَملأْ عِدالَ هؤلاءِ القَومِ طَعامًا قَدْرَ ما يُطيقُونَ حِملَه، وضَعْ فِضَّةَ كُلِّ واحدٍ في فَمِ عِدْلِهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:8; التَّكوين 42:20; التَّثْنِيَة 8:16"> وضَعْ كأسي الّتي مِن الفِضَّةِ في فَمِ عِدلِ أَصغَرِهِم معَ فِضَّتِهِ ثمَن قمحِهِ“. ففَعلَ كما أَمرَهُ يوسُفُ.</span>
<span className="v" id="v3" data-v="3"> فلمَّا أضاءَ الصُّبحُ اَنصرَفَ الرِّجالُ بِحميرِهِم.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 10:32; المَزامِير 35:12; الأمثال 17:13"> فما إنْ خرَجُوا مِنَ المدينَةِ واَبتَعَدُوا قليلًا حتى قالَ يوسُفُ لِوَكيلِ بَيتِهِ: ”قُمِ اَتْبَعْ هؤُلاءِ الرِّجالَ، فإذا لَحِقْتَ بهِم فقُلْ لهُم: لِماذا كافأتُمُ الخَيرَ بالشَّرِّ؟</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 30:27; اللّاويِّين 19:26; المُلُوكِ الثّانِي 21:6"> لِماذا سَرَقتُم كأسَ الفِضَّةِ الّتي يشربُ بِها سيِّدي، وبِها يَرى أحوالَ الغَيبِ؟ أسأتُم في ما فعَلْتُم“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6"> فلَحِقَ الرَّجُلُ بهِم وقالَ لهُم ذلِكَ الكلامَ.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 38:16–18; التَّكوين 37:18–32; المُلُوكِ الثّانِي 8:13"> فأجابوه: ”لِماذا يتَكَلَّمُ سيِّدي بِمِثلِ هذا الكلامِ؟ حَرامٌ على عبيدِكَ أنْ يعمَلوا عمَلًا كهذا.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 43:12; التَّكوين 42:35; ﺭﻭﻣﻴﺔ 13:9"> تِلكَ الفِضَّةُ الّتي وَجَدْناها في أفواهِ عِدالِنا رَدَدْناها إِليكَ مِنْ أرضِ كنعانَ، فكيفَ نسرِقُ مِنْ بَيتِ سيِّدِكَ فِضَّةً أو ذهَبًا؟</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 31:32; المَزامِير 7:3–5; أيُّوب 31:38–40"> إنْ وَجَدْتَ الكأسَ معَ أحدٍ مِنَّا نحنُ عبيدِكَ فاَقتُلْهُ، ونحنُ أيضًا نكونُ عبيدًا لكَ يا سيِّدي“.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 22:3; التَّكوين 44:17; التَّكوين 44:33"> فقالَ: ”حسَنًا، فليَكُنْ كما تقولونَ. مَنْ وَجَدْتَ الكأسَ معَهُ يكونُ لي عبدًا، وأنتُم تكونونَ أبرياءَ“.</span>
<span className="v" id="v11" data-v="11"> فأسرَعَ كُلُّ واحدٍ مِنهُم وأنزَلَ عِدْلَهُ إلى الأرضِ وفَتَحَهُ،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 44:26–32; التَّكوين 42:36–38; التَّكوين 43:14"> ففَتَّشَهُم مُبتَدِئًا بالأكبرِ حتّى اَنتَهى إلى الأصغرِ، فإذا الكأسُ في عِدْلِ بِنيامينَ.</span>
<span className="v" id="v13" data-v="13" data-xref="صموئيلَ الثّانِي 1:11; التَّكوين 37:29–34; صموئيلَ الثّانِي 1:2"> فمَزَّقوا ثِيابَهُم، وحَمَّلَ كُلُّ واحدٍ حِمارَهُ، ورَجَعوا إلى المدينةِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 50:18; التَّكوين 43:25; التَّكوين 43:16"> ودَخَلَ يَهوذا وإخوَتُه إلى بَيتِ يوسُفَ، وهُو بَعدُ هُناكَ، ووقَعوا ساجدينَ أمامَهُ إلى الأرضِ</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 4:10; التَّكوين 21:26; التَّكوين 3:13"> فقالَ لهُم يوسُفُ: ”ما هذا العمَلُ الّذي عَمِلْتُم؟ أما عَلِمْتُم أنَّ رَجُلًا مِثلي يَرى أحوالَ الغَيبِ؟“</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 44:9; دانيال 9:7; العَدَد 32:23"> فقالَ يَهوذا: ”بماذا نُجِيبُكَ يا سيِّدي، وماذا نقولُ لكَ، وكيفَ نتبَرَّأُ بَعدَ أنْ كشَفَ اللهُ جُرْمَنا، نحنُ عبيدُكَ؟ هُنا نحنُ عبيدٌ لكَ يا سيِّدي، نحنُ ومَنْ وَجَدْتَ الكأسَ معَهُ“.</span>
<span className="v" id="v17" data-v="17" data-xref="الأمثال 17:15; صموئيلَ الثّانِي 23:3; التَّكوين 18:25"> فأجابَ يوسُفُ: ”حَرامٌ عليَّ أنْ أفعَلَ هذا، بلِ الرَّجلُ الّذي وَجَدْتُ الكأسَ معَهُ هوَ يكونُ لي عبدًا، وأنتُم ترجِعونَ بِسلامٍ إلى أبيكُم“.</span>
</div>
</div>

## وساطة يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 32:22; التَّكوين 41:40; التَّكوين 18:30"> فتَقَدَّمَ إليه يَهوذا وقالَ: ”عَفوكَ يا سيِّدي. دَعْني أقولُ كلِمَةً على مَسمَعِكَ، ولا يَشتَدَّ غضَبُكَ عليَّ أنا عبدُكَ، فأنتَ مِثْلُ فِرعَونَ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 43:7; التَّكوين 43:29; التَّكوين 42:7–10"> سألْتَنا يا سيِّدي هل لكُم أبٌ أو أخٌ؟</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 37:3; التَّكوين 42:38; لُوقا 7:12"> فأجَبناكَ: لنا يا سيِّدي أبٌ شيخٌ، واَبنٌ صغيرٌ وُلِدَ لَه في شَيخوخَتِه، أخوهُ ماتَ وبقيَ هوَ وحدَهُ لأُمِّهِ، وأبوهُ يُحِبُّه.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 42:15; التَّكوين 42:20; عامُوس 9:4"> فقلْتَ لنا يا سيّدي: أحضِروهُ إليَّ حتّى أُلْقيَ نظري علَيهِ.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 44:30; التَّكوين 42:38"> فقُلْنا لكَ يا سيِّدي: لا يَقدِرُ الصَّغيرُ أنْ يترُكَ أباه، وإنْ ترَكَهُ يموتُ أبوه.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 43:5; التَّكوين 43:3; التَّكوين 42:15–20"> فقُلتَ لنا: إنْ لم يَنزِلْ أخوكُمُ الصَّغيرُ معَكُم، فلا تَعودوا ترَونَ وجهي.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 42:29–34"> فلمَّا صَعِدْنا إلى عبدِكَ أبي أخبَرناهُ بِما قُلتَ لنا يا سيِّدي.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 43:2; التَّكوين 43:5"> وقالَ أبونا: إرجِعوا اَشتَروا لنا قليلًا مِنَ الطَّعامِ.</span>
<span className="v" id="v26" data-v="26" data-xref="لُوقا 11:7; التَّكوين 43:4–5"> فقُلنا: لا نقدِرُ أنْ ننزِلَ، لأنَّنا لا نقدِرُ أنْ نرَى وجهَ الرَّجُلِ إلَّا إذا كانَ أخونا الصَّغيرُ معَنا.</span>
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 46:19; التَّكوين 35:16–18; التَّكوين 30:22–25"> فقالَ لنا عبدُكَ أَبي: أنتُم تعلَمونَ أنَّ اَمرأتي راحيلَ ولَدَت لي اَبنَينِ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 37:33; التَّكوين 37:13–14; التَّكوين 42:36"> فخَرَجَ أحدُهُما مِنْ بَيتي وقُلتُ مَزَّقَه حيوانٌ مُفترِسٌ، لأنِّي إلى الآنَ ما رأيتُ لَه وجهًا.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 42:38; التَّكوين 43:14; التَّثْنِيَة 31:17"> فإنْ أخَذْتُمُ اَبني هذا أيضًا مِنْ أمامي فأَصابَه أذىً، أنزَلْتُم شَيبَتي بالبُؤسِ إلى عالَمِ الأموات.</span>
<span className="v" id="v30" data-v="30" data-xref="صَمُوئيلَ الأوَّلُ 18:1; صموئيلَ الثّانِي 18:33; التَّكوين 44:31"> والآنَ إذا رجَعْنا إلى عبدِكَ أبينا، ولم يَكُنِ الصَّبيُّ معَنا، وحياةُ أبي مُتَعَلِّقَةٌ بِحياتِهِ،</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 44:29; التَّكوين 37:35; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:10"> يموتُ عِندَما يرى الغُلامَ مَفقودًا فَنُنزِلُ شَيبَةَ أبينا بِحَسرَةٍ إلى عالَمِ الأمواتِ.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 43:8–9; التَّكوين 43:16"> ثُمَّ إنَّني يا سيِّدي ضَمِنْتُ الصَّبيَّ لأبي، فقُلتُ لَه: أعودُ بِهِ أو أكونُ مُخطِئًا إليكَ طُولَ الزَّمانِ.</span>
<span className="v" id="v33" data-v="33" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:16; الخُرُوج 32:32; العِبرانِيّين 7:22"> والآنَ دَعني يا سيِّدي أبقَى مكانَ الصَّبيِّ عبدًا لكَ، وَليَعُدْ هوَ معَ إخوَتِهِ.</span>
<span className="v" id="v34" data-v="34" data-xref="أسْتِير 8:6; المَزامِير 116:3; المَزامِير 119:143"> وإلَّا فكَيفَ أعودُ إلى أبي ولا يكونُ الصَّبيُّ معي، فأرَى الشَّرَّ الّذي يَحِلُّ بِأبي“.</span>
</div>
