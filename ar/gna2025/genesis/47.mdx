## المثول بين يدي فرعون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 45:10; التَّكوين 46:31; التَّكوين 46:28"> فدَخلَ يوسُفُ على فِرعَونَ وقالَ لَه: ”أبي وإخوتي جاؤوا منْ أرضِ كنعانَ بِغنَمِهِم وبقَرِهِم وكُلِّ ما هوَ لهُم، وها هُم في أرضِ جاسانَ“.</span>
<span className="v" id="v2" data-v="2" data-xref="اعمال الرسل 7:13; كُولُوسِي 1:28; يَهُوذا 1:24"> وأخذَ يوسُفُ خَمْسَةَ رِجالٍ مِنْ إخوتِهِ وأحضَرَهم أمامَ فِرعونَ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 46:33–34; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:10; التَّكوين 4:2"> فقالَ فِرعَونُ لإخوةِ يوسُفَ: ”ما حِرْفَتُكُم؟“ فقالوا لِفِرعَونَ: ”نحنُ يا سيِّدي رُعاةُ غنَمٍ، نحنُ وآباؤُنا جميعًا“.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 46:34; التَّكوين 15:13; التَّثْنِيَة 26:5"> وقالوا لَه: ”جِئْنا لِنتَغَرَّبَ في أرضِكَ يا سيِّدَنا، لأنَّ لا مَرعى لِغنَمِنا مِن اَشتدادِ الجُوعِ في أرضِ كنعانَ. فاَسمَحْ لنا أنْ نُقيمَ بأرضِ جاسانَ“.</span>
<span className="v" id="v5" data-v="5"> فقالَ فِرعَونُ ليوسُفَ: ”أبوكَ وإخوتُكَ جاؤوا إليكَ،</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 18:21; الأمثال 22:29; التَّكوين 47:11"> فهذِهِ أرضُ مِصْرَ بَينَ يَدَيكَ، أنزِلْهُم بأجودِها، ودَعْهُم يُقيمُونَ بأرضِ جاسانَ، وإنْ كُنتَ تعلَمُ أنَّ فيهِم قادرينَ على العَملِ، فأوكِلْهُم على ماشيتي“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 47:10; صموئيلَ الثّانِي 8:10; صَمُوئيلَ الأوَّلُ 2:20"> وأدخَلَ يوسُفُ يعقوبَ أباه وأوقفَهُ بَينَ يَدَيّ فِرعَونَ، فبارَكَهُ يَعقوبُ</span>
<span className="v" id="v8" data-v="8"> فقالَ لَه فِرعونُ: ”كم لكَ مِنَ العُمرِ؟“</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 39:12; المَزامِير 39:5; أيُّوب 14:1"> فأجابَهُ يعقوبُ: ”أيّامُ غُربَتي مئةٌ وثلاثونَ سنَةً. قليلةً وسَيِّئةً كانَت أيّامُ حياتي، على غيرِ ما كانَت أيّامُ حياةِ آبائي في غُربَتِهم“.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 47:7; المَزامِير 129:8; العِبرانِيّين 7:7"> وباركَ يعقوبُ فِرعَونَ وخرَجَ مِنْ بَينِ يَدَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 47:6; الخُرُوج 1:11; الخُرُوج 12:37"> وأسكَنَ يوسُفُ أباه وإخوَتَه وأعطاهُم مُلْكًا في أرضِ مِصْرَ، في أجوَدِ موضِعٍ مِنها، وهوَ أرضُ رَعَمْسيسَ، كما أمرَ فِرعَونُ.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 45:11; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:8; التَّكوين 47:1"> وعيَّنَ يوسُفُ لأبيهِ وإخوتهِ وسائرِ أهلِ بَيتِ أبيهِ مِقدارَ ما يحتاجونَ إليهِ مِنَ الطَّعامِ على حسَبِ عيالِهِم.</span>
</div>
</div>

## خطة يوسف الزراعية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="اعمال الرسل 7:11; المُلُوكِ الأوَّلُ 18:5; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:9"> ونفَدَ الخبزُ في كُلِّ مكانٍ، واَشتدَّتِ المَجاعَةُ حتّى خارت قِوى أهلِ مِصْرَ وأرضِ كنعانَ منَ الجُّوعِ.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 41:56; لُوقا 16:10–12; لُوقا 16:1–2"> وكانَ يوسُفُ يَجمَعُ كُلَّ الفِضَّةِ الّتي كانوا يشتَرونَ بِها القمحَ ويَجيءُ بها إلى بَيتِ فِرعَونَ.</span>
<span className="v" id="v15" data-v="15" data-xref="القُضاة 8:8; المَزامِير 37:3; ﻣﺘﻰ 6:11"> فلمَّا نَفَدَتِ الفِضَّةُ مِنْ أرضِ مِصْرَ ومِنْ أرضِ كنعانَ أقبلَ المِصْريُّونَ إلى يوسُفَ وقالوا لَه: ”أعطِنا خبزًا. أنموتُ أمامَ عينَيكَ لأنَّ الفِضَّةَ نفَدَت؟“</span>
<span className="v" id="v16" data-v="16" data-xref="كُولُوسِي 4:5; دانيال 6:5–7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:32"> فقالَ لهُم يوسُفُ: ”إذا كانت فضَّتُكُم نَفَدَت، فهاتوا ماشيتَكُم لأُعطيَكُم عِوَضَها خبزًا“.</span>
<span className="v" id="v17" data-v="17" data-xref="أيُّوب 2:4; الخُرُوج 9:3; إشَعْياء 31:1"> فجاؤوا بِماشيَتهِم إلى يوسُفَ، فأعطاهُم خبزًا بَدَلَ الخَيلِ والماشيةِ مِنَ الغنَمِ والبقَرِ والحميرِ. أعطاهُم خُبزًا بِكُلِّ ماشيَتهِم في تِلكَ السَّنةِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الثّانِي 6:26; إرْمِيا 38:9"> فلمَّا اَنْقَضَت تِلكَ السَّنَةُ جاؤوا في السَّنةِ الثَّانيةِ وقالوا لَه: لا نُخفي علَيكَ يا سيِّدي أنَّ الفِضَّةَ نَفَدَت، ومُقْتَنانَا مِنَ البَهائِمِ هوَ عِندَك، وما بقِيَ عِندَنا لِنُعطيَكَ إلَّا أبدانَنا وأراضينا.</span>
<span className="v" id="v19" data-v="19" data-xref="نَحَمْيا 5:2–3; أيُّوب 2:4; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:11"> أَنَهلَكُ أمامَ عينَيكَ نحنُ وأراضينا؟ إشتَرِنا نحنُ وأراضيَنا بالخبزِ، فنَصيرَ بأراضينا عبيدًا لِفِرعَونَ، وأعطِنا بِذارًا فنَحيا ولا نموتَ، ولا تصِيرُ أراضينا بُورًا“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20"> فاَشترى يوسُفُ جميعَ أراضي المِصْريِّينَ لِفِرعَونَ، لأنَّ كُلَّ واحدٍ مِنهُم باعَ حقلَهُ مِنْ شِدَّةِ الجُّوعِ. فصارتِ الأرضُ لِفِرعَونَ</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 41:48"> والشَّعبُ عبيدًا لَه مِنْ أقصى حُدودِ مِصْرَ إلى أقصاها.</span>
<span className="v" id="v22" data-v="22" data-xref="عَزْرا 7:24; نَحَمْيا 13:10; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:10"> أمَّا أراضي الكَهنَةِ فلم يَشتَرِها، لأنَّ الكهَنَةَ كانوا يعتاشون ممَّا خَصَّصَهُم بِه فِرعَونُ، فاَستَغنَوا عَن بَيعِ أراضيهِم.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="المَزامِير 112:5; المَزامِير 41:1; الجامِعَة 11:6"> وقالَ يوسُفُ للشَّعبِ: ”أنا اَشْترَيتُكُم اليومَ أنتُم وأراضيَكُم لِفِرعَونَ، فخُذُوا لَكُم بِذارًا تزرَعُونه في الأرضِ.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 41:34; المَزامِير 41:1; المَزامِير 112:5"> وعِندَ الحَصادِ تُعطُونَ خُمْسَ غِلالِكُم لِفِرعَونَ، والأربَعةُ الأخماسُ الباقيةُ تكونُ لكُم بِذارًا للحُقولِ وطعامًا لكُم ولأهلِ بُيوتِكُم وعِيالِكُم“.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 33:15; الأمثال 11:26–27; التَّكوين 18:3"> فقالوا: ”أنقَذْتَ حياتَنا. لَيْتَنا نحظَى بِرضاكَ يا سيِّدي، فنَكونَ عبيدًا لِفِرعَونَ“.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 47:22; حِزْقِيال 7:24"> فجَعَلَ يوسُفُ هذا الإجراءَ فريضةً على أرضِ مِصْرَ إلى هذا اليومِ، وهوَ أنْ يكونَ الخُمْسُ لِفِرعَونَ، ما عدا أراضيَ الكهَنةِ وحدَها لأنَّها لم تَصِرْ لِفِرعَونَ.</span>
</div>
</div>

## وصية يعقوب قبل موته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 46:3; الخُرُوج 1:7; اعمال الرسل 7:17"> وأقامَ يعقوبُ والّذينَ معَه بِجاسانَ في أرضِ مِصْرَ، فتَمَلَّكوا فيها ونَمَوا وتكاثَروا.</span>
<span className="v" id="v28" data-v="28" data-xref="المَزامِير 90:10; التَّكوين 47:8–9; التَّكوين 37:2"> وعاشَ يعقوبُ في أرضِ مِصْرَ سَبْعَ عشْرَةَ سنَةً فصارَ جميعُ عُمْرِهِ مئةً وسَبْعًا وأربَعينَ سنَةً.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّثْنِيَة 31:14; التَّكوين 24:49; التَّكوين 24:2"> ولمَّا دنَت ساعتُهُ دعا اَبنَه يوسُفَ وقالَ لَه: ”بِرِضاكَ عليَّ ضَعْ يَدَكَ تَحتَ فَخذي واَحْلِفْ لي بأنْ تكونَ وفيًّا وأمينًا لي. لا تدفنّي في مِصْرَ</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 49:29–32; التَّكوين 23:19; التَّكوين 25:9"> بل إذا مُتُّ اَحمِلْني واَدفنِّي في مقبرةِ آبائي“. قالَ: ”سأفعَلُ كما قُلتَ“.</span>
<span className="v" id="v31" data-v="31" data-xref="المُلُوكِ الأوَّلُ 1:47; العِبرانِيّين 11:21; التَّكوين 24:3"> فقال لَه: ”اَحْلِفْ لي“. فحَلَفَ لَه يوسُفُ. فسجَدَ يعقوبُ على رأسِ سريرِهِ.</span>
</div>
