## يعقوب يستعد للقاء عيسو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:6; لُوقا 14:11; التَّكوين 33:8"> وأرسَلَ يعقوبُ رُسُلًا يسبقونَه إلى عِيسو أخيهِ في أرضِ سَعيرَ وبِلادِ أدُومَ،</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 33:8; التَّكوين 33:15; التَّكوين 33:11"> وأوصاهُم فقالَ لهُم: ”قولوا لِسيِّدِي عِيسو: هذا ما يقولُه لكَ عبدُكَ يعقوبُ: نزَلْتُ عِندَ لابانَ وأَقَمتُ إلى الآنَ،</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 33:1; التَّكوين 32:8; التَّكوين 32:11"> وصارَ لي بقَرٌ وحميرٌ وغنَمٌ وعبيدٌ وجَوارٍ. فرأيتُ أنْ أُرسِلَ مَنْ يُخبِرُكَ، يا سيّدي، حتى أَنَالَ رِضاكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 35:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:8–10; الخُرُوج 14:10"> فرجَعَ الرُّسلُ إلى يعقوبَ وقالوا: ”ذهَبنا إلى أخيكَ عيسو، فإذا هوَ قادِمٌ لِلقائِكَ ومعَهُ أربَعُ مئةِ رَجُلٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="ﻣﺘﻰ 10:16; التَّكوين 33:1–3"> فاَستولى على يعقوبَ الخَوفُ والضِّيقُ، فقَسَمَ الجماعةَ الّذينَ معَهُ والغنَمَ والبقَرَ والجِمالَ إلى فِرقتَينِ</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 31:42; المَزامِير 34:4–6; التَّكوين 31:3"> وقالَ في نفْسِه: ”إنْ صادَفَ عيسو إِحدى الفِرقتَينِ فضَربَها نَجَتِ الفِرقَةُ الأُخرى“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 18:35; التَّكوين 24:27; صموئيلَ الثّانِي 7:18"> وقالَ يعقوبُ: ”يا إلهَ أبي إِبراهيمَ وإلهَ أبي إِسحَقَ، أيُّها الرّبُّ الّذي قالَ لي: إرجعْ إلى أرضِكَ وإلى عَشِيرتِكَ وأنا أُحسِنُ إليكَ،</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 59:1–2; الأمثال 18:19; ﻣﺘﻰ 6:13"> أنا دونَ أنْ أستَحِقَّ كُلَّ ما أظهَرتَه لي أنا عبدُكَ، مِنْ رحمةٍ ووفاءٍ، عبَرْتُ هذا الأردنَّ وما لي إلَّا عصايَ، وأمَّا الآنَ فصارَ لي فِرقتانِ.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 28:13–15; العِبرانِيّين 6:17; تِيطُس 1:2"> نَجِّني مِنْ يَدِ أخي عيسو، فأنا أخافُ مِنهُ أنْ يَجيءَ فيَقتُلَنا، أنا والأُمَّهاتِ والبَنينَ،</span>
<span className="v" id="v13" data-v="13" data-xref="الأمثال 18:16; التَّكوين 43:11; الأمثال 19:6"> وأنتَ قلتَ لي: أنا أُحسِنُ إليكَ وأجعَلُ نسلَكَ كرَملِ البَحرِ الّذي لا يُحصَى لِكَثرَتِهِ“.</span>
<span className="v" id="v14" data-v="14" data-xref="أيُّوب 1:3; التَّثْنِيَة 8:18; أيُّوب 42:12"> وباتَ يعقوبُ هُناكَ تِلكَ اللَّيلةَ، واَنتَقَى مِمَّا جاءَ بِه معَه هَديَّةً لِعيسو أخيهِ:</span>
<span className="v" id="v15" data-v="15"> مئتَي عَنْزٍ وعِشْرينَ تَيسًا، ومِئتي نَعجَةٍ وعِشْرينَ كَبْشًا،</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 33:8–9; التَّكوين 32:20; الأمثال 2:11"> وثلاثينَ ناقةً مُرضِعةً وأولادَها، وأربَعينَ بقَرَةً وعشَرَةَ ثِيرانٍ وَعِشْرينَ أتانًا وعشَرَةَ حميرٍ</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 33:3"> وسَلَّمَ يعقوبُ هذِهِ كُلَّها قطيعًا قطيعًا، كُلًا على حِدَةٍ إلى أيدي عبيدهِ وقالَ لهُم: ”تَقَدَّمُوا أمامي واَجْعَلُوا فُسحَةً بينَ قطيعٍ وقطيعٍ“.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 32:4–5"> وأوصَى العَبدَ الأوَّلَ فقالَ لَه: ”إنْ صادَفكَ عيسو أخي وسألَكَ: لِمَن أنتَ وإلى أينَ تذهَبُ، ولِمَنْ هذا الّذي أمامَكَ؟</span>
<span className="v" id="v19" data-v="19"> فقُلْ: لِعَبدِك يعقوبَ، وهوَ هَديَّةٌ أرسَلَها إلى سيِّدِهِ عيسو. وها هو نفْسُهُ ورَاءَنا“.</span>
<span className="v" id="v20" data-v="20" data-xref="أيُّوب 42:8–9; الأمثال 21:14; التَّكوين 43:11"> وأوصى العَبدَين الثَّانيَ والثَّالثَ وجميعَ السَّائرينَ خَلْفَ القُطعانِ، بِمِثلِ ذلِكَ، فقالَ لهُم: ”ذلِكَ ما تقولونَه لِعيسو إذا لقيتُمُوه“.</span>
<span className="v" id="v21" data-v="21"> وتقولونَ أيضًا: ”ها هوَ عبدُكَ يعقوبُ نفْسُهُ ورَاءَنا“. فعَلَ يعقوبُ هذا لأنَّه قالَ في نَفسِه: ”أَستَعْطِفُه أوَّلًا بالهَديَّةِ الّتي تقَدَّمَتْني إليهِ، حتى إذا تلاقينا وجهًا إلى وجهٍ لَعَلَّهُ يعفو عنِّي“.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 3:16; التَّثْنِيَة 2:37; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:8"> فتَقَدَّمَتْهُ الهَديَّةُ، وباتَ هوَ تِلكَ اللَّيلةَ في المحَلَّةِ.</span>
</div>
</div>

## صراع يعقوب مع الله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23"> وقامَ في اللَّيلِ، فأخَذَ اَمرَأَتَيهِ وجارِيَتَيهِ وبنيهِ الأَحَدَ عشَرَ وعَبَرَ مخاضَةَ يبُّوقَ،</span>
<span className="v" id="v24" data-v="24" data-xref="هُوشَع 12:3–5; أفَسُس 6:12; لُوقا 13:24"> أخَذَهُم وَأرسَلَهُم عَبرَ الوادي معَ كُلِّ ما كانَ لَه.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:7–9; هُوشَع 12:3–4; ﻣﺘﻰ 26:41"> وبقيَ يعقوبُ وحدَهُ، فصارَعَهُ رَجُلٌ حتّى طُلوعِ الفَجرِ.</span>
<span className="v" id="v26" data-v="26" data-xref="هُوشَع 12:4; المَزامِير 115:12–13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:8–9"> ولمَّا رأَى أنَّه لا يقوى على يعقوبَ في هذا الصِّراعِ، ضرَبَ حُقَّ وِرْكِه فاَنخَلَعَ.</span>
<span className="v" id="v27" data-v="27"> وقالَ لِيعقوبَ: ”طَلَعَ الفجرُ فاَترُكْني!“ فقالَ يعقوبُ: ”لا أتْرُكُكَ حتّى تُبارِكَني“.</span>
<span className="v" id="v28" data-v="28" data-xref="رُؤيا يُوحَنّا 2:17; التَّكوين 35:10; إشَعْياء 62:2–4"> فقالَ الرَّجلُ: ”ما اَسمُكَ؟“ قالَ: ”اَسمي يعقوبُ“.</span>
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 9:6; القُضاة 13:16–18; أيُّوب 11:7"> فقالَ: ”لا يُدعَى اَسمُكَ يعقوبَ بَعدَ الآنَ بل إِسرائيلَ، لأنَّكَ غالَبْتَ اللهَ والنَّاسَ وغلَبْتَ“.</span>
<span className="v" id="v30" data-v="30" data-xref="الخُرُوج 24:10–11; العَدَد 12:8; إشَعْياء 6:5"> وسألَهُ يعقوبُ: ”أخبِرْني ما اَسمُكَ“. فقالَ: ”لماذا تسأَلُ عَنِ اَسمي“. وبارَكَهُ هُناكَ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:7; التَّكوين 19:15; القُضاة 8:8"> وسمَّى يعقوبُ ذلِكَ المَوضِعَ فنوئيلَ، وقالَ: ”لأنِّي رأيتُ اللهَ وجهًا إلى وجهٍ ونجَوتُ بِحياتي“.</span>
<span className="v" id="v32" data-v="32" data-xref="صَمُوئيلَ الأوَّلُ 5:5"> وأشرَقَت لهُ الشَّمسُ وهوَ يعبُرُ فَنوئيلَ عارِجًا مِنْ وِرْكِه.</span>
<span className="v" id="v33" data-v="33"> لِذلِكَ لا يأكُلُ بَنو إِسرائيلَ عِرقَ النَّسا الّذي في حُقِّ الوِرْكِ إلى هذا اليومِ، لأنَّ الرَّجلَ ضرَبَ حُقَّ وِرْكِ يعقوبَ على عِرقِ النَّسا.</span>
</div>
