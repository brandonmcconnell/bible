## يوسف يتعرف إلى إخوته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 18:15; اعمال الرسل 10:41; التَّكوين 43:30–31"> فلم يقدِرْ يوسُفُ أن يَضبِطَ نفْسَهُ أمامَ جميعِ القائِمينَ على خِدمَتِهِ فصاحَ: ”أخرِجُوا كُلَّ واحدٍ مِنْ هُنا“. فلم يَكُنْ عِندَه أحدٌ حينَ تعَرَّفَ يوسُفُ إلى إخوتِهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 46:29; اعمال الرسل 20:37; العَدَد 14:1"> فرَفَعَ صوتَهُ بالبُكاءِ فسَمِعَهُ المِصْرِيُّونَ وسَمِعَه أهلُ بَيتِ فِرعَونَ.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 7:13; اعمال الرسل 9:5; زَكَريّا 12:10"> وقالَ يوسُفُ لإخوتِهِ: ”أنا يوسُفُ. أحَيٌّ أبي بَعدُ؟“ فلم يقدِرْ إخوَتُه أنْ يُجيبُوه لِفَزَعِهِم مِنهُ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 37:28; ﻣﺘﻰ 14:27; التَّكوين 50:18"> فقالَ يوسُفُ لإخوتِهِ: ”تقَدَّموا إليَّ“ فتَقَدَّموا. فقالَ: ”أنا يوسُفُ أخوكُم الّذي بِعتُمُوه إلى مِصْرَ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 50:20; اعمال الرسل 7:9–15; التَّكوين 45:7–8"> والآنَ لا تَأْسَفُوا ولا تَستاؤوا لأنَّكُم بِعتُموني إلى هُنا، لأنَّ اللهَ أرسَلَني أمامَكُم لأحفَظَ حياتَكُم.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 47:23; التَّكوين 41:54; التَّكوين 41:56"> مَضَتِ الآنَ سنَتا جُوعٍ في الأرضِ، وبقيَ خَمْسُ سِنينَ لا فِلاحَةٌ فيها ولا حَصادٌ،</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 44:4; اعمال الرسل 7:35; المَزامِير 18:50"> فأرسَلَني اللهُ أمامَكُم لِيُبقيَ لكُم نَسلًا في الأرضِ ويُنجِّيَ الأحياءَ مِنكُم.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 15:16; ﺭﻭﻣﻴﺔ 9:16; يُوحَنّا 19:11"> فما أنتُمُ الّذينَ أرسَلْتُموني إلى هُنا، بل اللهُ. وهوَ جعَلَني وَصِيًّا عِندَ فِرعَونَ، وسيِّدًا لِجميعِ أهلِ بَيتِهِ، ومُتسَلِّطًا على كُلِّ أرضِ مِصْرَ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 45:19–20; التَّكوين 45:13"> فأسرِعوا بالعودَةِ إلى أبي وقولوا لَه: هذا ما يقولُهُ اَبنُكَ يوسُفُ: جعَلَني اللهُ سيِّدًا لجميعِ المِصْريِّينَ. تَعالَ إليَّ ولا تَتأخَّرْ،</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 9:26; الخُرُوج 8:22; التَّكوين 46:34–6"> فتُقيمَ بأرضِ جاسانَ وتكونَ قريبًا منِّي، أنتَ وبَنوكَ وبَنو بَنيك وغنَمُكَ وبقَرُكَ وكُلُّ ما هوَ لكَ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 47:12; التَّكوين 47:6; ﻣﺘﻰ 15:5–6"> وأعولُكَ هُناكَ لأنَّه بقيَ خَمْسُ سِنينَ جُوعٍ، فلا يُعوِزُكَ شيءٌ، أَنتَ وأهلُ بيتِكَ وكُلُّ ما هوَ لكَ</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 42:23; لُوقا 24:39; يُوحَنّا 20:27"> وهذِهِ عُيونُكُم ترى، وعَينا أخي بنيامينَ أيضًا، أنّي أنا الّذي يُخاطِبُكُم.</span>
<span className="v" id="v13" data-v="13" data-xref="اعمال الرسل 7:14; رُؤيا يُوحَنّا 21:23; رسالَةُ بُطرُسَ الأُولَى 1:10–12"> فأخبِروا أبي بِكُلِّ مَجدي في مِصْرَ وبِكُلِّ ما رَأيتُم، وأسرِعوا بالمَجيءِ بِه إلى هُنا“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 33:4; التَّكوين 29:11; التَّكوين 46:29"> وعانَقَ يوسُفُ بنيامينَ أخاهُ وبَكَى، وبَكَى بنيامينُ على عُنقِهِ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 15:20; صَمُوئيلَ الأوَّلُ 10:1; المَزامِير 77:4"> وقبَّلَ سائِرَ إخوتِهِ وبَكى معَهُم. وبَعدَ ذلِكَ أخذوا يُكَلِّمُونَه.</span>
</div>
</div>

## دعوة فرعون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 41:33; أسْتِير 1:21; التَّكوين 20:15"> وبلَغَ قَصرَ فِرعَونَ خبَرُ مَجيءِ إخوةِ يوسُفَ، فَسَرَّ الخبَرُ فِرعَونَ ورِجالَ حاشيتِهِ</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 42:25–26; التَّكوين 44:1–2"> فقالَ فِرعَونُ ليوسُفَ: ”قُلْ لإخوتِكَ: حَمِّلوا دَوَابَّكُم واَرجِعُوا إلى أرضِ كنعانَ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 27:28; العَدَد 18:12; التَّكوين 47:6"> وخُذوا أباكُم وأهلَ بيوتِكُم وتعالَوا إليَّ فأُعطيَكُم أجوَدَ أرض في مِصْرَ، وتأكُلوا خَيراتِ الأرضِ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 46:5; التَّكوين 45:27; التَّكوين 31:17–18"> وأُوصيكَ أنْ تقولَ لهُم أيضًا: خُذوا لكُم مِنْ أرضِ مِصْرَ مَركباتٍ لِأطفالِكُم ونِسائِكُم، واَحمِلوا أباكُم وتَعالوا.</span>
<span className="v" id="v20" data-v="20" data-xref="إشَعْياء 1:19; لُوقا 17:31; التَّثْنِيَة 7:16"> وَلا تتَوَجَّعْ قلوبُكُم على أملاكِكُم، فأجوَدُ جميعِ ما في أرضِ مِصْرَ هوَ لكُم“.</span>
</div>
</div>

## العودة إلى أرض كنعان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 45:19; حِزْقِيال 23:24; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:18"> ففعَلَ بَنو يعقوبَ كما أمرَهُم فِرعَونُ. وأعطاهُم يوسُفُ مَركباتٍ وزادًا للطَّريقِ</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 43:34; المُلُوكِ الثّانِي 5:5; القُضاة 14:12"> وأعطَى كُلَّ واحِدٍ مِنهُم ثيابًا فاخِرَةً، وخَصَّ بَنيامينَ بِثَلاثِ مئةٍ مِنَ الفِضَّةِ وخَمْسِ حُلَلٍ مِنَ الثِّيابِ،</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 24:10; الخُرُوج 16:3; التَّكوين 43:11"> وأرسَلَ إلى أبيهِ أيضًا عشَرَةَ حميرٍ مُحَمَّلَةً مِنْ خَير ما في مِصْرَ، وعشْرَ أُتُنٍ مُحَمَّلَةً قمحًا وخبزًا وزادًا لِسفَرِ أبيهِ.</span>
<span className="v" id="v24" data-v="24" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:13; المَزامِير 133:1–3; التَّكوين 42:21–22"> ثُمَّ صَرَفَ إخوتَه في سبيلهِم وقالَ لهُم: ”لا تُبْطِئوا في الطَّريقِ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25"> فصَعِدوا مِنْ مِصْرَ وجاؤوا إلى يعقوبَ أبيهِم في أرضِ كنعانَ.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 44:28; التَّكوين 37:35; لُوقا 24:34"> وقالوا لَه: ”يوسُفُ حَيٌّ بَعدُ، وهوَ مُتسلِّطٌ على جميعِ أرضِ مِصْرَ“. فجَمَدَ قلبُه ولم يُصَدِّقْهُم.</span>
<span className="v" id="v27" data-v="27" data-xref="المَزامِير 85:6; إشَعْياء 57:15; التَّكوين 45:19"> فلمَّا أخبَروه بِكُلِّ ما قالَه لهُم يوسُفُ، ورأى المَركباتِ الّتي أرسَلَها يوسُفُ لِتَحمِلَه إلى مِصْرَ، اَنتَعَشَت رُوحُهُ</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 2:28–30; يُوحَنّا 16:21–22; التَّكوين 46:30"> وقالَ لهُم: ”كَفاني أنَّ يوسُفَ اَبني حَيٌّ بَعدُ. أذهَبُ وأراهُ قَبلَ أنْ أموتَ“.</span>
</div>
