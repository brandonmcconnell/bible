## إسحق ذبيحة للرب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العِبرانِيّين 11:17; يَعقُوب 2:21; التَّثْنِيَة 8:2"> وبَعدَ هذِهِ الأحداثِ اَمتحنَ اللهُ إبراهيمَ، فقالَ لَه: ”يا إبراهيمُ!“ قالَ: ”نعم، ها أنا“.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 3:16; ﺭﻭﻣﻴﺔ 8:32; العِبرانِيّين 11:17"> قالَ: ”خُذْ إسحَقَ اَبنَكَ وحيدَكَ الّذي تُحِبُّهُ واَذهبْ إلى أرضِ مُوريَّةَ، وهناكَ أصعِدْهُ مُحرقةً على جبَلٍ أدُلُّكَ علَيه“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 119:60; ﻣﺘﻰ 10:37; لُوقا 14:26"> فبكَّرَ إبراهيمُ في الغدِ وأسرجَ حمارَهُ وشقَّقَ حطبًا لمُحرقةٍ، ثُمَّ أخذَ اَثنينِ مِنْ خَدمِهِ وإسحَقَ اَبنَهُ وسارَ في طريقِهِ إلى المَوضِعِ الّذي دلَّهُ اللهُ علَيه</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 13:32; ﻣﺘﻰ 17:23; هُوشَع 6:2"> وفي اليومِ الثَّالثِ رفَعَ إبراهيمُ عينيه فأبصرَ المكانَ مِنْ بَعيدٍ،</span>
<span className="v" id="v5" data-v="5" data-xref="العِبرانِيّين 11:19; العِبرانِيّين 12:1"> فقالَ لخادمَيهِ: ”اَنتظرا أنتما هُنا معَ الحمارِ، وأنا والصَّبيُّ نذهبُ إلى هُناكَ، فنَسجُدُ ونرجِعُ إليكُما“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 19:17; رسالَةُ بُطرُسَ الأُولَى 2:24; إشَعْياء 53:6"> وأخذَ إبراهيمُ حطبَ المُحرقةِ، ووضَعَهُ على ظهرِ إسحقَ اَبنهِ، وأمسكَ بيدهِ النَّارَ والسِّكِّينَ، وسارا كِلاهُما معًا.</span>
<span className="v" id="v7" data-v="7" data-xref="ﺭﻭﻣﻴﺔ 8:15; ﻣﺘﻰ 26:39; التَّكوين 8:20"> فقالَ إسحَقُ لأبيهِ: ”يا أبي!“ قالَ: ”نعم يا اَبني“. قالَ: ”هُنا النَّارُ والحطَبُ، فأينَ الخروفُ للمُحرقَةِ؟“</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 1:36; يُوحَنّا 1:29; رُؤيا يُوحَنّا 5:12"> فأجابَ إبراهيمُ: ”اللهُ يُدَبِّرُ لَه الخروفَ للمُحرقَةِ يا اَبني“. وسارا كِلاهُما معًا.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="يَعقُوب 2:21; العِبرانِيّين 11:17–19; فِيلِبِّي 2:7–8"> فلمَّا وصلا إلى المَوضِعِ الّذي دَلَّهُ اللهُ علَيه، بَنى إبراهيمُ هُناكَ المذبحَ، ورتَّبَ الحطَبَ، وربطَ إسحَقَ اَبنَهُ، ووضَعَهُ على المذبحِ فوقَ الحطَبِ.</span>
<span className="v" id="v10" data-v="10" data-xref="يَعقُوب 2:21–23; العِبرانِيّين 11:17–19; إشَعْياء 53:6–12"> ومدَّ إبراهيمُ يدَهُ، فأخذَ السِّكِّينَ ليذبَحَ اَبنَه.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 21:17; الخُرُوج 3:4; التَّكوين 16:9–10"> فناداهُ ملاكُ الرّبِّ مِنَ السَّماءِ وقالَ: ”إبراهيمُ، إبراهيمُ!“ قالَ: ”نعم، ها أنا“.</span>
<span className="v" id="v12" data-v="12" data-xref="يَعقُوب 2:21–22; يَعقُوب 2:18; صَمُوئيلَ الأوَّلُ 15:22"> قالَ: ”لا تمُدَّ يدَكَ إلى الصَّبيِّ ولا تفعَلْ بهِ شيئًا. الآنَ عرفتُ أنَّكَ تخافُ اللهَ، فما بَخُلْتَ عليَّ باَبنِكَ وحيدِكَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:13; المَزامِير 40:6–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:9–10"> فرفَعَ إبراهيمُ عينيهِ ونظَرَ فرأى وراءَهُ كَبْشًا عالقًا بقرنيهِ بينَ الشُّجَيراتِ، فأقبلَ على الكَبْشِ وأخذَهُ وقَدَّمَهُ مُحرقَةً بَدلَ اَبنِه.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّكوين 22:13; الخُرُوج 17:15; التَّكوين 22:8"> وسمَّى إبراهيمُ ذلِكَ المَوضِعَ ”الرّبُّ يُدَبِّر“. فيُقالُ إلى اليومِ: هذا جَبلُ الرّبِّ يدبِّرُ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15"> ونادى ملاكُ الرّبِّ إبراهيمَ ثانيةً مِنَ السَّماءِ</span>
<span className="v" id="v16" data-v="16" data-xref="العِبرانِيّين 6:13–14; لُوقا 1:73; المَزامِير 105:9"> وقالَ: ”بنفْسي أقسمْتُ، يقولُ الرّبُّ: بما أنَّكَ فعلْتَ هذا وما بَخُلْتَ باَبنِكَ وحيدِكَ،</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 15:5; التَّكوين 24:60; إرْمِيا 33:22"> فأُباركُكَ وأُكثِّرُ نسلَكَ كنجُومِ السَّماءِ والرَّملِ الّذي على شاطئِ البحرِ. ويَرِثُ نسلُكَ مُدُنَ أعدائهِ،</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 3:25; التَّكوين 12:3; غَلاطِيَّة 3:16"> ويتبارَكُ في نسلِكَ جميعُ أُمَمِ الأرضِ لأنَّكَ سَمِعْتَ لي“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّكوين 21:31; القُضاة 20:1; يَشُوع 15:28"> ثُمَّ رجَعَ إبراهيمُ إلى خادمَيهِ، فقاموا وذهبوا معًا إلى بئرَ سَبْعَ، وأقامَ إبراهيمُ هُناكَ.</span>
</div>
</div>

## نسل ناحور

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 11:29; التَّكوين 24:24; الأمثال 25:25"> وبَعدَ هذِهِ الأحداثِ قيلَ لإبراهيمَ ها مِلْكَةُ أيضًا ولَدتْ بَنينَ لناحورَ أخيكَ:</span>
<span className="v" id="v21" data-v="21" data-xref="أيُّوب 1:1; أيُّوب 32:2; المَزامِير 60:1–12"> عوصًا بكْرَه، وبُوزًا أخاهُ، وقموئيلَ أبا أرَامَ،</span>
<span className="v" id="v22" data-v="22"> وكاسدَ وحزوَ وفِلداشَ ويدلافَ وبتوئيلَ.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 24:15; التَّكوين 28:5; التَّكوين 24:51"> وولَدَ بتوئيلُ رفقةَ. هؤلاءِ الثَّمانيةُ ولَدتْهُم مِلكَةُ لناحورَ أخي إبراهيمَ.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 16:3; التَّكوين 25:6; الأمثال 15:25"> وأمَّا محظيَّتُه واَسمُها رؤومةُ، فولَدتْ هي أيضًا طابحَ وجاحَمَ وتاحَشَ ومَعكَةَ.</span>
