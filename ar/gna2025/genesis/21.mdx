## مولد إسحق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المَزامِير 12:6; التَّكوين 18:10; التَّكوين 18:14"> وتفقَّدَ الرّبُّ سارةَ كما قالَ، وفعَلَ لها كما وعَدَ.</span>
<span className="v" id="v2" data-v="2" data-xref="غَلاطِيَّة 4:22; العِبرانِيّين 11:11; التَّكوين 18:10"> فحَمَلت سارةُ وولَدت لإبراهيمَ اَبنًا في شيخوختهِ، في الوقتِ الّذي تكلَّمَ اللهُ عنهُ</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 17:19; ﺭﻭﻣﻴﺔ 9:7; التَّكوين 21:6"> فسَمَّاهُ إسحَقَ.</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 7:8; التَّثْنِيَة 12:32; يُوحَنّا 7:22–23"> وختَنَ إبراهيمُ إسحَقَ اَبنَهُ وهوَ اَبنُ ثمانيةِ أيّامٍ، كما أوصاهُ اللهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 17:17; ﺭﻭﻣﻴﺔ 4:19; التَّكوين 17:1"> وكانَ إبراهيمُ اَبنَ مئةِ سنَةٍ حينَ وُلِدَ إسحقُ.</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 126:2; إشَعْياء 54:1; التَّكوين 17:17"> وقالت سارةُ: ”اللهُ جعَلَني أضحَكُ، وكُلُّ مَنْ سَمِعَ يضحَكُ لي أيضًا“.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 18:11–12; إشَعْياء 66:8; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 1:10"> وقالت: ”مَنْ كانَ يقولُ إنَّ سارةَ ستُرضِعُ لإبراهيمَ بنَينَ؟ وها أنا وَلَدتُ لَه اَبنًا في شيخوختِهِ!“</span>
</div>
</div>

## طرد هاجر وإسماعيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="صَمُوئيلَ الأوَّلُ 1:22; هُوشَع 1:8; التَّكوين 40:20"> وكبُرَ الصَّبيُّ وفُطِمَ. وأقامَ إبراهيمُ وليمةً عظيمةً في يومِ فِطامِ إسحَقَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="غَلاطِيَّة 4:29; التَّكوين 16:15; غَلاطِيَّة 4:22"> ورأت سارةُ اَبنَ هاجرَ المِصْريَّةِ الّذي ولَدتْهُ لإبراهيمَ يلعبُ مع اَبنِها إسحَقَ،</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 8:35; رسالَةُ يُوحَنّا الأُولَى 2:19; غَلاطِيَّة 4:22–31"> فقالت لإبراهيمَ: ”أُطردْ هذِهِ الجاريةَ واَبنَها! فاَبنُ هذِهِ الجاريةِ لا يَرِثُ معَ اَبني إسحَقَ“.</span>
<span className="v" id="v11" data-v="11" data-xref="العِبرانِيّين 12:11; التَّكوين 17:18; ﻣﺘﻰ 10:37"> وساءَ إبراهيمَ هذا الكلامُ، لأنَّ إسماعيلَ، كانَ أيضًا اَبنَه.</span>
<span className="v" id="v12" data-v="12" data-xref="العِبرانِيّين 11:18; ﺭﻭﻣﻴﺔ 9:7–8; التَّكوين 17:19"> فقالَ لَه اللهُ: ”لا يَسوؤُكَ هذا الكلامُ على الصَّبيِّ وعلى جاريتِكَ. إسمعْ لكُلِّ ما تقولُهُ لكَ سارةُ، لأنَّ بإسحَقَ يكونُ لكَ نسلٌ.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 16:10; التَّكوين 17:20; التَّكوين 21:18"> واَبنُ الجاريةِ أيضًا أجعلُهُ أُمَّةً لأنَّهُ مِنْ صُلبِكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الجامِعَة 9:10; المَزامِير 119:60; التَّكوين 16:7"> فبكَّرَ إبراهيمُ في الغدِ وأخذَ خُبزًا وقِربةَ ماءٍ، فأعطاهُما لهاجرَ ووضَعَ الصَّبيَّ على كتِفِها وصرفَها، فمضَت تَهيمُ على وجهِها في صحراءِ بئرَ سَبْعَ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الثّانِي 3:9; المَزامِير 63:1; إرْمِيا 14:3"> ونفدَ الماءُ مِنَ القِربةِ، فألقت هاجرُ الصَّبيَّ تَحتَ إحدى الأشجارِ</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 49:15; التَّكوين 29:11; صَمُوئيلَ الأوَّلُ 24:16"> ومضَت فجلسَت قُبَالتَهُ على بُعْدِ رميَتَي قوسٍ وهيَ تقولُ في نفْسِها: ”لا أُريدُ أنْ أرى الولدَ يموتُ“. وفيما هيَ جالسةٌ رفعت صوتَها بالبُكاءِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 16:11; المَزامِير 50:15; الخُرُوج 3:7"> وسَمِعَ اللهُ صوتَ الصَّبيِّ، فنادى ملاكُ اللهِ هاجرَ مِنَ السَّماءِ وقالَ لها: ”ما لكِ يا هاجرُ؟ لا تخافي. سَمِعَ اللهُ صوتَ الصَّبيِّ حَيثُ هوَ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 21:13; التَّكوين 16:10; التَّكوين 17:20"> قُومي اَحملي الصَّبيَّ وخُذي بيدهِ، فسأجعَلُهُ أمَّةً عظيمةً“.</span>
<span className="v" id="v19" data-v="19" data-xref="العَدَد 22:31; المُلُوكِ الثّانِي 6:17–20; إشَعْياء 35:5–6"> وفتحَ اللهُ بصيرتَها فرأت بئرَ ماءٍ، فمضَت إلى البئرِ وملأتِ القِربةَ ماءً وسقتِ الصَّبيَّ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 28:15; التَّكوين 39:21; التَّكوين 16:12"> وكانَ اللهُ معَ الصَّبيِّ حتى كبُرَ، فأقامَ بالصَّحراءِ، وكان راميًا بالقوسِ.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 26:34–35; التَّكوين 24:3–4; العَدَد 12:16"> وحينَ أقامَ بصحراءِ فارانَ، زوَّجَتْهُ أمُّهُ باَمرأةٍ مِنْ أرضِ مِصْرَ.</span>
</div>
</div>

## إبراهيم وأبيمالك في بئر سبع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يَشُوع 3:7; التَّكوين 26:26; التَّكوين 26:28"> وفي تِلكَ الأيّامِ جاءَ أبيمالكُ معَ فيكولَ رئيسِ جيشهِ وقالَ لإبراهيمَ: ”أرى أنَّ اللهَ معَكَ في كُلِّ ما تعمَلُهُ،</span>
<span className="v" id="v23" data-v="23" data-xref="يَشُوع 2:12; التَّكوين 20:14; العِبرانِيّين 6:16"> فاَحْلِفْ لي هُنا باللهِ أنَّكَ لا تغدُرُ بي ولا بذُرِّيَّتي ونسلي، بل تُبادلُني معروفًا بمعروفٍ، أنا والأرضَ الّتي تغرَّبتَ فيها“.</span>
<span className="v" id="v24" data-v="24" data-xref="العِبرانِيّين 6:16; التَّكوين 14:13; ﺭﻭﻣﻴﺔ 12:18"> فقالَ إبراهيمُ: ”أحلِفُ“.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 26:15–22; التَّكوين 29:8; الأمثال 25:9"> وعاتبَ إبراهيمُ أبيمالكَ بخصوصِ البئرِ التي اَغتصَبها منهُ عبيدُ أبيمالكَ.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 13:7; المُلُوكِ الثّانِي 5:20–24"> فقالَ أبيمالكُ: ”لا أعرفُ مَنْ فعَلَ هذا، فلا أنتَ أخبرْتَني ولا أنا سَمِعْتُ بهِ إلى الآنَ“.</span>
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 14:22–23; إشَعْياء 32:8; ﺭﻭﻣﻴﺔ 1:31"> وأخذَ إبراهيمُ غنَمًا وبقَرًا، فأعطى أبيمالكَ وقطَعا كِلاهُما عهدًا.</span>
<span className="v" id="v28" data-v="28"> وأفردَ إبراهيمُ سَبْعَ نِعاجٍ مِنَ الغنَمِ على حِدَةٍ.</span>
<span className="v" id="v29" data-v="29" data-xref="صَمُوئيلَ الأوَّلُ 15:14; التَّكوين 33:8; الخُرُوج 12:26"> فقالَ لَه أبيمالكُ: ”ما هذِهِ السَّبْعُ النِّعاجِ الّتي أفردْتَها على حِدَةٍ؟“</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 31:52; يَشُوع 24:27; يَشُوع 22:27–28"> فأجابَ إبراهيمُ: ”سَبْعُ نعاجٍ تأخذُ مِنْ يَدي، شهادةً لي بأنِّي حفَرْتُ هذِهِ البئرَ“.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 26:33; التَّكوين 21:14; المُلُوكِ الأوَّلُ 4:25"> وسمَّى ذلِكَ المَوضِعَ بئرَ سَبْعَ، لأنَّ فيهِ تمَّ الإِتِّفاقُ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 13:17; التَّكوين 21:27; القُضاة 13:1"> وبَعدَما تعاهدا في بئرِ سَبْعَ، رجَعَ أبيمالكُ وفيكولُ رئيسُ جيشِه إلى أرضِ الفلِسطيّينَ</span>
<span className="v" id="v33" data-v="33" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:17; إشَعْياء 40:28; المَزامِير 90:2"> وغرَسَ إبراهيمُ شجَرًا في بئرِ سَبْعَ ودعا هُناكَ باَسمِ الرّبِّ الإلهِ السَّرمديِّ.</span>
<span className="v" id="v34" data-v="34" data-xref="أخبارِ الأيّامِ الأوّلُ 29:15; رسالَةُ بُطرُسَ الأُولَى 2:11; العِبرانِيّين 11:9"> وتغرَّبَ إبراهيمُ في أرضِ الفلِسطيِّينَ أيّامًا كثيرةً.</span>
</div>
</div>
