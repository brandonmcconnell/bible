## خراب سدوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 18:22; التَّكوين 18:1–5; العِبرانِيّين 13:2"> فجاءَ المَلاكانِ إلى سدومَ عِندَ الغُروبِ وكانَ لُوطٌ جالسًا بِبابِ المدينةِ، فلمَّا رآهُما قامَ لِلقائِهِما وسجدَ بِوجهِهِ إلى الأرضِ</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 18:4; لُوقا 24:28–29; العِبرانِيّين 13:2"> وقالَ: ”يا سيّديَّ، ميلا إلى بَيتِ عبدِكُما وبيتا واَغْسِلا أرجُلَكُما، وفي الصَّباحِ باكرًا تَستأنِفانِ سَفَرَكُما“. فقالا: ”لا، بل في السَّاحَةِ نَبيتُ“.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 18:6–8; القُضاة 6:19; صَمُوئيلَ الأوَّلُ 28:24"> فألَحَّ علَيهِما كثيرًا حتّى مالا إليهِ ودَخَلا بَيتَه، فعَمِلَ لهُما وليمةً وخبزَ فطيرًا فأكلا.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 13:13; الأمثال 4:16; الخُرُوج 23:2"> وقَبلَ أنْ يناما جاءَ رِجالُ سدومَ جميعًا، شُبَّانًا وشُيوخًا، وأحاطوا بالبَيتِ مِنْ كُلِّ جِهةٍ،</span>
<span className="v" id="v5" data-v="5" data-xref="القُضاة 19:22; إشَعْياء 3:9; يَهُوذا 1:7"> فنادوا لُوطًا وقالوا لهُ: ”أينَ الرَّجُلانِ اللَّذانِ دَخلا بَيتَكَ اللَّيلةَ؟ أخرِجْهُما إلينا حتى نُضاجِعَهُما“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="القُضاة 19:23"> فخَرجَ إليهِم لُوطٌ وأغلقَ البابَ وراءَه</span>
<span className="v" id="v7" data-v="7" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:9–11; القُضاة 19:23; اللّاويِّين 20:13"> وقالَ: ”لا تفعَلوا سُوءًا يا إخوتي.</span>
<span className="v" id="v8" data-v="8" data-xref="القُضاة 19:24; التَّكوين 19:31–38; القُضاة 9:15"> لي بِنتانِ ما ضاجعَتا رَجُلًا، أُخرِجُهما إليكُم فاَفعَلوا بهما ما يحلو لكُم. وأمَّا الرَّجُلانِ فلا تفعلوا بهما شيئًا، لأنَّهما في ضِيافتي“.</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 2:14; رسالَةُ بُطرُسَ الثّانيَة 2:7–8; التَّكوين 13:12"> فقالوا لَهُ: ”اَبتَعِدْ مِنْ هُنا! جِئتَ أيُّها الغريبُ لِتُقيمَ بَينَنا وتتحكَّمَ فينا. الآنَ نفعَلُ بِكَ أسوأَ مِمّا نفعَلُ بِهِما“. ودفعوا لُوطًا إلى الوراءِ وتقدَّموا إلى البابِ لِيَكسِرُوه.</span>
<span className="v" id="v10" data-v="10"> فمَدَّ الرَّجُلانِ أيديَهما وجذَبا لُوطًا إلى البَيتِ وأغلقا البابَ.</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 6:18; اعمال الرسل 13:11; التَّثْنِيَة 28:28–29"> وأمَّا الرِّجالُ الّذينَ على بابِ البَيتِ فضربَهُمُ الرَّجلانِ بالعَمَى، مِنْ صغيرِهِم إلى كبيرِهِم، فعجزُوا عَنْ أنْ يَجِدوا البابَ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 7:1; رسالَةُ بُطرُسَ الثّانيَة 2:7; رُؤيا يُوحَنّا 18:4"> وقالَ الرَّجُلانِ للُوطٍ: ”مَنْ لكَ أيضًا هُنا؟ إنْ كانَ لكَ أصهارٌ وبَنونَ وبَناتٌ وأقرباءُ آخرونَ في هذِهِ المدينةِ، فأخرِجْهُم مِنها.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 18:20; ﻣﺘﻰ 13:49–50; يَهُوذا 1:7"> فهذا المكانُ سَنُهلِكُه، لأنَّ الشَّكوى على أهلهِ بلغَت مَسامعَ الرّبِّ فأرسلَنا لِنُهلِكَهُم“.</span>
<span className="v" id="v14" data-v="14" data-xref="العَدَد 16:21; إرْمِيا 51:6; العَدَد 16:45"> فخرجَ لُوطٌ وقالَ لِصهرَيهِ الخاطبينِ بِنتَيهِ: ”قُومَا اَخرُجا مِنْ هُنا، لأنَّ الرّبَّ سَيُهلِكُ المدينةَ“. فكانَ كَمَنْ يَمزَحُ في نظَرِ صِهرَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="العَدَد 16:24–27; رُؤيا يُوحَنّا 18:4; التَّكوين 19:22"> فلمَّا طلَعَ الفَجرُ كانَ الملاكانِ يستعجلانِ لُوطًا ويقولانِ لهُ: ”قُمْ خُذِ اَمرأتَكَ واَبنتَيكَ الموجودَتينِ هُنا، لِئلَّا تَهلَكُوا معَ المدينةِ عِقابًا لها“.</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 9:15–16; تِيطُس 3:5; يُوحَنّا 6:44"> فلمَّا تباطأ لُوطٌ أمسَكَ الرَّجُلانِ بيدهِ وبيدِ اَمرأتهِ واَبنتَيهِ، لشفقةِ الرّبِّ علَيهِ، وأخرَجاهُ مِنَ المدينةِ وتركاهُ هُناكَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 24:16–18; لُوقا 9:62; فِيلِبِّي 3:13–14"> وبَينَما هُما يُخرجانِهِ مِنَ المدينةِ قالَ لَه أحدُهُما: ”أُنْجُ بنفْسِكَ. لا تَلتَفِتْ إلى ورائِكَ ولا تَقِفْ في السَّهْلِ كُلِّهِ، واَهرُبْ إلى الجَبلِ لِئلَّا تَهلِكَ“</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 13:6–8; اعمال الرسل 10:14; التَّكوين 32:26"> فقالَ لُوطٌ: ”لا يا سيّدي.</span>
<span className="v" id="v19" data-v="19" data-xref="صَمُوئيلَ الأوَّلُ 27:1; المَزامِير 103:1–22; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:14–16"> نِلْتُ رِضاكَ وغمرْتَني برحمتِكَ فأنقذتَ حياتي. ولكنّي لا أقدِرُ أنْ أهرُبَ إلى الجَبلِ، فرُبَّما لَحِقَني السُّوءُ فأموتُ.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 119:175; الأمثال 3:5–7; التَّكوين 19:30"> أمَّا تِلكَ المدينةُ فهيَ قريبةٌ وصغيرةٌ، فدَعْني أهربُ إليها، فأنجوَ لصِغَرِها بحياتي“.</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 145:19; المَزامِير 34:15; المَزامِير 102:17"> فقالَ لَهُ: ”إكرامًا لكَ لن أُدمِّرَ المدينةَ التي ذَكرْتَ.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 13:10; التَّكوين 14:2; التَّكوين 32:25–28"> أسرِعْ بالهرَبِ إلى هُناكَ، لأنِّي لن أفعَلَ شيئًا حتى تَصِلَ إليها“. ولذلِكَ سُمِّيتِ المدينةُ صُوغرَ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23"> فلمَّا أشرقتِ الشَّمسُ على الأرضِ ودَخلَ لُوطٌ مدينةَ صُوغرَ.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّثْنِيَة 29:23; إشَعْياء 13:19; عامُوس 4:11"> أمطرَ الرّبُّ على سدومَ وعَمورةَ كِبريتًا ونارًا مِنَ السَّماءِ،</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 107:34; التَّكوين 14:3; التَّكوين 13:10"> فدَمَّرَها معَ الوادي وجميعِ سُكَّانِ المُدُنِ ونباتِ الأرضِ.</span>
<span className="v" id="v26" data-v="26" data-xref="لُوقا 17:31–32; التَّكوين 19:17; الأمثال 14:14"> واَلْتفتتِ امرأةُ لُوطٍ إلى الوراءِ فصارت عَمودَ مِلحٍ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="حَبَقُّوق 2:1; المَزامِير 5:3; العِبرانِيّين 2:1"> وبكَّرَ إبراهيمُ في الغدِ إلى المكانِ الّذي وقفَ فيهِ أمامَ الرّبِّ،</span>
<span className="v" id="v28" data-v="28" data-xref="رُؤيا يُوحَنّا 18:9; رُؤيا يُوحَنّا 9:2; رُؤيا يُوحَنّا 21:8"> وتطلَّعَ إلى جِهةِ سدومَ وعَمورةَ والوادي كُلِّهِ، فرأى دُخانَ الأرضِ صاعدًا كدُخانِ الأتونِ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 8:1; التَّثْنِيَة 9:5; التَّكوين 18:23–33"> ولمَّا أهلكَ اللهُ مُدُنَ الوادي الّتي كانَ لُوطٌ يُقيمُ بها، تذَكَّرَ إبراهيمَ، فأخرجَ لُوطًا مِنْ وسَطِ الدَّمارِ.</span>
</div>
</div>

## أصل المؤابيين وبني عمون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 19:19; إشَعْياء 15:5; يَعقُوب 1:8"> وخافَ لُوطٌ أنْ يسكُنَ في صُوغرَ، فصعِدَ إلى الجبَلِ وأقامَ بالمغارةِ هوَ واَبنتاهُ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="التَّثْنِيَة 25:5; إشَعْياء 4:1; التَّكوين 16:2"> فقالتِ الكُبرى للصُّغرى: ”شاخَ أبونا وما في الأرضِ رَجلٌ يتزوَّجُنا على عادةِ أهلِ الأرضِ كُلِّهِم.</span>
<span className="v" id="v32" data-v="32" data-xref="اللّاويِّين 18:6–7; التَّكوين 11:3; التَّكوين 9:21"> تعالَي نسقي أبانا خمرًا ونضاجِعُهُ ونقيمُ مِنْ أبينا نسلًا“.</span>
<span className="v" id="v33" data-v="33" data-xref="حَبَقُّوق 2:15–16; الأمثال 23:29–35; الأمثال 20:1"> فسقتا أباهُما خمرًا تِلكَ اللَّيلةَ، وجاءتِ الكُبرى وضاجعت أباها وهوَ لا يَعلمُ بنيامِها ولا قيامِها.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="إرْمِيا 8:12; إرْمِيا 6:15; إشَعْياء 3:9"> وفي الغدِ قالتِ الكُبرى للصُّغرى: ”ضاجعتُ البارحةَ أبي، فلنَسْقِهِ خمرًا اللَّيلةَ أيضًا، وضاجعيهِ أنتِ لِنُقيمَ مِنْ أبينا نسلًا“.</span>
<span className="v" id="v35" data-v="35" data-xref="الأمثال 24:16; المَزامِير 8:4; رسالَةُ بُطرُسَ الأُولَى 4:7"> فسقتا أباهُما خمرًا تِلكَ اللّيلةَ أيضًا، وقامتِ الصُّغرى وضاجعَتْهُ وهوَ لا يَعلمُ بنيامِها ولا قيامِها.</span>
<span className="v" id="v36" data-v="36" data-xref="حَبَقُّوق 2:15; القُضاة 1:7; ﻣﺘﻰ 7:2"> فحملتِ اَبنتا لُوطٍ مِنْ أبيهما.</span>
<span className="v" id="v37" data-v="37" data-xref="التَّثْنِيَة 2:9; المُلُوكِ الثّانِي 3:1–27; العَدَد 21:29"> فولدتِ الكُبرى اَبنًا وسمَّتْهُ موأبَ، وهوَ أبو المؤابيِّينَ إلى اليومِ.</span>
<span className="v" id="v38" data-v="38" data-xref="التَّثْنِيَة 2:19; القُضاة 10:6–40; المَزامِير 83:4–8"> والصُّغرى أيضًا ولدتِ اَبنًا وسمَّتْهُ بنَ عمِّي، وهوَ أبو بَني عمُّونَ إلى اليومِ.</span>
</div>
