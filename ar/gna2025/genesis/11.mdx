## برج بابل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اعمال الرسل 2:6; صَفَنْيا 3:9; إشَعْياء 19:18"> وكانَ لأهلِ الأرضِ كُلِّها لُغَةٌ واحدةٌ وكلامٌ واحدٌ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 10:10; دانيال 1:2; التَّكوين 14:1"> فلمَّا رحَلوا مِنَ المَشرقِ وَجَدوا بُقعَةً في سَهلِ شِنْعارَ، فأقاموا هُناكَ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 14:10; الخُرُوج 1:14; الخُرُوج 2:3"> وقالَ بعضُهُم لِبعضٍ: ”تعالَوا نصنَع لِبْنًا ونَشْوِيهِ شيًّا“، فكانَ لهُمُ اللِّبْنُ بَدَلَ الحِجارَةِ والتُرابُ الأحمرُ بَدَلَ الطِّينِ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 1:28; يُوحَنّا 5:44; لُوقا 1:51"> وقالوا: ”تعالَوا نَبْنِ لنا مدينَةً وبُرجًا رأسُهُ في السَّماءِ. وَنُقِمْ لنا اَسمًا فلا نتَشتَّتُ على وجهِ الأرضِ كُلِّها“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 18:21; المَزامِير 33:13–14; يُوحَنّا 3:13"> ونَزَلَ الرّبُّ لِيَنظُرَ المدينَةَ والبُرجَ الّلذَينِ كانَ بَنو آدمَ يَبنونَهما،</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 2:1–4; التَّكوين 3:22; التَّكوين 11:1"> فقالَ الرّبُّ: ”ها هُم شعبٌ واحدٌ، ولهُم جميعًا لُغَةٌ واحدةٌ! ما هذا الّذي عَمِلوه إلَّا بِدايةً، ولن يصعُبَ علَيهم شيءٌ مِما يَنوونَ أنْ يعمَلوه!</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 33:10; المَزامِير 55:9; أيُّوب 5:12–13"> فلنَنزِلْ ونُبَلبِلْ هُناكَ لُغَتَهُم، حتّى لا يفهَمَ بعضُهُم لُغَةَ بعضٍ“.</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 1:51; التَّثْنِيَة 32:8; التَّكوين 10:25"> فشَتَّتَهُمُ الرّبُّ مِنْ هُناكَ على وجهِ الأرضِ كُلِّها، فكَفُّوا عَن بِناءِ المدينةِ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 10:10; التَّكوين 10:5; التَّكوين 10:20"> ولِهذا سُمِّيَت بابِلَ، لأنَّ الرّبَّ هُناكَ بَلبَلَ لُغَةَ النَّاسِ جميعًا، ومِنْ هُناكَ شَتَّتَهُمُ الرّبُّ على وجهِ الأرضِ كُلِّها.</span>
</div>
</div>

## مواليد سام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="أخبارِ الأيّامِ الأوّلُ 1:17–27; التَّكوين 10:21–22; لُوقا 3:34–36"> هؤلاءِ مَواليدُ سامٍ: لمَّا كانَ سامُ اَبنَ مئةِ سنَةٍ وَلَدَ أرفكْشادَ بَعدَ الطُّوفانِ بِسنَتينِ.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 9:7; المَزامِير 144:12; التَّكوين 1:28"> وعاشَ سامٌ بَعدَما وَلَدَ أرفَكْشادَ خَمْسَ مئَةِ سنَةٍ، ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="لُوقا 3:36"> وعاشَ أرفَكْشادُ خَمْسًا وثَلاثينَ سنَةً ووَلَدَ شالَحَ.</span>
<span className="v" id="v13" data-v="13"> وعاشَ أرفَكْشادُ بَعدَما وَلَدَ شَالَحَ أربَعَ مئةٍ وثَلاثَ سِنينَ، ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14"> وعاشَ شَالَحُ ثَلاثينَ سنَةً ووَلَدَ عابِرَ.</span>
<span className="v" id="v15" data-v="15"> وعاشَ شَالَحُ بَعدَما وَلَدَ عابِرَ أربَعَ مئةٍ وثَلاثَ سنِينَ ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 10:21; أخبارِ الأيّامِ الأوّلُ 1:19; العَدَد 24:24"> وعاشَ عابِرُ أربعًا وثلاثينَ سنَةً وَوَلدَ فالَجَ.</span>
<span className="v" id="v17" data-v="17"> وعاشَ عابِرُ بَعدَما وَلَدَ فالَجَ أربَعَ مئةٍ وثَلاثينَ سنَةً ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 3:35"> وعاشَ فالَجُ ثَلاثينَ سنَةً وَوَلدَ رَعُوَ.</span>
<span className="v" id="v19" data-v="19"> وعاشَ فالَجُ بعدَما وَلَد رَعُوَ مئتينِ وَتِسعَ سِنينَ ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="لُوقا 3:35"> وعاش رَعُوَ اَثنتَينِ وثَلاثينَ سنَةً ووَلَدَ سَرُوجَ.</span>
<span className="v" id="v21" data-v="21"> وعاشَ رَعُوَ بعدَما وَلَدَ سَرُوجَ مئتَينِ وَسَبْعَ سَنينَ ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="يَشُوع 24:2"> وعاشَ سَرُوجُ ثَلاثينَ سنَةً ووَلَدَ ناحورَ.</span>
<span className="v" id="v23" data-v="23"> وعاشَ سَرُوجُ بَعدَما وَلَدَ ناحورَ مئَتَي سنَةٍ، ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="لُوقا 3:34; يَشُوع 24:2"> وعاشَ ناحورُ تِسعًا وعِشْرينَ سنَةً ووَلَدَ تارَحَ.</span>
<span className="v" id="v25" data-v="25"> وعاشَ ناحورُ بعدَما وَلَدَ تارَحَ مئةً وتِسعَ عشْرَة سنةً، ووَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="يَشُوع 24:2; أخبارِ الأيّامِ الأوّلُ 1:26–27; التَّكوين 22:20–24"> وعاشَ تارَحُ سَبْعينَ سنَةً ووَلَدَ أبرامَ وناحورَ وهارَانَ.</span>
</div>
</div>

## مواليد تارح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التَّكوين 12:4; التَّكوين 14:12; التَّكوين 11:31"> وهؤلاءِ مَواليدُ تارَحَ: وَلَدَ تارَحُ أبرامَ وناحورَ وهارانَ. وهارانُ وَلَدَ لُوطًا.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 15:7; نَحَمْيا 9:7; اعمال الرسل 7:2–4"> وماتَ هارانُ قبلَ تارَحَ أبيهِ في أرضِ ميلادِه، في أورِ الكَلدانيِّينَ.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّكوين 17:15; التَّكوين 22:20; التَّكوين 24:15"> وتَزوَّجَ أبرامُ وناحورُ اَمْرَأتَينِ، اَسمُ اَمْرَأةِ أبرامَ سارايُ، واَسمُ اَمرأةِ ناحورَ مَلْكَةُ بنتُ هارانَ أبي مَلْكَةَ وأبي يِسْكَةَ.</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 29:31; التَّكوين 21:1–2; لُوقا 1:7"> وكانَت سارايُ عاقِرًا لا وَلَدَ لها.</span>
<span className="v" id="v31" data-v="31" data-xref="نَحَمْيا 9:7; اعمال الرسل 7:2–4; التَّكوين 24:10"> وأخَذَ تارَحُ أبرامَ اَبنَهُ، ولوطًا بنَ هارانَ حَفيدَهُ، وسارايَ كَنَّتَه، اَمْرأةَ أبرامَ اَبْنِه، فخَرَجَ معَهُم مِنْ أُورِ الكَلدانيِّينَ ليذهَبوا إلى أرضِ كنعانَ، فجاؤوا إلى حارانَ وأقاموا هُناك،</span>
<span className="v" id="v32" data-v="32"> وكانَ عُمْرُ تارَحَ مئَتينِ وخَمْسَ سِنينَ. وماتَ تارَحُ في حارانَ.</span>
