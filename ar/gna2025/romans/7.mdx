## التحرّر من الشَّريعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﺭﻭﻣﻴﺔ 6:14; ﺭﻭﻣﻴﺔ 7:6; ﺭﻭﻣﻴﺔ 6:3"> أنتُم لا تَجهَلونَ أيُّها الإخوةُ، وأنا أكَلِّمُ جماعَةً يَعرِفونَ الشَّريعةَ، أنْ لا سُلطَةَ لِلشَّريعةِ على الإنسانِ إلَّا وهوَ حيٌّ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 2:23–24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:39; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:4"> فالمرأةُ المُتزوِّجَةُ تَربُطُها الشَّريعةُ بالرَّجُلِ ما دامَ حيًّا، فإذا ماتَ تَحَرَّرَتْ مِنْ رِباطِ الشَّريعةِ هذا.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 20:14; ﻣﺘﻰ 5:32; مَرْقُس 10:6–12"> وإنْ صارَتْ إلى رَجُلٍ آخَرَ وزَوجُها حَيٌّ، فَهيَ زانيةٌ. ولكِنْ إذا ماتَ زَوجُها تَحرَّرَتْ مِنَ الشَّريعةِ، فلا تكونُ زانِيةً إنْ صارَتْ إلى رَجُلٍ آخَرَ.</span>
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:24; غَلاطِيَّة 5:22–23; إشَعْياء 54:5"> وهكذا أنتُم أيُّها الإخوةُ، مُتُّمْ عَنِ الشَّريعةِ بِجَسدِ المَسيحِ لتَصيروا إلى آخَرَ، إلى الّذي قامَ مِنْ بَينِ الأمواتِ، فتُثمِروا للهِ.</span>
<span className="v" id="v5" data-v="5" data-xref="أفَسُس 2:3; غَلاطِيَّة 5:16–17; غَلاطِيَّة 5:24"> فحينَ كُنّا نَحيا حياةَ الجسَدِ، كانَتِ الأهْواءُ الشِّرِّيرَةُ الّتي أثارَتْها الشَّريعةُ تَعمَلُ في أعضائِنا لتُثمِرَ لِلموتِ.</span>
<span className="v" id="v6" data-v="6" data-xref="غَلاطِيَّة 3:23–25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:6; ﺭﻭﻣﻴﺔ 7:4"> ولكنَّنا الآنَ تَحَرَّرْنا مِنَ الشَّريعةِ، لأنَّنا مُتْنا عَمّا كانَ يُقَيِّدُنا، حتّى نَعبُدَ اللهَ في نِظامِ الرُّوحِ الجديدِ، لا في نِظامِ الحَرْفِ القديمِ.</span>
</div>
</div>

## الشَّريعة والخطيئة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﺭﻭﻣﻴﺔ 3:20; ﺭﻭﻣﻴﺔ 4:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:56"> فماذا نَقولُ؟ أتكونُ الشَّريعةُ خَطيئَةً؟ كلّا! ولكنِّي ما عَرَفتُ الخَطيئَةَ إلَّا بالشَّريعةِ. فلولا قَولُها لي: ”لا تَشْتَهِ!“ لما عرَفْتُ الشَّهوَةَ.</span>
<span className="v" id="v8" data-v="8" data-xref="ﺭﻭﻣﻴﺔ 7:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:56; ﺭﻭﻣﻴﺔ 4:15"> ولكِنَّ الخَطيئَةَ وجَدَتْ في هذِهِ الوصيَّةِ فُرصَةً لتُثيرَ فيَّ كُلَّ شَهوَةٍ، لأنَّ الخَطيئَةَ بِلا شَريعةٍ مَيتةٌ.</span>
<span className="v" id="v9" data-v="9" data-xref="غَلاطِيَّة 3:10; ﺭﻭﻣﻴﺔ 8:7; يَعقُوب 2:10–11"> كُنتُ أحيا مِنْ قَبلُ بِلا شريعةٍ، فلمَّا جاءَتِ الوصيَّةُ، عاشَتِ الخَطيئَةُ ومُتُّ أنا.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:7; ﺭﻭﻣﻴﺔ 10:5; اللّاويِّين 18:5"> فإذا بالوصيَّةِ الّتي هيَ لِلحياةِ، قادَتْني أنا إلى الموتِ،</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 3:13; ﺭﻭﻣﻴﺔ 7:8; العِبرانِيّين 3:13"> لأنَّ الخَطيئَةَ اَتَّخَذَتْ مِنَ الوصيَّةِ سَبيلًا، فخَدَعَتْني بِها وقَتَلَتْني.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:8; المَزامِير 119:137; المَزامِير 19:7–12"> الشَّريعةُ ذاتُها، إذًا، مُقَدَّسةٌ، والوَصِيَّةُ، مُقَدَّسَةٌ وعادِلَةٌ وصالِحةٌ.</span>
<span className="v" id="v13" data-v="13" data-xref="يَعقُوب 1:13–15; غَلاطِيَّة 3:21; ﺭﻭﻣﻴﺔ 8:3"> فهَلْ صارَ الصَّالِحُ سَببًا لِموتي؟ كلّا! بَلْ هيَ الخَطيئَةُ تَذَرَّعَتْ بالصَّالِحِ فعَمِلَتْ لِمَوتي حتّى تَظهَرَ أنَّها خَطيئَةٌ، وتذَرَّعتْ بالوصيَّةِ حتّى تَبلُغَ أقصى حُدودِ الخَطيئَةِ.</span>
</div>
</div>

## صراع الإنسان بينه وبين نفسه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="المُلُوكِ الأوَّلُ 21:25; المُلُوكِ الثّانِي 17:17; ﺭﻭﻣﻴﺔ 7:18"> ونَحنُ نَعرِفُ أنَّ الشَّريعةَ روحِيَّةٌ، ولكِنِّي بَشَرٌ بِيعَ عَبدًا لِلخَطيئَةِ:</span>
<span className="v" id="v15" data-v="15" data-xref="غَلاطِيَّة 5:17; ﺭﻭﻣﻴﺔ 7:19–20; الجامِعَة 7:20"> لا أفهَمُ ما أعمَلُ، لأنَّ ما أُريدُه لا أعمَلُهُ، وما أكرَهُهُ أعمَلُهُ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 7:12; ﺭﻭﻣﻴﺔ 7:14; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:8"> وحينَ أعمَلُ ما لا أُريدُهُ، أوافِقُ الشَّريعةَ على أنَّها حقٌ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 7:20; فِيلِبِّي 3:8–9; ﺭﻭﻣﻴﺔ 7:18"> فلا أكونُ أنا الّذي يعمَلُ ما لا يُريدُهُ، بَلِ الخَطيئَةُ الّتي تَسكُنُ فيَّ،</span>
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 3:6; المَزامِير 51:5; غَلاطِيَّة 5:17"> لأنِّي أعلَمُ أنَّ الصّلاحَ لا يَسكُنُ فيَّ، أي في جسَدي. فإرادةُ الخَيرِ هِيَ بإِمكاني، وأمَّا عمَلُ الخَيرِ فلا.</span>
<span className="v" id="v19" data-v="19" data-xref="ﺭﻭﻣﻴﺔ 7:15"> فالخَيرُ الّذي أُريدُهُ لا أعمَلُهُ، والشَّرُّ الّذي لا أُريدُهُ أعمَلُه.</span>
<span className="v" id="v20" data-v="20" data-xref="ﺭﻭﻣﻴﺔ 7:17"> وإذا كُنتُ أعمَلُ ما لا أُريدُهُ، فما أنا الّذي يَعمَلُه، بَلِ الخَطيئَةُ الّتي تَسكُنُ فيَّ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="ﺭﻭﻣﻴﺔ 6:14; ﺭﻭﻣﻴﺔ 7:23; المَزامِير 119:133"> وهكذا أجِدُ أنِّي في حُكْمِ هذِهِ الشَّريعةِ، وهيَ أنِّي أُريدُ أنْ أعمَلَ الخَيرَ ولكِنَّ الشَّرَّ هوَ الّذي بإمكاني.</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 1:2; المَزامِير 40:8; المَزامِير 119:35"> وأنا في أعماقِ كِياني أبتَهِجُ بِشريعةِ اللهِ،</span>
<span className="v" id="v23" data-v="23" data-xref="غَلاطِيَّة 5:17; يَعقُوب 4:1; ﺭﻭﻣﻴﺔ 8:2"> ولكنِّي أشعُرُ بِشريعَةٍ ثانِيَةٍ في أعضائي تُقاوِمُ الشَّريعةَ الّتي يُقِرُّها عَقلي وتَجعَلُني أسيرًا لِشريعةِ الخَطيئَةِ الّتي هِيَ في أعضائي.</span>
<span className="v" id="v24" data-v="24" data-xref="ﺭﻭﻣﻴﺔ 6:6; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:18; تِيطُس 2:14"> ما أتعسني أنا الإنسانُ! فمَنْ يُنَجِّيني مِنْ جَسَدِ الموتِ هذا؟</span>
<span className="v" id="v25" data-v="25" data-xref="غَلاطِيَّة 5:17–24; كُولُوسِي 3:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:9–10"> الحمدُ للهِ بربِنا يَسوعَ المَسيحِ. فأنا بالعَقلِ أخضَعُ لِشريعةِ اللهِ، وبالجسَدِ لِشريعةِ الخَطيئَةِ.</span>
</div>
