## إعملوا ما يرضي الآخرين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:14; ﺭﻭﻣﻴﺔ 14:1; غَلاطِيَّة 6:1–2"> فعَلَينا نَحنُ الأقوِياءَ في الإيمانِ أنْ نَحتَمِلَ ضُعفَ الضُّعفاءِ، ولا نَطلُبُ ما يُرضي أنفُسَنا،</span>
<span className="v" id="v2" data-v="2" data-xref="ﺭﻭﻣﻴﺔ 14:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:24; فِيلِبِّي 2:4–5"> بَلْ ليَعمَلْ كُلُّ واحدٍ مِنَّا ما يُرضي أخاهُ لِخيرِ البُنيانِ المُشَتَركِ.</span>
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 69:9; فِيلِبِّي 2:8; يُوحَنّا 4:34"> وما طلَبَ المَسيحُ ما يُرضي نَفسَهُ، بَلْ كما جاءَ في الكِتابِ: ”شتائِمُ الّذينَ يَشتُمونَكَ وقَعَتْ علَيَّ“.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:11; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:16–17; ﺭﻭﻣﻴﺔ 4:23–24"> وكُلُّ ما جاءَ قَبلًا في الكُتُبِ المُقَدَّسَةِ إنَّما جاءَ ليُعَلِّمَنا كيفَ نَحصلُ على الرَّجاءِ بِما في هذِهِ الكُتُبِ مِنَ الصَّبرِ والعَزاءِ.</span>
<span className="v" id="v5" data-v="5" data-xref="فِيلِبِّي 2:4–5; رسالَةُ بُطرُسَ الأُولَى 3:8; ﺭﻭﻣﻴﺔ 12:16"> فليُعطِكُم إلهُ الصَّبرِ والعَزاءِ اتِّفاقَ الرَأيِ في ما بَينَكُم كما عَلَّمَنا المَسيحُ يَسوعُ،</span>
<span className="v" id="v6" data-v="6" data-xref="يُوحَنّا 10:29–30; رُؤيا يُوحَنّا 1:6; أفَسُس 1:3"> لِتُمَجِّدوا اللهَ أبا رَبِّنا يَسوعَ المَسيحِ بِقَلبٍ واحدٍ ولِسانٍ واحدٍ.</span>
</div>
</div>

## التبشير بين غير اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 10:40; يُوحَنّا 13:34; ﺭﻭﻣﻴﺔ 5:2"> فاَقبَلوا بَعضُكُم بَعضًا لِمَجدِ اللهِ كما قَبِلكُمُ المَسيحُ.</span>
<span className="v" id="v8" data-v="8" data-xref="غَلاطِيَّة 4:4–5; ﻣﺘﻰ 15:24; يُوحَنّا 1:11"> وأنا أقولُ لكُم إنَّ المَسيحَ صارَ خادِمَ اليَهودِ ليُظهِرَ أنَّ اللهَ صادِقٌ ويَفِيَ بِما وعَدَ بِه الآباءَ.</span>
<span className="v" id="v9" data-v="9" data-xref="صموئيلَ الثّانِي 22:50; المَزامِير 18:49; ﺭﻭﻣﻴﺔ 3:29"> أمَّا غَيرُ اليَهودِ فيُمجِّدونَ اللهَ على رَحمَتِهِ، كما جاءَ في الكِتابِ: ”لذلِكَ أُسبِّحُ بِحَمدِكَ بَينَ الأُمَمِ، وأُرتِّلُ لاَسمِكَ“.</span>
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 67:3–4; التَّثْنِيَة 32:43; إشَعْياء 42:10–12"> وجاءَ أيضًا: ”إفرَحي أيَّتُها الأُمَمُ معَ شَعبِ اللهِ!“</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 117:1"> وأيضًا: ”سَبِّحوا الرَّبَّ يا جميعَ الأُمَمِ! مَجِّدوهُ يا جميعَ الشُّعوبِ!“</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 11:10; إشَعْياء 11:1; ﻣﺘﻰ 12:21"> وقالَ إشَعْيا: ”سيَظهَرُ فَرعٌ مِنْ أصلِ يَسَّى، يَقومُ لِيَسودَ الأُمَمَ وعلَيهِ يكونُ رَجاءُ الشُّعوبِ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:16–17; ﺭﻭﻣﻴﺔ 12:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:8"> فلْيغمُرْكُم إلهُ الرَّجاءِ بالفرَحِ والسَّلامِ في الإيمانِ، حتّى يَفيضَ رَجاؤُكُم بِقُدرَةِ الرُّوحِ القُدُسِ!</span>
</div>
</div>

## خدمة بولس الرسولية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="كُولُوسِي 3:16; رسالَةُ بُطرُسَ الثّانيَة 1:12; رسالَةُ يُوحَنّا الأُولَى 2:21"> فأنا على ثِقَةٍ يا إخوتي بأنَّكُم مُمتَلِئونَ خَيرًا، وأنَّكُم فائِضونَ بالمَعرِفَةِ، قادِرونَ على أنْ يَنصَحَ بَعضُكُم بَعضًا.</span>
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 12:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:10; ﺭﻭﻣﻴﺔ 1:5"> ولكِنِّي كَتَبتُ إلَيكُم في بَعضِ الأُمورِ بِكَثيرٍ مِنَ الجُرأَةِ لأنبِّهَكُم إلَيها، وذلِكَ لِلنِّعمَةِ الّتي وهَبَها اللهُ لي</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 11:13; ﺭﻭﻣﻴﺔ 1:1; اعمال الرسل 9:15"> حتّى أكونَ خادِمَ المَسيحِ يَسوعَ عِندَ غَيرِ اليَهودِ. وأنا أخدُمُ بِشارَةَ اللهِ ككاهِنٍ، فيَصيرُ غَيرُ اليَهودِ قُربانًا مَقبولًا عِندَ اللهِ مُقَدَّسًا بالرُّوحِ القُدُسِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:14–16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:11–21; العِبرانِيّين 5:1"> ويَحِقُّ لي، إذًا، أنْ أفتَخِرَ في المَسيحِ يَسوعَ بِخِدمَتي للهِ،</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 15:12; ﺭﻭﻣﻴﺔ 1:5; اعمال الرسل 21:19"> لأنِّي لا أجرُؤُ أنْ أتكَلَّمَ إلَّا بِما عَمِلَهُ المَسيحُ على يَدي لِهِدايَةِ غَيرِ اليَهودِ إلى طاعَةِ اللهِ، بالقَولِ والفِعلِ</span>
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:12; العِبرانِيّين 2:4; اعمال الرسل 19:11–12"> وبِقُوَّةِ الآياتِ والمُعجِزاتِ وبِقُدرَةِ رُوحِ اللهِ. مِنْ أُورُشليمَ وما حَولَها إلى اللِّيريكُونَ أكمَلْتُ التَّبشيرَ بالمَسيحِ.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:13–16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:9–15; أفَسُس 2:20–22"> وكُنتُ حَريصًا أنْ لا أُبَشِّرَ حَيثُ سَمِعَ النَّاسُ باَسمِ المَسيحِ، لِئَلَّا أبنِيَ على أساسِ غَيري،</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 52:15; إشَعْياء 65:1"> فعَمِلتُ بِما جاءَ في الكِتابِ: ”الّذينَ ما بَشَّرَهُم بِه أحدٌ سيُبصِرونَ، والّذينَ ما سَمِعوا بِه سَيَفهمونَ“.</span>
</div>
</div>

## رغبة بولس في زيارة رومة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="ﺭﻭﻣﻴﺔ 1:13; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:17–18"> وهذا ما مَنَعَني مِرارًا مِنَ المَجيءِ إلَيكُم.</span>
<span className="v" id="v23" data-v="23" data-xref="ﺭﻭﻣﻴﺔ 15:32; ﺭﻭﻣﻴﺔ 1:10–12; ﺭﻭﻣﻴﺔ 15:29"> أمَّا الآنَ ولا مَجالَ عمَلٍ لي بَعدُ في هذِهِ الأقطارِ، ولي مِنْ عِدَّةِ سِنينَ شَوقٌ إلى المَجيءِ إلَيكُم،</span>
<span className="v" id="v24" data-v="24" data-xref="ﺭﻭﻣﻴﺔ 15:28; ﺭﻭﻣﻴﺔ 1:12; اعمال الرسل 15:3"> فأنا أرجو أن أراكُم عِندَ مُروري بِكُم في طريقي إلى إسبانِيَةَ وأستَعينَ بِكُم على السَّفَرِ إلَيها، بَعدَ أنْ أنعَمَ ولو قَليلًا بِلقائِكُم.</span>
<span className="v" id="v25" data-v="25" data-xref="اعمال الرسل 24:17; اعمال الرسل 19:21; اعمال الرسل 20:22"> ولكِنِّي الآنَ ذاهِبٌ إلى أُورُشليمَ في خِدمَةِ الإخوَةِ القِدِّيسينَ،</span>
<span className="v" id="v26" data-v="26" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:1–9; اعمال الرسل 11:27–30; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:2"> لأنَّ كنائِسَ مكِدونِيَّةَ وأخائِيَةَ اَستَحسَنوا أنْ يَتبَرَّعوا بِبَعضِ المَعونةِ لِلمُحتاجينَ مِنَ الإخوَةِ القِدِّيسينَ الّذينَ في أُورُشليمَ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:11; غَلاطِيَّة 6:6; فِلِيْمُون 1:19"> هُمُ اَستَحسَنوا ذلِكَ وهوَ حَقٌّ علَيهِم، لأنَّهُم شارَكوا أبناءَ سائِرِ الشُّعوبِ في خَيراتِهِمِ الرُّوحِيَّةِ فكانَ على هَؤُلاءِ أنْ يَخدِموهُم بِخيراتِهِمِ المادِيَّةِ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﺭﻭﻣﻴﺔ 15:24; فِيلِبِّي 4:17; كُولُوسِي 1:6"> وبَعدَ أنْ أقومَ بِهذِهِ المهمَّةِ وأُسلِّمَ إلَيهِم تِلكَ المَعونَةَ، أَمُرُّ بِكُم وأنا في طريقي إلى إسبانيَةَ.</span>
<span className="v" id="v29" data-v="29" data-xref="أفَسُس 4:13; ﺭﻭﻣﻴﺔ 1:10–12; المَزامِير 16:11"> وأعلمُ أنِّي إذا جِئتُ إلَيكم أجيءُ بِمِلءِ بَركَةِ المَسيحِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="كُولُوسِي 4:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:11"> فأُناشِدُكُم، أيُّها الإخوةُ، باَسمِ رَبِّنا يَسوعَ المَسيحِ وبِمَحبَّةِ الرُّوحِ، أنْ تُجاهِدوا مَعي بِرَفعِ صَلواتِكُم إلى اللهِ، مِنْ أجلي،</span>
<span className="v" id="v31" data-v="31" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:1"> ليُنقِذَني مِنْ أيدي الخارِجينَ على الإيمانِ في اليَهودِيَّةِ ويَجعلَ خِدمَتي في أُورُشليمَ مَقبولَةً عِندَ الإخوةِ القِدِّيسينَ،</span>
<span className="v" id="v32" data-v="32" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:18; فِلِيْمُون 1:20; فِلِيْمُون 1:7"> فأجيءَ إلَيكُم مَسرورًا إنْ شاءَ اللهُ، وأرتاحَ عِندَكُم.</span>
<span className="v" id="v33" data-v="33" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:11; فِيلِبِّي 4:9; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:23"> وليَكُنْ إلهُ السَّلامِ مَعكُم أجمعينَ. آمين.</span>
