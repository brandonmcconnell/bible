## الله ومختاروه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:7; ﺭﻭﻣﻴﺔ 1:9; غَلاطِيَّة 1:20"> أقولُ الحقَّ في المَسيحِ ولا أكذِبُ. فضَميري شاهِدٌ لي في الرُّوحِ القُدُسِ</span>
<span className="v" id="v2" data-v="2" data-xref="فِيلِبِّي 3:18; المَزامِير 119:136; إرْمِيا 13:17"> أنِّي حَزينٌ جدًّا وفي قلبي ألَمٌ لا يَنقَطِعُ،</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 32:32; أسْتِير 8:6; ﺭﻭﻣﻴﺔ 11:14"> وأنِّي أتمَنّى لَو كُنتُ أنا ذاتي مَحرومًا ومُنفصِلًا عَنِ المَسيحِ في سَبيلِ إخوَتي بَني قَومي في الجسَدِ.</span>
<span className="v" id="v4" data-v="4" data-xref="أفَسُس 2:12; المَزامِير 147:19; ﺭﻭﻣﻴﺔ 9:6"> هُمْ بَنو إِسرائيلَ الّذينَ جَعَلَهُمُ اللهُ أبناءَهُ، ولهُمُ المَجدُ والعُهودُ والشَّريعةُ والعِبادَةُ والوُعودُ،</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 10:30; يُوحَنّا 1:1–3; كُولُوسِي 1:16–19"> ومِنهُم كانَ الآباءُ وجاءَ المَسيحُ في الجسَدِ، وهوَ الكائِنُ على كُلِّ شيءٍ إلهًا مُباركًا إلى الأبَدِ. آمين.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="ﺭﻭﻣﻴﺔ 2:28–29; غَلاطِيَّة 6:16; ﺭﻭﻣﻴﺔ 11:1–2"> ولا أقولُ إنَّ وَعدَ اللهِ خابَ. فما كُلُّ بَني إِسرائيلَ هُمْ إِسرائيلُ،</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 21:12; العِبرانِيّين 11:18; يُوحَنّا 8:37–39"> ولا كُلُّ الّذينَ مِنْ نَسلِ إبراهيمَ هُمْ أبناءُ إبراهيمَ. قالَ اللهُ لإبراهيمَ: ”بإسحقَ يكونُ لكَ نَسلٌ“.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 1:13; ﺭﻭﻣﻴﺔ 4:11–16; غَلاطِيَّة 3:26–29"> فما أبناءُ الجسَدِ هُم أبناءُ اللهِ، بَلْ أبناءُ الوَعدِ هُمُ الّذينَ يَحسُبُهُمُ اللهُ نَسلَ إبراهيمَ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 18:10; التَّكوين 18:14; التَّكوين 17:21"> فكلامُ الوَعدِ هوَ هذا: ”سأعودُ في مِثلِ هذا الوَقتِ، ويكونُ لِسارَةَ اَبنٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 25:21–23; لُوقا 16:26; ﺭﻭﻣﻴﺔ 5:11"> وما هذا كُلُّ شيءٍ، بَلْ إنَّ رفقَةَ حَبِلَتْ مِنْ رَجُلٍ واحدٍ، مِنْ أبينا إسحقَ،</span>
<span className="v" id="v11" data-v="11" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:9; أفَسُس 1:4–5; رسالَةُ بُطرُسَ الثّانيَة 1:10"> وقَبلَ أنْ يولَدَ الصَّبيّانِ ويَعمَلا خَيرًا أو شَرًّا، ولِيَتِمَّ ما اَختارَهُ اللهُ بِتَدبيرِهِ القائِمِ على دَعوَتِهِ لا على الأعمالِ،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 25:22–23; صموئيلَ الثّانِي 8:14; المُلُوكِ الأوَّلُ 22:47"> قالَ اللهُ لِرفقَةَ: ”الأكبَرُ يَستعبدُهُ الأصغَرُ“،</span>
<span className="v" id="v13" data-v="13" data-xref="مَلاخِي 1:2–3; التَّثْنِيَة 21:15; التَّكوين 29:33"> على ما ورَدَ في الكِتابِ: ”أحبَبتُ يَعقوبَ وأبغَضتُ عِيسو“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="أخبارِ الأيّامِ الثّانِي 19:7; المَزامِير 145:17; التَّثْنِيَة 32:4"> فماذا نَقولُ؟ أيكونُ عِندَ اللهِ ظُلمٌ؟ كلَّا!</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 33:19; ﺭﻭﻣﻴﺔ 9:16; ميخا 7:18"> قالَ اللهُ لِموسى: ”أرحَمُ مَنْ أرحَمُ، وأُشفِقُ على مَنْ أُشفِقُ“.</span>
<span className="v" id="v16" data-v="16" data-xref="أفَسُس 2:8; ﻣﺘﻰ 11:25–26; رسالَةُ بُطرُسَ الأُولَى 2:9–10"> فالأمرُ لا يَعودُ إلى إرادةِ الإنسانِ ولا إلى سَعيِهِ، بَلْ إلى رَحمَةِ اللهِ وحدَها.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 9:16; الأمثال 16:4; الخُرُوج 14:17–18"> ففي الكِتابِ قالَ اللهُ لِفرعونَ: ”رفَعتُكَ لأُظهِرَ فيكَ قُدرَتي ويَدعوَ النَّاسُ باَسمي في الأرضِ كُلِّها“.</span>
<span className="v" id="v18" data-v="18" data-xref="يَشُوع 11:20; الخُرُوج 4:21; ﺭﻭﻣﻴﺔ 9:15–16"> فهوَ إذًا يَرحَمُ مَنْ يَشاءُ ويُقسّي قَلبَ مَنْ يَشاءُ.</span>
</div>
</div>

## غضب الله ورحمته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="دانيال 4:35; أخبارِ الأيّامِ الثّانِي 20:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:12"> ويَقولُ لي أحدُكُم: ”فلِماذا يَلومُنا اللهُ؟ مَنْ يَقدِرُ أنْ يُقاوِمَ مَشيئَتَه؟“</span>
<span className="v" id="v20" data-v="20" data-xref="إشَعْياء 29:16; إشَعْياء 45:9–11; إشَعْياء 64:8"> فأُجيبُ: مَنْ أنتَ أيُّها الإنسانُ حتّى تَعتَرِضَ على اللهِ؟ أيقولُ المَصنوعُ لِلصَّانِعِ: لِماذا صَنعتَني هكذا؟</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 64:8; الأمثال 16:4; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:20–21"> أما يَحِقُّ لِلخَزّافِ أنْ يَستَعْمِلَ طِينَهُ كما يَشاءُ، فيَصنَع مِنْ جَبلَةِ الطّينِ نَفسِها إناءً لاَستِعمالٍ شَريفٍ، وإناءً آخرَ لاَستِعمالٍ دَنيءٍ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الأمثال 16:4; يَهُوذا 1:4; ﺭﻭﻣﻴﺔ 9:21"> وكذلِكَ اللهُ، شاءَ أنْ يُظهِرَ غَضبَهُ ويُعلِنَ قُدرتَهُ، فاَحتَمَلَ بِصَبرٍ طويلٍ آنيةَ النَّقمَةِ الّتي لِلهلاكِ</span>
<span className="v" id="v23" data-v="23" data-xref="ﺭﻭﻣﻴﺔ 2:4; رسالَةُ بُطرُسَ الأُولَى 1:2–5; كُولُوسِي 1:27"> كما شاءَ أنْ يُعلِنَ فَيضَ مَجدِهِ في آنِيَةِ الرَّحمَةِ الّتي سبَقَ فأعَدَّها لِلمَجدِ،</span>
<span className="v" id="v24" data-v="24" data-xref="كُولُوسِي 3:11; اعمال الرسل 13:47–48; أفَسُس 3:6–8"> أي نَحنُ الّذينَ دَعاهُم لا مِنْ بَينِ اليَهودِ وحدَهُم، بَلْ مِنْ بَينِ سائِرِ الشُّعوبِ أيضًا.</span>
<span className="v" id="v25" data-v="25" data-xref="هُوشَع 2:23; رسالَةُ بُطرُسَ الأُولَى 2:10; يُوحَنّا 16:27"> وفي كِتابِ هوشَعَ أنَّ اللهَ قالَ: ”الّذي ما كانَ شَعبي سَأدعوهُ شَعبي، والّتي ما كانَتْ مَحبوبَتي سأَدعوها مَحبوبَتي،</span>
<span className="v" id="v26" data-v="26" data-xref="هُوشَع 1:9–10; رسالَةُ يُوحَنّا الأُولَى 3:1–3; غَلاطِيَّة 3:26"> وحَيثُ قِيلَ لهُم: ما أنتُم شعبي، يُدعَونَ أبناءَ اللهِ الحَيِّ“.</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 10:20–23; ﺭﻭﻣﻴﺔ 11:4–6; إشَعْياء 1:9"> ويكتُبُ إشَعْيا في كلامِهِ على إِسرائيلَ: ”وإنْ كانَ بَنو إِسرائيلَ عدَدَ رَملِ البحرِ، فلا يَخلُصُ مِنهُم إلَّا بَقِيَّةٌ،</span>
<span className="v" id="v28" data-v="28" data-xref="إشَعْياء 10:23; رُؤيا يُوحَنّا 19:11; اعمال الرسل 17:31"> لأنَّ الرَّبَّ سيَقضي في الأرضِ قَضاءً كاملًا سَريعًا عادِلًا.</span>
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 1:9; إشَعْياء 13:19; عامُوس 4:11"> وبِهذا أنبَأَ إشَعْيا فقَالَ: لَولا أنَّ رَبَّ الجُنودِ حَفِظَ لَنا نَسلًا، لَصِرْنا مثلَ سَدومَ وأشبهنا عَمورَةَ“.</span>
</div>
</div>

## إسرائيل والمسيح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="فِيلِبِّي 3:9; العِبرانِيّين 11:7; غَلاطِيَّة 2:16"> فماذا نقولُ؟ نَقولُ إنَّ الأُممَ الّذين ما سَعَوْا إلى البِرِّ تَبَرَّروا ولكِنْ بالإيمانِ،</span>
<span className="v" id="v31" data-v="31" data-xref="ﺭﻭﻣﻴﺔ 11:7; ﺭﻭﻣﻴﺔ 3:20; يَعقُوب 2:10–11"> أمَّا بَنو إِسرائيلَ الّذينَ سعَوا إلى شريعةٍ غايَتُها البِرُّ فَشِلوا في بُلوغِ غايَةِ الشَّريعةِ.</span>
<span className="v" id="v32" data-v="32" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:23; رسالَةُ بُطرُسَ الأُولَى 2:8; لُوقا 7:23"> ولِماذا؟ لأنَّهُم سَعَوا إلى هذا البِرِّ بالأعمالِ الّتي تَفرِضُها الشَّريعةُ لا بالإيمانِ، فصَدَموا حَجَرَ العَثرةِ،</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 28:16; إشَعْياء 8:14–15; ﺭﻭﻣﻴﺔ 10:11"> كما يقولُ الكِتابُ: ”ها أنا أضعُ في صِهيونَ حجَرَ عَثرَةٍ في طريقِ الشَّعبِ وصَخرَةَ سُقوطٍ، فمَنْ آمَنَ بِه لا يَخيب“.</span>
</div>
