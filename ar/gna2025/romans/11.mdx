## هل نبذ الله إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:22; صَمُوئيلَ الأوَّلُ 12:22; فِيلِبِّي 3:5"> لكِنِّي أقولُ: هَلْ نبَذَ اللهُ شَعبَهُ؟ كلَّا! فأنا نَفسي مِنْ بَني إِسرائيلَ، مِنْ نَسلِ إبراهيمَ وعَشيرَةِ بِنيامينَ.</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 94:14; ﺭﻭﻣﻴﺔ 8:29–30; الخُرُوج 32:1"> ما نبَذَ اللهُ شَعبَهُ وهوَ الّذي سبَقَ فاَختارَهُ. وأنتُم تَعرِفونَ ما قالَ الكِتابُ في إيليَّا حينَ شَكا بَني إِسرائيلَ إلى اللهِ، فقَالَ:</span>
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الأوَّلُ 19:10–18; المُلُوكِ الأوَّلُ 18:13; نَحَمْيا 9:26"> ”يا ربُّ، قَتَلوا أنبياءَكَ وهَدَموا كُلَّ مَذابِحِكَ وبَقِيتُ أنا وحدي، وهُم يُريدونَ أنْ يَقتُلوني“.</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الأوَّلُ 19:18; العَدَد 25:3; صَفَنْيا 1:4"> فَماذا أجابَهُ صوتُ اللهِ؟ أجابَهُ: ”أبْقَيتُ سَبعةَ آلافِ رَجُلٍ ما حَنَوا رُكبَةً لِبَعلٍ“.</span>
<span className="v" id="v5" data-v="5" data-xref="ﺭﻭﻣﻴﺔ 9:27; أفَسُس 1:5–6; ﺭﻭﻣﻴﺔ 9:11"> وفي الزَّمَنِ الحاضِرِ أيضًا بَقِيَّةٌ مِنَ النَّاسِ اَختارَها اللهُ بالنِّعمةِ.</span>
<span className="v" id="v6" data-v="6" data-xref="غَلاطِيَّة 2:21; ﺭﻭﻣﻴﺔ 4:4–5; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:9"> فإذا كانَ الاختِيارُ بالنِّعمةِ، فما هوَ إذًا بالأعمالِ، وإلَّا لما بَقِيَتِ النِّعمةُ نِعمةً.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﺭﻭﻣﻴﺔ 9:18; ﺭﻭﻣﻴﺔ 11:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:4"> فماذا بَعدُ؟ ما كانَ يَطلُبُهُ بنو إِسرائيلَ ولا يَنالونَهُ، نالَهُ الّذينَ اَختارَهُمُ اللهُ. أمَّا الباقونَ فقَسَتْ قُلوبُهُم،</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 29:4; إشَعْياء 29:10; حِزْقِيال 12:2"> كما جاءَ في الكِتابِ: ”أعطاهُمُ اللهُ عَقلًا خامِلًا وعُيونًا لا تُبصِرُ وآذانًا لا تَسمَعُ إلى هذا اليومِ“.</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 69:22–23; لُوقا 12:20; المَزامِير 28:4"> وقالَ داودُ: ”لِتكُنْ مَوائِدُهُم فخًّا لهُم وشَرَكًا وحجَرَ عَثرةٍ وعِقابًا.</span>
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 69:23; التَّثْنِيَة 28:64–68; رسالَةُ بُطرُسَ الثّانيَة 2:4"> لتُظلِمَ عُيونُهُم فَلا تُبصِرَ، ولتَكُنْ ظُهورُهُم مَحنِيَّةً كُلَّ حينٍ“.</span>
</div>
</div>

## خلاص غير اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 13:46–48; ﺭﻭﻣﻴﺔ 11:31; ﺭﻭﻣﻴﺔ 10:19"> وأمَّا أنا فأقولُ: هل زَلَّتْ قَدَمُ اليَهودِ ليَسقُطوا إلى الأبدِ؟ كلَّا! بَلْ بِزَلَّتِهِم صارَ الخلاصُ لِغيرِ اليَهودِ حتّى تَثورَ الغيرَةُ في بَني إِسرائيلَ.</span>
<span className="v" id="v12" data-v="12" data-xref="ﺭﻭﻣﻴﺔ 11:25; كُولُوسِي 1:27; ﺭﻭﻣﻴﺔ 11:15"> فإذا كانَ في زَلَّتِهِم غِنًى لِلعالَمِ وفي نُقصانِهِم غِنًى لِسائِرِ الشُّعوبِ، فكَمْ يكونُ الغِنى في اَكتِمالِهِم؟</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="اعمال الرسل 9:15; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:11–12; أفَسُس 3:8"> والآنَ أقولُ لِغيرِ اليَهودِ مِنكُم: ما دُمتُ رَسولًا إلى غَيرِ اليَهودِ، فأنا فَخورٌ بِرِسالّتي</span>
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:16; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 4:16; ﺭﻭﻣﻴﺔ 9:3"> لعَلِّي أُثيرُ غَيرَةَ بَني قومي فأُخَلِّصَ بَعضًا مِنهُم.</span>
<span className="v" id="v15" data-v="15" data-xref="أفَسُس 1:10; ﺭﻭﻣﻴﺔ 5:10–11; كُولُوسِي 1:20–21"> فإذا كانَ رَفضُهُم أدّى إلى مُصالَحَةِ العالَمِ معَ اللهِ، فهَلْ يكونُ قُبولُهُم إلَّا حياةً بَعدَ موتٍ؟</span>
<span className="v" id="v16" data-v="16" data-xref="حِزْقِيال 44:30; يَعقُوب 1:18; الأمثال 3:9"> وإذا كانَتِ الخَميرَةُ مُقَدَّسَةً، فالعَجينُ كُلُّهُ مُقَدَّسٌ. وإذا كانَ الأصلُ مُقَدَّسًا، فالفُروعُ مُقَدَّسَةٌ أيضًا.</span>
<span className="v" id="v17" data-v="17" data-xref="أفَسُس 2:11–13; أفَسُس 3:6; ﻣﺘﻰ 21:43"> فإذا قُطِعَتْ بَعضُ الفُروعِ، وكُنتَ أنتَ زيتونَةً برِّيَّةً فَطُعِّمْتَ لِتُشارِكَ الفُروعَ الباقِيةَ في أصلِ الشَّجرَةِ وخِصبِها،</span>
<span className="v" id="v18" data-v="18" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:12; ﺭﻭﻣﻴﺔ 11:20; أفَسُس 2:19–20"> فلا تَفتَخِرْ على الفُروعِ الّتي قُطِعَتْ. وكيفَ تَفتَخِرُ وأنتَ لا تَحمِلُ الأصلَ، بَلِ الأصلُ هوَ الّذي يَحمِلُكَ؟</span>
<span className="v" id="v19" data-v="19" data-xref="ﺭﻭﻣﻴﺔ 11:17; ﺭﻭﻣﻴﺔ 11:11–12; ﺭﻭﻣﻴﺔ 11:23–24"> ولكنَّكَ تَقولُ: ”قُطِعَتْ تِلكَ الفُروعُ حتّى أُطعَّمَ أنا!“</span>
<span className="v" id="v20" data-v="20" data-xref="ﺭﻭﻣﻴﺔ 12:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:12"> حسنًا! هيَ قُطِعَتْ لِعَدَمِ إيمانِها، وأنتَ باقٍ لإيمانِكَ، فلا تَفتَخِرْ بَلْ خَفْ.</span>
<span className="v" id="v21" data-v="21" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:4–9; إرْمِيا 25:29; إرْمِيا 49:12"> فَإن كانَ اللهُ لم يُبقِ على الفُروعِ الطَّبيعِيَّةِ فهَلْ يُبقي علَيكَ؟</span>
<span className="v" id="v22" data-v="22" data-xref="العِبرانِيّين 3:14; العِبرانِيّين 3:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:2"> فاَعتَبِرْ بِلينِ اللهِ وشِدَّتِهِ فالشِّدَّةُ على الّذينَ سقَطوا، واللينُ لكَ إذا ثبَتَّ أهلًا لِهذا اللِّينِ، وإلَّا فتُقطَعُ أنتَ أيضًا.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:16; زَكَريّا 12:10; ﻣﺘﻰ 23:39"> أمَّا هُم، فإذا تَوَقَّفوا عَنْ عدَمِ إيمانِهِم يُطَعِّمَهُمْ اللهُ، لأنَّ اللهَ قادِرٌ على أنْ يُطعِمَّهُم ثانيةً.</span>
<span className="v" id="v24" data-v="24" data-xref="ﺭﻭﻣﻴﺔ 11:17–18; ﺭﻭﻣﻴﺔ 11:30"> فإذا كانَ اللهُ قَطَعَكَ مِنْ زيتونَةٍ برِّيَّةٍ تَنتَمي إلَيها بِطبيعتِكَ، وطَعَّمَكَ خِلافًا لِطبيعَتِكَ في زيتونَةٍ جيِّدَةٍ، فما أحقَّ الّذينَ يَنتَمونَ إلى زيتونَتِهِم بالطَّبيعَةِ بأنْ يُطعِّمَهُمُ اللهُ فيها.</span>
</div>
</div>

## رحمة الله تشمل الجميع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="لُوقا 21:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:14–16; ﺭﻭﻣﻴﺔ 16:25"> فأنا لا أُريدُ، أيُّها الإخوةُ، أنْ تَجهَلوا هذا السِّرَّ لِئَلَّا تَحسَبُوا أنفُسَكُم عُقلاءَ، وهوَ أنَّ قِسمًا مِنْ بَني إِسرائيلَ قَسَّى قَلبَهُ إلى أنْ يكمُلَ عَددُ المُؤمِنينَ مِنْ سائِرِ الأُمَمِ.</span>
<span className="v" id="v26" data-v="26" data-xref="إشَعْياء 59:20; إرْمِيا 31:31–37; المَزامِير 14:7"> وهكذا يَخلُصُ جميعُ بَني إِسرائيلَ، كما جاءَ في الكِتابِ: ”مِنْ صِهيونَ يَجيءُ المُخَلِّصُ ويُزيلُ الكُفرَ عَنْ بَني يَعقوبَ.</span>
<span className="v" id="v27" data-v="27" data-xref="إرْمِيا 31:31–34; إشَعْياء 59:21; حِزْقِيال 36:25–29"> ويكونُ هذا عَهدي لَهُم حينَ أمحو خَطاياهُم“.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 10:15; التَّثْنِيَة 7:7–8; ﻣﺘﻰ 21:43"> فاليَهودُ مِنْ حيثُ البِشارَةُ هُم أعداءُ اللهِ لِخيرِكُم. وأمَّا مِنْ حيثُ اَختِيارُ اللهِ، فَهُم أحبّاؤُهُ إكرامًا للآباءِ.</span>
<span className="v" id="v29" data-v="29" data-xref="العِبرانِيّين 7:21; العَدَد 23:19; مَلاخِي 3:6"> ولا نَدامَةَ في هِباتِ اللهِ ودَعَوتِهِ.</span>
<span className="v" id="v30" data-v="30" data-xref="كُولُوسِي 3:7; أفَسُس 2:1–2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:25"> فكَما عَصَيتُمُ اللهَ مِنْ قَبلُ ورَحِمَكُمُ الآنَ لعِصيانِهِم،</span>
<span className="v" id="v31" data-v="31" data-xref="ﺭﻭﻣﻴﺔ 11:15; ﺭﻭﻣﻴﺔ 10:16; ﺭﻭﻣﻴﺔ 11:25"> فكذلِكَ هُمْ عَصَوا اللهَ الآنَ ليَرحَمَهُم كما رَحِمَكُم،</span>
<span className="v" id="v32" data-v="32" data-xref="غَلاطِيَّة 3:22; ﺭﻭﻣﻴﺔ 3:9; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:4–6"> لأنَّ اللهَ جعَلَ البشَرَ كُلَّهُم سُجناءَ العِصيانِ حتّى يَرحَمَهُم جميعًا.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="أيُّوب 5:9; أيُّوب 11:7–9; المَزامِير 92:5"> ما أعمَقَ غِنى اللهِ وحِكمَتَهُ وعِلْمَه! وما أصعَبَ إدراكَ أحكامِهِ وفَهمَ طُرُقِهِ؟</span>
<span className="v" id="v34" data-v="34" data-xref="إشَعْياء 40:13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:16; أيُّوب 15:8"> فالكِتابُ يَقولُ: ”مَنِ الّذي عرَفَ فِكْرَ الرَّبِّ، أو مَنِ الّذي كانَ لَه مُشيرًا؟</span>
<span className="v" id="v35" data-v="35" data-xref="أيُّوب 41:11; أيُّوب 35:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:7"> ومَنِ الّذي بادَرَهُ بالعَطاءِ لِيُبادِلَهُ بِالمِثلِ؟“</span>
<span className="v" id="v36" data-v="36" data-xref="كُولُوسِي 1:15–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:6; العِبرانِيّين 2:10"> فكُلُّ شيءٍ مِنهُ وبِه وإلَيهِ. فَلَهُ المَجدُ إلى الأبدِ. آمين.</span>
</div>
