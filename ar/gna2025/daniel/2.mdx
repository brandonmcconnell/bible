## حلم نبوخذنصر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="دانيال 4:5; أسْتِير 6:1; أيُّوب 33:15–17"> وفي السَّنَةِ الثَّانيَةِ مِنْ عَهدِ نبوخذنَصَّرَ المَلِكِ، حَلُمَ نَبوخذنَصَّرُ أحلامًا أزعَجَتْهُ ومنَعَت عَنهُ النَّومَ.</span>
<span className="v" id="v2" data-v="2" data-xref="دانيال 1:20; دانيال 4:6; التَّكوين 41:8"> فأمرَ أنْ يُدعَى السَّحَرَةُ والمَجوسُ والعَرَّافونَ والمُنَجِّمُونَ ليُفَسِّروا لَه أحلامَهُ، فجاؤوا ووقَفوا أمامَه.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 40:8; التَّكوين 41:15; دانيال 4:5"> فقالَ لهُم: ”حَلُمْتُ حُلُمًا فانْزَعَجْتُ، وأُريدُ أنْ أعرفَ ما هوَ“.</span>
<span className="v" id="v4" data-v="4" data-xref="دانيال 3:9; دانيال 5:10; عَزْرا 4:7"> فأجابَه المُنَجِّمُونَ بالآراميَّةِ: ”أيُّها المَلِكُ عِشْتَ إلى الأبدِ. أخبِرْنا بالحُلُمِ فنُبَيِّنَ تفسيرَهُ“.</span>
<span className="v" id="v5" data-v="5" data-xref="دانيال 3:29; عَزْرا 6:11; المُلُوكِ الثّانِي 10:27"> فقالَ لهُمُ المَلِكُ: ”قُلتُ ولا مرَدَّ لِقولي: إنْ لم تُعلِمونيَ الحُلُمَ وتفسيرَه أُقَطِّعُكُم قِطَعًا وأجعَلُ بُيوتَكُم مَزابِلَ،</span>
<span className="v" id="v6" data-v="6" data-xref="دانيال 5:7; دانيال 2:48; دانيال 5:29"> وإنْ أعلَمْتُموني الحُلُمَ وبَيَّنتُم تفسيرَهُ تَنالونَ منِّي هدايا وجَوائِزَ وإكرامًا كثيرًا. فأعلِموني الحُلُمَ وبَيِّنوا لي تفسيرَهُ“.</span>
<span className="v" id="v7" data-v="7" data-xref="دانيال 2:4; دانيال 2:9; الجامِعَة 10:4"> فأجابُوهُ ثانيَةً: ”أخبِرْنا أيُّها المَلِكُ بالحُلُمِ فنُبَيِّنَ تفسيرَهُ“.</span>
<span className="v" id="v8" data-v="8" data-xref="كُولُوسِي 4:5; أفَسُس 5:16"> فقالَ لهُمُ المَلِكُ: ”أعلَمُ عِلْمَ اليَقينِ أنَّكُم تُحاوِلونَ كَسْبَ الوقتِ عِندَما رأيتُم أنْ لا مرَدَّ لِقولي.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 41:23; دانيال 7:25; أسْتِير 4:11"> إنْ كُنتُم لا تُعلِموني الحُلُمَ، فعِقابُكُم واحدٌ، لأنَّكُمُ اتَّفَقْتُم معًا على كلامٍ كاذِبٍ فاسِدٍ تتكلَّمونَ بهِ أمامي، لَعلَّ الأحوالَ تتَغَيَّرُ معَ الوقتِ. لذلِكَ أخبِروني بِالحُلُمِ، فأعلَمَ أنَّكُم قادِرونَ أنْ تُبَيِّنوا لي تفسيرَهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="دانيال 2:27"> فقالَ المُنَجِّمونَ أمامَ المَلِكِ: ”ما مِنْ إنسانٍ في الأرضِ يَقدِرُ أنْ يُبَيِّنَ ما يأمُرُنا بهِ المَلِكُ، وما مِنْ مَلِكٍ عظيمِ السُّلطانِ سأَلَ ساحِرًا أو مَجوسيًّا أو مُنَجِّمًا عَنْ أمرٍ مِثلِ هذا.</span>
<span className="v" id="v11" data-v="11" data-xref="دانيال 5:11; إشَعْياء 57:15; التَّكوين 41:39"> والأمرُ الّذي سأَلَ عَنهُ المَلِكُ عويصٌ، ولا أحدَ يُبَيِّنُه للمَلِكِ غيرُ الآلهةِ الّذينَ لا يسكُنونَ معَ البشَرِ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="دانيال 3:13; ﻣﺘﻰ 2:16; دانيال 3:19"> عِندَ ذلِكَ غَضِبَ المَلِكُ واَغْتاظَ جدًّا، وأمرَ بِإِبادَةِ جميعِ حُكَماءِ بابِلَ.</span>
<span className="v" id="v13" data-v="13" data-xref="دانيال 1:19–20; أسْتِير 3:12–15; إشَعْياء 10:1"> فصدَرَ الحُكْمُ بقَتلِهِم، ومِنهُم دانيالُ ورفاقُهُ.</span>
<span className="v" id="v14" data-v="14" data-xref="دانيال 2:24; إرْمِيا 52:12; صموئيلَ الثّانِي 20:16–22"> فراجَعَ دانيالُ، باستِعطافٍ وتأديبٍ، أريُوخَ قائِدَ حرَسِ المَلِكِ، وهوَ الّذي خرَجَ لِيَقتُلَ حُكَماءَ بابِلَ</span>
<span className="v" id="v15" data-v="15"> بِسُلطَةٍ مِنَ المَلِكِ، وقالَ لَه: ”لماذا هذا الحُكْمُ الشَّديدُ مِنَ المَلِكِ؟“ فأعلَمَ أريُوخُ دانيالَ بالأمرِ.</span>
<span className="v" id="v16" data-v="16" data-xref="دانيال 1:18–19; دانيال 2:9–11"> فدخَلَ دانيالُ على المَلِكِ وطلَبَ إليهِ أنْ يُعطيَهُ وقتًا لِيُبَيِّنَ لَه تفسيرَ الحُلُمِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="دانيال 1:6–7; دانيال 3:12; دانيال 1:11"> وذهَبَ دانيالُ إلى بَيتِه وأعلَمَ حَنَنيا وميشائيلَ وعَزَريا رفاقَه بِالأمرِ،</span>
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 33:3; ﻣﺘﻰ 18:19; ﻣﺘﻰ 18:12"> لِيَطلُبوا مِنْ إلهِ السَّماءِ أنْ يرحَمَهُم جميعًا ويكشِفَ لهُم سِرَّ هذا الحُلُمِ لِئلَّا يُبادوا معَ سائِرِ حُكُماءِ بابِلَ.</span>
</div>
</div>

## دانيال يفسر الحلم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="العَدَد 12:6; عامُوس 3:7; دانيال 7:7"> وكانَ أنِ اَنْكَشفَ السِّرُّ لِدانيالَ في رُؤيا ليلٍ، فبارَكَ إلهَ السَّماءِ.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 113:2; المَزامِير 145:1–2; رُؤيا يُوحَنّا 5:12"> وقالَ: ”لِيَكُنِ إِسْمُ اللهِ مُبارَكًا مِنَ الأزَلِ وإلى الأبدِ. فلَهُ الحكمَةُ والجَبَروتُ،</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 75:5–7; المَزامِير 31:14–15; المَزامِير 113:7–8"> وهوَ الّذي يُغَيِّرُ الأوقاتَ والأزمِنَةَ ويَعزِلُ المُلوكَ ويُقيمُهُم ويهَبُ الحكمَةَ للحُكَماءِ والمعرِفَةَ للفُهَماءِ.</span>
<span className="v" id="v22" data-v="22" data-xref="أيُّوب 12:22; رسالَةُ يُوحَنّا الأُولَى 1:5; إرْمِيا 23:24"> هوَ الّذي يكشِفُ الأعماقَ والخفايا، ويَعلَمُ ما في الظُّلمَةِ، وعِندَهُ يُضيءُ النُّورُ.</span>
<span className="v" id="v23" data-v="23" data-xref="دانيال 2:18; الخُرُوج 3:15; دانيال 2:29–30"> إِيَّاكَ أحمَدُ يا إِلهَ آبائي. وإِيَّاكَ أُسَبِّحُ، لأنَّكَ وهَبْتَ ليَ الحكمَةَ والقُدْرَةَ، وبُحْتَ ليَ الآنَ بِما طلَبْناهُ مِنكَ، فَأعلَمْتَنا بِما نقولُ للمَلِكِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 27:24; دانيال 2:12–14"> ثُمَّ دخَلَ دانيالُ على أريوخَ الّذي توَلَّى بِأمرِ المَلِكِ إبادَةَ حُكَماءِ بابِلَ وقالَ لَه: ”لا تُبِدْ حُكَماءَ بابِلَ. أدخِلْني إلى أمامِ المَلِكِ فأُبَيِّنَ لَه تفسيرَ الحُلُمِ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:27–28; دانيال 6:13; دانيال 5:13"> فأدخَلَهُ أريوخُ مُسرِعًا إلى أمامِ المَلِكِ وقالَ لَه: ”وجَدْتُ رَجُلًا مِنْ بَني يَهوذا المَسبيِّينَ يُعْلِمُكَ بِتفسيرِ الحُلُمِ“.</span>
<span className="v" id="v26" data-v="26" data-xref="دانيال 1:7; دانيال 5:16; دانيال 5:12"> فقالَ المَلِكُ لِدانيالَ الّذي اسمُهُ بَلطَشاصَّرُ: ”أتَقدِرُ أنتَ أنْ تُعلِمَني بِالحُلُمِ الّذي رأيتُهُ وتُبَيِّنَ لي تفسيرَهُ؟“</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 47:12–14; دانيال 2:2; إشَعْياء 44:25"> فأجابَهُ دانيالُ: ”السِّرُّ الّذي تسأَلُ عَنهُ، أيُّها المَلِكُ، لا يَقدِرُ الحُكَماءُ ولا المَجوسُ ولا السَّحَرَةُ ولا المُنَجِّمونَ أنْ يُبَيِّنوهُ لكَ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّكوين 40:8; عامُوس 4:13; دانيال 10:14"> لكنَّ في السَّماءِ إلهًا يكشِفُ الأسرارَ فأُعلِمُكَ، أيُّها المَلِكُ نَبوخذنَصَّرُ، بِما سيكونُ بَعدَ هذِهِ الأيّامِ. حلُمُكَ وما تراءى لكَ، وأنتَ نائِمٌ في فِراشِكَ هوَ هذا:</span>
<span className="v" id="v29" data-v="29" data-xref="دانيال 2:28; عامُوس 4:13; دانيال 2:47"> جاءَتكَ أيُّها المَلِكُ، وأنتَ نائِمٌ في فِراشِكَ، أفكارٌ في ما سيكونُ بَعدَ هذِهِ الأيّامِ، واللهُ الّذي يكشِفُ الأسرارَ أعلَمَكَ بِسِرِّ ما سيكونُ.</span>
<span className="v" id="v30" data-v="30" data-xref="التَّكوين 41:16; اعمال الرسل 3:12; دانيال 2:17–18"> وهذا السِّرُّ اَنْكَشَفَ لي، لا لِحكمَةٍ فيَّ أكثرَ مِنْ سائِرِ الأحياءِ، ولكنْ لِأُعلِمَكَ أيُّها المَلِكُ بِتفسيرِ حُلُمِكَ وأفكارِ قلبِكَ.</span>
<span className="v" id="v31" data-v="31" data-xref="حَبَقُّوق 1:7; حِزْقِيال 28:7; دانيال 7:3–17"> أنتَ أيُّها المَلِكُ رأيتَ فإذا بِتِمثالٍ عظيمٍ هائِلٍ كثيرِ البَهاءِ. كانَ واقِفًا أمامَكَ وكانَ مَنظَرُهُ رهيبًا.</span>
<span className="v" id="v32" data-v="32" data-xref="إرْمِيا 51:7; دانيال 7:4–6; رُؤيا يُوحَنّا 17:4"> وكانَ رأسُهُ مِنْ ذهَبٍ خالصٍ، وصَدْرُهُ وذِراعاهُ مِنْ فِضَّةٍ، وبَطنُهُ وفَخذاهُ مِنْ نُحاسٍ،</span>
<span className="v" id="v33" data-v="33" data-xref="دانيال 7:19–26; دانيال 7:7–8; دانيال 2:40–43"> وساقاهُ مِنْ حديدٍ وقَدَماهُ بعضُهُما مِنْ حديدٍ والبعضُ مِنْ خزَفٍ،</span>
<span className="v" id="v34" data-v="34" data-xref="دانيال 2:44–45; دانيال 8:25; زَكَريّا 12:3"> وبَينَما أنتَ تَنظُرُ إليهِ اَنقطَعَ حجَرٌ مِنَ الجبَلِ مِنْ دونِ أنْ تَلْمَسَهُ يَدٌ، فضَرَبَ التِّمثالَ على قدَمَيهِ اللَّتينِ مِنْ حديدٍ وخزَفٍ وسحَقَهُما.</span>
<span className="v" id="v35" data-v="35" data-xref="إشَعْياء 2:2–3; المَزامِير 37:10; رُؤيا يُوحَنّا 20:11"> فاَنسَحَقَ الحديدُ والخزَفُ والنُّحاسُ والفِضَّةُ والذَّهَبُ معًا، وصارَت كُلُّها كَتِبْنِ البَيدَرِ في الصَّيفِ، فحَمَلَتْها الرِّيحُ وما وُجِدَ لها أثَرٌ. أمَّا الحجَرُ الّذي ضرَبَ التِّمثالَ، فصارَ جبَلًا كبيرًا ومَلأَ الأرضَ كُلَّها.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="دانيال 2:23–24"> ”هذا هوَ الحُلُمُ. أمَّا تفسيرُه فأُخبرُكَ بهِ أيُّها المَلِكُ:</span>
<span className="v" id="v37" data-v="37" data-xref="حِزْقِيال 26:7; إرْمِيا 27:6–7; عَزْرا 7:12"> أنتَ أيُّها المَلِكُ مَلِكُ المُلوكِ، لِأنَّ إلهَ السَّماءِ وهبَكَ المُلْكَ والعِزَّةَ والقُدْرَةَ والجَلالَ،</span>
<span className="v" id="v38" data-v="38" data-xref="دانيال 4:21–22; المَزامِير 50:10–11; إرْمِيا 27:5–7"> وكُلَّ ما يَسكُنُه بَنو البشَرِ ووُحوشُ البَرِّ وطُيورُ السَّماءِ وهَبَهُ لكَ وسَلَّطَكَ علَيهِ جميعًا، فأنتَ الرَّأسُ الّذي مِنْ ذهَبٍ.</span>
<span className="v" id="v39" data-v="39" data-xref="دانيال 5:28–31; إشَعْياء 44:28–5; دانيال 2:32"> وبَعدَكَ تقومُ مَملَكَةٌ أُخرى أصغَرُ مِنْ مَملَكَتِكَ، ثُمَّ مَملَكَةٌ ثالِثَةٌ سِلاحُها مِنْ نُحاسٍ فتَتَسَلَّطُ على كُلِّ الأرضِ.</span>
<span className="v" id="v40" data-v="40" data-xref="دانيال 7:7; دانيال 7:19–26; دانيال 11:36–45"> ثُمَّ مَملَكَةٌ رابِعَةٌ يكونُ سِلاحُها صَلْبًا كالحديدِ، لِأنَّ الحديدَ يَسحَقُ ويَطحَنُ كُلَّ شيءٍ. فكما أنَّ الحديدَ يُحَطِّمُ، كذلِكَ تَسحَقُ هذِهِ المَملَكَةُ وتُحَطِّمُ جميعَ تِلكَ المَمالِكِ.</span>
<span className="v" id="v41" data-v="41" data-xref="رُؤيا يُوحَنّا 17:12; دانيال 2:33–35; رُؤيا يُوحَنّا 12:3"> كما رأيتَ أنَّ بَعضَ القدَمَينِ والأصابِعِ مِنْ خزَفِ الفخَّارِ والبَعضَ الآخرَ مِنْ حديدٍ، فلذلِكَ تكونُ المَملَكَةُ مُنقَسمَةً، ولكنْ فيها قسوةُ الحديدِ لأنَّ الحديدَ مِثلَما رأيتَ مُختَلِطٌ بِخزَفٍ مِنَ الطِّينِ.</span>
<span className="v" id="v42" data-v="42" data-xref="دانيال 7:24; رُؤيا يُوحَنّا 13:1"> وكما أنَّ أصابِعَ القدَمَينِ، بَعضُها مِنْ حديدٍ وبَعضُها مِنْ خزَفٍ، فكذلِكَ يكونُ بَعضُ المَملَكَةِ صَلْبًا والبَعضُ سَريعَ الانكِسارِ.</span>
<span className="v" id="v43" data-v="43"> ورأيتَ أنَّ الحديدَ مُختَلِطٌ بِخزَفِ الطِّينِ فهذا يَعني أنَّ مُلوكَ تِلكَ المَملَكَةِ يَختَلِطونَ بَعضُهُم معَ بَعضٍ بالزَّواجِ، فلا يَلتَحِمونَ كما أنَّ الحديدَ لا يَختَلِطُ بالخزَفِ.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 28:18; لُوقا 1:32–33; دانيال 6:26"> وفي أيّامِ هؤلاءِ المُلوكِ يُقيمُ إلهُ السَّماءِ مَملَكَةً لا تَخرَبُ أبدًا، ولا يَغلِب سُلطانَها شعبٌ آخَرُ، فتَسحَقُ وتَفْني جميعَ تِلكَ المَمالِكِ، وهيَ تَثبُتُ إلى الأبدِ.</span>
<span className="v" id="v45" data-v="45" data-xref="إشَعْياء 28:16; التَّكوين 41:32; التَّثْنِيَة 10:17"> ورأيتَ أنَّ حجَرًا اَنقَطَعَ مِنَ الجبَلِ مِنْ دُونِ أنْ تَلمَسَهُ يَدٌ، فسَحَقَ الحديدَ والنُّحاسَ والخزَفَ والفِضَّةَ والذَّهَبَ، فهذا يَعني أنَّ الإلهَ العظيمَ أعلَمَ المَلِكَ ما سيكونُ بَعدَ هذِهِ الأيّامِ. حُلُمُكَ صحيحٌ وتفسيري لَه صادِقٌ“.</span>
