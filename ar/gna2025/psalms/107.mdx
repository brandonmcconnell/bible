## الرّب يخلّصنا من الضيق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المَزامِير 106:1; أخبارِ الأيّامِ الأوّلُ 16:34; أخبارِ الأيّامِ الثّانِي 7:3"> إحمدوا الرّبَّ لأنَّهُ صالحٌ، لِأنَّ إلى الأبدِ رحمتَهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:18–19; لُوقا 1:74; غَلاطِيَّة 3:13"> لِيقُلْ ذلِكَ الّذينَ اَفْتَداهُم، اَفْتَداهُمُ الرّبُّ مِنْ يَدِ الضِّيقِ</span>
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 106:47; إشَعْياء 43:5–6; رُؤيا يُوحَنّا 5:9"> وجَمَعَهُم مِنْ كلِّ البُلدانِ، مِنَ المشرِقِ والمغرِبِ، والشِّمالِ والجَنوبِ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 32:10; حِزْقِيال 34:12; حِزْقِيال 34:6"> بعضُهُم تاهَ في برِّيَّةٍ مُقفِرَةٍ ولم يجدوا طريقًا لهُم إلى مدينةٍ يُقيمونَ بها.</span>
<span className="v" id="v5" data-v="5" data-xref="مَرْقُس 8:2–3; صَمُوئيلَ الأوَّلُ 30:11–12; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 2:19"> كانوا جياعًا عِطاشًا تخورُ نُفوسُهُم فيهِم،</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 50:15; المَزامِير 91:15; إرْمِيا 29:12–14"> فصرَخوا إلى الرّبِّ في ضيقِهِم، فأنقَذَهُم مِنْ سُوءِ حالِهِم</span>
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 31:9; المَزامِير 77:20; العِبرانِيّين 11:9–10"> وهَداهُم طريقًا مُستقيمًا توصِلُهم إلى مدينةٍ لِلإقامةِ.</span>
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 107:31; المَزامِير 147:1; دانيال 4:2–3"> فَليَحمَدوا الرّبَّ على رَحمتِهِ وعلى عَجائِبِه لِبَني البشَرِ،</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 5:6; إرْمِيا 31:25; المَزامِير 34:10"> لأنَّه أشبَعَ النَّفْسَ المُشتَهيةَ وملأَ النَّفْسَ الرَّاغِبةَ خَيرًا.</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 1:79; ميخا 7:8; ﺭﻭﻣﻴﺔ 6:20–21"> بعضُهُم في الظَّلامِ وظِلِّ الموتِ جلَسوا أسرى يُعانونَ القُيودَ،</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 106:43; المَزامِير 78:40; ﺭﻭﻣﻴﺔ 1:28"> لأنَّهُم عَصَوا كلامَ اللهِ واَسْتَهانُوا بِمَشورَةِ العليِّ.</span>
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 22:11; الخُرُوج 5:18–19; الخُرُوج 2:23"> أذَلَّ قُلوبَهُم بالتَّعَبِ، وتساقَطوا ولا نصيرَ لهُم.</span>
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 107:28; المَزامِير 107:6; القُضاة 4:3"> صرَخوا إلى الرّبِّ في ضيقِهِم، فخَلَّصَهُم مِنْ سُوءِ حالِهِم.</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 42:16; أفَسُس 5:8; المَزامِير 116:16"> أخرَجَهُم مِنَ الظَّلامِ وظِلِّ الموتِ وقَطَعَ لهُم قُيودَهُم.</span>
<span className="v" id="v15" data-v="15" data-xref="المَزامِير 107:21; المَزامِير 107:31; المَزامِير 107:8"> فلْيَحمَدوا الرّبَّ على رَحمتِهِ وعلى عَجائِبِهِ لِبَني البشَرِ،</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 45:1–2; القُضاة 16:3; ميخا 2:13"> لأنَّهُ حطَّمَ أبوابَ النُّحاسِ وكسَرَ مَغاليقَ الحديدِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:39; الأمثال 1:22; المَزامِير 38:1–8"> وبعضُهُم حَمْقى لِسُلوكِ المَعاصي، ومِنْ آثامِهِم كانوا يُعانونَ.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 9:13; المَزامِير 88:3; أيُّوب 33:19–22"> عافَت نُفوسُهُم كُلَّ طعَامٍ وبَلَغوا إلى أبوابِ الموتِ.</span>
<span className="v" id="v19" data-v="19" data-xref="المَزامِير 78:34–35; المَزامِير 34:4–6; المَزامِير 107:13"> صرَخوا إلى الرّبِّ في ضيقِهِم، فخَلَّصَهُم مِنْ سُوءِ حالِهِم.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 8:8; المَزامِير 103:3–4; المَزامِير 147:15"> أرسلَ كلِمَتهُ فشَفاهُم ونَجَّاهُم مِنْ مَهالِكِهم.</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 107:8; المَزامِير 107:15; أخبارِ الأيّامِ الثّانِي 32:25"> فَليَحمَدوا الرّبَّ على رَحمتِهِ وعلى عَجائِبِه لِبَني البشَرِ.</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 50:14; المَزامِير 118:17; المَزامِير 9:11"> وليَذبَحوا ذَبائِحَ الحَمدِ، وليُخْبِروا عَنْ أعمالِهِ بالتَّرنيمِ.</span>
<span className="v" id="v23" data-v="23" data-xref="رُؤيا يُوحَنّا 18:17; المَزامِير 48:7; حِزْقِيال 27:26"> وآخرونَ أبحروا في السُّفُنِ. ليُتاجروا في المياهِ الغزيرةِ،</span>
<span className="v" id="v24" data-v="24" data-xref="المَزامِير 104:24–27; المَزامِير 95:5; أيُّوب 38:8–11"> فرأوا أعمالَ الرّبِّ وعَجائِبَهُ في اللُّجَجِ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻳﻮﻧﺎﻥ 1:4; ﻣﺘﻰ 8:24; المَزامِير 148:8"> أمرَ فثارَت ريحٌ عاصِفةٌ وهَيَّجَت أمواجَ البحرِ،</span>
<span className="v" id="v26" data-v="26" data-xref="المَزامِير 22:14; المَزامِير 119:28; إشَعْياء 13:7"> فكانت تعلو إلى السَّماءِ وتهبُطُ بِهِم إلى الأعماقِ، فَتَرَجَّحَت حياتُهُم في الخَطَرِ.</span>
<span className="v" id="v27" data-v="27" data-xref="أيُّوب 12:25; إشَعْياء 29:9; إشَعْياء 19:3"> تَرَنَّحوا ومالوا كالسُّكارى، وكُلُّ حِكمَتِهِم ذهَبَت عَبَثًا.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 8:25; المَزامِير 107:13; المَزامِير 107:19"> فصرَخوا إلى الرّبِّ في ضيقِهِم، فأنقَذَهُم مِنْ سُوءِ حالِهم.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 8:26; المَزامِير 65:7; لُوقا 8:23–25"> هدَّأَ الزَّوبعَةَ فَسكَنَت وسكَتَت أمواجُ البحرِ.</span>
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 6:21"> ففَرِحوا حينَ سكَنَت وهداهُم إلى ميناءٍ أرادوها.</span>
<span className="v" id="v31" data-v="31" data-xref="المَزامِير 107:8; المَزامِير 103:2; المَزامِير 107:21"> فَليَحمَدوا الرّبَّ على رَحمتِهِ وعلى عَجائبه لِبَني البشَرِ،</span>
<span className="v" id="v32" data-v="32" data-xref="المَزامِير 22:25; المَزامِير 22:22; المَزامِير 99:5"> وليَرفَعوهُ في جماعةِ الشَّعبِ، وليُهَلِّلوا لَه في مَجلِسِ شيوخِهِم.</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 50:2; إشَعْياء 42:15; عامُوس 4:7–8"> يجعَلُ الأنهارَ قَفرًا ومجاريَ المياهِ جَفافًا.</span>
<span className="v" id="v34" data-v="34" data-xref="التَّكوين 14:3; التَّكوين 13:10; حِزْقِيال 47:11"> والتُّربَةَ المُخصِبةَ مِلْحًا لِشَرِّ المُقيمينَ بها.</span>
<span className="v" id="v35" data-v="35" data-xref="المَزامِير 114:8; إشَعْياء 35:6–7; إشَعْياء 41:17–19"> يجعَلُ القَفْرَ مَرْجًا والأرضَ القاحلةَ مجاريَ مياهٍ،</span>
<span className="v" id="v36" data-v="36" data-xref="المَزامِير 107:7; لُوقا 1:53; اعمال الرسل 17:26"> فيسكُنُ هُناكَ الجياعُ ويُقيمونَ لهُم مدينةَ سُكْنى.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:10; إرْمِيا 29:5"> يزرَعونَ حُقولًا ويغرِسون كُرومًا فتثمِرُ لهُم غِلالًا.</span>
<span className="v" id="v38" data-v="38" data-xref="الخُرُوج 1:7; التَّكوين 12:2; التَّكوين 17:20"> يُبارِكُهُم فَيكثُرونَ جِدًّا ولا يُقلِّلُ بَهائِمَهُم.</span>
<span className="v" id="v39" data-v="39" data-xref="الخُرُوج 2:23–24; المُلُوكِ الثّانِي 10:32; أخبارِ الأيّامِ الثّانِي 15:5–6"> حينَ يَقِلُّونَ ويَنحَنُون ذُلًا تحت ثِقْلِ الأذى والحَسْرَةِ.</span>
<span className="v" id="v40" data-v="40" data-xref="أيُّوب 12:24; أيُّوب 12:21; دانيال 4:33"> يصُبُّ الذُّلَّ على أسيادِهِم ويقودُهم في تيهٍ بِلا طريقٍ.</span>
<span className="v" id="v41" data-v="41" data-xref="صَمُوئيلَ الأوَّلُ 2:8; أيُّوب 21:11; أسْتِير 8:15–17"> يرفعُ البؤساءَ مِنْ عَنائِهِم، ويُكثِرُ عَشائِرَهُم كقُطْعانِ الغنَمِ.</span>
<span className="v" id="v42" data-v="42" data-xref="المَزامِير 63:11; أيُّوب 22:19; ﺭﻭﻣﻴﺔ 3:19"> فيَنظُرُ المُستَقِيمونَ ويفرَحونَ، ويَسُدُّ كُلُّ جائرٍ فَمَهُ.</span>
<span className="v" id="v43" data-v="43" data-xref="هُوشَع 14:9; إرْمِيا 9:12; المَزامِير 64:9"> مَنْ كانَ حكيمًا يُراعي هذا، ويَتَبَيَّنُ مَراحمَ الرّبِّ.</span>
