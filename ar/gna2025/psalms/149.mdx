<div className="bp">

<span className="v" id="v2" data-v="2" data-xref="المَزامِير 100:1–3; المَزامِير 95:6; إشَعْياء 52:7"> يفرَحُ بَنو إِسرائيلَ بِخالِقِهم. يبتَهِجُ بَنو صِهيَونَ بِمَلِكِهِم.</span>
<span className="v" id="v3" data-v="3" data-xref="إرْمِيا 31:13; الخُرُوج 15:20; المَزامِير 81:2"> يُهَلِّلونَ لاَسمِهِ بالرَّقْصِ. يُرَتِّلونَ بالدُّفِّ والكَنَّارةِ</span>

<span className="v" id="v6" data-v="6" data-xref="العِبرانِيّين 4:12; المَزامِير 66:17; رُؤيا يُوحَنّا 1:16"> يُعَظِّمونَ اللهَ مِلْءَ أفواهِهِم، وبأيديهِم سيفٌ ذو حَدَّينِ.</span>
<span className="v" id="v7" data-v="7" data-xref="صَمُوئيلَ الأوَّلُ 15:2–3; رُؤيا يُوحَنّا 19:11–21; زَكَريّا 14:17–19"> ينتَقِمونَ مِنْ جميعِ الأمَمِ ويُؤَدِّبونَ جميعَ الشُّعوبِ.</span>

<span className="v" id="v9" data-v="9" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:2–3; المَزامِير 148:14; المَزامِير 137:8"> يُنزِلونَ بِهِمِ القضاءَ المكتوبَ، فَيَتمَجَّدُ جميعُ أتقيائِهِ. هَلِّلويا.</span>
