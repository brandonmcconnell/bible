## الإيمان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:18; ﺭﻭﻣﻴﺔ 8:24–25"> الإيمانُ هوَ الوُثوقُ بِما نَرجوهُ وتَصديقُ ما لا نَراهُ،</span>
<span className="v" id="v2" data-v="2" data-xref="العِبرانِيّين 11:39; العِبرانِيّين 11:4"> وبِه شَهِدَ اللهُ لِلقُدَماءِ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 33:6; رسالَةُ بُطرُسَ الثّانيَة 3:5; يُوحَنّا 1:3"> بالإيمانِ نُدرِكُ أنَّ اللهَ خَلَقَ الكَونَ بِكلِمَةٍ مِنهُ، فصَدَرَ ما نَراهُ مِمَّا لا نَراهُ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="العِبرانِيّين 12:24; التَّكوين 4:3–5; رسالَةُ يُوحَنّا الأُولَى 3:11–12"> بالإيمانِ قَدَّمَ هابيلُ للهِ ذَبيحَةً أفضَلَ مِنْ ذَبيحةِ قايينَ، وبالإيمانِ شَهِدَ اللهُ لَه أنَّهُ مِنَ الأبرارِ عِندَما رَضِيَ بِقرابينِهِ، وبالإيمانِ ما زالَ يتكَلَّمُ بَعدَ مَوتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 5:21–24; يَهُوذا 1:14; العِبرانِيّين 11:6"> بالإيمانِ رفَعَ اللهُ أخْنُوخَ إلَيهِ مِنْ غَيرِ أنْ يَرى المَوتَ، فما وجَدَهُ أحدٌ لأنَّ اللهَ رَفعَهُ إلَيهِ. والكِتابُ شَهِدَ لَه قَبلَ رَفْعِهِ بأنَّهُ أرضى اللهَ،</span>
<span className="v" id="v6" data-v="6" data-xref="الأمثال 8:17; إرْمِيا 29:13–14; ﻣﺘﻰ 6:33"> وبِغَيرِ الإِيمانِ يَستَحيلُ إرضاءُ اللهِ، لأنَّ الّذي يتَقَرَّبُ إلى اللهِ يَجِبُ أنْ يُؤمِنَ بأنَّهُ موجودٌ وأنَّهُ يُكافئُ الّذينَ يَطلُبونَه.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 6:13–1; رسالَةُ بُطرُسَ الأُولَى 3:20; حِزْقِيال 14:14"> بالإيمانِ اَتَّعَظَ نُوحٌ فبَنى فُلْكًا لِخلاصِ أهلِ بَيتِهِ عِندَما أنذَرَهُ اللهُ بِما سيَحدُثُ مِنْ أُمورٍ لا يَراها. وهكذا حكَمَ على العالَمِ وورِثَ البِرَّ ثَمَرَةً للإيمانِ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="التَّكوين 12:1–4; اعمال الرسل 7:2–4; ﻣﺘﻰ 7:24–25"> بالإيمانِ لبَّى إبراهيمُ دَعوَةَ اللهِ فخَرَجَ إلى بلَدٍ وعَدَهُ اللهُ بِه ميراثًا، خرَجَ وهوَ لا يَعرِفُ إلى أينَ يَذهَبُ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 12:8; التَّكوين 13:18; التَّكوين 13:3"> وبالإيمانِ نَزَلَ في أرضِ الميعادِ كأنَّهُ في أرضٍ غريبَةٍ، وأقامَ في الخِيامِ معَ إسحقَ ويَعقوبَ شَريكَيهِ في الوَعدِ ذاتِهِ،</span>
<span className="v" id="v10" data-v="10" data-xref="العِبرانِيّين 13:14; فِيلِبِّي 3:20; العِبرانِيّين 12:22"> لأنَّهُ كانَ ينتَظِرُ المدينةَ الثّابِتَةَ على أُسُسٍ واللهُ مُهَندِسُها وبانيها.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﺭﻭﻣﻴﺔ 4:20–21; العِبرانِيّين 10:23; التَّكوين 18:11–14"> بالإيمانِ نالَتْ سارَةُ نَفسُها القُدرَةَ على أنْ تَحبَلَ معَ أنَّها عاقِرٌ جاوَزَتِ السِّنَّ، لأنَّها اَعتَبَرَت أنَّ الّذي وعَدَ أمينٌ،</span>
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 22:17; التَّكوين 32:12; التَّكوين 15:5"> فولَدَتْ مِنْ رَجُلٍ واحدٍ قارَبَ المَوتَ نَسلًا كثيرًا مِثلَ نُجومِ السَّماءِ ولا حَصرَ لَه كالرِّمالِ الّتي على شاطِئِ البحرِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 23:4; المَزامِير 39:12; العِبرانِيّين 11:27"> وفي الإيمانِ ماتَ هَؤُلاءِ كُلُّهُم دونَ أن يَنالوا ما وعَدَ اللهُ بِه، ولكنَّهُم رَأَوهُ وحَيَّوهُ عَنْ بُعدٍ. واعتَرَفوا بأنَّهُم غُرَباءُ نُزَلاءُ في الأرضِ،</span>
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:18–7; العِبرانِيّين 11:16; العِبرانِيّين 13:14"> والّذينَ يَقولونَ هذا القَولَ يُبَرهِنونَ أنَّهُم يَطلُبونَ وطَنًا.</span>
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 24:6–8; التَّكوين 12:10; التَّكوين 32:9–11"> ولَو كانوا ذكَروا الوَطنَ الّذي خَرَجوا مِنهُ، لكانَ لهُم فُرصَةٌ لِلعَودَةِ إلَيهِ.</span>
<span className="v" id="v16" data-v="16" data-xref="العِبرانِيّين 13:14; العِبرانِيّين 11:10; فِيلِبِّي 3:20"> ولكنَّهُم كانوا يَشتاقونَ إلى وطَنٍ أفضَلَ مِنهُ، أي إلى الوَطَنِ السَّماوِيِّ. لذلِكَ لا يَستَحي اللهُ أنْ يكونَ إلهَهُم، فهوَ الّذي أعَدَّ لهُم مدينةً.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 22:16; التَّكوين 22:1–12; يَعقُوب 2:21–24"> بالإيمانِ قَدَّمَ إبراهيمُ اَبنَهُ الوحيدَ إسحقَ ذَبيحةً عِندَما اَمتحَنَهُ اللهُ، قدَّمَهُ وهوَ الّذي أعطاهُ اللهُ الوَعدَ</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 21:12; ﺭﻭﻣﻴﺔ 9:7; التَّكوين 17:19"> وقالَ لَه: ”بإسحقَ يكونُ لَكَ نَسلٌ“.</span>
<span className="v" id="v19" data-v="19" data-xref="ﺭﻭﻣﻴﺔ 4:17–21; ﻣﺘﻰ 9:28; أفَسُس 3:20"> واَعتقَدَ إبراهيمُ أنَّ اللهَ قادِرٌ أنْ يُقيمَ الأمواتَ. لِذلِكَ عادَ إلَيهِ اَبنُهُ إسحقُ وفي هذا رَمزٌ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 27:27–40; التَّكوين 28:2–3"> بالإيمانِ بارَكَ إسحقُ يَعقوبَ وعيسو لِخَيراتِ المُستَقبَلِ.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 47:31–1; التَّكوين 48:5–22"> وبالإيمانِ بارَكَ يَعقوبُ، لمَّا حَضَرَهُ المَوتُ، كُلًا مِن اَبنَي يوسُفَ وسجَدَ للهِ وهوَ مُستَنِدٌ إلى طرَفِ عَصاهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 13:19; التَّكوين 50:24–25; يَشُوع 24:32"> وبالإيمانِ ذكَرَ يوسُفُ عِندَ موتِهِ خُروجَ بَني إِسرائيلَ مِنْ مِصْرَ وأوصى أين يَدفنونَ عِظامَه.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 1:22; الخُرُوج 1:16; العِبرانِيّين 13:6"> بالإيمانِ أخفى والِدا موسى اَبنَهُما ثَلاثةَ أشهُرٍ بَعدَ مَولِدِهِ، لأنَّهُما رأيا الطِّفلَ جَميلًا وما أخافَهُما أمرُ المَلِكِ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 2:10–11; اعمال الرسل 7:21–24"> بالإيمانِ رفَضَ موسى، بَعدَما كبِرَ، أنْ يُدعى اَبنًا لِبنتِ فِرعَونَ،</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 84:10; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:3–10; أيُّوب 36:21"> وفَضَّلَ أنْ يُشارِكَ شَعبَ اللهِ في الذُّلِّ على التَّمتُّعِ الزائِلِ بِالخَطيئَةِ،</span>
<span className="v" id="v26" data-v="26" data-xref="العِبرانِيّين 10:35; العِبرانِيّين 11:6; رسالَةُ بُطرُسَ الأُولَى 4:14"> واَعتبَرَ عارَ المَسيحِ أغنى مِنْ كُنوزِ مِصْرَ، لأنَّهُ تَطلَّعَ إلى ما سيَنالُهُ مِنْ ثَوابٍ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:18; العِبرانِيّين 11:13; العِبرانِيّين 11:1"> بِالإيمانِ ترَكَ موسى مِصْرَ دونَ أنْ يَخافَ مِنْ غَضَبِ المَلِكِ، وثَبَتَ على عَزمِهِ كأنَّهُ يَرى ما لا تَراهُ عَينٌ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 12:21–30; العِبرانِيّين 9:19; العِبرانِيّين 12:24"> بِالإيمانِ أقامَ الفِصحَ ورَشَّ الدَّمَ، لِئَلَّا يَمَسَّ مَلاكُ المَوتِ أيَّ بِكرٍ لِبَني إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="الخُرُوج 14:13–21; المَزامِير 106:9–11; المَزامِير 78:13"> بالإيمانِ عبَرَ بَنو إِسرائيلَ البحرَ الأحمَرَ كأنَّهُ بَرٌّ، ولمَّا حاوَلَ المِصْرِيُّونَ عُبورَهُ غَرِقوا.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يَشُوع 6:3–20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:4–5"> بِالإيمانِ سَقَطَت أسوارُ أريحا بَعدَما طافَ بِها بَنو إِسرائيلَ سَبعَةَ أيّامٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="يَعقُوب 2:25; يَشُوع 6:22–25; ﻣﺘﻰ 1:5"> بِالإيمانِ نَجَتْ راحابُ البَغِيُّ مِنَ الهَلاكِ معَ العُصاةِ، لأنَّها رَحَّبَت بِالجاسوسَينِ.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="صَمُوئيلَ الأوَّلُ 16:1; صَمُوئيلَ الأوَّلُ 1:20; صَمُوئيلَ الأوَّلُ 16:13"> وماذا أقولُ بَعدُ؟ الوَقتُ يَضيقُ بي إذا أخبَرتُ عنْ جِدعَونَ وباراقَ وشَمشونَ ويَفتاحَ وداودَ وصَموئيلَ والأنبياءِ.</span>
<span className="v" id="v33" data-v="33" data-xref="المَزامِير 144:1–2; العِبرانِيّين 6:12–15; العِبرانِيّين 10:36"> فهُم بِالإيمانِ أخضَعوا المَمالِكَ وأقاموا العَدلَ ونالوا ما وعَدَ بِه اللهُ وسَدُّوا أفواهَ الأُسودِ</span>
<span className="v" id="v34" data-v="34" data-xref="المُلُوكِ الثّانِي 6:16–18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:9–10; صَمُوئيلَ الأوَّلُ 17:51–52"> وأخمَدوا لَهيبَ النِّيرانِ ونَجَوا مِنْ حَدِّ السَّيفِ وتغَلَّبوا على الضُّعفِ وصاروا أبطالًا في الحَربِ وهزَموا جُيوشَ الغُرَباءِ،</span>
<span className="v" id="v35" data-v="35" data-xref="المُلُوكِ الثّانِي 4:27–37; المُلُوكِ الأوَّلُ 17:22–24; فِيلِبِّي 3:11"> واَستَعادَ نِساءٌ أمواتَهُنَّ بالقِيامَةِ. واَحتمَلَ بَعضُهُمُ التَّعذيبَ ورَفَضوا النَّجاةَ في سَبيلِ القيامَةِ إلى حَياةٍ أفضَلَ،</span>
<span className="v" id="v36" data-v="36" data-xref="إرْمِيا 20:2; التَّكوين 39:20; إرْمِيا 37:15–21"> وقاسى آخَرونَ الهُزءَ والجَلْدَ، بَلِ القُيودَ والسِّجنَ.</span>
<span className="v" id="v37" data-v="37" data-xref="أخبارِ الأيّامِ الثّانِي 24:21; إرْمِيا 26:23; المُلُوكِ الأوَّلُ 19:10"> ورُجِموا ونُشِروا وقُتِلوا بِحَدِّ السَّيفِ وتَشَرَّدوا لابِسينَ جُلودَ الغَنَمِ والماعِزِ مَحرومينَ مَقهورينَ مَظلومينَ،</span>
<span className="v" id="v38" data-v="38" data-xref="المُلُوكِ الأوَّلُ 18:4; صَمُوئيلَ الأوَّلُ 22:1; المُلُوكِ الأوَّلُ 19:9"> لا يَستَحِقُّهُمُ العالَمُ، فتاهوا في البَراري والجِبالِ والمَغاوِرِ وكُهوفِ الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="لُوقا 10:23–24; العِبرانِيّين 11:13; العِبرانِيّين 11:2"> وما حصَلَ هَؤُلاءِ على الوَعدِ معَ أنَّهُ مَشهودٌ لهُم بِالإيمانِ،</span>
<span className="v" id="v40" data-v="40" data-xref="رُؤيا يُوحَنّا 6:11; العِبرانِيّين 8:6; العِبرانِيّين 10:11–14"> لأنَّ اللهَ أعَدَّ لنا مَصيرًا أفضَلَ مِنْ مَصيرِهِم وشاءَ أنْ لا يَصيروا كامِلينَ إلَّا مَعَنا.</span>
</div>
