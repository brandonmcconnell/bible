## شرائع تقريب القربان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 23:10; العَدَد 15:18; اللّاويِّين 25:2"> ”قُلْ لِبَني إِسرائيلَ إذا دخَلتُمُ الأرضَ الّتي أعطيتُكُم لِتقيموا بِها،</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 29:18; التَّكوين 8:21; فِيلِبِّي 4:18"> وقرَّبْتُم مِنَ البقَرِ والغنَمِ وَقيدةً لِلرّبِّ، مُحرَقةً أو ذبيحةَ وَفاءِ نَذْرٍ أو تَبرُّعًا أو فريضةً في أعيادِكُم، تُرضي رائحَتُها الرّبَّ،</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 29:40; اللّاويِّين 23:13; اللّاويِّين 6:14"> فليُقَرِّبْ صاحِبُ هذا القربانِ تقدمةً للرّبِّ تكونُ عُشْرَ قُفَّةٍ مِنَ الدَّقيقِ مَلتوتًا بِرُبْعِ خابيةٍ مِنَ الزَّيتِ</span>
<span className="v" id="v5" data-v="5" data-xref="العَدَد 28:7; العَدَد 28:14; القُضاة 9:13"> ورُبْعَ خابيةٍ مِنَ الخمرِ سكيبًا يُصعِدُهُ معَ المُحرَقةِ أو معَ الذَّبيحةِ لِلخروفِ الواحدِ.</span>
<span className="v" id="v6" data-v="6" data-xref="العَدَد 28:12–14; العَدَد 15:4"> أمَّا لِلكَبْشِ فيُقَرِّبُ تَقدمةً تَكونُ عِشرينَ قُفَّةً مِنَ الدَّقيقِ مَلتوتةً بِثُلْثِ خابيةٍ مِنَ الزَّيتِ</span>
<span className="v" id="v7" data-v="7"> وثُلْثِ خابيةٍ مِنَ الخمرِ سكيبًا يُقدِّمُه رائحةَ رِضًى لِلرّبِّ.</span>
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 3:1; اللّاويِّين 7:11–18; اللّاويِّين 1:3"> وإذا قرَّبَ عِجْلًا للرّبِّ، مُحرَقةً أو ذبيحةً أو وَفاءَ نَذْرٍ أو ذبيحةَ سلامةٍ،</span>
<span className="v" id="v9" data-v="9" data-xref="العَدَد 28:14; العَدَد 28:12; اللّاويِّين 6:14"> فليُقَرِّبْ معَ العِجْلِ تقدِمةً تكونُ ثَلاثةَ أعشارِ القُفَّةِ مِنَ الدَّقيقِ مَلتوتةً بِنِصفِ خابيةٍ مِنَ الزَّيتِ،</span>
<span className="v" id="v10" data-v="10" data-xref="العَدَد 15:5; العَدَد 6:15"> ويُقَدِّمُ معَه سكيبًا يكونُ نِصفَ خابيةٍ مِنَ الخمرِ، وَقيدةً تُرضي رَائحَتُها الرّبَّ.</span>
<span className="v" id="v11" data-v="11"> ومِثلَ ذلِكَ يُقَرِّبُ صاحبُ القربانِ معَ كُلِّ ثَورٍ وكُلِّ كَبْشٍ وكُلِّ رأسٍ مِنَ الغنَمِ أوِ المعَزِ.</span>
<span className="v" id="v12" data-v="12"> وإذا قَرَّبَ أكثرَ مِنْ حيوانٍ واحدٍ تُزادُ التَّقدِمةُ بِحسَبِ العدَدِ.</span>
<span className="v" id="v13" data-v="13"> ذلِكَ ما يُقَرِّبُه كُلُّ إِسرائيليٍّ أصيلٍ إذا أصعدَ مُحرَقةً تُرضي رائحَتُها الرّبَّ.</span>
<span className="v" id="v14" data-v="14"> وأيُّ دَخيلٍ نزَلَ بِكُم أو سكَنَ فيما بَينَكُم مدَى أجيالِكُم وصَنعَ وَقيدةً تُرضي رائحَتُها الرّبَّ، فليَصنعْ كما تصنَعونَ.</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 9:14; العَدَد 15:29; الخُرُوج 12:49"> فريضَةٌ واحدةٌ أبديَّةٌ لَكم ولِلدَّخيلِ النَّازِلِ بَينَكُم مدَى أجيالِكُم، مِثْلُكُم يكونُ الدَّخيلُ أمامَ الرّبِّ.</span>
<span className="v" id="v16" data-v="16"> شريعةٌ وحُكْمٌ واحدٌ يكونانِ لكم ولِلدَّخيلِ النَّازلِ فيما بَينَكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v18" data-v="18" data-xref="العَدَد 15:2; التَّثْنِيَة 26:1–15"> ”قُلْ لِبَني إِسرائيلَ: إذا دَخَلتُمُ الأرضَ الّتي سأُدخِلُكُم إيَّاها،</span>
<span className="v" id="v19" data-v="19" data-xref="يَشُوع 5:11–12"> فقَدِّموا مِمَّا تُطعِمُكُمُ الأرضُ تقدِمةً لِلرّبِّ:</span>
<span className="v" id="v20" data-v="20" data-xref="اللّاويِّين 2:14; الخُرُوج 34:26; التَّثْنِيَة 26:2–10"> مِنْ أوَّلِ عجينكُم تُقَدِّمونَ رغيفًا تقدِمةً خاصَّةً كَتِلكَ الّتي تُقَدِّمونَها مِنَ الحُبوبِ على بيادِرِكُم،</span>
<span className="v" id="v21" data-v="21" data-xref="الخُرُوج 29:28; العَدَد 18:26"> وتكونُ تقدِمةً لِلرّبِّ مدَى أجيالِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اللّاويِّين 4:2; اللّاويِّين 4:27; اللّاويِّين 5:15–17"> ”وإنْ سَهَوتُم فما عَمِلْتُم بجميعِ هذِهِ الوصايا الّتي أمرَ الرّبُّ بِها موسى،</span>
<span className="v" id="v23" data-v="23"> وجاءَت على لِسانِهِ مُنذُ أمرَهُ الرّبُّ بِها فصاعِدًا إلى مدَى أجيالِكُم،</span>
<span className="v" id="v24" data-v="24" data-xref="اللّاويِّين 4:23; اللّاويِّين 4:13–21; اللّاويِّين 5:15"> فإنْ خَفيَ السَّهوُ على عُيونِ الجماعةِ، فَلتُقَرِّبِ الجماعةُ كُلُّها عِجْلًا مِنَ البقَرِ مُحرَقةً تُرضي رائحَتُها الرّبَّ، معَ تقدِمَتِهِ وسكيبِهِ مِنَ الخمرِ، بِحسَبِ الفَرائِضِ، وتيسًا مِنَ المعَزِ ذبيحةَ خطيئةٍ،</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 4:20; ﺭﻭﻣﻴﺔ 3:25; العِبرانِيّين 2:17"> فَيُكَفِّرَ الكاهنُ عَن جماعةِ بَني إِسرائيلَ فيُسامِحَهُمُ الرّبُّ، لأنَّ خطيئَتَهُم سَهْوٌ وجاؤوا بِقربانِهم وَقيدةً للرّبِّ وذبيحةَ خطيئةٍ أمامَ الرّبِّ عَنْ سَهْوِهِم.</span>
<span className="v" id="v26" data-v="26" data-xref="العَدَد 15:24"> كلُّ جماعةِ بَني إِسرائيلَ والدُّخلاءِ النَّازِلينَ فيما بَينَهُم يُسامِحُهُمُ الرّبُّ لأنَّهُم خَطِئوا سَهْوًا.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 4:27–28; اعمال الرسل 17:30; اعمال الرسل 3:17"> ”وإنْ خَطِئَ أحدٌ مِنهُم سَهْوًا، فليُقَرِّبْ عَنزًا حَوليَّةً ذبيحةَ خطيئةٍ،</span>
<span className="v" id="v28" data-v="28" data-xref="اللّاويِّين 4:35"> فيُكفِّرُ الكاهنُ عَنهُ فيُسامِحُهُ الرّبُّ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 15:15; اللّاويِّين 16:29; اللّاويِّين 17:15"> لِلأصيلِ مِنْ بَني إِسرائيلَ ولِلدَّخيلِ النَّازلِ فيما بَينَهُم شريعةٌ واحدةٌ تكونُ لكُم لِمَنْ خَطئَ سَهْوًا.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="العِبرانِيّين 10:26; المَزامِير 19:13; التَّثْنِيَة 1:43"> ”وأيُّ أصيلٍ أو دخيلٍ خَطئَ عَمْدًا يكونُ جَدَّفَ على الرّبِّ فيُقطَعُ مِنْ بَينِ شعبِهِ،</span>
<span className="v" id="v31" data-v="31" data-xref="صموئيلَ الثّانِي 12:9; الأمثال 13:13; حِزْقِيال 18:20"> لأنَّهُ اَسْتهانَ بِكلامِ الرّبِّ ونقضَ وصيَّتَهُ. يُقطَعُ قَطْعًا ويتَحَمَّلُ عاقِبةَ خطيئَتِهِ.</span>
</div>
</div>

## الرّجل الذي خالف شريعة السبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 35:2–3; الخُرُوج 31:14–15; الخُرُوج 20:8–10"> وحينَ كانَ بَنو إِسرائيلَ في البرِّيَّةِ وجَدوا رَجُلًا يَحتَطِبُ حطَبًا في يومِ السَّبْتِ،</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 8:3–20"> فقادَهُ الّذينَ وجَدوهُ إلى موسى وهرونَ وكُلِّ الجماعةِ.</span>
<span className="v" id="v34" data-v="34" data-xref="اللّاويِّين 24:12; العَدَد 9:8"> فألقَوهُ في السِّجنِ لأن ما يَفعلونَ بِهِ لم يُعلَنْ لهُم.</span>
<span className="v" id="v35" data-v="35" data-xref="اعمال الرسل 7:58; الخُرُوج 31:14–15; المُلُوكِ الأوَّلُ 21:13"> فقالَ الرّبُّ لِموسى: ”يُقتَلُ الرَّجلُ قَتْلًا. تَرجُمُهُ كُلُّ الجماعةِ بِالحجارةِ في خارجِ المَحلَّةِ“.</span>
<span className="v" id="v36" data-v="36" data-xref="يَشُوع 7:25"> فأخرجَتْهُ الجماعةُ كُلُّها إلى خارجِ المَحلَّةِ ورَجموهُ بِالحجارةِ، فماتَ كما أمرَ الرّبُّ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 23:5; التَّثْنِيَة 22:12; ﻣﺘﻰ 9:20"> ”قُلْ لبَني إِسرائيلَ أنْ يَصنَعوا لهُم أهدابًا على أذيالِ ثيابِهِم مدَى أجيالِكُم، ويجعَلوا على أهدابِ الذَّيلِ سِلْكًا أزرَقَ اللَّونِ،</span>
<span className="v" id="v39" data-v="39" data-xref="المَزامِير 106:39; الأمثال 28:26; المَزامِير 73:27"> فترَونَها وتَذكرونَ جميعَ وصايا الرّبِّ وتَعملونَ بِها، ولا تَضِلُّونَ بِاَتِّباعِ ما قد يكونُ في قلوبِكُم وعيونِكُم مِنْ مَيلٍ إلى خيانةِ الرّبِّ.</span>
<span className="v" id="v40" data-v="40" data-xref="ﺭﻭﻣﻴﺔ 12:1; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:7; اللّاويِّين 11:44–45"> تتَذكَّرونَ وتَعملونَ بجميعِ وصايايَ وتكونونَ مُكرَّسينَ لإلَهِكُم.</span>
<span className="v" id="v41" data-v="41" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:9–10; حِزْقِيال 36:25–27; العِبرانِيّين 11:16"> أنا الرّبُّ إلهُكُم الّذي أخرَجَكُم مِنْ أرضِ مِصْرَ لِيكونَ لكُم إلهًا. أنا الرّبُّ إلهُكُم“.</span>
</div>
