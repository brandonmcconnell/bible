## الإنتصار على الكنعانيين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="القُضاة 1:16; العَدَد 33:40; يَشُوع 12:14"> وسمِعَ الكنعانيُّ ملِكُ عَرادَ المُقيمُ بالجَنوبِ أنَّ بَني إِسرائيلَ جاؤوا على طريقِ أتاريمَ، فقاتَلَهُم وسَبى بَعضَهُم.</span>
<span className="v" id="v2" data-v="2" data-xref="القُضاة 11:30; التَّكوين 28:20; صَمُوئيلَ الأوَّلُ 1:11"> فنَذَر بَنو إِسرائيلَ نَذْرًا للرّبِّ وقالوا: ”إنْ أسلَمْتَ هؤلاءِ القومَ إلى أيدينا أبَحْنا حُرمَةَ مُدُنِهِم“.</span>
<span className="v" id="v3" data-v="3" data-xref="العَدَد 14:45; التَّثْنِيَة 1:44; المَزامِير 10:17"> فسَمِعَ الرّبُّ صوتَ بَني إِسرائيلَ وأسلَمَ الكنعانيِّين إلى أيديهِم فحَلَّلوا إهلاكَهُم، هُم ومُدُنَهُم، وسَمَّوا ذلِكَ الموضِعَ حُرْمةَ.</span>
</div>
</div>

## الحية النحاسية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 6:9; القُضاة 11:18; العَدَد 32:7"> ثُمَّ رحَلوا مِنْ جبَلِ هُورَ على طريقِ البحرِ الأحمرِ ليَدوروا مِنْ حَولِ أرضِ أدومَ، فبَدَأ الشَّعبُ يَلهَثُ في الطَّريقِ.</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 78:19; العَدَد 16:13–14; الخُرُوج 17:2–3"> وتكلَّموا على اللهِ وعلى موسى وقالوا: ”لماذا أصعَدْتَنا مِنْ مِصْرَ لِنموتَ في البرِّيَّةِ؟ فلا خُبزَ لنا ولا ماءَ ونُفوسُنا قضَت مِنْ هذا الطَّعامِ القليلِ“.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:9; إرْمِيا 8:17; التَّثْنِيَة 8:15"> فأرسلَ الرّبُّ على الشَّعبِ حيَّاتٍ ناريَّةً، فلَدَغَتِ الشَّعبَ وماتَ قومٌ كثيرونَ مِنْ بَني إِسرائيلَ.</span>
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 8:24; العَدَد 11:2; المَزامِير 78:34"> فجاءَ الشَّعبُ إلى موسى وقالوا: ”خَطِئنا حينَ تكَلَّمْنا على الرّبِّ وعلَيكَ، فصَلِّ إلى الرّبِّ حتّى يُزيلَ عنَّا الحيَّاتِ“. فصلَّى موسى لأجلِ الشَّعبِ.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 3:14; المَزامِير 106:43–45; المَزامِير 145:8"> فقالَ الرّبُّ لموسى: ”إصنَعْ لكَ حيَّةً واَرفَعْها على ساريةٍ، فكُلُّ مَلدوغٍ يَنظُرُ إليها يَحيا“.</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 3:14–15; العِبرانِيّين 12:2; يُوحَنّا 6:40"> فصَنعَ موسى حيَّةً مِنْ نُحاسٍ ورفَعَها على ساريةٍ، فكانَ أيُّ إنسانٍ لدَغَتْهُ حيَّةٌ ونظَرَ إلى الحيَّةِ النُحاسيَّةِ يَحيا.</span>
</div>
</div>

## من هور إلى وادي موآب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="العَدَد 33:43–45"> ثُمَّ رحَلَ بَنو إِسرائيلَ ونَزَلوا أوبوتَ.</span>
<span className="v" id="v11" data-v="11"> ومِنْ أوبوتَ نزَلوا عَيِيِّ عَبَّاريمَ في البرِّيَّةِ، تُجاهَ مُوآبَ، جِهَةَ مَشرِقِ الشَّمسِ.</span>
<span className="v" id="v12" data-v="12" data-xref="التَّثْنِيَة 2:13–14"> ورحَلوا مِنْ هُناكَ ونزَلوا وادي زارَدَ.</span>
<span className="v" id="v13" data-v="13" data-xref="القُضاة 11:18; العَدَد 22:36; إشَعْياء 16:2"> ثُمَّ رحَلوا مِنْ هُناكَ وعبَروا نهرَ أرنونَ ونزَلوا بالبرِّيَّةِ المُتاخمَةِ أرضَ الأموريِّينَ، لأنَّ نهرَ أرنونَ هوَ الحَدُّ الفاصِلُ بينَ الموآبيِّينَ والأموريِّينَ.</span>
<span className="v" id="v14" data-v="14" data-xref="يَشُوع 10:13; صموئيلَ الثّانِي 1:18"> ولذلِكَ يُقالُ في كتابِ حروبِ الرّبِّ: ”مدينةُ واهِبَ في أرضِ سوفَة وأوديَةِ أرنونَ</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 21:28; التَّثْنِيَة 2:18; التَّثْنِيَة 2:9"> ومصبِّ الأوديةِ الّذي يَميلُ إلى مدينةِ عارَ ويستَنِدُ إلى تُخُمِ مُوآبَ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="العَدَد 20:8; إشَعْياء 12:3; إشَعْياء 41:17–18"> ثُمَّ رحَلوا مِنْ هُناكَ إلى بِئْرٍ قالَ الرّبُّ عِندَها يومًا لموسى: ”إجمَعِ الشَّعبَ حتّى أعطيَهُم ماءً“.</span>
<span className="v" id="v17" data-v="17" data-xref="يَعقُوب 5:13; الخُرُوج 15:1–2; المَزامِير 106:12"> في ذلِكَ الحينِ أنشَدَ إِسرائيلُ هذا النَّشيدَ:<br />”أخْرجي يا بِئْرُ ماءَكِ!<br />غَنُّوا لها.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إشَعْياء 33:22; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:17–18; يَعقُوب 4:12"> بِئْرٌ حفَرَها الرُّؤساءُ<br />ووَسَّعَها أشرافُ الشَّعبِ<br />بالصَّولجانِ والعِصيِّ.<br />عطِيَّةٌ مِنَ البرِّيَّةِ“.<br />ومِنَ البِئْرِ رحَلوا</span>
<span className="v" id="v19" data-v="19"> إلى نَحْليئيلَ، ومِنْ نَحْليئيلَ إلى باموتَ،</span>
<span className="v" id="v20" data-v="20" data-xref="العَدَد 23:28; العَدَد 23:14; التَّثْنِيَة 1:5"> ومِنْ باموتَ في الوادي الّذي في حقلِ موآبَ، تحتَ رأسِ جبَلِ فِسْجَةَ الّذي يُشرِفُ على الصَّحراءِ.</span>
</div>
</div>

## الإنتصار على سيحون وعوج

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التَّثْنِيَة 2:26–28; العَدَد 20:14–19; القُضاة 11:19–21"> وأرسلَ بَنو إِسرائيلَ رُسُلًا إلى سيحونَ، ملِكِ الأموريِّينَ، يقولونَ لَه:</span>
<span className="v" id="v22" data-v="22" data-xref="العَدَد 20:17"> ”دَعنا نعبُرُ أرضَكَ، على أنْ لا نميلَ إلى حقلٍ ولا كرمٍ ولا نشربَ ماءَ بِئْرٍ، وإنَّما نسيرُ في الطَّريقِ العامِ إلى أنْ نعبُرَ أرضَكَ“.</span>
<span className="v" id="v23" data-v="23" data-xref="القُضاة 11:20; التَّثْنِيَة 29:7–8; إشَعْياء 15:4"> فرفَضَ سيحونُ أنْ يعبُرَ بَنو إِسرائيلَ أرضَهُ، وجمَعَ قومَهُ وخرَجَ للقائِهِم إلى البرِّيَّةِ، ووَصَل إلى ياهَصَ وحارَبَهُم هُناكَ.</span>
<span className="v" id="v24" data-v="24" data-xref="عامُوس 2:9; يَشُوع 24:8; التَّكوين 32:22"> فضَربَهُ بَنو إِسرائيلَ بحَدِّ السَّيفِ، واَمْتَلكوا أرضَه مِنْ أرنونَ إلى يَبّوقَ إلى أرضِ بَني عَمُّونَ، لأنَّ تُخُمَ أرضِ بَني عَمُّونَ كانَ مَنيعًا.</span>
<span className="v" id="v25" data-v="25" data-xref="العَدَد 32:33–42; إشَعْياء 16:8–9; إرْمِيا 48:2"> وأخذَ بَنو إِسرائيلَ جميعَ مُدُنِ الأموريِّينَ، ومِنها حَشْبونُ وجميعُ توابِعِها،</span>
<span className="v" id="v26" data-v="26"> لأنَّ حَشْبونَ هيَ مدينَةُ سيحونَ، ملِكِ الأموريِّينَ، وكانَ حارَبَ ملِكَ موآبَ قَبلًا فأخذَ مِنْ يَدِهِ جميعَ أرضِهِ إلى أرنونَ.</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 14:4; العَدَد 21:14; حَبَقُّوق 2:6"> لذلِكَ يقولُ ضاربو الأمثالِ:<br />”أُدْخُلوا حَشْبونَ لتُبنى<br />وتُشَيَّدَ مدينةُ سيحونَ،</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="العَدَد 21:15; إرْمِيا 48:45–46; إشَعْياء 15:1–2"> لأنَّ نارًا خرَجَت مِنْ حَشْبونَ<br />ولهيبًا مِنْ مدينةِ سيحونَ،<br />فأكَلَت مدينةَ عارٍ في موآبَ<br />واَلتهَمَت مَشارفَ أرنونَ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="القُضاة 11:24; المُلُوكِ الثّانِي 23:13; إرْمِيا 48:46"> وَيلٌ لكَ يا شعبَ موآبَ.<br />هَلَكْتَ يا شعبَ الإلهِ كَموشَ.<br />جعَلَ بَنيهِ مُشَرَّدينَ<br />وبَناتِه سَبيًا للملِكِ،<br />لملِكِ الأموريِّينَ سيحونَ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="إرْمِيا 48:18; إشَعْياء 15:2; إرْمِيا 48:22"> لَكِنْ نيرُ الأموريِّينَ بادَ،<br />مِنْ حَشْبونَ بادَ إلى ديبونَ،<br />ومِنْ نوفَ إلى عندِ ميدبا“.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="يَشُوع 13:8–12; العَدَد 32:33–42; التَّثْنِيَة 3:16–17"> فأقامَ بَنو إِسرائيلَ بأرضِ الأموريِّينَ،</span>
<span className="v" id="v32" data-v="32" data-xref="إرْمِيا 48:32; العَدَد 32:1; العَدَد 32:35"> وأرسلَ موسى مَنْ يتَجَسَّسُ مدينةَ يَعزيرَ، فأخَذوا توابِعَها وطَرَدوا الأموريِّينَ الّذينَ هُناكَ.</span>
<span className="v" id="v33" data-v="33" data-xref="يَشُوع 13:12; التَّثْنِيَة 1:4; التَّثْنِيَة 29:7"> ثُمَّ تحَوَّلوا وصَعِدوا في طريقِ باشانَ، فخرَجَ عوجُ ملِكُ باشانَ لمُحاربتِهِم، هوَ وجميعُ قومِهِ، في أذْرعي.</span>
<span className="v" id="v34" data-v="34" data-xref="المَزامِير 135:10–11; يَشُوع 8:7; التَّثْنِيَة 3:2–3"> فقالَ الرّبُّ لموسى: ”لا تَخَفْ مِنْ عوجَ، فأنا أسلَمْتُه إلى يدِكَ هوَ وجميعُ قومِهِ وأرضِهِ، تفعَلُ بهِ كما فعَلْتَ بسيحونَ ملِكِ الأموريِّينَ المُقيمِ في حَشْبونَ“.</span>
<span className="v" id="v35" data-v="35" data-xref="التَّثْنِيَة 29:7–8; يَشُوع 13:12; التَّثْنِيَة 3:3–17"> فضَربَهُ بَنو إِسرائيلَ، هوَ وبَنوه وجميعُ قومِهِ، حتّى لم يَبقَ لَه شريدٌ، واَمْتَلكوا أرضَهُ.</span>
</div>
