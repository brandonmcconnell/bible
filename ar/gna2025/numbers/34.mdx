## حدود الأرض

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="التَّكوين 17:8; المَزامِير 105:11; التَّثْنِيَة 1:7–8"> ”قُلْ لبَني إِسرائيلَ إنَّكُم داخِلونَ أرضَ كنعانَ، وهيَ الأرضُ الّتي أورَثتُكُم إيَّاها، وهذِهِ حُدودُها:</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 14:3; حِزْقِيال 47:13; يَشُوع 3:16"> الحَدُّ الجنوبيُّ يمتَدُّ مِنْ برِّيَّةِ صينَ على جانِبِ أدومَ، فيكونُ مِنْ طَرَفِ البحرِ المَيتِ شرقًا.</span>
<span className="v" id="v4" data-v="4" data-xref="العَدَد 32:8; يَشُوع 15:3–4; العَدَد 13:26"> ثُمَّ يستَديرُ مِنْ جنوبِ عَقبَةِ عَقرِبِّيمَ ويعبُرُ إلى صينَ، وينفُذُ مِنَ الجنوبِ إلى قادِشَ بَرنيعَ، ويخرُجُ إلى حَصَرِ أدَّارَ، ويعبُرُ إلى عَصْمونَ.</span>
<span className="v" id="v5" data-v="5" data-xref="يَشُوع 15:4; التَّكوين 15:18; المُلُوكِ الأوَّلُ 8:65"> ثُمَّ يستَديرُ الحَدُّ مِنْ عَصْمونَ إلى نهر مِصْرَ نافِذًا إلى البحرِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="يَشُوع 1:4; حِزْقِيال 47:20; حِزْقِيال 47:15"> ”وأمَّا الحَدُّ الغربيُّ فيكونُ البحرُ الكبيرُ تُخُمًا. ذلكَ يكونُ لكُم تُخُمَ الغربِ،</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 33:37; العَدَد 34:6; العَدَد 34:3"> وهذا يكونُ لكُم التُّخُمُ الشِّماليُّ: مِنَ البحرِ الكبيرِ تَخُطُّونَ حَدًّا إلى جبَلِ هُورَ،</span>
<span className="v" id="v8" data-v="8" data-xref="العَدَد 13:21; المُلُوكِ الثّانِي 14:25; يَشُوع 13:5–6"> ومِنْ جبَلِ هُورَ تَخُطُّونَ حَدًّا إلى مدخَلِ حَماة وينفُذُ إلى صدَدَ،</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 47:17"> ثُمَّ يخرُجُ إلى زِفْرونَ وينتَهي عندَ حَصَرِ عَينانَ. ذلِكَ يكونُ الحَدُّ الشِّماليُّ.</span>
<span className="v" id="v10" data-v="10"> وتخُطُّونَ الحَدَّ الشَّرقيَّ مِنْ حَصَرِ عَينانَ إلى شفامَ،</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 23:33; التَّثْنِيَة 3:17; يَشُوع 11:2"> ثُمَّ ينحَدِرُ مِنْ شفامَ إلى رِبْلَةَ، شرقيَّ عَينٍ، حتّى يُلامِسَ جانِبَ بحرِ كنَّارة شرقًا،</span>
<span className="v" id="v12" data-v="12" data-xref="العَدَد 34:3; التَّكوين 13:10; التَّكوين 19:24–26"> ومِنْ هُناكَ ينحَدِرُ إلى الأُردُنِّ وينفُذُ إلى البحرِ المَيتِ. تلكَ تكونُ لكُم حُدودَ أرضِكُم مِنْ كُلِّ جِهةٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="العَدَد 34:1; يَشُوع 14:1–5"> فقالَ موسى لبَني إِسرائيلَ: ”تلكَ هي الأرضُ الّتي تأخُذونَ فيها ميراثًا بالقُرعةِ كما أمرَ الرّبُّ أنْ تُعطَى للتِّسعَةِ الأسباطِ ونِصْفِ السِّبْطِ،</span>
<span className="v" id="v14" data-v="14" data-xref="العَدَد 32:33; العَدَد 32:23; يَشُوع 14:2–3"> لأنَّ سِبْطَ بَني رَأوبينَ وسِبْطَ بَني جادَ ونِصْفَ سِبْطِ بَني منَسَّى أخَذوا ميراثَهُم بحسَبِ بُيوتِ آبائِهِم.</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 32:32"> هذانِ السِّبْطانِ ونِصْفُ السِّبْطِ أخَذوا ميراثَهُم مِنْ عَبرِ الأُردُنِّ عندَ أريحا شرقًا“.</span>
</div>
</div>

## قسمة الأرض

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v17" data-v="17" data-xref="يَشُوع 14:1; يَشُوع 19:51; العَدَد 13:8"> ”هذِهِ أسماءُ الرِّجالِ الّذينَ يَقسِمونَ لكُم الأرضَ: ألِعازارُ الكاهنُ، ويَشوعُ بنُ نونٍ،</span>
<span className="v" id="v18" data-v="18" data-xref="العَدَد 1:4–16"> ورئيسٌ مِنْ كُلِّ سِبْطٍ يُساعِدُهُما في ذلِكَ.</span>
<span className="v" id="v19" data-v="19" data-xref="العَدَد 26:65; العَدَد 13:30; العَدَد 14:24"> وهذِهِ أسماؤُهُم: مِنْ سِبْطِ يَهوذا كالِبُ بن يَفُنَّا،</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 49:5"> ومِنْ سِبْطِ بَني شِمعونَ شَموئيلُ بنُ عَمِّيهودَ،</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 49:27; المَزامِير 68:27"> ومِنْ سِبْطِ بنيامينَ أليدادُ بنُ كَسْلونَ،</span>
<span className="v" id="v22" data-v="22"> ومِنْ سِبْطِ بَني دانَ الرَّئيسُ بُقِّي بنُ يُجْلي،</span>
<span className="v" id="v23" data-v="23"> ومِنْ بَني يوسُفَ مِنْ سِبْطِ بَني منَسَّى الرَّئيسُ حَنِّيئيلُ بنُ إيفودَ،</span>
<span className="v" id="v24" data-v="24"> ومِنْ سِبْطِ أفرايِمَ الرَّئيسُ قموئيلُ بنُ شِفْطانَ،</span>
<span className="v" id="v25" data-v="25"> ومِنْ سِبْطِ بَني زَبولونَ الرَّئيسُ أليصافانُ بنُ فَرْناخَ،</span>
<span className="v" id="v26" data-v="26"> ومِنْ سِبْطِ بَني يسَّاكرَ الرَّئيسُ فَلْطِيئِيلُ بنُ عزَّانَ،</span>
<span className="v" id="v27" data-v="27"> ومِنْ سِبْطِ بَني أشيرَ الرَّئيسُ أخيهودُ بنُ شَلومي،</span>
<span className="v" id="v28" data-v="28"> ومِنْ سِبْطِ بَني نَفتالي الرَّئيسُ فَدَهْئيلُ بنُ عَمِّيهودَ“.</span>
<span className="v" id="v29" data-v="29" data-xref="يَشُوع 19:51; العَدَد 34:18"> أولئِكَ هُمُ الّذينَ أمرَهُمُ الرّبُّ أنْ يقسِموا لبَني إِسرائيلَ أرضَ كَنعانَ.</span>
</div>
