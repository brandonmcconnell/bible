## تذمر بني إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّثْنِيَة 1:45; العَدَد 11:1–4"> فرَفَعَ كُلُّ الجماعةِ أصواتَهُم وصرَخوا وبَكَوا في تِلكَ اللَّيلةِ.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 15:24; الخُرُوج 17:3; العَدَد 16:41"> ولامَ جميعُ بَني إِسرائيلَ موسى وهرونَ وقالوا لهُما: ”يا ليتَنا مُتْنا في أرضِ مِصْرَ أو في هذِهِ البَرِّيَّةِ!</span>
<span className="v" id="v3" data-v="3" data-xref="العَدَد 14:31–32; إرْمِيا 9:3; المَزامِير 78:40"> لِماذا جاءَ بِنا الرّبُّ إلى هذِهِ الأرضِ حتّى نسقُطَ تَحتَ السَّيفِ وتصيرَ نِساؤُنا وأطفالُنا غَنيمةً لِلعَدُوِّ؟ أما كانَ خَيرًا لنا أنْ نَرجِعَ إلى مِصْرَ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 7:39; نَحَمْيا 9:16–17; التَّثْنِيَة 17:16"> وقالَ بَعضُهُم لِبَعضٍ: ”نُقيمُ رئيسًا علَينا ونَرجِعُ إلى مِصْرَ“.</span>
<span className="v" id="v5" data-v="5" data-xref="العَدَد 16:4; العَدَد 16:22; العَدَد 16:45"> فوَقعَ موسى وهرونُ على وجهَيهِما تضَرُّعًا أمامَ جماعةِ بَني إِسرائيلَ</span>
<span className="v" id="v6" data-v="6" data-xref="العَدَد 14:30; العَدَد 13:30; العَدَد 14:38"> ومَزَّقَ يَشوعُ بنُ نُونٍ وكالِبُ بنُ يُفَنَّا مِمَّنْ تجَسَّسوا الأرضَ ثيابَهُما</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 13:27; التَّثْنِيَة 1:25; التَّثْنِيَة 8:7–9"> وقالا لِكُلِّ جماعةِ بَني إِسرائيلَ: ”الأرضُ الّتي مَرَرْنا فيها لِنَتَجَسَّسَها أَرضٌ جَيِّدةٌ جِدًّا جِدًّا.</span>
<span className="v" id="v8" data-v="8" data-xref="صموئيلَ الثّانِي 22:20; إشَعْياء 62:4; العَدَد 13:27"> إنْ كانَ الرّبُّ راضيًا علَينا، فهوَ يُدخِلُنا تِلكَ الأرضَ ويَهبُها لنا أرضًا تَدُرُّ لَبَنًا وعسَلًا.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّثْنِيَة 7:18; ﺭﻭﻣﻴﺔ 8:31; التَّثْنِيَة 31:6"> لَكنْ على الرّبِّ لا تتمَرَّدوا ولا تخافوا أهلَ تِلكَ الأرضِ، فهُم لُقمَةٌ سائِغةٌ لنا، وإلهُهُم تخلَّى عَنهُم والربُّ إلهُنا معَنا فلا تَرهَبوهُم“.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 17:4; صَمُوئيلَ الأوَّلُ 30:6; اللّاويِّين 9:23"> فقالَتِ الجماعةُ كُلُّها: ”هيَّا نَرجِمْهُما بِالحجارةِ“. فظَهرَ مَجدُ الرّبِّ في خَيمةِ الاجتِماعِ لجميعِ بَني إِسرائيلَ.</span>
</div>
</div>

## موسى يتشفع بالشعب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 78:32; المَزامِير 106:24; المَزامِير 78:22"> وكلَّمَ الرّبُّ موسى فقالَ: ”إلى متى يَستَهينُ بي هؤلاءِ الشَّعبُ، وإلى متى لا يُؤمِنونَ بي معَ جميعِ المُعجِزاتِ الّتي صَنَعتُها فيما بَينَهُم؟</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 32:10; العَدَد 25:9; صموئيلَ الثّانِي 24:12–15"> سأضرِبُهُم بالوَبَاءِ وأُفنيهِم وأجعَلُكَ أنتَ أمَّةً أعظَمَ وأقوى مِنهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 106:23; حِزْقِيال 20:9; التَّثْنِيَة 32:27"> فقالَ موسى لِلرّبِّ: ”أنتَ أخرَجْتَ هذا الشَّعبَ مِنْ بَينِ المِصْريِّينَ بِقُدرتِكَ، فإذا سَمِعَ المِصْرِيُّونَ ما فعَلْتَ بِشعبِكَ</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 15:14; الخُرُوج 33:11; التَّثْنِيَة 5:4"> يُخبِرونَ أهلَ هذِهِ الأرضِ. وهؤلاءِ سَمِعوا أنَّكَ يا ربُّ حالٌّ بَينَ شعبِكَ الّذينَ ظَهَرتَ لهُم وجهًا إلى وجهٍ، وأقَمتَ سَحابَتَكَ فوقَهُم، وسِرْتَ أمامَهُم بِعَمودِ سَحابٍ نهارًا، وبِعَمود نارٍ ليلًا.</span>
<span className="v" id="v15" data-v="15" data-xref="القُضاة 6:16"> فإذا قَتَلتَهُم عَنْ بِكْرَةِ أبيهِم تقولُ الأُمَمُ الّتي سَمِعَت بِأخبارِكَ تِلكَ:</span>
<span className="v" id="v16" data-v="16" data-xref="التَّثْنِيَة 9:28; يَشُوع 7:9; يَشُوع 7:7"> لم يَقدِرِ الرّبُّ أنْ يُدخِلَ شعبَهُ إلى الأرضِ الّتي حلَفَ لهُم علَيها، فَقَتلَهُم في البَرِّيَّةِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 9:6; ﻣﺘﻰ 9:8; ميخا 3:8"> والآنَ أرِنا يا ربُّ قُدرَتَك واَفْعَلْ ما وَعَدْتَنا بهِ حينَ قُلتَ</span>
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 34:6–7; المَزامِير 145:8; الخُرُوج 20:5"> إنَّكَ أنتَ الرّبُّ البطيءُ عَنِ الغضَبِ، الكثيرُ الرَّحمةِ، الغافِرُ الذَّنبَ والإثمَ، لكنَّكَ لا تُبرِئُ بل تُعاقِبُ البنينَ على ذُنوبِ الآباءِ إلى الجيلِ الثَّالثِ والرَّابعِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 34:9; المَزامِير 78:38; المَزامِير 106:45"> والآنَ فاَصْفَحْ يا ربُّ عَنْ إثْمِ هؤلاءِ الشَّعبِ لِكثرَةِ رَحمَتِكَ، كما غَفَرتَ لهُم مِنْ مِصْرَ إلى هُنا“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="ميخا 7:18–20; يَعقُوب 5:16; المَزامِير 106:23"> فقالَ الرّبُّ: ”صَفَحتُ بِحسَبِ قولِكَ.</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 72:19; حَبَقُّوق 2:14; إشَعْياء 6:3"> لَكنّي أُقسِمُ بِحياتي وبِمَجدي الّذي يَملأُ الأرضَ كُلَّها</span>
<span className="v" id="v22" data-v="22" data-xref="أيُّوب 19:3; التَّكوين 31:7; الخُرُوج 32:1"> لا أحدَ مِنْ جميعِ الّذينَ رَأَوا مَجدي ومُعجِزاتي الّتي صَنَعتُها في مِصْرَ وفي البرِّيَّةِ، وجَرَّبوني عَشْرَ مرَّاتٍ ولم يسمَعوا لِقولي</span>
<span className="v" id="v23" data-v="23" data-xref="العَدَد 32:11; العَدَد 26:64; حِزْقِيال 20:15"> سيَرى الأرضَ الّتي أقسَمتُ علَيها لآبائِهِم. كُلُّ مَنِ اَسْتهانَ بي لن يَراها.</span>
<span className="v" id="v24" data-v="24" data-xref="العَدَد 14:6–9; يَشُوع 14:6–14; أفَسُس 6:6"> وأمَّا عبدي كالِبُ، فبِما أنَّه تَحلَّى بِروحِ الانقيادِ التَّامِ لي، فسأُدخِلُهُ الأرضَ الّتي تجَسَّسَها، ونَسلُهُ يَرِثُها.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 1:40; العَدَد 13:29; المَزامِير 81:11–13"> والآنَ، فالعَمالقةُ والكنعانيُّونَ مُقيمونَ بالأغوارِ، فميلوا في غَدٍ واَرْحَلوا إلى البرِّيَّةِ على طريقِ البحرِ الأحمرِ“.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26"> وكلَّمَ الرّبُّ موسى وهرونَ فقالَ:</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 16:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:10; الخُرُوج 16:28"> ”إلى متى أحتَمِلُ هؤلاءِ القومَ الأشرارَ الّذينَ يُلقونَ اللَّومَ عليَّ؟ سَمِعْتُ لومَ بَني إِسرائيلَ لكَ</span>
<span className="v" id="v28" data-v="28" data-xref="العَدَد 14:21; العِبرانِيّين 3:17; العَدَد 14:2"> فَقُلْ لهُم: حَيٌّ أنا يقولُ الرّبُّ سأصنَعُ بِكُم كما تكَلَّمتُم على مَسامعي.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 1:45; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:5; العِبرانِيّين 3:17"> ففي هذِهِ البرِّيَّةِ تسقُطُ جُثَثُكُم. والمَحصيُّونُ مِنكُم مِنِ اَبْنِ عِشرينَ سنَةً فصاعِدًا، مِمَّنْ ألقَوا اللَّومَ عليَّ</span>
<span className="v" id="v30" data-v="30" data-xref="العَدَد 26:65; العَدَد 14:38; التَّثْنِيَة 1:36–38"> لن يَدخُلوا الأرضَ الّتي رفَعْتُ يَدي مُقْسِمًا أنْ أُسكِنَهُم فيها، ما عدا كالِبَ بنَ يُفَنَّا ويَشوعَ بنَ نُون.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّثْنِيَة 1:39; المَزامِير 106:24; العَدَد 26:64"> وأطفالُكُمُ الّذينَ قُلْتُم إنَّهُم يصيرونَ غنيمةً لِلعَدوِّ سأُدخِلُهُمُ الأرضَ الّتي رفَضتُموها وهُم سيَسكُنونَها.</span>
<span className="v" id="v32" data-v="32" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:5; العَدَد 14:29; العِبرانِيّين 3:17"> وأمَّا جُثَثُكُم أنتُم فتَسقُطُ في البرِّيَّةِ.</span>
<span className="v" id="v33" data-v="33" data-xref="حِزْقِيال 23:35; العَدَد 32:13; التَّثْنِيَة 2:14"> وبَنوكُم يكونونَ رُعاةً في البرِّيَّةِ مُدَّةَ أربعينَ سنَةً، ويتَحمَّلونَ عاقِبةَ خيانَتِكُم إلى أنْ تَفنى آخِرُ جُثَثِكُم في البرِّيَّةِ.</span>
<span className="v" id="v34" data-v="34" data-xref="حِزْقِيال 4:6; العَدَد 13:25; إرْمِيا 18:9–10"> بِعدَدِ الأيّامِ الّتي تجَسَّستُمُ الأرضَ فيها، وهيَ أربعونَ يومًا، تتَحمَّلونَ عاقبَةَ آثامِكُم أربعينَ سنَةً، كُلَّ يومِ بِسنَةٍ، فتَعرفونَ كيفَ تكونُ مُعاداتي لكُم.</span>
<span className="v" id="v35" data-v="35" data-xref="العَدَد 23:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:5; العِبرانِيّين 3:19"> أنا الرّبُّ تَكلَّمتُ، فأفعلُ ذلِكَ بكُلِّ هذِهِ الجماعةِ الشِّرِّيرةِ الّتي تُهَدِّدُني. في هذِهِ البرِّيَّةِ يَنقرضونَ وهُنا يَموتونَ“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="العَدَد 13:31–33; العَدَد 13:4–16"> وأمَّا الرِّجالُ الّذينَ أرسلَهُم موسى لِيتَجَسَّسوا الأرضَ ورَجِعوا وجَعلوا كُلَّ الجماعةِ تلومُه لأنَّهُم أشاعُوا بِرداءَةِ الأرضِ،</span>
<span className="v" id="v37" data-v="37" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:10; العَدَد 16:49; العِبرانِيّين 3:17"> فماتوا بِضَربةٍ أمامَ الرّبِّ.</span>
<span className="v" id="v38" data-v="38" data-xref="يَشُوع 14:6–10; العَدَد 26:65"> ولم يَسلَمْ مِنهُم إلَّا يَشوعُ بنُ نُون وكالِبُ بنُ يُفَنَّا.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="الخُرُوج 33:4; ﻣﺘﻰ 8:12; العِبرانِيّين 12:17"> ولمَّا أخبرَ موسى بَني إِسرائيلَ بما قالَهُ الرّبُّ بَكَوا بُكاءً شديدًا.</span>
<span className="v" id="v40" data-v="40" data-xref="التَّثْنِيَة 1:41; ﻣﺘﻰ 7:21–23; لُوقا 13:25"> ثُمَّ بَكَّروا في الصَّباحِ وصَعِدوا إلى رأسِ الجبَلِ وقالوا: ”ها نحنُ صاعدونَ إلى الموضِعِ الّذي أخبرَنا الرّبُّ عَنهُ، فنَعتَرِفُ بِأنَّنا خَطِئْنا“.</span>
<span className="v" id="v41" data-v="41" data-xref="أخبارِ الأيّامِ الثّانِي 24:20; أيُّوب 4:9; إرْمِيا 2:37"> فقالَ لهُم موسى: ”لماذا تُخالِفونَ أمرَ الرّبِّ؟ فهذا لا يُفيدُ.</span>
<span className="v" id="v42" data-v="42" data-xref="التَّثْنِيَة 1:42; يَشُوع 7:8; يَشُوع 7:12"> لا تَصعَدوا. الرّبُّ لن يكونَ معَكُم وستَنهَزِمونَ أمامَ أعدائِكُم.</span>
<span className="v" id="v43" data-v="43" data-xref="العَدَد 13:29; العَدَد 14:25; أخبارِ الأيّامِ الثّانِي 15:2"> فالعَمالقةُ والكنعانيُّونَ هُناكَ أمامَكُم، فتَسقطونَ في الحربِ لأنَّكُمُ اَرْتَدَدْتُم عَنِ الرّبِّ فلا يكونُ الرّبُّ معَكُم“.</span>
<span className="v" id="v44" data-v="44" data-xref="التَّثْنِيَة 1:43; صَمُوئيلَ الأوَّلُ 4:3–11; العَدَد 31:6"> فتَجَبَّروا وصَعِدوا إلى رأسِ الجبَلِ، لكنَّ موسى وتابوتَ عَهدِ الرّبِّ لم يُغادرا وسَطَ المَحلَّةِ.</span>
<span className="v" id="v45" data-v="45" data-xref="العَدَد 21:3; القُضاة 1:17; التَّثْنِيَة 1:44"> فنَزَلَ العَمالقةُ والكنعانيُّونَ المُقيمونَ بِذلكَ الجبَلِ، فضربوهُم وهَزموهُم إلى حُرمَةَ.</span>
