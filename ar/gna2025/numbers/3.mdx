## بنو هرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 6:20; التَّكوين 10:1; الخُرُوج 6:16"> وهذِهِ مَواليدُ هرونَ وموسى يومَ كلَّمَ الرّبُّ موسى في جبَلِ سيناءَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 6:23; العَدَد 26:60; الخُرُوج 28:1"> هذِهِ أسماءُ بَني هرونَ: نادابُ البِكْرُ، ثُمَّ أبيهو وألعازارُ وإيثامارُ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 28:41; اللّاويِّين 8:30; اللّاويِّين 8:12"> هؤلاءِ مُسِحُوا وكُرِّسوا لِخِدمةِ الكهَنوتِ.</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 10:1–2; العَدَد 26:61"> وماتَ نادابُ وأبيهو لأنَّهما قرَّبا نارًا غَيرَ مُقدَّسةٍ أمامَ الرّبِّ في برِّيَّةِ سيناءَ، وما كانَ لهما بَنونَ. وكانَ ألعازارُ وإيثامارُ يقومانِ بِخدمةِ الكهَنوتِ بِمحضَرِ هرونَ أبيهِما.</span>
</div>
</div>

## اللّاويّون خدّام للكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 10:8; العَدَد 2:33; العَدَد 1:49–53"> ”قَدِّمْ سِبْطَ لاوي لِيَقِفوا بَينَ يدَيِ هرونَ الكاهنِ ويَخدِموهُ</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 1:50; العَدَد 8:15; العَدَد 8:11"> ويتَوَلَّوا عَنهُ وعَنِ الجماعةِ حِراسةَ خَيمةِ الاجتِماعِ ويَقوموا بِخدمةِ المَسكِنِ،</span>
<span className="v" id="v8" data-v="8" data-xref="العَدَد 4:33; العَدَد 10:17; إشَعْياء 52:11"> ويُحافِظوا على جميعِ أمتِعَةِ خَيمةِ الاجتِماعِ، ويَنوبوا عَنْ بَني إِسرائيلَ في ما هوَ مطلوبٌ مِنهُم في خِدمةِ المَسكِنِ.</span>
<span className="v" id="v9" data-v="9" data-xref="العَدَد 8:19; العَدَد 18:6–7; أفَسُس 4:8"> وأوكِلِ اللَّاويِّينَ إلى هرونَ وبَنيهِ، فَهُم مَوهوبونَ لَه هِبَةً مِنْ بَينِ بَني إِسرائيلَ.</span>
<span className="v" id="v10" data-v="10" data-xref="العَدَد 1:51; العَدَد 3:38; ﺭﻭﻣﻴﺔ 12:7"> وخَصِّصْ هرونَ وبَنيهِ لِخِدمةِ الكهَنوتِ، ومَنِ اَقْتَرَبَ لِخِدمتِهِ سِواهُم يُقْتَلْ“.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v12" data-v="12" data-xref="العَدَد 3:41; العَدَد 8:16; العَدَد 18:6"> ”أخذْتُ اللَّاويِّينَ مِنْ بَني إِسرائيلَ بدَلَ كُلِّ بِكْرٍ فاتِحِ رَحِم مِنْ بَني إِسرائيلَ، فيكونُ اللَّاويُّونَ لي،</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 2:23; الخُرُوج 13:12; الخُرُوج 13:2"> لأنَّ كلَّ بِكْرٍ هوَ لي، يومَ أهلَكْتُ كُلَّ بِكْرٍ في أرضِ مِصْرَ. قدَّسْتُ لي كُلَّ بِكْرٍ في إِسرائيلَ مِنَ النَّاسِ والبَهائمِ. فَهُم لي يكونونَ أنا الرّبُّ“.</span>
</div>
</div>

## إحصاء بني لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14"> وكلَّمَ الرّبُّ موسى في برِّيَّةِ سيناءَ فقالَ:</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 26:62; مَرْقُس 10:14; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:15"> ”تَعُدُّ بَني لاوي بِحسَبِ عائلاتِهِم وعشائرِهِم، كُلُّ ذَكَرٍ مِنْ اَبنِ شهرٍ فصاعِدًا“.</span>
<span className="v" id="v16" data-v="16" data-xref="العَدَد 3:39; العَدَد 4:37; العَدَد 4:49"> فعَدَّهُم موسى كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّكوين 46:11; الخُرُوج 6:16–19; العَدَد 15:5–23"> وهؤلاءِ بَنو لاوي بأسمائِهِم: جَرشونُ وقَهاتُ ومَراري.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الأوّلُ 6:17; أخبارِ الأيّامِ الأوّلُ 26:1–32; الخُرُوج 6:17–19"> وهذانِ اَسْمَا اَبنَي جَرشونَ بِحسَبِ عَشائرِهِما: لِبْني وشِمْعي.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 6:18; أخبارِ الأيّامِ الأوّلُ 6:18; أخبارِ الأيّامِ الأوّلُ 15:17–21"> وبَنو قَهاتَ بِحسَبِ عشائرِهِم: عَمْرامُ ويصهارُ وحبرونُ وعُزِّيئيلُ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 6:19; أخبارِ الأيّامِ الأوّلُ 6:19; العَدَد 3:33"> واَبنا مَراري بِحسَبِ عشائرِهِما: مَحْلي وموشي. تِلكَ عشائِرُ اللَّاويِّينَ بِحسَبِ عائلاتِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21"> لجَرشونَ عشيرةُ لِبْني وعَشيرةُ شِمْعي،</span>
<span className="v" id="v22" data-v="22" data-xref="العَدَد 4:38–40"> والمَعدُودونَ مِنهُما بإحصاءِ كُلِّ ذَكَرٍ مِنِ اَبنِ شهرٍ فصاعِدًا سَبعَةُ آلافٍ وخَمسُ مئةٍ.</span>
<span className="v" id="v23" data-v="23" data-xref="العَدَد 1:53; العَدَد 2:17"> وهاتانِ العَشيرَتانِ تَنزِلانِ وراءَ المَسكِنِ جِهَةَ الغربِ،</span>
<span className="v" id="v24" data-v="24"> والرَّئيسُ علَيها ألياسافُ بنُ لابِلَ.</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 25:9; أخبارِ الأيّامِ الأوّلُ 23:32; أخبارِ الأيّامِ الأوّلُ 9:14–33"> وما يَحرُسُه بَنو جِرشونَ في خَيمةِ الاجتِماعِ هوَ المَسكِنُ، والخَيمَةُ وغِطاؤُها، وسِتارُ بابِ خَيمةِ الاجتِماعِ،</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 35:18; الخُرُوج 38:9–16; الخُرُوج 27:9–16"> وسُجوفُ الرِّواقِ وسِتارُ بابِهِ الّذي حَولَ المَسكِن وحَولَ المذبَحِ مُحيطًا بِهِما، والحِبالُ الّتي لتِلكَ كُلِّها.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="أخبارِ الأيّامِ الأوّلُ 26:23; العَدَد 3:19; أخبارِ الأيّامِ الأوّلُ 23:12"> ولِقَهاتَ عَشيرةُ العِمراميِّين وعَشيرَةُ اليِصْهاريِّينَ وعَشيرةُ الحَبرونيِّينَ وعَشيرةُ العُزِّيئيليِّينَ،</span>
<span className="v" id="v28" data-v="28" data-xref="العَدَد 3:31; العَدَد 4:35–36; العَدَد 3:7"> والمَعدُودونَ مِنها بإحصاءِ كُلِّ ذَكَرٍ مِنْ اَبنِ شهرٍ فصاعِدًا ثمانيةُ آلافٍ وسِتُّ مئةٍ. وهُم يَقومونَ بِحراسةِ الأواني المُقدَّسةِ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 1:53; العَدَد 3:23; العَدَد 2:10"> ويَنزِلونَ إلى جانبِ المَسكِنِ جِهةَ الجنوبِ،</span>
<span className="v" id="v30" data-v="30"> والرَّئيسُ علَيهم أليصافانُ بنُ عُزِّيئيلَ.</span>
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 36:35–36; الخُرُوج 27:1–8; الخُرُوج 26:31–33"> وما يَحرِسونَه هوَ تابوتُ العَهدِ والمائِدةُ والمَنارةُ والمذبَحانِ وأمتِعةُ القُدْسِ الّتي يَخدِمونَ بِها والسِّتارُ وما يتعَلَّقُ بِخِدمةِ تِلكَ كُلِّها.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="العَدَد 4:27; العَدَد 4:16; العَدَد 20:25–28"> ورئيسُ رؤساءِ اللَّاويِّينَ، ألعازارُ بنُ هرونَ الكاهنِ، يكونُ رقيبًا على القائِمينَ بِحِراسةِ المَقْدِسِ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="العَدَد 3:20; أخبارِ الأيّامِ الأوّلُ 6:19; أخبارِ الأيّامِ الأوّلُ 23:21"> ولِمَراري عَشيرةُ المَحْلَوِيِّينَ وعَشيرةُ الموشَوِيِّينَ،</span>
<span className="v" id="v34" data-v="34" data-xref="العَدَد 2:9; العَدَد 2:11; العَدَد 1:21"> المَعدُودونَ مِنهُما بإحصاءِ كُلِّ ذَكَرٍ مِنِ اَبنِ شهرٍ فصاعِدًا سِتَّةُ آلافٍ ومِئتانِ.</span>
<span className="v" id="v35" data-v="35" data-xref="العَدَد 1:53; العَدَد 2:25; العَدَد 3:28–29"> والرَّئيسُ علَيهِم صُورِيئيلُ بنُ أبيحائيلَ، وهُم ينزِلونَ إلى جانِبِ المَسكِنِ جِهةَ الشَّمالِ.</span>
<span className="v" id="v36" data-v="36" data-xref="الخُرُوج 26:32; الخُرُوج 26:37; الخُرُوج 26:15–29"> وما يتَوكَّلُ بَنو مَراريَ على حِراستِهِ هوَ ألواحُ المَسكِنِ وعوارِضُهُ وأعمِدتُهُ وقواعِدُه وكُلُّ أمتِعَتِه وما يتعلَّقُ بِخِدمةِ تِلكَ كُلِّها،</span>
<span className="v" id="v37" data-v="37"> وأعمِدَةُ الرِّواقِ الّتي حَوالَيهِ وقواعِدُها وأوتَادُها وحِبالُها.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="العَدَد 3:10; العَدَد 1:53; العَدَد 2:3"> وينزِلُ أمامَ المَسكِنِ، تُجاهَ خيمةِ الاجتِماعِ جِهةَ المشرِقِ، موسى وهرونُ وبَنوهُ لِحراسةِ المكانِ المُقَدَّسِ عَنْ بَني إِسرائيلَ. ومَنِ اَقتَرَبَ مِنهُ سِواهُم يُقتَلُ.</span>
<span className="v" id="v39" data-v="39" data-xref="العَدَد 26:62; ﻣﺘﻰ 7:14; العَدَد 4:47–48"> جميعُ عدَدِ اللَّاويِّينَ الّذينَ عَدَّهُم موسى وهرونُ كما أمرَ الربُّ، على حسَبِ عشائرِهِم، مِنِ اَبنِ شهرٍ فصاعِدًا، اَثنانِ وعِشرونَ ألفًا.</span>
</div>
</div>

## اللّاويّون فداء كل بكر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="العَدَد 3:15; لُوقا 10:20; المَزامِير 87:6"> وكلَّمَ الرّبُّ موسى فقالَ: ”تَعُدُّ كُلَّ بِكْرٍ ذَكَرٍ مِنْ بَني إِسرائيلَ مِنِ اَبنِ شهرٍ فصاعِدًا وتُسَجِّلُهُم بأسمائِهِم.</span>
<span className="v" id="v41" data-v="41" data-xref="العَدَد 3:12; العَدَد 3:45; العَدَد 8:16"> واَجْعَلِ اللَّاويِّينَ لي أنا الرّبُّ بدَلَ كُلِّ بِكْرٍ مِنْ بَني إِسرائيلَ، واَجْعَلْ بَهائِمَ اللَّاويِّينَ بدَلَ كُلِّ بِكْرٍ مِنْ بَهائِمِ بَني إِسرائيلَ“.</span>
<span className="v" id="v42" data-v="42"> فعَدَّ موسى كُلَّ بِكْرٍ مِنْ بَني إِسرائيلَ، كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v43" data-v="43" data-xref="العَدَد 3:39"> فكانَت جُملَةُ المَعدُودينَ مِنَ الأبكارِ الذُّكورِ المُسَجَّلينَ بأسمائِهِم مِنِ اَبنِ شهرٍ فصاعِدًا اَثنَينِ وعِشرينَ ألفًا ومئَتَينِ وثَلاثةً وسَبعينَ.</span>
</div>
<div className="bp">
<span className="v" id="v44" data-v="44"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v45" data-v="45" data-xref="العَدَد 3:12; العَدَد 3:40–41"> ”إجْعَلِ اللَّاويِّينَ بدَلَ كُلِّ بِكْرٍ مِنْ بَني إِسرائيلَ وبَهَائِمَ اللَّاويِّينَ بدَلَ بَهائمِهِم فيصيرَ اللَّاويُّونَ لي أنا الرّبُّ.</span>
<span className="v" id="v46" data-v="46" data-xref="الخُرُوج 13:13; العَدَد 18:15–16"> وأمَّا فِداءُ المئَتَينِ والثَّلاثةِ والسَّبعينَ الزَّائِدينَ على اللَّاويّينَ مِنْ بُكورِ بَني إِسرائيلَ</span>
<span className="v" id="v47" data-v="47" data-xref="الخُرُوج 30:13; اللّاويِّين 27:6; العَدَد 18:16"> فتأخُذُ خمسةَ مَثاقيلَ مِنَ الفِضَّةِ لِكُلِّ نفْسٍ مِنهُم، كُلُّ مِثقالٍ بالسِّعرِ الرَّسميِّ عِشرونَ دِرهَمًا،</span>
<span className="v" id="v48" data-v="48"> وتدفعُ الفِضَّةَ إلى هرونَ وبَنيه فِداءَ الزَّائِدينَ علَيهِم“.</span>
<span className="v" id="v49" data-v="49"> فأخذَ موسى الفِضَّةَ مِنَ الزَّائِدينَ على الّذينَ اَفْتَداهُمُ اللَّاويُّونَ</span>
<span className="v" id="v50" data-v="50" data-xref="العَدَد 3:46–47; رسالَةُ بُطرُسَ الأُولَى 3:18; ﻣﺘﻰ 20:28"> مِنْ بُكُورِ بَني إِسرائيلَ. أخذَها موسى ألفًا وثلاثَ مئةٍ وخمسةً وسِتِّينَ مِثقالًا بالوَزنِ الرَّسميِّ</span>
<span className="v" id="v51" data-v="51" data-xref="العَدَد 3:48; العَدَد 16:15; صَمُوئيلَ الأوَّلُ 12:3–4"> ودفَعَها إلى هرونَ وبَنيهِ، كما أمرَ الرّبُّ.</span>
</div>
</div>
