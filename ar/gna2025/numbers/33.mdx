## الرحيل من مصر إلى موآب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المَزامِير 77:20; ميخا 6:4; الخُرُوج 12:51"> هذِهِ مَراحِلُ بَني إِسرائيلَ بعدَما خرَجوا مِنْ أرضِ مِصْرَ بجيوشِهِم، على يَدِ موسى وهرونَ.</span>
<span className="v" id="v2" data-v="2" data-xref="العَدَد 9:17–23; التَّثْنِيَة 10:11; العَدَد 10:6"> كتَبَها موسى مرحلَةً مرحلَةً بعدَ خُروجِهِم، كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 14:8; الخُرُوج 12:37; الخُرُوج 13:4"> رحَلوا مِنْ مدينةِ رَعْمَسيسَ في اليومِ الخامسَ عشَرَ مِنَ الشَّهرِ الأوَّلِ، في غَدِ الفِصْحِ. خرجوا مُنتَصِرينَ أمامَ مشهدٍ مِنْ جميعِ المِصْريِّينَ،</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 12:12; إشَعْياء 19:1; الخُرُوج 18:11"> فيما كانوا يَدفُنونَ أبكارَهُمُ الّذينَ ضرَبَهُمُ الرّبُّ وأجرى حُكمَه بآلهَتِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 12:37"> فرحَلَ بَنو إِسرائيلَ مِنْ رَعْمَسيسَ ونزَلوا بسكُّوتَ.</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 13:20"> واَرتحَلوا مِنْ سكُّوتَ ونَزَلوا بإيثامَ الّتي هي في طَرَفِ البرِّيَّةِ.</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 14:2; الخُرُوج 14:9; العَدَد 33:8"> ورحَلوا مِنْ إيثامَ ورجَعوا إلى فَمِ الحيروثِ الّتي تُجاهَ بَعْلَ صَفونَ، ونزَلوا أمامَ مَجْدَلَ.</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 15:22–26; الخُرُوج 14:21–31"> ورحَلوا مِنْ فَمِ الحيروثِ، وعبَروا البحرَ الأحمرَ إلى برِّيَّةِ شورَ، وساروا مسافةَ ثَلاثَةِ أيّامٍ في برِّيَّةِ إيثامَ ونزَلوا بمارَّةَ.</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 15:27"> ورحَلوا مِنْ مارَّةَ إلى إيليمَ، وفي إيليمَ اَثنَتا عَشْرَةَ عينِ ماءٍ وسَبعونَ نَخلَةً، فنزَلوا هُناكَ.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 16:1; الخُرُوج 17:1"> واَرتحَلوا مِنْ إيليمَ ونزَلوا على البحرِ الأحمرِ.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 16:1"> واَرتحَلوا مِنَ البحرِ الأحمرِ ونزَلوا ببرِّيَّةِ سينَ.</span>
<span className="v" id="v12" data-v="12"> ورحَلوا مِنْ برِّيَّةِ سينَ ونزَلوا بدُفقَةَ.</span>
<span className="v" id="v13" data-v="13"> واَرتحَلوا مِنْ دُفقَةَ ونزَلوا بألوشَ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 17:1–8; الخُرُوج 19:2"> ورحَلوا مِنْ ألوشَ فنَزَلوا برَفيديمَ، ولم يكُنْ هُناكَ ماءٌ للشَّعبِ يشربونَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 19:1–2; الخُرُوج 16:1"> ورحَلوا مِنْ رَفيديمَ ونزَلوا على التَّوالي ببرِّيَّةِ سيناءَ، وقَبْروتَ هتَّأوَة وحَصيروتَ، ورِثْمَة، ورِمُّونَ فارَصَ، ولِبْنَة، ورِسَّة، وقُهْيلاتَة، وجَبَلِ شافَرَ، وحَرادَة، ومَقْهَيْلوت، وتاحَت، وتارَحَ، ومِثْقَة، وحَشْمونَةَ، ومَسيروتَ، وبَني يَعْقانَ، وحورَ الجِدْجادِ، ويُطْباتَ، وعَبْرونَةَ، وعِصْيونَ جابَرَ، وبرِّيَّةِ صينَ وهي قادِشُ، وجبَلِ هُورَ في طَرَفِ أرضِ أدومَ.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="التَّثْنِيَة 10:6; التَّثْنِيَة 32:50; العَدَد 20:24–28"> فصَعِدَ هرونُ إلى جبَلِ هُورَ، بأمرِ الرّبِّ، وماتَ هُناكَ في السَّنةِ الأربعينَ لخُروجِ بَني إِسرائيلَ مِنْ أرضِ مِصْرَ، في اليومِ الأوَّلِ مِنَ الشَّهرِ الخامِسِ.</span>
<span className="v" id="v39" data-v="39"> وكانَ هرونُ اَبنَ مئةٍ وثَلاثٍ وعِشرينَ سنَةً حينَ ماتَ في جبَلِ هُورَ.</span>
<span className="v" id="v40" data-v="40" data-xref="العَدَد 21:1–9"> وسَمِعَ الكنعانيُّ مَلِكُ عَرادَ بمَجيءِ بَني إِسرائيلَ وهوَ ساكِنٌ في الجنوبِ في أرضِ كنعانَ.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="العَدَد 21:4"> ورحَلوا مِنْ جبَلِ هُورَ ونزَلوا على التَّوالي بِصَلْمونَةَ، وفونونَ، وأوبوتَ، وتلالِ العَباريمِ في حُدودِ موآبَ، وديبونِ جادَ، وعَلْمونِ دِبلاتايِمَ، وجبالِ عباريمَ تُجاهَ نَبو، وسَهلِ موآبَ عَبرَ الأُردُنِّ مِنْ أريحا، بينَ بَيتِ يَشيموتَ وآبَلِ شِطِّيمَ.</span>
</div>
</div>

## وصايا الرّبّ قبل عبور الأردن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v50" data-v="50" data-xref="العَدَد 33:48–49"> وكلَّمَ الرّبُّ موسى في سَهلِ موآبَ عَبرَ الأُردُنِّ مِنْ أريحا فقالَ:</span>
<span className="v" id="v51" data-v="51" data-xref="يَشُوع 3:17; التَّثْنِيَة 9:1; التَّثْنِيَة 7:1"> ”قُلْ لبَني إِسرائيلَ: ستَعبُرونَ الأُردُنَّ إلى أرضِ كنعانَ،</span>
<span className="v" id="v52" data-v="52" data-xref="الخُرُوج 23:24; اللّاويِّين 26:1; التَّثْنِيَة 20:16–18"> فتَطرُدونَ جميعَ سُكَّانِها مِنْ أمامِكُم، وتُبيدونَ جميعَ مَنقوشاتِهِم وأصنامِهِم المسبوكَةِ، وتَهدِمونَ مَعابدَ آلهَتهِمِ المُرتَفِعةِ.</span>
<span className="v" id="v53" data-v="53" data-xref="التَّثْنِيَة 11:31; المَزامِير 115:16; التَّثْنِيَة 32:8"> وتَملِكونَ الأرضَ وتُقيمونَ بها لأنِّي أعطَيتُها لكُم ميراثًا.</span>
<span className="v" id="v54" data-v="54" data-xref="العَدَد 26:53–56; يَشُوع 18:11–48; يَشُوع 15:1–12"> تَرِثونَ الأرضَ بالقُرعَةِ على حسَبِ عَشائرِكُم. الكَثيرُ تُكثِّرونَ لَه نصيبَهُ، والقليلُ تُقَلِّلونَ لَه، وما أصابَ أحدَكُم بالقُرعَة يكونُ لَه ميراثًا على حسَبِ أسباطِ آبائِكُم.</span>
<span className="v" id="v55" data-v="55" data-xref="القُضاة 2:3; المَزامِير 106:34–36; التَّثْنِيَة 7:16"> وإنْ لم تَطرُدوا أهلَ الأرضِ مِنْ أمامِكُم، يَكونوا مَنْ تُبْقونَهُ مِنهُم كمِخْرَزٍ في عُيونِكُم وكشوكةٍ في خَواصِرِكُم، يُضايِقونَكُم في الأرضِ الّتي أنتُم مُقيمونَ بها.</span>
<span className="v" id="v56" data-v="56" data-xref="لُوقا 21:23–24; التَّثْنِيَة 28:63; اللّاويِّين 20:23"> فأفعَلُ بِكُم كما نَوَيتُ أنْ أفعَلَ بِهِم“.</span>
</div>
