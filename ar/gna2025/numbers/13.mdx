## إرسال الجواسيس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 1:22–25; العَدَد 32:8; العَدَد 34:18"> ”تُرسِلُ رِجالًا يتَجَسَّسونَ أرضَ كنعانَ الّتي أعطَيتُها لبَني إِسرائيلَ. تُرسِلُهُم رَجُلًا واحدًا مِنْ كُلِّ سِبْطٍ ويكونونَ كُلُّهم رُؤساءَ أسباطِهِم“.</span>
<span className="v" id="v3" data-v="3" data-xref="العَدَد 12:16; العَدَد 32:8; التَّثْنِيَة 1:19"> فأرسَلَهُم موسى مِنْ برِّيَّةِ فارانَ، كما قالَ الرّبُّ، كُلُّهم مِنْ رُؤساءِ أسباطِ بَني إِسرائيلَ.</span>
<span className="v" id="v4" data-v="4"> وهذِهِ أسماؤُهُم: مِنْ سِبْطِ رَأوبينَ شَمُّوعُ بنُ زَكُّورَ،</span>
<span className="v" id="v5" data-v="5"> ومِنْ سِبْطِ شِمعونَ شافاطُ بنُ حوري،</span>
<span className="v" id="v6" data-v="6" data-xref="العَدَد 34:19; العَدَد 14:6; العَدَد 13:30"> ومِنْ سِبْطِ يَهوذا كالِبُ بنُ يَفُنَّا،</span>
<span className="v" id="v7" data-v="7"> ومِنْ سِبْطِ يسَّاكِرَ يَجْآلُ بنُ يوسُفَ،</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 24:13; العَدَد 11:28; العَدَد 13:16"> ومِنْ سِبْطِ أفرايمَ هوشَعُ بنُ نُونٍ،</span>
<span className="v" id="v9" data-v="9"> ومِنْ سِبْطِ بنيامينَ فَلْطي بنُ رافو.</span>
<span className="v" id="v10" data-v="10"> ومِنْ سِبْطِ زَبولونَ جَدِّيئيلُ بنُ سُودي،</span>
<span className="v" id="v11" data-v="11"> ومِنْ سِبْطِ منَسَّى بنِ يوسُفَ جِدِّي بنُ سُوسي،</span>
<span className="v" id="v12" data-v="12"> ومِنْ سِبْطِ دانَ عَمِّيئيلُ بنُ جَمَلِّي،</span>
<span className="v" id="v13" data-v="13"> ومِنْ سِبْطِ أشيرَ سَتورُ بنُ ميخائيلَ،</span>
<span className="v" id="v14" data-v="14"> ومِنْ سِبْطِ نَفتالي نَحْبي بنُ وَفْسي،</span>
<span className="v" id="v15" data-v="15"> ومِنْ سِبْطِ جادَ جأُوئيلُ بنُ ماكي.</span>
<span className="v" id="v16" data-v="16" data-xref="العَدَد 13:8; الخُرُوج 17:9; العَدَد 14:6"> تِلكَ أسماءُ الرِّجالِ الّذينَ أرسَلَهُم موسى ليَتَجَسَّسوا أرضَ كنعانَ، وسَمَّى موسى هوشَعَ بنَ نُونَ يَشوعَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="القُضاة 1:9; التَّكوين 12:9; القُضاة 1:19"> وأرسَلَهُم موسى ليَتَجَسَّسوا أرضَ كنعانَ وقالَ لهُم: ”تَطلَعونَ مِنْ هُناكَ مِنْ جَنوبي أرضِ كنعانَ، وتصعَدونَ الجبَلَ</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 34:14; الخُرُوج 3:8"> وتنظُرونَ إلى الأرضِ كيفَ هيَ، والشَّعبِ المُقيمِ بِها أشَديدٌ هوَ أم ضعيفٌ؟ أقليلٌ أم كثيرٌ؟</span>
<span className="v" id="v19" data-v="19"> وكيفَ الأرضُ الّتي هوَ مُقيمٌ بِها، أجَيِّدةٌ هيَ أم رَديئةٌ؟ وما المُدُنُ، أخيامٌ أم حُصونٌ؟</span>
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 31:23; التَّثْنِيَة 31:6–8; حِزْقِيال 34:14"> وكيفَ الأرضُ، أمُخْصِبةٌ هيَ أم عقيمةٌ؟ أفيها شجَرٌ أم لا؟ وتَشجَّعوا وخُذوا مِنْ ثَمَرِها. وكانَت في ذلِكَ الوقتِ أيّامُ بَواكيرِ العِنبِ“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="العَدَد 20:1; العَدَد 33:36; العَدَد 27:14"> فصَعِدوا وتجَسَّسُوا الأرضَ مِنْ بَرِّيَّةِ صينَ إلى رَحوبَ عِندَ مدخلِ حماةٍ.</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 78:12; المَزامِير 78:43; إشَعْياء 19:11"> صَعِدُوا أوَّلًا مِنَ الجنوبِ وجاؤوا إلى حبرونَ، وكانَ هُناكَ أخيمانُ وشيشاي وتَلمايُ بَنو عناقَ، وكانَتْ حبرونُ بُنِيَت قَبلَ صُوعَنَ بِمِصْرَ بِسَبْعِ سِنينَ.</span>
<span className="v" id="v23" data-v="23" data-xref="العَدَد 13:24; العَدَد 32:9; التَّثْنِيَة 1:24–25"> ثُمَّ هبَطوا وادي أشكولَ، وقَطَعوا مِنْ هُناك غُصنًا بِعُنقودٍ واحدٍ مِنَ العِنبِ وحَمَلوهُ لِثقْلِه بِعَتَلَةٍ فيما بَينَ اَثْنَينِ مِنهُم، معَ شيءٍ مِنَ الرُّمَّانِ والتِّينِ.</span>
<span className="v" id="v24" data-v="24"> فسُمِّيَ الموضِعُ وادي أشكولَ بِسبَبِ العُنقودِ الّذي قطَفَهُ مِنْ هُناكَ بَنو إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 34:28; الخُرُوج 24:18; العَدَد 14:33–34"> ورَجَعوا مِنْ تَجسُّسِ الأرضِ بَعدَ أربَعينَ يومًا،</span>
<span className="v" id="v26" data-v="26" data-xref="العَدَد 32:8; العَدَد 20:1; التَّثْنِيَة 1:19"> وساروا حتّى جاؤوا إلى موسى وهرونَ وكُلِّ جماعةِ بَني إِسرائيلَ في برِّيَّةِ فارانَ، في قادِشَ، ورَدُّوا خبَرًا علَيهِم وأرَوهُم ثمَرَ الأرضِ.</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 3:8; الخُرُوج 3:17; الخُرُوج 13:5"> وقَصُّوا علَيهِم وقالوا: ”ذهَبْنا إلى الأرضِ الّتي أرسَلْتَنا إليها، فإذا هيَ بالحقيقةِ تَدُرُّ لَبَنًا وعسَلًا، وهذا ثمَرُها،</span>
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 9:1–2; التَّثْنِيَة 1:28; العَدَد 13:33"> غَيرَ أنَّ الشَّعبَ السَّاكنينَ فيها أقوياءُ والمُدُنَ حصينةٌ عظيمةٌ جِدًّا، ورَأينا هُناكَ بَني عَناقَ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 14:43; القُضاة 6:3; التَّكوين 15:19–21"> فيها العَمالقةُ مُقيمونَ بأرضِ الجنوبِ، والحِثِّيُّونَ واليبوسيُّونَ والأموريُّونَ مُقيمونَ بِالجبَلِ، والكنعانيُّونَ مُقيمونَ عِندَ البحرِ وعلى مَجرى الأُردُنِّ“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="العَدَد 14:24; ﺭﻭﻣﻴﺔ 8:31; فِيلِبِّي 4:13"> وكانَ كالِبُ يَزجُرُ الشَّعبَ ويَرُدُّهُ عَنْ موسى ويقولُ: ”بل نصعَدُ ونرِثُ الأرضَ لأنَّنا قادِرونَ علَيها“.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّثْنِيَة 1:28; يَشُوع 14:8; العِبرانِيّين 3:19"> وأمَّا الّذينَ ذهَبوا لِلتَّجَسُّسِ معَهُ فقالوا: ”لا نَقدِرُ أنْ نصعَدَ إلى هُناكَ لأنَّ القَومَ أقوى مِنَّا“.</span>
<span className="v" id="v32" data-v="32" data-xref="عامُوس 2:9; العَدَد 14:36–37; حِزْقِيال 36:13–14"> وأشاعوا عِندَ بَني إِسرائيلَ بِرَداءَةِ الأرضِ الّتي تَجَسَّسوها وقالوا: ”الأرضُ الّتي مَرَرْنا فيها لِنَتَجَسَّسَها هيَ أرضٌ تأكُلُ أهلَها، وجميعُ الشَّعبِ الّذينَ رَأيناهُم فيها أُناسٌ طِوالُ القاماتِ.</span>
<span className="v" id="v33" data-v="33" data-xref="التَّثْنِيَة 1:28; إشَعْياء 40:22; التَّكوين 6:4"> وشاهَدْنا هُناكَ مِنَ الجَبابرةِ جَبابرةَ بَني عَناقَ، فَصِرْنا في نظَرِنا صِغارًا كالجَرادِ، وكذلِكَ في نظَرِهِم“.</span>
</div>
