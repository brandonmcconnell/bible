## مدن بني لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العَدَد 22:1; العَدَد 31:12; العَدَد 33:50"> وكلَّمَ الرّبُّ موسى في سَهلِ موآبَ، عَبرَ الأُردُنِّ مِنْ أريحا، فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="يَشُوع 14:3–4; حِزْقِيال 48:8; يَشُوع 21:2–42"> ”قُلْ لبَني إِسرائيلَ أنْ يُعطوا اللَّاويِّينَ مِنْ ميراثِ مُلكِهِم مُدُنًا يسكُنونَها ومراعيَ حولَ المُدُنِ،</span>
<span className="v" id="v3" data-v="3" data-xref="أخبارِ الأيّامِ الثّانِي 11:14; يَشُوع 21:11; حِزْقِيال 45:2"> لبهائمِهِم ومَواشيهِم وكُلِّ حَيواناتِهِم</span>
<span className="v" id="v4" data-v="4"> والمَراعي الّتي تُعطونَها للَّاويِّينَ تكونُ ألفَ ذِراعٍ مِنْ سورِ المدينةِ إلى خارِجٍ على مُحيطِها.</span>
<span className="v" id="v5" data-v="5"> فتكونُ مَساحةُ الجانِبِ الشَّرقيِّ مِنْ خارِجِ المدينةِ ألفَي ذِراعٍ ومَساحةُ الجانِبِ الجنوبي ألفَي ذِراعٍ، ومَساحةُ الجانِبِ الغربي ألفَي ذِراعٍ، ومَساحةُ الجانِبِ الشِّماليِّ ألفَي ذِراعٍ، والمدينةُ في الوسَطِ. تلكَ هي مَساحةُ المَراعي</span>
<span className="v" id="v6" data-v="6" data-xref="يَشُوع 21:13; يَشُوع 21:3; المَزامِير 9:9"> والمُدُنُ الّتي تُعطونَها للَّاويِّينَ هيَ ستُّ مُدُنٍ تكونُ ملجأً يهرُبُ إليها القاتِلُ، واَثْنَتانِ وأربَعونَ مدينةً،</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 21:3–42; أخبارِ الأيّامِ الأوّلُ 6:54–81"> فتصيرُ جميعُ المُدُنِ الّتي تُعطونَها للَّاويِّينَ ثمانيَ وأربعينَ مدينةً بمَراعيها.</span>
<span className="v" id="v8" data-v="8" data-xref="العَدَد 26:54; العَدَد 33:54; يَشُوع 21:1–42"> والمدُنُ الّتي تُعطونَها مِنْ مُلْكِ بَني إِسرائيلَ تكونُ على قدْرِ الميراثِ الّذي ورِثَهُ كُلُّ سِبْطٍ. فمِمَّنْ أخذَ كثيرًا مِنْ بَني إِسرائيلَ تأخُذونَ كثيرًا، ومِمَّنْ أخذَ قليلًا تأخُذونَ قليلًا“.</span>
</div>
</div>

## مدن الملجأ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 12:9; التَّثْنِيَة 19:1–2; اللّاويِّين 14:34"> ”قُلْ لبَني إِسرائيلَ: إذا أنتُم عبَرْتُمُ الأُردُنَّ إلى أرضِ كنعانَ،</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 21:13; يَشُوع 20:2; العَدَد 35:6"> فعَيِّنوا لكُم مُدُنًا تكونُ مُدُنَ ملجإٍ يهرُبُ إليها مَنْ قتَلَ نفْسًا سَهْوًا،</span>
<span className="v" id="v12" data-v="12" data-xref="يَشُوع 20:3–6; التَّثْنِيَة 19:6; يَشُوع 20:9"> فلا يثأرُ منهُ وَليُّ القتيلِ حتّى يقِفَ أمامَ الجماعةِ للمُحاكمةِ.</span>
<span className="v" id="v13" data-v="13"> والمُدُنُ الّتي تُعَيِّنونَها لهذا الغرضِ ستُّ مُدُنٍ،</span>
<span className="v" id="v14" data-v="14" data-xref="التَّثْنِيَة 19:8–10; التَّثْنِيَة 4:41–43; يَشُوع 20:7–9"> ثَلاثٌ مِنها في عَبرِ الأُردُنِّ وثَلاثٌ في أرضِ كنعانَ،</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 15:16; غَلاطِيَّة 3:28; ﺭﻭﻣﻴﺔ 3:29"> وتكونُ لبَني إِسرائيلَ وللغريبِ والنَّزيلِ فيما بينَكُم، فيَهرُبُ إليها كُلُّ مَنْ قتَلَ نفْسًا سَهْوًا.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="اللّاويِّين 24:17; التَّكوين 9:5–6; التَّثْنِيَة 19:11–13"> ”إنْ ضرَبَ أحدٌ أحدًا بآلةِ حديدٍ فماتَ فهوَ قاتلٌ، والقاتلُ يُقتَل.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 21:18"> وإنْ رَماهُ بِحجرٍ مِمَّا يُقتَلُ بهِ فماتَ فهوَ قاتِلٌ، والقاتِلُ يُقتَلُ.</span>
<span className="v" id="v18" data-v="18"> وإنْ رَماهُ بِآلةٍ مِنْ خَشَبٍ مِمَّا يُقتَلُ بهِ فماتَ، فهوَ قاتِلٌ، والقاتِلُ يُقتَلُ.</span>
<span className="v" id="v19" data-v="19" data-xref="يَشُوع 20:3; التَّثْنِيَة 19:6; التَّثْنِيَة 19:12"> ووَليُّ القاتلِ هوَ يَقتُلُ القاتِل حينَ يُصادِفُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 21:14; التَّثْنِيَة 19:11; صموئيلَ الثّانِي 3:27"> ”وإنْ دفَعَ أحدٌ أحدًا عَنْ بُغضٍ، أو ألقى عليهِ شيئًا مُتَعَمَّدًا فماتَ،</span>
<span className="v" id="v21" data-v="21"> أو ضرَبَهُ بيَدِهِ عَنْ عَداوةٍ فماتَ، فالقاتِلُ يُقتَلُ، ووَليُّ القَتيلِ هوَ يَقتُلُهُ حينَ يُصادِفُهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="العَدَد 35:11; الخُرُوج 21:13; يَشُوع 20:3"> لكنْ إنْ دفَعَهُ مُصادَفةً بلا عداوةٍ، أو ألقى علَيهِ آلةً ما بغَيرِ تعَمُّدٍ،</span>
<span className="v" id="v23" data-v="23"> أو أسقَطَ علَيهِ حجَرًا مِمَّا يُقتَلُ بهِ عَنْ غيرِ رُؤيَةٍ فماتَ، وما كانَ عدُوًّا لَه ولا طالِبًا لَه سوءًا،</span>
<span className="v" id="v24" data-v="24" data-xref="يَشُوع 20:6; العَدَد 35:12"> فلتَحكُمِ الجماعةُ بينَ القاتل ووَليِّ القتيلِ بِمُقتَضى هذِهِ الأحكامِ،</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 29:7; اللّاويِّين 4:3; ﺭﻭﻣﻴﺔ 3:24–26"> وتُخلِّصُ الجماعةُ القاتلَ مِنْ يَدِ وَليِّ القتيلِ وتردُّهُ إلى المدينةِ الّتي لجَأ إليها هرَبًا، فيُقيمَ بها حتّى يموتَ الكاهنُ العظيمُ الّذي مُسِحَ بالزَّيتِ المُقَدَّسِ.</span>
<span className="v" id="v26" data-v="26"> فإنْ خرَجَ القاتلُ مِنْ حُدودِ مدينةِ ملجَئِهِ الّتي هربَ إليها،</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 22:2; التَّثْنِيَة 19:10; التَّثْنِيَة 19:6"> فصادَفَهُ وَليُّ القتيلِ فقتَلَهُ، فلا جُرْمَ علَيهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="العِبرانِيّين 3:14; العِبرانِيّين 10:26–30; العِبرانِيّين 6:4–8"> فليُقِمْ في مدينةِ ملجئِهِ إلى أنْ يموتَ الكاهنُ العظيمُ، وبعدَ موتِهِ يرجعُ إلى أرضِ مُلْكِهِ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 27:11; العَدَد 27:1"> فلتكُنْ لكُم هذِهِ الفَرائضُ أحكامًا مدَى أجيالِكُم، في جميعِ مَساكِنِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="العِبرانِيّين 10:28; التَّثْنِيَة 19:15; ﻣﺘﻰ 18:16"> ”كُلُّ مَنْ قتَلَ نفْسًا، فبِشهادةِ شُهودٍ يُقتَلُ القاتِلُ، لا بِشهادةِ شاهِدٍ واحدٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّثْنِيَة 19:11–13; التَّكوين 9:5–6; المَزامِير 51:14"> ولا تأخُذوا فِديَةً عَنْ نفْسِ قاتلٍ يستَوجِبُ القتلَ، بل يُقتَلُ.</span>
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 4:12; غَلاطِيَّة 3:10–13; غَلاطِيَّة 2:21"> ولا تأخُذوا فِديَةً عَنْ قاتلٍ ليَهرُبَ إلى مدينةِ ملجَئِه، ولا ليَعودَ فيُقيمَ بالأرضِ قبلَ موتِ الكاهنِ.</span>
<span className="v" id="v33" data-v="33" data-xref="لُوقا 11:50–51; المَزامِير 106:38; هُوشَع 4:2–3"> لا تُدَنِّسوا الأرضَ الّتي أنتُم فيها، لأنَّ الدَّمَ يُدَنِّسُ الأرضَ ولا يُكَفِّرُ عَنها الدَّمُ الّذي سُفِكَ علَيها إلَّا بدَمِ سافكِهِ.</span>
<span className="v" id="v34" data-v="34" data-xref="العَدَد 5:3; المَزامِير 135:21; رُؤيا يُوحَنّا 21:27"> فلا تُنَجِّسوا الأرضَ الّتي أنتُم ساكِنونَ فيها وأنا مُقيمٌ بوسَطِها، لأنِّي أنا الرّبُّ مُقيمٌ فيما بينَ بَني إِسرائيلَ“.</span>
</div>
