## مثل الزارع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 5:1–3; مَرْقُس 2:13; لُوقا 8:4–10"> وعادَ إلى التَّعليمِ بِجانبِ البحرِ، فتجَمَّعَ حَولَهُ جُمهورٌ كبيرٌ جدًّا، حتّى إنَّهُ صَعِدَ إلى قارِبٍ في البحرِ وجلَسَ فيهِ، والجمعُ كلُّهُم في البَرِّ على شاطئِ البحرِ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 13:34–35; مَرْقُس 3:23; مَرْقُس 4:11"> فعلَّمَهُم بالأمثالِ أشياءَ كثيرةً. وقالَ لهُم في تَعليمِهِ:</span>
<span className="v" id="v3" data-v="3" data-xref="يَعقُوب 2:5; ﻣﺘﻰ 13:26; مَرْقُس 4:26–29"> ”إسمَعوا! خرَجَ الزَّارعُ لِيزرَعَ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 13:19; مَرْقُس 4:15; ﻣﺘﻰ 13:4"> وبَينَما هوَ يَزرَعُ، وقَعَ بَعضُ الحبِّ على جانبِ الطَّريقِ، فجاءَتِ الطُّيورُ وأكلَتْهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="حِزْقِيال 36:26; لُوقا 8:6; لُوقا 8:13"> ووقَعَ بَعضُهُ على أرضٍ صخريَّةٍ قليلةِ التُّرابِ، فنَبَتَ في الحالِ لأنَّ تُرابَهُ كانَ بلا عُمْقٍ.</span>
<span className="v" id="v6" data-v="6" data-xref="يَعقُوب 1:11; إرْمِيا 17:5–8; ﻳﻮﻧﺎﻥ 4:8"> فلمَّا أشرَقَتِ الشَّمسُ اَحترَقَ، وكانَ بِلا جُذورٍ فيَبِسَ.</span>
<span className="v" id="v7" data-v="7" data-xref="إرْمِيا 4:3; لُوقا 8:14; التَّكوين 3:17–18"> ومِنهُ ما وقَعَ بَينَ الشَّوكِ، فطَلَعَ الشَّوكُ وخنَقَهُ فما أعطى ثَمرًا.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 15:5; كُولُوسِي 1:6; مَرْقُس 4:20"> ومِنهُ ما وقَعَ على أرضٍ طيِّبةٍ، فنَبَتَ ونَما وأعطى ثَمَرًا، فأثمَرَ بَعضُهُ ثلاثينَ، وبَعضُهُ ستِّينَ، وبَعضُهُ مِئةً“.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 11:15; مَرْقُس 4:23–24; رُؤيا يُوحَنّا 3:22"> وقالَ يَسوعُ: ”مَنْ كانَ لَه أُذْنانِ تَسمَعانِ، فَليَسمَعْ“.</span>
</div>
</div>

## مغزى الأمثال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 13:10–17; مَرْقُس 4:34; الأمثال 13:20"> فلمَّا كانَ على اَنفِرادٍ، سألَهُ أتباعُهُ والرُّسُلُ الإثنا عشَرَ عَنْ مَغزى الأمثالِ،</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 11:25; لُوقا 8:10; رسالَةُ يُوحَنّا الأُولَى 5:20"> فقَالَ لهُم: ”أنتُم أُعطِيتُم سرَّ مَلكوتِ اللهِ. وأمَّا الّذينَ مِنْ خارِجٍ فيَسمَعونَ كُلَّ شيءٍ بالأمثالِ،</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 6:9–10; إشَعْياء 44:18; ﻣﺘﻰ 13:14–15"> حتّى إنَّهُم:<br />مَهما نَظَروا لا يُبصِرونَ،<br />ومَهما سَمِعوا لا يَفهَمونَ،<br />لِئلَّا يَتوبوا<br />فتُغفَرَ لهُم خطاياهُم“.</span>
</div>

## تفسير مثل الزارع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="رُؤيا يُوحَنّا 3:19; ﻣﺘﻰ 13:51–52; مَرْقُس 7:17–18"> ثُمَّ قالَ لِتلاميذِهِ: ”أما تَفهَمونَ هذا المَثلَ؟ كيفَ، إذًا، تَفهَمونَ غَيرَهُ مِنَ الأمثالِ؟</span>
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 8:4; ﻣﺘﻰ 13:19; ﻣﺘﻰ 13:37"> الزّارِعُ يَزرَعُ كلامَ اللهِ،</span>
<span className="v" id="v15" data-v="15" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:8; ﻣﺘﻰ 13:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:11"> وبَعضُ النَّاسِ مِثلُ الزَّرعِ الّذي يقَعُ على جانِبِ الطَّريقِ، يَسمَعونَ كلامَ اللهِ فيُسرِعُ الشَّيطانُ إليهِم ويَنتَزِعُ الكلامَ المَزروعَ فيهِم.</span>
<span className="v" id="v16" data-v="16" data-xref="مَرْقُس 6:20; لُوقا 8:13; ﻣﺘﻰ 13:20–21"> وبَعضُ النَّاسِ مِثلُ الزَّرعِ في أرضٍ صَخريَّةٍ، ما إنْ يَسمَعوا كلامَ اللهِ حتّى يَقبَلوهُ فَرِحينَ،</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 11:6; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 3:3–5; ﻣﺘﻰ 12:31"> ولكِنْ لا عُمقَ لهُم في نُفوسِهِم، فلا يَثبُتونَ على حالٍ. فإذا حدَثَ ضِيقٌ أوِ اَضطِهادٌ مِنْ أجلِ كلامِ اللهِ، اَرتَدّوا عَنهُ في الحالِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 8:14; إرْمِيا 4:3; ﻣﺘﻰ 13:22"> وبَعضُ النَّاسِ مِثلُ الزَّرعِ بَينَ الأشواكِ، يَسمَعونَ كلامَ اللهِ،</span>
<span className="v" id="v19" data-v="19" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:15–17; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9–10; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:17"> ولكِنْ هُمومُ الدُّنيا ومَحبَّةُ الغِنى وسائرُ الشَّهَواتِ تَدخُلُ في قُلوبِهِم وتَخنُقُ كلامَ اللهِ فلا يُثمِرُ.</span>
<span className="v" id="v20" data-v="20" data-xref="كُولُوسِي 1:10; لُوقا 8:15; يُوحَنّا 15:4–5"> وبَعضُ النَّاسِ مِثلُ الزَّرعِ في الأرضِ الطيِّبَةِ، يَسمَعونَ كلامَ اللهِ ويَقبَلونَهُ فيُثمِرونَ: مِنهُم مَنْ يُثمِرُ ثلاثينَ، ومِنهُم ستّينَ، ومِنهُم مئةً“.</span>
</div>
</div>

## مثل الزرع الّذي ينمو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 13:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:6–9; لُوقا 8:11"> وقالَ: ”يُشبِهُ مَلكوتُ اللهِ رجُلًا يَبذُرُ الزَّرعَ في حَقلِهِ.</span>
<span className="v" id="v27" data-v="27" data-xref="الجامِعَة 11:5; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 1:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:37–38"> فيَنامُ في اللَّيلِ ويَقومُ في النَّهارِ، والزَّرعُ يَنبُتُ ويَنمو، وهوَ لا يَعرِفُ كيفَ كانَ ذلِكَ.</span>
<span className="v" id="v28" data-v="28" data-xref="مَرْقُس 4:31–32; هُوشَع 6:3; فِيلِبِّي 1:9–11"> فالأرضُ مِنْ ذاتِها تُنبِتُ العُشبَ أوَّلًا، ثُمَّ السُّنبُلَ، ثُمَّ القَمحَ الّذي يَملأُ السُّنبُلَ.</span>

<span className="v" id="v32" data-v="32" data-xref="دانيال 4:20–22; دانيال 4:10–14; حِزْقِيال 31:3–10"> ولكنَّها بَعدَ الزَّرعِ تَرتَفِعُ وتَصيرُ أكبَرَ النَّباتِ، وتمُدُّ غُصونًا كبيرةً، حتّى إنَّ طُيورَ السَّماءِ تَجيءُ وتُعشِّشُ في ظِلِّها“.</span>

## يسوع يهدئ العاصفة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 8:18; لُوقا 8:22; لُوقا 8:25"> وفي مساءِ ذلِكَ اليومِ، قالَ لهُم: ”تعالَوا نَعبُرُ إلى الشَّاطئِ المُقابِلِ“.</span>
<span className="v" id="v36" data-v="36" data-xref="مَرْقُس 3:9; مَرْقُس 4:1; مَرْقُس 5:2"> فتركوا الجُموعَ وساروا بيَسوعَ في القارِبِ الّذي كانَ فيهِ، وكانَت معَهُ قوارِبُ أُخرى.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 8:23–24; أيُّوب 1:19; لُوقا 8:22–23"> فهَبّت عاصِفَةٌ شَديدةٌ وأخذَتِ الأمواجُ تَضرِبُ القارِبَ حتّى كادَ يَمتلِئُ.</span>
<span className="v" id="v38" data-v="38" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:7; العِبرانِيّين 4:15; إشَعْياء 54:6–8"> وكانَ يَسوعُ نائِمًا في مُؤخَّرِ القارِبِ، ورأسُهُ على مِخَدَّةٍ. فأيقظوهُ وقالوا لَه: ”يا مُعلِّمُ، أما يَهمُّكَ أنَّنا نهلِكُ؟“</span>
<span className="v" id="v39" data-v="39" data-xref="المَزامِير 107:29; المَزامِير 89:9; المَزامِير 65:7"> فقامَ واَنتهَرَ الرِّيحَ وقالَ للبحرِ: ”أُصمُتْ! إخرَسْ!“ فسكَنَتِ الرِّيحُ وسادَ هُدوءٌ تامٌ.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="لُوقا 8:25; ﻣﺘﻰ 8:26; ﻣﺘﻰ 14:31"> وقالَ يَسوعُ لِتلاميذِهِ: ”ما لكُم خائِفينَ؟ أما عندَكُم إيمانٌ بَعدُ؟“</span>
<span className="v" id="v41" data-v="41" data-xref="لُوقا 8:25; أيُّوب 38:11; المَزامِير 89:7"> ولكنَّهُم كانوا في فزَعٍ شديدٍ، وقالَ بَعضُهُم لبَعضٍ: ”مَنْ هذا؟ حتّى الرِّيحُ والبحرُ يُطيعانِهِ!“</span>
</div>
