## الشفاء في السَّبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 12:9–14; لُوقا 6:6–11; مَرْقُس 1:21"> ورجَعَ يَسوعُ إلى المجمَعِ، فوجَدَ فيهِ رجُلًا يدُهُ يابِسَةٌ.</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 37:32; لُوقا 14:1; لُوقا 6:7"> وكانَ هُناكَ جَماعةٌ يُراقِبونَهُ ليَرَوا هل يشفيهِ في السَّبتِ، فيَتَّهموهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="فِيلِبِّي 1:28–30; إشَعْياء 42:4; لُوقا 6:8"> فقَالَ للرَّجُلِ الّذي يدُهُ يابسةٌ: ”قُمْ في وَسْطِ المَجمَعِ!“</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 6:9; ﻣﺘﻰ 12:10–12; هُوشَع 6:6"> وقالَ لِلحاضِرينَ: ”أيَحِلُّ في السَّبتِ عمَلُ الخيرِ أمْ عمَلُ الشَّرِّ؟ إنقاذُ نَفسٍ أم إهلاكُها؟“ فسكَتوا.</span>
<span className="v" id="v5" data-v="5" data-xref="لُوقا 6:10; أفَسُس 4:18; ﺭﻭﻣﻴﺔ 11:25"> فأجالَ يَسوعُ نظَرَهُ فيهِم وهوَ غاضِبٌ حزينٌ لِقَساوةِ قلوبِهِم، وقالَ لِلرَّجُلِ: ”مُدَّ يَدَكَ!“ فمَدَّها فَعادت صحيحةً كالأخرى.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 22:16; مَرْقُس 12:13; ﻣﺘﻰ 12:14"> فخرَجَ الفَرِّيسيُّونَ وتَشاوَروا معَ الهِيرودسيِّينَ ليقتُلوا يَسوعَ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 4:25; لُوقا 6:17; ﻣﺘﻰ 12:15"> فاَنصَرَفَ معَ تلاميذِهِ إلى بحرِ الجليلِ، وتَبِعَهُ جُمهورٌ كبيرٌ مِنَ الجليلِ واليهودِيَّةِ،</span>
<span className="v" id="v8" data-v="8" data-xref="حِزْقِيال 35:15; إشَعْياء 34:5; ﻣﺘﻰ 11:21"> ومِنْ أُورُشليمَ وأدوميةَ وعَبْرِ الأُردُنِّ ونواحي صُورَ وصَيدا. وهؤُلاءِ سَمِعوا بأعمالِهِ فجاؤُوا إلَيهِ.</span>
<span className="v" id="v9" data-v="9" data-xref="مَرْقُس 5:30; يُوحَنّا 6:15"> فأمرَ تلاميذَهُ بأنْ يُهَيِّئوا لَه قارِبًا حتّى لا يزحَمَهُ الجَمعُ،</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 14:36; مَرْقُس 6:56; ﻣﺘﻰ 4:23"> لأنَّهُ شَفى كثيرًا مِنَ النَّاسِ، حتّى أخَذَ كُلُّ مريضٍ يشُقُّ طريقَهُ إلَيهِ ليَلمُسَهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 4:41; مَرْقُس 1:23–24; لُوقا 8:28"> وكانَ الّذينَ فيهِم أرواحٌ نَجِسَةٌ يَسجُدونَ لَه إذا رأَوهُ ويَصيحونَ: ”أنتَ اَبنُ اللهِ!“</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 12:16; مَرْقُس 1:25; مَرْقُس 1:34"> فكانَ يأْمُرُهُم بشِدَّةٍ أنْ لا يُعلِنوا أمرَهُ.</span>
</div>
</div>

## اختيار الرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="لُوقا 9:1; لُوقا 6:12–16; ﻣﺘﻰ 10:1–4"> وصَعِدَ إلى الجبَلِ ودَعا الّذينَ أرادَهُم فحَضَروا إلَيهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 1:8; لُوقا 9:1–6; يُوحَنّا 15:16"> فأقامَ مِنهُم اَثنَي عشَرَ سَمّاهُم رُسُلًا يُرافِقونَهُ فيُرْسِلُهُم مُبَشِّرينَ،</span>
<span className="v" id="v15" data-v="15"> ولهُم سُلطانٌ بِه يَطرُدونَ الشَّياطينَ.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 1:42; اعمال الرسل 1:13; ﻣﺘﻰ 16:16–18"> وهؤُلاءِ الاثنا عشَرَ هُم: سِمْعانُ وسَمّاهُ يَسوعُ بُطرُسَ،</span>
<span className="v" id="v17" data-v="17" data-xref="العِبرانِيّين 4:12; مَرْقُس 14:33; إرْمِيا 23:29"> ويَعقوبُ ويوحنَّا اَبنا زَبْدي وسَمّاهُما بُوانَرْجِسَ، أيْ اَبنَي الرَّعدِ،</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 1:13; مَرْقُس 2:14; ﻣﺘﻰ 9:9"> وأندَراوُسُ وفيلِبُّسُ وبَرثولوماوُسُ، ومتَّى وتوما، ويَعقوبُ بنُ حَلْفى وتَدَّاوُسُ وسِمْعانُ الوطنيُّ الغيورُ،</span>
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 6:71; يُوحَنّا 13:2; اعمال الرسل 1:16–25"> ويَهوذا أسخَرْيوطُ الّذي أسلَمَ يَسوعَ.</span>
</div>
</div>

## يسوع وبعلزبول

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 6:31; يُوحَنّا 4:31–34; مَرْقُس 9:28"> وجاءَ يَسوعُ إلى البيتِ، فعادَ النَّاسُ إلى الازدِحامِ، حتّى تعَذَّرَ على يَسوعَ وتلاميذِه أنْ يأكُلوا.</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 26:24; يُوحَنّا 10:20; مَرْقُس 3:31"> وسمِعَ أقرِباؤُهُ، فجاؤُوا ليأخُذوهُ لأنَّ بعضَ النَّاسِ قالوا: ”فقَدَ صوابَهُ“.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 15:1; ﻣﺘﻰ 10:25; ﻣﺘﻰ 9:34"> وأمَّا مُعَلِّمو الشَّريعَةِ الّذينَ نَزَلوا مِنْ أورُشليمَ، فقَالوا: ”فيهِ بعلُزَبول، وهوَ بِرئيسِ الشَّياطينِ يَطرُدُ الشَّياطينَ“.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="لُوقا 11:17–23; ﻣﺘﻰ 12:25–30; ﻣﺘﻰ 13:34"> فدَعاهُم وكلَّمَهُم بأمثالٍ، قالَ: ”كيفَ يُمكِنُ لِلشَّيطانِ أنْ يَطرُدَ الشَّيطانَ؟</span>
<span className="v" id="v24" data-v="24" data-xref="زَكَريّا 11:14; إشَعْياء 19:2–3; صموئيلَ الثّانِي 20:6"> فإذا اَنقسَمتْ مَملكةٌ لا تَثبُتُ،</span>
<span className="v" id="v25" data-v="25" data-xref="يَعقُوب 3:16; المَزامِير 133:1; غَلاطِيَّة 5:15"> أوِ اَنقسَمَ بَيتٌ لا يَثبُتُ.</span>
<span className="v" id="v26" data-v="26"> وإذا ثارَ الشَّيطانُ على ذاتِهِ واَنقسَمَ لا يَثبُتُ، بل يزولُ.</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 49:24–26; ﻣﺘﻰ 12:29; لُوقا 11:21–23"> لا يقدِرُ أحدٌ أن يدخُلَ بَيتَ رجُلٍ قويٍّ ويَنهبَ أمتِعتَهُ إلَّا إذا قيَّدَ هذا الرَّجُلَ القَويَّ أوَّلًا، ثمَّ يَنهَبُ بَيتَهُ.</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 12:10; ﻣﺘﻰ 12:31–32; رسالَةُ يُوحَنّا الأُولَى 5:16"> الحقَّ أقولُ لكُم: كُلُّ خَطيئةٍ وكُلُّ تَجديفٍ مَهما كانَ، يَغفِرُهُما اللهُ للنَّاسِ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 12:31–32; لُوقا 12:10; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 1:9"> وأمَّا مَنْ جَدَّفَ على الرُّوحِ القُدُسِ، فلا مَغفِرةَ لَه أبدًا، بل تَبقى خَطيئتُه أبديَّةً“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 10:20; مَرْقُس 3:22"> وبِهذا الكلامِ ردَّ على الّذينَ قالوا: ”فيهِ روحٌ نَجِسٌ!“</span>
</div>
</div>

## أم يسوع وإخوته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="لُوقا 8:19–21; ﻣﺘﻰ 12:46–50"> وجاءَت أمُّهُ وإخوَتُهُ، فَوقفوا في خارجِ البَيتِ وأرسَلوا إلَيهِ يَدعونَهُ.</span>
<span className="v" id="v32" data-v="32"> وكانَ يَجلِسُ حولَهُ جمعٌ كبيرٌ، فقَالوا لَه: ”أمُّكَ وإخوتُكَ وأخواتُكَ في خارِجِ البَيتِ يَطلبونَكَ“.</span>
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 3:21; التَّثْنِيَة 33:9; مَرْقُس 6:3"> فأجابَهُم: ”مَنْ هيَ أُمّي ومَنْ هُم إخوَتي؟“</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 12:49–50; لُوقا 11:27–28; ﺭﻭﻣﻴﺔ 8:29"> ونظَرَ إلى الجالسينَ حولَهُ وقالَ: ”هؤُلاءِ هُم أُمّي وإخوتي!</span>
<span className="v" id="v35" data-v="35" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:17; ﻣﺘﻰ 7:21; رسالَةُ يُوحَنّا الأُولَى 3:22–23"> لأنَّ مَنْ يَعمَلُ بمشيئةِ اللهِ هوَ أخي وأُختي وأُمّي“.</span>
</div>
