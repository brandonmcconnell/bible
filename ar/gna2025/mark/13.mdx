## يسوع يُنبئ بخراب الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 24:1–51; حِزْقِيال 11:22–23; حِزْقِيال 10:19"> ولمَّا خرَجَ مِنَ الهَيكَلِ، قالَ لَه واحدٌ مِنْ تلاميذِهِ: ”يا مُعَلِّمُ، أُنظُرْ ما أروَعَ هذِهِ الحِجارَةَ وهذِهِ الأبنِيَةَ!“</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 21:6; ﻣﺘﻰ 24:2; المُلُوكِ الأوَّلُ 9:7–8"> فأجابَهُ يَسوعُ: ”أتَرى هذِهِ الأبنيَةَ العظيمَةَ؟ لن يَبقى مِنها هُنا حجَرٌ على حجَرٍ، بَلْ يُهدَمُ كُلُّهُ“.</span>
</div>
</div>

## الاضطراب والاضطهاد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 21:1; مَرْقُس 4:34; مَرْقُس 14:33"> وبَينَما هوَ جالِسٌ في جبَلِ الزَيتونِ، تُجاهَ الهَيكَلِ، سألَهُ بُطرُسُ ويَعقوبُ ويوحنَّا وأندراوُسُ على اَنفِرادٍ:</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 1:6–7; ﻣﺘﻰ 24:3; دانيال 12:6"> ”أخبِرْنا متى يَحدُثُ هذا الخَرابُ، وما هيَ العَلامةُ الّتي تَدُلُّ على قُربِ حُدوثِهِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="كُولُوسِي 2:8; أفَسُس 5:6; إرْمِيا 29:8"> فأجابَهُم يَسوعُ: ”إيَّاكُم أنْ يُضلِّلَكُم أحدٌ.</span>
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 13:22; إرْمِيا 23:21–25; يُوحَنّا 8:24"> سيَجيءُ كثيرٌ مِنَ النَّاسِ مُنتَحِلينَ اَسمي، فيقولونَ: أنا هوَ المَسيحُ! ويَخدَعونَ أُناسًا كثيرينَ.</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 21:9–11; يُوحَنّا 14:27; ﻣﺘﻰ 24:6–7"> فإذا سَمِعتُم بِالحُروبِ وبأخبارِ الحُروبِ فلا تَرتَعِبوا، لأنَّ هذا لا بُدَّ أنْ يَحدُثَ، ولكنْ لا تكونُ هيَ الآخِرَةُ.</span>
<span className="v" id="v8" data-v="8" data-xref="ﻣﺘﻰ 24:8; إشَعْياء 37:3; المَزامِير 48:6"> ستَقومُ أمَّةٌ على أمَّةٍ ومَملكةٌ على مَملكةٍ، وتقَعُ الزَّلازِلُ في أماكِنَ كثيرةٍ، وتَحدُثُ مَجاعاتٌ، وهذا كُلُّه بَدءُ الأوجاعِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 10:17–18; يُوحَنّا 16:2; فِيلِبِّي 1:29"> وأمَّا أنتُم، فكونوا على حَذَرٍ. سيُسلمونكُم إلى المَحاكِمِ، ويَضربونكُم في المَجامِعِ، ويَسوقونَكُم إلى الحُكَّامِ والمُلوكِ مِنْ أجلي لتَشْهَدوا عِندَهُم،</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 24:14; كُولُوسِي 1:23; ﺭﻭﻣﻴﺔ 10:18"> لأنَّهُ يَجِبُ قَبلَ الآخِرَةِ إعلانُ البِشارةِ إلى جميعِ الشُّعوبِ.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 21:14–15; لُوقا 12:11–12; ﻣﺘﻰ 10:19–21"> وعِندَما يأْخُذونَكُم ليُسلِّموكُم لا تَهتمُّوا مِنْ قَبلُ كيفَ تَتكلَّمونَ، بل تكلَّموا بِما يُوحى إلَيكُم في حِينِه، لأنَّ الرُّوحَ القُدُسَ هوَ المُتكلِّمُ لا أنتُم.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 10:21; لُوقا 12:52–53; لُوقا 21:16"> سيُسلِّمُ الأخُ أخاهُ إلى الموتِ، والأبُ إِبنَهُ، ويتَمرَّدُ الأبناءُ على الآباءِ ويقتُلونَهُم،</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 15:18–19; ﻣﺘﻰ 10:22; ﻣﺘﻰ 24:13"> ويُبغِضُكُم جميعُ النَّاسِ مِنْ أجلِ اَسمي. ومَنْ يَثْبُت إلى النّهايةِ يَخْلُص.</span>
</div>
</div>

## الخراب العظيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="دانيال 9:27; دانيال 12:11; دانيال 11:31"> وإذا رَأيتُم ”نجاسةَ الخَرابِ“ قائِمةً حَيثُ يَجب أن لا تكونَ، (إفهمْ هذا أيُّها القارِئُ)، فَلْيَهرُبْ إلى الجبالِ مَنْ كانَ في اليهودِيَّةِ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 17:31–33; العِبرانِيّين 11:7; فِيلِبِّي 3:7–8"> ومَنْ كانَ على السَّطحِ، فلا يَنزِلْ إلى البَيتِ ليأخُذَ مِنهُ شيئًا.</span>
<span className="v" id="v16" data-v="16"> ومَنْ كانَ في الحَقلِ، فلا يَرجِعْ ليأخُذَ ثَوبَهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:10; لُوقا 23:29; هُوشَع 9:14"> الويلُ لِلحَبالى والمُرضِعاتِ في تِلكَ الأيّامِ.</span>
<span className="v" id="v18" data-v="18"> صَلُّوا حتّى لا يَحدُثَ هذا الخَرابُ في الشِّتاءِ،</span>
<span className="v" id="v19" data-v="19" data-xref="دانيال 12:1; يُوئيل 2:2; ﻣﺘﻰ 24:21"> فستَنزِلُ في تِلكَ الأيّامِ نكبَةٌ ما حَدَثَ مِثلُها مُنذُ بَدءِ العالَمِ الّذي خلَقَهُ اللهُ إلى اليومِ، ولنْ يَحدُثَ.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 24:22; زَكَريّا 13:8–9; إشَعْياء 65:8–9"> ولولا أنَّ الرَّبَّ جعَلَ تِلكَ الأيّامَ قَصيرةً، لما نَجا أحَدٌ مِنَ البَشَرِ. ولكنْ مِنْ أجلِ خاصَّتِهِ الّذينَ اَختارَهُم قَصَّرَ تِلكَ الأيّامَ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="لُوقا 21:8; ﻣﺘﻰ 24:23–25; ﻣﺘﻰ 24:5"> فإذا قالَ لكُم أحَدٌ: ها هوَ المَسيحُ هُنا! أو ها هوَ هُناكَ! فلا تُصدِّقُوهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 24:24; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:8–14; مَرْقُس 13:6"> فسيَظهَرُ مُسَحاءُ دَجَّالونَ وأنبياءُ كَذَّابُونَ يَعمَلونَ آياتٍ ومُعجِزاتٍ، ولو أمكَنَهُم لَضَلَّلوا الّذينَ اَختارَهُمُ اللهُ.</span>
<span className="v" id="v23" data-v="23" data-xref="رسالَةُ بُطرُسَ الثّانيَة 3:17; مَرْقُس 13:5; يُوحَنّا 14:29"> فكونوا أنتُمُ على حَذَرٍ. ها أنا أنبأتُكُم بِكُلِّ شيءٍ.</span>
</div>
</div>

## مجيء ابن الإنسان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 13:10; اعمال الرسل 2:19–20; رُؤيا يُوحَنّا 6:12–14"> وفي تِلكَ الأيّامِ، بَعدَ زَمَنِ الضّيقِ، تُظلِمُ الشّمسُ ولا يُضيءُ القمَرُ.</span>
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 34:4; رُؤيا يُوحَنّا 6:13"> وتَتَساقَطُ النُّجومُ مِنَ السَّماءِ وتَتزَعزعُ قوّاتُ السَّماءِ.</span>
<span className="v" id="v26" data-v="26" data-xref="رُؤيا يُوحَنّا 1:7; ﻣﺘﻰ 16:27; ﻣﺘﻰ 24:30"> وفي ذلِكَ الحينِ يَرى النَّاسُ اَبنَ الإنسانِ آتيًا في السَّحابِ بِكُلِّ عِزَّةٍ وجَلالٍ.</span>
<span className="v" id="v27" data-v="27" data-xref="زَكَريّا 2:6; ﻣﺘﻰ 24:31; التَّثْنِيَة 30:4"> فيُرسِلُ مَلائِكتَهُ إلى جِهاتِ الرّياحِ الأربَعِ ليَجمَعوا مُختاريهِ مِنْ أقصى الأرضِ إلى أقصى السَّماءِ.</span>
</div>
</div>

## عبرة شجرة الّتين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="لُوقا 21:29–31; ﻣﺘﻰ 24:32–33"> خُذوا مِنْ شجَرَةِ التِّينِ عِبرةً: إذا لانَت أغصانُها وأوْرَقَت، عَرَفتُم أنَّ الصَّيفَ قَريبٌ.</span>
<span className="v" id="v29" data-v="29" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:17–18; يَعقُوب 5:9; حِزْقِيال 12:25–28"> وكذلِكَ إذا رأيتُم هذا كُلَّهُ يَحدُثُ، فاَعلَموا أنَّ الوَقتَ قَريبٌ على الأبوابِ.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 24:34; لُوقا 21:32; ﻣﺘﻰ 16:28"> الحقَّ أقولُ لكُم: لنْ يَنقَضيَ هذا الجِيلُ حتّى يَتِمَّ هذا كُلُّهُ.</span>
<span className="v" id="v31" data-v="31" data-xref="إشَعْياء 40:8; إشَعْياء 51:6; ﻣﺘﻰ 24:35"> السَّماءُ والأرضُ تَزولانِ وكلامي لَنْ يَزولَ.</span>
</div>

## السهر الدائم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 1:7; ﻣﺘﻰ 24:36–42; ﻣﺘﻰ 25:13"> وأمَّا ذلِكَ اليومُ أو تِلكَ السَّاعةُ فلا يَعرِفُهُما أحَدٌ، لا الملائِكَةُ في السَّماءِ ولا الابنُ، إلَّا الآبَ.</span>
<span className="v" id="v33" data-v="33" data-xref="أفَسُس 6:18; لُوقا 12:40; ﺭﻭﻣﻴﺔ 13:11–12"> فكونوا على حَذَرٍ واَسْهَروا، لأنَّكُم لا تَعرِفونَ متى يَجيءُ الوقتُ.</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 25:14–30; يُوحَنّا 10:3; لُوقا 12:36–40"> وذلِكَ كمَثَلِ رَجُلٍ سافَرَ وترَكَ بَيتَهُ وسَلَّمهُ إلى خَدَمِهِ، كُلُّ واحدٍ وعَمَلُهُ، وأوصى البوّابَ بالسَّهَرِ.</span>
<span className="v" id="v35" data-v="35" data-xref="مَرْقُس 13:33; ﻣﺘﻰ 24:42; ﻣﺘﻰ 24:44"> فاَسهَروا، لأنَّكُم لا تَعرفونَ متى يَجيءُ رَبُّ البَيتِ، أفي المَساءِ أم في مُنْتَصفِ اللَّيلِ أم عِندَ صِياحِ الدّيكِ أم في الصّباحِ،</span>
<span className="v" id="v36" data-v="36" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:6–7; ﺭﻭﻣﻴﺔ 13:11–14; ﻣﺘﻰ 24:48–51"> لِئلَّا يَجيءَ فَجْأةً فيَجِدَكُم نيامًا.</span>
<span className="v" id="v37" data-v="37" data-xref="مَرْقُس 13:33; لُوقا 12:41–46; مَرْقُس 13:35"> وما قُلتُهُ لكُم أقولُهُ لِجميعِ النَّاسِ: اَسهَرُوا!“</span>
