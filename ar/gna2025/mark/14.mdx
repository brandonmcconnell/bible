## محاولة قتل يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 22:1–2; الخُرُوج 12:6–20; يُوحَنّا 13:1"> وقَبلَ الفِصحِ وعيدِ الفَطيرِ بِيَومَينِ، كانَ رُؤساءُ الكَهنَةِ ومُعَلِّمو الشَّريعةِ يَبحَثونَ كيفَ يُمسِكونَ يَسوعَ بِحيلَةٍ ليَقتُلوهُ.</span>

## امرأة تسكب الطيب على يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="لُوقا 7:37–39; يُوحَنّا 12:1–8; يُوحَنّا 11:2"> وبَينَما يَسوعُ في بَيتِ عَنيا، عِندَ سِمعانَ الأبرصِ، يتَناولُ الطَّعامَ، جاءَتِ اَمرأةٌ بِيَدِها قارورَةُ طِيبٍ غالي الثَمَنِ مِنَ النارِدينِ النَّقيِّ، فكسَرَتِ القارورَةَ وسكَبَتْهُ على رأسِهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 12:4–5; ﻣﺘﻰ 26:8–9; مَلاخِي 1:12–13"> فاَستاءَ بَعضُ الحاضِرينَ وقالوا في ما بَينَهُم: ”ما هذا الإسرافُ في الطِّيبِ؟</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 13:29; أفَسُس 4:28; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:10"> كانَ يُمكِنُ بَيعُهُ بأكثرَ من ثلاثِ مئَةِ دينارٍ تُوزَّعُ على الفُقراءِ!“ وأخذوا يَلومونَ المرأةَ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="تِيطُس 3:14; العِبرانِيّين 10:24; العِبرانِيّين 13:21"> فقَالَ يَسوعُ: ”أُترُكوها، لماذا تُزعِجونَها؟ هذا عمَلٌ صالِحٌ عَمِلَتهُ لي.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 15:11; ﻣﺘﻰ 26:11; يَعقُوب 2:14–16"> فالفُقراءُ عِندَكُم في كُلِّ حينٍ، ومتى أرَدتُم تَقْدِرونَ أنْ تُحسِنوا إليهِم. وأمَّا أنا، فلا أكونُ في كُلِّ حينٍ عِندَكُم.</span>
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12; يُوحَنّا 12:7; مَرْقُس 15:42–1"> وهذِهِ المرأةُ عَمِلَت ما تَقدِرُ علَيهِ، فسكَبَتِ الطِّيبَ على جَسَدي لِتُهَيِّئَهُ للدَّفنِ.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 26:12–13; مَرْقُس 16:15; ﻣﺘﻰ 24:14"> الحقَّ أقولُ لكُم: أينَما تُعلَنُ البِشارَةُ في العالَمِ كُلِّهِ، يُحدَّثُ أيضًا بِعَمَلِها هذا، إِحياءً لذِكرِها“.</span>
</div>
</div>

## خيانة يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="لُوقا 22:3–6; ﻣﺘﻰ 26:14–16; يُوحَنّا 13:2"> وذهَبَ يَهوذا أسخريوط، أحَدُ التَّلاميذِ الاثنَي عشَرَ، إلى رُؤساءِ الكَهنَةِ ليُسلِّمَ إليهِم يَسوعَ.</span>
<span className="v" id="v11" data-v="11" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:10; هُوشَع 7:3; يَهُوذا 1:11"> ففَرِحوا لكلامِهِ، ووَعَدُوهُ بأنْ يُعطوهُ مالًا. فأخَذَ يَتَرقَّبُ الفُرصَةَ ليُسْلِمَهُ.</span>
</div>
</div>

## عشاء الفصح مع التَّلاميذ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 12:6; الخُرُوج 12:18; ﻣﺘﻰ 26:17–19"> وفي أوَّلِ يومٍ مِنْ عيدِ الفَطيرِ، حينَ تُذبَحُ الخِرافُ لِعَشاءِ الفِصحِ سألَهُ تلاميذُهُ: ”إلى أينَ تُريدُ أنْ نَذهَبَ لِنُهَيِّئَ لكَ عَشاءَ الفِصحِ لِتأكُلَهُ؟“</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 22:10–13; ﻣﺘﻰ 26:18–19; لُوقا 19:30–33"> فأرسَلَ اَثنينِ مِنْ تلاميذِهِ وقالَ لهُما: ”إذهَبا إلى المدينةِ، فيُلاقِيكُما رَجُلٌ يَحمِلُ جَرَّةَ ماءٍ فاَتبَعاهُ.</span>
<span className="v" id="v14" data-v="14" data-xref="رُؤيا يُوحَنّا 3:20; مَرْقُس 11:3; يُوحَنّا 13:13"> وعِندَما يَدخُلُ بَيتًا قولا لِربِّ البَيتِ: يقولُ المُعَلِّمُ: أينَ غُرفَتي الّتي آكُلُ فيها عَشاءَ الفِصحِ معَ تلاميذي؟</span>
<span className="v" id="v15" data-v="15" data-xref="اعمال الرسل 1:13; المَزامِير 110:3; الأمثال 21:1–2"> فيُريكُما في أعلى البَيتِ غُرفَةً واسِعةً مَفروشَةً مُجهَّزةً، فهيِّئاهُ لنا هُناكَ“.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 16:4; لُوقا 22:13; لُوقا 22:35"> فذَهبَ التِّلميذانِ ودَخلا المدينةَ، فوَجدا كما قالَ لهُما وهَيَّأا عَشاءَ الفِصحِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="لُوقا 22:14; يُوحَنّا 13:21–26; ﻣﺘﻰ 26:20–24"> ولمَّا كانَ المساءُ، جاءَ معَ تلاميذِهِ الاثنَي عشَرَ.</span>
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 41:9; يُوحَنّا 6:47; يُوحَنّا 6:70"> وبَينَما هُم جالِسونَ لِلطَّعامِ، قالَ يَسوعُ: ”الحقَّ أقولُ لكُم: واحدٌ مِنكُم سيُسلِمُني، وهوَ يأكُلُ مَعي“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="لُوقا 22:21–23; ﻣﺘﻰ 26:22; يُوحَنّا 13:22"> فحَزِنَ التَّلاميذُ وأخَذوا يَسألونَهُ، واحدًا فواحدًا: ”هل أنا هوَ؟“</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 13:26; ﻣﺘﻰ 26:23; يُوحَنّا 6:71"> فقَالَ لهُم: ”هوَ واحدٌ مِنَ الاثنَي عشَرَ، وهوَ الّذي يَغمِسُ يدَهُ في الصَّحنِ مَعي.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 26:24–25; إشَعْياء 53:1–12; يُوحَنّا 19:28"> واَبنُ الإنسانِ سيَموتُ كما جاءَ عَنهُ في الكُتُبِ المُقَدَّسَةِ، ولكِنَّ الويلَ لِمنْ يُسلِّمُ اَبنَ الإنسانِ! كانَ خيرًا لَهُ أنْ لا يولَدَ“.</span>
</div>
</div>

## عشاء الرّبّ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 26:26–29; مَرْقُس 14:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:16–17"> وبَينَما هُم يأكُلونَ، أخَذَ خُبزًا وبارَكَ وكَسَرهُ وناوَلَهُم وقالَ: ”خُذوا، هذا هوَ جَسَدي“.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="مَرْقُس 14:22; ﻣﺘﻰ 26:27; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:16"> وأخذَ كأسًا وشكَرَ وناوَلَهُم، فشَرِبوا مِنها كُلُّهُم،</span>
<span className="v" id="v24" data-v="24" data-xref="زَكَريّا 9:11; الخُرُوج 24:8; رُؤيا يُوحَنّا 7:9–17"> وقالَ لهُم: ”هذا هوَ دَمي، دمُ العَهدِ الّذي يُسفَكُ مِنْ أجلِ أُناسٍ كَثيرينَ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 26:29; المَزامِير 104:15; لُوقا 22:16–18"> الحقَّ أقولُ لكُم: لا أشرَبُ بَعدَ الآنَ مِنْ عَصيرِ الكرمَةِ، حتّى يَجيءَ يومٌ فيهِ أشرَبُهُ جديدًا في مَلكوتِ اللهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 26:30; لُوقا 22:39; اعمال الرسل 16:25"> ثُمَّ سَبَّحوا وخرَجوا إلى جبَلِ الزَّيتُونِ.</span>
</div>
</div>

## يسوع يُنبئ بإنكار بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="زَكَريّا 13:7; يُوحَنّا 16:32; ﻣﺘﻰ 26:31"> فقَالَ لهُم يَسوعُ: ”ستَترُكوني كُلُّكُم، لأنَّ الكِتابَ يقولُ: سأَضرِبُ الرّاعي، فتَتَبدَّدُ الخِرافُ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 28:16; مَرْقُس 16:7; ﻣﺘﻰ 28:7"> ولكِنْ بَعدَ قيامتي أسبُقُكُم إلى الجَليلِ“.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 26:33–35; لُوقا 22:33–34; يُوحَنّا 13:36–38"> فقَالَ لَهُ بُطرُسُ: ”لَو تَركوك كُلُّهُم، فأنا لَنْ أترُكَكَ!“</span>
<span className="v" id="v30" data-v="30" data-xref="يُوحَنّا 13:38; مَرْقُس 14:66–72; التَّكوين 1:5"> فأجابَهُ يَسوعُ: ”الحقَّ أقولُ لكَ يا بُطرُسُ: اليومَ، في هذِهِ اللَّيلةِ، قَبلَ أنْ يَصيحَ الدّيكُ مرَّتَينِ، تُنكِرُني ثلاثَ مرَّاتٍ“.</span>
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 19:8; يُوحَنّا 13:37; الأمثال 16:18"> فأصرَّ بُطرُسُ على قولِهِ: ”لا أُنكِرُكَ وإنْ كانَ علَيَّ أنْ أموتَ مَعكَ!“ وهكذا قالَ التَّلاميذُ كُلُّهُم.</span>
</div>
</div>

## يسوع يصلّي في جتسيماني

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 14:36; مَرْقُس 14:39; المَزامِير 18:5–6"> وجاؤُوا إلى مكانٍ اَسمُهُ جَتْسِيماني، فقَالَ لِتلاميذِهِ: ”أُقعُدوا هُنا، بَينَما أنا أُصَلِّي“.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="لُوقا 22:44; العِبرانِيّين 5:7; مَرْقُس 5:37"> وأخَذَ مَعهُ بُطرُسَ ويَعقوبَ ويوحنَّا، وبَدأَ يشعُرُ بالرَّهبَةِ والكآبَةِ.</span>
<span className="v" id="v34" data-v="34" data-xref="يُوحَنّا 12:27; مَرْقُس 13:35–37; رسالَةُ بُطرُسَ الأُولَى 5:8"> فقَالَ لهُم: ”نَفسي حَزينةٌ حتّى الموتِ. إِنتَظِروا هُنا واَسهَروا!“</span>
<span className="v" id="v35" data-v="35" data-xref="مَرْقُس 14:41; التَّثْنِيَة 9:18; العِبرانِيّين 5:7"> واَبتَعدَ قليلًا ووقَعَ إلى الأرضِ يُصلِّي حتّى تَعبُرَ عَنهُ ساعَةُ الألَمِ، إنْ كانَ مُمكِنًا.</span>
<span className="v" id="v36" data-v="36" data-xref="غَلاطِيَّة 4:6; ﺭﻭﻣﻴﺔ 8:15–16; يُوحَنّا 5:30"> فقَالَ: ”أبَّا، يا أبَّا! أنتَ قادِرٌ على كُلِّ شيءٍ، فأبْعِدْ عَنِّي هذِهِ الكأسَ. ولكِنْ لا كما أنا أُريدُ، بَلْ كما أنتَ تُريدُ“.</span>
</div>
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 26:40; مَرْقُس 14:29–31; العِبرانِيّين 12:3"> ورَجَعَ فوَجدَهُم نِيامًا، فقَالَ لبُطرُسَ: ”أنائمٌ أنتَ يا سِمْعانُ؟ أما قَدِرتَ أنْ تَسهَرَ ساعةً واحدةً؟</span>
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 26:41; ﻣﺘﻰ 24:42; لُوقا 21:36"> إِسهَروا وصَلُّوا، لِئَلَّا تقَعوا في التَّجربَةِ. الروحُ راغبةٌ ولكِنَّ الجسَدَ ضَعيفٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="ﻣﺘﻰ 26:42–44; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:8; ﻣﺘﻰ 6:7"> واَبتَعَدَ ثانيةً وصَلَّى، فرَدَّدَ الكلامَ ذاتَهُ.</span>
<span className="v" id="v40" data-v="40" data-xref="مَرْقُس 9:33–34; ﺭﻭﻣﻴﺔ 3:19; التَّكوين 44:16"> ورَجَعَ أيضًا فوجَدَهُم نيامًا، لأنَّ النُّعاسَ أثقَلَ جُفونَهُم وحاروا بِماذا يُجيبونَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="مَرْقُس 14:35; مَرْقُس 9:31; يُوحَنّا 17:1"> ورجَعَ في المرَّةِ الثَّالثةِ وقالَ لهُم: ”أنيامٌ بَعدُ ومُستريحونَ؟ يكفي! جاءَتِ السَّاعةُ. ها هوَ اَبنُ الإنسانِ يُسلَّمُ إلى أيدي الخاطِئينَ.</span>
<span className="v" id="v42" data-v="42" data-xref="ﻣﺘﻰ 26:46; يُوحَنّا 18:1–2"> قوموا نَنْصرِف! إِقترَبَ الّذي يُسَلِّمُني!“</span>
</div>
</div>

## اعتقال يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="يُوحَنّا 18:3–11; لُوقا 22:47–53; ﻣﺘﻰ 26:47–56"> وبَينَما هوَ يتكَلَّمُ، وصَلَ يَهوذا، أحدُ التَّلاميذِ الاثني عشَرَ، على رأسِ عِصابةٍ تَحمِلُ السُّيوفَ والعِصِيَّ، أرسَلَها رُؤساءُ الكَهنَةِ ومُعَلِّمو الشَّريعةِ والشُّيوخُ.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 26:48–50; الأمثال 27:6; المَزامِير 55:20–21"> وكانَ الّذي أسلَمَهُ أعطاهُم علامَةً قالَ: ”هوَ الّذي أُقَبِّلُهُ، فأمسِكُوهُ وخُذوهُ في حِراسةٍ شديدةٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="ﻣﺘﻰ 23:7–10; مَلاخِي 1:6; مَرْقُس 12:14"> ولمَّا وصلَ يَهوذا، دنا في الحالِ إلى يَسوعَ وقالَ لَهُ: ”يا مُعَلِّمُ!“ وقَبَّلَهُ.</span>
<span className="v" id="v46" data-v="46" data-xref="يُوحَنّا 18:12; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:20; اعمال الرسل 2:23"> فألقَوا أيديَهُم علَيهِ وأمسَكوهُ.</span>
<span className="v" id="v47" data-v="47" data-xref="يُوحَنّا 18:10–11; ﻣﺘﻰ 26:51–54; لُوقا 22:49–51"> فاَستَلَّ أحدُ الحاضِرينَ سيفَهُ، وضرَبَ خادِمَ رئيسِ الكَهنَةِ فقَطَعَ أُذنَهُ.</span>

<span className="v" id="v49" data-v="49" data-xref="يُوحَنّا 18:20; يُوحَنّا 8:2; لُوقا 22:37"> كُنتُ كُلَّ يومٍ بَينَكُم أعلِّمُ في الهَيكَلِ فما أمسكتُموني، ولكِنْ حدَثَ هذا لِتَتِمَّ الكُتُبُ المُقدَّسةُ“.</span>

## يسوع في مجلس اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v53" data-v="53" data-xref="ﻣﺘﻰ 26:57–68; يُوحَنّا 18:12–14; لُوقا 22:54–62"> وأخَذوا يَسوعَ إلى رَئيسِ الكَهنَةِ، فاَجتَمعَ رُؤساءُ الكَهنَةِ والشُّيوخُ ومُعَلِّمو الشَّريعةِ كُلُّهُم.</span>
<span className="v" id="v54" data-v="54" data-xref="يُوحَنّا 18:18; ﻣﺘﻰ 26:3; يُوحَنّا 18:25"> وتَبِعَهُ بُطرُسُ عَنْ بُعدٍ إلى دارِ رَئيسِ الكَهنَةِ، فدَخَلَ وقعَدَ معَ الحَرَسِ يتَدَفّأُ عِندَ النَّارِ.</span>
<span className="v" id="v55" data-v="55" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:16–18; ﻣﺘﻰ 26:59–60; ﻣﺘﻰ 5:22"> وكانَ رُؤساءُ الكَهنَةِ وجميعُ أعضاءِ المَجلِسِ يَطلُبونَ شَهادَةً على يَسوعَ ليَقتُلوهُ فلا يَجدونَ،</span>
<span className="v" id="v56" data-v="56"> لأنَّ أُناسًا كَثيرينَ شَهِدوا علَيهِ زُورًا فتَناقَضَت شَهاداتُهُم.</span>
<span className="v" id="v57" data-v="57" data-xref="اعمال الرسل 6:13–14; ﻣﺘﻰ 26:60–61; يُوحَنّا 2:18–21"> فقامَ بَعضُهُم وشَهِدوا علَيهِ زُورًا، قالوا:</span>
<span className="v" id="v58" data-v="58" data-xref="يُوحَنّا 2:19; العِبرانِيّين 9:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:1"> ”نَحنُ سَمِعناهُ يَقولُ: سأهدِمُ هذا الهَيكَلَ المَصنوعَ بالأيدي، وأبني في ثلاثةِ أيّامٍ هَيكَلًا آخَرَ غَيرَ مَصنوعٍ بالأيدي“.</span>
<span className="v" id="v59" data-v="59"> وفي هذا أيضًا تناقَضَت شَهاداتُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v60" data-v="60" data-xref="يُوحَنّا 19:9–10; ﻣﺘﻰ 26:62–63; مَرْقُس 15:3–5"> فقامَ رَئيسُ الكَهنَةِ في وسَطِ المَجلِسِ وسألَ يَسوعَ: ”أما تُجيبُ بِشيءٍ؟ ما هذا الّذي يَشهَدونَ بهِ علَيكَ؟“</span>
<span className="v" id="v61" data-v="61" data-xref="ﻣﺘﻰ 26:63–64; مَرْقُس 15:2; إشَعْياء 53:7"> فظَلَّ ساكِتًا، لا يَقولُ كَلِمةً. وسألَهُ رئيسُ الكَهنَةِ ثانيةً: ”هل أنتَ المَسيحُ اَبنُ اللهِ المُبارَكِ؟“</span>
<span className="v" id="v62" data-v="62" data-xref="دانيال 7:13–14; المَزامِير 110:1; العِبرانِيّين 1:3"> فأجابَهُ يَسوعُ: ”أنا هوَ. وسَتَرَونَ اَبنَ الإنسانِ جالِسًا عَنْ يَمينِ اللهِ القديرِ، وآتيًا معَ سحابِ السَّماءِ!“</span>
<span className="v" id="v63" data-v="63" data-xref="ﻣﺘﻰ 26:65; إشَعْياء 36:22–1; العَدَد 14:6"> فشَقَّ رَئيسُ الكَهنَةِ ثيابَهُ وقالَ: ”أنَحتاجُ بَعدُ إلى شُهودٍ؟</span>
<span className="v" id="v64" data-v="64" data-xref="اللّاويِّين 24:16; يُوحَنّا 10:31–33; يُوحَنّا 19:7"> سَمِعتُم تَجديفَهُ، فما رأيُكُم؟“ فحَكَموا علَيهِ كُلُّهُم بأنَّهُ يَستوجِبُ الموتَ.</span>
</div>
<div className="bp">
<span className="v" id="v65" data-v="65" data-xref="ﻣﺘﻰ 26:67–68; اعمال الرسل 23:2; إشَعْياء 50:6"> وأخذَ بَعضُهُم يَبصُقونَ علَيهِ، ويُغَطُّونَ وجهَهُ ويَلطِمونَهُ ويقولونَ لَه: ”تَنَبّأْ“. وتناولَهُ الحَرَسُ بالضَّربِ.</span>

## بطرس ينكر يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v66" data-v="66" data-xref="مَرْقُس 14:54; ﻣﺘﻰ 26:69–75; يُوحَنّا 18:25–27"> وبَينَما بطرسُ في السّاحةِ السُفلى مِنَ الدَّارِ، مَرَّت جارِيةٌ مِنْ جواري رئيسِ الكَهنَةِ.</span>
<span className="v" id="v67" data-v="67" data-xref="مَرْقُس 14:54; مَرْقُس 10:47; ﻣﺘﻰ 2:23"> فلمَّا رأَت بطرسَ يَتَدَفّأُ، نَظرَتْ إلَيهِ وقالَت لَهُ: ”أنتَ أيضًا كُنتَ معَ يَسوعَ النّاصريِّ!“</span>
<span className="v" id="v68" data-v="68" data-xref="مَرْقُس 14:29–31; ﻣﺘﻰ 26:71–72; يُوحَنّا 13:36–38"> فأنكَرَ قالَ: ”لا أعرِفُ ولا أفهَمُ ما تَقولينَ!“ وخرَجَ إلى الدِّهليزِ، فصاحَ الدّيكُ.</span>
<span className="v" id="v69" data-v="69" data-xref="لُوقا 22:58; مَرْقُس 14:38; غَلاطِيَّة 6:1"> فرأَتْهُ الجارِيةُ، وأخذت تَقولُ لِلحاضرينَ: ”هذا مِنهُم!“</span>
<span className="v" id="v70" data-v="70" data-xref="لُوقا 22:59–60; ﻣﺘﻰ 26:73–74; اعمال الرسل 2:7"> فأنكرَ أيضًا. وبَعدَ قليلٍ، قالَ الحاضِرونَ لبطرسَ: ”لا شكَّ أنَّكَ واحِدٌ مِنهُم، لأنَّكَ مِنَ الجليلِ“.</span>
<span className="v" id="v71" data-v="71" data-xref="إرْمِيا 17:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:12; المُلُوكِ الثّانِي 10:32"> فأخَذَ يَلعَنُ ويَحلِفُ: ”أنا لا أعرِفُ هذا الرَّجُلَ الّذي تَعنونَ“.</span>
<span className="v" id="v72" data-v="72" data-xref="مَرْقُس 14:30; مَرْقُس 14:68; حِزْقِيال 7:16"> وفي الحالِ صاحَ الدّيكُ مرَّةً ثانيةً، فتَذكَّرَ بطرسُ قولَ يَسوعَ: ”قَبلَ أنْ يَصيحَ الدّيكُ مرَّتَينِ، تُنكِرُني ثلاثَ مرَّاتٍ“. وأخذَ يَبكي.</span>
