<div className="bp">

<span className="v" id="v2" data-v="2" data-xref="مَلاخِي 3:1; ﻣﺘﻰ 11:10; لُوقا 1:76"> بَدأتْ كما كَتبَ النَّبيُّ إشَعْيا:<br />”ها أنا أُرسِلُ رَسولي قُدَّامَكَ<br />ليُهيِّئَ طَريقَكَ</span>

<span className="v" id="v5" data-v="5" data-xref="اللّاويِّين 26:40–42; اعمال الرسل 19:18; ﻣﺘﻰ 3:5–6"> وكانوا يَخرُجونَ إلَيهِ مِنْ جميعِ بِلادِ اليَهوديَّةِ وأُورُشليمَ فيُعَمِّدهُم في نهرِ الأُردُنِّ، مُعتَرِفينَ بِخطاياهُم.</span>

<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الثّانِي 1:8; اللّاويِّين 11:22; ﻣﺘﻰ 3:4"> وكانَ يوحنَّا يَلبَسُ ثَوبًا مِنْ وبَرِ الجِمالِ، وعلى وَسْطِهِ حِزامٌ مِنْ جِلدٍ، ويَقتاتُ مِنَ الجَرادِ والعسَلِ البرِّيِّ.</span>
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 13:25; لُوقا 3:16; يُوحَنّا 1:27"> وكانَ يُبشِّرُ فيقولُ: ”يَجيءُ بَعدي مَنْ هوَ أقوى منِّي. مَنْ لا أحسبُ نفْسي أهلًا لأنْ أنحَنيَ وأحُلَّ رِباطَ حِذائِهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 1:5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:13; اعمال الرسل 19:4–6"> أنا عَمَّدتكُم بالماءِ، وأمَّا هوَ فيُعمِّدُكُم بالرُّوحِ القُدُسِ“.</span>
</div>

## يوحنا يعمّد يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 3:13–17; ﻣﺘﻰ 2:23; لُوقا 3:21–22"> وفي تِلكَ الأيّامِ جاءَ يَسوعُ مِنَ النّاصِرَةِ الّتي في الجَليلِ، وتَعمَّدَ على يَدِ يوحنَّا في نَهرِ الأُردُنِّ.</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 1:31–34; لُوقا 3:22; إشَعْياء 42:1"> ولمَّا صعدَ يَسوعُ مِنَ الماءِ رأى السَّماواتِ تنفَتِحُ والرُّوحَ القُدُسَ يَنزِلُ علَيهِ كأنَّهُ حَمامةٌ.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 42:1; مَرْقُس 9:7; ﻣﺘﻰ 3:17"> وقالَ صوتٌ مِنَ السَّماءِ: ”أنتَ اَبني الحبيبُ، بِكَ رَضِيتُ“.</span>
</div>
</div>

## يسوع في الجليل: التَّلاميذ الأولون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 4:23; ﻣﺘﻰ 4:12; لُوقا 8:1"> وبَعدَ اَعتِقالِ يوحنَّا، جاءَ يَسوعُ إلى الجليلِ يُعلِنُ بِشارةَ اللهِ،</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 3:2; اعمال الرسل 2:36–38; ﻣﺘﻰ 4:17"> فيقولُ: ”تَمَّ الزَّمانُ واَقترَبَ مَلكوتُ اللهِ. فتُوبوا وآمنوا بالإنجيلِ“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 4:18–22; يُوحَنّا 1:40–42; يُوحَنّا 6:8"> وبَينَما هوَ يَمشي على شاطئِ بحرِ الجَليلِ، رأى صيَّادَينِ هُما سِمْعانُ وأخوهُ أندَرَاوُسُ يُلقيانِ الشَّبكَةَ في البحرِ،</span>
<span className="v" id="v17" data-v="17" data-xref="لُوقا 5:10; ﻣﺘﻰ 4:19–20; اعمال الرسل 2:38–41"> فقَالَ لهُما يَسوعُ: ”إِتبعاني أجعَلْكُما صيّادَي بَشَرٍ“.</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 18:28–30; لُوقا 5:11; مَرْقُس 10:28–31"> فتَركا شِباكَهُما في الحالِ وتَبِعاهُ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 5:37; اعمال الرسل 1:13; مَرْقُس 10:35"> ومشَى قليلًا، فَرأى يَعقوبَ بنَ زَبدي وأخاهُ يوحنَّا، وهُما في القارِبِ يُصلِحانِ شِباكَهُما.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 10:37; لُوقا 14:26; مَرْقُس 10:29"> فما إنْ دَعاهُما، حتّى تَركا أباهُما زَبدي في القارِبِ معَ مُعاوِنيهِ وتَبِعاهُ.</span>
</div>
</div>

## يسوع يطرد روحًا نجسًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 6:2; مَرْقُس 1:39; ﻣﺘﻰ 4:23"> وجاؤُوا إلى كَفرناحومَ، فدَخَلَ المَجمعَ في السَّبتِ وأخَذَ يُعلِّمُ.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 7:28–29; ﻣﺘﻰ 13:54; لُوقا 4:32"> فتَعَجَّبوا من تعليمِهِ، لأنَّهُ كانَ يُعلِّمُهُم مِثلَ مَنْ لَه سُلطانٌ، لا مِثلَ مُعَلِّمي الشَّريعةِ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="مَرْقُس 9:25; مَرْقُس 7:25; مَرْقُس 1:34"> وكانَ في المَجمعِ رجُلٌ فيهِ رُوحٌ نَجِسٌ، فأخذَ يَصيحُ:</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 8:29; يَعقُوب 2:19; اعمال الرسل 3:14"> ”ما لنا ولكَ، يا يَسوعُ النّاصريُّ؟ أجِئتَ لِتُهلِكَنا؟ أنا أعرِفُ مَنْ أنتَ: أنتَ قُدُّوسُ اللهِ!“</span>
<span className="v" id="v25" data-v="25" data-xref="مَرْقُس 1:34; لُوقا 4:41; اعمال الرسل 16:17"> فاَنتَهَرَهُ يَسوعُ، قالَ: ”إخرَسْ واَخرُجْ مِنَ الرَّجُلِ!“</span>
<span className="v" id="v26" data-v="26" data-xref="مَرْقُس 9:26; مَرْقُس 9:20; لُوقا 11:22"> فصرَعَهُ الرُّوحُ النَّجِسُ، وصرَخَ صَرخةً قويَّةً وخرَجَ مِنهُ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 15:31; مَرْقُس 7:37; لُوقا 9:1"> فتَعجَّبَ النَّاسُ كُلُّهم وتَساءَلوا: ”ما هذا؟ أتَعليمٌ جديدٌ يُلقى بِسُلطانٍ؟ حتّى الأرواحُ النَّجِسةُ يأمُرُها فتُطيعُهُ!“</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 9:31; لُوقا 4:37; مَرْقُس 1:45"> وذاعَ صيتُ يَسوعَ في جميعِ أنحاءِ الجَليلِ.</span>
</div>
</div>

## شفاء حماة بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="لُوقا 4:38–39; ﻣﺘﻰ 8:14–15; مَرْقُس 1:21"> ولمَّا خرَجَ مِنَ المَجمعِ، جاءَ معَ يَعقوبَ ويوحنَّا إلى بيتِ سِمْعانَ وأندَراوسَ.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:5; مَرْقُس 5:23; يُوحَنّا 11:3"> وكانَت حَماةُ سِمْعانَ طَريحةَ الفِراشِ بالحُمَّى، فأخبَروهُ عَنها.</span>
<span className="v" id="v31" data-v="31" data-xref="اعمال الرسل 9:41; المَزامِير 116:12; مَرْقُس 5:41"> فدَنا مِنها وأمسَكَ يَدَها وأنهَضَها. فتَركَتْها الحُمَّى وأخذَت تَخدُمُهُم.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 8:16; مَرْقُس 3:2; لُوقا 4:40"> وعِندَ المساءِ، بَعدَ غُروبِ الشَّمسِ، حمَلَ النَّاسُ إلَيهِ جميعَ المرضى والّذينَ فيهِم شياطينُ.</span>
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 1:5; اعمال الرسل 13:44"> وتجمَّعَ أهلُ المدينةِ كُلُّهُم على البابِ،</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 4:41; مَرْقُس 3:12; ﻣﺘﻰ 4:23"> فشَفى كثيرًا مِنَ المُصابينَ بمُختَلفِ الأمراضِ، وطَردَ كثيرًا مِنَ الشَّياطينِ، ومنَعَ الشَّياطينَ أنْ تتكلَّمَ لأنَّها عَرَفتْهُ.</span>
</div>
</div>

## شفاء أبرص

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="لُوقا 5:12–14; ﻣﺘﻰ 8:2–4; اللّاويِّين 13:1–14"> وجاءَهُ أبرصُ يَتوسَّلُ إليهِ، فسَجَدَ وقالَ لَه: ”إنْ أرَدتَ طَهَّرتَني“.</span>
<span className="v" id="v41" data-v="41" data-xref="ﻣﺘﻰ 9:36; العِبرانِيّين 2:17; العِبرانِيّين 4:15"> فأشفَقَ علَيهِ يَسوعُ ومدَّ يدَهُ ولَمَسهُ وقالَ لَه: ”أريدُ، فاَطْهُرْ!“</span>
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 15:3; مَرْقُس 5:29; يُوحَنّا 4:50–53"> فزالَ عَنهُ البرَصُ في الحالِ وطَهَرَ.</span>
</div>
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 9:30; مَرْقُس 7:36; لُوقا 8:56"> فاَنتَهَرَهُ يَسوعُ وصرَفَهُ،</span>
<span className="v" id="v44" data-v="44" data-xref="اللّاويِّين 14:1–32; لُوقا 17:14; ﻣﺘﻰ 8:4"> بَعدَما قالَ لَه: ”إيَّاكَ أن تُخبِرَ أحدًا بِشَيءٍ. ولكِنِ إِذهَبْ إلى الكاهنِ وأرِهِ نفسَكَ، ثُمَّ قَدِّمْ عَنْ شِفائِكَ ما أمَرَ بِه موسى، شَهادةً عِندَهُم“.</span>
<span className="v" id="v45" data-v="45" data-xref="مَرْقُس 2:13; ﻣﺘﻰ 9:31; يُوحَنّا 6:2"> ولكِنَّ الرَّجُلَ اَنصَرَفَ وأخَذَ يُذيعُ الخبَرَ ويَنشُرُهُ في كُلِّ مكانٍ. حتّى تعذَّرَ على يَسوعَ أنْ يدخُلَ علانيَّةً إلى أيَّةِ مدينةٍ. فأقامَ في أماكنَ مُقفِرَةٍ. وكانَ النَّاسُ يَجيئونَ إلَيهِ مِنْ كُلِّ مكانٍ.</span>
