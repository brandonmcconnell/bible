## الناصرة ترفض يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 4:16–30; ﻣﺘﻰ 13:54–58; ﻣﺘﻰ 2:23"> وخرَجَ مِنْ هُناكَ وجاءَ إلى بلدِهِ يَتبَعُهُ تلاميذُهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 7:15; يُوحَنّا 6:42; اعمال الرسل 4:13–14"> وفي السَّبتِ أخَذَ يُعلِّمُ في المَجمعِ. فتَعجَّبَ أكثرُ النَّاسِ حينَ سَمِعوهُ وقالوا: ”مِنْ أينَ لَه هذا؟ وما هذِهِ الحِكمَةُ المُعطاةُ لَه وهذِهِ المُعجزاتُ الّتي تَجري على يَدَيهِ؟</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 12:46; ﻣﺘﻰ 13:55–57; يُوحَنّا 6:42"> أما هوَ النَّجّارُ اَبنُ مَريمَ، وأخو يَعقوبَ ويوسي ويَهوذا وسِمعانَ؟ أما أخَواتُهُ عِندَنا هُنا؟“ ورَفَضوهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 4:44; لُوقا 4:24; ﻣﺘﻰ 13:57"> فقَالَ لهُم يَسوعُ: ”لا نَبِيَّ بلا كرامةٍ إلَّا في وطَنِهِ وبَينَ أقربائِهِ وأهلِ بَيتِهِ“.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 13:58; مَرْقُس 9:23; مَرْقُس 5:23"> وتَعذَّرَ على يَسوعَ أنْ يَصنَعَ أيَّةَ مُعجِزَةٍ هُناكَ، سِوى أنَّهُ وضَعَ يدَيهِ على بَعضِ المَرضى فشَفاهُم.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻣﺘﻰ 9:35; ﻣﺘﻰ 8:10; لُوقا 13:22"> وكانَ يتَعجَّبُ مِنْ قِلَّةِ إيمانِهِم. ثُمَّ سارَ في القُرى المُجاوِرَةِ يُعلِّمُ.</span>
</div>
</div>

## يسوع يرسل التَّلاميذ الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الجامِعَة 4:9–10; لُوقا 10:17–20; ﻣﺘﻰ 10:1–5"> ودَعا إليهِ التَّلاميذَ الاثنَي عشَرَ وأخَذَ يُرسِلُهُم اَثنَينِ اَثنَينِ ليُبَشِّروا، وأعطاهُم سُلطانًا على الأرواحِ النَّجِسَةِ.</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 9:3; ﻣﺘﻰ 10:9–10; لُوقا 10:4"> وأوصاهُم قالَ: ”لا تأخُذوا للطَّريقِ شيئًا سِوى عَصًا: لا خُبزًا، ولا كيسًا، ولا نُقودًا في جُيوبِكُم،</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 12:8; أفَسُس 6:15"> بَلِ اَنتَعِلوا حِذاءً، ولا تَلبَسوا ثوبَينِ“.</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 10:7–8; لُوقا 9:4; ﻣﺘﻰ 10:11–13"> وقالَ لهُم: ”وحَيثُما دَخَلتُم بَيتًا، فأقيموا فيهِ إلى أنْ تَرحَلوا.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 10:14–15; اعمال الرسل 18:6; لُوقا 10:10–15"> وإذا جِئْتُم إلى مكانٍ وما قَبِلَكُم أهلُهُ ولا سَمِعوا كلامَكُم، فاَرْحَلوا عَنهُ واَنفُضوا الغُبارَ عَنْ أقدامِكُم نَذيرًا لهُم!“</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 4:17; لُوقا 13:3; اعمال الرسل 20:21"> فخَرَجوا يَدعونَ النَّاسَ إلى التَّوبةِ،</span>
<span className="v" id="v13" data-v="13" data-xref="يَعقُوب 5:14–15; مَرْقُس 6:7; لُوقا 10:17"> وطَرَدوا كثيرًا مِنَ الشَّياطينِ، ودَهَنوا بالزَّيتِ كثيرًا مِنَ المَرضى فشَفَوهُم.</span>
</div>
</div>

## موت يوحنا المعمدان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="لُوقا 9:7–9; مَرْقُس 6:14–29; لُوقا 23:7–12"> وسمِعَ المَلِكُ هِيرودُسُ بأخبارِ يَسوعَ، لأنَّ اَسمَهُ أصبحَ مَشهورًا. وكانَ بَعضُ النَّاسِ يَقولونَ: ”قامَ يوحنَّا المَعمدانُ مِنْ بَينِ الأمواتِ، ولذلِكَ تَجري المُعْجِزاتُ على يدِهِ“.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 16:14; مَلاخِي 4:5; ﻣﺘﻰ 21:11"> وقالَ آخرونَ: ”هوَ إيلِيَّا“، وآخرونَ: ”هوَ نَبيٌّ كسائِرِ الأنبياءِ“.</span>
<span className="v" id="v16" data-v="16" data-xref="رُؤيا يُوحَنّا 11:10–13; ﻣﺘﻰ 27:4; لُوقا 9:9"> فلمَّا سَمِعَ هِيرودُسُ قالَ: ”هذا يوحنَّا الّذي قَطعتُ أنا رأسَهُ، قامَ مِنْ بَينِ الأمواتِ!“</span>
<span className="v" id="v17" data-v="17" data-xref="لُوقا 3:19–20; ﻣﺘﻰ 11:2; لُوقا 3:1"> وكانَ هِيرودُسُ أمسَكَ يوحنَّا وقَيَّدَهُ في السِّجنِ، مِنْ أجلِ هِيرودِيَّةَ الّتي تَزوَّجَها وهيَ اَمرأةُ أخيهِ فيلبُّسَ.</span>
<span className="v" id="v18" data-v="18" data-xref="اللّاويِّين 20:21; اللّاويِّين 18:16; المُلُوكِ الأوَّلُ 22:14"> فكانَ يوحنَّا يقولُ لَه: ”لا يَحلُّ لكَ أنْ تأخُذَ اَمرأةَ أخيكَ“.</span>
<span className="v" id="v19" data-v="19" data-xref="الجامِعَة 7:9; أفَسُس 4:26–27; التَّكوين 39:17–20"> وكانَت هيرودِيَّةُ ناقِمَةً علَيهِ تُريدُ قَتلَه فلا تَقدِرُ،</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 21:26; ﻣﺘﻰ 14:5; أخبارِ الأيّامِ الثّانِي 24:15–22"> لأنَّ هِيرودُسَ كانَ يَهابُهُ ويَحميه لعِلْمِهِ أنَّهُ رَجُلٌ بارٌّ وقِدِّيسٌ. وكانَ يَسُرُّهُ أنْ يَستمِعَ إليهِ، معَ أنَّهُ حارَ فيهِ كثيرًا.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="أسْتِير 2:18; التَّكوين 40:20; رسالَةُ بُطرُسَ الأُولَى 4:3"> وسنَحَتِ الفُرصةُ لِهيرودِيَّةَ عِندَما أقامَ هِيرودُسُ في ذِكرى مَولِدِه وَليمَةً للنُّبَلاءِ وكِبارِ القادَةِ وأعيانِ الجَليلِ.</span>
<span className="v" id="v22" data-v="22" data-xref="أسْتِير 1:10–12; ﻣﺘﻰ 14:6; دانيال 5:2"> فدخَلَتِ اَبنَةُ هيرودِيَّةَ ورَقَصَتْ، فأعجَبَت هِيرودُسَ والمَدعوِّينَ. فقَالَ المَلِكُ لِلفَتاةِ: ”أُطلُبي ما شِئتِ فأُعطيَكِ“.</span>
<span className="v" id="v23" data-v="23" data-xref="أسْتِير 7:2; أسْتِير 5:3; أسْتِير 5:6"> وحلَفَ لها يَمينًا مُشدَّدًا قالَ: ”أُعطيكِ كُلَّ ما تَطلُبينَ، ولَو نِصفَ مَملكَتي!“</span>
<span className="v" id="v24" data-v="24" data-xref="أيُّوب 31:31; حِزْقِيال 19:2–3; التَّكوين 27:8–11"> فخَرجَتِ الفَتاةُ وسألَت أُمَّها: ”ماذا أطلُبُ؟“ فأجابَتْها: ”رأسَ يوحنَّا المَعمدانِ!“</span>
<span className="v" id="v25" data-v="25" data-xref="ﺭﻭﻣﻴﺔ 3:15; العَدَد 7:19–89; العَدَد 7:13"> فأَسرَعَت إلى المَلِكِ وقالَت لَه: ”أُريدُ أن تُعطِيَني الآنَ على طَبَقٍ رأسَ يوحنَّا المَعمَدانِ!“</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 27:24–25; ﻣﺘﻰ 14:9; ﻣﺘﻰ 27:3–5"> فحَزِنَ المَلِكُ كثيرًا، ولكِنَّهُ أرادَ أنْ لا يَرُدَّ طَلَبَها مِنْ أجلِ اليمينِ الّتي حلَفَها أمامَ المَدعوِّينَ.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 14:10–11"> فأرسَلَ في الحالِ جُنديًّا وأمرَهُ بأنْ يَجيءَ برَأسِ يوحنَّا. فذهَبَ وقطَعَ رأسَهُ في السِّجنِ،</span>
<span className="v" id="v28" data-v="28"> وجاءَ بِه على طَبَقٍ وسلَّمَهُ إلى الفَتاةِ فحمَلَتْهُ إلى أُمِّها.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 14:12; أخبارِ الأيّامِ الثّانِي 24:16; ﻣﺘﻰ 27:57–60"> وبلَغَ الخبَرُ تلاميذَ يوحنَّا، فجاؤُوا وأخَذوا جُثَّتَهُ ووضَعوها في قَبرٍ.</span>
</div>
</div>

## يسوع يطعم خمسة آلاف رجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="لُوقا 9:10; لُوقا 10:17; ﻣﺘﻰ 10:2"> واَجتَمَعَ الرُّسُلُ عِندَ يَسوعَ، وأخبروهُ بِكُلِّ ما عَمِلوا وعَلَّموا.</span>
<span className="v" id="v31" data-v="31" data-xref="مَرْقُس 3:20; يُوحَنّا 6:1; مَرْقُس 1:45"> وكانَ كثيرٌ مِنَ النَّاسِ يَروحونَ ويَجيئونَ، فلا يَتركونَ لهُم فُرصةً ليأكُلوا طعامَهُم، فقَالَ لهُم يَسوعُ: ”تعالَوا أنتُم وحدَكُم إلى مكانٍ مُقفِرٍ واَستَريحوا قليلًا“.</span>
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 8:2–9; مَرْقُس 6:45; مَرْقُس 4:36"> فذَهبوا في القارِبِ وحدَهُم إلى مكانٍ مُقفِرٍ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 6:2; ﻣﺘﻰ 15:29–31; مَرْقُس 6:54–55"> فرآهُمُ النَّاسُ ذاهبينَ، وعرَفَ كثيرٌ مِنهُم إلى أينَ، فأسرَعوا مِنْ جميعِ المُدُنِ مَشيًا على الأقدامِ وسَبقوهُم إلى ذلِكَ المكانِ.</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 9:36; لُوقا 9:11; العَدَد 27:17"> فلمَّا نزَلَ يَسوعُ مِنَ القارِبِ رأى جَمْعًا كبيرًا، فأشفَقَ علَيهِم، لأنَّهُم كانوا مِثلَ غنَمٍ لا راعيَ لها، وأخَذَ يُعلِّمُهُم أشياء كثيرةً.</span>
<span className="v" id="v35" data-v="35" data-xref="يُوحَنّا 6:5–15; لُوقا 9:12–17; ﻣﺘﻰ 14:15–21"> وفاتَ الوقتُ، فدَنا مِنهُ تلاميذُهُ وقالوا لَهُ: ”فاتَ الوقتُ، وهذا مكانٌ مُقفِرٌ،</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 15:23; مَرْقُس 3:21; مَرْقُس 5:31"> فقُلْ لِلنَّاسِ أنْ يَنصرِفوا إلى المَزارِعِ والقُرى المُجاوِرَةِ ليَشتَروا لهُم ما يأكُلونَ“.</span>
<span className="v" id="v37" data-v="37" data-xref="المُلُوكِ الثّانِي 4:42–44; ﻣﺘﻰ 15:32–33; مَرْقُس 8:2–3"> فأجابَهُم يَسوعُ: ”أعطوهُم أنتُم ما يأكُلونَ“. فقَالوا: ”أتُريدُنا أنْ نَذهَبَ ونَشترِيَ خُبزًا بمئَتَي دينارٍ ونُعطيَهُم لِيأكُلوا؟“</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 15:34; مَرْقُس 8:5; يُوحَنّا 6:9"> فقَالَ يَسوعُ: ”كَم رغيفًا عِندَكُم؟ إِذهَبوا واَنظُروا“. فلمَّا عَرَفوا ما عِندَهُم، قالوا لَهُ: ”خَمسةُ أرغِفَةٍ وسَمكَتانِ“.</span>
<span className="v" id="v39" data-v="39" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:40; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:33; ﻣﺘﻰ 15:35"> فأمرَهُم أنْ يُقعِدوا النَّاسَ جماعةً جماعةً على العُشْبِ الأخضَرِ.</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 9:14–15"> فقَعَدُوا صُفوفًا صُفوفًا، في بعضِها مِئةٌ وفي بَعضِها خَمسونَ.</span>
<span className="v" id="v41" data-v="41" data-xref="ﻣﺘﻰ 14:19; مَرْقُس 7:34; لُوقا 9:16"> وأخَذَ يَسوعُ الأرغِفَةَ الخَمسةَ والسَّمكتَينِ ورفَعَ عَينَيهِ نَحوَ السَّماءِ وبارَكَ وكَسَرَ الأرغِفَةَ وناوَلَ تلاميذَهُ ليُوزِّعوها على النَّاسِ، وقَسَّمَ السَّمكتَينِ علَيهِم جميعًا.</span>
<span className="v" id="v42" data-v="42" data-xref="لُوقا 9:17; مَرْقُس 8:8–9; ﻣﺘﻰ 15:37–38"> فأكلوا كُلُّهُم حتّى شَبِعوا.</span>
<span className="v" id="v43" data-v="43" data-xref="مَرْقُس 8:19–20"> ثُمَّ رَفَعوا اَثنتَي عَشْرَةَ قُفَّةً مَملوءَةً مِنَ الكِسَرِ وفَضَلاتِ السَّمكَتَينِ.</span>
<span className="v" id="v44" data-v="44"> وكانَ الّذينَ أكَلوا مِنَ الأرغِفَةِ نَحوَ خَمسَةِ آلافِ رَجُلٍ.</span>
</div>
</div>

## يسوع يمشي على الماء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="ﻣﺘﻰ 14:22–33; يُوحَنّا 6:15–21; مَرْقُس 6:32"> وأمَرَ يَسوعُ تلاميذَهُ أنْ يَركَبوا القارِبَ ويَسبِقوهُ إلى بَيتَ صيدا، عِندَ الشَّاطئِ المقابلِ، حتّى يَصرِفَ الجُموعَ.</span>
<span className="v" id="v46" data-v="46" data-xref="مَرْقُس 1:35; ﻣﺘﻰ 6:6; لُوقا 6:12"> فلمَّا صرَفَهُم ذهَبَ إلى الجبَلِ ليُصلِّي.</span>
<span className="v" id="v47" data-v="47" data-xref="ﻣﺘﻰ 14:23; يُوحَنّا 6:16–17"> وعِندَ المَساءِ، كانَ القارِبُ في وَسْطِ البحرِ، ويَسوعُ وحدَهُ على البَرِّ.</span>
<span className="v" id="v48" data-v="48" data-xref="لُوقا 24:28; لُوقا 12:38; أيُّوب 9:8"> ورأى يَسوعُ أنَّ تلاميذَهُ يُلاقونَ مَشقَّةً في التَّجذيفِ، لأنَّ الرِّيحَ كانَت ضِدَّهُم. فَجاءَ إليهِم قَبلَ الفَجرِ ماشِيًا على البحرِ، وكادَ يمُرُّ بِهِم.</span>
<span className="v" id="v49" data-v="49" data-xref="ﻣﺘﻰ 14:25–26; لُوقا 24:37; أيُّوب 9:8"> فلمَّا رأوهُ ماشِيًا على البحرِ ظنُّوهُ شَبَحًا فصَرَخوا،</span>
