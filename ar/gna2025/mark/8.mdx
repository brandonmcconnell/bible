## يسوع يطعم أربعة آلاف رجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 15:32–39; مَرْقُس 6:34–44"> وتَجمَّعَ في تِلكَ الأيّامِ جُمهورٌ كبيرٌ، وما كانَ مَعَهُم شيءٌ يأكلونَهُ. فَدَعا يَسوعُ تلاميذَهُ وقالَ لهُم:</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 9:36; ميخا 7:19; المَزامِير 103:13"> ”أُشفِقُ على هذا الجَمعِ فَهُم مِنْ ثلاثةِ أيّامٍ مَعي، ولا طعامَ لهُم.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 40:31; القُضاة 8:4–6; صَمُوئيلَ الأوَّلُ 30:10–12"> وإنْ صرَفْتُهم إلى بُيوتِهِم صائِمينَ، خارَت قِواهُم في الطَّريقِ، ومِنهُم مَنْ جاءَ مِنْ مكانٍ بَعيدٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الثّانِي 4:42–44; المُلُوكِ الثّانِي 7:2; مَرْقُس 6:52"> فأجابَهُ تلاميذُهُ: ”كيفَ لأحَدٍ أنْ يُشبعَ هؤُلاءِ النَّاسَ خُبزًا هُنا في البرِّيَّةِ؟“</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 14:15–17; ﻣﺘﻰ 15:34; لُوقا 9:13"> فسألَهُم: ”كم رَغيفًا عِندَكُم؟“ قالوا: ”سَبعَةٌ“.</span>
<span className="v" id="v6" data-v="6" data-xref="كُولُوسِي 3:17; ﻣﺘﻰ 15:35–36; يُوحَنّا 6:10–11"> فأمَرَ يَسوعُ الجَمعَ أنْ يَقعُدوا على الأرضِ، وأخَذَ الأرغِفَةَ السَبعَةَ وشكَرَ وكسَرَها وأعطى تلاميذَهُ لِيُوَزِّعوها على الجَمعِ، فوَزَّعُوها علَيهِم.</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 14:19; يُوحَنّا 21:5; لُوقا 24:41–42"> وكانَ معَ التَّلاميذِ بعضُ سَمكاتٍ صِغارٍ، فبارَكَها يَسوعُ وطَلبَ مِنْ تلاميذِهِ أنْ يُوَزِّعوها أيضًا.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الثّانِي 4:42–44; المَزامِير 145:16; ﻣﺘﻰ 16:10"> فأكَلَ النَّاسُ حتّى شَبِعوا، ثُمَّ رفَعوا مِمَّا فضَلَ مِنَ الكِسَرِ سَبعَ سِلالٍ.</span>
<span className="v" id="v9" data-v="9"> وكانوا نحوَ أربَعةِ آلافٍ. فَصرَفَهُم يَسوعُ</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 15:39"> وركِبَ القارِبَ معَ تلاميذِهِ وجاءَ إلى نواحي دَلمانُوثَةَ.</span>
</div>
</div>

## الفريسيون يطلبون آية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:9; ﻣﺘﻰ 12:38; التَّثْنِيَة 6:16"> وأقبَلَ إليهِ بَعضُ الفَرِّيسيِّينَ وأخذوا يُجادِلونَهُ طالِبينَ مِنهُ آيةً مِنَ السَّماءِ ليُجرِّبوهُ،</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 12:39–40; مَرْقُس 7:34; ﻣﺘﻰ 16:4"> فتَنَهَّدَ مِنْ أعماقِ قَلبِه وقالَ: ”ما لِهذا الجِيلِ يَطلُبُ آيةً؟ الحقَّ أقولُ لكُم: لن يُعطى هذا الجِيلُ آيةً!“</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 12:36; اعمال الرسل 13:45–46; المَزامِير 81:12"> وتَركَهُم ورَجَعَ في القارِبِ إلى الشَّاطئِ المُقابِلِ.</span>
</div>
</div>

## خمير الفريسيين وخمير هيرودس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 16:5"> ونَسيَ التَّلاميذُ أن يتزَوَّدوا خُبزًا، وما كانَ معَهُم في القارِبِ سِوى رغيفٍ واحدٍ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:6–8; ﻣﺘﻰ 16:6; ﻣﺘﻰ 16:11–12"> وأَوصاهُم يَسوعُ قالَ: ”إنتَبِهوا، إيَّاكُم وخميرَ الفَرِّيسيِّينَ وخميرَ هيرُودُسَ!“</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="لُوقا 9:46; لُوقا 20:5; ﻣﺘﻰ 16:7–8"> فتَساءَلوا: ”هل لأنْ لا خُبزَ مَعنا؟“</span>
<span className="v" id="v17" data-v="17" data-xref="مَرْقُس 6:52; إشَعْياء 63:17; مَرْقُس 2:8"> فَعرفَ يَسوعُ وقالَ لهُم: ”لماذا تُفكِّرونَ أنِّي قُلتُ هذا لأنْ لا خُبزَ معَكُم؟ أما أدْركتُم بَعدُ وفَهِمتُم؟ أعَمِيت قُلوبُكُم إلى هذا الحدِّ؟</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 28:26–27; ﻣﺘﻰ 13:14–15; إرْمِيا 5:21"> أمْ لكُم عُيونٌ ولا تُبصِرونَ، وآذانٌ ولا تَسمَعونَ؟ ألا تَذكُرونَ</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 9:12–17; ﻣﺘﻰ 14:17–21; يُوحَنّا 6:5–13"> عِندَما كَسَرْتُ الأرغِفَةَ الخَمسَةَ لِلخَمسَةِ الآلافِ، كم قُفَّةً مملوءةً مِنَ الكِسَرِ رَفَعتُم؟ قالوا: ”إِثنَتَي عَشْرَةَ قُفَّةً!“</span>
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 15:34–38; مَرْقُس 8:1–9"> قالَ: ”وعِندَما كَسَرتُ الأرغِفَةَ السَّبعَةَ لِلأربَعةِ الآلافِ، كم سلَّةً مَملوءةً مِنَ الفَضَلاتِ رَفَعتُم؟ قالوا: ”سبعَ سِلالٍ!“</span>
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 6:52; ﻣﺘﻰ 16:11–12; مَرْقُس 8:12"> فقَالَ لهُم: ”أما فَهِمْتُم بَعدُ؟“</span>
</div>
</div>

## شفاء أعمى في بيت صيدا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 11:21; مَرْقُس 6:45; لُوقا 10:13"> ووَصَلوا إلى بَيتَ صَيدا، فجاؤُوا إليهِ بِأعمى وتَوسَّلوا إلَيهِ أنْ يَلمِسَهُ.</span>
<span className="v" id="v23" data-v="23" data-xref="مَرْقُس 7:33; يُوحَنّا 9:6–7; مَرْقُس 5:23"> فأخَذَ بِيَدِ الأعمى وَقادَهُ إلى خارِجِ القَريةِ، وهُناكَ تَفَلَ في عَينَيهِ، ووَضَعَ يَدَيهِ علَيهِ وسألَهُ: ”أتُبْصِرُ شيئًا؟“</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 32:3; إشَعْياء 29:18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:9–12"> فتَطلَّعَ وقالَ: ”أُبصِرُ النَّاسَ وأراهُم يَمشُونَ كأنَّهُم أشجارٌ“.</span>
<span className="v" id="v25" data-v="25" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:9; رسالَةُ بُطرُسَ الثّانيَة 3:18; الأمثال 4:18"> فوَضَعَ يَسوعُ يدَيهِ مرَّةً أُخرى على عَينَي الرَّجُلِ، فأبصَرَ جيِّدًا وعادَ صَحيحًا يرى كُلَّ شيءٍ واضِحًا.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 8:4; مَرْقُس 8:23; مَرْقُس 5:43"> فأرسَلهُ إلى بيتِهِ وقالَ لَه: ”لا تَدخُلِ القَريةَ“.</span>
</div>
</div>

## بطرس يشهد بحقيقة يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 16:13–20; لُوقا 9:18–20"> وسارَ يَسوعُ وتلاميذُهُ إلى قُرى قَيصريَّةِ فيلبُّسَ، وفي الطَّريقِ سألَ تلاميذَهُ: ”مَنْ أنا في رأيِ النَّاسِ؟“</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 1:21; ﻣﺘﻰ 14:2; مَلاخِي 4:5"> فأَجابوهُ: ”يوحنَّا المَعمَدانُ، وبَعضُهُم يَقولُ: إيليَّا، وبَعضُهُم: أحدُ الأنبياءِ“.</span>
<span className="v" id="v29" data-v="29" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:15; ﻣﺘﻰ 16:15–16; رسالَةُ يُوحَنّا الأُولَى 5:1"> فسألَهُم: ”ومَنْ أنا في رأيِكُم أنتُم؟“ فأجابَ بُطرُسُ: ”أنتَ المَسيحُ!“</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 9:21; ﻣﺘﻰ 16:20; مَرْقُس 9:9"> فأمرَهُم أنْ لا يُخبِروا أحدًا.</span>
</div>

## يسوع يُنبئ أول مرة بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="مَرْقُس 10:33–34; يُوحَنّا 2:19; لُوقا 24:26"> وبدأَ يُعلِّمُهُم أنَّ اَبنَ الإنسانِ يَجبُ أنْ يتألَّمَ كثيرًا، وأنْ يَرفُضَهُ الشُّيوخُ ورُؤَساءُ الكَهنَةِ ومُعَلِّمو الشَّريعَةِ، وأنْ يَموتَ قَتْلًا، وأنْ يَقومَ بَعدَ ثلاثَةِ أيّامٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 16:22; يُوحَنّا 16:25; يُوحَنّا 16:29"> وكانَ يَقولُ هذا الكلامَ علانيَّةً. فاَنفَردَ بِه بُطرُسُ وأخَذَ يُعاتِبُهُ.</span>
<span className="v" id="v33" data-v="33" data-xref="ﻣﺘﻰ 4:10; فِيلِبِّي 3:19; رسالَةُ يُوحَنّا الأُولَى 2:15"> فاَلتَفتَ ورأى تلاميذَهُ، فوَبَّخَ بُطرُسَ بِقولِهِ: ”إذهَبْ وَرَائي يا شَيطانُ، لأنَّ أفكارَكَ هذِهِ أفكارُ البَشرِ، لا أفكارُ اللهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 10:38; لُوقا 14:26–27; يُوحَنّا 10:27"> ودَعا الجُموعَ وتلاميذَهُ وقالَ لهُم: ”مَنْ أرادَ أنْ يَتبَعَني، فلْيُنكِرْ نَفسَهُ ويَحمِلْ صليبَهُ ويَتبَعْني.</span>
<span className="v" id="v35" data-v="35" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:11–13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:23; لُوقا 17:33"> لأنَّ الّذي يُريدُ أنْ يُخلِّصَ حَياتَهُ يَخسَرُها، ولكن الّذي يَخسَرُ حَياتَهُ في سَبيلي وسَبيلِ البِشارَةِ يُخَلِّصُها.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 16:26; لُوقا 9:25; المَزامِير 49:17"> فماذا يَنفَعُ الإنسانَ لو رَبِحَ العالَمَ كُلَّهُ وخَسِرَ نَفسَهُ؟</span>
<span className="v" id="v37" data-v="37" data-xref="المَزامِير 49:7–8; رسالَةُ بُطرُسَ الأُولَى 1:18–19"> وبماذا يَفدي الإنسانُ نَفسَهُ؟</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 12:8–9; ﺭﻭﻣﻴﺔ 1:16; رسالَةُ يُوحَنّا الأُولَى 2:23"> لأنَّ مَنْ يَستَحي بي وبِكَلامي في هذا الجِيلِ الخائِنِ الشِّرِّيرِ، يَستَحي بِه اَبنُ الإنسانِ متى جاءَ في مَجدِ أبيهِ معَ الملائِكَةِ الأطهارِ“.</span>
