## شفاء كسيح في كفرناحوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 4:47; لُوقا 5:18; لُوقا 18:35–38"> ورجَعَ يَسوعُ بَعدَ أيّامٍ إلى كَفْرَناحومَ، فسَمِعَ النَّاسُ أنَّهُ في البَيتِ.</span>
<span className="v" id="v2" data-v="2" data-xref="مَرْقُس 2:13; مَرْقُس 1:45; المَزامِير 40:9"> فتجَمَّعَ مِنهُم عددٌ كبيرٌ ملأَ المكانَ حتّى عِندَ البابِ، فوَعظَهُم بِكلامِ اللهِ.</span>
<span className="v" id="v3" data-v="3" data-xref="لُوقا 5:18–26; ﻣﺘﻰ 9:1–8; ﻣﺘﻰ 4:24"> وجاءَ إليهِ أربعةُ رِجالٍ يَحمِلونَ كسيحًا.</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 5:19; التَّثْنِيَة 22:8"> فلمَّا عَجِزوا عَن الوُصولِ بِه إليهِ لِكثرَةِ الزِّحامِ، نقَبُوا السَّقفَ وكشَفوا فوقَ المكانِ الّذي كانَ فيهِ يَسوعُ ودلُّوا الكسيحَ وهوَ على فراشِهِ.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 9:2; يَعقُوب 5:15; مَرْقُس 2:9–10"> فلمَّا رأى يَسوعُ إيمانَهُم قالَ للكسيحِ: ”يا اَبني، مَغفورَةٌ لكَ خَطاياكَ!“</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="لُوقا 5:21–22; مَرْقُس 8:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:5"> وكانَ بَينَ الحُضورِ بعضُ مُعَلِّمي الشَّريعةِ، فقَالوا في أنفُسِهِم:</span>
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 43:25; لُوقا 5:21; ميخا 7:18"> ”كيفَ يتكَلَّمُ هذا الرَّجُلُ كلامًا كهذا؟ فهوَ يُجدِّفُ! مَنْ يَقدِرُ أن يَغفِرَ الخطايا إلَّا اللهُ وَحدَهُ؟“</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 139:2; اعمال الرسل 8:22; لُوقا 5:22"> وعرَفَ يَسوعُ في سِرِّه أفكارَهُم، قالَ لهُم: ”ما هذِهِ الأفكارُ في قلوبِكُم؟</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 5:22–25; ﻣﺘﻰ 9:5"> أيُّما أسهَلُ: أنْ يُقالَ لِهذا الكسيحِ: مَغفورَةٌ لكَ خَطاياكَ، أمْ أنْ يُقالَ لَه: قُمْ واَحمِلْ فِراشَكَ واَمشِ؟</span>
<span className="v" id="v10" data-v="10" data-xref="دانيال 7:13–14; ﻣﺘﻰ 9:6–8; يُوحَنّا 5:20–27"> سأُريكُم أنَّ اَبنَ الإنسانِ لَه سُلطانٌ على الأرضِ ليَغفِرَ الخَطايا“. وقالَ لِلكسيحِ:</span>
<span className="v" id="v11" data-v="11" data-xref="مَرْقُس 1:41; يُوحَنّا 5:8–10; يُوحَنّا 6:63"> ”أقولُ لكَ: قُمْ واَحْمِلْ فِراشَكَ واَذهبْ إلى بَيتِكَ!“</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 9:33; ﻣﺘﻰ 9:8; لُوقا 7:16"> فقامَ الرَّجُلُ وحمَلَ فِراشَهُ في الحالِ وخرَجَ بِمَشْهدٍ مِنَ الحاضِرينَ. فتَعجَّبوا كُلُّهُم ومَجَّدوا اللهَ وقالوا: ”ما رأينا مِثلَ هذا في حَياتِنا!“</span>
</div>
</div>

## يسوع يدعو لاوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 4:1; الأمثال 1:20–22; لُوقا 19:48"> ورجَعَ يَسوعُ إلى شاطئِ بحرِ الجليلِ. وجاءَهُ جُمهورٌ مِنَ النَّاسِ فأخَذَ يُعلِّمُهُم.</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 5:27–32; لُوقا 6:15; ﻣﺘﻰ 9:9–13"> وبَينَما هوَ سائِرٌ رأى لاوِيَ بنَ حَلْفى جالِسًا في بَيتِ الجبايةِ. فقَالَ لَه يَسوعُ: ”إِتبَعْني!“ فقامَ وتَبِعَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="لُوقا 15:1; لُوقا 5:29–30; ﻣﺘﻰ 9:10–11"> وكانَ يَسوعُ يأكُلُ في بَيتِ لاوِي، فجلَسَ معَهُ ومعَ تلاميذِهِ كثيرونَ مِنَ الّذينَ تَبِعوهُ مِنْ جُباةِ الضَّرائبِ والخاطِئينَ.</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 19:10; لُوقا 19:7; لُوقا 18:11"> فلمَّا رأى بَعضُ مُعَلِّمي الشَّريعةِ مِنَ الفَرِّيسيِّينَ أنَّهُ يأكُلُ معَ جُباةِ الضَّرائِبِ والخاطِئينَ، قالوا لِتلاميذِهِ: ”ما بالُهُ يأكُلُ ويَشربُ معَ جُباةِ الضَّرائِبِ والخاطئينَ!“</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 9:12–13; لُوقا 15:7; لُوقا 5:31–32"> فسَمِعَ يَسوعُ كلامَهُم، فقَالَ لهُم: ”لا يَحتاجُ الأصِحّاءُ إلى طبيبٍ، بلِ المَرضى. ما جِئتُ لأدعُوَ الأبرارَ، بلِ الخاطِئينَ“.</span>
</div>
</div>

## الصوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 18:12; ﻣﺘﻰ 6:18; ﻣﺘﻰ 9:14–17"> وكانَ تلاميذُ يوحنَّا والفَرِّيسيُّونَ صائِمينَ، فَجاءَ بعضُ النَّاسِ إلى يَسوعَ وقالوا لَه: ”لماذا يَصومُ تلاميذُ يوحنَّا وتلاميذُ الفَرِّيسيِّينَ، ولا يَصومُ تلاميذُكَ؟“</span>
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 25:1–10; التَّكوين 29:22; القُضاة 14:10–11"> فقَالَ لهُم: ”أتَنْتَظِرونَ مِنْ أهلِ العريسِ أنْ يَصوموا والعريسُ بينَهُم؟ فما دامَ العريسُ بينَهُم، لا يَقدِرونَ أنْ يَصوموا.</span>
<span className="v" id="v20" data-v="20" data-xref="لُوقا 17:22; رُؤيا يُوحَنّا 21:9; إشَعْياء 54:5"> ولكِنْ يَجيءُ وقتٌ يُرفَعُ فيهِ العريسُ مِنْ بَينِهِم وفي ذلِكَ الوقتِ يَصومونَ.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 9:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:13; إشَعْياء 57:16"> ما مِنْ أحدٍ يَرقَعُ ثوبًا عتيقًا برُقعةٍ مِنْ قماشٍ جديدٍ، لِئلَّا تَنكَمِشَ فتَنتَزع الرُقعةُ الجديدةُ شَيئًا مِنَ الثَّوبِ العَتيقِ فيتَّسع الخَرْقُ.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 9:17; المَزامِير 119:83; لُوقا 5:37–38"> وما مِنْ أحدٍ يضَعُ خَمرًا جديدةً في أوعِيَةٍ مِنْ جلدٍ عتيقةٍ، لِئلَّا تَشُقَّ الخَمرُ الجديدةُ الأوعيةَ، فَتَتلف الخَمرُ والأَوعيةُ معًا. ولكِنْ للخَمرِ الجديدةِ أوعِيَةٌ جديدةٌ!“</span>
</div>
</div>

## السَّبت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 12:1–8; التَّثْنِيَة 23:24–25; لُوقا 6:1–5"> ومَرَّ في السَّبتِ بَينَ المزارعِ، فأخَذَ تلاميذُهُ يَقطِفونَ السُّنبُلَ وهُم سائِرونَ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 20:10; العِبرانِيّين 12:3; ﻣﺘﻰ 12:2"> فقَالَ لَه الفَرِّيسيُّونَ: ”أُنظُرْ! لماذا يَعمَلُ تلاميذُكَ ما لا يَحلُّ في السَّبتِ؟“</span>
<span className="v" id="v25" data-v="25" data-xref="صَمُوئيلَ الأوَّلُ 21:3–6; لُوقا 10:26; ﻣﺘﻰ 22:31"> فقَالَ لهُم: ”أما قَرأتُم ما عمِلَ داودُ عِندَما أحوجَهُ الجُوعُ هوَ ورِجالُهُ؟</span>
<span className="v" id="v26" data-v="26" data-xref="صموئيلَ الثّانِي 8:17; اللّاويِّين 24:5–9; صَمُوئيلَ الأوَّلُ 21:1"> كيفَ دخَلَ بَيتَ اللهِ في أيّامِ أبياتارَ رئيسِ الكَهنَةِ، فأكَلَ خُبزَ القُربانِ وأعطى مِنهُ رِجالَهُ، وأكْلُهُ لا يَحِلُّ إلَّا للكَهنَةِ“.</span>
<span className="v" id="v27" data-v="27" data-xref="كُولُوسِي 2:16; لُوقا 6:9; التَّثْنِيَة 5:14"> وقالَ يَسوعُ: ”اللهُ جعَلَ السَّبتَ للإنسانِ، وما جعلَ الإنسانَ للسَّبتِ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 12:8; رُؤيا يُوحَنّا 1:10; يُوحَنّا 5:17"> فاَبنُ الإنسانِ رَبُّ السَّبتِ أيضًا“.</span>
</div>
