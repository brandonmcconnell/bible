## قيامة يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 23:56–12; يُوحَنّا 20:1–10; ﻣﺘﻰ 28:1–10"> ولمَّا مَضى السَّبتُ، اَشتَرَت مَريَمُ المَجدَليَّةُ، ومَريَمُ أمُّ يَعقوبَ، وسالومةُ، بَعضَ الطِّيبِ ليَذهبنَ ويَسكُبْنَهُ على جَسدِ يَسوعَ.</span>
<span className="v" id="v2" data-v="2" data-xref="يُوحَنّا 20:1; لُوقا 24:1; ﻣﺘﻰ 28:1"> وفي صباحِ يومِ الأحدِ، عِندَ طُلوعِ الشَّمسِ، جِئنَ إلى القبرِ.</span>
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 15:46–47; ﻣﺘﻰ 27:60–66"> وكانَ يَقولُ بَعضُهُنَّ لبَعضٍ: ”مَنْ يُدحرِجُ لنا الحجَرَ عَنْ بابِ القَبرِ؟“</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 28:2–4; لُوقا 24:2; يُوحَنّا 20:1"> فلمَّا تَطلَّعْنَ وجَدنَ الحجَرَ مُدحرَجًا، وكانَ كبيرًا جدًّا.</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 20:11–12; لُوقا 24:3–5; يُوحَنّا 20:8"> فدخَلْنَ القَبرَ، فرأينَ شابًا جالِسًا عَنِ اليَمينِ، علَيهِ ثوبٌ أبيضُ، فاَرتَعَبْنَ.</span>
<span className="v" id="v6" data-v="6" data-xref="رُؤيا يُوحَنّا 1:17–18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:3–7; لُوقا 24:46"> فقَالَ لهُنَّ: ”لا تَرتَعِبْنَ! أنتُنَّ تَطلُبنَ يَسوعَ النّاصريَّ المَصلوبَ. ما هوَ هُنا، بل قامَ. وهذا هوَ المكانُ الّذي وضَعوهُ فيهِ.</span>
<span className="v" id="v7" data-v="7" data-xref="مَرْقُس 14:28; ﻣﺘﻰ 26:32; ﻣﺘﻰ 28:10"> فاَذهَبْنَ وقُلْنَ لِتلاميذِه ولبطرسَ: هوَ يَسبِقُكُم إلى الجليلِ، وهُناكَ تَرَوْنَهُ كما قالَ لكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="لُوقا 24:22–24; لُوقا 24:9–11; مَرْقُس 16:5–6"> فخَرَجْنَ مِنَ القبرِ هارِباتٍ مِنْ شِدَّةِ الحَيرةِ والفَزَعِ. وما أخْبرنَ أحدًا بشيءٍ لأنَّهُنَّ كُنَّ خائِفاتٍ.</span>
</div>
</div>

## يسوع يظهر لمريم المجدلية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 20:14–19; لُوقا 8:2; يُوحَنّا 20:12"> &#91;وبَعدَما قامَ يَسوعُ في صَباحِ الأحدِ، ظَهرَ أوَّلًا لِمَريَمَ المَجدَليَّةِ الّتي أخرَجَ مِنها سَبعةَ شياطينَ.</span>
<span className="v" id="v10" data-v="10" data-xref="يُوحَنّا 20:18; لُوقا 24:17; يُوحَنّا 16:6"> فذَهَبَت وأخبَرَت تلاميذَهُ، وكانوا يَنوحونَ ويَبكونَ،</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 24:11; مَرْقُس 16:13–14; مَرْقُس 9:19"> فما صَدَّقوها عِندَما سَمِعوا أنَّهُ حيٌّ وأنَّها رأتْهُ.</span>
</div>
</div>

<div className="bt" dir="rtl">
<div className="bp">

<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 20:25; لُوقا 24:33–35; يُوحَنّا 20:8"> فرَجَعا وأخبرا الآخرينَ، فما صدَّقوهُما.</span>
</div>
</div>

## يسوع يظهر للرسل الأحد عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:5; لُوقا 24:36–43; العِبرانِيّين 3:7–8"> وظهَرَ آخِرَ مرَّةٍ لِتلاميذِهِ الأحَدَ عشَرَ، وهُم يتَناولونَ الطَّعامَ، فلامَهُم على قِلَّةِ إيمانِهِم وقَساوَةِ قُلوبِهِم، لأنَّهُم ما صَدَّقوا الّذينَ شاهَدوهُ بَعدَما قامَ.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 28:19; مَرْقُس 13:10; اعمال الرسل 1:8"> وقالَ لهُم: ”إِذهَبوا إلى العالَمِ كُلِّهِ، وأعلِنوا البِشارةَ إلى النَّاسِ أجمعينَ.</span>
<span className="v" id="v16" data-v="16" data-xref="يُوحَنّا 11:25–26; يُوحَنّا 3:36; ﺭﻭﻣﻴﺔ 10:9–14"> كُلُّ مَنْ يُؤمِنُ ويتَعَمَّدُ يَخلُصُ، ومَنْ لا يُؤمِنُ يَهلِكُ.</span>
<span className="v" id="v17" data-v="17" data-xref="اعمال الرسل 5:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:28; اعمال الرسل 8:7"> والّذينَ يُؤمِنونَ تُسانِدُهُم هذِهِ الآياتُ: يَطرُدونَ الشَّياطينَ باَسمي، ويَتكلَّمونَ بِلُغاتٍ جَديدةٍ،</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 10:19; يَعقُوب 5:14–15; اعمال الرسل 28:3–6"> ويُمسِكونَ بأيديهِم الحيّاتِ. وإنْ شَرِبوا السُمَّ لا يُصيبُهُم أذًى، ويَضعونَ أيديَهُم على المَرضى فيَشفونَهُم“.</span>
</div>
</div>

<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:4–5; اعمال الرسل 14:3; ﺭﻭﻣﻴﺔ 15:19"> وأمَّا التَّلاميذُ، فذَهَبوا يُبشِّرونَ في كُلِّ مكانٍ، والرَّبُّ يُعينُهُم ويُؤيِّدُ كلامَهُم بِما يُسانِدُهُ مِن الآياتِ&#93;.</span>
