## الزواج والطلاق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 10:40; مَرْقُس 4:2; يُوحَنّا 11:7"> وقامَ يَسوعُ مِنْ هُناكَ وجاءَ إلى بلادِ اليهوديَّةِ مِنْ عَبرِ الأُردُنِّ، فأقبلت إليهِ الجموعُ وأخَذَ يعلّمُهُم كعادَتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:10–11; لُوقا 11:53–54; يُوحَنّا 8:6"> فدَنا بَعضُ الفَرِّيسيِّينَ وسَألوهُ ليُحرِجوهُ: ”أيَحِلُّ لِلرَّجُلِ أنْ يُطَلِّقَ اَمرَأتَهُ؟“</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 8:20; غَلاطِيَّة 4:21; لُوقا 10:25"> فأجابَهُم: ”بماذا أوصاكُم موسى؟“</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 24:1–4; ﻣﺘﻰ 5:31–32; إشَعْياء 50:1"> قالوا: ”أجازَ موسى لِلرَجُلِ أنْ يَكتُبَ لاَمرأتِهِ كِتابَ طَلاقٍ فتُطلَّقُ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 19:8; نَحَمْيا 9:26; اعمال الرسل 7:51"> فقَالَ لهُم يَسوعُ: ”لِقساوَةِ قُلوبِكُم كَتبَ لكُم موسى هذِهِ الوصيَّةَ.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 1:27; التَّكوين 5:2; التَّكوين 1:1"> فمِنْ بَدءِ الخَليقَةِ جَعلَهُما اللهُ ذكَرًا وأُنثى.</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 2:24; ﻣﺘﻰ 19:5–6; أفَسُس 5:31"> ولذلِكَ يَترُكُ الرَّجُلُ أباهُ وأُمَّهُ ويتَّحِدُ باَمرأتِهِ،</span>
<span className="v" id="v8" data-v="8" data-xref="أفَسُس 5:28; التَّكوين 2:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:16"> فيَصيرُ الاثنانِ جسَدًا واحدًا. فلا يكونانِ اَثنَينِ، بل جَسدٌ واحدٌ.</span>
<span className="v" id="v9" data-v="9" data-xref="ﺭﻭﻣﻴﺔ 7:1–3; أخبارِ الأيّامِ الأوّلُ 7:10–13"> وما جَمَعَهُ اللهُ لا يُفَرِّقُهُ الإنسانُ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="مَرْقُس 9:33; مَرْقُس 9:28; مَرْقُس 4:10"> ولمَّا دَخَلوا البَيتَ، سألَهُ التَّلاميذُ عَنْ هذا الأمرِ،</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 5:31–32; ﻣﺘﻰ 19:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:10–11"> فقَالَ لهُم: ”مَنْ طلَّقَ اَمرأتَهُ وتَزوَّجَ غَيرَها زنى علَيها،</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:13"> وإنْ طَلَّقتِ اَمرأةٌ زوجَها وتَزوَّجت غيرَهُ زَنَت“.</span>

## الرجل الغني

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 19:16–30; لُوقا 18:18–30; ﺭﻭﻣﻴﺔ 2:7"> وخرَجَ إلى الطَّريقِ، فأسرَعَ إلَيهِ رَجُلٌ وسَجَدَ لَه وسألَهُ: ”أيُّها المُعَلِّمُ الصَّالِحُ، ماذا أعمَلُ لأرِثَ الحياةَ الأبديَّةَ؟“</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 18:19; ﻣﺘﻰ 19:17; ﺭﻭﻣﻴﺔ 3:12"> فقَالَ لَه يَسوعُ: ”لماذا تَدعوني صالِحًا؟ لا صالِحَ إلَّا اللهَ وحدَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 20:12–17; ﺭﻭﻣﻴﺔ 13:9; يَعقُوب 2:11"> أنتَ تَعرِفُ الوصايا: لا تَقتُلْ، لا تَزْنِ، لا تَسرِقْ، لا تَشهَدْ بالزّورِ، لا تَظلِمْ، أكرِمْ أباكَ وأُمَّكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="حِزْقِيال 33:31–32; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:5; فِيلِبِّي 3:6"> فأجابَهُ الرَّجُلُ: ”يا مُعَلِّمُ، مِنْ أيّامِ صِبايَ عَمِلتُ بِهذِهِ الوصايا كُلِّها“.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 6:19–21; رُؤيا يُوحَنّا 2:4; لُوقا 18:22"> فنَظَرَ إلَيهِ يَسوعُ بِمحَبَّةٍ وقالَ لَه: ”يُعْوِزُكَ شيءٌ واحدٌ: اَذهَبْ بِعْ كُلَّ ما تَملِكُه ووَزِّعْ ثَمنَهُ على الفُقَراءِ، فيكونَ لكَ كَنْزٌ في السَّماءِ، وتَعالَ اَتْبَعني“.</span>
<span className="v" id="v22" data-v="22" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:15–16; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9–10; لُوقا 12:15"> فحَزِنَ الرَّجُلُ لِهذا الكلامِ ومَضى كَئيبًا، لأنَّهُ كانَ يَملِكُ أموالًا كثيرةً.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:26; لُوقا 18:24; مَرْقُس 3:5"> فنَظَرَ يَسوعُ حولَهُ وقالَ لِتلاميذِهِ: ”ما أصعَبَ دُخولَ الأغنياءِ إلى مَلكوتِ اللهِ!“</span>
<span className="v" id="v24" data-v="24" data-xref="يَعقُوب 5:1–3; الأمثال 11:28; المَزامِير 52:7"> فاَستَغْربَ التَّلاميذُ كلامَهُ، فقَالَ لهُم ثانيةً: ”يا أبنائي، ما أصعَبَ الدُّخولَ إلى مَلكوتِ اللهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 19:24–25; لُوقا 18:25; ﻣﺘﻰ 23:24"> فَمُرورُ الجمَلِ في ثَقْبِ الإبرةِ أسهلُ مِنْ دُخولِ الغنيِّ إلى مَلكوتِ اللهِ“.</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 16:31; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:23; ﺭﻭﻣﻴﺔ 10:9–13"> فزادَ اَستِغرابُهُم وتَساءَلوا: ”مَنْ يُمكِنُهُ أنْ يخلُصَ، إذًا؟“</span>
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 19:26; إرْمِيا 32:27; أيُّوب 42:2"> فنظَرَ إلَيهِم يَسوعُ وقالَ: ”هذا شيءٌ غَيرُ مُمكِنٍ عِندَ النَّاسِ لا عِندَ اللهِ، فعِندَ اللهِ كُلُّ شيءٍ مُمكِنٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 19:27–30; لُوقا 18:28–30; لُوقا 14:33"> فقَالَ لَه بُطرُسُ: ”ها نَحنُ تَركنا كُلَّ شيءٍ وتَبِعْناكَ“.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:23; ﻣﺘﻰ 5:10–11; مَرْقُس 8:35"> فأجابَهُ يَسوعُ: ”الحقَّ أقولُ لكُم: ما مِنْ أحدٍ ترَكَ بَيتًا أو إخوَةً أو أخواتٍ أو أُمًّا أو أبًا أو أولادًا أو حُقولًا مِنْ أجلي ومِنْ أجلِ البِشارَةِ،</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 18:30; فِيلِبِّي 3:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:10"> إلَّا نالَ في هذِهِ الدُّنيا، معَ الاضطِهاداتِ، مئةَ ضُعفٍ مِنَ البيوتِ والإخوةِ والأخواتِ والأُمَّهاتِ والأولادِ والحُقولِ، ونالَ في الآخِرَةِ الحياةَ الأبديَّةَ.</span>
<span className="v" id="v31" data-v="31" data-xref="ﻣﺘﻰ 19:30; لُوقا 13:30; ﻣﺘﻰ 20:16"> وكثيرٌ مِنَ الأوَّلينَ يَصيرونَ آخرينَ، ومِنَ الآخِرينَ يَصيرونَ أوَّلينَ“.</span>
</div>
</div>

## يسوع يُنبئ مرة ثالثة بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 20:17–19; لُوقا 18:31–34; لُوقا 9:51"> وكانوا في الطَّريقِ صاعِدينَ إلى أُورُشليمَ ويَسوعُ يتَقَدَّمُهُم. وكانَ التَّلاميذُ في حَيرةٍ، والّذينَ يَتبَعونَهُ خائِفينَ. فاَنفَرَدَ بالاثني عشَرَ مرَّةً أُخرى وأخَذَ يُكلِّمُهُم بِما سيَحدُثُ لَه،</span>
<span className="v" id="v33" data-v="33" data-xref="ﻣﺘﻰ 27:2; ﻣﺘﻰ 20:17–19; مَرْقُس 8:31"> فقَالَ: ”ها نَحنُ صاعِدونَ إلى أُورُشليمَ، وسيُسلَّمُ اَبنُ الإنسانِ إلى رُؤساءِ الكَهنَةِ ومُعَلِّمي الشَّريعَةِ، فيَحكُمونَ علَيهِ بالموتِ ويُسلِّمونَهُ إلى حُكَّامٍ غُرباءَ،</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 26:67; ﻣﺘﻰ 16:21; مَرْقُس 14:65"> فيَستَهزِئونَ بِه، ويَبصِقونَ علَيهِ ويَجلِدونَهُ ويَقتُلونَهُ، وبَعدَ ثلاثةِ أيّامٍ يَقومُ“.</span>
</div>

## طلب يعقوب ويوحنا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 20:20–28; صموئيلَ الثّانِي 14:4–11; المُلُوكِ الأوَّلُ 2:20"> ودَنا إليهِ يَعقوبُ ويوحنَّا، اَبنا زَبْدي، وقالا لَه: ”يا مُعَلِّمُ، نُريدُ أنْ تُلَبِّيَ طَلَبَنا“.</span>
<span className="v" id="v36" data-v="36" data-xref="يُوحَنّا 15:7; المُلُوكِ الأوَّلُ 3:5–15; مَرْقُس 10:51"> فقَالَ لهُما: ”ماذا تُريدانِ أنْ أعمَلَ لكُما؟“</span>
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 19:28; مَرْقُس 16:19; لُوقا 24:26"> فأجابا: ”أعطِنا أنْ نَجلِسَ، واحدٌ عَنْ يَمينِكَ وواحدٌ عَنْ شِمالِكَ في مَجدِكَ“.</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 12:50; ﻣﺘﻰ 20:21–22; المَزامِير 75:8"> فقَالَ لهُما يَسوعُ: ”أنتُما لا تَعرِفانِ ما تَطلُبانِ: أتقدِرانِ أنْ تَشرَبا الكأسَ الّتي سأشرَبُها، أو تَقبلا مَعمودِيَّةَ الآلامِ الّتي سأقبَلُها؟“</span>
<span className="v" id="v39" data-v="39" data-xref="اعمال الرسل 12:2; رُؤيا يُوحَنّا 1:9; يُوحَنّا 17:14"> فأجابا: ”نَقدِرُ“. فقَالَ لهُما: ”نَعَم، الكأسُ الّتي أشرَبُها تَشرَبانِها، ومَعمودِيَّةُ الآلامِ الّتي أقبَلُها تقبَلانِها.</span>
<span className="v" id="v40" data-v="40" data-xref="ﻣﺘﻰ 20:23; ﻣﺘﻰ 25:34; يُوحَنّا 17:24"> وأمَّا الجُلوسُ عَنْ يَميني أو عَنْ شِمالي، فلا يَحقُّ لي أنْ أُعطِيَهُ، لأنَّهُ للَّذينَ هيَّأهُ اللهُ لهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="ﻣﺘﻰ 20:24; الأمثال 13:10; ﺭﻭﻣﻴﺔ 12:10"> وسَمِعَ التَّلاميذُ العَشرَةُ هذا الكلامَ، فغَضِبوا على يَعقوبَ ويوحنَّا.</span>
<span className="v" id="v42" data-v="42" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:3; لُوقا 22:25; ﻣﺘﻰ 20:25"> فدَعاهُم يَسوعُ وقالَ لهُم: ”تَعلَمونَ أنَّ رُؤَساءَ الأُمَمِ يَسودونَها، وأنَّ عُظَماءَها يتَسَلَّطونَ علَيها.</span>
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 23:8–12; لُوقا 14:11; مَرْقُس 9:35"> فلا يكُنْ هذا فيكُم، بل مَنْ أرادَ أنْ يكونَ عَظيمًا فيكُم، فلْيكُنْ لكُم خادِمًا.</span>
<span className="v" id="v44" data-v="44"> ومَنْ أرادَ أنْ يكونَ الأوَّلَ فيكُم، فليَكُنْ لجَميعِكُم عَبدًا.</span>
<span className="v" id="v45" data-v="45" data-xref="ﻣﺘﻰ 20:28; فِيلِبِّي 2:5–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:9"> لأنَّ اَبنَ الإنسانِ جاءَ لا ليَخدِمَهُ النَّاسُ، بل ليَخدِمَهُم ويَفديَ بِحياتِهِ كثيرًا مِنهُم“.</span>
</div>
</div>

## شفاء برتيماوس الأعمى

<div className="bt" dir="rtl">
