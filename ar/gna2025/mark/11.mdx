## يسوع يدخل أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 21:1–11; يُوحَنّا 12:14–19; لُوقا 19:29–40"> ولمَّا قَرُبوا مِنْ أُورُشليمَ ووَصَلوا إلى بَيتِ فاجي وبَيتِ عَنيا، عِندَ جبَلِ الزَّيتونِ، أرسَلَ يَسوعُ اَثنينِ مِنْ تلاميذِهِ</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 21:2–3; لُوقا 19:30–31"> وقالَ لهُما: ”إِذهَبا إلى القريَةِ الّتي أمامَكُما، وحالَما تَدخُلانِها تَجِدانِ جَحشًا مَربوطًا ما ركِبَ علَيهِ أحدٌ، فَحُلَّا رِباطَهُ وجيئا بِه.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 10:36; مَرْقُس 14:15; المَزامِير 110:3"> وإنْ سألَكُما أحدٌ: لِماذا تَفعَلانِ هذا، فَقولا: الرَّبُّ مُحتاجٌ إليهِ، وسيُعيدُهُ إلى هُنا في الحالِ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 26:19; لُوقا 19:32–34; ﻣﺘﻰ 21:6–7"> فذَهَبَ التِّلميذانِ فوجَدا جَحشًا مَربوطًا عِندَ بابٍ على الطَّريقِ، فحَلَّا رباطَهُ.</span>
<span className="v" id="v5" data-v="5"> فسألَهُما بَعضُ الّذينَ كانوا هناكَ: ”ما بالُكُما تَحُلَّانِ رِباطَ الجحشِ؟“</span>
<span className="v" id="v6" data-v="6"> فَقالا لهُم كما أوصاهُما يَسوعُ، فتَركوهُما.</span>
<span className="v" id="v7" data-v="7" data-xref="زَكَريّا 9:9; لُوقا 19:35–36; ﻣﺘﻰ 21:7–8"> فجاءَ التِّلميذانِ بالجحشِ إلى يَسوعَ، ووضَعا ثوبَيهِما علَيهِ فركِبَه.</span>
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 23:40"> وبَسَطَ كثيرٌ مِنَ النَّاسِ ثيابَهُم على الطَّريقِ، وقطَعَ آخرونَ أغصانًا مِنَ الحُقولِ.</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 118:25–26; ﻣﺘﻰ 21:9; ﻣﺘﻰ 23:39"> وكانَ الّذينَ يتَقَدَّمونَ يَسوعَ والّذينَ يَتْبعونَهُ يَهتِفُونَ: ”المَجدُ للهِ! تبارَكَ الآتي باَسمِ الرَّبِّ.</span>
<span className="v" id="v10" data-v="10" data-xref="حِزْقِيال 37:24–25; لُوقا 2:14; لُوقا 19:38–40"> تباركَتِ المَملكةُ الآتِيَةُ، مَملكةُ أبينا داودَ. المَجدُ في العُلى!“</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 8:1–2; لُوقا 19:41–45; لُوقا 21:37–38"> ودخَلَ يَسوعُ أُورُشليمَ والهَيكَلَ، ونَظرَ كُلَّ شيءٍ فيهِ. وكانَ الوَقتُ فاتَ، فخَرَجَ إلى بَيتِ عَنيا معَ التَّلاميذِ الاثنَي عشَرَ.</span>
</div>
</div>

## يسوع يلعن شجرة التين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 21:18–22; ﻣﺘﻰ 4:2; يُوحَنّا 19:28"> ولمَّا خَرَجوا في الغَدِ مِنْ بَيتِ عَنيا أحسَّ بالجوعِ.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 21:19; لُوقا 13:6–9; إشَعْياء 5:7"> ورأى عَنْ بُعدٍ شجَرَةَ تِينٍ مُورِقَةً، فقَصَدها راجيًا أنْ يَجِدَ علَيها بَعضَ الثَّمَرِ. فلمَّا وصَلَ إليها، ما وجَدَ علَيها غَيرَ الوَرَقِ، لأنَّ وَقتَ التِّينِ ما حانَ بَعدُ.</span>
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 21:19; ﻣﺘﻰ 7:19; ﻣﺘﻰ 3:10"> فقَالَ لها: ”لا يأكُلْ أحَدٌ ثَمرًا مِنكِ إلى الأبدِ“. وسَمِعَ تلاميذُهُ ما قالَ.</span>
</div>
</div>

## يسوع يطرد الباعة من الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 2:13–17; ﻣﺘﻰ 21:12–16; لُوقا 19:45–47"> وجاؤُوا إلى أُورُشليمَ، فدخَلَ الهَيكَلَ وأخَذَ يَطرُدُ الّذينَ يَبيعونَ ويَشتَرونَ فيهِ. وقَلَبَ مناضِدَ الصَّيارِفَةِ ومَقاعِدَ باعَةِ الحَمامِ،</span>
<span className="v" id="v16" data-v="16"> ومنَعَ كُلَّ مَنْ يَحمِلُ بِضاعَةً أن يمُرَّ مِنْ داخِلِ الهَيكَلِ.</span>
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 56:7; إرْمِيا 7:11; لُوقا 19:46"> وأخَذَ يُعلِّمُهُم فيَقولُ: ”أما جاءَ في الكِتابِ: بَيتي بيتُ صلاةٍ لِجَميعِ الأُمَمِ، وأنتُم جَعلتُموهُ مَغارةَ لُصوصٍ؟“</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 7:28; مَرْقُس 12:12; مَرْقُس 14:1–2"> وسَمِعَ رُؤَساءُ الكَهنَةِ ومُعَلِّمو الشَّريعَةِ هذا الكلامَ، فتَشاوروا كيفَ يَقتُلونَهُ، وكانوا يَخافونَهُ لأنَّ الشَّعبَ كُلَّهُ كانَ مُعجَبًا بِتَعليمِهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 21:37; مَرْقُس 11:11; يُوحَنّا 12:36"> وعِندَ المساءِ خَرَجوا مِنَ المدينةِ.</span>
</div>
</div>

## عبرة شجرة التين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="ﻣﺘﻰ 15:13; مَرْقُس 11:14; يُوحَنّا 15:6"> وبَينَما هُمْ راجِعونَ في الصَّباحِ، رأَوا شجَرَة التِّينِ يابِسَةً مِنْ أصولِها.</span>
<span className="v" id="v21" data-v="21" data-xref="الأمثال 3:33; ﻣﺘﻰ 23:7; ﻣﺘﻰ 25:41"> وتَذكَّرَ بُطرُسُ كلامَ يَسوعَ فقَالَ لَهُ: ”أُنظُرْ، يا مُعَلِّمُ! التّينةُ الّتي لعَنْتَها يَبِسَت“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 62:8; يُوحَنّا 14:1; مَرْقُس 9:23"> فقَالَ لهُم يَسوعُ: ”آمِنوا بِاللهِ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 17:20; يُوحَنّا 15:7; ﻣﺘﻰ 21:21"> الحقَّ أقولُ لكُم: مَنْ قالَ لِهذا الجبَلِ: قُمْ واَنطَرِحْ في البحرِ، وهوَ لا يَشُكُّ في قَلبِه، بَلْ يُؤمِنُ بِأنَّ ما يقولُهُ سيكونُ، تَمَّ لَهُ ذلِكَ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 21:22; رسالَةُ يُوحَنّا الأُولَى 5:14–15; رسالَةُ يُوحَنّا الأُولَى 3:22"> ولِهذا أقولُ لكُم: كُلُّ ما تَطلُبونَهُ في صَلواتِكُم، آمِنوا بأنَّكُم نِلتُموهُ يَتِمُّ لكُم.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 6:14–15; كُولُوسِي 3:13; أفَسُس 4:32"> وإذا قُمتُم لِلصلاةِ وكانَ لكُم شيءٌ على أحَدٍ فاَغفِروا لَهُ، حتّى يغفِرَ لكُم أبوكُمُ الّذي في السَّماواتِ زلَّاتِكُم.</span>
<span className="v" id="v26" data-v="26"> &#91;وإنْ كُنتُم لا تَغفِرونَ للآخرينَ، لا يَغفِرُ لكُم أبوكُمُ الّذي في السَّماواتِ زلَّاتِكُم“&#93;.</span>
</div>
</div>

## السلطة المعطاة ليسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="لُوقا 20:1–8; ﻣﺘﻰ 21:23–27; المَزامِير 2:1–5"> ورجَعوا إلى أُورُشليمَ. وبَينَما هوَ يتمشَّى في الهَيكَلِ، جاءَ إليهِ رُؤَساءُ الكَهنَةِ ومُعَلِّمو الشَّريعةِ وشُيوخُ الشَّعبِ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 2:14; اعمال الرسل 7:38–39; العَدَد 16:3"> فقَالوا لَهُ: ”بأيَّةِ سُلطَةٍ تَعمَلُ هذِهِ الأعمالَ؟ بل مَنْ أعطاكَ السُّلطَةَ لِتَعمَلَها؟“</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="لُوقا 20:3–8; ﻣﺘﻰ 21:24; إشَعْياء 52:13"> فأجابَهُم يَسوعُ: ”وأنا أسألُكُم سُؤالًا واحدًا. أجيبوني، فأقول لكُم بأيَّةِ سُلطَةٍ أعمَلُ هذِهِ الأعمالَ:</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 3:1–20; يُوحَنّا 3:25–36; مَرْقُس 1:1–11"> مِنْ أينَ ليوحنَّا سُلطةُ المَعموديَّةِ؟ أمِنَ السَّماءِ أمْ مِنَ النَّاسِ؟“</span>
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 1:29; يُوحَنّا 1:34; ﻣﺘﻰ 21:31–32"> فقَالوا في أنفُسِهِم: ”إنْ قُلنا مِنَ اللهِ يَقولُ: فلِماذا ما آمنتُم بِه؟</span>
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 21:46; مَرْقُس 12:12; ﻣﺘﻰ 14:5"> فهَلْ نَقولُ مِنَ النَّاسِ؟“ لكِنَّهم كانوا يَخافونَ الشَّعبَ، لأنَّ الشَّعبَ كُلَّهُ كانَ مُقتَنِعًا بأنَّ يوحنَّا نَبيٌّ.</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 6:9–10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:3–4; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:10–12"> فأجابوا يَسوعَ: ”لا نَعرِفُ!“ فقَالَ لهُم يَسوعُ: ”وأنا لا أقولُ لكُم بأيَّةِ سُلطَةٍ أعمَلُ هذِهِ الأعمالَ“.</span>
</div>
