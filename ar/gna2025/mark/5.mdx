## طرد الأرواح النجسة وغرق الخنازير

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 8:26–39; ﻣﺘﻰ 8:28–34; مَرْقُس 4:35"> ووَصَلوا إلى الشَّاطئِ الآخَرِ مِنْ بحرِ الجليلِ، في ناحِيَةِ الجَراسيِّينَ.</span>
<span className="v" id="v2" data-v="2" data-xref="مَرْقُس 1:23; إشَعْياء 65:4; مَرْقُس 5:8"> ولمَّا نزَلَ مِنَ القاربِ اَستقبَلَهُ رجُلٌ خرَجَ مِنَ المَقابِرِ، وفيهِ رُوحٌ نَجِسٌ.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 65:4; لُوقا 8:29; دانيال 4:32–33"> وكانَ يُقيمُ هُناكَ، ولا يَقدِرُ أحَدٌ أنْ يَربُطَهُ حتّى بسِلسِلَةٍ.</span>
<span className="v" id="v4" data-v="4" data-xref="يَعقُوب 3:7–8"> فكثيرًا ما رَبَطوهُ بالقُيودِ والسلاسِلِ، فكانَ يُقطِّعُ السَّلاسِلَ ويُكسِّرُ القُيودَ، ولا يَقوى أحَدٌ على ضَبْطِهِ.</span>
<span className="v" id="v5" data-v="5" data-xref="أيُّوب 2:7–8; المُلُوكِ الأوَّلُ 18:28; يُوحَنّا 8:44"> وكانَ طَوالَ اللَّيلِ والنَّهارِ في المَقابِرِ والجِبالِ يَصرُخُ ويُجرِّحُ جسَدَهُ بالحِجارَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="يَعقُوب 2:19; المَزامِير 66:3; لُوقا 4:41"> فلمَّا شاهَدَ يَسوعَ عَنْ بُعْدٍ، أسرَعَ إلَيهِ وسجَدَ لَه</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 8:29; اعمال الرسل 16:17; ﻣﺘﻰ 26:63"> وصاحَ بِأَعلى صوتِهِ: ”ما لي ولَكَ، يا يَسوعُ اَبنَ اللهِ العَليِّ؟ أستَحلِفُكَ باللهِ، لا تُعَذِّبْني!“</span>
<span className="v" id="v8" data-v="8" data-xref="مَرْقُس 1:25; مَرْقُس 9:25–26; اعمال الرسل 16:18"> لأنَّ يَسوعَ قالَ لَه: ”أيُّها الرُّوحُ النَّجِسُ أُخرُجْ مِنْ هذا الرَّجُلِ!“</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 8:30; مَرْقُس 5:15; ﻣﺘﻰ 26:53"> فسألَهُ يَسوعُ: ”ما اَسمُكَ؟“ فأجابَ: ”إِسمي جَيشٌ، لأنَّنا كَثيرونَ“.</span>
<span className="v" id="v10" data-v="10" data-xref="مَرْقُس 3:22; مَرْقُس 5:13"> وتَوسَّلَ كثيرًا إلى يَسوعَ أنْ لا يَطرُدَ الأرواحَ النَّجِسَةَ مِنْهُ إلى خارِجِ المِنطَقَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اللّاويِّين 11:7–8; إشَعْياء 65:4; إشَعْياء 66:3"> وكانَ هُناكَ قَطيعٌ كبيرٌ مِنَ الخنازيرِ يَرعى قُرْبَ الجبَلِ.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:11; أيُّوب 1:10–12; لُوقا 22:31–32"> فتَوسَّلَتِ الأرواحُ النَّجِسَةُ إلى يَسوعَ بِقولِها: ”أرسِلْنا إلى تِلكَ الخنازيرِ لنَدخُلَ فيها“.</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الأوَّلُ 22:22; رسالَةُ بُطرُسَ الأُولَى 3:22; أيُّوب 2:6"> فأَذِنَ لها. فخَرَجتِ الأرواحُ النَّجِسَةُ ودَخَلَت في الخنازيرِ. فاَندَفعَ القَطيعُ مِنَ المُنحَدَرِ إلى البحرِ، وعدَدُهُ نَحوَ ألفينِ، فغَرِقَ فيهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 8:34; ﻣﺘﻰ 8:33"> وهرَبَ الرُّعاةُ ونَشروا الخبَرَ في المدينةِ والقُرى، فخرَجَ النَّاسُ ليَرَوا ما جرى.</span>
<span className="v" id="v15" data-v="15" data-xref="كُولُوسِي 1:13; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:7; صَمُوئيلَ الأوَّلُ 6:20–21"> فلمَّا وصَلوا إلى يَسوعَ، شاهدوا الرَّجُلَ الّذي كانَ فيهِ جَيشٌ مِنَ الشَّياطينِ جالِسًا هُناكَ لابِسًا سليمَ العَقلِ. فاَستَولى علَيهِمِ الخوفُ.</span>
<span className="v" id="v16" data-v="16"> فأخبرَهُمُ الشُّهودُ بِما جرى للرَّجُلِ الّذي كانَت فيهِ الشَّياطينُ وبِما أصابَ الخنازيرَ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 8:34; مَرْقُس 1:24; لُوقا 8:37"> فطَلبوا إلى يَسوعَ أن يَرحَلَ عَنْ ديارِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 8:38–39; فِيلِبِّي 1:23–24; مَرْقُس 5:17"> وبَينَما هوَ يركَبُ القارِبَ، طلَبَ إلَيهِ الّذي كانَت فيهِ الشَّياطينُ أن يأخُذَهُ معَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="المَزامِير 66:16; يُوحَنّا 4:29; إشَعْياء 38:9–20"> فما أذِنَ لَه، بل قالَ لَه: ”إِرجِعْ إلى بَيتِكَ وإلى أهلِكَ وأخْبِرْهُم بِما عَمِلَ الرَّبُّ لَكَ وكيفَ رَحِمَكَ“.</span>
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 7:31; ﻣﺘﻰ 4:25"> فذهَبَ الرَّجُلُ وأخذَ يُنادي في المُدُنِ العَشْرِ بِما عَمِلَ يَسوعُ لَه، وكانَ جميعُ النَّاسِ يَتعجَّبونَ.</span>
</div>
</div>

## ابنة يايروس والمرأة الّتي لمست ثوب يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 9:1; مَرْقُس 4:1; لُوقا 8:40"> وعبَرَ يَسوعُ في القارِبِ إلى الشَّاطئِ المُقابِلِ، فتجَمَّعَ حَولَهُ على الشَّاطئِ جُمهورٌ كبيرٌ.</span>
<span className="v" id="v22" data-v="22" data-xref="لُوقا 13:14; اعمال الرسل 18:8; مَرْقُس 5:35–36"> وجاءَ رجُلٌ مِنْ رُؤساءِ المَجمعِ اَسمُهُ يايرُسُ. فلمَّا رأى يَسوعَ وقَعَ على قَدمَيهِ،</span>
<span className="v" id="v23" data-v="23" data-xref="لُوقا 13:13; لُوقا 4:40; مَرْقُس 16:18"> وتوَسَّلَ إليهِ كثيرًا بِقولِهِ: ”إِبنَتي الصَّغيرةُ على فِراشِ الموتِ! تَعالَ وضَعْ يدَكَ علَيها، فتَشفى وتَحيا!“</span>
<span className="v" id="v24" data-v="24" data-xref="مَرْقُس 5:31; مَرْقُس 3:20; لُوقا 7:6"> فذهَبَ معَهُ وتَبِعَهُ جَمعٌ كبيرٌ يَزحمُهُ مِنْ كُلِّ جانِبٍ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 15:25–27; لُوقا 8:43–44; ﻣﺘﻰ 9:20–22"> وكانَت هُناكَ اَمرَأةٌ مُصابَةٌ بِنَزْفِ الدَّمِ مِنِ اَثنتَي عَشْرَةَ سنَةً،</span>
<span className="v" id="v26" data-v="26" data-xref="إرْمِيا 30:12–13; إرْمِيا 8:22; المَزامِير 108:12"> عالَجَها أطِبّاءُ كثيرونَ، وأنفَقَت كُلَّ ما تَملِكُ، فما اَستَفادَت شيئًا، لا بل صارَت مِنْ سيِّئٍ إلى أسوأَ.</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 19:12; ﻣﺘﻰ 14:36; اعمال الرسل 5:15"> فلمَّا سَمِعَت بأخبارِ يَسوعَ، دخَلَت بَينَ الجُموعِ مِنْ خَلفٍ ولَمَسَت ثَوبَهُ،</span>
<span className="v" id="v28" data-v="28"> لأنَّها قالَت في نَفْسِها: ”يكْفي أنْ ألمسَ ثيابَهُ لأُشفى“.</span>
<span className="v" id="v29" data-v="29" data-xref="مَرْقُس 5:34; المَزامِير 30:2; مَرْقُس 3:10"> فاَنقَطعَ نَزْفُ دمِها في الحالِ، وأحَسَّت في جِسمِها أنَّها شُفِيَت مِنْ دائِها.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="لُوقا 6:19; لُوقا 8:46; لُوقا 5:17"> وشعَرَ يَسوعُ في الحالِ بِقُوَّةٍ خَرجَت مِنهُ. فاَلتَفتَ إلى الجُموعِ وقالَ: ”مَنْ لَمَسَ ثيابي؟“</span>
<span className="v" id="v31" data-v="31" data-xref="لُوقا 8:45; لُوقا 9:12"> فقَالَ لَه تلاميذُهُ: ”تَرى النَّاسَ يَزحَمونَكَ وتَسألُ: مَنْ لَمَسني؟“</span>
<span className="v" id="v32" data-v="32"> ولكِنَّهُ نظَرَ حَولَهُ ليَرى الّتي فَعلَت ذلِكَ.</span>
<span className="v" id="v33" data-v="33" data-xref="لُوقا 8:47; لُوقا 1:29; لُوقا 1:12"> وخافَتِ المَرأةُ واَرتَعبَت لِعِلمِها بِما جرَى لها، فجاءَت وسَجدَت لَه وأخبرَتهُ بالحقيقةِ كُلِّها.</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 8:48; ﻣﺘﻰ 9:22; لُوقا 7:50"> فقَالَ لها: ”يا اَبنتي إِيمانُكِ شَفاكِ. فاَذهَبي بِسلامٍ، وتَعافَي مِنْ دائِكِ“.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="لُوقا 8:49; يُوحَنّا 11:39; مَرْقُس 5:22"> وبَينَما يَسوعُ يتكَلَّمُ وصَلَ رِجالٌ مِنْ دارِ رَئيسِ المَجمعِ يَقولونَ: ”ماتَتِ اَبنتُكَ، فلا حاجَةَ إلى إزعاجِ المُعلِّمِ“.</span>
<span className="v" id="v36" data-v="36" data-xref="لُوقا 8:50; ﻣﺘﻰ 9:28–29; يُوحَنّا 11:40"> فتَجاهَلَ يَسوعُ كلامَهُم، وقالَ لِرئيسِ المَجمعِ: ”لا تَخفْ، يكفي أن تُؤمِنَ“.</span>
<span className="v" id="v37" data-v="37" data-xref="اعمال الرسل 9:40; مَرْقُس 14:33; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:1"> وما أذِنَ لأحَدٍ أنْ يُرافِقَهُ إلَّا بُطرُسَ ويَعقوبَ ويوحنَّا أخا يَعقوبَ.</span>
<span className="v" id="v38" data-v="38" data-xref="ﻣﺘﻰ 9:23–24; لُوقا 8:52–53; مَرْقُس 5:22"> ولمَّا وصَلوا إلى دارِ رَئيسِ المَجمعِ، رأى يَسوعُ الضَّجيجَ وبُكاءَ النَّاسِ وعَويلَهُم.</span>
<span className="v" id="v39" data-v="39" data-xref="اعمال الرسل 20:10; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:10; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:13–14"> فدَخَلَ وقالَ لهُم: ”لِماذا تَضجُّونَ وتَبكونَ؟ ما ماتَتِ الصَّبيَّةُ، لكِنَّها نائِمةٌ!“</span>
<span className="v" id="v40" data-v="40" data-xref="نَحَمْيا 2:19; المُلُوكِ الثّانِي 4:33; ﻣﺘﻰ 9:24–25"> فضَحِكوا علَيهِ. فأخرَجَهُم جَميعًا، ودخَلَ بأبي الصَّبيَّةِ وأُمِّها والّذينَ كانوا معَهُ إلى حَيثُ كانَتِ الصَّبيَّةُ.</span>
<span className="v" id="v41" data-v="41" data-xref="مَرْقُس 1:31; لُوقا 7:14–15; لُوقا 8:54–55"> فأخَذَ بِيَدِها وقالَ لها: ”طَلِيثا قُوم!“ أيْ ”يا صَبيَّةُ أقولُ لَكِ: قُومي!“</span>
<span className="v" id="v42" data-v="42" data-xref="مَرْقُس 4:41; اعمال الرسل 3:10–13; مَرْقُس 7:37"> فقامَتْ في الحالِ وأخذَتْ تَمشي، وكانَتِ اَبنةَ اَثنتَي عَشْرَةَ سنَةً، ففي الحالِ تَعَجَّبوا كثيرًا.</span>
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 8:4; ﻣﺘﻰ 9:30; مَرْقُس 1:43"> فأوصاهُم يَسوعُ بشِدَّةٍ أنْ لا يَعلَمَ أحَدٌ بِما حدَثَ، وأمرَهُم أنْ يُطعِموها.</span>
</div>
