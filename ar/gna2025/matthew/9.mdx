## يسوع يشفي كسيحًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 4:13; لُوقا 8:37; ﻣﺘﻰ 8:18"> فركِبَ يَسوعُ القارِبَ وعَبَر البُحَيْرةَ راجِعًا إلى مدينتِهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="مَرْقُس 2:1–12; لُوقا 5:18–26; إرْمِيا 31:33–34"> فجاءَهُ بَعضُ النَّاسِ بِكَسيحٍ مُلقًى على سَريرٍ. فلمَّا رأى يَسوعُ إيمانَهُم قالَ للكَسيحِ: ”تَشَجَّعْ يا اَبني، مَغْفورَةٌ لكَ خَطاياكَ“.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻣﺘﻰ 26:65; لُوقا 5:21; اللّاويِّين 24:16"> فقَالَ بَعضُ مُعلِّمي الشَّريعةِ في أنفُسِهِم: ”هذا الرَّجُلُ يُجدِّفُ!“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="لُوقا 6:8; ﻣﺘﻰ 12:25; رُؤيا يُوحَنّا 2:23"> وعرَفَ يَسوعُ أفكارَهُم، فقَالَ: ”لماذا تَظُنُّونَ السُّوءَ في قُلوبِكُم؟</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 5:8–14; مَرْقُس 2:9–12; اعمال الرسل 3:16"> أيُّما أسهَلُ؟ أنْ يُقالَ: مغفورَةٌ لكَ خَطاياكَ، أمْ أنْ يُقالَ: قُمْ واَمشِ؟</span>
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 43:25; ﻣﺘﻰ 9:5; يُوحَنّا 20:21–23"> سأُريكُمْ أنَّ اَبنَ الإنسانِ لَه سُلطانٌ على الأرضِ ليَغفِرَ الخطايا“. وقالَ لِلكسيحِ: ”قُمْ واَحمِلْ سَريرَكَ واَذهَبْ إلى بيتِكَ“.</span>
<span className="v" id="v7" data-v="7"> فقامَ الرَّجُلُ وذهَبَ إلى بَيتِه.</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 7:16; ﻣﺘﻰ 15:31; اعمال الرسل 4:21"> فلمَّا شاهَدَ النَّاسُ ما جرَى، خافوا ومَجَّدوا اللهَ الّذي أعطى البشَرَ مِثلَ هذا السُّلطانِ.</span>
</div>
</div>

## يسوع يدعو متَّى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="لُوقا 5:27–38; مَرْقُس 2:14–22; لُوقا 19:2–10"> وسارَ يَسوعُ مِنْ هُناكَ، فرأى رَجُلًا جالِسًا في بَيتِ الجِبايةِ اَسمُهُ متَّى. فَقالَ لَه يَسوعُ: ”إِتْبَعني“. فقامَ وتَبِعَهُ.</span>
<span className="v" id="v10" data-v="10" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:13–16; لُوقا 5:29–32; مَرْقُس 2:15–17"> وبَينَما يَسوعُ يأكُلُ في بَيتِ متَّى، جاءَ كثيرٌ من جُباةِ الضَّرائبِ والخاطِئينَ وجلَسُوا معَ يَسوعَ وتلاميذِهِ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 11:19; لُوقا 5:30; مَرْقُس 2:16"> ورأى بعضُ الفَرِّيسيِّينَ ذلِكَ، فقَالوا لتلاميذِهِ: ”لِماذا يأكُلُ مُعلِّمُكُم معَ جُباةِ الضَّرائبِ والخاطِئينَ؟“</span>
<span className="v" id="v12" data-v="12" data-xref="لُوقا 5:31; رُؤيا يُوحَنّا 3:17–18; مَرْقُس 2:17"> فسَمِعَ يَسوعُ كلامَهُم، فأجابَ: ”لا يَحتاجُ الأصِحَّاءُ إلى طَبيبٍ، بل المَرضى.</span>
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 2:17; لُوقا 19:10; هُوشَع 6:6"> فاَذهَبوا وتَعلَّموا مَعنى هذِهِ الآيةِ: أُريدُ رَحمةً لا ذبيحةً. وما جِئتُ لأدعُوَ الأبرارَ، بَلِ الخاطِئينَ“.</span>
</div>
</div>

## الصوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="لُوقا 5:33–39; مَرْقُس 2:18–22; ﻣﺘﻰ 6:16"> وجاءَ تلاميذُ يوحنَّا المَعمَدانِ إلى يَسوعَ وقالوا لَه: ”لِماذا نَصومُ نَحنُ والفرِّيسيُّونَ كثيرًا، وتلاميذُكَ لا يَصومونَ؟</span>
<span className="v" id="v15" data-v="15" data-xref="رُؤيا يُوحَنّا 19:9; يُوحَنّا 3:29; رُؤيا يُوحَنّا 21:2"> فأجابَهُم يَسوعُ: ”أتَنتَظِرونَ مِنْ أهلِ العَريسِ أنْ يَحزَنوا، والعَريسُ بَينَهُم؟ لكنْ يَجيءُ وقتٌ يُرفَعُ فيهِ العَريسُ مِنْ بَينِهِم فيَصومونَ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:1–2; يُوحَنّا 16:12; المَزامِير 125:3"> ما مِنْ أحَدٍ يَرقَعُ ثَوبًا عَتيقًا بِرُقعَةٍ مِنْ قماشٍ جَديدٍ، لأنَّها تَنكَمِش فَتَنْتَزِع شَيئًا مِنَ الثَّوبِ العَتيقِ فيتَّسِعُ الخَرْقُ.</span>
<span className="v" id="v17" data-v="17" data-xref="يَشُوع 9:4; أيُّوب 32:19; المَزامِير 119:83"> وما مِنْ أحدٍ يَضَعُ خَمْرًا جَديدةً في أوعِيَةِ جِلْدٍ عَتيقَةٍ، لِئلَّا تَنشَقَّ الأوعِيَةُ فتَسيلَ الخمرُ وتَتْلَفَ الأوعِيَةُ. بل تُوضَعُ الخمرُ الجَديدةُ في أوعِيَةٍ جَديدَةٍ، فتَسلَمُ الخمرُ والأوعِيَةُ“.</span>
</div>
</div>

## إحياء صبية وشفاء اَمرأة لمست ثوب يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 8:41–56; مَرْقُس 5:22–43; ﻣﺘﻰ 8:2"> وبَينَما هوَ يَتكلَّمُ، دَنا مِنهُ رَئيسٌ يَهوديٌّ وسجَدَ لَه وقالَ: ”الآنَ ماتَتِ اَبنَتي. تَعالَ وضَعْ يدَكَ علَيها فتَحيا“.</span>
<span className="v" id="v19" data-v="19" data-xref="اعمال الرسل 10:38; غَلاطِيَّة 6:9–10; ﻣﺘﻰ 8:7"> فقامَ يَسوعُ وتَبِعَهُ معَ تلاميذِهِ.</span>
<span className="v" id="v20" data-v="20" data-xref="لُوقا 8:43–56; ﻣﺘﻰ 14:36; مَرْقُس 6:56"> وكانَت هُناكَ اَمرأةٌ مُصابةٌ بنَزْفِ الدَّمِ مِن اَثنتَي عَشْرَةَ سنَةً، فدَنَتْ مِنْ خَلفِ يَسوعَ ولَمَستْ طَرَفَ ثوبِهِ،</span>
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 5:26–33; لُوقا 8:45–47; اعمال الرسل 19:12"> لأنَّها قالَت في نَفسِها: ”يكفي أنْ ألمُسَ ثوبَهُ لأُشفى“.</span>
<span className="v" id="v22" data-v="22" data-xref="لُوقا 7:50; ﻣﺘﻰ 15:28; ﻣﺘﻰ 9:2"> فاَلتَفَت يَسوعُ فَرآها وقالَ: ”ثِقي يا اَبنَتي، إيمانُكِ شَفاكِ“. فشُفِيَتِ المَرأةُ مِنْ تِلكَ السَّاعَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="أخبارِ الأيّامِ الثّانِي 35:25; مَرْقُس 5:35–40; لُوقا 8:49–51"> ولمَّا وصَلَ يَسوعُ إلى بَيتِ الرئيسِ اليهوديِّ رأى الندَّابينَ والنَّاسَ في اَضطرابٍ،</span>
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 20:10; يُوحَنّا 11:11–13; ﻣﺘﻰ 27:39–43"> فقَالَ: ”إخرُجوا! ما ماتَتِ الصَّبيَّةُ، لكنَّها نائمةٌ!“ فضَحِكوا علَيهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="مَرْقُس 9:27; مَرْقُس 5:41; لُوقا 8:54"> وبَعدَما أخْرَجَ النَّاسَ، دخَلَ وأخَذَ بيَدِ الصَّبيَّةِ فقامَتْ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 4:24; ﻣﺘﻰ 14:1–2; مَرْقُس 1:45"> واَنتشرَ الخبَرُ في تِلكَ الأنحاءِ كُلِّها.</span>
</div>
</div>

## يسوع يشفي أعميين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="ﻣﺘﻰ 15:22; ﻣﺘﻰ 20:30–31; مَرْقُس 10:46–48"> وسارَ يَسوعُ مِنْ هُناكَ، فتَبِعَهُ أعمَيانِ يَصيحانِ: ”إِرْحَمْنا، يا اَبنَ داودَ!“</span>
<span className="v" id="v28" data-v="28" data-xref="مَرْقُس 9:23–24; ﻣﺘﻰ 13:58; ﻣﺘﻰ 8:2"> ولمَّا دخَلَ البَيتَ، دَنا مِنهُ الأعمَيانِ. فقَالَ لهُما يَسوعُ: ”أتؤمنانِ بأنِّي قادِرٌ على ذلِكَ؟“ فأجابا: ”نَعَم، يا سيِّدُ!“</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 15:28; مَرْقُس 10:52; ﻣﺘﻰ 8:13"> فلمَسَ يَسوعُ أعيُنَهُما وقالَ: ”فليكُنْ لكُما على قَدْرِ إيمانِكُما“.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 8:4; لُوقا 5:14; ﻣﺘﻰ 17:9"> فاَنفَتَحَتْ أعيُنُهُما. وأنذرَهُما يَسوعُ فقَالَ: ”إيَّاكُما أن يَعلَمَ أحدٌ!“</span>
<span className="v" id="v31" data-v="31" data-xref="مَرْقُس 7:36; ﻣﺘﻰ 9:26; مَرْقُس 1:44–45"> ولكنَّهُما خَرَجا ونَشرا الخبَرَ في تِلكَ الأنحاءِ كُلِّها.</span>
</div>
</div>

## يسوع يشفي أخرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 4:24; ﻣﺘﻰ 12:22–24; لُوقا 11:14"> ولمَّا خرَجَ الأعميانِ، جاءَهُ بَعضُ النَّاسِ بأخرسَ، فيهِ شَيطانٌ.</span>
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 35:6; مَرْقُس 7:32–37; الخُرُوج 4:11–12"> فلمَّا طَرَدَ يَسوعُ الشَّيطانَ، تكلَّمَ الأخرسُ. فتعَجَّبَ الجُموعُ وقالوا: ”ما رأينا مِثلَ هذا في إِسرائيلَ!“</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 11:15; مَرْقُس 3:22; ﻣﺘﻰ 12:23–24"> ولكنَّ الفَرِّيسيِّينَ قالوا: ”برئيسِ الشَّياطينِ يَطرُدُ الشَّياطينَ!“</span>
</div>
</div>

## يسوع يشفق على الناس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 4:23–24; اعمال الرسل 10:38; مَرْقُس 6:56"> وطافَ يَسوعُ في جميعِ المُدُنِ والقُرى يُعلِّمُ في المجامِعِ ويُعلِنُ بِشارةَ المَلكوتِ ويَشفي النَّاسَ مِنْ كُلِّ مَرَضٍ وَداءٍ.</span>
<span className="v" id="v36" data-v="36" data-xref="مَرْقُس 6:34; العَدَد 27:17; إرْمِيا 50:6"> ولمَّا رأى الجُموعَ اَمتلأ قَلبُهُ بالشَّفَقَةِ علَيهِم، لأنَّهُم كانوا بائِسينَ مُشتَّتينَ مِثلَ غنَمٍ لا راعِيَ لها.</span>
<span className="v" id="v37" data-v="37" data-xref="لُوقا 10:2; مَرْقُس 16:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:9"> فقَالَ لِتَلاميذِهِ: ”الحَصادُ كثيرٌ، ولكنَّ العُمَّالَ قَليلونَ.</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 10:1–2; إرْمِيا 3:15; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:1"> فاَطلُبوا مِنْ رَبِّ الحَصادِ أنْ يُرسِلَ عُمَّالًا إلى حَصادِهِ“.</span>
</div>
