<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 9:1–6; مَرْقُس 3:13–15; مَرْقُس 6:7–13"> ودَعا يَسوعُ تلاميذَهُ الاثنيَ عشَرَ وأعْطاهُم سُلطانًا يَطرُدونَ بِه الأرواحَ النَّجسَةَ ويَشْفونَ النَّاسَ مِنْ كُلِّ داءٍ ومرَضٍ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 4:21; اعمال الرسل 1:13; لُوقا 5:10"> وهذِهِ أسماءُ الرُّسُلِ الاثني عشَرَ: أوَّلُهُم سِمْعانُ المُلَقَّبُ بِبُطرُسَ وأخوهُ أندَراوُسُ، ويَعقوبُ بنُ زَبدي وأخوهُ يوحنَّا،</span>
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 3:18; اعمال الرسل 1:13; مَرْقُس 2:14"> وفيلُبُّسُ وبَرْثولوماوُسُ، وتوما ومتَّى جابي الضَّرائبِ، ويَعقوبُ بنُ حَلْفَى وتَدَّاوسُ،</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 26:14; ﻣﺘﻰ 26:47; يُوحَنّا 13:2"> وسِمْعانُ الوطنيُّ الغَيورُ، ويَهوذا الإسخَرْيوطيُّ الّذي أسلَمَ يَسوعَ.</span>
</div>

## يسوع يرسل الاثني عشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 4:9; اعمال الرسل 22:21–23; المُلُوكِ الثّانِي 17:24–41"> وأرسَلَ يَسوعُ هؤُلاءِ التَّلاميذَ الاثنَي عشَرَ وأوْصاهُم قالَ: ”لا تَقصِدوا أرضًا وثَنِيَّةً ولا تَدْخُلوا مدينةً سامِريَّةً،</span>
<span className="v" id="v6" data-v="6" data-xref="إرْمِيا 50:6; حِزْقِيال 34:6; إشَعْياء 53:6"> بَلِ اَذْهَبوا إلى الخِرافِ الضَّالَّةِ مِنْ بَني إِسرائيلَ،</span>
<span className="v" id="v7" data-v="7" data-xref="اعمال الرسل 28:31; ﻣﺘﻰ 3:2; لُوقا 9:60"> وبَشِّروا في الطَّريقِ بأنَّ مَلكوتَ السَّماواتِ اَقتَرَبَ.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 3:6; لُوقا 10:9; اعمال الرسل 4:30"> واَشفوا المَرضى، وأقيموا الموتَى، وطَهِّروا البُرْصَ، واَطرُدوا الشَّياطينَ. مجَّانًا أخَذتُمْ، فمَجَّانًا أعْطُوا.</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 22:35; لُوقا 9:3–5; لُوقا 10:4–12"> لا تَحمِلوا نُقودًا مِنْ ذَهَبٍ ولا مِنْ فِضَّةٍ ولا مِنْ نُحاسٍ في جُيوبِكُم،</span>
<span className="v" id="v10" data-v="10" data-xref="لُوقا 10:7–12; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:17–18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:4–14"> ولا كِيسًا لِلطَّريقِ ولا ثوبًا آخَرَ ولا حِذاءً ولا عصًا، لأنَّ العامِلَ يَسْتَحِقُّ طعامَهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 10:7–8; أيُّوب 31:32; القُضاة 19:16–21"> وأيَّةَ مدينةٍ أو قريةٍ دَخَلْتُم، فاَستَخبِروا عَنِ المُستحِقِّ فيها، وأقيموا عِندَهُ إلى أنْ تَرحَلوا.</span>
<span className="v" id="v12" data-v="12" data-xref="صَمُوئيلَ الأوَّلُ 25:6; لُوقا 10:5–6; اعمال الرسل 10:36"> وإذا دَخلْتُم بَيتًا فسَلِّموا علَيهِ.</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 10:6; المَزامِير 35:13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:16"> فإنْ كانَ أهلًا للسَّلامِ، حلَّ سلامُكُم بِه، وإلَّا رجَعَ سَلامُكُم إلَيكُم.</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 10:10–11; اعمال الرسل 18:6; نَحَمْيا 5:13"> وإذا اَمتَنَعَ بَيتٌ أو مدينةٌ عَنْ قَبولِكُم أو سَماعِ كلامِكُم، فاَتركوا المكانَ واَنْفُضوا الغُبارَ عَنْ أقدامِكُم.</span>
<span className="v" id="v15" data-v="15" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:6; رسالَةُ يُوحَنّا الأُولَى 4:17; يُوحَنّا 15:22–24"> الحقَّ أقولُ لكُم: سيكونُ مَصيرُ سَدومَ وعَمورَةَ يومَ الحِسابِ أكثرَ اَحتِمالًا مِنْ مَصيرِ تِلكَ المدينةِ.</span>
</div>
</div>

## زمن الاضطهاد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="لُوقا 10:3; فِيلِبِّي 2:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:20"> ”ها أنا أُرسِلُكُم مِثلَ الخِرافِ بَينَ الذِّئابِ فكونوا حَذِرينَ كالحيَّاتِ، وُدَعاءَ كالحَمام.</span>
<span className="v" id="v17" data-v="17" data-xref="مَرْقُس 13:9; ﻣﺘﻰ 23:34; اعمال الرسل 26:11"> واَنتَبِهوا، لأنَّ النَّاسَ سَيُسلِّمونَكم إلى المَحاكِمِ، ويَجلِدونَكُم في المجامِعِ،</span>
<span className="v" id="v18" data-v="18" data-xref="اعمال الرسل 5:25–27; ﻣﺘﻰ 8:4; المَزامِير 2:1–6"> ويَسوقونَكم إلى الحُكّامِ والمُلوكِ مِنْ أجلي، لتَشْهَدوا عِندَهُم وعِندَ سائِرِ الشُّعوبِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 4:12; مَرْقُس 13:11–13; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:17"> فلا تَهتَمُّوا حِينَ يُسْلِمونَكُم كيفَ أو بِماذا تَتكلَّمونَ، لأنَّكُم سَتُعطَوْنَ في حينِهِ ما تَتكلَّمونَ بِه.</span>
<span className="v" id="v20" data-v="20" data-xref="لُوقا 21:15; اعمال الرسل 6:10; لُوقا 12:12"> فما أنتُمُ المُتكَلِّمونَ، بَلْ رُوحُ أبيكمُ السَّماويِّ يَتكَلَّمُ فيكُم.</span>
<span className="v" id="v21" data-v="21" data-xref="لُوقا 21:16–17; ميخا 7:5–6; مَرْقُس 13:12–13"> سيُسلِمُ الأخُ أخاهُ إلى الموتِ، والأبُ اَبنَهُ، ويتَمَرَّدُ الأبناءُ على الآباءِ ويَقتُلونَهُم،</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 24:13; يَعقُوب 1:12; رُؤيا يُوحَنّا 2:10"> ويُبْغِضُكُم جَميعُ النَّاسِ مِنْ أجلِ اَسْمي. والّذي يَثبُتُ إلى النِّهايةِ يَخلُصُ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 16:28; اعمال الرسل 8:1; مَرْقُس 13:26"> وإذا اَضطَهَدوكُم في مدينةٍ، فاَهرُبوا إلى غَيرِها. الحقَّ أقولُ لكُم: لن تُنْهوا عَمَلكُم في مُدُنِ إِسرائيلَ كُلِّها حتّى يَجيءَ اَبنُ الإنسانِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 15:20; لُوقا 6:40; يُوحَنّا 13:16"> لا تِلميذَ أعظَمُ مِنْ مُعلِّمِهِ، ولا خادِمَ أعظَمُ مِنْ سيِّدِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="لُوقا 11:15; ﻣﺘﻰ 12:24; ﻣﺘﻰ 9:34"> يكفي التِّلميذَ أنْ يكونَ مِثلَ مُعلِّمِهِ والخادِمَ مِثلَ سيِّدِهِ. إذا كان ربُّ البَيتِ قيلَ لَه بَعلَزَبولُ، فكَيفَ أهلُ بيتِهِ؟</span>
</div>
</div>

## مخافة الله

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="لُوقا 8:17; مَرْقُس 4:22; إرْمِيا 1:8"> ”لا تَخافوهُم. فما مِنْ مَستورٍ إلَّا سَيَنكَشِفُ، ولا مِنْ خفِيٍّ إلَّا سَيُظهَرُ.</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 5:20; يُوحَنّا 16:13; يُوحَنّا 16:1"> وما أقولُهُ لكُم في الظَّلامِ، قولوهُ في النُّورِ. وما تَسمَعونَهُ هَمسًا، نادوا بِه على السُّطوحِ.</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 12:4–5; إشَعْياء 8:12–13; رُؤيا يُوحَنّا 2:10"> لا تخافوا الّذينَ يَقتُلونَ الجسَدَ ولا يَقدرونَ أنْ يقتُلوا النَّفسَ، بَل خافوا الّذي يَقدِرُ أنْ يُهلِكَ الجسَدَ والنَّفسَ معًا في جَهَنَّم.</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 12:6–7; المَزامِير 104:27–30"> أما يُباعُ عُصْفورانِ بدِرْهَمٍ واحدٍ؟ ومعَ ذلِكَ لا يقَعُ واحدٌ منهُما إلى الأرضِ إلَّا بِعِلمِ أبيكُمُ السَّماويِّ.</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 12:7; صَمُوئيلَ الأوَّلُ 14:45; اعمال الرسل 27:34"> أمَّا أنتُم، فشَعرُ رُؤُوسِكُم نفسُهُ معدودٌ كُلُّهُ.</span>
<span className="v" id="v31" data-v="31" data-xref="لُوقا 12:24; المَزامِير 8:5; ﻣﺘﻰ 6:26"> لا تخافوا، أنتُم أفضَلُ مِنْ عصافيرَ كثيرةٍ.</span>
</div>
</div>

<div className="bp">

<div className="bp">

<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 12:25; مَرْقُس 8:35–36; لُوقا 17:33"> مَنْ حَفِظَ حياتَهُ يَخسَرُها، ومَنْ خَسِرَ حياتَهُ مِنْ أجلي يَحفَظُها.</span>

## الجزاء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="يُوحَنّا 13:20; لُوقا 9:48; لُوقا 10:16"> ”مَنْ قَبِلَكُم قَبِلَني، ومَنْ قَبِلَني قَبِلَ الّذي أرسَلَني.</span>
<span className="v" id="v41" data-v="41" data-xref="رسالَةُ يُوحَنّا الثّالثَة 1:5–8; العِبرانِيّين 6:10; المُلُوكِ الثّانِي 4:16–17"> مَنْ قَبِلَ نَبيًّا لأنَّهُ نَبيٌّ، فَجَزاءَ نَبيٍّ يَنالُ. ومَنْ قَبِلَ رَجُلًا بارًّا لأنَّهُ رجُلٌ بارٌّ، فجَزاءَ رَجُلٍ بارٍّ يَنالُ.</span>
<span className="v" id="v42" data-v="42" data-xref="العِبرانِيّين 6:10; ﻣﺘﻰ 25:40; مَرْقُس 9:41–42"> ومَنْ سَقى أحدَ هؤُلاءِ الصِّغارِ ولَو كأسَ ماءٍ بارِدٍ لأنَّهُ تِلميذي، فأجرُهُ، الحقَّ أقولُ لكُم، لن يَضيعَ“.</span>
