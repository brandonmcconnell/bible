## الزواج والطلاق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يُوحَنّا 10:40; ﻣﺘﻰ 19:1–9; ﻣﺘﻰ 7:28"> ولمَّا أتَمَّ يَسوعُ هذا الكلامَ، تَرَكَ الجَليلَ وجاءَ إلى بِلادِ اليَهودِيَّةِ مِنْ عَبرِ الأُردُنِّ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 12:15; مَرْقُس 6:55–56; ﻣﺘﻰ 15:30–31"> فتبِعَتْهُ جُموعٌ كبيرةٌ، فشَفاهُم هُناكَ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 10:2; العِبرانِيّين 3:9; ﻣﺘﻰ 16:1"> ودَنا إلَيهِ بَعضُ الفَرِّيسيِّينَ وسألوهُ لِيُحرِجوهُ: ”أيَحِلُّ لِلرَّجُلِ أنْ يُطلِّقَ اَمرأتَهُ لأيِّ سَبَبٍ كانَ؟“</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 2:23; التَّكوين 1:27; التَّكوين 5:2"> فأجابَهُم: ”أما قَرأْتُم أنَّ الخالِقَ مِنَ البَدءِ جعَلَهُما ذكَرًا وأُنثى</span>
<span className="v" id="v5" data-v="5" data-xref="التَّكوين 2:21–24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:2"> وقالَ: لذلِكَ يَترُكُ الرَّجُلُ أباهُ وأُمَّهُ ويَتَّحِدُ باَمرأتَهِ، فيَصيرُ الاثنانِ جسَدًا واحدًا؟</span>
<span className="v" id="v6" data-v="6" data-xref="العِبرانِيّين 13:4; مَرْقُس 10:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:10–14"> فلا يكونانِ اثنينِ، بل جسَدٌ واحدٌ. وما جمَعَهُ اللهُ لا يُفرِّقُهُ الإنسانُ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 24:1–4; ﻣﺘﻰ 5:31; مَرْقُس 10:4"> وسألَه الفَرِّيسيُّونَ: ”فلِماذا أوصى موسى بأنْ يُعطيَ الرَّجُلُ اَمرأتَهُ كِتابَ طَلاقٍ فتُطلَّقُ؟“</span>
<span className="v" id="v8" data-v="8" data-xref="مَلاخِي 2:13–14; التَّكوين 2:24; مَرْقُس 10:5"> فأجابَهُم يَسوعُ: ”لِقساوَةِ قُلوبِكُم أجازَ لكُم موسى أنْ تُطلِّقوا نِساءَكُم. وما كانَ الأمرُ مِنَ البَدءِ هكذا.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 5:32; لُوقا 16:18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:39"> أمَّا أنا فأقولُ لكُم: مَنْ طلَّقَ اَمرأتَهُ إلَّا في حالَةِ الزِّنى وتزَوَّجَ غَيرَها زنى“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:32–35; الأمثال 19:13–14; الأمثال 21:19"> فقَالَ لَه تلاميذُهُ: ”إذا كانَت هذِهِ حالُ الرَّجُلِ معَ المرأةِ، فخَيرٌ لَه أنْ لا يتَزوَّجَ“.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:35; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:7–9"> فأجابَهُم يَسوعُ: ”لا يَقبلُ هذا الكلامَ إلَّا الّذينَ أُعطِيَ لهُم أن يَقبَلوهُ.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:32–38; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:5; إشَعْياء 56:3–4"> ففي النَّاسِ مَنْ ولَدَتْهُم أُمَّهاتُهُم عاجِزينَ عَنِ الزَّواجِ، وفيهِم مَنْ جَعلَهُمُ النَّاسُ هكذا، وفيهِم مَنْ لا يَتزَوَّجونَ مِنْ أجلِ مَلكوتِ السَّماواتِ. فمَنْ قدِرَ أنْ يَفهَمَ فليَفْهَم“.</span>
</div>
</div>

## يسوع يبارك الأطفال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 10:13–16; لُوقا 18:15–17; ﻣﺘﻰ 18:2–5"> وجاءَهُ بَعضُ النَّاسِ بأطفالٍ ليَضَعَ يَدَيْهِ علَيهِم ويُصلِّيَ، فاَنتهَرَهُمُ التَّلاميذُ.</span>
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 18:3; لُوقا 18:16–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 14:20"> فقَالَ يَسوعُ: ”دَعُوا الأطفالَ يأْتُونَ إليَّ ولا تَمنَعوهُم، لأنَّ لأمثالِ هؤلاءِ مَلكوتَ السَّماواتِ“.</span>
<span className="v" id="v15" data-v="15" data-xref="مَرْقُس 10:16; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:14"> ووضَعَ يَدَيْهِ علَيْهِم ومَضى مِنْ هُناكَ.</span>
</div>
</div>

## الشاب الغني

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="مَرْقُس 10:17–30; يُوحَنّا 3:15; يَهُوذا 1:21"> وأقبَلَ إليهِ شابٌ وقالَ لَه: ”أيُّها المُعَلِّمُ، ماذا أعمَلُ مِنَ الصَّلاحِ لأنالَ الحَياةَ الأبدِيَّةَ؟“ فأجابَهُ يَسوعُ: ”</span>
<span className="v" id="v17" data-v="17" data-xref="اللّاويِّين 18:5; لُوقا 10:26–28; غَلاطِيَّة 3:11–13"> لِماذا تَسألُني عمَّا هوَ بارٌّ؟ لا بارَّ إلَّا واحدٌ. إذا أَرَدْتَ أنْ تَدخُلَ الحياةَ فاَعمَلْ بالوصايا“.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّثْنِيَة 5:16–21; الخُرُوج 20:12–17; مَرْقُس 10:19"> فقَالَ لَه: ”أيَّ وصايا؟“ فقَالَ يَسوعُ: ”لا تَقتُلْ، لا تَزْنِ، لا تَسرِقْ، لا تَشهَدْ بالزُّورِ،</span>
<span className="v" id="v19" data-v="19" data-xref="اللّاويِّين 19:18; الخُرُوج 20:12; ﻣﺘﻰ 22:39"> أكرِمْ أباكَ وأُمَّكَ، أحِبَّ قريبَكَ مِثلما تُحبُّ نَفسَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="غَلاطِيَّة 3:24; مَرْقُس 10:20–21; فِيلِبِّي 3:6"> فقَالَ لَه الشَّابُّ: ”عَمِلتُ بِهذِهِ الوصايا كُلِّها، فما يَعوزُني؟“</span>
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 10:21; لُوقا 18:22; لُوقا 12:33"> أجابَهُ يَسوعُ: ”إذا أردتَ أنْ تكونَ كامِلًا، فاَذهَبْ وبِعْ ما تملِكُهُ ووَزِّعْ ثمَنَهُ على الفُقراءِ، فيكونَ لكَ كنزٌ في السَّماواتِ، وتعالَ اَتْبَعْني!“</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 6:24; حِزْقِيال 33:31; ﻣﺘﻰ 13:22"> فلمَّا سَمِعَ الشَّابُّ هذا الكلامَ مَضى حَزينًا لأنَّهُ كانَ يملِكُ أموالًا كثيرةً.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9–10; ﻣﺘﻰ 13:22; مَرْقُس 10:23"> وقالَ يَسوعُ لِتلاميذِهِ: ”الحقَّ أقولُ لكُم: يَصعُبُ على الغنيِّ أنْ يَدخُلَ مَلكوتَ السَّماواتِ.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 5:44; لُوقا 18:25; مَرْقُس 10:24–25"> بل أقولُ لكُم: مُرورُ الجمَلِ في ثَقبِ الإبرةِ أسهَلُ مِنْ دُخولِ الغنيِّ مَلكوتَ اللهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="ﺭﻭﻣﻴﺔ 10:13; ﻣﺘﻰ 24:22; ﺭﻭﻣﻴﺔ 11:5–7"> فتعجَّبَ التَّلاميذُ كثيرًا مِنْ هذا الكلامِ وقالوا: ”مَنْ يُمكِنُهُ أنْ يَخلُصَ، إذًا؟“</span>
<span className="v" id="v26" data-v="26" data-xref="إرْمِيا 32:27; لُوقا 1:37; أيُّوب 42:2"> فنَظَرَ إلَيهِم يَسوعُ وقالَ لهُم: ”هذا شيءٌ غيرُ مُمكنٍ عِندَ النَّاسِ، أمَّا عِندَ اللهِ فكُلُّ شيءٍ مُمكِنٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="فِيلِبِّي 3:8; لُوقا 18:28; لُوقا 5:11"> وقالَ لَه بُطرُسُ: ”ها نَحنُ تَركْنا كُلَّ شيءٍ وتَبِعْناكَ، فماذا يكونُ نَصيبُنا؟“</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="لُوقا 22:28–30; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:2–3; ﻣﺘﻰ 25:31"> فأجابَ يَسوعُ: ”الحقَّ أقولُ لكُم: متى جلَسَ اَبنُ الإنسانِ على عَرشِ مَجدِهِ عِندَ تَجديدِ كُلِّ شيءٍ، تَجلِسونَ أنتُم الّذينَ تَبِعوني على اَثني عشَرَ عَرشًا لتَدينوا عَشائرَ إِسرائيلَ الاثني عشَرَ.</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 18:29–30; مَرْقُس 10:29–30; فِيلِبِّي 3:8"> وكُلُّ مَنْ ترَكَ بيوتًا، أو إخوَةً أو أخواتٍ، أو أبًا، أو أُمًّا، أو أبناءً، أو حُقولًا مِنْ أجلِ اَسمي، يَنالُ مِئةَ ضِعفٍ ويَرِثُ الحياةَ الأبديَّةَ.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 20:16; لُوقا 13:30; مَرْقُس 10:31"> وكثيرٌ مِنَ الأوَّلينَ يَصيرونَ آخِرينَ، ومِنَ الآخِرينَ يَصيرونَ أوَّلينَ.</span>
</div>
