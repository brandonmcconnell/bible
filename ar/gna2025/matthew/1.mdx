## نسب يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 22:18; ﺭﻭﻣﻴﺔ 1:3; غَلاطِيَّة 3:16"> هذا نسَبُ يسوعَ المسيحِ اَبنِ داودَ اَبنِ إبراهيمَ:</span>
<span className="v" id="v2" data-v="2" data-xref="اعمال الرسل 7:8; العِبرانِيّين 11:17–18; التَّكوين 46:8–27"> إبراهيمُ ولَدَ إسحق. وإسحق وَلَدَ يَعقوبَ. ويَعقوبُ ولَدَ يَهوذا وإخوتَه.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 46:12; التَّكوين 38:29–30; أخبارِ الأيّامِ الأوّلُ 4:1"> ويَهوذا ولَدَ فارِصَ وزارَحَ مِنْ ثامارَ. وفارِصُ ولَدَ حَصْرونَ. وحَصْرونُ ولَدَ أرامَ.</span>
<span className="v" id="v4" data-v="4" data-xref="أخبارِ الأيّامِ الأوّلُ 2:10–12; العَدَد 7:12; العَدَد 2:3"> وأرامُ ولَدَ عَمِّينادابَ. وعَمِّينادابُ ولَدَ نَحْشونَ. ونَحْشونُ ولَدَ سَلَمونَ.</span>
<span className="v" id="v5" data-v="5" data-xref="العِبرانِيّين 11:31; راعُوث 1:16–17; يَعقُوب 2:25"> وسَلَمونُ ولَدَ بُوعَزَ مِنْ راحابَ. وبُوعَزُ ولَدَ عُوبيدَ مِنْ راعُوثَ. وعُوبيدُ ولَدَ يَسّى.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 15:5; صَمُوئيلَ الأوَّلُ 17:12; صموئيلَ الثّانِي 12:24–25"> ويَسّى ولَدَ داودَ المَلِكَ.<br />وداودُ ولَدَ سُلَيْمانَ مِن اَمرأةِ أوريَّا.</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الثّانِي 14:1–15; المُلُوكِ الأوَّلُ 14:31; المُلُوكِ الأوَّلُ 11:43–24"> وسُلَيْمانُ ولَدَ رَحْبَعامَ. ورَحْبَعامُ ولَدَ أبيّا. وأبيّا ولَدَ آسا.</span>
<span className="v" id="v8" data-v="8" data-xref="أخبارِ الأيّامِ الأوّلُ 3:11; أخبارِ الأيّامِ الثّانِي 26:1–23; المُلُوكِ الثّانِي 8:16"> وآسا ولَدَ يوشافاطَ. ويوشافاطُ ولَدَ يُورامَ. ويُورامُ ولَدَ عُزِّيّا.</span>
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 7:1–13; أخبارِ الأيّامِ الثّانِي 26:21; أخبارِ الأيّامِ الثّانِي 27:1–32"> وعُزِّيّا ولَدَ يُوثامَ. ويُوثامُ ولَدَ أحازَ. وأحازُ ولَدَ حَزْقِيّا.</span>
<span className="v" id="v10" data-v="10" data-xref="أخبارِ الأيّامِ الثّانِي 32:33–27; المُلُوكِ الأوَّلُ 13:2; أخبارِ الأيّامِ الأوّلُ 3:13–15"> وحَزْقِيّا ولَدَ مَنَسّى. ومَنَسّى ولَدَ آمونَ. وآمونُ ولَدَ يوشِيّا.</span>
<span className="v" id="v11" data-v="11" data-xref="إرْمِيا 27:20; إرْمِيا 52:28–30; دانيال 1:2"> ويوشِيّا ولَدَ يَكُنيّا وإخوَتَه زَمَنَ السبْيِ إلى بابِلَ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الأوّلُ 3:17; عَزْرا 3:2; لُوقا 3:27"> وبَعْدَ السّبْي إلى بابِلَ يَكُنِيّا ولَدَ شَأَلْتَئيلَ. وشَأَلْتَئيلُ ولَدَ زَرُبابِلَ.</span>
<span className="v" id="v13" data-v="13"> وزَرُبابِلُ ولَدَ أبيهُودَ. وأبيهُودُ ولَدَ ألِياقيمَ. وألِياقيمُ ولَدَ عازُورَ.</span>
<span className="v" id="v14" data-v="14"> وَعازُورُ ولَدَ صادُوقَ. وصادُوقُ ولَدَ أَخيمَ. وأخيمُ ولَدَ أليُودَ.</span>
<span className="v" id="v15" data-v="15"> وأليُودُ ولَدَ أليعازَرَ. وأليعازَرُ ولَدَ مَتّانَ. ومَتّانُ ولَدَ يَعقوبَ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 27:17; يُوحَنّا 4:25; لُوقا 3:23"> ويَعقوبُ ولَدَ يوسفَ رَجُلَ مَرْيمَ الّتي ولَدَتْ يَسوعَ الّذي يُدعى المَسيحَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 1:11–12; المُلُوكِ الثّانِي 24:14; ﻣﺘﻰ 11:2"> فمَجْموعُ الأجْيالِ مِنْ إبراهيمَ إلى داودَ أرْبَعَةَ عَشَرَ جيلًا. ومِنْ داودَ إلى سَبْيِ بابِلَ أرْبَعَةَ عَشَرَ جيلًا. ومِنْ سَبْيِ بابِلَ إلى المَسيحِ أرْبَعَةَ عَشَرَ جيلًا.</span>
</div>
</div>

## ميلاد يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="غَلاطِيَّة 4:4–5; لُوقا 1:27–38; أيُّوب 14:4"> وهذِهِ سيرَةُ ميلادِ يَسوعَ المَسيحِ: كانَت أُمُّهُ مَريَمُ مَخْطوبَةً ليوسفَ، فَتبيَّنَ قَبْلَ أنْ تَسْكُنَ مَعَهُ أنَّها حُبْلى مِنَ الرُّوحِ القُدُسِ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 22:21–24; التَّثْنِيَة 24:1–4; يُوحَنّا 8:4–5"> وكانَ يوسفُ رَجُلًا بارًّا فَما أرادَ أنْ يكْشِفَ أمْرَها، فَعزَمَ على أنْ يَترُكَها سِرًّا.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="الأمثال 3:5–6; لُوقا 1:19; ﻣﺘﻰ 1:18"> وبَينَما هوَ يُفَكِّرُ في هذا الأمْرِ، ظَهَرَ لَه مَلاكُ الرَّبِّ في الحُلُمِ وقالَ لَه: ”يا يوسفُ اَبنَ داودَ، لا تخَفْ أنْ تأخُذَ مَرْيمَ اَمرأةً لكَ. فَهيَ حُبْلى مِنَ الروحِ القُدُسِ،</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 13:23; اعمال الرسل 4:12; يُوحَنّا 1:29"> وسَتَلِدُ اَبنًا تُسمّيهِ يَسوعَ، لأنَّهُ يُخَلِّصُ شعْبَهُ مِنْ خَطاياهُمْ“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="لُوقا 24:44; ﻣﺘﻰ 2:23; المُلُوكِ الأوَّلُ 8:15"> حَدَثَ هذا كُلُّه لِيَتِمَّ ما قالَ الرَّبُّ بلِسانِ النَّبيِّ:</span>
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 7:14; إشَعْياء 9:6–7; المَزامِير 46:11"> ”سَتحْبَلُ العَذْراءُ، فتَلِدُ اَبْنًا يُدْعى ”عِمّانوئيلَ“، أي اللهُ مَعَنا.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 15:14; يَعقُوب 2:21–26; يُوحَنّا 2:5–8"> فلمَّا قامَ يوسفُ مِنَ النَّومِ، عَمِلَ بِما أمَرَهُ مَلاكُ الرَّبِّ. فَجاءَ باَمْرَأتِهِ إلى بَيتِه،</span>
<span className="v" id="v25" data-v="25" data-xref="لُوقا 2:21; الخُرُوج 13:2; ﻣﺘﻰ 1:21"> ولكِنَّهُ ما عَرَفَها حتّى ولَدَتِ اَبْنَها فَسَمّاهُ يَسوعَ.</span>
</div>
</div>
