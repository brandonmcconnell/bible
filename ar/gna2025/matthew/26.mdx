## محاولة قتل يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 7:28; ﻣﺘﻰ 19:1"> ولمَّا أتمَّ يَسوعُ هذا الكلامَ كُلَّهُ، قالَ لِتلاميذِهِ:</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 22:1–2; مَرْقُس 14:1–2; يُوحَنّا 13:1–2"> ”تَعرِفون أنَّ الفِصحَ يَقَعُ بَعدَ يَومينِ، وفيهِ يُسلَّمُ اَبنُ الإنسانِ ليُصلَبَ“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="يُوحَنّا 11:47–53; يُوحَنّا 18:24; مَرْقُس 14:54"> واَجتَمَعَ في ذلِكَ الحينِ رُؤساءُ الكهَنةِ وشُيوخُ الشَّعبِ في دارِ قيافا رَئيسِ الكَهنَةِ،</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 12:14; المَزامِير 2:2; اعمال الرسل 13:10"> وتَشاوروا ليُمسِكوا يَسوعَ بحيلَةٍ ويقتُلوهُ.</span>

</div>
</div>

## امرأة تسكب الطيب على يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 14:3–9; لُوقا 7:37–39; يُوحَنّا 12:1–8"> وبَينَما يَسوعُ في بَيتَ عَنْيا عِندَ سِمْعانَ الأبرصِ،</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 12:2–3; لُوقا 7:46; لُوقا 7:37–38"> دَنَتْ مِنهُ اَمرأةٌ تَحمِلُ قارورةَ طِيبٍ غالي الثَّمَنِ، فسكَبَتهُ على رأسِهِ وهوَ يَتناولُ الطعامَ.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 12:4–6; مَرْقُس 14:4; الخُرُوج 5:17"> فلمَّا رأى التَّلاميذُ ما عمِلَتْ، اَستاؤُوا وقالوا: ”ما هذا الإسرافُ؟</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 12:5–6; مَرْقُس 14:5; يَشُوع 7:20–21"> كانَ يُمكِنُ أنْ يُباعَ غالِيًا ويُوزَّعَ ثَمنُهُ على الفُقَراءِ!“</span>
<span className="v" id="v10" data-v="10" data-xref="تِيطُس 3:8; العِبرانِيّين 13:21; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:10"> فعَرفَ يَسوعُ وقالَ لهُم: ”لِماذا تُزعِجونَ هذِهِ المَرأةَ؟ فهيَ عَمِلَتْ لي عَملًا صالِحًا.</span>
<span className="v" id="v11" data-v="11" data-xref="التَّثْنِيَة 15:11; مَرْقُس 14:7; يُوحَنّا 12:8"> فالفُقراءُ عِندَكُم في كُلِّ حينٍ، وأمَّا أنا فلا أكونُ في كُلِّ حينٍ عِندَكُم.</span>
<span className="v" id="v12" data-v="12" data-xref="مَرْقُس 14:8; يُوحَنّا 12:7; مَرْقُس 16:1"> وإذا كانَت سكبَتْ هذا الطِّيبَ على جَسدي، فَلِتُهَيِّئَهُ للدَفْنِ.</span>
<span className="v" id="v13" data-v="13" data-xref="مَرْقُس 14:9; ﻣﺘﻰ 24:14; المَزامِير 112:6"> الحقَّ أقولُ لكُم: أينَما تُعلَنُ هذِهِ البِشارةُ في العالَمِ كُلِّهِ، يُحدَّثُ أيضًا بعمَلِها هذا، إحياءً لِذكرِها“.</span>
</div>
</div>

## خيانة يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="لُوقا 22:3–6; ﻣﺘﻰ 10:4; مَرْقُس 14:10–11"> وفي ذلِكَ الوقتِ ذَهبَ أحدُ التَّلاميذِ الاثنَي عشَرَ، وهوَ يهوذا الملقَّبُ بالإسْخَريوطِيِّ، إلى رُؤساءِ الكهَنَةِ</span>
<span className="v" id="v15" data-v="15" data-xref="زَكَريّا 11:12–13; التَّكوين 37:26–28; الخُرُوج 21:32"> وقالَ لهُم: ”ماذا تُعطوني لأُسَلِّمَ إلَيكُم يَسوعَ؟“ فوَعدوهُ بثلاثينَ مِنَ الفِضَّةِ.</span>
<span className="v" id="v16" data-v="16" data-xref="مَرْقُس 14:11; لُوقا 22:6"> وأخَذَ يَهوذا مِنْ تِلكَ السَّاعةِ يترَقَّبُ الفُرصةَ ليُسَلِّمَ يَسوعَ.</span>
</div>
</div>

## عشاء الفصح مع التَّلاميذ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="لُوقا 22:7–13; مَرْقُس 14:12–16; الخُرُوج 12:18–20"> وفي أوّلِ يومٍ من عِيدِ الفَطير، جاءَ التَّلاميذُ إلى يَسوعَ وقالوا لَه: ”أينَ تُريدُ أنْ نُهيِّئَ لكَ عشاءَ الفِصحِ؟“</span>
<span className="v" id="v18" data-v="18" data-xref="مَرْقُس 14:13–16; يُوحَنّا 7:6; يُوحَنّا 17:1"> فأجابَهُم: ”إذهَبوا إلى فُلانٍ في المدينةِ وقولوا لَه: يقولُ المُعلِّمُ: جاءَتْ ساعَتي، وسأتناوَلُ عَشاءَ الفِصحِ في بَيتِكَ معَ تلاميذي“.</span>
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 2:5; يُوحَنّا 15:14; ﻣﺘﻰ 21:6"> فعَمِلَ التّلاميذُ ما أمرَهُم بِه يَسوعُ وهيَّأوا عَشاءَ الفِصحِ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 14:17–21; يُوحَنّا 13:21; لُوقا 22:14–16"> وفي المَساءِ، جلَسَ يَسوعُ لِلطَّعامِ معَ تلاميذِهِ الاثني عشَرَ.</span>
<span className="v" id="v21" data-v="21" data-xref="يُوحَنّا 13:21; العِبرانِيّين 4:13; يُوحَنّا 6:70–71"> وبَينَما هُمْ يأكُلونَ، قالَ يَسوعُ: ”الحقَّ أقولُ لكُم: واحدٌ مِنكُم سيُسلِّمُني“.</span>
<span className="v" id="v22" data-v="22" data-xref="يُوحَنّا 13:22–25; لُوقا 22:23; مَرْقُس 14:19–20"> فحَزِنَ التَّلاميذُ كثيرًا وأخَذوا يسألونَهُ، واحدًا واحدًا: ”هل أنا هوَ، يا ربُّ؟“</span>
<span className="v" id="v23" data-v="23" data-xref="يُوحَنّا 13:18; المَزامِير 41:9; يُوحَنّا 13:26–28"> فأجابَهُم: ”مَنْ يَغمِسُ خُبزَهُ في الصَّحنِ معي هوَ الّذي سَيُسلِّمُني.</span>
<span className="v" id="v24" data-v="24" data-xref="اعمال الرسل 26:22–23; لُوقا 24:46; ﻣﺘﻰ 18:7"> فاَبنُ الإنسانِ سيَموتُ كما جاءَ عَنهُ في الكِتابِ، ولكِنَّ الويلَ لمن يُسلِّمُ اَبنَ الإنسانِ! كانَ خيرًا لَه أنْ لا يُولدَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 26:64; ﻣﺘﻰ 27:11; لُوقا 22:70"> فسألَهُ يَهوذا الّذي سيُسلِّمُهُ: ”هل أنا هوَ، يا مُعَلِّمُ؟“ فأجابَهُ يَسوعُ: ”أنتَ قُلتَ“.</span>
</div>
</div>

## عشاء الرّبّ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:23–29; لُوقا 22:18–20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:16–17"> وبَينَما هُم يأكُلونَ، أخذَ يَسوعُ خُبزًا وبارَكَ وكَسَرَهُ وناوَلَ تلاميذَهُ وقالَ: ”خُذوا كُلوا، هذا هوَ جَسَدي“.</span>
<span className="v" id="v27" data-v="27" data-xref="إشَعْياء 25:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:16; المَزامِير 116:13"> وأخَذَ كأسًا وشكَرَ وناوَلَهُم وقالَ: ”إشرَبوا مِنها كُلُّكُم.</span>
<span className="v" id="v28" data-v="28" data-xref="العِبرانِيّين 10:4–14; العِبرانِيّين 9:14–22; اللّاويِّين 17:11"> هذا هوَ دَمي، دمُ العَهدِ الّذي يُسفَكُ مِنْ أجلِ أُناسٍ كثيرينَ. لِغُفرانِ الخطايا.</span>
<span className="v" id="v29" data-v="29" data-xref="العِبرانِيّين 12:2; رُؤيا يُوحَنّا 7:17; مَرْقُس 14:25"> أقولُ لكُم: لا أشرَبُ بَعدَ اليومِ مِنْ عَصيرِ الكَرمةِ هذا، حتّى يَجيءَ يومٌ فيهِ أشرَبُهُ مَعكُم جَديدًا في مَلكوتِ أبي“.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 21:1; أفَسُس 5:19–20; لُوقا 22:39"> ثُمَّ سبَّحوا وخَرَجوا إلى جبَلِ الزَّيْتونِ.</span>
</div>
</div>

## يسوع يُنبئ بإنكار بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="زَكَريّا 13:7; مَرْقُس 14:27–28; يُوحَنّا 16:32"> وقالَ لهُم يَسوعُ: ”في هذِهِ اللَّيلَةِ ستَترُكوني كُلُّكُم، فالكِتابُ يَقولُ: سَأضرِبُ الرّاعيَ، فتَتَبدَّدُ خِرافُ القَطيعِ.</span>
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 28:10; لُوقا 18:33–34; ﻣﺘﻰ 28:16"> ولكِنْ بَعدَ قيامَتي مِنْ بَينِ الأمواتِ أسبُقُكُم إلى الجليلِ“.</span>
<span className="v" id="v33" data-v="33" data-xref="لُوقا 22:33; يُوحَنّا 13:36–38; الأمثال 28:25–26"> فقَالَ بُطرُسُ: ”لَو تَركوكَ كُلُّهُم، فأنا لن أترُكَكَ“.</span>
<span className="v" id="v34" data-v="34" data-xref="يُوحَنّا 13:38; لُوقا 22:34; ﻣﺘﻰ 26:75"> فقَالَ لَه يَسوعُ: ”الحقَّ أقولُ لكَ: في هذِهِ اللَّيلَةِ، قَبلَ أن يَصيحَ الدّيكُ، تُنكِرُني ثلاثَ مرَّاتٍ“.</span>
<span className="v" id="v35" data-v="35" data-xref="ﺭﻭﻣﻴﺔ 11:20; يُوحَنّا 13:37; الأمثال 29:23"> فأجابَهُ بُطرُسُ: ”لا أُنكِرُكَ وإنْ كانَ علَيَّ أن أموتَ معَكَ“. وهكذا قالَ التَّلاميذُ كُلُّهُم.</span>
</div>
</div>

## يسوع يصلّي في جتسماني

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="مَرْقُس 14:32–42; ﻣﺘﻰ 26:42; لُوقا 22:39–46"> ثُمَّ جاءَ يَسوعُ معَ تلاميذِهِ إلى موضِعٍ اَسمُهُ جَتْسِماني، فقَالَ لهُم: ”أُقعُدوا هُنا، حتّى أذهَبَ وأُصلِّيَ هُناكَ“.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 4:21; ﻣﺘﻰ 17:1; مَرْقُس 5:37"> وأخَذَ مَعهُ بُطرُسَ واَبنيْ زَبَدي، وبَدأَ يَشعُرُ بالحُزنِ والكآبَةِ.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 12:27; رسالَةُ بُطرُسَ الأُولَى 3:18; رسالَةُ بُطرُسَ الأُولَى 2:24"> فقَالَ لهُم: ”نفسي حَزينَةٌ حتّى الموتِ. إِنتَظِروا هُنا واَسهَروا مَعي“.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 6:38; فِيلِبِّي 2:8; لُوقا 22:41–42"> واَبتَعَدَ عنهُم قَليلًا واَرتَمى على وجهِهِ وصلَّى فَقالَ: ”إنْ أمكَنَ يا أبي، فلْتَعبُرْ عنِّي هذِهِ الكأسُ. ولكن لا كما أنا أُريدُ، بل كما أنتَ تُريدُ“.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="مَرْقُس 14:37; لُوقا 22:45; ﻣﺘﻰ 26:43"> ورجَعَ إلى التَّلاميذِ فوجَدَهُم نِيامًا، فقَالَ لبُطرُسَ: ”أهكذا لا تَقدِرونَ أنْ تَسهَروا مَعي ساعةً واحدةً؟</span>
<span className="v" id="v41" data-v="41" data-xref="مَرْقُس 14:38; لُوقا 22:46; أفَسُس 6:18"> إسهَروا وصلُّوا لِئلَّا تَقَعُوا في التَّجرِبَةِ. الرُّوحُ راغِبةٌ، ولكنَّ الجسَدَ ضَعيفٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="ﻣﺘﻰ 26:39; مَرْقُس 14:39–40; العِبرانِيّين 4:15"> واَبتَعدَ ثانيةً وصلّى، فقَالَ: ”يا أبي، إذا كانَ لا يُمكِنُ أنْ تَعبُرَ عنِّي هذِهِ الكأسُ، إلَّا أنْ أشرَبَها، فلْتكُنْ مَشيئتُكَ“.</span>
<span className="v" id="v43" data-v="43" data-xref="لُوقا 9:32; ﺭﻭﻣﻴﺔ 13:1; الأمثال 23:34"> ثُمَّ رجَعَ فوَجَدَهُم نِيامًا، لأنَّ النُّعاسَ أثقَلَ جُفونَهُم.</span>
</div>
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:8; لُوقا 18:1; ﻣﺘﻰ 6:7"> فتَركَهُم وعادَ إلى الصَّلاةِ مرَّةً ثالِثةً، فردَّدَ الكلامَ نفسَهُ.</span>
<span className="v" id="v45" data-v="45" data-xref="يُوحَنّا 13:1; يُوحَنّا 12:27; مَرْقُس 14:41–42"> ثُمَّ رجَعَ إلى التَّلاميذِ وقالَ لهُم: ”أنِيامٌ بَعدُ ومُستَريحونَ؟ جاءَتِ السّاعَةُ الَّتي فيها يُسلَّمُ ابنُ الإنسان إلى أيدي الخاطِئينَ.</span>
<span className="v" id="v46" data-v="46" data-xref="لُوقا 9:51; يُوحَنّا 14:31; صَمُوئيلَ الأوَّلُ 17:48"> قوموا نَنصرِفُ! إِقترَبَ الّذي يُسَلِّمُني“.</span>
</div>
</div>

## القبض على يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v47" data-v="47" data-xref="اعمال الرسل 1:16; ﻣﺘﻰ 26:55; مَرْقُس 14:43–50"> وبَينَما يَسوعُ يتكَلَّمُ وصَلَ يَهوذا، أحدُ التَّلاميذِ الاثنَي عشَرَ، على رأسِ عِصابةٍ كبيرةٍ تَحمِلُ السُّيوفَ والعِصِيَّ، أرسلَها رُؤساءُ الكَهنَةِ وشُيوخُ الشَّعبِ.</span>
</div>

<div className="bp">

<span className="v" id="v58" data-v="58" data-xref="يُوحَنّا 7:32; يُوحَنّا 18:15–16; يُوحَنّا 18:25"> وتَبِعَه بُطرُسُ عَنْ بُعدٍ إلى دارِ رَئيسِ الكَهنَةِ. فدخَلَ وقَعدَ معَ الحَرَسِ ليرى النِهايةَ.</span>
</div>
<div className="bp">
<span className="v" id="v59" data-v="59" data-xref="اعمال الرسل 6:11–13; التَّثْنِيَة 19:16–21; مَرْقُس 14:55–56"> وكانَ رُؤساءُ الكَهنَةِ وجميعُ أعضاءِ المَجلِسِ يَطلُبونَ شَهادةَ زُورٍ على يَسوعَ ليَقتُلوهُ،</span>
<span className="v" id="v60" data-v="60" data-xref="التَّثْنِيَة 19:15; المَزامِير 35:11; المَزامِير 27:12"> فما وجَدوا، معَ أنَّ كثيرًا مِنْ شهودِ الزّورِ تَقدَّموا بشَهاداتِهِم. ثُمَّ قامَ شاهدانِ</span>
<span className="v" id="v61" data-v="61" data-xref="يُوحَنّا 2:19–21; ﻣﺘﻰ 27:40; مَرْقُس 15:29"> وقالا: ”هذا الرَّجُلُ قالَ: أقدِرُ أنْ أهدِمَ هَيكَلَ اللهِ وأبنِيَهُ في ثلاثةِ أيّامٍ“.</span>
<span className="v" id="v62" data-v="62" data-xref="يُوحَنّا 19:9–11; يُوحَنّا 18:19–24; مَرْقُس 14:60"> فقامَ رئيسُ الكَهنَةِ وقالَ ليَسوعَ: ”أما تُجيبُ بِشيءٍ؟ ما هذا الّذي يَشهَدانِ بِه علَيكَ؟“</span>
<span className="v" id="v63" data-v="63" data-xref="اللّاويِّين 5:1; ﻣﺘﻰ 16:16; إشَعْياء 53:7"> فظَلَّ يَسوعُ ساكِتًا. فقَالَ لَه رَئيسُ الكَهنَةِ: ”أستَحلِفُكَ باللهِ الحيِّ أنْ تَقولَ لنا: هَل أنتَ المَسيحُ اَبنُ اللهِ؟“</span>
<span className="v" id="v64" data-v="64" data-xref="دانيال 7:13; مَرْقُس 14:62; المَزامِير 110:1"> فأجابَ يَسوعُ: ”أنتَ قُلتَ. وأنا أقولُ لكُم: سترَوْنَ بَعدَ اليومِ اَبنَ الإنسانِ جالِسًا عَنْ يَمينِ اللهِ القَديرِ وآتيًا على سَحابِ السَّماءِ!“</span>

<span className="v" id="v66" data-v="66" data-xref="يُوحَنّا 19:7; اعمال الرسل 7:52; اللّاويِّين 24:11–16"> فما رأيُكُم؟“ فأجابوهُ: ”يَسْتَوجِبُ الموتَ!“</span>

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v69" data-v="69" data-xref="المُلُوكِ الأوَّلُ 19:13; ﻣﺘﻰ 26:71; اعمال الرسل 5:37"> وكانَ بُطرُسُ قاعِدًا في ساحَةِ الدَّارِ، فدنَتْ إلَيهِ جاريَةٌ وقالَت: ”أنتَ أيضًا كُنتَ معَ يَسوعَ الجليليّ!“</span>
<span className="v" id="v70" data-v="70" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:12; رُؤيا يُوحَنّا 21:8; ﻣﺘﻰ 26:40–43"> فأنكَرَ أمامَ جميعِ الحاضرينَ، قالَ: ”لا أفهَمُ ما تَقولينَ“.</span>
<span className="v" id="v71" data-v="71" data-xref="مَرْقُس 14:68–69; يُوحَنّا 18:25–27; لُوقا 22:58"> وخرَجَ إلى مَدخَلِ السّاحةِ، فَرأتهُ جاريةٌ أخرى. فقَالَت لِمَن كانوا هُناكَ: ”هذا الرَّجُلُ كانَ معَ يَسوعَ النّاصريِّ!“</span>

<span className="v" id="v74" data-v="74" data-xref="رُؤيا يُوحَنّا 3:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:22; مَرْقُس 14:71–72"> فأخذَ يَلعَنُ ويحلِفُ: ”أنا لا أعرِفُ هذا الرَّجُلَ“. فصاحَ الدّيكُ في الحالِ،</span>
<span className="v" id="v75" data-v="75" data-xref="ﻣﺘﻰ 26:34; ﺭﻭﻣﻴﺔ 7:18–20; لُوقا 22:31–34"> فتذكَّرَ بُطرُسُ قَولَ يَسوعَ: ”قَبلَ أنْ يَصيحَ الدّيكُ تُنكِرُني ثلاثَ مرَّاتٍ“. فخرَجَ وبكى بُكاءً مُرًّا.</span>
