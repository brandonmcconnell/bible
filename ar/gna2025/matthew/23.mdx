## يسوع يحذر من معلمي الشريعة والفريسيين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 12:1; لُوقا 20:45–46; ﻣﺘﻰ 15:10–20"> وخاطَبَ يَسوعُ الجُموعَ وتلاميذَهُ،</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 20:46; مَرْقُس 12:38; مَلاخِي 2:7"> قالَ: ”مُعلِّمو الشَّريعةِ والفَرِّيسيُّونَ على كُرسِيِّ موسى جالسونَ،</span>
<span className="v" id="v3" data-v="3" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:5; تِيطُس 1:16; ﺭﻭﻣﻴﺔ 2:19–24"> فاَفعَلوا كُلَّ ما يَقولونَهُ لكُم واَعمَلوا بِه. ولكِنْ لا تَعمَلوا مِثلَ أعمالِهِم، لأنَّهُم يَقولونَ ولا يَفعلونَ:</span>
<span className="v" id="v4" data-v="4" data-xref="لُوقا 11:46; اعمال الرسل 15:10; غَلاطِيَّة 6:13"> يَحزِمونَ أحمالًا ثَقيلَةً شاقَّةَ الحَمْلِ ويُلْقونَها على أكتافِ النَّاسِ، ولكنَّهُم لا يُحَرِّكونَ إصْبَعًا تُعينُهُم على حَمْلِها.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 6:1–16; يُوحَنّا 12:43; فِيلِبِّي 2:3"> وهُمْ لا يَعمَلونَ عَمَلًا إلَّا لِيُشاهِدَهُمُ النَّاسُ: يجعَلونَ عَصائِبَهُم عريضَةً على جِباهِهِم وسواعِدِهم، ويُطَوِّلونَ أطرافَ ثِيابِهِم،</span>
<span className="v" id="v6" data-v="6" data-xref="رسالَةُ يُوحَنّا الثّالثَة 1:9; لُوقا 11:43–54; مَرْقُس 12:38–39"> ويُحبُّونَ مَقاعِدَ الشَّرَفِ في الوَلائِمِ ومكانَ الصَّدارَةِ في المجامِعِ</span>
<span className="v" id="v7" data-v="7" data-xref="يُوحَنّا 1:38; يُوحَنّا 1:49; يُوحَنّا 3:26"> والتَّحِيّاتِ في الأسواقِ، وأنْ يَدْعُوَهُمُ النَّاسُ: يا مُعلِّمُ.</span>
<span className="v" id="v8" data-v="8" data-xref="يَعقُوب 3:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:12–13; رسالَةُ بُطرُسَ الأُولَى 5:3"> أمَّا أنتُم فلا تَسمَحوا بأنْ يَدْعُوَكُم أحدٌ: يا مُعلِّمُ، لأنّكُم كُلَّكُم إخوَةٌ ولكُم مُعلِّمٌ واحدٌ.</span>
<span className="v" id="v9" data-v="9" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:1; العِبرانِيّين 12:9; مَلاخِي 1:6"> ولا تَدْعوا أحدًا على الأرضِ يا أبانا، لأنَّ لكُم أبًا واحدًا هوَ الآبُ السَّماويُّ.</span>
<span className="v" id="v10" data-v="10"> ولا تَسمَحوا بأنْ يَدْعوَكُم أحدٌ: يا سيِّدُ، لأنَّ لكُم سيِّدًا واحدًا هوَ المَسيحُ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:19; غَلاطِيَّة 5:13; يُوحَنّا 13:14–15"> وليكُنْ أكبَرُكُم خادِمًا لكُم.</span>
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:5; يَعقُوب 4:6; الأمثال 29:23"> فَمَن يَرفعْ نَفسَهُ يَنخفِضْ، ومَنْ يَخفِضْ نفسَهُ يرتَفِعْ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="لُوقا 11:52; ﻣﺘﻰ 23:29; ﻣﺘﻰ 23:23"> الوَيلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيِّينَ المُراؤونَ! تُغلِقونَ مَلكوتَ السَّماواتِ في وُجوهِ النَّاسِ، فلا أنتُم تَدخُلونَ، ولا تَترُكونَ الدَّاخلينَ يَدخُلونَ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14"> &#91;الوَيلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيُّونَ المُراؤونَ! تأكُلونَ بُيوتَ الأرامِلِ وأنتُمْ تُظهِرونَ أنَّكُم تُطيلونَ الصَّلاةَ، سيَنالُكُم أشدُّ العِقابِ&#93;.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="أفَسُس 2:3; اعمال الرسل 13:10; اعمال الرسل 13:43"> الوَيلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيُّونَ المُراؤونَ! تَقطَعونَ البحرَ والبَرَّ لتكسِبوا واحدًا إلى دِيانَتِكُم، فإذا نَجَحتُم، جَعَلْتموهُ يستَحِقُّ جَهنَّمَ ضِعفَ ما أنتُم تَستَحِقُّونَ!</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 15:14; ﻣﺘﻰ 5:33–35; يَعقُوب 5:12"> الوَيلُ لكُم أيُّها القادَةُ العُميانُ! تَقولونَ: مَنْ حَلفَ بالهَيكلِ لا يَلتزِمُ بيَمينِهِ، ولكنْ مَنْ حَلَفَ بذَهَبِ الهَيكَلِ يَلتَزِمُ بيَمينِهِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 23:19; العَدَد 16:38–39; المَزامِير 94:8"> فأيُّما أعظمُ، أيُّها الجُهَّالُ العُميانُ؟ الذَّهَبُ أمِ الهَيكَلُ الّذي قَدَّسَ الذَّهَبَ؟</span>
<span className="v" id="v18" data-v="18"> وتَقولونَ: مَنْ حلَفَ بالمَذبحِ لا يلتَزمُ بيمينِهِ، ولكِنْ من حَلفَ بالقُربانِ الّذي على المذبَحِ يَلتَزِمُ بيَمينِهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 29:37; الخُرُوج 30:29"> فأيُّما أعظمُ، أيُّها العُميانُ؟ القُربانُ أمِ المَذبحُ الّذي يُقدِّسُ القُربانَ؟</span>
<span className="v" id="v20" data-v="20"> أما ترَوْن أنَّ الّذي يَحلِفُ بالمذبَحِ يَحلِف بِه وبِكُلِّ ما علَيهِ،</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 26:8; المُلُوكِ الأوَّلُ 8:13; أخبارِ الأيّامِ الثّانِي 6:2"> والّذي يَحلِفُ بالهَيكَلِ يَحلِفُ بِه وباللهِ السَّاكنِ فيهِ،</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 5:34; المَزامِير 11:4; إشَعْياء 66:1"> والّذي يحلِفُ بالسَّماءِ يحلِفُ بعَرشِ اللهِ وبالجالِسِ علَيهِ؟</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="لُوقا 11:42; ميخا 6:8; هُوشَع 6:6"> الوَيلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيُّونَ المُراؤونَ! تُعطُونَ العُشْرَ مِنَ النَّعْنعِ والصَّعتَرِ والكَمُّونِ، ولكنَّكُم تُهمِلونَ أهمَّ ما في الشَّريعةِ: العَدلَ والرَّحمةَ والصِّدقَ، وهذا ما كانَ يَجبُ علَيكُم أنْ تَعمَلوا بِه مِنْ دونِ أن تُهمِلوا ذاكَ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 19:24; ﻣﺘﻰ 27:6–8; ﻣﺘﻰ 15:2–6"> أيُّها القادَةُ العُميانُ! تُصَفُّونَ الماءَ مِنَ البَعوضَةِ، ولكنَّكُم تَبتَلِعونَ الجمَلَ.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="لُوقا 11:39–40; ﻣﺘﻰ 15:19–20; مَرْقُس 7:4–13"> الويلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيُّونَ المُراؤونَ! تُطَهِّرونَ ظاهِرَ الكأسِ والصَّحنِ، وباطنُهُما مُمتلِئٌ بِما حصَلتُم علَيهِ بالنَّهبِ والطَمَعِ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1; يَعقُوب 4:8; العِبرانِيّين 10:22"> أيُّها الفَرِّيسيُّ الأعمى! طَهِّرْ أوَّلًا باطنَ الوِعاءِ، فيَصيرَ الظَّاهِرُ مِثلَهُ طاهرًا.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="لُوقا 11:44; اعمال الرسل 23:3; العَدَد 19:16"> الويلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيُّونَ المُراؤونَ! أنتُم كالقُبورِ المبيَّضَةِ، ظاهرُها جميلٌ وباطِنُها مُمتَلئٌ بعِظامِ الموتى وبكُلِّ فسادٍ.</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 16:15; المَزامِير 51:6; ﻣﺘﻰ 23:5"> وأنتُم كذلِكَ، تَظهَرونَ لِلنَّاسِ أبرارَ وباطِنُكُم كُلُّهُ رِياءٌ وشَرٌّ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="لُوقا 11:47–48; اعمال الرسل 2:29"> الوَيلُ لكُم يا مُعَلِّمي الشَّريعةِ والفَرِّيسيُّونَ المُراؤونَ! تَبْنونُ قُبورَ الأنبياءِ وتُزَيِّنونَ مدافِنَ الأتقياءِ،</span>
<span className="v" id="v30" data-v="30" data-xref="إرْمِيا 2:30; ﻣﺘﻰ 23:34–35; ﻣﺘﻰ 21:35–36"> وتَقولونَ: لو عِشنا في زَمَنِ آبائِنا، لما شارَكْناهُم في سَفْكِ دَمِ الأنبياءِ.</span>
<span className="v" id="v31" data-v="31" data-xref="اعمال الرسل 7:51–52; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:15–16; يَشُوع 24:22"> فتَشهَدونَ على أنفُسِكُم بأنَّكُم أبناءُ الّذينَ قَتلوا الأنبياءَ.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّكوين 15:16; العَدَد 32:14; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:16"> فتَمِّموا أنتُم ما بَدأَ بِه آباؤُكُم.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="ﻣﺘﻰ 3:7; ﻣﺘﻰ 12:34; ﻣﺘﻰ 5:22"> أيُّها الحيّاتُ أولادَ الأفاعي! كيفَ ستَهرُبونَ مِنْ عِقابِ جَهنَّمَ؟</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 10:16–17; ﻣﺘﻰ 13:52; كُولُوسِي 1:28"> لذلِكَ سأُرسِلُ إلَيكُم أنبياءَ وحُكَماءَ ومُعَلِّمينَ، فمِنهُم مَنْ تَقتُلونَ وتَصلِبونَ، ومِنهُم مَنْ تَجلِدونَ في مجامِعِكُم وتُطارِدونَ مِنْ مدينةٍ إلى مدينةٍ،</span>
<span className="v" id="v35" data-v="35" data-xref="أخبارِ الأيّامِ الثّانِي 24:20–22; زَكَريّا 1:1; العِبرانِيّين 11:4"> حتّى يَنزِلَ بِكُم العِقابُ على سَفكِ كُلِّ دمٍ بريءٍ على الأرضِ، مِنْ دمِ هابيلَ الصِّدّيقِ إلى دمِ زكريّا بنِ بَرَخِيّا الّذي قتَلتُموهُ بَينَ المَذبحِ وبَيتِ اللهِ.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 10:23; ﻣﺘﻰ 24:34; مَرْقُس 13:30–31"> الحقَّ أقولُ لكُم: هذا كُلُّهُ سيَقَعُ على هذا الجِيلِ!</span>

## محبة يسوع لأورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v37" data-v="37" data-xref="لُوقا 13:34–35; راعُوث 2:12; المَزامِير 91:4"> ”أُورُشليمُ، أُورُشليمُ! يا قاتِلَةَ الأنبياءِ وراجِمةَ المُرسَلينَ إلَيها. كَم مَرَّةٍ أرَدتُ أنْ أجمَعَ أبناءَكِ، مِثلَما تَجمَعُ الدَّجاجةُ فِراخَها تَحتَ جَناحَيْها، فما أردتُم.</span>
<span className="v" id="v38" data-v="38" data-xref="إرْمِيا 22:5; لُوقا 13:35; لُوقا 21:24"> وها هوَ بَيتُكُم مَتروكٌ لكُم خَرابًا.</span>
<span className="v" id="v39" data-v="39" data-xref="المَزامِير 118:26; ﻣﺘﻰ 21:9; ﺭﻭﻣﻴﺔ 11:25"> أقولُ لكُم: لَن تَرَوْني إلَّا يومَ تهتِفونَ: تَبارَكَ الآتي بِاَسمِ الرَّبِّ“.</span>
</div>
