## التجلي ومجيء إيليَّا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 9:28–36; ﻣﺘﻰ 26:37; مَرْقُس 5:37"> وبَعدَ سِتَّةِ أيّامٍ أخَذَ يَسوعُ بُطرُسَ ويَعقوبَ وأخاهُ يوحنَّا، واَنفَرَدَ بِهِم على جبَلٍ مُرتَفِعٍ،</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 104:2; ﻣﺘﻰ 28:3; لُوقا 9:29"> وتَجلَّى بمَشهَدٍ مِنهُم، فأشرقَ وجْهُهُ كالشَّمسِ وصارَتْ ثيابُهُ بَيضاءَ كالنُّورِ.</span>
<span className="v" id="v3" data-v="3" data-xref="لُوقا 24:44; مَرْقُس 9:4; مَلاخِي 4:5"> وظهَرَ لهُم موسى وإيليَّا يُكلِّمانِ يَسوعَ.</span>
<span className="v" id="v4" data-v="4" data-xref="مَرْقُس 9:5–6; لُوقا 9:33; يُوحَنّا 14:8–9"> فقَالَ بُطرُسُ ليَسوعَ: ”يا ربُّ، ما أجمَلَ أن نكونَ هُنا: فإن شِئتَ، نَصَبتُ هُنا ثلاثَ مظالَّ: واحِدةً لكَ وواحِدةً لموسى وواحِدةً لإيليَّا“.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 3:17; ﻣﺘﻰ 12:18; التَّثْنِيَة 18:15"> وبَينَما هوَ يتكلَّمُ، ظلَّلتْهُم سَحابَةٌ مُضيئَةٌ، وقالَ صوتٌ مِنَ السَّحابةِ: ”هذا هوَ اَبني الحبيبُ الّذي بِه رَضِيتُ، فلَهُ اَسمَعوا!“</span>
<span className="v" id="v6" data-v="6" data-xref="اعمال الرسل 26:14; رسالَةُ بُطرُسَ الثّانيَة 1:18; دانيال 8:17"> فلمَّا سَمِعَ التَّلاميذُ هذا الصَّوتَ وقَعوا على وجوهِهِم وهُمْ في خوفٍ شديدٍ.</span>
<span className="v" id="v7" data-v="7" data-xref="دانيال 10:18; ﻣﺘﻰ 14:27; دانيال 8:18"> فدَنا يَسوعُ ولَمَسَهُم وقالَ لهُم: ”قوموا، لا تخافوا“.</span>
<span className="v" id="v8" data-v="8" data-xref="لُوقا 9:36; مَرْقُس 9:8; اعمال الرسل 12:10–11"> فرَفَعوا عُيونَهُم، فما رأوا إلَّا يَسوعَ وحدَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="مَرْقُس 8:30; مَرْقُس 9:9–13; ﻣﺘﻰ 17:23"> وبَينَما هُمْ نازلونَ مِنَ الجبَلِ، أوصاهُم يَسوعُ قالَ: ”لا تُخبِروا أحدًا بِما رأيتُم إلى أنْ يقومَ اَبنُ الإنسانِ مِنْ بَينِ الأمواتِ“.</span>
<span className="v" id="v10" data-v="10" data-xref="مَلاخِي 4:5–6; ﻣﺘﻰ 11:14; يُوحَنّا 1:21"> فسألَهُ التَّلاميذُ: ”لِماذا يقولُ مُعَلِّمو الشَّريعَةِ: يجبُ أنْ يَجيءَ إيليَّا أوَّلًا؟“</span>
<span className="v" id="v11" data-v="11" data-xref="اعمال الرسل 3:21; لُوقا 1:16–17; مَلاخِي 4:6"> فأجابَهُم: ”نَعم، يَجيءُ إيليَّا ويُصلِحُ كُلَّ شَيءٍ.</span>
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 53:3–12; ﻣﺘﻰ 16:21; اعمال الرسل 13:24–28"> ولكنِّي أقولُ لكُم: جاءَ إيليَّا فما عَرَفوهُ، بَلْ فَعلوا بِه على هَواهُم. وكذلِكَ اَبنُ الإنسانِ سيتألَّمُ على أيديهِم“.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 11:14"> ففَهِمَ التَّلاميذُ أنَّهُ كانَ يُكلِّمُهُم عَنْ يوحنَّا المَعمدانِ.</span>
</div>
</div>

## يسوع يشفي صبيًّا فيه شيطان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="مَرْقُس 9:14–29; مَرْقُس 1:40; اعمال الرسل 10:25–26"> ولمَّا رَجَعُوا إلى الجُموعِ، أقبلَ إلَيهِ رَجُلٌ وسَجَدَ</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 4:24; أيُّوب 1:10–19; مَرْقُس 9:20–22"> وقالَ لَه: ”إرحمِ اَبني يا سيِّدي، لأنَّهُ يُصابُ بالصَّرَعِ ويتَألَّمُ ألمًا شديدًا. وكثيرًا ما يَقَعُ في النَّارِ وفي الماءِ.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 17:19–20; لُوقا 9:40; اعمال الرسل 19:15–16"> وجِئْتُ بِه إلى تلاميذِكَ، فما قَدِروا أنْ يَشفُوهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="العَدَد 14:11; ﻣﺘﻰ 6:30; مَرْقُس 16:14"> فأجابَ يَسوعُ: ”أيُّها الجِيلُ غَيرُ المُؤمِنِ الفاسِدُ! إلى متى أبْقى معكُم؟ وإلى متى أحتَمِلُكُم؟ قَدِّموا الصَّبيَّ إليَّ هُنا!“</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 9:22; مَرْقُس 1:34; مَرْقُس 5:8"> واَنتهَرَهُ يَسوعُ، فَخرَجَ الشَّيطانُ مِنَ الصَّبيِّ، فشُفِيَ في الحالِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 9:28; مَرْقُس 4:10"> فاَنفَرَدَ التَّلاميذُ بيَسوعَ وسألُوهُ: ”لِماذا عَجِزْنا نَحنُ عَنْ أنْ نَطرُدَهُ؟“</span>
<span className="v" id="v20" data-v="20" data-xref="لُوقا 17:6; مَرْقُس 11:23; لُوقا 1:37"> فأجابَهُم: ”لِقِلَّةِ إيمانِكُم! الحقَّ أقولُ لكُم: لو كانَ لكُم إيمانٌ بِمقدارِ حَبَّةٍ مِنْ خَرْدَلٍ، لَقُلتُم لِهذا الجبَلِ: إِنتَقِلْ مِنْ هُنا إلى هُناكَ فَينتَقِلُ، ولَمَا عَجِزتُم عَنْ شَيءٍ.</span>
<span className="v" id="v21" data-v="21"> &#91;وهذا الجِنْسُ مِنَ الشَّياطينِ لا يُطرَدُ إلَّا بالصَّلاةِ والصَّومِ&#93;“.</span>
</div>
</div>

## يسوع يُنبئ مرَّة ثانية بموته وقيامته

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 16:21; لُوقا 18:31–34; ﻣﺘﻰ 17:23"> وكانَ التَّلاميذُ مُجتَمِعينَ في الجليلِ، فقَالَ لهُم يَسوعُ: ”سيُسلَّمُ اَبنُ الإنسانِ إلى أيدي النَّاسِ،</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 16:21; إشَعْياء 53:10–12; مَرْقُس 8:31"> فيَقتُلونَهُ، وفي اليومِ الثَّالِثِ يَقومُ مِنْ بَينِ الأمواتِ“. فحَزِنَ التَّلاميذُ كثيرًا.</span>
</div>
</div>

## يسوع يدفع ضريبة الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 38:26; الخُرُوج 30:13; مَرْقُس 9:33"> وعِندَما رجَعَ يَسوعُ وتلاميذُهُ إلى كَفْرَناحومَ، جاءَ جُباةُ ضَريبةِ الهَيكَلِ إلى بُطرُسَ وسألوهُ: ”أما يُوفي مُعَلِّمُكُم ضَريبةَ الهَيكَلِ؟“</span>
<span className="v" id="v25" data-v="25" data-xref="ﻣﺘﻰ 22:21; ﺭﻭﻣﻴﺔ 13:6–7; ﻣﺘﻰ 22:17"> فأجابَ: ”نعم“. فلمَّا دخَلَ بُطرُسُ إلى البَيتِ، عاجَلَهُ يَسوعُ بِقولِهِ: ”ما رأيُكَ، يا سِمْعانُ؟ مِمَّنْ يأخُذُ مُلوكُ الأرضِ الجِبايَةَ أو الجِزيَةَ؟ أمِنْ أَبناءِ البِلادِ أم مِنَ الغُرَباءِ؟“</span>
<span className="v" id="v26" data-v="26"> فأجابَ بُطرُسُ: ”مِنَ الغُرَباءِ“. فقَالَ لَه يَسوعُ: ”إذًا، فالأبناءُ أحرارٌ في أمرِ إيفائِها.</span>
<span className="v" id="v27" data-v="27" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:19–22"> لكنَّنا لا نُريدُ أنْ نُحرِجَ أحدًا، فاَذهبْ إلى البحرِ وألقِ الصِّنَّارَةَ، وأَمسِكْ أوَّلَ سَمكَةٍ تَخرُجُ واَفتَحْ فمَها تَجِدْ فيهِ قِطعةً بأربعةِ دَراهمَ، فخُذْها واَدفَعْها إلَيهِم عنِّي وعَنْكَ“.</span>
</div>
