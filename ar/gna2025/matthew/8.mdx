## شفاء الأبرص

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 8:18; ﻣﺘﻰ 4:25–1; لُوقا 5:15"> ولمَّا نزَلَ يَسوعُ مِنَ الجبَلِ، تَبِعَتْهُ جُموعٌ كبيرةٌ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 15:25; لُوقا 17:12–19; ﻣﺘﻰ 18:26"> ودَنا مِنهُ أبرَصُ، فسجَدَ لَه وقالَ: ”يا سيِّدي، إنْ أرَدْتَ فأنْتَ قادِرٌ أنْ تُطَهِّرَني“.</span>
<span className="v" id="v3" data-v="3" data-xref="لُوقا 5:13; المَزامِير 33:9; المُلُوكِ الثّانِي 5:14"> فمَدَّ يَسوعُ يدَه ولَمَسهُ وقالَ: ”أُريدُ، فاَطهُرْ!“ فطَهُرَ مِنْ بَرَصِهِ في الحالِ.</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 14:2–32; لُوقا 5:14; لُوقا 17:14"> فقَالَ لَه يَسوعُ: ”إيَّاكَ أن تُخبِرَ أحدًا. ولكن اَذهَبْ إلى الكاهنِ وأرِهِ نَفسَكَ. ثُمَّ قَدِّمِ القُربانَ الّذي أمرَ بِه موسى، شَهادةً عِندَهُم“.</span>
</div>
</div>

## شفاء خادم الضابط

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="لُوقا 7:1–10; اعمال الرسل 27:43; اعمال الرسل 10:1–33"> ودخَلَ يَسوعُ كَفْرَناحومَ، فجاءَهُ ضابِطٌ رومانِيٌّ وتَوَسَّلَ إلَيهِ بِقولِهِ:</span>
<span className="v" id="v6" data-v="6" data-xref="كُولُوسِي 4:1; ﻣﺘﻰ 4:24; اعمال الرسل 9:33"> ”يا سيِّدُ، خادِمي طَريحُ الفِراشِ في البَيتِ يَتوَجَّعُ كثيرًا ولا يَقدِرُ أنْ يَتحرَّكَ“.</span>
<span className="v" id="v7" data-v="7" data-xref="ﻣﺘﻰ 9:18–19; لُوقا 7:6; مَرْقُس 5:23–24"> فقَالَ لَه يَسوعُ: ”أنا ذاهبٌ لأشفِيَهُ“.</span>
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 107:20; لُوقا 7:6–7; المَزامِير 33:9"> فأجابَ الضَّابِطُ: ”أنا لا أستحِقُّ، يا سيِّدي، أنْ تَدخُلَ تَحتَ سقفِ بَيتي. ولكِنْ يكفي أنْ تَقولَ كَلِمَةً فيُشفى خادِمي.</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 4:35–36; تِيطُس 2:9; كُولُوسِي 3:22"> فأنا مَرؤوسٌ ولي جُنودٌ تَحتَ أمري، أقولُ لِهذا: إذهَبْ! فيذهَبُ، ولِلآخَرِ: تَعالَ! فيجيءُ، ولِخادِمي: إعمَلْ هذا، فيَعْمَلُ“.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 15:28; لُوقا 7:9; مَرْقُس 6:6"> فتعجَّبَ يَسوعُ مِنْ كلامِهِ وقالَ لِلَّذينَ يَتبَعونَهُ: ”الحقَّ أقولُ لكُم: ما وجَدتُ مِثلَ هذا الإيمانِ عِندَ أحَدٍ في إِسْرائيلَ.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 13:28–29; أفَسُس 3:6; غَلاطِيَّة 3:28–29"> أقولُ لكُم: كثيرونَ مِنَ النَّاسِ سيَجيئونَ مِنَ المَشرِقِ والمَغرِبِ ويَجلِسونَ إلى المائدةِ معَ إبراهيمَ وإسحقَ ويعقوبَ في مَلكوتِ السَّماواتِ.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 25:30; لُوقا 13:28; ﻣﺘﻰ 21:43"> وأمَّا مَنْ كانَ لَهُمُ المَلكوتُ، فيُطْرَحونَ خارِجًا في الظُّلمَةِ، وهُناكَ البُكاءُ وصَريفُ الأسنانِ“.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 9:29–30; مَرْقُس 9:23; يُوحَنّا 4:50"> وقالَ يَسوعُ للضَّابِطِ: ”إذهَبْ، وليَكُنْ لكَ على قَدْرِ إيمانِكَ“. فشُفِيَ الخادِمُ في تِلكَ السَّاعةِ.</span>
</div>
</div>

## شفاء حماة بطرس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="لُوقا 4:38–41; مَرْقُس 1:29–34; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:5"> ودَخَلَ يَسوعُ إلى بَيتِ بُطرُسَ، فوَجَدَ حَماةَ بُطرُسَ طَريحَةَ الفِراشِ بالحُمَّى.</span>
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الثّانِي 13:21; لُوقا 4:38–39; لُوقا 8:54"> فلَمَسَ يدَها، فتَركَتْها الحُمَّى. فقامَت وأخَذَتْ تَخدُمُه.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 15:26; ﻣﺘﻰ 14:14; مَرْقُس 1:32–34"> وعِندَ المساءِ، جاءَهُ النَّاسُ بِكثيرٍ مِنَ الّذينَ فيهِم شَياطينُ، فأخرَجَها بِكَلِمةٍ مِنهُ، وشفَى جميعَ المَرضى.</span>
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 53:4; رسالَةُ بُطرُسَ الأُولَى 2:24; ﻣﺘﻰ 1:22"> فتَمَّ ما قالَ النَّبيُّ إشَعْيا: ”أخذَ أوجاعَنا وحمَلَ أمراضَنا“.</span>
</div>
</div>

## يسوع أم العالم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="لُوقا 4:42–43; لُوقا 8:22; مَرْقُس 4:35"> ورأى يَسوعُ جُمهورًا حَولَهُ، فأمَرَ تلاميذَهُ بالعُبورِ إلى الشّاطِئِ المُقابِلِ.</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 22:33–34; يُوحَنّا 13:36–38; لُوقا 14:25–27"> فدَنا مِنهُ أحَدُ مُعلِّمي الشَّريعةِ وقالَ لَه: ”يا مُعلِّمُ، أتبَعُكَ أيْنَما تَذهَبُ“.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 84:3; لُوقا 2:7; إشَعْياء 53:2–3"> فأجابَهُ يَسوعُ: ”للثَّعالِبِ أوكارٌ، ولِطُيورِ السَّماءِ أعشاشٌ، وأمَّا اَبنُ الإنسانِ، فلا يَجِدُ أينَ يُسنِدُ رأسَهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="لُوقا 9:59–62; اللّاويِّين 21:11–12; العَدَد 6:6–7"> وقالَ لَه واحدٌ مِنْ تلاميذِهِ: ”يا سيِّدُ، دَعْني أذهَبُ أوّلًا وأدفِنُ أبي“.</span>
<span className="v" id="v22" data-v="22" data-xref="كُولُوسِي 2:13; لُوقا 15:32; أفَسُس 5:14"> فقَالَ لَه يَسوعُ: ”إِتبَعْني واَترُكِ المَوتى يَدفِنونَ مَوتاهُم!“</span>
</div>
</div>

## يسوع يهدئ العاصفة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="لُوقا 8:22–25; مَرْقُس 4:36–41; ﻣﺘﻰ 9:1"> ورَكِبَ يَسوعُ القارِبَ، فتَبِعَهُ تلاميذُهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="لُوقا 8:23; المَزامِير 107:23–27; مَرْقُس 4:37–38"> وهَبَّتْ عاصِفةٌ شديدةٌ في البَحرِ حتّى غَمَرتِ الأمواجُ القارِبَ. وكانَ يَسوعُ نائمًا.</span>
<span className="v" id="v25" data-v="25" data-xref="ﻳﻮﻧﺎﻥ 1:6; أخبارِ الأيّامِ الثّانِي 14:11; مَرْقُس 4:38–39"> فدَنا مِنهُ تلاميذُهُ وأيقَظوهُ وقالوا لَه: ”نَجِّنا يا سيِّدُ، فنَحنُ نَهلِكُ!“</span>
<span className="v" id="v26" data-v="26" data-xref="إشَعْياء 41:10–14; المَزامِير 65:7; المَزامِير 107:28–30"> فأجابَهُم يَسوعُ: ”ما لكُمْ خائِفينَ، يا قليلي الإيمانِ؟“ وقامَ واَنتَهرَ الرِّياحَ والبحرَ، فحَدَثَ هُدوءٌ تامٌّ.</span>
<span className="v" id="v27" data-v="27" data-xref="مَرْقُس 7:37; مَرْقُس 6:51; ﻣﺘﻰ 14:33"> فتعَجَّبَ النَّاسُ وقالوا: ”مَنْ هذا حتّى تُطيعَهُ الرِّياحُ والبحرُ؟“</span>
</div>
</div>

## طرد الشياطين وغرق الخنازير

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="لُوقا 8:26–39; مَرْقُس 5:1–20; اعمال الرسل 10:38"> ولمَّا وصَلَ يَسوعُ إلى الشَّاطئِ المُقابِلِ في ناحيةِ الجدرييِّنَ اَستقْبَلَهُ رَجُلانِ خَرَجا مِنَ المَقابِرِ، وفيهما شياطينُ. وكانا شَرِسَيْنِ جدًّا، حتّى لا يَقدِرَ أحدٌ أن يمُرَّ مِنْ تِلكَ الطَّريقِ.</span>
<span className="v" id="v29" data-v="29" data-xref="مَرْقُس 1:24; لُوقا 4:34; يَعقُوب 2:19"> فأخذا يَصيحانِ: ”ما لنا ولكَ، يا اَبنَ اللهِ؟ أجِئتَ إلى هُنا لِتُعذِّبَنا قَبلَ الأوانِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="التَّثْنِيَة 14:8; اللّاويِّين 11:7; مَرْقُس 5:11"> وكانَ يَرعَى بَعيدًا مِنْ هُناكَ قطيعٌ كبيرٌ مِنَ الخنازيرِ،</span>
<span className="v" id="v31" data-v="31" data-xref="رُؤيا يُوحَنّا 20:1–2; لُوقا 8:30–33; مَرْقُس 5:12"> فتوَسَّلَ الشَّياطينُ إلى يَسوعَ بقولِهِم: ”إنْ طرَدْتَنا، فأرْسِلنا إلى قطيعِ الخنازيرِ“.</span>
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 5:13; اعمال الرسل 2:23; رُؤيا يُوحَنّا 20:7"> فقَالَ لهُم: ”إِذهَبوا!“ فخَرجوا ودَخَلوا في الخنازيرِ، فاَندفَعَ القطيعُ كُلُّهُ مِنَ المُنحَدَرِ إلى البحرِ وهَلَكَ في الماءِ.</span>
<span className="v" id="v33" data-v="33" data-xref="لُوقا 8:34–36; مَرْقُس 5:14–16; اعمال الرسل 19:15–17"> فهرَبَ الرُّعاةُ إلى المدينةِ وأخبَروا بِما حدَثَ وبِما جرَى للرَّجُلَينِ اللَّذَينِ كانَ فيهِما شياطينُ.</span>
<span className="v" id="v34" data-v="34" data-xref="اعمال الرسل 16:39; لُوقا 8:37–39; لُوقا 5:8"> فخَرَجَ أهلُ المدينةِ كُلُّهُم إلى يَسوعَ. ولمَّا رأوْهُ طلَبوا إلَيهِ أنْ يَرحَلَ عَنْ دِيارِهِم.</span>
</div>
