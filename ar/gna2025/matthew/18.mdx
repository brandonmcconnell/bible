## الأعظم في ملكوت السماوات

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 9:33–37; لُوقا 9:46–48; ﻣﺘﻰ 20:20–28"> ودَنا التَّلاميذُ في ذلِكَ الوَقتِ إلى يَسوعَ وسألوهُ: ”مَنْ هوَ الأعظَمُ في مَلكوتِ السَّماواتِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 19:13–14; مَرْقُس 9:36–37; المُلُوكِ الأوَّلُ 3:7"> فدَعا يَسوعُ طِفلًا وأقامَهُ في وسَطِهِم</span>
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 10:14–15; رسالَةُ بُطرُسَ الأُولَى 2:2; ﻣﺘﻰ 19:14"> وقالَ: ”الحقَّ أقولُ لكُم: إنْ كُنتُم لا تَتَغيَّرونَ وتَصيرونَ مِثلَ الأطفالِ، فلن تَدخُلوا مَلكوتَ السَّماواتِ.</span>
<span className="v" id="v4" data-v="4" data-xref="يَعقُوب 4:10; لُوقا 14:11; مَرْقُس 10:43"> مَن اَتَّضعَ وصارَ مِثلَ هذا الطِّفلِ، فهوَ الأعظمُ في مَلكوتِ السَّماواتِ.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 25:40; مَرْقُس 9:37; يُوحَنّا 13:20"> ومَنْ قَبِلَ طِفلًا مِثلَهُ باَسمي يكونُ قبِلَني.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 9:42; لُوقا 17:1–2; ﻣﺘﻰ 18:10"> مَنْ أوقعَ أحَدَ هؤُلاءِ الصِّغارِ المؤمنينَ بي في الخَطيئةِ، فخَيرٌ لَه أنْ يُعلَّقَ في عُنُقهِ حجَرُ طَحْنٍ كبيرٍ ويُرمى في أعماقِ البحرِ.</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 17:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:19; ﻣﺘﻰ 26:24"> الويلُ لِلعالَمِ ممَّا يُوقِعُ النَّاسَ في الخطيئةِ! ولا بُدَّ أنْ يَحدُثَ ما يُوقِعُ في الخَطيئةِ، ولكنَّ الويلَ لمَنْ يُسَبِّبُ حُدوثَهُ!</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="ﻣﺘﻰ 5:29–30; فِيلِبِّي 3:8–9; ﺭﻭﻣﻴﺔ 13:12"> فإذا أوقَعَتكَ يَدُكَ أو رِجلُكَ في الخَطيئةِ، فاَقْطَعْها وألْقِها عَنكَ، لأنَّهُ خَيرٌ لكَ أنْ تَدخُلَ الحياةَ الأبديَّةَ ولكَ يدٌ أو رِجلٌ واحدةٌ، مِنْ أنْ يكونَ لكَ يَدانِ ورِجلانِ وتُلقى في النَّارِ الأبديَّةِ.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 14:22; ﻣﺘﻰ 18:8; ﻣﺘﻰ 5:29"> وإذا أوقَعَتكَ عَينُكَ في الخَطيئةِ، فاَقلَعْها وألقِها عَنكَ، لأنَّهُ خَيرٌ لكَ أنْ تَدخُلَ الحياةَ الأبديَّةَ ولكَ عَينٌ واحدةٌ، مِنْ أنْ يكونَ لكَ عَينانِ وتُلقى في نارِ جَهنَّمَ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 91:11; المَزامِير 34:7; العِبرانِيّين 1:14"> إيَّاكُم أنْ تَحتقروا أحدًا مِنْ هَؤلاءِ الصِّغارِ. أقولُ لكُم: إنَّ ملائِكَتَهُم في السَّماواتِ يُشاهِدونَ كُلَّ حِينٍ وجهَ أبي الّذي في السَّماواتِ.</span>
<span className="v" id="v11" data-v="11"> &#91;فاَبنُ الإنسانِ جاءَ ليُخلِّصَ الهالِكينَ&#93;.</span>
</div>
</div>

## الخروف الضال

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="حِزْقِيال 34:12; لُوقا 15:4–7; حِزْقِيال 34:16"> ”وما قولُكُم؟ إنْ كانَ لِرجلٍ مِئَةُ خَروفٍ وضَلَّ واحدٌ مِنها، ألا يَترُكُ التِّسعةَ والتِّسعينَ في الجِبالِ ويَبحَثُ عَن الخَروفِ الضَّالِّ؟</span>
<span className="v" id="v13" data-v="13" data-xref="صَفَنْيا 3:17; لُوقا 15:5–10; يَعقُوب 2:13"> وإذا وجَدَهُ، ألا يَفرَحُ بِه؟ الحقَّ أقولُ لكُم: إنَّهُ يَفرَحُ بِه أكثرَ مِنْ فَرَحِهِ بالتِّسعةِ والتِّسعينَ الَّتي ما ضَلَّت.</span>
<span className="v" id="v14" data-v="14" data-xref="يُوحَنّا 17:12; يُوحَنّا 6:39–40; رسالَةُ بُطرُسَ الثّانيَة 3:9"> وهكذا لا يُريدُ أبوكُمُ الّذي في السَّماواتِ أنْ يَهلِكَ واحدٌ مِنْ هَؤلاءِ الصِّغارِ.</span>
</div>
</div>

## التسامح الأخوي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="اللّاويِّين 19:17; لُوقا 17:3–4; غَلاطِيَّة 6:1"> ”إذا خَطِئَ أخوكَ إليكَ، فاَذهَبْ إلَيهِ وعاتِبْهُ بَينَكَ وبَينَهُ، فإذا سَمِعَ لكَ تكونُ رَبِحتَ أخاكَ.</span>
<span className="v" id="v16" data-v="16" data-xref="التَّثْنِيَة 19:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 13:1; يُوحَنّا 8:17"> وإنْ رَفَض أنْ يَسمعَ لكَ، فَخُذْ معَكَ رَجُلًا أو رَجُلينِ، حتّى تُثْبِتَ كُلَّ شيءٍ بِشَهادَةِ شاهِدَينِ أو ثلاثَةٍ.</span>
<span className="v" id="v17" data-v="17" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:9–7; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:14–15"> فإنْ رَفَضَ أنْ يَسمَعَ لهُم، فقُلْ لِلكنيسةِ، وإنْ رَفَضَ أنْ يَسمَعَ لِلكنيسةِ، فَعامِلْهُ كأنَّهُ وثَنيٌّ أو جابي ضرائبَ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 16:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:10; يُوحَنّا 20:23"> الحقَّ أقولُ لكُم: ما تَرْبُطونَهُ في الأرضِ يكونُ مَربوطًا في السَّماءِ، وما تَحُلُّونَهُ في الأرضِ يكونُ مَحلولًا في السَّماءِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 7:7; رسالَةُ يُوحَنّا الأُولَى 5:14–16; مَرْقُس 11:24"> الحقَّ أقولُ لكُم: إذا اَتَّفَقَ اَثنانِ مِنكُم في الأرضِ أنْ يَطلُبا حاجةً، حَصَلا علَيها مِنْ أبي الّذي في السَّماواتِ.</span>
<span className="v" id="v20" data-v="20" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:4; رُؤيا يُوحَنّا 21:3; ﻣﺘﻰ 28:20"> فأينَما اَجتمعَ اَثنانِ أو ثلاثَةٌ باَسمي، كُنتُ هُناكَ بَينَهُم“.</span>
</div>
</div>

## مثل العبد الّذي لا يغفر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="لُوقا 17:3–4; ﻣﺘﻰ 18:15"> فدَنا بُطرُسُ وقالَ لِيَسوعَ: ”يا ربُّ، كَم مرَّةً يَخطَأُ إليَّ أخي وأَغفِرُ لَهُ؟ أسبعَ مَرَّاتٍ؟“</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="كُولُوسِي 3:13; مَرْقُس 11:25; ﻣﺘﻰ 6:14–15"> فأجابَهُ يَسوعُ: ”لا سَبعَ مرَّاتٍ، بل سَبعينَ مرَّةً سبعَ مرَّاتٍ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 13:24; ﺭﻭﻣﻴﺔ 14:12; ﻣﺘﻰ 3:2"> فمَلكوتُ السَّماواتِ يُشبِهُ مَلِكًا أرادَ أنْ يُحاسِبَ عَبيدَهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="المَزامِير 38:4; المَزامِير 130:3–4; لُوقا 16:5"> فلمَّا بَدَأَ يُحاسِبُهُم، جِيءَ إلَيهِ بِواحدٍ مِنهُم علَيهِ عَشَرةُ آلافِ دِرهَمٍ مِنَ الفِضَّةِ.</span>
<span className="v" id="v25" data-v="25" data-xref="المُلُوكِ الثّانِي 4:1; نَحَمْيا 5:5; لُوقا 7:42"> وكانَ لا يَملِكُ ما يُوفي، فأمَرَ سيِّدُهُ بأنْ يُباعَ هوَ واَمرأتُهُ وأولادُهُ وجميعُ ما يَملِكُ حتّى يُوفيَهُ دَينَهُ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 18:29; لُوقا 7:43; ﻣﺘﻰ 8:2"> فركَعَ العبدُ لَه ساجدًا وقالَ: أمهِلني فأُوفيَكَ كُلَّ ما لك علَيَّ!</span>
<span className="v" id="v27" data-v="27" data-xref="المَزامِير 145:8; نَحَمْيا 9:17; المَزامِير 86:15"> فأشفَقَ علَيهِ سيِّدُهُ وأطلَقَهُ وأعفاهُ مِنَ الدَّينِ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻣﺘﻰ 20:2; حِزْقِيال 45:9; التَّثْنِيَة 15:2"> ولمَّا خرَجَ الرَّجلُ لَقِيَ عَبْدًا مِنْ أصحابِه كانَ لَه علَيهِ مئةُ دينارٍ، فأمسكَهُ بِعُنُقِهِ حتّى كادَ يَخنُقُهُ وهوَ يقولُ لَه: أوفِني ما لي علَيكَ!</span>
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 18:26; ﻣﺘﻰ 6:12; فِلِيْمُون 1:18–19"> فركَعَ صاحِبُهُ يَرجوهُ ويقولُ: أمْهِلْني، فأوفيكَ.</span>
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الأوَّلُ 22:27; المُلُوكِ الأوَّلُ 21:27–29"> فما أرادَ، بل أخَذَهُ وألقاهُ في السِّجنِ حتّى يُوفيَهُ الدَّينَ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="ﺭﻭﻣﻴﺔ 12:15; العِبرانِيّين 13:3; مَرْقُس 3:5"> ورأى العَبيدُ أصحابُهُ ما جرى، فاَستاؤوا كثيرًا وذَهَبوا وأَخْبَروا سيِّدَهُم بِكُلِّ ما جرى.</span>
<span className="v" id="v32" data-v="32" data-xref="لُوقا 19:22; ﻣﺘﻰ 25:26; ﺭﻭﻣﻴﺔ 3:19"> فدَعاهُ سيِّدُهُ وقالَ لَه: يا عَبدَ السُوءِ! أعْفَيْتُكَ مِنْ دَينِكَ كُلِّهِ، لأنَّكَ رَجَوْتَني.</span>
<span className="v" id="v33" data-v="33" data-xref="كُولُوسِي 3:13; ﻣﺘﻰ 6:12; أفَسُس 4:32–2"> أفما كانَ يَجبُ علَيكَ أنْ تَرحَمَ صاحِبَكَ مِثلَما رحَمتُكَ؟</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 5:25–26; يَعقُوب 2:13; لُوقا 12:58–59"> وغَضِبَ سيِّدُهُ كثيرًا، فسَلَّمَهُ إلى الجَلَّادينَ حتّى يُوفِيَهُ كُلَّ ما لَه علَيهِ.</span>
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 6:14–15; يَعقُوب 2:13; لُوقا 6:37–38"> هكذا يَفعَلُ بِكُم أبي السَّماويُّ إنْ كانَ كُلُّ واحدٍ مِنكُم لا يَغفِرُ لأخيهِ مِنْ كُلِّ قَلبِهِ“.</span>
</div>
