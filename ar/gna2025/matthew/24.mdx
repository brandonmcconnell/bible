## يسوع يُنبئ بخراب الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 13:1–37; لُوقا 21:5–36; حِزْقِيال 8:6"> وخرَجَ يَسوعُ مِنَ الهَيكَلِ. وبَينَما هوَ يَبتَعِدُ عَنهُ، دَنا إلَيهِ تلاميذُهُ يُوجِّهونَ نظَرَهُ إلى أَبنِيةِ الهَيكَلِ.</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 19:44; إرْمِيا 26:18; ميخا 3:12"> فقَالَ لهُم: ”أتَرَونَ هذا كُلَّهُ؟ الحقَّ أقولُ لكُم: لن يُترَكَ هُنا حجَرٌ على حجَرٍ، بل يُهدَمُ كُلُّهُ“.</span>
</div>
</div>

## الاضطراب والاضطهاد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 13:3–4; اعمال الرسل 1:7; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:1–11"> وبَينَما يَسوعُ جالسٌ في جبَلِ الزَّيتونِ، سألَهُ تلاميذُهُ على اَنفِرادٍ: ”أخبِرْنا متى يَحدُثُ هذا الخَرابُ، وما هِيَ علامَةُ مَجيئِكَ واَنقِضاءِ الدَّهرِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="رسالَةُ يُوحَنّا الأُولَى 4:1; إرْمِيا 29:8; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:3"> فأجابَهُم يَسوعُ: ”إِنتَبِهوا لِئلَّا يُضلِّلَكُم أحدٌ.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 24:11; ﻣﺘﻰ 24:24; إرْمِيا 14:14"> سيَجيءُ كثيرٌ مِنَ النَّاسِ مُنتَحِلينَ اَسمي، فيقولونَ: أنا هوَ المَسيحُ! ويَخدعونَ كثيرًا مِنَ النَّاسِ.</span>
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 13:7–8; لُوقا 21:9; رسالَةُ بُطرُسَ الأُولَى 3:14–15"> وستَسمَعونَ بالحُروبِ وبِأخبارِ الحُروبِ، فإيَّاكُم أن تَفزَعوا. فهذا لا بُدَّ مِنهُ، ولكنَّها لا تكونُ هيَ الآخِرةَ.</span>
<span className="v" id="v7" data-v="7" data-xref="لُوقا 21:11; رُؤيا يُوحَنّا 6:8; إشَعْياء 19:2"> ستَقومُ أُمَّةٌ على أُمَّةٍ، ومَملَكَةٌ على مملَكَةٍ، وتَحدُثُ مجاعاتٌ وزَلازِلُ في أماكِنَ كثيرةٍ.</span>
<span className="v" id="v8" data-v="8" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:3; رسالَةُ بُطرُسَ الأُولَى 4:17–18; التَّثْنِيَة 28:59"> وهذا كُلُّهُ بَدءُ الأوجاعِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 16:2; يُوحَنّا 15:19–20; لُوقا 21:16–17"> وفي ذلِكَ الوَقتِ يُسلِّمونَكُم إلى العَذابِ ويَقتُلونَكُم. وتُبغِضُكُم جميعُ الأُممِ مِنْ أجلِ اسمي.</span>
<span className="v" id="v10" data-v="10" data-xref="ميخا 7:5–6; لُوقا 21:16; ﻣﺘﻰ 11:6"> ويَرتَدُّ عَنِ الإيمانِ كثيرٌ مِن النَّاسِ، ويَخونُ بعضُهُم بَعضًا ويُبغِضُ واحدُهُم الآخَرَ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 24:24; ﻣﺘﻰ 7:15; ﻣﺘﻰ 24:5"> ويَظهَرُ أنبياءُ كذَّابونَ كثيرونَ ويُضلِّلونَ كثيرًا مِنَ النَّاسِ.</span>
<span className="v" id="v12" data-v="12" data-xref="رُؤيا يُوحَنّا 2:4–5; يَعقُوب 4:1–4; رُؤيا يُوحَنّا 3:15"> ويَعُمُّ الفَسادُ، فَتبرُدُ المَحبَّةُ في أكثرِ القُلوبِ.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 10:22; مَرْقُس 13:13; رُؤيا يُوحَنّا 2:10"> ومَنْ يَثبُتْ إلى النِّهايَةِ يَخلُصْ.</span>
<span className="v" id="v14" data-v="14" data-xref="رُؤيا يُوحَنّا 14:6; ﺭﻭﻣﻴﺔ 10:18; ﺭﻭﻣﻴﺔ 16:25–26"> وتَجيءُ النِّهايةُ بَعدَما تُعلَنُ بِشارةُ مَلكوتِ اللهِ هذِهِ في العالَمِ كُلِّهِ، شَهادةً لي عِندَ الأُمَمِ كُلِّها.</span>
</div>
</div>

## الخراب العظيم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="دانيال 9:27; دانيال 12:11; مَرْقُس 13:14"> ”فإذا رَأيتُم ”نَجاسَةَ الخَرابِ“ الّتي تكلَّمَ علَيها النَّبيُّ دانيالُ قائِمةً في المكانِ المُقَدَّسِ (إفهَمْ هذا أيُّها القارئُ)،</span>
<span className="v" id="v16" data-v="16" data-xref="التَّكوين 19:15–17; لُوقا 21:21–22; العِبرانِيّين 11:7"> فَلْيَهرُبْ إلى الجِبالِ مَنْ كانَ في اليهودِيَّةِ.</span>
<span className="v" id="v17" data-v="17" data-xref="لُوقا 17:31–33; مَرْقُس 13:15–16; ﻣﺘﻰ 10:27"> ومَنْ كانَ على السَّطحِ، فلا يَنزِلْ لِيأخُذَ مِنَ البَيتِ حوائجَهُ.</span>
<span className="v" id="v18" data-v="18"> ومَنْ كانَ في الحَقْلِ فلا يَرجِعْ ليَأخُذَ ثَوبَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 23:29–30; التَّثْنِيَة 28:53–56; لُوقا 21:23"> الوَيلُ لِلحَبالى والمُرضِعاتِ في تِلكَ الأيّامِ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 16:29; اعمال الرسل 1:12"> صَلُّوا لِئلَّا يكونَ هرَبُكُم في الشِّتاءِ أو في السَّبتِ.</span>
<span className="v" id="v21" data-v="21" data-xref="دانيال 12:1; يُوئيل 2:2; مَلاخِي 4:1"> فسَتَنزِلُ في ذلِكَ الوقتِ نكبةٌ ما حدَثَ مِثلُها مُنذُ بَدءِ العالَمِ إلى اليومِ، ولن يَحدُثَ.</span>
<span className="v" id="v22" data-v="22" data-xref="مَرْقُس 13:20; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:10; ﻣﺘﻰ 24:31"> ولولا أنَّ اللهَ جعَلَ تِلكَ الأيّامَ قَصيرةً، لَما نَجا أحدٌ مِنَ البشَرِ. ولكِنْ مِنْ أجلِ الّذينَ اَختارَهُم جعَلَ تِلكَ الأيّامَ قصيرةً.</span>
<span className="v" id="v23" data-v="23" data-xref="لُوقا 21:8; لُوقا 17:23–24; التَّثْنِيَة 13:1–3"> فإذا قالَ لكُم أحدٌ: ها هوَ المَسيحُ هُنا، أو ها هوَ هُناكَ! فلا تُصدِّقوهُ،</span>
<span className="v" id="v24" data-v="24" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:9–11; مَرْقُس 13:22; رُؤيا يُوحَنّا 13:13–14"> فسيَظهرُ مُسَحاءُ دجّالونَ وأنبياءُ كذّابونَ، يَصنَعونَ الآياتِ والعَجائبَ العَظيمةَ ليُضَلِّلوا، إنْ أمكَنَ، حتّى الّذينَ اَختارَهُمُ اللهُ.</span>
<span className="v" id="v25" data-v="25" data-xref="يُوحَنّا 16:1; لُوقا 21:13; إشَعْياء 48:5–6"> ها أنا أُنذِرُكُم.</span>
<span className="v" id="v26" data-v="26" data-xref="لُوقا 3:2–3; إشَعْياء 40:3; ﻣﺘﻰ 3:1"> فإنْ قالوا لكُم: ها هوَ في البرِّيَّةِ! فلا تَخرُجوا إلى هُناكَ، أو ها هوَ في داخِلِ البُيوتِ! فلا تُصدِّقوا،</span>
<span className="v" id="v27" data-v="27" data-xref="لُوقا 17:24–37; زَكَريّا 9:14; أيُّوب 37:3"> لأنَّ مجيءَ اَبنِ الإنسانِ يكونُ مِثلَ البَرقِ الّذي يلمَعُ مِنَ المشرقِ ويُضيءُ في المغرِبِ.</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 17:37; أيُّوب 39:27–30; التَّثْنِيَة 28:49"> وحَيثُ تكونُ الجِيفَةُ تَجتَمِعُ النُّسورُ.</span>
</div>
</div>

<div className="bt" dir="rtl">
<div className="bp">

<span className="v" id="v30" data-v="30" data-xref="رُؤيا يُوحَنّا 1:7; دانيال 7:13; لُوقا 21:27"> وتَظهَرُ في ذلِكَ الحينِ علامةُ اَبنِ الإنسانِ في السَّماءِ، فتَنتَحِبُ جميعُ قبائِلِ الأرضِ، ويَرى النَّاسُ اَبنَ الإنسانِ آتيًا على سَحابِ السَّماءِ في كُلِّ عِزَّةٍ وجلالٍ.</span>
<span className="v" id="v31" data-v="31" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:16; مَرْقُس 13:27; إشَعْياء 27:13"> فيُرسِلُ ملائِكَتَهُ بِبوقٍ عَظيمِ الصَوتِ إلى جِهاتِ الرّياحِ الأربعِ ليجمَعوا مُختاريهِ مِنْ أقصى السَّماواتِ إلى أقصاها.</span>

<div className="bp">

<span className="v" id="v33" data-v="33" data-xref="يَعقُوب 5:9; العِبرانِيّين 10:37; حِزْقِيال 7:2–14"> وكذلِكَ إذا رأيتُم هذا كُلَّهُ، فاَعلَموا أنَّ الوقتَ قريبٌ على الأبوابِ.</span>
<span className="v" id="v34" data-v="34" data-xref="مَرْقُس 13:30–31; لُوقا 21:32–33; ﻣﺘﻰ 16:28"> الحقَّ أقولُ لكُم: لن ينقضِيَ هذا الجِيلُ حتّى يتِمَّ هذا كُلُّهُ.</span>
<span className="v" id="v35" data-v="35" data-xref="إشَعْياء 40:8; رسالَةُ بُطرُسَ الأُولَى 1:25; العَدَد 23:19"> السَّماءُ والأرضُ تَزولانِ وكلامي لنْ يَزولَ.</span>

## السهر الدائم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="مَرْقُس 13:32; اعمال الرسل 1:7; ﻣﺘﻰ 24:44"> ”أمَّا ذلِكَ اليومُ وتِلكَ السَّاعةُ فلا يَعرِفُهُما أحدٌ، لا ملائِكةُ السَّماواتِ ولا الاَبنُ، إلَّا الآبُ وحدَهُ.</span>
<span className="v" id="v37" data-v="37" data-xref="لُوقا 17:26–27; التَّكوين 6:1–7; رسالَةُ بُطرُسَ الثّانيَة 2:5"> وكما حدَثَ في أيّامِ نوحٍ فكذلِكَ يَحدُثُ عِندَ مجيءِ اَبنِ الإنسانِ.</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 17:26–28; ﺭﻭﻣﻴﺔ 13:13–14; لُوقا 14:18–20"> كانَ النَّاسُ في الأيّامِ الّتي سَبَقتِ الطُوفانَ يأكُلونَ ويَشرَبونَ ويَتَزاوَجونَ، إلى يومِ دخَلَ نوحٌ الفُلكَ.</span>
<span className="v" id="v39" data-v="39" data-xref="يُوحَنّا 3:20; ﻣﺘﻰ 24:37; اعمال الرسل 13:41"> وما كانوا ينتَظِرونَ شيئًا، حتّى جاءَ الطُوفانُ فأغرَقهُم كُلَّهُم. وهكذا يَحدُثُ عِندَ مَجيءِ اَبنِ الإنسانِ:</span>
<span className="v" id="v40" data-v="40" data-xref="لُوقا 17:34–37; رسالَةُ بُطرُسَ الثّانيَة 2:7–9; رسالَةُ بُطرُسَ الثّانيَة 2:5"> فيكونُ رَجُلانِ في الحقلِ، فيُؤخذُ أحدُهُما ويُترَكُ الآخَرُ.</span>
<span className="v" id="v41" data-v="41" data-xref="الخُرُوج 11:5; إشَعْياء 47:2; لُوقا 17:35"> وتكونُ اَمرأتانِ على حجَرِ الطحنِ، فتُؤخَذُ إحداهُما وتُتركُ الأُخرَى.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="لُوقا 21:36; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:6; ﻣﺘﻰ 25:13"> فاَسهَروا، لأنَّكُم لا تَعرِفونَ أيَّ يومٍ يَجيءُ رَبُّكُم.</span>
<span className="v" id="v43" data-v="43" data-xref="لُوقا 12:39; ﻣﺘﻰ 24:44; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:2–6"> واَعلَموا أنَّ رَبَّ البَيتِ لو عَرَفَ في أيَّةِ ساعَةٍ مِنَ اللَّيلِ يَجيءُ اللِّصُّ، لسَهِرَ وما تَركَهُ يَنقُبُ بَيتَهُ.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 25:10; ﻣﺘﻰ 24:42; ﻣﺘﻰ 25:13"> فكونوا أنتُم أيضًا على اَستِعدادٍ، لأنَّ اَبنَ الإنسانِ يَجيءُ في ساعةٍ لا تَنتَظِرونَها.</span>
</div>

<div className="bp">

<span className="v" id="v46" data-v="46" data-xref="رُؤيا يُوحَنّا 16:15; لُوقا 12:43; لُوقا 12:37"> هنيئًا لذلِكَ الخادِمِ الّذي يَجدُهُ سيِّدُهُ عِندَ عَودَتِهِ يقومُ بِعَمَلِهِ هذا.</span>
<span className="v" id="v47" data-v="47" data-xref="ﻣﺘﻰ 25:23; ﻣﺘﻰ 25:21; لُوقا 12:37"> الحقَّ أقولُ لكُم: إنَّه يُوكِلُ إلَيهِ جميعَ أموالِهِ.</span>
