## المجوس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ميخا 5:2; يُوحَنّا 7:42; لُوقا 2:4–7"> ولمَّا وُلِدَ يَسوعُ في بَيتَ لَحْمِ اليَهودِيَّةِ، على عَهْدِ المَلِكِ هِيرودُسَ، جاءَ إلى أُورُشليمَ مَجوسٌ مِنَ المَشرقِ</span>
<span className="v" id="v2" data-v="2" data-xref="العَدَد 24:17; إرْمِيا 23:5; رُؤيا يُوحَنّا 22:16"> وقالوا: ”أينَ هوَ المَولودُ، مَلِكُ اليَهودِ؟ رَأَيْنا نَجْمَهُ في المَشْرِقِ فَجِئْنا لِنَسْجُدَ لَه“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 17:6–7; يُوحَنّا 11:47–48; ﻣﺘﻰ 23:37"> وسَمِعَ المَلِكُ هِيرودُسُ، فاَضْطَرَبَ هوَ وكُلُّ أُورُشليمَ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻣﺘﻰ 21:23; ﻣﺘﻰ 7:29; اعمال الرسل 6:12"> فجَمَعَ كُلَّ رُؤساءِ الكَهَنةِ ومُعَلِّمي الشَّعْبِ وسألَهُم: ”أينَ يولَدُ المَسيحُ؟“</span>
<span className="v" id="v5" data-v="5" data-xref="يُوحَنّا 7:42; راعُوث 1:1; صَمُوئيلَ الأوَّلُ 16:1"> فأجابوا: ”في بَيتَ لَحْمِ اليَهودِيَّةِ، لأنَّ هذا ما كَتَبَ النَبِيُّ:</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="حِزْقِيال 34:23–25; ميخا 5:2; إشَعْياء 9:6–7"> ”يا بَيتَ لَحْمُ، أرضَ يَهوذا،<br />ما أنتِ الصُّغْرى في مُدُنِ يَهوذا،<br />لأنَّ مِنكِ يَخْرُجُ رَئيسٌ<br />يَرعى شَعْبي إِسرائيلَ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 1:10; صَمُوئيلَ الأوَّلُ 18:21; المَزامِير 55:21"> فَدَعا هيرودُسُ المَجوسَ سِرًّا وتَحقَّقَ مِنْهُم مَتى ظَهَرَ النَّجْمُ،</span>
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:19–20; إرْمِيا 41:5–7; لُوقا 20:20–21"> ثُمَّ أرسَلَهُم إلى بَيتَ لَحْمَ وقالَ لَهُم: ”اَذْهَبوا واَبْحَثوا جيِّدًا عَنِ الطِّفلِ. فإذا وجَدْتُموهُ، فأَخْبِروني حتّى أذهَبَ أنا أيضًا وأسْجُدَ لَه“.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="ﻣﺘﻰ 2:2; الأمثال 8:17; المَزامِير 25:12"> فلمَّا سَمِعوا كلامَ المَلِكِ اَنْصَرَفوا. وبَينَما هُمْ في الطَّريقِ إذا النَّجْمُ الّذي رَأَوْهُ في المَشْرقِ، يَتَقَدَّمُهُمْ حتّى بَلَغَ المكانَ الّذي فيهِ الطِّفلُ فوَقَفَ فَوْقَه.</span>
<span className="v" id="v10" data-v="10" data-xref="المَزامِير 105:3; لُوقا 2:20; لُوقا 2:10"> فلمَّا رَأوا النَّجْمَ فَرِحوا فَرَحًا عَظيمًا جِدًّا،</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 60:6; المَزامِير 72:10; لُوقا 2:16"> ودَخَلوا البَيتَ فوَجَدوا الطِّفْلَ معَ أُمِّهِ مَرْيَمَ. فرَكَعوا وسَجَدوا لَه، ثُمَّ فَتَحوا أَكْياسَهُمْ وأهْدَوْا إلَيهِ ذَهَبًا وبَخورًا ومُرًّا.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 2:22; ﻣﺘﻰ 2:19; ﻣﺘﻰ 2:13"> وأنْذَرَهُمُ اللهُ في الحُلُمِ أنْ لا يَرجِعوا إلى هيرودُسَ، فأخَذوا طَريقًا آخَرَ إلى بِلادِهِم.</span>
</div>
</div>

## الهرب إلى مصر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="اعمال الرسل 10:7; اعمال الرسل 5:19; ﻣﺘﻰ 1:20"> وبَعدَما اَنْصرَفَ المَجوسُ، ظَهَرَ مَلاكُ الرَّبِّ لِيوسفَ في الحُلُمِ وقالَ لَه: ”قُمْ، خُذِ الطِّفْلَ وأُمَّهُ واَهربْ إلى مِصْرَ وأقِمْ فيها، حتّى أقولَ لكَ متى تَعودُ، لأنَّ هيرودُسَ سيَبحَثُ عَنِ الطِّفْلِ ليَقتُلَهُ“.</span>
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 2:20–21; اعمال الرسل 26:21; ﻣﺘﻰ 1:24"> فقامَ يوسفُ وأخذَ الطِّفْلَ وأُمَّهُ ليلًا ورحَلَ إلى مِصْرَ.</span>
<span className="v" id="v15" data-v="15" data-xref="هُوشَع 11:1; الخُرُوج 4:22; العَدَد 24:8"> فأقامَ فيها إلى أنْ ماتَ هيرودُسُ، ليتِمَّ ما قالَ الرّبُّ بِلسانِ النَّبيِّ: ”مِنْ مِصْرَ دَعَوْتُ اَبني“.</span>
</div>
</div>

## مقتل أولاد بيت لحم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 59:7; الأمثال 28:15; القُضاة 16:10"> فَلمَّا رَأى هيرودُسُ أنَّ المَجوسَ اَستهزَأوا بِه، غَضِبَ جدًّا وأمرَ بقَتلِ كُلِّ طِفْلٍ في بَيتَ لحمَ وجِوارِها، مِنِ اَبنِ سَنَتَينِ فَما دونَ ذلِكَ، حسَبَ الوَقتِ الّذي تحقَّقَهُ مِنَ المَجوسِ،</span>
<span className="v" id="v17" data-v="17"> فتَمَّ ما قالَ النَّبيُّ إرْميا:</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 31:15; إرْمِيا 9:17–21; إرْمِيا 4:31"> ”صُراخٌ سُمِعَ في الرّامَةِ،<br />بُكاءٌ ونَحيبٌ كثيرٌ،<br />راحيلُ تَبكي على أولادِها<br />ولا تُريدُ أنْ تَتَعزّى،<br />لأنَّهُم زالوا عَنِ الوجودِ“.</span>

## الرجوع من مصر إلى الناصرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 1:20; دانيال 8:25; ﻣﺘﻰ 2:12–13"> ولمَّا ماتَ هِيرودُسُ ظهَرَ ملاكُ الرَّبِّ ليوسفَ في الحُلُمِ، وهوَ في مِصْرَ</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 4:19; المُلُوكِ الأوَّلُ 12:1–3; ﻣﺘﻰ 2:13"> وقالَ لَه: ”قُمْ، خُذِ الطِّفْلَ وأُمَّهُ واَرجِعْ إلى أرضِ إِسرائيلَ، لأنَّ الّذينَ أرادوا أنْ يَقتُلوهُ ماتوا“.</span>
<span className="v" id="v21" data-v="21" data-xref="العِبرانِيّين 11:8; التَّكوين 6:22"> فقامَ وأخَذَ الطِّفْلَ وأُمَّهُ ورَجَعَ إلى أرضِ إِسرائيلَ.</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 48:17–18; لُوقا 2:39; يُوحَنّا 7:52"> لكِنَّهُ سَمِعَ أنَّ أرخيلاوُسَ يَملِكُ على اليَهودِيَّةِ خلَفًا لأبيهِ هِيرودُسَ، فخافَ أن يذهَبَ إلَيها. فأَنذَرَهُ اللهُ في الحُلُمِ، فلَجأَ إلى الجَليلِ.</span>
<span className="v" id="v23" data-v="23" data-xref="القُضاة 13:5; يُوحَنّا 1:45–46; مَرْقُس 1:24"> وجاءَ إلى مدينةٍ اَسمُها النّاصِرَةُ فسكَنَ فيها، لِيَتمَّ ما قالَ الأنبياءُ: ”يُدعى ناصِريًّا“.</span>
</div>
