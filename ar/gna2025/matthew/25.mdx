## مثل الوزنات

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="مَرْقُس 13:34; لُوقا 19:12–27; رسالَةُ بُطرُسَ الأُولَى 4:9–11"> ”ويُشبِهُ مَلكوتُ السَّماواتِ رجُلًا أرادَ السَّفَرَ، فدَعا خدَمَهُ وسَلَّمَ إلَيهِم أموالَهُ،</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 18:24; لُوقا 19:13–14; لُوقا 12:48"> كُلُّ واحدٍ مِنهُم على قَدْرِ طاقَتِهِ. فأعطى الأوّلَ خمسَ وزَناتٍ مِنَ الفِضَّةِ، والثَّاني وزْنَتَينِ، والثَّالثَ وزنَةً واحدةً وسافرَ.</span>
<span className="v" id="v16" data-v="16" data-xref="رسالَةُ يُوحَنّا الثّالثَة 1:5–8; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:17–18; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:6"> فأسرَعَ الّذي أخذَ الوَزَناتِ الخَمسَ إلى المتُاجَرةِ بِها، فربِحَ خَمسَ وزَناتٍ.</span>
<span className="v" id="v17" data-v="17" data-xref="الأمثال 3:9–10; أفَسُس 5:16; رسالَةُ بُطرُسَ الأُولَى 4:10"> وكذلِكَ الّذي أخذَ الوَزْنَتينِ، فرَبِحَ وزْنَتينِ.</span>
<span className="v" id="v18" data-v="18" data-xref="رسالَةُ بُطرُسَ الثّانيَة 1:8; حَجَّي 1:2–4; مَلاخِي 1:10"> وأمَّا الّذي أخذَ الوَزْنَة الواحدَةَ، فذهَبَ وحفَرَ حُفْرةً في الأرضِ ودفَنَ مالَ سيِّدِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:12–15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:10; ﺭﻭﻣﻴﺔ 14:7–12"> وبَعدَ مُدّةٍ طويلةٍ، رجَعَ سيِّدُ هؤُلاءِ الخَدَمِ وحاسَبَهُم.</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 20:24; لُوقا 19:16–17; كُولُوسِي 1:29"> فجاءَ الّذي أخَذَ الوَزَناتِ الخَمسَ، فدَفَعَ خَمسَ وزَناتٍ مَعَها وقالَ: يا سيِّدي، أعطيتَني خَمسَ وَزَناتٍ، فخُذْ خَمسَ وزَناتٍ رَبِحتُها.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 25:23; لُوقا 16:10; يُوحَنّا 12:26"> فقَالَ لَه سيِّدُهُ: أحسَنتَ، أيُّها الخادمُ البارُّ الأمينُ! كُنتَ أمينًا على القليلِ، فسَأُقيمُكَ على الكَثيرِ: أُدخُلْ نَعيمَ سيِّدِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:7–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:1–3"> وجاءَ الّذي أخَذَ الوَزْنتَينِ، فقَالَ: يا سيِّدي، أعطَيتَني وزْنَتَينِ، فخُذْ معَهُما وزْنتَينِ رَبِحتُهُما.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 25:21; مَرْقُس 14:8–9; مَرْقُس 12:41–44"> فَقالَ لَه سيِّدُهُ: أحسنتَ، أيُّها الخادِمُ البارُّ الأمينُ! كُنتَ أمينًا على القَليلِ، فسأُقيمُكَ على الكَثيرِ: أُدخُلْ نَعيمَ سيِّدِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="لُوقا 19:20–22; لُوقا 15:29; مَلاخِي 3:14–15"> وجاءَ الّذي أخَذَ الوَزْنةَ الواحِدَةَ، فقَالَ: يا سيِّدُ، عَرَفْتُكَ رجُلًا قاسِيًا، تحصِدُ حيثُ لا تَزرَعُ، وتَجمَعُ حيث لا تَبذُرُ،</span>
<span className="v" id="v25" data-v="25" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:6–7; ﺭﻭﻣﻴﺔ 8:15; رُؤيا يُوحَنّا 21:8"> فخِفتُ. فذَهبتُ ودفَنْتُ مالَكَ في الأرضِ، وها هوَ مالُكَ.</span>
<span className="v" id="v26" data-v="26" data-xref="ﻣﺘﻰ 18:32; أيُّوب 15:5–6"> فأجابَهُ سيِّدُهُ: يا لَكَ من خادِمٍ شِرِّيرٍ كَسلانَ! عَرَفتَني أحصِدُ حَيثُ لا أزرَعُ وأجمَعُ حيثُ لا أبذُرُ،</span>
<span className="v" id="v27" data-v="27" data-xref="التَّثْنِيَة 23:19–20; ﺭﻭﻣﻴﺔ 3:19; يَهُوذا 1:15"> فكانَ علَيكَ أنْ تضَعَ مالي عِندَ الصَّيارِفَةِ، وكُنتُ في عَودتي أستَرِدُّهُ معَ الفائِدَةِ.</span>
<span className="v" id="v28" data-v="28" data-xref="لُوقا 19:24; لُوقا 10:42"> وقالَ لخَدَمِه: خُذوا مِنهُ الوَزْنَةَ واَدْفَعوها إلى صاحِبِ الوَزَناتِ العَشْرِ،</span>
<span className="v" id="v29" data-v="29" data-xref="لُوقا 8:18; مَرْقُس 4:25; ﻣﺘﻰ 13:12"> لأنَّ مَنْ كانَ لَه شيءٌ، يُزادُ فيَفيضُ. ومَنْ لا شيءَ لَه، يُؤخذُ مِنهُ حتّى الّذي لَه.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺘﻰ 8:12; رُؤيا يُوحَنّا 3:15–16; ﻣﺘﻰ 13:50"> وهذا الخادِمُ الّذي لا نَفْعَ مِنهُ، إِطرَحوهُ خارِجًا في الظّلامِ. فهُناكَ البُكاءُ وصَريفُ الأسنانِ.</span>
</div>

## يوم الدينونة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="رُؤيا يُوحَنّا 1:7; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 1:7–8; ﻣﺘﻰ 19:28"> ”ومَتى جاءَ اَبنُ الإنسانِ في مَجدِهِ، ومعَهُ جميعُ ملائِكَتِهِ يَجلِسُ على عرشِهِ المَجيدِ،</span>
<span className="v" id="v32" data-v="32" data-xref="مَلاخِي 3:18; ﻣﺘﻰ 13:49; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:10"> وتَحتَشِدُ أمامَهُ جميعُ الشُّعوبِ، فيُفرِزُ بَعضَهُم عَنْ بَعضٍ، مِثلَما يُفرِزُ الرّاعي الخِرافَ عَنِ الجداءِ،</span>
<span className="v" id="v33" data-v="33" data-xref="المَزامِير 100:3; يُوحَنّا 10:26–28; يُوحَنّا 21:15–17"> فيَجعَلُ الخِرافَ عَنْ يَمينِهِ والجِداءَ عن شِمالِه.</span>
<span className="v" id="v34" data-v="34" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:19–20; يَعقُوب 2:5; رُؤيا يُوحَنّا 21:7"> ويقولُ المَلِكُ للَّذينَ عن يَمينِهِ: تَعالَوا، يا مَنْ باركهُم أبي، رِثُوا المَلكوتَ الّذي هَيَّأهُ لكُم مُنذُ إنشاءِ العالَمِ،</span>
<span className="v" id="v35" data-v="35" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:16–19; يَعقُوب 2:15–16; التَّثْنِيَة 15:7–11"> لأنِّي جُعتُ فأطعَمتُموني، وعَطِشتُ فسَقَيْتُموني، وكُنتُ غَريبًا فآوَيْتُموني،</span>
<span className="v" id="v36" data-v="36" data-xref="يَعقُوب 1:27; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:16–18; يَعقُوب 5:14–15"> وعُريانًا فكَسَوْتُموني، ومَريضًا فَزُرتُموني، وسَجينًا فجِئتُم إليَّ.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻣﺘﻰ 6:3; أخبارِ الأيّامِ الأوّلُ 29:14; رسالَةُ بُطرُسَ الأُولَى 5:5–6"> فيُجيبُهُ الأَبرارُ: يا ربُّ، متى رأيناكَ جوعانَ فأطْعَمناكَ؟ أو عَطشانَ فَسَقيْناكَ؟</span>
<span className="v" id="v38" data-v="38"> ومتى رَأيناكَ غَريبًا فآويناكَ؟ أو عُريانًا فكَسوناكَ؟</span>

</div>

<span className="v" id="v43" data-v="43"> وكُنتُ غَريبًا فما آوَيْتموني، وعُريانًا فما كَسَوْتُموني، ومريضًا وسَجينًا فما زُرْتُموني.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 7:22; لُوقا 10:29; ﻣﺘﻰ 25:24–27"> فيُجيبُهُ هؤُلاءِ: يا ربُّ، متى رأيناكَ جوعانَ أو عَطشانَ، غريبًا أو عُريانًا، مريضًا أو سَجينًا، وما أسْعَفْناكَ؟</span>
