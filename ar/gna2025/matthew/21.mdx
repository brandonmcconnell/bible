## يسوع يدخل أورشليم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="زَكَريّا 14:4; مَرْقُس 11:1–10; لُوقا 21:37"> ولمَّا قَرُبوا مِنْ أُورُشليمَ ووَصَلوا إلى بَيتِ فاجي عِندَ جبَلِ الزَّيتونِ، أرسَلَ يَسوعُ اَثنَينِ مِنْ تلاميذِهِ،</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 19:30–32; مَرْقُس 11:2–3; يُوحَنّا 2:5–8"> وقالَ لهُما: ”إِذهَبا إلى القريةِ الّتي أمامكُما، تَجِدا أتانًا مربوطةً وجَحشُها مَعها، فحُلَّا رِباطَهُما وجيئا بِهِما إليَّ.</span>
<span className="v" id="v3" data-v="3" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:1–2; اعمال الرسل 17:25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:9"> وإنْ قالَ لكُما أحدٌ شيئًا، فأجيبا: ”الرّبُّ مُحتاجٌ إلَيهِما، وسيُعيدُهُما في الحالِ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="زَكَريّا 9:9; يُوحَنّا 12:15; ﻣﺘﻰ 1:22"> وكانَ هذا لِيَتِمَّ ما قالَ النَّبيُّ:</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="زَكَريّا 9:9–10; إشَعْياء 62:11; إشَعْياء 12:6"> ”قولوا لابنَةِ صِهيونَ:<br />ها هوَ مَلِكُكِ قادِمٌ إلَيكِ<br />وديعًا راكِبًا على أتانٍ<br />وجَحشٍ اَبنِ أتانٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 39:43; يُوحَنّا 15:14; الخُرُوج 40:16"> فذهَبَ التِّلميذانِ وفَعَلا ما أمرَهُما بِه يَسوعُ</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الثّانِي 9:13; مَرْقُس 11:4–8; لُوقا 19:32–35"> وجاءا بالأتانِ والجَحشِ. ثُمَّ وضَعا علَيهِما ثَوبَيْهِما، فركِبَ يَسوعُ.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الثّانِي 9:13; يُوحَنّا 12:13; اللّاويِّين 23:40"> وبَسَطَ كثيرٌ مِنَ النَّاسِ ثيابَهُم على الطَّريقِ، وقطَعَ آخَرونَ أغصانَ الشَّجرِ وفَرَشوا بِها الطَّريقَ.</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 118:24–26; لُوقا 2:14; يُوحَنّا 12:13–15"> وكانَتِ الجُموعُ الّتي تتَقَدَّمُ يَسوعَ والّتي تَتْبَعُهُ تَهتِفُ: ”المَجْدُ لاَبنِ داودَ! تبارَكَ الآتي بِاَسمِ الرَّبِّ! المجدُ في العُلى!“</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="اعمال الرسل 9:5; راعُوث 1:19; يُوحَنّا 12:16–19"> ولمَّا دخَلَ يَسوعُ أُورُشليمَ ضَجّتِ المدينةُ كُلُّها وسألَتْ: ”مَنْ هذا؟“</span>
<span className="v" id="v11" data-v="11" data-xref="يُوحَنّا 6:14; لُوقا 7:16; يُوحَنّا 7:40"> فأجابَتِ الجُموعُ: ”هذا هوَ النَّبيُّ يَسوعُ مِنْ ناصرةِ الجليلِ“.</span>
</div>
</div>

## يسوع يطرد الباعة من الهيكل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 2:14–17; لُوقا 19:45–47; مَرْقُس 11:15–18"> ودخَلَ يَسوعُ الهَيكلَ وطرَدَ جميعَ الّذينَ يَبيعونَ ويَشتَرونَ فيهِ، فقَلَبَ مناضِدَ الصَّيارِفَةِ ومَقاعِدَ باعَةِ الحَمامِ،</span>
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 7:11; إشَعْياء 56:7; لُوقا 19:46"> وقال لهُم: ”جاءَ في الكِتابِ: بَيتي بَيتُ الصَّلاةِ، وأنتُم جَعَلْتُموهُ مغارَةَ لُصوصٍ!“</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="اعمال الرسل 3:1–9; اعمال الرسل 10:38; ﻣﺘﻰ 11:4–5"> وجاءَ إلَيهِ العُرجُ والعُميانُ وهوَ في الهَيكلِ فشَفاهُم.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 21:9; لُوقا 19:39–40; إشَعْياء 26:11"> فغَضِبَ رُؤساءُ الكَهنَةِ ومُعلِّمو الشَّريعةِ عِندَما رَأَوا المُعْجزاتِ الّتي صَنَعَها، وغاظَهُم هُتافُ الأولادِ في الهَيكَلِ: ”المجدُ لاَبنِ داودَ!“</span>
<span className="v" id="v16" data-v="16" data-xref="المَزامِير 8:2; ﻣﺘﻰ 11:25; لُوقا 19:39–40"> فقَالوا لَه: ”أتَسمَعُ ما يَقولُ هؤُلاءِ؟“ فأجابَهُم: ”نعم، أما قرأتُم هذِهِ الآيةَ: مِنْ أفواهِ الصِّغارِ والأطفالِ أخرَجْتَ كلامَ الحمدِ؟“</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 11:18; مَرْقُس 11:11; لُوقا 24:50"> ثُمَّ تَركَهُم وخرَجَ مِنَ المدينةِ إلى بَيتِ عَنيا وباتَ فيها.</span>
</div>
</div>

## يسوع يلعن شجرة الّتين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 4:2; العِبرانِيّين 4:15; ﻣﺘﻰ 12:1"> وبَينَما هوَ راجِعٌ إلى المدينةِ في الصَّباحِ، أحَسَّ بالجوعِ</span>
<span className="v" id="v19" data-v="19" data-xref="يُوحَنّا 15:6; مَرْقُس 11:14; تِيطُس 1:16"> فَجاءَ إلى شَجرَةِ تِينٍ رَآها على جانِبِ الطَّريقِ، فما وجَدَ علَيها غَيرَ الوَرقِ. فقَالَ لها: ”لن تُثمِري إلى الأبدِ!“ فيَبِسَتِ التّينةُ في الحالِ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="إشَعْياء 40:6–8; مَرْقُس 11:20–21; يَعقُوب 1:10–11"> ورأى التَّلاميذُ ما جرى، فتَعجَّبوا وقالوا: ”كيفَ يَبِسَتِ التّينةُ في الحالِ؟“</span>
<span className="v" id="v21" data-v="21" data-xref="يَعقُوب 1:6; ﻣﺘﻰ 17:20; مَرْقُس 11:22–23"> فأجابَهُم يَسوعُ: ”الحقَّ أقولُ لكُم: لو كُنتُم تؤمنونَ ولا تَشُكُّونَ، لَفعَلْتُم بِهذِهِ التّينةِ مِثلَما فعَلتُ، لا بلْ كُنتُم إذا قُلتُم لِهذا الجبَلِ: قُمْ واَنطَرِحْ في البحرِ، يكونُ لكُم ذلِكَ.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 7:7; رسالَةُ يُوحَنّا الأُولَى 5:14–15; مَرْقُس 11:24"> فكُلُّ شيءٍ تَطلُبونَهُ وأنتُم تُصلُّونَ بإيمانٍ، تنالونَهُ“.</span>
</div>
</div>

## السلطة المعطاة ليسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="اعمال الرسل 4:7; مَرْقُس 11:27–33; أخبارِ الأيّامِ الأوّلُ 24:1–19"> ودخَلَ يَسوعُ الهَيكلَ. وبَينَما هوَ يُعَلِّمُ، جاءَ إلَيهِ رؤَساءُ الكَهنَةِ وشُيوخُ الشَّعبِ وقالوا لَه: ”بأيَّةِ سُلطَةٍ تَعمَلُ هذِهِ الأعمالَ؟ ومَنْ أعطاكَ هذِهِ السُّلطَةَ؟“</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="لُوقا 6:9; كُولُوسِي 4:6; ﻣﺘﻰ 10:16"> فأجابَهُم يَسوعُ: ”وأنا أسألُكُم سُؤالًا واحدًا، إن أجَبْتُموني عَنهُ، قُلْتُ لكُم بأيَّةِ سُلطَةٍ أعمَلُ هذِهِ الأعمالَ:</span>
<span className="v" id="v25" data-v="25" data-xref="رسالَةُ يُوحَنّا الأُولَى 3:20; مَرْقُس 1:1–11; يُوحَنّا 3:18"> مِنْ أينَ ليوحنَّا سُلطةُ المَعمودِيَّةِ؟ مِنَ السَّماءِ أمْ مِنَ النَّاسِ؟“<br />فقَالوا في أنفُسِهِم: ”إنْ قُلنا مِنَ اللهِ، يُجيبُنا: فلِماذا ما آمنتُم بِه؟</span>
<span className="v" id="v26" data-v="26" data-xref="مَرْقُس 6:20; ﻣﺘﻰ 14:5; يُوحَنّا 5:35"> وإنْ قُلنا مِنَ النَّاسِ، نَخافُ الشَّعبَ، لأنَّهُم كُلَّهم يَعُدُّون يوحنَّا نبيًّا“.</span>
<span className="v" id="v27" data-v="27" data-xref="ﺭﻭﻣﻴﺔ 1:18–22; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:9–10; لُوقا 20:7–8"> فأجابوا يَسوعَ: ”لا نَعرِفُ“. فقَالَ لهُم: ”وأنا لا أقولُ لكُم بأيَّةِ سُلطَةٍ أعمَلُ هذِهِ الأعمالَ“.</span>
</div>
</div>

## مثل الابنينِ

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:58; ﻣﺘﻰ 20:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:15"> وقالَ يَسوعُ: ”ما رأيُكُم؟ كانَ لِرَجُلٍ اَبنانِ. فجاءَ إلى الأوَّلِ وقالَ لَه: يا اَبني، اَذهَبِ اليومَ واَعمَلْ في كرمي.</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 18:28–32; ﻣﺘﻰ 3:2–8; إشَعْياء 55:6–7"> فأجابَهُ: لا أريدُ. ولكِنَّه نَدِمَ بَعدَ حينٍ وذهَبَ إلى الكَرمِ.</span>
<span className="v" id="v30" data-v="30" data-xref="تِيطُس 1:16; حِزْقِيال 33:31; ﻣﺘﻰ 23:3"> وجاءَ إلى الابنِ الآخَرِ وطَلَبَ مِنهُ ما طلَبَهُ مِنَ الأوَّلِ، فأجابَهُ: أنا ذاهِبٌ، يا سيِّدي! ولكِنَّه ما ذهَبَ.</span>
<span className="v" id="v31" data-v="31" data-xref="لُوقا 7:29; لُوقا 7:37–50; ﻣﺘﻰ 7:21"> فأيَّهُما عَمِلَ إرادةَ أبيهِ؟“ قالوا: ”الأوَّلُ“. فقَالَ لهُم يَسوعُ: ”الحقَّ أقولُ لكُم: جُباةُ الضَّرائبِ والزَّواني يَسبِقونكُم إلى مَلكوتِ اللهِ.</span>
<span className="v" id="v32" data-v="32" data-xref="لُوقا 7:37–50; زَكَريّا 7:11–12; لُوقا 7:29–30"> جاءَكُم يوحنَّا المَعمَدانُ سالِكًا طَريقَ الحَقِّ فما آمنتُم بِه وآمنَ بِه جُباةُ الضَّرائبِ والزَّواني. وأنتُم رأيتُم ذلِكَ، فما ندِمتُم ولو بَعدَ حينٍ فتُؤْمنوا بِه“.</span>
</div>
</div>

## مثل الكرَّامين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="إشَعْياء 5:1–4; مَرْقُس 12:1–12; لُوقا 20:9–19"> ”إسمَعوا مَثَلًا آخَرَ: غرَسَ رجُلٌ كرمًا، فسَيَّجَهُ وحفَرَ فيهِ مَعْصَرَةً وبَنى بُرجًا وسَلَّمَهُ إلى بَعضِ الكرَّامينَ وسافَرَ.</span>
<span className="v" id="v34" data-v="34" data-xref="لُوقا 20:10–19; إشَعْياء 5:4; زَكَريّا 1:3–6"> فلمَّا جاءَ يومُ القِطافِ، أرسَلَ خدَمَهُ إليهِم ليأخُذوا ثمَرَهُ.</span>
<span className="v" id="v35" data-v="35" data-xref="أخبارِ الأيّامِ الثّانِي 36:15–16; ﻣﺘﻰ 5:12; العِبرانِيّين 11:36–37"> فأمسَكَ الكرَّامونَ خدَمَهُ وضَرَبوا واحدًا منهُم، وقَتَلوا غَيرَهُ، ورَجَموا الآخَرَ.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 22:4"> فأرسَلَ صاحِبُ الكرمِ خَدَمًا غَيرَهُم أكثرَ عددًا مِنَ الأوَّلينَ، ففَعَلوا بِهِم ما فَعلوهُ بالأوَّلينَ.</span>
<span className="v" id="v37" data-v="37" data-xref="يُوحَنّا 3:16; يُوحَنّا 3:35–36; صَفَنْيا 3:7"> وفي آخِر الأمرِ أرسلَ إلَيهِم اَبنَهُ وقالَ: سَيَهابونَ اَبني.</span>
<span className="v" id="v38" data-v="38" data-xref="لُوقا 20:14; التَّكوين 37:18–20; مَرْقُس 12:7–8"> فلمَّا رأى الكرَّامونَ الاَبنَ قالوا في ما بَينَهُم: ها هوَ الوارِثُ! تعالَوْا نَقْتُلُه ونأخُذُ ميراثَهُ!</span>
<span className="v" id="v39" data-v="39" data-xref="اعمال الرسل 2:23; يَعقُوب 5:6; لُوقا 22:52–54"> فأمسكوهُ ورمَوْهُ في خارِجِ الكرمِ وقَتَلوهُ.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="العِبرانِيّين 10:29; مَرْقُس 12:9; لُوقا 20:15–16"> فماذا يفعَلُ صاحِبُ الكرمِ بِهؤلاءِ الكرَّامينَ عِندَ رُجوعِهِ؟“</span>
<span className="v" id="v41" data-v="41" data-xref="اعمال الرسل 18:6; ﻣﺘﻰ 21:43; اعمال الرسل 28:28"> قالوا لَه: ”يَقتُلُ هؤُلاءِ الأشرارَ قَتلًا ويُسلِّمُ الكرمَ إلى كرّامينَ آخرينَ يُعطونَهُ الثَّمرَ في حينِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="المَزامِير 118:22–23; إشَعْياء 28:16; اعمال الرسل 4:11"> فقَالَ لهُم يَسوعُ: ”أما قرأتُم في الكُتُبِ المُقَدَّسةِ: الحجَرُ الّذي رَفضَهُ البنّاؤونَ صارَ رأسَ الزّاويَةِ؟ هذا ما صنَعَهُ الرَّبُّ، فيا للْعجَبِ!</span>
</div>
<div className="bp">
<span className="v" id="v43" data-v="43" data-xref="ﻣﺘﻰ 21:41; ﻣﺘﻰ 8:11–12; رسالَةُ بُطرُسَ الأُولَى 2:9"> لذلِكَ أقولُ لكُم: سيأخُذُ اللهُ مَلكوتَهُ مِنكُم ويُسلِّمُهُ إلى شعبٍ يَجعلُهُ يُثمِرُ.</span>
<span className="v" id="v44" data-v="44" data-xref="إشَعْياء 8:14–15; رسالَةُ بُطرُسَ الأُولَى 2:8; دانيال 2:34–35"> (مَنْ وقَعَ على هذا الحَجَرِ تَهَشَّمَ. ومَنْ وقَعَ هذا الحجَرُ علَيهِ سَحقَهُ)“.</span>
</div>
