## التمسك الأعمى بالتقاليد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 3:22; ﻣﺘﻰ 23:2; لُوقا 5:21"> وأقبَلَ إلى يَسوعَ بَعضُ الفَرِّيسيِّينَ ومُعلِّمي الشَّريعةِ مِنْ أُورُشليمَ، فسألوهُ:</span>
<span className="v" id="v2" data-v="2" data-xref="لُوقا 11:38; مَرْقُس 7:2; كُولُوسِي 2:20–23"> ”لِماذا يُخالِفُ تلاميذُكَ تقاليدَ القُدَماءِ، فلا يَغسِلونَ أيديَهُم قَبلَ الطَّعامِ؟“</span>
<span className="v" id="v3" data-v="3" data-xref="كُولُوسِي 2:23; تِيطُس 1:14; كُولُوسِي 2:8"> فأجابَهُم يَسوعُ: ”ولِماذا تُخالِفونَ أنتُم وصيَّةَ اللهِ مِنْ أجلِ تَقاليدِكُم؟</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 20:12; اللّاويِّين 20:9; الخُرُوج 21:17"> قالَ اللهُ: أكرِمْ أباكَ وأمَّكَ، ومَنْ لعنَ أباهُ أو أمَّهُ فموتًا يَموتُ.</span>
<span className="v" id="v5" data-v="5" data-xref="مَرْقُس 7:10–13; اعمال الرسل 5:29; ﻣﺘﻰ 23:16–18"> وأمَّا أنتُمْ فتَقولونَ: مَنْ كانَ عِندَهُ ما يُساعِدُ بِه أباهُ أو أمَّهُ وقالَ لَهُما: هذا تقدمَةٌ للهِ،</span>
<span className="v" id="v6" data-v="6" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:8; إرْمِيا 8:8; مَرْقُس 7:13"> فلا يلزَمُهُ أنْ يُكرِمَ أباهُ. وهكذا أبطَلْتُم كلامَ اللهِ مِنْ أجلِ تقاليدِكُم.</span>
<span className="v" id="v7" data-v="7" data-xref="مَرْقُس 7:6; ﻣﺘﻰ 23:23–29; اعمال الرسل 28:25–27"> يا مُراؤون، صَدَقَ إشَعْيا في نُبوءتِهِ عنكُم حينَ قالَ:</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 29:13; حِزْقِيال 33:31; اعمال الرسل 8:21"> هذا الشَّعبُ يُكرمُني بِشَفَتيهِ،<br />وأمَّا قَلبُهُ فبعيدٌ عنِّي.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="إشَعْياء 29:13; مَرْقُس 7:7; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:4"> وهوَ باطِلًا يَعبُدُني<br />بِتعاليمَ وضَعَها البشرُ“.</span>
</div>
</div>

## ما ينجس الإنسان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="مَرْقُس 7:14–15; لُوقا 20:45–47; لُوقا 24:45"> ثُمَّ دَعا الجُموعَ وقالَ لهُم: ”إِسمَعوا واَفهَموا:</span>
<span className="v" id="v11" data-v="11" data-xref="مَرْقُس 7:15; ﻣﺘﻰ 15:18–20; يَعقُوب 3:5–8"> ما يَدخُلُ الفَمَ لا يُنجِّسُ الإنسانَ، بَلْ ما يَخرُجُ مِنَ الفمِ هوَ الّذي يُنَجِّسُ الإنسانَ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المُلُوكِ الأوَّلُ 22:13–14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:32–33; غَلاطِيَّة 2:5"> فتقَدَّمَ تلاميذُهُ وقالوا لَه: ”أتعرِفُ أنَّ الفَرِّيسيِّينَ اَستاؤوا عِندَما سَمِعوا كلامَكَ هذا؟</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 15:2; إشَعْياء 60:21; يُوحَنّا 15:6"> فأجابَهُم: ”كُلُّ غَرْسٍ لا يَغرِسُه أبي السَّماويُّ يُقلَعُ.</span>
<span className="v" id="v14" data-v="14" data-xref="لُوقا 6:39; إشَعْياء 56:10; ﻣﺘﻰ 23:16–24"> أُترُكوهُم! هُمْ عُميانٌ قادةُ عُميانٍِ. وإذا كانَ الأعمى يَقودُ الأعمى، سقَطا معًا في حُفرةٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 13:36; مَرْقُس 4:34; يُوحَنّا 16:29"> فقَالَ لَه بُطرُسُ: ”فَسِّرْ لنا هذا المَثَلَ“.</span>
<span className="v" id="v16" data-v="16" data-xref="ﻣﺘﻰ 16:9; مَرْقُس 7:18; مَرْقُس 9:32"> فأجابَ: ”أأنتُم حتّى الآنَ لا تَفْهَمونَ؟</span>
<span className="v" id="v17" data-v="17" data-xref="لُوقا 6:45; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:13; ﻣﺘﻰ 7:19–20"> ألا تَعرِفونَ أنَّ ما يَدخُلُ فمَ الإنسانِ يَنْزِلُ إلى الجوفِ، ومِنهُ إلى خارجِ الجسَدِ؟</span>
<span className="v" id="v18" data-v="18" data-xref="الأمثال 15:28; ﻣﺘﻰ 12:34; يَعقُوب 3:6–10"> وأمَّا ما يَخرُجُ مِنَ الفَمِ، فمِنَ القلبِ يَخرُجُ، وهوَ يُنجِّسُ الإنسانَ.</span>
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 7:21–23; غَلاطِيَّة 5:19–21; إرْمِيا 17:9"> لأنَّ مِنَ القَلبِ تَخرُجُ الأفكارُ الشِّرِّيرةُ: القَتلُ والزِّنى والفِسقُ والسَّرقَةُ وشَهادَةُ الزُّورِ والنَّميمةُ،</span>
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 7:3–4; أفَسُس 5:3–6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:18–20"> وهيَ الّتي تُنَجِّسُ الإنسانَ. أمَّا الأكلُ بأيدٍ غيرِ مَغسولةٍ، فلا يُنجِّسُ الإنسانَ“.</span>
</div>
</div>

## إيمان المرأة الكنعانية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 7:24–30; ﻣﺘﻰ 11:21–23; القُضاة 1:31"> وخرَجَ يَسوعُ مِنْ هُناكَ وجاءَ إلى نواحي صورَ وصيدا.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 9:27; مَرْقُس 7:25–26; ﻣﺘﻰ 17:15"> فأَقبلَتْ إلَيهِ اَمرأةٌ كَنْعانِيّةٌ مِنْ تِلكَ البلادِ وصاحَتِ: ”إِرْحَمني، يا ربُّ، يا اَبن داودَ! إِبنتي فيها شَيطانٌ، ويُعذِّبُها كثيرًا“.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 14:15; مَرْقُس 10:47–48; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:8"> فما أجابَها يَسوعُ بكَلِمَةٍ. فَدنا تلاميذُهُ وتَوَسَّلوا إلَيهِ بقولِهِم: ”إِصرِفْها عنَّا، لأنَّها تَتبَعُنا بِصياحِها!“</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 10:5–6; ﺭﻭﻣﻴﺔ 15:8; إرْمِيا 50:6–7"> فأجابَهُم يَسوعُ: ”ما أرسلَني اللهُ إلَّا إلى الخِرافِ الضَّالَّةِ مِنْ بَني إِسرائيلَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّكوين 32:26; ﻣﺘﻰ 20:31; ﻣﺘﻰ 14:33"> ولكنَّ المرأةَ جاءَتْ فسَجَدَتْ لَه وقالَت: ”ساعِدْني، يا ربُّ!“</span>
<span className="v" id="v26" data-v="26" data-xref="مَرْقُس 7:27–28; ﻣﺘﻰ 7:6; فِيلِبِّي 3:2"> فأجابَها: ”لا يَجوزُ أنْ يُؤخذَ خُبزُ البَنينَ ويُرمى إلى جراءِ الكِلابِ“.</span>
<span className="v" id="v27" data-v="27" data-xref="أفَسُس 3:8; لُوقا 16:21; أيُّوب 40:4–5"> فقَالَت لَه المَرأةُ: ”نَعم، يا سيِّدي! حتّى جراءُ الكلابُ تأكُلُ مِنَ الفُتاتِ الّذي يَتَساقَطُ عَنْ موائدِ أصحابِها“.</span>
<span className="v" id="v28" data-v="28" data-xref="مَرْقُس 9:23–24; ﻣﺘﻰ 8:13; المَزامِير 145:19"> فأجابَها يَسوعُ: ”ما أعظَمَ إيمانَكِ، يا اَمرأةُ! فلْيكُنْ لَكِ ما تُريدينَ“. فشُفِيَت اَبنَتُها مِنْ تِلكَ السَّاعةِ.</span>
</div>
</div>

## يسوع يشفي كثيرين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="ﻣﺘﻰ 4:18; يُوحَنّا 6:1; ﻣﺘﻰ 5:1"> واَنتَقَلَ يَسوعُ مِنْ هُناكَ إلى شاطئِ بحرِ الجليلِ، فصَعِدَ الجبَلَ وجَلَسَ هُناكَ.</span>
<span className="v" id="v30" data-v="30" data-xref="لُوقا 6:17–19; اعمال الرسل 5:15–16; ﻣﺘﻰ 14:35–36"> فجاءَتْهُ جُموعٌ كبيرةٌ ومَعَهُم عُرْجٌ وعُميانٌ ومُقعَدُونَ وخُرسٌ وغَيرُهُم كَثيرونَ، فطَرحوهُم عِندَ قدَمَيْهِ فشفاهُم.</span>
<span className="v" id="v31" data-v="31" data-xref="ﻣﺘﻰ 9:8; لُوقا 18:43; لُوقا 17:15–18"> فتعجَّبَ النَّاسُ عِندَما رأوا الخُرسَ يَتكَلَّمونَ، والعُرجَ يُشفَونَ، والمُقعَدينَ يَمشُونَ، والعُميانَ يُبصِرونَ. فمَجَّدوا إلهَ إِسرائيلَ.</span>
</div>
</div>

## يسوع يطعم أربعة آلاف رجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 27:33; لُوقا 12:29–30; ﻣﺘﻰ 9:36"> ودَعا يَسوعُ تلاميذَهُ وقالَ لهُم: ”أُشفِقُ على هذا الجَمعِ، فهُم مِنْ ثلاثَةِ أيّامٍ يُلازِمونَني، وما عِندَهُم ما يأكُلونَ. فلا أريدُ أن أصرِفَهُم صائِمينَ، لِئلَّا تَخورَ قِواهُم في الطَّريقِ“.</span>
<span className="v" id="v33" data-v="33" data-xref="مَرْقُس 6:37; ﻣﺘﻰ 14:15; مَرْقُس 8:4–5"> فقَالَ لَه التَّلاميذُ: ”مِنْ أينَ لنا في هذِهِ البرِّيَّةِ خُبزٌ يُشبِعُ مِثلَ هذا الجمعِ؟“</span>
<span className="v" id="v34" data-v="34" data-xref="ﻣﺘﻰ 16:9–10; لُوقا 24:41–42; يُوحَنّا 21:9–10"> فقَالَ لهُم يَسوعُ: ”كَمْ رَغيفًا عندَكُم؟“ أجابوا: ”سَبْعةُ أرغِفَةٍ وبعضُ سَمكاتٍ صِغارٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="لُوقا 9:14–16; مَرْقُس 6:39–40; يُوحَنّا 6:10"> فأمَرَ يَسوعُ الجَمعَ أنْ يَقعُدوا على الأرضِ،</span>
<span className="v" id="v36" data-v="36" data-xref="لُوقا 22:19; ﺭﻭﻣﻴﺔ 14:6; يُوحَنّا 6:11"> وأخَذَ الأرغِفَةَ السَّبعَةَ والسَّمكاتِ، وشَكَرَ وكَسَرَها وأعطى تلاميذَهُ، والتَّلاميذُ أعطَوْا الجُموعَ.</span>
<span className="v" id="v37" data-v="37" data-xref="مَرْقُس 8:19–21; ﻣﺘﻰ 16:9–10; ﻣﺘﻰ 15:33"> فأكلوا كُلُّهُم حتّى شَبِعوا، ثُمَّ رَفَعوا ما فضَلَ مِنَ الكِسَرِ سَبعَ سِلالٍ مُمتَلئَةٍ.</span>
<span className="v" id="v38" data-v="38"> وكانَ الّذينَ أكلوا أربعةَ آلافِ رجُلٍ ما عدا النِّساءَ والأولادَ.</span>
<span className="v" id="v39" data-v="39" data-xref="مَرْقُس 8:10; ﻣﺘﻰ 14:22"> وصَرفَ يَسوعُ الجُموعَ ورَكِبَ القارِبَ وجاءَ إلى أرضِ مَجْدَان.</span>
</div>
