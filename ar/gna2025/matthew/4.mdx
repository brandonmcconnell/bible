## إبليس يجرب يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العِبرانِيّين 2:18; لُوقا 4:1–13; مَرْقُس 1:12–15"> وقادَ الرُّوحُ القُدُسُ يَسوعَ إلى البرِّيَّةِ ليُجَرِّبَهُ إِبليسُ.</span>
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 9:18; الخُرُوج 34:28; المُلُوكِ الأوَّلُ 19:8"> فصامَ أربعينَ يومًا وأربعينَ لَيلةً حتّى جاعَ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّكوين 3:1–5; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 3:5; العَدَد 11:4–6"> فَدنا مِنهُ المُجَرِّبُ وقالَ لَه: ”إنْ كُنْتَ اَبنَ اللهِ، فقُلْ لِهذِهِ الحِجارَةِ أنْ تَصيرَ خُبزًا“.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 8:3; لُوقا 4:4; يُوحَنّا 6:63"> فأجابَهُ: ”يقولُ الكِتابُ: ما بِالخبزِ وحدَهُ يحيا الإنسانُ، بل بكلِّ كَلِمَةٍ تَخرُجُ مِنْ فمِ اللهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="لُوقا 4:9; ﻣﺘﻰ 27:53; رُؤيا يُوحَنّا 11:2"> وأخذَهُ إبليسُ إلى المدينةِ المُقَدَّسَةِ، فأوْقَفَهُ على شُرفَةِ الهَيكل</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 91:11–12; العِبرانِيّين 1:14; المَزامِير 34:7"> وقالَ لَه: ”إنْ كُنتَ اَبنَ اللهِ فأَلقِ بِنَفسِكَ إلى الأسفَلِ، لأنَّ الكِتابَ يقولُ: يُوصي ملائِكَتَهُ بكَ، فيَحمِلونَكَ على أيديهِم لئلَّا تَصطَدِمَ رِجلُكَ بِحجرٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 6:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:9; المَزامِير 78:18"> فأجابَهُ يَسوعُ: ”يقولُ الكِتابُ أيضًا: لا تُجرِّبِ الرَّبَّ إلهَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:15–16; رُؤيا يُوحَنّا 11:15; المَزامِير 49:16–17"> وأخَذَهُ إبليسُ إلى جبَلٍ عالٍ جدًّا، فَأراهُ جَميعَ مَمالِكِ الدُّنيا ومجدَها</span>
<span className="v" id="v9" data-v="9" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:4; دانيال 5:18–19; يُوحَنّا 16:11"> وقالَ لَه: ”أُعطِيكَ هذا كلَّهُ، إنْ سجَدْتَ لي وعَبدْتَني“.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 6:13–14; التَّثْنِيَة 10:20; يَشُوع 24:14"> فأجابَهُ يَسوعُ: ”إِبتَعِدْ عنِّي يا شَيطانُ! لأنَّ الكِتابَ يقولُ: للرّبِّ إلهِكَ تَسجُدُ، وإيّاهُ وحدَهُ تَعبُدُ“.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﻣﺘﻰ 26:53; العِبرانِيّين 1:14; لُوقا 22:43"> ثُمَّ تَركَهُ إبليسُ، فجاءَ بَعضُ الملائِكةِ يخدِمونَهُ.</span>
</div>
</div>

## يسوع يبشّر في الجليل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="مَرْقُس 1:14; لُوقا 3:20; ﻣﺘﻰ 14:3"> وسَمِعَ يَسوعُ باَعتِقالِ يوحنَّا، فرجَعَ إلى الجليلِ.</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 4:46; مَرْقُس 1:21; ﻣﺘﻰ 11:23"> ثُمَّ ترَكَ النَّاصِرةَ وسكَنَ في كَفْرَناحومَ على شاطئِ بحرِ الجليلِ في بلادِ زَبولونَ ونَفتالي،</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 9:1–2; ﻣﺘﻰ 2:23; لُوقا 24:44"> ليَتِمَّ ما قالَ النَّبيُّ إشَعْيا:</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 9:1; المُلُوكِ الثّانِي 15:29; يَشُوع 21:32"> ”يا أرضَ زَبولونَ وأرضَ نَفتالي،<br />على طريقِ البحرِ، عَبْرَ الأردنِ،<br />يا جليلَ الأُمَمِ!</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 9:2; إشَعْياء 42:6–7; إشَعْياء 60:1–3"> الشَّعْبُ الجالِسُ في الظَّلامِ<br />رأى نورًا ساطِعًا،<br />والجالِسونَ في أرضِ المَوتِ وَظِلالِهِ<br />أشرَقَ علَيهِمِ النُّورُ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 3:2; اعمال الرسل 2:38; مَرْقُس 1:14–15"> وبدأَ يَسوعُ مِنْ ذلِكَ الوقتِ يُبشِّرُ فيَقولُ: ”توبوا، لأنَّ مَلكوتَ السَّماواتِ اَقتَرَبَ“.</span>
</div>
</div>

## يسوع يدعو التَّلاميذ الأوّلين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="يُوحَنّا 1:40–42; ﻣﺘﻰ 10:2; لُوقا 6:14"> وكانَ يَسوعُ يَمشي على شاطئِ بحرِ الجليلِ، فرأى أخَوَينِ هُما سِمعانُ المُلقَّبُ بِبُطرُسَ وأخوهُ أندراوُسُ يُلقِيانِ الشَّبكَةَ في البحرِ، لأنَّهُما كانا صيَّادَيْنِ.</span>
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 1:17–18; يُوحَنّا 12:26; ﻣﺘﻰ 16:24"> فقَالَ لَهُما: ”إتبَعاني، أجعَلْكُما صيَّادَيْ بشرٍ“.</span>
<span className="v" id="v20" data-v="20" data-xref="مَرْقُس 10:28–31; ﻣﺘﻰ 19:27; لُوقا 18:28–30"> فتَركا شِباكَهُما في الحالِ وتَبِعاهُ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="مَرْقُس 1:19–20; ﻣﺘﻰ 10:2; مَرْقُس 3:17"> وسارَ مِنْ هُناكَ فَرأى أخوَينِ آخَرينِ، هُما يعقوبُ بنُ زَبدي وأخوهُ يوحنَّا، مَعَ أبيهِما زَبدي في قارِبٍ يُصلِحانِ شِباكَهُما، فدَعاهُما إلَيهِ.</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 10:37; لُوقا 9:59–60; التَّثْنِيَة 33:9–10"> فتَركا القارِبَ وأباهُما في الحالِ وتَبِعاهُ.</span>
</div>
</div>

## يسوع يعلّم ويبشّر ويشفي المرضى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="ﻣﺘﻰ 9:35; مَرْقُس 1:39; اعمال الرسل 10:38"> وكانَ يَسوعُ يَسيرُ في أنحاءِ الجليلِ، يُعلِّمُ في المجامعِ ويُعلِنُ إنجيلَ المَلكوتِ ويَشفي النَّاسَ مِنْ كُلِّ مَرَضٍ وداءٍ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 8:6; ﻣﺘﻰ 17:15; لُوقا 5:15"> فاَنتَشرَ صيتُهُ في سوريةَ كُلِّها، فجاؤُوا إلَيهِ بِجميعِ المُصابينَ بأوجاعٍ وأمراضٍ متنوِّعَةٍ: مِنْ مَصروعينَ ومُقْعَدينَ والّذينَ بِهِمْ شياطينُ، فشفاهُم.</span>
<span className="v" id="v25" data-v="25" data-xref="لُوقا 6:17; مَرْقُس 5:20; مَرْقُس 3:7–8"> فتَبِعَتْهُ جموعٌ كبيرةٌ مِنَ الجليلِ والمُدُنِ العَشْر وأُورُشليمَ واليهوديَّةِ وعَبْرِ الأُردنِ.</span>
</div>
