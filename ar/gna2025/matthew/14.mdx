## موت يوحنا المعمدان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="لُوقا 3:1; مَرْقُس 8:15; لُوقا 9:7–9"> وفي ذلِكَ الوَقتِ سمِعَ الوالي هيرودُسُ أخبارَ يَسوعَ،</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 16:14; يُوحَنّا 10:41; مَرْقُس 8:28"> فقَالَ لحاشِيَتِهِ: ”هذا يوحنَّا المَعمدانُ قامَ مِنْ بَينِ الأمواتِ، ولذلِكَ تَجري المُعْجِزاتُ على يَدِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="مَرْقُس 6:17; لُوقا 3:19–20; ﻣﺘﻰ 4:12"> وكانَ هيرودُسُ أمسَكَ يوحنَّا وقَيَّدَهُ وسَجَنَهُ مِنْ أجلِ هيرودِيَّةَ اَمرأةِ أخيهِ فيلبُّسَ،</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 20:21; اللّاويِّين 18:16; مَرْقُس 6:18"> لأنَّ يوحنَّا كانَ يقولُ لَه: ”لا يَحِلُّ لَكَ أنْ تَتَزوَّجَها“.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻣﺘﻰ 21:26; ﻣﺘﻰ 11:9; مَرْقُس 6:19–20"> وأرادَ أنْ يَقتُلَهُ، فخافَ مِنَ الشَّعبِ لأنَّهُم كانوا يَعدُّونَهُ نَبيًّا.</span>
<span className="v" id="v6" data-v="6" data-xref="مَرْقُس 6:21–23; التَّكوين 40:20; هُوشَع 1:5–6"> ولمَّا أقامَ هيرودُسُ ذِكرى مَولِدِهِ، رقَصَتِ اَبنَةُ هيرودِيَّةَ في الحَفلةِ، فأعجَبَتْ هيرودُسَ،</span>
<span className="v" id="v7" data-v="7" data-xref="أسْتِير 5:3; أسْتِير 5:6; أسْتِير 7:2"> فأقسَمَ لها أنْ يُعطِيَها ما تَشاءُ.</span>
<span className="v" id="v8" data-v="8" data-xref="مَرْقُس 6:24; الأمثال 29:10; الأمثال 1:16"> فلقَّنَتْها أمُّها، فقَالَت لِهيرودُسَ: ”أعطِني هُنا على طَبَقٍ رَأسَ يوحنَّا المَعمدانِ!“</span>
<span className="v" id="v9" data-v="9" data-xref="يُوحَنّا 19:12–16; صَمُوئيلَ الأوَّلُ 14:28; مَرْقُس 6:26"> فحَزِنَ المَلِكُ، ولكنَّهُ أمَرَ بإعطائِها ما تُريدُ، مِنْ أجلِ اليَمينِ الّتي حَلَفَها على مسامِعِ الحاضرينَ.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 17:12; ﻣﺘﻰ 21:35–36; لُوقا 9:9"> وأرسَلَ جُنديًّا، فقَطَعَ رأسَ يوحنَّا في السِّجن</span>
<span className="v" id="v11" data-v="11" data-xref="رُؤيا يُوحَنّا 16:6; التَّكوين 49:7; حِزْقِيال 16:3–4"> وجاءَ بِه على طبَقٍ. وسلَّمَهُ إلى الفتاةِ، فحَمَلَتْهُ إلى أُمِّها.</span>
<span className="v" id="v12" data-v="12" data-xref="اعمال الرسل 8:2; ﻣﺘﻰ 27:58–61"> وجاءَ تلاميذُ يوحنَّا، فحَمَلوا الجُثَّةَ ودَفَنوها، ثُمَّ ذَهَبوا وأخبَروا يَسوعَ.</span>
</div>
</div>

## يسوع يطعم خمسة آلاف رجل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="لُوقا 9:10–17; ﻣﺘﻰ 14:1–2; مَرْقُس 6:30–44"> فلمَّا سَمِعَ يَسوعُ، خرَجَ مِنْ هُناكَ في قارِبٍ إلى مكانٍ مُقْفِرٍ يَعتَزِلُ فيهِ. وعرَفَ النَّاسُ، فتَبِعوهُ مِنَ المُدُنِ مَشيًا على الأقدامِ.</span>
<span className="v" id="v14" data-v="14" data-xref="العِبرانِيّين 4:15; ﻣﺘﻰ 9:36; العِبرانِيّين 2:17"> فلمَّا خرَجَ مِنَ القاربِ رأى جُموعًا كبيرةً، فأشفَقَ علَيهِم وشفَى مَرضاهُم.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 15:23; مَرْقُس 8:3; مَرْقُس 6:35–36"> وفي المساءِ، دَنا مِنهُ تلاميذُهُ وقالوا: ”فاتَ الوقتُ، وهذا المكانُ مُقفِرٌ، فقُلْ لِلنَّاسِ أنْ يَنصرِفوا إلى القُرى لِيشتَروا لهُم طعامًا“.</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الثّانِي 4:42–44; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:2–3; الجامِعَة 11:2"> فأجابَهُم يَسوعُ: ”لا داعيَ لاَنصرافِهِم. أعطوهُم أنتُم ما يأكلونَ“.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 15:33–34; مَرْقُس 6:37–38; لُوقا 9:13"> فقَالوا لَه: ”ما عِندَنا هُنا غيرُ خَمسةِ أرغِفةٍ وسَمكتَينِ“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18"> فقَالَ يَسوعُ: ”هاتوا ما عندَكُم“.</span>
<span className="v" id="v19" data-v="19" data-xref="مَرْقُس 8:6–7; لُوقا 24:30; صَمُوئيلَ الأوَّلُ 9:13"> ثُمَّ أمَرَ الجُموعَ أنْ يَقعُدوا على العُشبِ، وأخَذَ الأرغِفَةَ الخَمسَةَ والسَّمكتَينِ، ورَفَعَ عَينَيهِ نحوَ السَّماءِ وبارَكَ وكسَرَ الأرغِفَةَ وأعطَى تلامِيذَهُ، والتَّلاميذُ أعطَوا الجُموعَ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 16:12; مَرْقُس 6:42–44; ﻣﺘﻰ 16:8–10"> فأكلوا كُلُّهُم حتّى شَبِعوا، ثُمَّ رَفَعوا اَثنتي عَشْرةَ قُفَّةً مملوءةً مِنَ الكِسَرِ الّتي فَضَلَتْ.</span>
<span className="v" id="v21" data-v="21" data-xref="فِيلِبِّي 4:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:8–11; اعمال الرسل 4:4"> وكانَ الّذينَ أكلوا نحوَ خَمسةِ آلافِ رجُلٍ، ما عدا النِّساءَ والأولادَ.</span>
</div>
</div>

## يسوع يمشي على الماء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="مَرْقُس 6:45–51; يُوحَنّا 6:15–21; ﻣﺘﻰ 15:39"> وأمرَ يَسوعُ تلاميذَهُ أن يَركبوا القارِبَ في الحالِ ويَسبِقوهُ إلى الشَّاطِئِ المُقابلِ حتّى يَصرِفَ الجُموعَ.</span>
<span className="v" id="v23" data-v="23" data-xref="لُوقا 6:12; مَرْقُس 6:46; ﻣﺘﻰ 26:36"> ولمَّا صرَفَهُم صَعِدَ إلى الجبَلِ ليصَلِّيَ في العُزلَةِ. وكانَ وحدَهُ هُناكَ عِندَما جاءَ المساءُ.</span>
<span className="v" id="v24" data-v="24" data-xref="يُوحَنّا 6:18; مَرْقُس 6:48; ﻣﺘﻰ 8:24"> وأمَّا القارِبُ فاَبتَعدَ كثيرًا عَنِ الشَّاطئِ وطَغَتِ الأمواجُ علَيهِ، لأنَّ الرِّيحَ كانَت مُخالِفَةً لَه.</span>
<span className="v" id="v25" data-v="25" data-xref="أيُّوب 9:8; يُوحَنّا 6:19; مَرْقُس 6:48"> وقَبلَ الفَجرِ، جاءَ يَسوعُ إلى تلاميذِهِ ماشيًا على البَحرِ.</span>
<span className="v" id="v26" data-v="26" data-xref="مَرْقُس 6:49–50; رُؤيا يُوحَنّا 1:17; لُوقا 1:11–12"> فلمَّا رآهُ التَّلاميذُ ماشيًا على البَحرِ اَرتَعبوا وقالوا: ”هذا شَبَحٌ!“ وصَرَخوا مِنْ شِدَّةِ الخَوفِ.</span>
<span className="v" id="v27" data-v="27" data-xref="اعمال الرسل 23:11; رُؤيا يُوحَنّا 1:17–18; يُوحَنّا 16:33"> فقَالَ لهُم يَسوعُ في الحالِ: ”تَشجَّعوا. أنا هوَ، لا تخافوا!“</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 6:68; ﺭﻭﻣﻴﺔ 12:3; ﻣﺘﻰ 26:33–35"> فقَالَ لَه بُطرُسُ: ”إنْ كُنتَ أنتَ هوَ، يا ربُّ، فَمُرْني أنْ أجيءَ إلَيكَ على الماءِ“.</span>
<span className="v" id="v29" data-v="29" data-xref="فِيلِبِّي 4:13; مَرْقُس 9:23; اعمال الرسل 3:16"> فأجابَهُ يَسوعُ: ”تعالَ“. فنَزَلَ بُطرُسُ مِنَ القارِبِ ومشَى على الماءِ نحوَ يَسوعَ.</span>
<span className="v" id="v30" data-v="30" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:54–57; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:16–17; مَرْقُس 14:38"> ولكنَّهُ خافَ عِندَما رأى الرِّيحَ شديدةً فأخَذَ يَغرَقُ، فَصرَخ: ”نَجِّني، يا ربُّ!“</span>
<span className="v" id="v31" data-v="31" data-xref="يَعقُوب 1:6–8; المَزامِير 138:7; التَّكوين 22:14"> فمَدَّ يَسوعُ يدَهُ في الحالِ وأمسكَهُ وقالَ لَه: ”يا قليلَ الإيمانِ، لِماذا شكَكْتَ؟“</span>
<span className="v" id="v32" data-v="32" data-xref="المَزامِير 107:29–30; مَرْقُس 4:41; يُوحَنّا 6:21"> ولمَّا صَعِدا إلى القارِبِ هَدأَتِ الرِّيحُ.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 11:27; المَزامِير 2:7; ﺭﻭﻣﻴﺔ 1:4"> فسجَدَ لَه الّذينَ كانوا في القارِبِ وقالوا: ”بالحقيقةِ أنتَ اَبنُ اللهِ!“</span>
</div>
</div>

## يسوع يشفي من أمراض كثيرة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="مَرْقُس 6:53–56; لُوقا 5:1; يُوحَنّا 6:24–25"> وعَبَرَ يَسوعُ وتلاميذُهُ إلى بَرِّ جَنّيسارَتَ.</span>
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 4:24–25; مَرْقُس 6:55; مَرْقُس 1:28–34"> فلمَّا عرَفَ أهلُ البَلْدَةِ يَسوعَ، نَشروا الخبَرَ في تِلكَ الأنحاءِ كُلِّها. فجاؤُوهُ بالمَرضى</span>
<span className="v" id="v36" data-v="36" data-xref="لُوقا 6:19; مَرْقُس 3:10; الخُرُوج 28:33–43"> وطَلَبوا إلَيهِ أنْ يَلمُسوا ولو طرَفَ ثوبِهِ. فكانَ كُلُّ مَنْ يَلمُسُه يُشفى.</span>
</div>
