## يسوع عند بيلاطس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="مَرْقُس 15:1; يُوحَنّا 18:28; ميخا 2:1"> ولمَّا طلَعَ الصُّبحُ، تَشاورَ جميعُ رُؤساءِ الكَهنَةِ وشُيوخُ الشَّعبِ على يَسوعَ ليَقتُلوهُ.</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 20:19; اعمال الرسل 3:13; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:13"> ثُمَّ قَيَّدوهُ وأخَذوهُ وأسلَموهُ إلى الحاكِمِ بيلاطُسَ.</span>
</div>
</div>

## موت يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:10; لُوقا 22:47–48; أيُّوب 20:15–29"> فلمَّا رأى يَهوذا الّذي أسلَمَ يَسوعَ أنَّهُم حكَموا علَيهِ، ندِمَ ورَدَّ الثَّلاثينَ مِنَ الفِضَّةِ إلى رُؤساءِ الكَهنَةِ والشُّيوخِ،</span>
<span className="v" id="v4" data-v="4" data-xref="صَمُوئيلَ الأوَّلُ 15:30; تِيطُس 1:16; اعمال الرسل 18:15–17"> وقالَ لهُم: ”خَطِئتُ حينَ أسلَمتُ دمًا بريئًا“. فقَالوا لَه: ”ما علَينا؟ دَبِّرْ أنتَ أمرَكَ“.</span>
<span className="v" id="v5" data-v="5" data-xref="اعمال الرسل 1:18–19; صموئيلَ الثّانِي 17:23; المُلُوكِ الأوَّلُ 16:18"> فرَمى يَهوذا الفِضَّةَ في الهَيكلِ واَنْصرفَ، ثُمَّ ذهَبَ وشَنقَ نفسَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="لُوقا 6:7–9; التَّثْنِيَة 23:18; إشَعْياء 61:8"> فأخَذَ رُؤساءُ الكَهنَةِ الفِضَّةَ وقالوا: ”هذِهِ ثمنُ دمٍ، فلا يَحِلُّ لنا أنْ نضَعَها في صُندوقِ الهَيكَلِ“.</span>
<span className="v" id="v7" data-v="7"> فاَتَّفَقوا أنْ يَشتَروا بِها حَقلَ الخَزّافِ ليَجعَلوهُ مَقبرَةً لِلغُرَباءِ.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 1:19; ﻣﺘﻰ 28:15; يَشُوع 4:9"> ولِهذا يُسَمّيهِ الناسُ حقلَ الدَّمِ إلى هذا اليومِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="زَكَريّا 11:12–13; ﻣﺘﻰ 1:22; اللّاويِّين 27:2–7"> فتَمَّ ما قالَهُ النَّبيُّ إرميا: ”وأخذوا الثَّلاثينَ مِنَ الفِضَّةِ، وهيَ ما اَتَّفقَ بَعضُ بَني إِسرائيلَ على أنْ يكونَ ثمنُهُ،</span>
<span className="v" id="v10" data-v="10" data-xref="زَكَريّا 11:13"> ودَفَعوها ثَمنًا لِحَقلِ الخزّافِ. هكذا أمرَني الرَّبُّ“.</span>
</div>
</div>

## بيلاطس يسأل يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="لُوقا 23:2–3; مَرْقُس 15:2–5; ﻣﺘﻰ 26:25"> ووقَفَ يَسوعُ أمامَ الحاكِمِ فسألَهُ الحاكِمُ: ”أأنتَ مَلِكُ اليَهودِ؟“ فأجابَهُ يَسوعُ: ”أنتَ قُلتَ“.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻣﺘﻰ 26:63; إشَعْياء 53:7; ﻣﺘﻰ 27:14"> وكانَ رُؤساءُ الكَهنَةِ والشُّيوخِ يتَّهِمونَهُ، فلا يُجيبُ بِشيءٍ.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻣﺘﻰ 26:62; يُوحَنّا 18:35; اعمال الرسل 22:24"> فقَالَ له بيلاطُسُ: ”أما تسمَعُ ما يَشهَدونَ بِه علَيكَ؟“</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 8:18; المَزامِير 71:7; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:9"> فما أجابَهُ يَسوعُ عَنْ شيءٍ، حتّى تَعجَّبَ الحاكِمُ كثيرًا.</span>
</div>
</div>

## الحكم على يسوع بالموت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 18:38–40; لُوقا 23:16; يُوحَنّا 19:16"> وكانَ مِنْ عادَةِ الحاكِمِ في كُلِّ عيدٍ أن يُطلِقَ واحدًا مِن السُّجَناءِ يَختارُهُ الشّعبُ.</span>
<span className="v" id="v16" data-v="16" data-xref="مَرْقُس 15:7; اعمال الرسل 3:14; لُوقا 23:25"> وكانَ عِندَهُم في ذلِكَ الحينِ سَجينٌ شهيرٌ اَسمُهُ يشوعُ باراباسُ.</span>
<span className="v" id="v17" data-v="17" data-xref="يُوحَنّا 19:15; المُلُوكِ الأوَّلُ 18:21; مَرْقُس 15:9–12"> فلمَّا تَجَمْهرَ النَّاسُ سألَهُم بيلاطُسُ: ”مَنْ تُريدونَ أنْ أُطلِقَ لكُم: يشوعُ باراباسُ أمْ يَسوعُ الّذي يُقالُ لَه المَسيحُ؟“</span>
<span className="v" id="v18" data-v="18" data-xref="الأمثال 27:4; اعمال الرسل 5:17; مَرْقُس 15:10"> وكان بيلاطُسُ يَعرِفُ أنَّهُم مِنْ حَسَدِهِم أسلموا يَسوعَ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="زَكَريّا 9:9; الأمثال 29:1; ﻣﺘﻰ 27:24"> وبَينَما بيلاطُسُ على كُرسِيِّ القَضاءِ، أرسَلَتْ إلَيهِ اَمرأتُهُ تَقولُ: ”إيَّاكَ وهذا الرَّجُلَ البارَّ، لأنِّي تألَّمتُ اللَّيلَةَ في الحُلُمِ كثيرًا مِنْ أجلِهِ“.</span>
<span className="v" id="v20" data-v="20" data-xref="اعمال الرسل 3:14–15; يُوحَنّا 19:15–16; يُوحَنّا 18:40"> لكنَّ رُؤساءَ الكَهنَةِ والشُّيوخَ حَرَّضوا الجُموعَ على أنْ يَطلُبوا باراباسَ ويَقتُلوا يَسوعَ.</span>
<span className="v" id="v21" data-v="21"> فلمَّا سألَهُمُ الحاكمُ: ”أيُّهُما تُريدونَ أنْ أُطلِقَ لكُم؟“ أجابوا: ”باراباسُ!“</span>
<span className="v" id="v22" data-v="22" data-xref="إشَعْياء 53:2–3; إشَعْياء 49:7; اعمال الرسل 13:38"> فقَالَ لهُم بيلاطُسُ: ”وماذا أفعَلُ بيَسوعَ الّذي يُقالُ لَه المَسيحُ؟“ فأجابوا كُلُّهُم: ”إصْلِبْهُ!“</span>
<span className="v" id="v23" data-v="23" data-xref="التَّكوين 37:18–19; اعمال الرسل 7:57; اعمال الرسل 23:12–15"> قالَ لهُم: ”وَأيَّ شَرٍّ فَعلَ؟“ فاَرتفَعَ صياحُهُم: ”إصْلِبْهُ!“</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="ﻣﺘﻰ 27:4; المَزامِير 26:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:21"> فلمَّا رأى بيلاطُسُ أنَّهُ ما اَستفادَ شيئًا، بلِ اَشتَدَّ الاَضطِرابُ، أخذَ ماءً وغسَلَ يَديهِ أمامَ الجُموعِ وقالَ: ”أنا بَريءٌ مِنْ دَمِ هذا البارّ! دَبِّروا أنتُم أمرَهُ“.</span>
<span className="v" id="v25" data-v="25" data-xref="يَشُوع 2:19; اعمال الرسل 5:28; اعمال الرسل 7:52"> فأجابَ الشّعبُ كُلُّهُ: ”دمُهُ علَينا وعلى أولادِنا!“</span>
<span className="v" id="v26" data-v="26" data-xref="إشَعْياء 53:5; يُوحَنّا 19:1; مَرْقُس 15:15"> فأطلَقَ لهُم باراباسَ، أمَّا يَسوعُ فجلَدَهُ وأسلَمَهُ ليُصْلَبَ.</span>
</div>
</div>

## الجنود يستهزئون بيسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="يُوحَنّا 18:28; يُوحَنّا 18:33; اعمال الرسل 10:1"> فأخذَ جُنودُ الحاكِمِ يَسوعَ إلى قَصرِ الحاكِمِ وجَمعوا الكَتيبةَ كُلَّها،</span>
<span className="v" id="v28" data-v="28" data-xref="مَرْقُس 15:17; يُوحَنّا 19:2–5; لُوقا 23:11"> فنزَعوا عَنهُ ثيابَهُ وألبَسوهُ ثَوبًا قِرمِزيًّا،</span>
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 53:3; ﻣﺘﻰ 20:19; المَزامِير 69:19–20"> وضَفَروا إكليلًا مِنْ شَوكٍ ووضَعوهُ على رأسِهِ، وجَعَلوا في يَمينِهِ قصَبَةً، ثُمَّ رَكَعوا أمامَهُ واَستَهزأوا بِه فقَالوا: ”السّلامُ علَيكَ يا مَلِكَ اليَهودِ!“</span>
<span className="v" id="v30" data-v="30" data-xref="إشَعْياء 50:6; ﻣﺘﻰ 26:67; إشَعْياء 53:7"> وأمسكوا القصَبَةَ وأخَذوا يَضرِبونَهُ بِها على رأسِهِ وهُم يَبصُقونَ عليهِ.</span>
<span className="v" id="v31" data-v="31" data-xref="إشَعْياء 53:7; المُلُوكِ الأوَّلُ 21:13; المُلُوكِ الأوَّلُ 21:10"> وبَعدَما اَستَهزَأوا بِه نَزَعوا عَنهُ الثَّوبَ القِرمِزيَّ، وألبَسوهُ ثيابَهُ وساقوهُ ليُصلَبَ.</span>
</div>
</div>

## يسوع على الصليب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="مَرْقُس 15:21; ﻣﺘﻰ 16:24; لُوقا 23:26"> وبَينَما هُمْ خارِجونَ مِنَ المدينةِ صادَفوا رَجُلًا مِنْ قَيرينَ اَسمُهُ سِمْعانُ، فسَخَّروهُ ليَحمِلَ صَليبَ يَسوعَ.</span>
<span className="v" id="v33" data-v="33" data-xref="يُوحَنّا 19:17–19; مَرْقُس 15:22–32; مَرْقُس 15:34"> ولمَّا وصَلوا إلى المكانِ الّذي يُقالُ لَه الجُلجُثَةُ، أي ”مَوضِعُ الجُمجُمَةِ“</span>
<span className="v" id="v34" data-v="34" data-xref="المَزامِير 69:21; مَرْقُس 15:23; ﻣﺘﻰ 27:48"> أعطَوْهُ خَمرًا مَمزوجَةً بِالمُرِّ، فلمَّا ذاقَها رفَضَ أنْ يَشرَبَها.</span>
<span className="v" id="v35" data-v="35" data-xref="المَزامِير 22:18; مَرْقُس 15:24–32; لُوقا 23:34"> فصَلبوهُ واَقتَرعوا على ثيابِهِ واَقتَسموها.</span>
<span className="v" id="v36" data-v="36" data-xref="ﻣﺘﻰ 27:54; مَرْقُس 15:44; مَرْقُس 15:39"> وجَلَسوا هُناكَ يَحرُسونَه.</span>
<span className="v" id="v37" data-v="37" data-xref="لُوقا 23:38; يُوحَنّا 19:19–22; مَرْقُس 15:26"> ووضَعوا فَوقَ رأسِهِ لافِتَةً مكتوبًا فيها سَببُ الحُكمِ علَيهِ: ”هذا يَسوعُ، مَلِكُ اليَهودِ“.</span>
<span className="v" id="v38" data-v="38" data-xref="يُوحَنّا 19:18; لُوقا 23:39–43; إشَعْياء 53:12"> وصَلَبوا مَعهُ لِصَّينِ، واحدًا عَنْ يَمينِهِ وواحدًا عَنْ شِمالِهِ.</span>
<span className="v" id="v39" data-v="39" data-xref="المَزامِير 109:25; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:12; المَزامِير 69:7–12"> وكانَ المارَّةُ يَهُزُّونَ رُؤوسَهُم ويَشتِمونَهُ ويَقولونَ:</span>
<span className="v" id="v40" data-v="40" data-xref="ﻣﺘﻰ 26:61; ﻣﺘﻰ 4:6; ﻣﺘﻰ 4:3"> ”يا هادِمَ الهَيكَلِ وبانِيَهُ في ثلاثَةِ أيّامٍ، إنْ كُنتَ اَبنَ اللهِ، فخلِّصْ نفسَكَ واَنزِلْ عَنِ الصَّليبِ“.</span>
<span className="v" id="v41" data-v="41" data-xref="المَزامِير 35:26; إشَعْياء 28:22; المَزامِير 22:12–13"> وكانَ رُؤساءُ الكَهنَةِ ومُعلِّمو الشّريعَةِ والشُّيوخُ يَستهزِئونَ بِه، فيَقولونَ:</span>
<span className="v" id="v42" data-v="42" data-xref="يُوحَنّا 1:49; ﻣﺘﻰ 27:37; اعمال الرسل 4:14"> ”خَلَّصَ غيرَهُ، ولا يَقدِرُ أنْ يُخلِّصَ نفسَهُ! هوَ مَلِكُ إِسرائيلَ، فلْيَنزِلِ الآنَ عَنِ الصَّليبِ لِنؤمِنَ بِه!</span>
<span className="v" id="v43" data-v="43" data-xref="المَزامِير 22:8; يُوحَنّا 10:36; إشَعْياء 37:10"> توَكَّلَ على اللهِ وقالَ: أنا اَبنُ اللهِ، فليُنقِذْهُ اللهُ الآنَ إنْ كانَ راضيًا عنهُ“.</span>
<span className="v" id="v44" data-v="44" data-xref="ﻣﺘﻰ 27:38; مَرْقُس 15:32; لُوقا 23:39–43"> وعيَّرَهُ اللِّصانِ المَصلوبانِ مَعهُ أيضًا، فقَالا مِثلَ هذا الكلامِ.</span>
</div>
</div>

## موت يسوع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="عامُوس 8:9; مَرْقُس 15:33–34; لُوقا 23:44–45"> وعِندَ الظُّهرِ خيَّمَ على الأرضِ كُلِّها ظلامٌ حتّى السَّاعةِ الثَّالِثةِ.</span>
<span className="v" id="v46" data-v="46" data-xref="المَزامِير 22:1; العِبرانِيّين 5:7; إشَعْياء 53:10"> ونحوَ السَّاعةِ الثَّالثةِ صرَخَ يَسوعُ بِصوتٍ عَظيمٍ: ”إيلي، إيلي، لِما شَبقتاني؟“ أي ”إلهي، إلهي، لماذا تَركتَني؟“</span>
<span className="v" id="v47" data-v="47" data-xref="مَلاخِي 4:5; مَرْقُس 15:35–36; ﻣﺘﻰ 11:14"> فسَمِعَ بَعضُ الحاضرينَ هُناكَ، فقَالوا: ”ها هوَ يُنادي إيليَّا!“</span>
<span className="v" id="v48" data-v="48" data-xref="المَزامِير 69:21; ﻣﺘﻰ 27:34; لُوقا 23:36"> وأسرَعَ واحدٌ مِنهُم إلى إسفِنْجَةٍ، فبَلَّلَها بالخَلِّ ووضَعَها على طرَفِ قَصبَةٍ ورَفَعها إلَيهِ لِيَشرَبَ.</span>
<span className="v" id="v49" data-v="49"> فقَالَ لَه الآخرونَ: ”إِنتَظِرْ لِنرى هَلْ يَجيءُ إيليَّا ليُخَلِّصَهُ!“</span>
<span className="v" id="v50" data-v="50" data-xref="يُوحَنّا 19:30; إشَعْياء 53:9–12; المَزامِير 22:14–15"> وصرَخَ يَسوعُ مرَّةً ثانيةً صَرْخَةً قَوِيَّةً وأسلَمَ الرُّوحَ.</span>
</div>
