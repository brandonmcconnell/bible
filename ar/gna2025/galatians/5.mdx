## الحريّة المسيحية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 3:17; غَلاطِيَّة 5:13; رسالَةُ بُطرُسَ الأُولَى 2:16"> فالمَسيحُ حَرَّرَنا لِنكونَ أحرارًا. فاَثبُتوا، إذًا، ولا تَعودوا إلى نِيرِ العُبودِيَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="غَلاطِيَّة 5:6; اعمال الرسل 15:1; ﺭﻭﻣﻴﺔ 9:31–32"> فأنا بولُسُ أقولُ لكُم: إذا اَخْتَتَنتُم، فلا يُفيدُكمُ المَسيحُ شيئًا.</span>
<span className="v" id="v3" data-v="3" data-xref="غَلاطِيَّة 3:10; ﺭﻭﻣﻴﺔ 2:25; يَعقُوب 2:10–11"> وأشهَدُ مَرَّةً أُخرى لِكُلِّ مَنْ يَختَتِنُ بِأنَّهُ مُلزَمٌ أنْ يَعمَلَ بأحكامِ الشَّريعَةِ كُلِّها.</span>
<span className="v" id="v4" data-v="4" data-xref="ﺭﻭﻣﻴﺔ 9:31–32; رسالَةُ بُطرُسَ الثّانيَة 2:20–22; ﺭﻭﻣﻴﺔ 11:6"> والّذينَ مِنكُم يَطلُبونَ أنْ يَتَبَرَّروا بالشَّريعَةِ، يَقطَعونَ كُلَّ صِلَةٍ لهُم بالمَسيحِ ويَسقُطونَ عَنِ النِّعمَةِ.</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 130:5; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 1:10; المَزامِير 62:5"> أمَّا نَحنُ، فنَنتَظِرُ على رَجاءِ أنْ يُبرِّرَنا اللهُ بالإيمانِ بقُدرَةِ الرُّوحِ.</span>
<span className="v" id="v6" data-v="6" data-xref="غَلاطِيَّة 6:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:19; يَعقُوب 2:14–26"> فَفي المَسيحِ يَسوعَ لا الخِتانُ ولا عَدَمُهُ يَنفَعُ شيئًا، بَلِ الإيمانُ العامِلُ بالمحبَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:24; غَلاطِيَّة 3:1; العِبرانِيّين 12:1"> كُنتُم في سَيْرِكُم على ما يُرامُ، فمَنْ صَدَّكُم ورَدَّكُم عَنْ طاعَةِ الحقِّ؟</span>
<span className="v" id="v8" data-v="8" data-xref="غَلاطِيَّة 1:6; ﺭﻭﻣﻴﺔ 8:28"> ما كانَ هذا الإغراءُ مِنَ الّذي دَعاكُم.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:33; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:6–7; مَرْقُس 8:15"> قليلٌ مِنَ الخَميرِ يُخمِّرُ العَجينَ كُلَّهُ.</span>
<span className="v" id="v10" data-v="10" data-xref="غَلاطِيَّة 5:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:3; غَلاطِيَّة 1:7"> ولي ثِقَةٌ بِكُم في الرَّبِّ أنَّكُم لَنْ تَقبَلوا رأيًا آخرَ. وكُلُّ مَنْ يُوقِعُ البَلبَلَةَ بَينَكُم سيَنالُ عِقابَهُ، أيًّا كانَ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:23; غَلاطِيَّة 6:12; غَلاطِيَّة 4:29"> وأنا، أيُّها الإخوَةُ لَو كُنتُ أدعو إلى الخِتانِ، فلِماذا أُعاني الاضطهادَ إلى اليومِ، أما كانَ يَزولُ العائِقُ الّذي في الصَّليبِ؟</span>
<span className="v" id="v12" data-v="12" data-xref="غَلاطِيَّة 5:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:13; تِيطُس 3:10"> لَيتَ الّذينَ يُوقِعونَ البَلبَلَةَ بَينَكُم يَقطَعونَ هُم أعضاءَهُم!</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:9"> فأنتُم، يا إخوَتي، دَعاكُمُ اللهُ لتَكونوا أحرارًا، ولكِنْ لا تَجعَلوا هذِهِ الحُرِّيَّةَ حُجَّةً لإرضاءِ شَهَواتِ الجسَدِ، بَل اَخدُموا بَعضُكُم بَعضًا بِالمَحبَّةِ.</span>
<span className="v" id="v14" data-v="14" data-xref="ﻣﺘﻰ 7:12; اللّاويِّين 19:18; غَلاطِيَّة 6:2"> فالشَّريعَةُ كُلُّها تكتَمِلُ في وَصيَّةٍ واحدةٍ: ”أَحِبَّ قَريبَكَ مِثلَما تُحِبُّ نَفسَكَ“.</span>
<span className="v" id="v15" data-v="15" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:3; يَعقُوب 3:14–3; غَلاطِيَّة 5:26"> أمَّا إذا كُنتُم تَنهَشونَ وتَأكُلونَ بَعضُكُم بَعضًا، فاَنتَبِهوا أنْ لا يُفني واحِدُكُمُ الآخرَ.</span>
</div>

## الروح والجسد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="غَلاطِيَّة 5:24–25; ﺭﻭﻣﻴﺔ 8:12–14; غَلاطِيَّة 6:8"> وأقولُ لكُم: أُسلُكوا في الرُّوحِ ولا تُشبِعوا شَهوَةَ الجَسَدِ.</span>
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 8:13; يُوحَنّا 3:6; ﺭﻭﻣﻴﺔ 8:5–8"> فما يَشتَهيهِ الجَسَدُ يُناقِضُ الرُّوحَ، وما يَشتَهيهِ الرُّوحُ يُناقِضُ الجَسَدَ. كُلٌّ مِنهُما يُقاوِمُ الآخَرَ لِئَلَّا تَعمَلوا ما تُريدونَ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﺭﻭﻣﻴﺔ 8:14; ﺭﻭﻣﻴﺔ 6:14–15; غَلاطِيَّة 5:25"> فإذا كانَ الرُّوحُ يَقودُكُم، فَما أنتُم في حُكمِ الشَّريعَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:9–10; كُولُوسِي 3:5–8; ﻣﺘﻰ 15:18–19"> وأمَّا أعمالُ الجَسَدِ فهِيَ ظاهِرَةٌ: الزِّنى والدَّعارَةُ والفجورُ</span>
<span className="v" id="v20" data-v="20" data-xref="تِيطُس 3:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:19; رُؤيا يُوحَنّا 21:8"> وعِبادَةُ الأوثانِ والسِّحرُ والعَداوَةُ والشِّقاقُ والغَيرَةُ والغَضَبُ والدَّسُّ والخِصامُ والتَّحزبُ</span>
<span className="v" id="v21" data-v="21" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:9–10; ﺭﻭﻣﻴﺔ 8:13; ﺭﻭﻣﻴﺔ 13:13"> والحسَدُ والسِّكرُ والعَربدةُ وما أشبَهُ. وأُنبِّهُكمُ الآنَ، كما نَبَّهْتُكمُ مِنْ قَبلُ، أنَّ الّذينَ يَعمَلونَ هذِهِ الأعمالَ لا يَرِثونَ مَلكوتَ اللهِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="كُولُوسِي 3:12–17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:4–7; يُوحَنّا 15:5"> أمَّا ثمَرُ الرُّوحِ فهوَ المَحبَّةُ والفَرَحُ والسَّلامُ والصَّبرُ واللُّطفُ والصَّلاحُ والأمانَةُ</span>
<span className="v" id="v23" data-v="23" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:9; اعمال الرسل 24:25; تِيطُس 2:2"> والوَداعَةُ والعَفافُ. وما مِنْ شَريعَةٍ تنهى عَنْ هذِهِ الأشياءِ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﺭﻭﻣﻴﺔ 6:6; ﺭﻭﻣﻴﺔ 13:14; ﺭﻭﻣﻴﺔ 8:13"> والّذينَ هُم لِلمَسيحِ يَسوعَ صلَبوا جَسَدَهُم بِكُلِّ ما فيهِ مِنْ أهواءٍ وشَهواتٍ.</span>
<span className="v" id="v25" data-v="25" data-xref="غَلاطِيَّة 5:16; ﺭﻭﻣﻴﺔ 8:4–5; ﺭﻭﻣﻴﺔ 8:2"> فإذا كُنّا نَحيا بالرُّوح، فعلَينا أنْ نَسلُكَ طريقَ الرُّوحِ،</span>
<span className="v" id="v26" data-v="26" data-xref="يَعقُوب 3:14–16; فِيلِبِّي 2:1–3; يَعقُوب 4:16"> فلا نَتكبَّرُ ولا يَتَحدَّى ولا يَحسُدُ بَعضُنا بَعضًا.</span>
