## يوآش ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أخبارِ الأيّامِ الأوّلُ 3:11; المُلُوكِ الثّانِي 11:21–15"> وكانَ يوآشُ اَبنَ سَبعِ سِنينَ حينَ ملَكَ، وملَكَ أربَعينَ سنَةً بِأورُشليمَ، وكانَ اَسْمُ أمِّهِ ظَبْيَةَ مِنْ بِئرَ سَبْعَ.</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الثّانِي 26:4–5; أخبارِ الأيّامِ الثّانِي 25:2; أخبارِ الأيّامِ الثّانِي 24:17–22"> وعَمِلَ يوآشُ ما هوَ قَويمٌ في نظَرِ الرّبِّ كُلَّ أيّامِ يُوياداعَ الكاهنِ</span>
<span className="v" id="v3" data-v="3" data-xref="أخبارِ الأيّامِ الثّانِي 24:15; ﻣﺘﻰ 19:4–8; التَّكوين 4:19"> الّذي زَوَّجَهُ بِاَمرَأتَينِ فوَلَدَ بَنينَ وبَناتٍ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4"> وبَعدَ ذلِكَ عزَمَ يوآشُ على تَجديدِ هَيكلِ الرّبِّ</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الثّانِي 12:4–7; أخبارِ الأيّامِ الثّانِي 34:8–9; أخبارِ الأيّامِ الثّانِي 29:3"> فجَمَعَ الكهَنةَ واللَّاويِّينَ وقالَ لهُم: ”أخرُجوا إلى مُدُنِ يَهوذا واَجمَعوا مالًا مِنْ جميعِ الشَّعبِ لِتَرميمِ هَيكلِ إلهِكُم سنَةً بَعدَ سنَةٍ، وعَجِّلوا“. غَيرَ أنَّ اللَّاويِّينَ لم يُعَجِّلوا.</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 30:12–16; العَدَد 1:50; العَدَد 17:7–8"> فاَستَدعى المَلِكُ رئيسَهُم يُوياداعَ وسألَهُ: ”لِماذا لم تَطلُبْ مِنَ اللَّاويِّينَ أنْ يأخُذوا مِنْ يَهوذا وأورُشليمَ الضَّريبةَ الّتي فرَضَها موسى عبدُ الرّبِّ على شعبِهِ لِلعِنايةِ بِخَيمةِ الشَّهادةِ؟</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الثّانِي 21:17; أخبارِ الأيّامِ الثّانِي 28:22–24; المُلُوكِ الثّانِي 12:4"> فعَثَلْيا الأثيمةُ وأتباعُها أهمَلوا الهَيكلَ واَستَعمَلوا جميعَ أدَواتِهِ المُكرَّسَةِ لِلرّبِّ في عِبادةِ البَعلِ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="مَرْقُس 12:41; المُلُوكِ الثّانِي 12:8–9"> وبِناءً على أمرِ المَلِكِ عَمِلوا صُندوقًا ووضَعوهُ عِندَ بابِ الهَيكلِ مِنَ الخارِجِ،</span>
<span className="v" id="v9" data-v="9" data-xref="أخبارِ الأيّامِ الثّانِي 24:6; ﻣﺘﻰ 17:24–27"> ونادوا في يَهوذا وأورُشليمَ بِأنْ يدفَعَ الشَّعبُ الضَّريبَةَ الّتي فرَضَها علَيهِ موسى عبدُ اللهِ في البرِّيَّةِ.</span>
<span className="v" id="v10" data-v="10" data-xref="اعمال الرسل 2:45–47; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:7; أخبارِ الأيّامِ الأوّلُ 29:9"> ففَرِحَ جميعُ الشَّعبِ وأعيانُهُ، بِهذا وتَوافَدوا إلى دَفعِ الضَّريبةِ في الصُّندوقِ حتّى اَمتَلأَ.</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 12:10–12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 16:2"> وكُلَّما اَمتَلأَ كانَ يَحمِلُهُ اللَّاويُّونَ إلى ديوانِ المَلِكِ فيُفرِغُهُ رئيسُ الدِّيوانِ ومُعاوِنُ رئيسِ الكهَنةِ، ثُمَّ يأخُذانِهِ ويَرُدَّانِهِ إلى مكانِهِ. وهكذا كانوا يفعَلونَ يومًا فيومًا حتّى جمَعوا مِنَ المالِ شيئًا كثيرًا.</span>
<span className="v" id="v12" data-v="12" data-xref="المُلُوكِ الأوَّلُ 5:15; أخبارِ الأيّامِ الثّانِي 34:9–11"> فأعطاهُ المَلِكُ ويُوياداعُ لِلقائِمينَ بِالعمَلِ على ترميمِ الهَيكلِ، فاَستَأجَروا نحَّاتينَ ونجَّارينَ وحدَّادينَ ونحَّاسينَ لِهذا الغرَضِ.</span>
<span className="v" id="v13" data-v="13" data-xref="نَحَمْيا 4:7; مَرْقُس 13:1–2; حَجَّي 2:3"> وبذَلَ هؤلاءِ العُمَّالُ جَهدَهُم وتَوَفَّقوا بِما عَمِلَت أيديهِم، وأعادوا هَيكلَ اللهِ إلى ما كانَ علَيهِ سابِقًا، وثَبَّتوا أركانَهُ.</span>
<span className="v" id="v14" data-v="14" data-xref="العَدَد 28:2–29; الأمثال 27:22; المُلُوكِ الثّانِي 12:13–14"> ولمَّا فَرَغوا مِنْ ذلِكَ، أحضَروا ما تبَقَّى مِنَ المالِ إلى المَلِكِ ويُوياداعَ، فعَمِلا بهِ آنيةً وطاساتِ ذهبٍ وفِضَّةٍ لِلخِدمَةِ في الهَيكلِ وتَقديمِ المُحرَقاتِ بِلا اَنقِطاعٍ كُلَّ أيّامِ يُوياداعَ.</span>
</div>
</div>

## وفاة يوياداع والعودة إلى الأصنام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="التَّكوين 47:9; أخبارِ الأيّامِ الأوّلُ 23:1; المَزامِير 90:10"> وشاخَ يُوياداعُ وشَبِعَ مِنَ الأيّامِ وماتَ وهوَ اَبنُ مئَةٍ وثَلاثينَ سنَةً.</span>
<span className="v" id="v16" data-v="16" data-xref="العِبرانِيّين 6:10; أخبارِ الأيّامِ الثّانِي 31:20; نَحَمْيا 13:14"> فدَفَنوهُ في مدينةِ داوُدَ معَ المُلوكِ اَعتِرافًا بِفَضلِهِ على إِسرائيلَ وبِجَهدِهِ الّذي بذَلَهُ في خِدمةِ اللهِ وهَيكلِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="التَّثْنِيَة 31:27; الأمثال 26:28; اعمال الرسل 20:29–30"> وبَعدَ وفاةِ يُوياداعَ جاءَ زُعَماءُ يَهوذا وأكَّدوا وَلاءَهُم لِلمَلِكِ، فقَرَّبَهُم وجعَلَهُم مِنْ مُستَشاريهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="يَشُوع 22:20; أخبارِ الأيّامِ الثّانِي 19:2; أفَسُس 5:6"> فأهمَلَ الشَّعبُ هَيكلَ الرّبِّ إلهِ آبائِهِم وعبَدوا أصنامَ أشيرةَ، فحَلَّ غضَبُ اللهِ على يَهوذا وأورُشليمَ لِمَعصيَتِهم هذِهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 42:23; لُوقا 20:9–15; ﻣﺘﻰ 13:9"> فأرسلَ إليهِم أنبياءَ ليُنذِروهُم ويَرُدُّوهُم إلى الرّبِّ، فلم يَستَمِعوا.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="العَدَد 14:41; إرْمِيا 5:25; أخبارِ الأيّامِ الثّانِي 15:1–2"> ثُمَّ اَستَولى روحُ اللهِ على زكريّا بنِ يُوياداعَ الكاهنِ فوقَفَ أمامَ الشَّعبِ وقالَ لهُم: ”هذا ما قالَ الرّبُّ: لِماذا تُخالِفونَ وصايا الرّبِّ، فتُنزِلونَ الشَّقاءَ بِأنفُسِكُم؟ أهمَلتُمُ الرّبَّ فأهمَلَكُم“.</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 7:58–59; نَحَمْيا 9:26; إرْمِيا 38:4–6"> فثاروا على زَكريّا ورجَموهُ بِالحجارَةِ بِأمرٍ مِنَ المَلِكِ يوآشَ في دارِ الهَيكلِ.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 9:5; لُوقا 11:51; رُؤيا يُوحَنّا 6:9–11"> ولم يَذكُرِ المَلِكُ ما كانَ لِيُوياداعَ أبي زَكريّا مِنَ الفَضلِ علَيهِ، بل قتَلَ اَبنَهُ الّذي قالَ عِندَ موتِهِ: ”الرّبُّ يرى وينتَقِمُ“.</span>
</div>
</div>

## آخر أيام يوآش

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="المُلُوكِ الثّانِي 12:17–18; التَّثْنِيَة 32:35; المُلُوكِ الأوَّلُ 20:22"> وفي خريفِ تِلكَ السَّنةِ زحَفَ جيشُ آرامَ على يَهوذا وأورُشليمَ وقتَلوا جميعَ أعيانِ الشَّعبِ وأرسَلوا كُلَّ غَنائِمِهِم إلى مَلِكِ دِمشقَ،</span>
<span className="v" id="v24" data-v="24" data-xref="أخبارِ الأيّامِ الثّانِي 22:8; اللّاويِّين 26:37; اللّاويِّين 26:8"> ومعَ أنَّ الجيشَ الآراميَّ كانَ قليلَ العدَدِ، فإنَّ الرّبَّ أسلَمَ إلى أيديهِم جيشَ يَهوذا العظيمَ جِدًّا، لأنَّ الشَّعبَ تَركوا الرّبَّ إلهَ آبائِهِم. وأتَمَّ جيشُ آرامَ قضاءَ الرّبِّ على يوآشَ،</span>
<span className="v" id="v25" data-v="25" data-xref="أخبارِ الأيّامِ الثّانِي 24:21–22; أخبارِ الأيّامِ الثّانِي 28:27; أخبارِ الأيّامِ الثّانِي 21:18–20"> إذ تَركوه مُصابًا بِجِراحٍ كثيرةٍ، فاَتَّفَقَ علَيهِ خَدَمُهُ اَنتِقامًا لِدَمِ اَبنِ يُوياداعَ الكاهنِ وقتَلوهُ في سريرِهِ. فدُفنَ في مدينةِ داوُدَ، لا في مقابِرِ المُلوكِ.</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الثّانِي 12:21"> وكانَ مِمَّنْ تآمرَ علَيهِ زابادُ بنُ شَمعَةَ العَمُّونيَّةِ، ويوزابادُ بنُ شِمْريتَ الموآبيَّةِ.</span>
<span className="v" id="v27" data-v="27" data-xref="أخبارِ الأيّامِ الثّانِي 16:11; أخبارِ الأيّامِ الثّانِي 13:22; أخبارِ الأيّامِ الثّانِي 24:12"> أمَّا سيرَةُ بَنيهِ، وكثرةُ النُّبوءاتِ علَيهِ، وترميمُهُ هَيكلَ الرّبِّ، هذا كُلُّهُ مُدوَّنٌ في شرحِ سِفرِ المُلوكِ. وملَكَ أمَصْيا اَبنُهُ مكانَهُ.</span>
</div>
