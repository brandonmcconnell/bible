## يوآحاز ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إرْمِيا 22:11; أخبارِ الأيّامِ الثّانِي 33:25; أخبارِ الأيّامِ الثّانِي 26:1"> فأقامَ شعبُ يَهوذا يوآحازَ بنَ يوشيَّا مَلِكًا علَيهِم مكانَ أبيهِ بأورُشليمَ.</span>
<span className="v" id="v2" data-v="2"> وكانَ يوآحازُ اَبنَ ثَلاثٍ وعِشرينَ سنَةً حينَ ملَكَ، وملَكَ ثلاثَةَ أشهرٍ بأورُشليمَ.</span>
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الثّانِي 23:33"> فعَزَلَهُ نَكْوُ مَلِكُ مِصْرَ وفرَضَ على يَهوذا ضَريبةً قَدْرُها سِتَّةَ عشَرَ قِنْطارًا مِنَ الفِضَّةِ، وأربَعةَ عشَرَ رَطْلًا مِنَ الذَّهبِ.</span>
<span className="v" id="v4" data-v="4" data-xref="إرْمِيا 22:10–12; المُلُوكِ الثّانِي 23:34–35; حِزْقِيال 19:3–4"> وأقامَ نَكْوُ مَلِكُ مِصْرَ ألياقيمَ أخا يوآحازَ مَلِكًا على يَهوذا وأورُشليمَ، وبَدَّلَ اَسمَهُ يوياقيمَ. وأمَّا يوآحازُ فأخذَهُ نَكْوُ وجاءَ بهِ إلى مِصْرَ.</span>
</div>
</div>

## يوياقيم ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الثّانِي 23:36–37; إرْمِيا 22:13–19; إرْمِيا 26:1"> وكانَ يوياقيمُ اَبنَ خمسٍ وعِشرينَ سنَةً حينَ ملَكَ، وملَكَ إحدى عَشْرَةَ سنَةً بأورُشليمَ، وفعَلَ الشَّرَّ في نظَرِ الرّبِّ إلَهِهِ.</span>
<span className="v" id="v6" data-v="6" data-xref="أخبارِ الأيّامِ الثّانِي 33:11; المُلُوكِ الثّانِي 24:5–6; حِزْقِيال 19:5–9"> فزَحَفَ نَبوخذنَصَّرُ مَلِكُ بابِلَ على يَهوذا، وأسَرَ يوياقيمَ وقَيَّدهُ بِسِلسِلَتَينِ مِنْ نُحاسٍ وجاءَ بهِ إلى بابِلَ.</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الثّانِي 24:13; دانيال 5:2–4; إرْمِيا 27:16–18"> وأخذَ نَبوخذنَصَّرُ آنيَةً مِنَ الهَيكلِ ووضَعَها في قصرِهِ بِبابِلَ.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الثّانِي 24:5–6; ﻣﺘﻰ 1:11–12; أخبارِ الأيّامِ الأوّلُ 3:16–17"> وما تبَقَّى مِنْ أخبارِ يوياقيمَ وما فعَلَهُ مِنْ مَعصياتٍ واَرتكَبَهُ مِنْ شُرورٍ مُدوَّنٌ في سِفرِ مُلوكِ إِسرائيلَ ويَهوذا. وملَكَ يوياكينُ اَبنُهُ مكانَهُ.</span>
</div>
</div>

## يوياكين ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الثّانِي 24:8–17"> وكانَ يوياكينُ اَبنَ ثَماني عَشْرَةَ سنَةً حينَ ملَكَ، وملَكَ ثلاثةَ أشهرٍ وعشَرَةَ أيّامٍ بأورُشليمَ، وفعَلَ الشَّرَّ في نظَرِ الرّبِّ:</span>
<span className="v" id="v10" data-v="10" data-xref="إرْمِيا 37:1; أخبارِ الأيّامِ الثّانِي 36:7; المُلُوكِ الثّانِي 24:10–17"> وفي ربيعِ تِلكَ السَّنةِ أرسلَ المَلِكُ نَبوخذنَصَّرُ رُسُلًا إلى أورُشليمَ، فجاؤوا إلى بابِلَ بيوياكينَ وما كانَ في هَيكلِ الرّبِّ مِنْ آنيةٍ نَفيسةٍ وأقاموا صِدْقيَّا عَمَّ يوياكينَ مَلِكًا على يَهوذا وأورُشليمَ.</span>
</div>
</div>

## صدقيّا ملك يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الثّانِي 24:18–20; إرْمِيا 52:1–3; إرْمِيا 27:1"> وكانَ صِدْقيَّا اَبنَ إحدى وعِشرينَ سنَةً حينَ ملَكَ، وملَكَ إحدى عَشْرَةَ سنَةً بِأورُشليمَ.</span>
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الثّانِي 33:23; إرْمِيا 37:2–21; إرْمِيا 38:14–28"> وفعَلَ الشَّرَّ في نظَرِ الرّبِّ إلَهِهِ ولم يُصغِ بِخُشوعٍ لإرميا النَّبيِّ الّذي كانَ يَنطِقُ بِكلامِ الرّبِّ.</span>
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 52:2–3; أخبارِ الأيّامِ الثّانِي 30:8; حِزْقِيال 17:11–20"> وتمَرَّدَ أيضًا على المَلِكِ نَبوخذنَصَّرَ الّذي حَلَّفَهُ باللهِ أنْ يُخْلِصَ لَه. وعانَدَ الرّبَّ إلهَ إِسرائيلَ وقَسَّى قلبَهُ ولم يَرجِعْ إليهِ.</span>
<span className="v" id="v14" data-v="14" data-xref="دانيال 9:8; حِزْقِيال 22:6; ميخا 3:9–11"> وبالَغَ جميعُ رُؤساءِ الكهَنةِ وزُعَماءُ الشَّعبِ في خيانةِ الرّبِّ باَرتِكابِ مَعاصي الأُمَمِ حَولهُم، ونَجَّسوا بَيتَ الرّبِّ الّذي قَدَّسَهُ الرّبُّ في أورُشليمَ.<br />سقوط أورشليم</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 35:15; إرْمِيا 7:13; إرْمِيا 25:3–4"> ولم يتَوَقَّفِ الرّبُّ إلهُ آبائِهِم عَنْ إنذارِهِم على ألسِنَةِ أنبيائِهِ لأنَّهُ أشفَقَ على شعبِهِ وعلى هَيكلِهِ.</span>
<span className="v" id="v16" data-v="16" data-xref="إرْمِيا 5:12–13; أخبارِ الأيّامِ الثّانِي 30:10; لُوقا 22:63–64"> فكانوا يَستَهزِئونَ بِرُسُلِ اللهِ وأنبيائِهِ ويَحتَقِرونَ كلامَهُ على ألْسِنَتِهِم، حتّى ثارَ غضَبُ الرّبِّ على شعبِهِ ولم يَعُدْ هُناكَ مِنْ عِلاجٍ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="المَزامِير 74:20; عَزْرا 9:7; حِزْقِيال 9:5–7"> فأرسلَ الرّبُّ مَلِكَ بابِلَ لِمُحارَبَتِهِم، فقَتَل خِيرَةَ شُبَّانِهِم بِالسَّيفِ حتّى في بَيتِ مَقْدسِهِم، ولم يُشفِقْ على فَتًى أو عذراءَ، ولا على شيخٍ أو كَهْلٍ مُقْعَدٍ، بل أسلَمَ الرّبُّ الجميعَ إلى يَدِهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الثّانِي 36:7; أخبارِ الأيّامِ الثّانِي 36:10; إشَعْياء 39:6"> ونهَبَ جميعَ آنيَةِ الهَيكلِ، الكبيرةِ والصَّغيرةِ، وكُنوزَهُ وكُنوزَ المَلِكِ وأعوانِهِ وأخذَها إلى بابِلَ.</span>
<span className="v" id="v19" data-v="19" data-xref="المَزامِير 79:1; المُلُوكِ الأوَّلُ 9:8; إرْمِيا 52:13–15"> وأحرَقَ بِالنَّارِ هَيكلَ اللهِ وجميعَ قُصورِ أورُشليمَ، وهدَمَ سُورَ المدينةِ، وأتلَفَ كُلَّ نَفيسٍ فيها.</span>
<span className="v" id="v20" data-v="20" data-xref="إرْمِيا 27:7; عَزْرا 1:1–11; أخبارِ الأيّامِ الثّانِي 36:22"> والّذينَ نَجَوا مِنَ السَّيفِ سَباهُم إلى بابِلَ حَيثُ صاروا عبيدًا لَه ولِبَنيهِ حتى قيامِ دولَةِ الفُرسِ.</span>
<span className="v" id="v21" data-v="21" data-xref="إرْمِيا 29:10; اللّاويِّين 26:34–35; زَكَريّا 1:12"> وبِذلِكَ تَمَّ ما قالَهُ الرّبُّ بِلِسانِ إرمِيا: ”تَبقى الأرضُ بُورًا سَبعينَ سنَةً، حتّى تُعَوِّضَ عَمَّا فاتَها مِنَ السُّبوتِ الّتي لم تستَرِحْ فيها طُولَ تِلكَ المُدَّةِ“.</span>
</div>
</div>

## كورش يأمر اليهود بالعودة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="إرْمِيا 25:12; عَزْرا 1:1–3; إرْمِيا 29:10"> وفي السَّنةِ الأولى لِكورَشَ مَلِكِ فارِسَ حقَّقَ الرّبُّ ما وعَدَ بهِ على لِسانِ إرميا، إذ ألهَمَ كورَشَ بِأنْ يكتُبَ هذا النِّداءَ ويُذيعَهُ في مَملكَتِهِ كُلِّها:</span>
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 44:26–28; عَزْرا 7:13; ﺭﻭﻣﻴﺔ 8:31"> ”هذا ما يقولُ كورَشُ مَلِكُ الفُرسِ: أعطانيَ الرّبُّ إلهُ السَّماواتِ جميعَ مَمالِكِ الأرضِ وأوصاني بِأنْ أبنيَ لَه هَيكلًا في أورُشليمَ الّتي بيَهوذا. فمَنْ كانَ مِنكُم مِنْ شعبِهِ، فليَذهَبْ إلى هُناكَ، والرّبُّ إلهُهُ معَهُ“.</span>
</div>
