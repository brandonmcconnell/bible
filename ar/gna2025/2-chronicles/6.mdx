## سليمان يخاطب الشعب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المَزامِير 18:8–11; اللّاويِّين 16:2; الخُرُوج 20:21"> ثُمَّ قالَ سليمانُ: ”قُلتَ يا رَبُّ إنَّكَ تَسكُنُ في الظَّلامِ.</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الثّانِي 2:4–6; المَزامِير 135:21; أخبارِ الأيّامِ الأوّلُ 28:6"> والآنَ يا ربُّ بَنَيتُ لكَ هَيكلًا مجيدًا، مكانًا لِسُكْناكَ إلى الأبدِ“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الأوَّلُ 8:14; أخبارِ الأيّامِ الثّانِي 29:29; المُلُوكِ الأوَّلُ 8:55–61"> واَلتَفتَ المَلِكُ إلى كُلِّ جماعةِ إِسرائيلَ وهُم وقُوفٌ، وباركَهُم</span>
<span className="v" id="v4" data-v="4" data-xref="المَزامِير 72:18–19; ﻣﺘﻰ 24:35; المَزامِير 68:32–35"> وقالَ: ”تبارَكَ الرّبُّ إلهُ إِسرائيلَ الّذي تَمَّمَ بيَدِهِ ما وعَدَ بِفَمِهِ داوُدَ أبي، قائلًا:</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 20:24; المُلُوكِ الأوَّلُ 8:16; صَمُوئيلَ الأوَّلُ 13:13–14"> مُنذُ يومِ أخرَجتُ شعبي إِسرائيلَ مِنْ مِصْرَ لم أختَرْ مدينةً مِنْ جميعِ مُدُنِ أسباطِ إِسرائيلَ ليُبْنَى لي فيها هَيكلٌ يكونُ اَسمي فيهِ، ولا اَختَرْتُ رَجُلًا يكونُ رئيسًا لِشعبي إِسرائيلَ.</span>
<span className="v" id="v6" data-v="6" data-xref="أخبارِ الأيّامِ الأوّلُ 28:4; المَزامِير 132:13; أخبارِ الأيّامِ الثّانِي 12:13"> لكنِّي اليومَ اَختَرْتُ أورُشليمَ ليكونَ اَسمي فيها واَختَرْتُ داوُدَ ليكونَ رئيسًا على بَني إِسرائيلَ شعبي“.<br />ثُمَّ قالَ سليمانُ:</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الأوّلُ 17:1; المُلُوكِ الأوَّلُ 5:3; أخبارِ الأيّامِ الأوّلُ 22:7"> ”وكانَ في نيَّةِ داوُدَ أبي أنْ يَبنيَ هَيكلًا لاَسمِ الرّبِّ إلهِ إِسرائيلَ،</span>
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12; مَرْقُس 14:8; المُلُوكِ الأوَّلُ 8:18–21"> فقالَ لَه الرّبُّ: نَوَيتَ في قلبِكَ أنْ تَبنيَ هَيكلًا لاَسمي، فحَسَنٌ ما نَوَيتَ.</span>
<span className="v" id="v9" data-v="9" data-xref="صموئيلَ الثّانِي 7:12–13; أخبارِ الأيّامِ الأوّلُ 17:4; أخبارِ الأيّامِ الأوّلُ 17:11–12"> لكِنْ أنتَ لا تَبني البَيتَ، بلِ اَبنُكَ الّذي يخرُجُ مِنْ صُلْبِكَ هوَ الّذي يَبنيهِ.</span>
<span className="v" id="v10" data-v="10" data-xref="الجامِعَة 1:4; أخبارِ الأيّامِ الأوّلُ 28:5; أخبارِ الأيّامِ الأوّلُ 17:11"> وحقَّقَ الرّبُّ قولَهُ، وقُمْتُ أنا مكانَ داوُدَ أبي وجلَسْتُ على عرشِ إِسرائيلَ، كما قالَ الرّبُّ، وبَنَيتُ الهَيكلَ لاَسمِ الرّبِّ إلهِ إِسرائيلَ.</span>
<span className="v" id="v11" data-v="11" data-xref="أخبارِ الأيّامِ الثّانِي 5:10; أخبارِ الأيّامِ الثّانِي 5:7; المُلُوكِ الأوَّلُ 8:21"> وأعدَدْتُ مكانًا لِتابوتِ العَهدِ الّذي قطَعَهُ الرّبُّ لِبَني إِسرائيلَ“.</span>
</div>
</div>

## صلاة سليمان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="المَزامِير 63:4; إشَعْياء 50:10; أيُّوب 11:13"> ثُمَّ نهَضَ سليمانُ أمامَ مَذبَحِ الرّبِّ، بِحُضورِ كُلِّ جماعَةِ إِسرائيلَ، ورفَعَ يَدَيهِ للصَّلاةِ</span>
<span className="v" id="v13" data-v="13" data-xref="المُلُوكِ الأوَّلُ 8:54; نَحَمْيا 8:4; المَزامِير 95:6"> وكانَ سليمانُ صنَعَ مِنبرًا مِنْ نُحاسٍ وأقامَهُ وسَطَ الدَّارِ، طُولُهُ خمسُ أذرُعٍ وعَرضُهُ خمسُ أذرُعٍ واَرْتِفاعُهُ ثَلاثُ أذرُعٍ، فصَعِدَ علَيهِ وسجَدَ على رُكبَتَيهِ أمامَ جماعةِ بَني إِسرائيلَ ورفَعَ يَدَيهِ نحوَ السَّماءِ.</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 15:11; التَّثْنِيَة 7:9; نَحَمْيا 1:5"> وقالَ: ”أيُّها الرّبُّ إله إِسرائيلَ، لا إلهَ مِثلُكَ في السَّماءِ ولا على الأرضِ، حافِظُ العَهدِ وصانِعُ الرَّحمةِ لِعَبيدِكَ الّذينَ يَخدِمونَكَ بِكُلِّ قلوبِهِم،</span>
<span className="v" id="v15" data-v="15" data-xref="أخبارِ الأيّامِ الأوّلُ 22:9–10; صموئيلَ الثّانِي 7:12; المُلُوكِ الأوَّلُ 8:24"> أنتَ يا مَنْ حقَّقَ بِيَدِهِ اليومَ ما قالَهُ بِفَمِهِ لِعَبدِهِ داوُدَ أبي.</span>
<span className="v" id="v16" data-v="16" data-xref="المُلُوكِ الأوَّلُ 2:4; يُوحَنّا 15:14–15; أخبارِ الأيّامِ الثّانِي 7:18"> والآنَ أيُّها الرّبُّ إله إِسرائيلَ تَمِّمْ لِعَبدِكَ داوُدَ أبي ما وعَدْتَهُ بهِ: لا يَنقَطِعُ مِنْ نسلِكَ رجُلٌ يجلِسُ على عرشِ إِسرائيلَ، إنْ عَمِلَ بَنوكَ بِشَريعتي وخَدموني كما خدَمتَني أنتَ.</span>
<span className="v" id="v17" data-v="17" data-xref="إرْمِيا 11:5; إشَعْياء 41:17; أخبارِ الأيّامِ الثّانِي 6:4"> والآنَ يا إلهَ إِسرائيلَ، ليتَحَقَّقِ الوَعْدُ الّذي وعَدْتَ بهِ عَبدَكَ داوُدَ أبي.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الثّانِي 2:6; إشَعْياء 66:1; المَزامِير 139:7–10"> ”ولكِنْ هل تسكُنُ الأرضَ حَقًّا يا اللهُ؟ حتّى السَّماواتُ وسماواتُ السَّماواتِ لا تسَعُكَ، فكَيفَ يسَعُكَ هذا الهَيكلُ الّذي بَنَيتُهُ لكَ؟</span>
<span className="v" id="v19" data-v="19" data-xref="المُلُوكِ الأوَّلُ 8:28; لُوقا 18:1–7; يُوحَنّا 17:20"> إلتَفِتْ إلى صلاتي وتضَرُّعي أنا عبدَكَ أيُّها الرّبُّ إلهي، واَسمَعِ الدُّعاءَ والصَّلاةَ الّتي أرفعُها إليكَ اليومَ.</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 34:15; دانيال 6:10; المَزامِير 121:5"> لِتكُنْ عيناكَ مَفتوحَتَينِ ليلًا نهارًا على هذا الهَيكلِ الّذي قُلتَ يكونُ اَسمُكَ فيهِ لِتَسمَعَ الصَّلاةَ الّتي أرفعُها أنا عبدكَ في هذا المكانِ،</span>
<span className="v" id="v21" data-v="21" data-xref="ميخا 7:18; دانيال 9:19; إشَعْياء 43:25"> واَستَجِبْ إلى تضَرُّعِ عبدِكَ وبَني إِسرائيلَ شعبِكَ الّذينَ يُصَلُّونَ في اَتِّجاهِ هذا المكانِ، واَسمَعْ أنتَ مِنْ مَسكِنِكَ في السَّماءِ وإذا سَمِعْتَ فاَغفِرْ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 22:11; اللّاويِّين 5:1; العَدَد 5:19–22"> ”إذا اتَّهمَ أحدُهُمُ الآخَرَ بالإساءةِ إليهِ وجيءَ بهِ إلى مذبَحِكَ في هذا الهَيكلِ ليَحلِفَ اليَمينَ أنَّهُ بَريءٌ،</span>
<span className="v" id="v23" data-v="23" data-xref="إشَعْياء 3:10–11; الأمثال 17:15; إرْمِيا 51:56"> فاَسمَعْ أنتَ مِنَ السَّماءِ واَحكُمْ بَينَ عبيدِكَ، فتَدينَ المُذنِبَ وتُعاقِبَهُ لِما فعَلَ وتُبَرِّئَ البَريءَ وتُكافِئَهُ على بَراءَتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="اللّاويِّين 26:17; المُلُوكِ الثّانِي 17:7–18; يَشُوع 7:8"> ”وإذا اَنهَزَمَ بَنو إِسرائيلَ شعبُكَ أمامَ أعدائِهِم بِسبَبِ خطيئتِهِم إليكَ، ثُمَّ تابوا إليكَ واَعْتَرفوا باَسمِكَ وصلّوا وتضَرَّعُوا إليكَ في هذا المكانِ،</span>
<span className="v" id="v25" data-v="25" data-xref="إرْمِيا 33:6–13; عَزْرا 1:1–6; التَّكوين 13:15"> فاَسمَعْ أنتَ مِنَ السَّماءِ واَغفِرْ لهُم خطيئتَهُم ورُدَّهُم إلى الأرضِ الّتي أعطيتَها لهُم ولآبائِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 28:23; التَّثْنِيَة 11:17; اللّاويِّين 26:19"> ”وإذا حبَسْتَ المطَرَ في السَّماءِ عَنهُم بِسبَبِ خطيئَتِهِم إليكَ، وصلّوا في هذا المكانِ واَعْتَرفوا باَسمِكَ ورجَعوا عَنْ خطيئَتِهم بَعدَما أذلَلتَهُم،</span>
<span className="v" id="v27" data-v="27" data-xref="المُلُوكِ الأوَّلُ 8:35–36; إرْمِيا 5:24; زَكَريّا 10:1"> فاَسمَعْ أنتَ مِنَ السَّماءِ واَغفِرْ خطيئةَ عبيدِكَ وشعبكَ بَني إِسرائيلَ، واَهدِهِمِ الطَّريقَ الصَّالحَ الّذي يسلُكونَهُ، وأنزِلْ مطَرًا على أرضِكَ هذِهِ الّتي أعطَيتَها مُلْكًا لِشعبِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="رُؤيا يُوحَنّا 9:3–11; يُوئيل 2:25; راعُوث 1:1"> ”وإذا حدَثَ في الأرضِ جُوعٌ أو وَباءٌ أو أصابَ الزَّرْعَ حَرٌّ شديدٌ أو فيهِ حَلَّ الجَرادُ أوِ النَّمْلُ، أو إذا حاصَرَهُم أعداؤُهُم في أيِّ مدينةٍ مِنْ مُدُنِهِم، أو نزَلَت بِهِم آفةٌ أو داءٌ،</span>
<span className="v" id="v29" data-v="29" data-xref="المَزامِير 50:15; المَزامِير 32:2–6; أخبارِ الأيّامِ الثّانِي 6:12–13"> أو إذا شعَرَ أيُّ واحدٍ مِنْ بَني إِسرائيلَ شعبِكَ بالنَّدَمِ في قلبِهِ على ما فعَلَهُ مِنْ سُوءٍ، فصلَّى إليكَ وتضرَّعَ رافِعًا يَديهِ صوبَ هذا الهَيكلِ،</span>
<span className="v" id="v30" data-v="30" data-xref="أخبارِ الأيّامِ الأوّلُ 28:9; صَمُوئيلَ الأوَّلُ 16:7; رُؤيا يُوحَنّا 2:23"> فاَسمَعْ أنتَ مِنَ السَّماءِ، مِنْ مكانِ سُكْناكَ، واَغفِرْ لِكُلِّ واحدٍ، أعِنْهُ وجازِهِ، بِحسَبِ عمَلِهِ ومعرِفتِكَ ما في قلبِهِ لأنَّكَ أنتَ وحدَكَ تَعرِفُ قلوبَ جميعِ البشَرِ،</span>
<span className="v" id="v31" data-v="31" data-xref="اعمال الرسل 9:31; أيُّوب 28:28; المَزامِير 128:1"> فيخافُكَ بَنو إِسرائيلَ ويسلُكونَ في طُرُقِكَ كُلَّ أيّامِ حياتِهِم على وجهِ الأرضِ الّتي أعطَيتَها لآبائِنا.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="إشَعْياء 56:3–7; المُلُوكِ الأوَّلُ 8:41–43; ﻣﺘﻰ 8:10–11"> ”وإذا جاءَ الغريبُ الّذي لا ينتَمي إلى بَني إِسرائيلَ شعبِكَ، إذا جاءَ مِنْ أرضٍ بعيدةٍ مِنْ أجلِ اَسمِكَ، فالنَّاسُ يسمَعونَ باَسمِكَ العظيمِ ويدِكَ القديرةِ وذِراعِكَ المَمدودةِ، وصلّى في هذا الهَيكلِ</span>
<span className="v" id="v33" data-v="33" data-xref="أخبارِ الأيّامِ الثّانِي 7:14; المُلُوكِ الأوَّلُ 8:16; رُؤيا يُوحَنّا 11:15"> فاَسمَعْ أنتَ مِنَ السَّماءِ، مِنْ مكانِ سُكْناكَ واَستَجِب ما يرجوهُ مِنكَ هذا الغريبُ ليَعرِفَ جميعُ أُمَمِ الأرضِ اَسمَكَ ويخافوكَ مِثلَ بَني إِسرائيلَ شعبِكَ ويَعلموا أنَّ اَسمَكَ في هذا الهَيكلِ الّذي بَنَيتُهُ لكَ.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="أخبارِ الأيّامِ الثّانِي 20:6–13; أخبارِ الأيّامِ الثّانِي 18:31; أخبارِ الأيّامِ الثّانِي 20:4"> ”وإذا خرَجَ شعبُكَ لِمُحارَبةِ أعدائِهِم حسَبَ مشيئَتِكَ، وصَلّوا إليكَ جِهَةَ المدينةِ الّتي اَختَرْتَها والهَيكلِ الّذي بَنَيتُهُ لاَسمِكَ،</span>
<span className="v" id="v35" data-v="35" data-xref="دانيال 9:17–19; المَزامِير 9:3–4; إرْمِيا 5:28"> فاَسمَعْ أنتَ مِنَ السَّماءِ صلاتَهُم وتضَرُّعَهُم واَنصُرْ حَقَّهُم.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="الجامِعَة 7:20; يَعقُوب 3:2; رسالَةُ يُوحَنّا الأُولَى 1:8–10"> ”وإذا خَطِئوا إليكَ، كما لا بُدَّ أنْ يَخطَأَ كُلُّ إنسانٍ، وغَضِبْتَ علَيهِم وهزَمْتَهُم أمامَ أعدائِهِم، فسَبَوهُم إلى بِلادٍ بعيدةٍ أو قريبةٍ</span>
<span className="v" id="v37" data-v="37" data-xref="إرْمِيا 31:18–20; دانيال 9:5–11; إشَعْياء 64:6–12"> ثُمَّ رجَعُوا إلى نُفوسِهِم في الأرضِ الّتي سُبُوا إليها، فَتابوا وتضَرَّعوا إليكَ وقالوا: خَطِئنا وأثِمْنا وأجرَمْنا،</span>
