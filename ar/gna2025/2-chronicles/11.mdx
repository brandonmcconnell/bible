## نبوءة شمعيا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المُلُوكِ الأوَّلُ 12:21–24; المَزامِير 33:10; المَزامِير 33:16"> ولمَّا وصَلَ رَحُبعامُ إلى أورُشليمَ جمَعَ مئَةً وثَمانينَ ألفَ جُنديٍّ باسِلٍ مِنْ كُلِّ بَيتِ يَهوذا وسِبْطِ بنيامينَ، ليُحاربوا بَيتَ إِسرائيلَ ويَردُّوهُم إلى طاعَةِ المَلِكِ.</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الثّانِي 12:15; أخبارِ الأيّامِ الثّانِي 12:5–7; أخبارِ الأيّامِ الثّانِي 8:14"> فقالَ اللهُ لِشَمَعْيا النَّبيّ:</span>
<span className="v" id="v3" data-v="3" data-xref="المُلُوكِ الثّانِي 17:34; التَّكوين 49:28; رُؤيا يُوحَنّا 7:4–8"> ”قُلْ لِرَحُبعامَ بنِ سُليمانَ مَلِكِ يَهوذا وكُلِّ بَيتِ يَهوذا وبنيامينَ وباقي الشَّعبِ:</span>
<span className="v" id="v4" data-v="4" data-xref="التَّكوين 50:20; المُلُوكِ الأوَّلُ 11:29–38; أخبارِ الأيّامِ الثّانِي 10:15–16"> هذا ما قالَ الرّبُّ: لا تُقاتِلوا إخوتَكُم بَيتَ إِسرائيلَ، بل عودوا إلى بُيوتِكُم لأنِّي أنا الّذي أرَدْتُ أنْ يَحدُثَ ما حدَثَ“. فسَمِعوا لِكلامِ الرّبِّ ورَجَعوا عَنْ مُحارَبةِ يَرُبعامَ.</span>
</div>

## رحبعام يحصن المدن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الثّانِي 8:2–6; أخبارِ الأيّامِ الثّانِي 14:6–7; إشَعْياء 22:8–11"> وأقامَ رَحُبعامُ بأورُشليمَ وبَنى مُدُنًا مُحَصَّنَةً في يَهوذا.</span>
<span className="v" id="v6" data-v="6" data-xref="نَحَمْيا 3:5; عامُوس 1:1; أخبارِ الأيّامِ الأوّلُ 4:32"> وهي بَيتُ لَحمَ وعيطمُ وتَقوعُ</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 15:35; صَمُوئيلَ الأوَّلُ 22:1; يَشُوع 15:58"> وبَيتُ صورَ وسوكو وعدُلَّامُ</span>
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 54:1; يَشُوع 15:24; صَمُوئيلَ الأوَّلُ 23:19"> وجتُّ ومريشَةُ وزيفُ</span>
<span className="v" id="v9" data-v="9" data-xref="يَشُوع 15:39; يَشُوع 15:35; يَشُوع 10:5"> وأدورايمُ ولخيشُ وعزيقَةُ</span>
<span className="v" id="v10" data-v="10" data-xref="العَدَد 13:22; يَشُوع 15:33; يَشُوع 14:14"> وصَرْعَةُ وأيلونُ وحبرونُ، وهذِهِ المُدُنُ كُلُّها في يَهوذا أو بنيامينَ</span>
<span className="v" id="v11" data-v="11" data-xref="أخبارِ الأيّامِ الثّانِي 17:19; إشَعْياء 22:10–11; أخبارِ الأيّامِ الثّانِي 11:23"> وعَزَّزَ الحُصونَ وجعَلَ فيها قادَةً ومَخازِنَ طعامٍ وزيتٍ وخمرٍ</span>
<span className="v" id="v12" data-v="12" data-xref="صموئيلَ الثّانِي 13:22; أخبارِ الأيّامِ الثّانِي 26:14–15; أخبارِ الأيّامِ الثّانِي 32:5"> وتروسًا ورِماحًا. وبِذَلكَ ثَبَّتَ سيطرَتَهُ على أرضِ يَهوذا وبنيامينَ سيطَرَةً تامَّةً.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13"> وجاءَ إلى رَحُبعامَ الكهَنةُ واللَّاويُّونَ الّذينَ في أرضِ إِسرائيلَ</span>
<span className="v" id="v14" data-v="14" data-xref="أخبارِ الأيّامِ الثّانِي 13:9; العَدَد 35:2–5; المُلُوكِ الأوَّلُ 12:28–33"> تاركينَ مَراعيَهم وسائِرَ أملاكِهِم جاؤُوا إلى يَهوذا وأورُشليمَ لأنَّ يربعامَ مَلِكَ إِسرائيلَ وبنيهِ الّذينَ خلَفوهُ مَنَعوهُم مِنْ مُمارَسةِ الكهنوتِ للرّبِّ،</span>
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الأوَّلُ 12:28; المُلُوكِ الأوَّلُ 12:31; المُلُوكِ الأوَّلُ 13:33"> ذلِكَ أنَّ يَرُبعامَ أقامَ كهنةً للمَذابحِ على المُرتفعاتِ بأصنامٍ مَصنوعةٍ على شَكلِ عُجولٍ وتُيوسٍ.</span>
<span className="v" id="v16" data-v="16" data-xref="أخبارِ الأيّامِ الثّانِي 15:9; اعمال الرسل 11:23; المَزامِير 84:5–7"> ولَحِقَ بالكهَنةِ مِنْ أسباطِ إِسرائيلَ جميعُ الّذينَ أرادوا بِكُلِّ قلوبِهِم أنْ يَعبُدوا الرّبَّ إلهَ إِسرائيلَ، فيُقَدِّموا الذَّبائحَ للرّبِّ إلهِ آبائِهِم في أورُشليمَ.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الثّانِي 12:1; أخبارِ الأيّامِ الثّانِي 1:1–12; أخبارِ الأيّامِ الثّانِي 7:17–19"> فعَزَّزوا مَملَكَةَ يَهوذا وسانَدوا رَحُبعامَ بنَ سليمانَ ثَلاثَ سِنينَ، لأنَّه سلَكَ في طريقِ داوُدَ وسليمانَ أبيهِ طُولَ تِلكَ المُدَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="صَمُوئيلَ الأوَّلُ 16:6; أخبارِ الأيّامِ الأوّلُ 27:18; صَمُوئيلَ الأوَّلُ 17:13"> وتزَوَّجَ رَحُبعامُ مَحْلَةَ بِنتَ يريموثَ بنِ داوُدَ مِنِ اَمرأتِهِ أبيجايلَ بِنتِ أليابَ بنِ يَسَّى،</span>
<span className="v" id="v19" data-v="19"> فوَلَدَت لَه ثَلاثةَ بَنينَ هُم: يعوشُ وشمريا وزاهَمُ.</span>
<span className="v" id="v20" data-v="20" data-xref="أخبارِ الأيّامِ الثّانِي 13:2; أخبارِ الأيّامِ الثّانِي 11:21; أخبارِ الأيّامِ الثّانِي 12:16"> ثُمَّ تزَوَّجَ مَعكَةَ بِنتَ أبشالومَ، فوَلَدَت لَه أبيَّا وعِتَّايَ وزيزا وشلوميتَ</span>
<span className="v" id="v21" data-v="21" data-xref="القُضاة 8:30; التَّثْنِيَة 17:17; المُلُوكِ الأوَّلُ 11:3"> وأحبَّ رَحُبعامُ مَعكَةَ بِنتَ أبشالومَ، أكثَرَ مِنْ جَميعِ زوجاتِهِ وجواريهِ. وكانَ لَه ثَمانيَ عَشْرةَ زوجةً وسِتُّونَ جاريةً، ووَلَدَ ثَمانيةً وعِشرينَ اَبنًا وسِتِّينَ بِنتًا.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 21:15–17; أخبارِ الأيّامِ الأوّلُ 29:1; أخبارِ الأيّامِ الأوّلُ 5:1–2"> وأقامَ رَحُبعامُ اَبنَهُ أبيَّا مِنْ زوجتِهِ مَعكَةَ رئيسًا على إخوتِهِ لأنَّه نوى أنْ يَجعَلَهُ خلفًا لَه</span>
<span className="v" id="v23" data-v="23" data-xref="أخبارِ الأيّامِ الثّانِي 21:3; التَّكوين 25:6; لُوقا 16:8"> وتصَرَّفَ بِفِطنَةٍ، فعيَّنَ جميعَ بَنيهِ حُكَّامًا في جميعِ المُدُنِ المُحَصَّنَةِ في أرضِ يَهوذا وبنيامينَ وأغدقَ علَيهِمِ المدَدَ بِسَخاءٍ، وأخذَ لهُم نِساءً كثيراتٍ.</span>
</div>
