## إنتقام اليهود من أعدائهم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="أسْتِير 3:13; أسْتِير 8:12; المَزامِير 30:11"> وفي اليوم الثَّالثَ عشَرَ مِنَ الشَّهرِ الثَّاني عشَرَ الّذي هوَ شهرُ آذارَ، لمَّا حانَ تنفيذُ أمرِ المَلِكِ وحُكْمِهِ واَرتجَى أعداءُ اليهودِ التسلُّطَ علَيهِم، اَنقلبَ ذلِكَ فكانَ أنْ تسلَّطَ اليهودُ على مُبغِضيهِم.</span>
<span className="v" id="v2" data-v="2" data-xref="أسْتِير 8:17; أسْتِير 8:11; المَزامِير 71:13"> واَجتمعَ اليهودُ في مُدُنِهِم، في كُلِّ أقاليمِ أحشويروشَ المَلِكِ، ليقبضوا على جميعِ الّذينَ يَطلبونَ الإساءةَ إليهم، فلم يقفْ أحدٌ في وُجوهِهِم، لأنَّ الخوفَ مِنهُم اَستولى على جميعِ الشُّعوبِ.</span>
<span className="v" id="v3" data-v="3" data-xref="عَزْرا 8:36; أسْتِير 8:5; دانيال 6:1–2"> وكانَ كُلُّ رؤساءِ الأقاليمِ والحُكَّامُ والولاةُ وعمَّالُ المَلِكِ يُساعدونَ اليهودَ، لأنَّ مَخافةَ مُرْدخايَ اَستولت علَيهِم.</span>
<span className="v" id="v4" data-v="4" data-xref="صموئيلَ الثّانِي 3:1; أخبارِ الأيّامِ الأوّلُ 11:9; صَمُوئيلَ الأوَّلُ 2:30"> كانَ نفوذُهُ كبيرًا في بَيتِ الملِكِ، وصيتُهُ يتعاظمُ في جميعِ الأقاليمِ.</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 18:34–40; المَزامِير 20:7–8; المَزامِير 149:6–9"> فضرَبَ اليهودُ جميعَ أعدائِهِم بالسَّيفِ وفتكوا بهِم كما شاؤوا.</span>
<span className="v" id="v6" data-v="6" data-xref="أسْتِير 3:15"> وفي شُوشنَ العاصمةِ قتلَ اليهودُ وأهلكوا خمسَ مئَةِ رَجلٍ،</span>
<span className="v" id="v7" data-v="7"> وقتلوا فرشنداتا ودلفونَ وأسقاتا</span>
<span className="v" id="v8" data-v="8"> وفوراتا وأدليا وأريداتا</span>
<span className="v" id="v9" data-v="9"> وفرمشتا وأريسايَ وأريدايَ وويزاتا</span>
<span className="v" id="v10" data-v="10" data-xref="أسْتِير 5:11; أسْتِير 8:11; التَّكوين 14:23"> وهُم عشَرَةُ أبناءِ هامانَ بنِ همداتا عدوِّ اليهودِ. قتلوهُم، لكنَّهُم لم يَمدُّوا أيديَهُم إلى غنيمةٍ.</span>
</div>
</div>

## مزيد من القتلى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11"> في ذلِكَ اليومِ رُفِعَ إحصاءٌ بعَددِ المقتولينَ في شُوشَنَ العاصمةِ إلى المَلِكِ،</span>
<span className="v" id="v12" data-v="12" data-xref="أسْتِير 7:2; أسْتِير 5:6"> فقالَ لأسْتِيرَ الملكةِ: ”قتلَ اليهودُ وأهلكوا في شُوشَنَ العاصمةِ وحدَها خَمسَ مئَةِ رجلٍ مِنْ بَينِهم بنو هامانَ العشَرَةُ، فماذا يكونونَ فَعلوا في باقي أقاليمِ المملكةِ؟ والآنَ فما مَطلبُكِ فأعطيه لكِ، وما مُرادُكِ بَعدُ فأقضيه؟“</span>
<span className="v" id="v13" data-v="13" data-xref="أسْتِير 8:11; التَّثْنِيَة 21:23; أسْتِير 9:15"> فقالَت أستيرُ: ”إنْ حَسُنَ عِندَ المَلِكِ، فَليُبِحْ لليهودِ الّذينَ في شُوشَنَ أنْ يفعلوا غدًا أيضًا كما فعلوا في هذا اليومِ، ويُعلِّقوا بَني هامانَ العشَرَةَ على خشَباتٍ“.</span>
<span className="v" id="v14" data-v="14"> فأمرَ المَلِكُ بذَلِكَ وأُذِيعَ أمرُهُ في شُوشَنَ. فعلَّقوا بني هامانَ العشَرَةَ.</span>
<span className="v" id="v15" data-v="15" data-xref="أسْتِير 8:11; أسْتِير 9:10; أسْتِير 9:16"> واَجتمعَ أيضًا اليهودُ الّذينَ في شُوشَنَ، في اليومِ الرَّابعَ عشَرَ مِنْ شهرِ آذارَ، وقتلوا ثلاثَ مئَةِ رَجلٍ في شُوشَنَ، لكنَّهُم لم يَمدُّوا أيدِيَهُم إلى غنيمةٍ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="أسْتِير 8:11; أسْتِير 9:2; اللّاويِّين 26:7–8"> واَجتمعَ سائرُ اليهودِ الّذينَ في أقاليمِ المَلِكِ للدِّفاعِ عَنْ أنْفُسِهِم والاستراحةِ مِنْ أعدائِهِم، فقتلوا خَمسةً وسبعينَ ألفًا، لكنَّهُم لم يَمدُّوا أيديَهُم إلى غَنيمةٍ.</span>
<span className="v" id="v17" data-v="17" data-xref="أسْتِير 9:21; أسْتِير 9:1; أسْتِير 8:9"> فعلوا ذلِكَ في اليومِ الثَّالثَ عشَرَ مِنْ شهرِ آذارَ واَستراحوا في اليومِ الرَّابعَ عشَرَ منهُ وجَعلوهُ يومَ وليمةٍ وفرحٍ.</span>
<span className="v" id="v18" data-v="18" data-xref="أسْتِير 9:21; أسْتِير 9:1; أسْتِير 9:15"> وأمَّا اليهودُ الّذينَ في شُوشَنَ فاَجتمعوا في الثَّالثَ عشَرَ مِنهُ وفي الرَّابعَ عشَرَ واَستراحوا في الخامسَ عشَرَ مِنهُ وجعلوهُ يومَ وليمةٍ وفرحٍ.</span>
<span className="v" id="v19" data-v="19" data-xref="أسْتِير 8:17; رُؤيا يُوحَنّا 11:10; نَحَمْيا 8:10–12"> لذلِكَ جعَلَ اليهودُ الّذينَ في القُرى، السَّاكنونَ مُدُنًا غَيرَ مُسوَّرةٍ، اليومَ الرَّابعَ عشَرَ مِنْ شهرِ آذارَ يومَ فرحٍ ووليمةٍ ويومَ خَيرٍ وتبادُلِ العطايا بَعضُهُم معَ بَعضٍ.</span>
</div>
</div>

## رسائل مردخاي إلى اليهود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 31:19–22; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:10–11; المَزامِير 145:4–12"> وكتَبَ مُرْدخايُ هذِهِ الأحداثَ وأرسلَ رسائلَ إلى جميعِ اليهودِ الّذينَ في جميعِ أقاليمِ المَلِكِ أحشويروشَ مِنْ أدناها إلى أقصاها.</span>
<span className="v" id="v21" data-v="21"> وأوجبَ علَيهِم أنْ يُعيِّدوا في الرَّابعَ عشَرَ مِنْ شهرِ آذارَ واليومِ الخامسَ عشَرَ مِنهُ في كُلِّ سَنةٍ،</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 30:11; إشَعْياء 14:3; أسْتِير 9:19"> في اليومَينِ اللَّذَينِ اَستراحَ فيهما اليهودُ مِنْ أعدائِهِم، في الشَّهرِ الّذي تَحوَّلَ لهُم الحُزنُ فيه إلى فَرحٍ والنُّواحُ إلى يومِ اَبتهاجٍ، ليجعلوهُما يومَيْ وليمةٍ وفَرحٍ وتبادُلِ العطايا، بَعضُهُم معَ بَعضٍ، والإحسانِ للفقراءِ.</span>
<span className="v" id="v23" data-v="23"> فاَتَّخذَ اليهودُ ما اَبتدأوا بإجرائِهِ وما كتبَ بهِ إليهِم مُرْدخايُ قانونًا لهُم.</span>
<span className="v" id="v24" data-v="24" data-xref="أسْتِير 3:5–13; أسْتِير 9:10"> ولمَّا كانَ هامانُ بنُ همداتا الأجاجيُّ عَدوَّ جميعِ اليهودِ ونوى علَيهِم ليُهلِكَهم وألقى فُورًا أيْ قرعةً ليُفنيَهُم ويُبيدَهم،</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 7:16; المَزامِير 140:9; المَزامِير 141:10"> لولا أستيرُ الّتي جاءت إلى أمامِ المَلِكِ، فأمرَ برسالةٍ مِنهُ أنْ يَرتدَّ على رأسِ هامانَ تدبيرُهُ الخَبيثُ الّذي دبَّرَهُ على اليهودِ وأنْ يُعلَّقَ هوَ وبنوهُ على الخشَباتِ</span>
<span className="v" id="v26" data-v="26" data-xref="أسْتِير 9:20; العَدَد 16:40; حِزْقِيال 39:11"> فلذلِكَ دعا اليهودُ هذينِ اليومَينِ فوريم مِنْ لفظةِ فُورَ،</span>
<span className="v" id="v27" data-v="27" data-xref="زَكَريّا 2:11; إشَعْياء 56:6; أسْتِير 8:17"> فَرضَ اليهودُ وأوجبوهُ على أنفُسِهِم وعلى نَسلِهِم وعلى كُلِّ مَنْ يَتَّصلُ بهِمْ أن لا يُبطِلَ تعييدَهُم بهذَينِ اليومَينِ ويُعيِّدَهما في كُلِّ جيلٍ</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 13:8–9; يَشُوع 4:7; المَزامِير 103:2"> وفي كُلِّ عشيرةٍ وكُلِّ إقليمٍ وكُلِّ مدينةٍ، وأنَّ يومَيْ فُوريمَ هذَينِ لا يُبطَلانِ مِنْ أعيادِ اليهودِ ولا يُمْحَى ذِكرُهُما عِندَ الأجيالِ الآتيةِ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="أسْتِير 9:20; أسْتِير 2:15; أسْتِير 8:10"> وكتبت أستيرُ المَلِكةُ بِنتُ أبيحائيلَ ومُرْدخايُ اليهوديُّ بكُلِّ سُلطةٍ لإثباتِ رسالةِ فُوريمَ الثَّانيةِ هذِهِ</span>
<span className="v" id="v30" data-v="30" data-xref="أسْتِير 1:1; أسْتِير 8:9; زَكَريّا 8:19"> وأرسَلا بالرَّسائِلِ إلى جميعِ اليهودِ في المئَةِ والسَّبعةِ والعِشرينَ إقليمًا مِنْ مَملَكةِ أحشويروشَ يتمنَّيانِ فيها لليهودِ السَّلامَ والأمانَ</span>
<span className="v" id="v31" data-v="31" data-xref="أسْتِير 4:3; أسْتِير 4:16; ﻳﻮﻧﺎﻥ 3:2–9"> ويَدعوَانهِم لإثباتِ يَومَيْ فُوريمَ هذَينِ في أوقاتِهِما كما فَرضَ مُرْدخايُ اليهوديُّ وأستيرُ المَلِكةُ، وكما أوجبوا على أنفُسِهِم وعلى أجيالِهِم الآتيةِ أمورَ الصِّيامِ والنُّواحِ.</span>
<span className="v" id="v32" data-v="32" data-xref="أسْتِير 9:26"> وأثبتَ أمرُ أستيرَ أحكامَ فُوريمَ هذِهِ وكُتِبَت في السِّجلِّ.</span>
