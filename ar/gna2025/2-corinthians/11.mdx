## بولس والرسل الكذّابون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:21; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:19"> لَيتَكُم تَحتَمِلوني، ولَو أظهَرْتُ قَليلًا مِنَ الحَماقَةِ. نعَمْ، اَحتَمِلوني!</span>
<span className="v" id="v2" data-v="2" data-xref="هُوشَع 2:19–20; أفَسُس 5:26–27; إشَعْياء 54:5"> فأنا أغارُ علَيكُم غَيرَةَ اللهِ لأنِّي خَطَبتُكُم لِرَجلٍ واحدٍ وهوَ المَسيحُ، لأُقَدِّمَكُم إلَيهِ عَذراءَ طاهِرَةً.</span>
<span className="v" id="v3" data-v="3" data-xref="كُولُوسِي 2:8; غَلاطِيَّة 1:6; أفَسُس 4:14"> لكنِّي أخافُ أنْ تَزوغَ بَصائِرُكُم عَنِ الصِّدقِ والوَلاءِ الخالِصِ للمَسيحِ، مِثلَ حَوّاءَ الّتي أغوَتْها الحَيَّةُ بِحيلَتِها.</span>
<span className="v" id="v4" data-v="4" data-xref="غَلاطِيَّة 1:6–8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:11; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:5"> فلَو جاءَكُم أحَدٌ يُبَشِّركُم بِيَسوعَ آخرَ غَيرِ الّذي بَشَّرناكُم بِه، أو يَعرُضُ علَيكُم رُوحًا غَيرَ الّذي نِلتُموهُ، وبِشارَةً غَيرَ الّتي تَلقَّيتُموها. لكُنتُم اَحتَملتُموهُ أحسنَ احتِمالٍ.</span>
<span className="v" id="v5" data-v="5" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:11–12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:10; غَلاطِيَّة 2:6–9"> ولا أظُنُّ أنِّي أقَلُّ شأْنًا مِنْ أُولئِكَ ”الرُّسُلِ العِظامِ!“</span>
<span className="v" id="v6" data-v="6" data-xref="أفَسُس 3:4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:2"> فإنْ أعوَزَتني الفَصاحَةُ، فلا تُعوِزُني المَعرِفَةُ. وهذا ما أظهَرناهُ لكُم جميعًا في كُلِّ شيءٍ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:13; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:9; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:1"> فهَلْ أَخطَأتُ حينَ حَمَلْتُ إلَيكُم مَجّانًا بِشارَةَ اللهِ وأذْلَلْتُ نَفسي لِتَرتَفِعوا أنتُم؟</span>
<span className="v" id="v8" data-v="8" data-xref="فِيلِبِّي 4:18; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:9; فِيلِبِّي 4:14–16"> حَرَمتُ كنائِسَ أُخْرى وأخَذتُ مِنها النَّفقَةَ لِخِدمَتِكُم.</span>
<span className="v" id="v9" data-v="9" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 3:8–9; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:9; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:6"> وما ثَقَّلتُ على أحدٍ مِنكُم حينَ كانَت بي حاجَةٌ وأنا بَينَكُم، فالإخوَةُ الّذينَ جاؤُوا مِنْ مكدونِيَّةَ سَدُّوا حاجَتي. وهكذا حَرَصتُ أنْ لا أُثَقِّلَ علَيكُم في شيءٍ وسأحرِصُ أيضًا.</span>
<span className="v" id="v10" data-v="10" data-xref="ﺭﻭﻣﻴﺔ 9:1; ﺭﻭﻣﻴﺔ 1:9; اعمال الرسل 18:12"> وبِحقِّ المَسيحِ فِيَّ، لَنْ يَسلُبَني أحدٌ هذِهِ المَفخَرَةَ في بلادِ أخائيةَ.</span>
<span className="v" id="v11" data-v="11" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:11–12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:3"> ولِماذا؟ ألأنِّي لا أُحِبُّكُم؟ اللهُ يَعلَمُ كَمْ أنا أُحِبُّكُم.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:12"> وما أعمَلُهُ الآنَ سأعمَلُهُ حتّى أُبطِلَ دَعوى الّذينَ يُحاوِلونَ أنْ يَدَّعوا فيما يُفاخِرونَ بِه أنَّهُم يَعمَلونَ مِثلَما نَعمَلُ.</span>
<span className="v" id="v13" data-v="13" data-xref="غَلاطِيَّة 1:7; رُؤيا يُوحَنّا 2:2; رسالَةُ يُوحَنّا الأُولَى 4:1"> هُمْ رُسُلٌ كَذّابونَ وعامِلونَ مُخادِعونَ يَظهَرونَ بِمَظهَرِ رُسُلِ المَسيحِ.</span>
<span className="v" id="v14" data-v="14" data-xref="غَلاطِيَّة 1:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3"> ولا عَجَبَ، فالشَّيطانُ نَفسُهُ يَظهَرُ بِمَظهَرِ مَلاكِ النُّورِ،</span>
<span className="v" id="v15" data-v="15" data-xref="فِيلِبِّي 3:19; رُؤيا يُوحَنّا 9:11; إشَعْياء 9:14–15"> فلا أقَلَّ مِنْ أنْ يَظهَرَ خَدَمُهُ بِمَظهَرِ الخَدَمِ الأبرارَ. هَؤلاءِ عاقِبَتُهُم على قَدْرِ أعمالِهِم.</span>
</div>

## آلام بولس في خدمة البشارة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:11"> أُكَرِّرُ القَولَ: لا يَظُنُّ أحدٌ أنِّي جاهِلٌ، وإلَّا فاَقبَلوني ولَو كَجاهِلٍ لِيكونَ لي شيءٌ أُفاخِرُ بِه.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:12; فِيلِبِّي 3:3–6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:4"> ما أقولُهُ هُنا لا أَقولُهُ وَفْقًا للرَّبِّ، بَلْ أقولُهُ كجاهِلٍ لَه الجُرأةُ أنْ يُفاخِرَ.</span>
<span className="v" id="v18" data-v="18" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:21–23; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:12–18"> ولكِنْ ما دامَ كثيرٌ مِنكُم يُفاخِرونَ مُفاخَرَةً بَشرِيَّةً، فلي أنا أيضًا أنْ أُفاخِرَ.</span>
<span className="v" id="v19" data-v="19" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:1"> وأنتُمُ العُقَلاءُ تَحتَمِلونَ الجُهَلاءَ بِسُرورٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="غَلاطِيَّة 2:4; غَلاطِيَّة 4:3; غَلاطِيَّة 5:1"> نعَمْ، تَحتَمِلونَ كُلَّ مَنْ يَستَعبِدُكُم ويَنهَبُكُم ويَسلُبُكُم ويَتكَبَّرُ علَيكُم ويَلطِمُكُم.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 10:1–2"> فيا لَلخَجَلِ! يَظهرُ أنَّنا كُنَّا ضُعفاءَ في هذا السَّبيلِ.<br />فالّذي يُباهونَ بِه – وكلامي كلامُ جاهِلٍ – أُباهي بِه أيضًا.</span>
<span className="v" id="v22" data-v="22" data-xref="فِيلِبِّي 3:5; ﺭﻭﻣﻴﺔ 11:1; ﺭﻭﻣﻴﺔ 9:4"> إنْ كانوا عِبرانيِّينَ فأنا عِبرانِيٌّ، أو إِسرائيليِّينَ فأنا إِسرائيليٌّ، أو مِنْ ذُرِّيَّةِ إبراهيمَ فأنا مِنْ ذُرِّيَّةِ إبراهيمَ.</span>
<span className="v" id="v23" data-v="23" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:4–5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:9–10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:9"> وإنْ كانوا خَدَمَ المَسيحِ – أقولُ هذا كأحمقَ – فأنا أفوقُهُم: في الجِهادِ جاهَدْتُ أكثرَ مِنهُم، في دُخولِ السُّجونِ قاسَيتُ أكثرَ مِنهُم، في الضَّرْبِ تَحمَّلْتُهُ أكثَرَ مِنهُم بِكثيرٍ وتَعرَّضتُ لِلمَوتِ مِرارًا،</span>
<span className="v" id="v24" data-v="24" data-xref="مَرْقُس 13:9; التَّثْنِيَة 25:2–3; ﻣﺘﻰ 10:17"> جَلَدَني اليَهودُ خَمسَ مَرَّاتٍ تِسعًا وثَلاثينَ جلدَةً.</span>
<span className="v" id="v25" data-v="25" data-xref="اعمال الرسل 14:19; اعمال الرسل 16:37; العِبرانِيّين 11:37"> وضرَبَني الرُّومانيُّونَ بِالعِصِيِّ ثلاثَ مَرَّاتٍ، ورَجَمَني النَّاسُ مَرَّةً واحدةً، واَنكسَرَتْ بيَ السَّفينَةُ ثلاثَ مَرَّاتٍ، وقَضَيتُ نَهارًا ولَيلَةً في عَرْضِ البَحرِ.</span>
<span className="v" id="v26" data-v="26" data-xref="اعمال الرسل 14:5; اعمال الرسل 17:5; اعمال الرسل 13:50"> وفي أسفاري المُتَعَدِّدةِ تَعرَّضتُ لِخَطَرِ الأنهارِ واللُّصوصِ، ولِخطَرِ اليَهودِ وغَيرِ اليَهودِ، وواجَهْتُ أخطارًا في المُدُنِ، وأخطارًا في البَراري، وأخطارًا في البَحرِ، وأخطارًا مِنَ الإخوَةِ الكَذّابينَ.</span>
<span className="v" id="v27" data-v="27" data-xref="فِيلِبِّي 4:12; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:5; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 2:9"> عانَيتُ الكَدَّ والتَّعَبَ والسَّهَرَ الدّائِمَ، والجُوعَ والعَطَشَ والصَّومَ الكثيرَ والبَردَ والعُريَ.</span>
<span className="v" id="v28" data-v="28" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:17; ﺭﻭﻣﻴﺔ 15:16; اعمال الرسل 20:18–35"> وهذا كُلُّهُ إلى جانِبِ ما أُعانيهِ كُلَّ يومٍ مِن اَهتمامٍ بِجميعِ الكنَّائِسِ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 8:13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:22; غَلاطِيَّة 4:8–20"> فمَنْ يَضعُفُ وأنا لا أضعُفُ معَهُ، ومَنْ يقَعُ في الخَطيئَةِ وأنا لا أحتَرِقُ مِنَ الحُزنِ علَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:5–11; الأمثال 25:27; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:3"> إنْ كانَ لا بُدَّ مِنَ الافتخارِ، فأنا أفتَخِرُ بِضُعفي.</span>
<span className="v" id="v31" data-v="31" data-xref="ﺭﻭﻣﻴﺔ 9:5; أفَسُس 1:3; يُوحَنّا 10:30"> واللهُ أبو رَبِّنا يَسوعَ – تَباركَ إلى الأبَدِ – يَعرِفُ أنِّي لا أكذِبُ،</span>
<span className="v" id="v32" data-v="32" data-xref="اعمال الرسل 9:24–25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:26"> وأنَّ والي المَلِكِ الحارِثِ على دِمشقَ أمَرَ بِحراسَةِ المدينةِ لِلقَبضِ علَيَّ.</span>
<span className="v" id="v33" data-v="33" data-xref="اعمال الرسل 9:25; صَمُوئيلَ الأوَّلُ 19:12; يَشُوع 2:18"> ولكِنَّ الإخوَةَ وضَعوني في قُفَّةٍ وأنزَلوني مِنْ كُوَّةٍ في السُّورِ، فنَجَوتُ مِنْ يَدَيْه.</span>
