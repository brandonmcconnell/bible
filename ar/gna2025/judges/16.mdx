## شمشون ودليلة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يَشُوع 15:47; عَزْرا 9:1–2; التَّكوين 38:16–18"> ثمَّ ذهَبَ شَمشونُ إلى غَزَّةَ فصادَفَ هُناكَ اَمرَأةً بَغيًّا، فدَخلَ علَيها.</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 118:10–12; صَمُوئيلَ الأوَّلُ 23:26; اعمال الرسل 9:24"> فقيلَ لأهلِ غَزَّةَ: ”شَمشونُ هُنا“، فأحاطوا بِالمكانِ وكمَنوا لَه كُلَّ اللَّيلِ عِندَ بابِ المدينةِ. وسَكَنوا اللَّيلَ كُلَّهُ وقالوا: ”عِندَ الصُّبحِ نقتُلُهُ“.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 2:24; المَزامِير 107:16; إشَعْياء 63:1–5"> فنامَ شَمشونُ إلى نِصفِ اللَّيلِ، ثُمَّ قامَ وأخذَ مِصراعَي بابِ المدينةِ بِقائِمتيهِ وقلَعَ البابَ ومِغلاقَهُ وحمَلَهُ على كَتِفِهِ وصَعِدَ إلى رأسِ الجبَلِ الّذي قُبالَةَ حبرونَ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:6; الأمثال 22:14; نَحَمْيا 13:26"> وبَعدَ ذلِكَ أحبَّ شَمشونُ اَمرَأةً في وادي سورَقَ اَسمُها دليلةُ.</span>
<span className="v" id="v5" data-v="5" data-xref="يَشُوع 13:3; القُضاة 14:15; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:9–10"> فصَعِدَ إليها زُعماءُ الفِلسطيِّينَ وقالوا لها: ”خادِعيهِ واَعرِفي بِماذا قوَّتُهُ العظيمةُ وكيفَ نتَمكَّنُ مِنهُ حتّى نوثِقَهُ ونتَغلَّبَ علَيهِ، وكُلُّ واحدٍ مِنَّا يدفَعُ إليكِ ألفًا ومئةً مِنَ الفِضَّةِ“.</span>
<span className="v" id="v6" data-v="6" data-xref="الأمثال 7:21; الأمثال 6:26; الأمثال 22:14"> فقالَت دليلةُ لِشَمشونَ: ”أخبِرني بِماذا قوَّتُكَ العظيمةُ، وبِماذا توثَقُ فتُغلَبُ؟“</span>
<span className="v" id="v7" data-v="7" data-xref="صَمُوئيلَ الأوَّلُ 21:2–3; ﺭﻭﻣﻴﺔ 3:8; صَمُوئيلَ الأوَّلُ 27:10"> فقالَ لها: ”إذا أوثَقوني بِسبعةِ أوتارٍ طَريئةٍ لم تَجِفَّ بَعدُ، فإنِّي أضعُفُ وأصيرُ كواحدٍ مِنَ النَّاسِ“.</span>
<span className="v" id="v8" data-v="8" data-xref="الجامِعَة 7:26"> فأعطاها الفِلسطيّونَ سبعةَ أوتارٍ طَريئةٍ لم تَجِفَّ بَعدُ، فشَدَّتهُ بِها،</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 58:9"> وكمَنوا لَه عِندَها في الغُرفةِ ثمَّ قالَت لَه: ”هجَمَ علَيكَ الفِلسطيّونَ يا شَمشونُ“. فقطَعَ شَمشونُ الأوتارَ كما يُقطَعُ خَيطُ الكتَّانِ إذا لامَسَتْهُ النَّارُ. ولم يَعلَمِ الفِلسطيّونَ بِماذا قوَّتُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="القُضاة 16:13; لُوقا 22:48; القُضاة 16:7"> فقالَت لَه دليلةُ: ”خدَعتَني وكَذبْتَ عليَّ، فأخبِرني الآنَ بِماذا توثَقُ؟“</span>
<span className="v" id="v11" data-v="11" data-xref="الأمثال 13:5; أفَسُس 4:25; الأمثال 13:3"> فقالَ لها: ”إنْ أوثَقوني بِحبالٍ جديدةٍ لم تُستَعمَلْ مِنْ قَبلُ، فإنِّي أضعُفُ وأصيرُ كواحدٍ مِنَ النَّاسِ“.</span>
<span className="v" id="v12" data-v="12"> فأخذَت دليلةُ حِبالًا جديدةً وأوثقَتْهُ بِها، وقالَت لَه: ”هجمَ علَيكَ الفِلسطيّونَ يا شَمشونُ“. وكانَ الكمينُ عِندَها مُختَبِئًا في الغُرفةِ، فقطَعَ شَمشونُ الحِبالَ عَنْ ذِراعَيهِ كما يُقطَعُ الخَيطُ.</span>
<span className="v" id="v13" data-v="13"> فقالَت لَه دليلةُ: ”إلى متى تَخدَعُني وتكذِبُ عليَّ، فأخبِرني بِماذا توثَقُ“. فقالَ لها: ”إذا نُسِجَتِ الخُصَلُ السَّبْعُ في رأسي بِالنَّولِ وشدَدتِها بِالوَتَدِ، فإنِّي أضعُفُ وأصيرُ كواحدٍ مِنَ النَّاسِ“.</span>
<span className="v" id="v14" data-v="14" data-xref="المَزامِير 106:43; عَزْرا 9:13–14"> وفيما هوَ نائِمٌ أخَذَت دليلةُ الخُصَلَ السَّبْعَ في رأسِهِ ونسَجتْها بِالنَولِ وشَدَّتها إلى الوتَدِ وقالَت لَه: ”هجَمَ علَيكَ الفِلسطيّونَ يا شَمشونُ“. فأفاقَ مِنْ نومِهِ وقَلَعَ وتَدَ النَّسيجِ وما فيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="القُضاة 14:16; رسالَةُ يُوحَنّا الأُولَى 2:15–16; يُوحَنّا 14:21–24"> فقالَت لَه دليلةُ: ”كيفَ تقولُ إنَّكَ تُحِبُّني وأنتَ لا تَثِقُ بي؟ وها أنتَ لِلمَرَّةِ الثَّالِثةِ تَخدَعُني ولا تُخبِرُني بِماذا قوَّتُكَ العظيمةُ“.</span>
<span className="v" id="v16" data-v="16" data-xref="الأمثال 7:21–23; لُوقا 18:5; مَرْقُس 14:24"> ولمَّا كانَت تُضايِقُه بِكلامِها كُلَّ يومٍ وتُلِحُّ علَيهِ ضاقَت نفْسُهُ حتّى الموتِ،</span>
<span className="v" id="v17" data-v="17" data-xref="ميخا 7:5; القُضاة 13:5; الأمثال 12:23"> فأطلَعَها على كُلِّ ما في قلبِهِ وقالَ لها: ”لم يُقَصَّ شَعري يومًا لأنِّي نَذيرُ اللهِ مِنْ بَطنِ أمِّي، فإنْ قُصَّ شَعري فارَقَتْني قوَّتي وضَعُفتُ وصِرتُ كواحدٍ مِنَ النَّاسِ“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="ﻣﺘﻰ 26:15; القُضاة 16:5; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:10"> وأحَسَّت دليلةُ أنَّهُ أخبَرَها بِكُلِّ ما في قلبِهِ، فأرسلَت ودَعَت زُعماءَ الفِلسطيِّينَ إليها وقالَت لهُم: ”تَعالَوا إليَّ هذِهِ المرَّةَ فإنَّهُ كاشَفَني بِكُلِّ ما في قلبِهِ“. فأتَوا إليها والِفضَّةُ بأيديهِم.</span>
<span className="v" id="v19" data-v="19" data-xref="الأمثال 7:26–27; الأمثال 7:21–23; الجامِعَة 7:26"> فأنامَتهُ على رُكبَتَيها ودَعَت رَجُلًا فحَلقَ الخُصَلَ السَّبْعَ في رأسِهِ وبَدأت بِتَعذيبِهِ بَعدَ أنْ فارَقَتهُ قُوَّتُهُ.</span>
<span className="v" id="v20" data-v="20" data-xref="العَدَد 14:42–43; صَمُوئيلَ الأوَّلُ 16:14; يَشُوع 7:12"> وقالَت لَه: ”هجَمَ علَيكَ الفِلسطيّونَ يا شَمشونُ“. فأفاقَ مِنْ نومِهِ وحَسِبَ أنَّهُ كما في السَّابِقِ سَينتَفِضُ في وجهِهِم ويُنقِذُ نفْسَهُ. وهوَ لا يَعلَمُ أنَّ الرّبَّ فارَقَهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="الأمثال 2:19; الأمثال 14:14; إشَعْياء 47:2"> فقبضَ علَيهِ الفِلسطيّونَ وقلَعوا عينَيهِ واَقتادوهُ إلى غَزَّةَ وقيَّدوهُ بِسلسلتَينِ مِنْ نُحاسٍ وجعَلوهُ يَطحَنُ في السِّجنِ.</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 107:13–14; المَزامِير 106:44–45; اللّاويِّين 26:44"> وأخذَ شَعرُ رأسِهِ ينمو مِنْ جديدٍ.</span>
</div>
</div>

## موت شمشون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="صَمُوئيلَ الأوَّلُ 5:2–5; الأمثال 24:17; المَزامِير 35:15–16"> فاَجتَمَعَ زُعماءُ الفِلسطيِّينَ لِيُقدِّموا ذبيحةً عظيمةً لِداجونَ إلهِهِم تعبيرًا عَنِ الفرَحِ، وقالوا: ”سَلَّمَ إلهُنا عَدُوَّنا شَمشونَ إلى أيدينا“.</span>
<span className="v" id="v24" data-v="24" data-xref="دانيال 5:4; دانيال 5:23; القُضاة 15:16"> ولمَّا رَآهُ الشَّعبُ مَجَّدوا إلهَهُم وقالوا: ”سَلَّمَ إلهُنا إلى أيدينا عَدُوَّنا الّذي خَرَّبَ أرضَنا وأكثرَ قتلانا“.</span>
<span className="v" id="v25" data-v="25" data-xref="صموئيلَ الثّانِي 13:28; القُضاة 19:6; القُضاة 9:27"> فلمَّا اَمتَلأت قُلوبُهُم زَهْوًا قالوا: ”هاتُوا شَمشونَ لِنَضحكَ علَيهِ“. فَجاؤُوا بِشَمشونَ مِنَ السِّجنِ وضَحِكوا علَيهِ وأقاموهُ بَينَ الأعمِدةِ.</span>
<span className="v" id="v26" data-v="26"> فقالَ شَمشونُ لِلصَّبيِّ الآخِذِ بِيدِهِ: ”دَعني ألمُسُ الأعمِدةَ الّتي يقومُ علَيها هذا البَيتُ حتّى أتَّكِئَ علَيها“.</span>
<span className="v" id="v27" data-v="27" data-xref="التَّثْنِيَة 22:8; صموئيلَ الثّانِي 11:2; يَشُوع 2:8"> وكانَ البَيتُ يَغُصُّ بِالرِّجالِ والنِّساءِ، وكانَ هُناكَ جميعُ زُعَماءِ الفِلسطيِّينَ وعلى السَّطحِ نحوَ ثَلاثةِ آلافٍ مِنَ الرِّجالِ والنِّساءِ يتفرَّجونَ على شَمشونَ وهُم يضحكونَ علَيهِ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="إرْمِيا 15:15; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 3:31–32; المَزامِير 143:12"> فدعا شَمشونُ الرّبَّ وقالَ: ”يا سيِّدي الرّبَّ اَذكُرني وشَدِّدني هذِهِ المرَّةَ أيضًا يا إلهي، لأنتَقِمَ لِعينَيَّ مِنَ الفِلسطيِّينَ دَفعةً واحدةً“.</span>
<span className="v" id="v29" data-v="29"> ثُمَّ قبضَ على العَمودينِ اللَّذَينِ في الوسَطِ، القائِمِ علَيهِما البَيتُ، آخِذًا أحدَهُما بِيمينِهِ والآخَرَ بِشمالِهِ واَتَّكَأَ علَيهِما</span>
<span className="v" id="v30" data-v="30" data-xref="فِيلِبِّي 2:8; ﻣﺘﻰ 16:25; ﻣﺘﻰ 24:38–39"> وقالَ: ”عليَّ وعلى الفِلسطيِّينَ لِيَسقُطِ البَيتُ“. ودفَعَ العَمودَينِ بِشِدَّةٍ فسقطَ البَيتُ على الزُّعماءِ وعلى جميعِ النَّاسِ الّذينَ في البَيتِ. فكانَ الموتى الّذينَ قتَلَهُم في موتِهِ أكثرَ مِنَ الّذينَ قتَلَهُم في حياتِهِ.</span>
<span className="v" id="v31" data-v="31" data-xref="القُضاة 15:20; القُضاة 13:25; القُضاة 13:2"> فنزَلَ إخوتُهُ وجميعُ أهلِ بَيتِهِ، فحَملوهُ وصَعِدوا بهِ ودَفَنوهُ بَينَ صَرعَةَ وأشتأولَ، في قبرِ مَنوحَ أبيهِ. وكانَ تَولَّى القضاءَ على بَني إِسرائيلَ عِشرينَ سنَةً.</span>
</div>
