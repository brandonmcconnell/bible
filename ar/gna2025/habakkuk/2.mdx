## الرّبّ يجيب حبقوق

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 21:8; إشَعْياء 62:6; صموئيلَ الثّانِي 18:24"> أقِفُ على مَرصَدي وأنتَصِبُ على حِصْني وأرقُبُ لأرى ماذا يقولُ ليَ الرّبُّ، وماذا يُجيبُ عَنْ شكوايَ.</span>
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 30:8; رُؤيا يُوحَنّا 14:13; رُؤيا يُوحَنّا 1:18–19"> فأجابَني الرّبُّ: ”أُكتُبْ هذِهِ الرُّؤْيا واَنْقُشْها على الألواحِ حتّى تَسهُلَ قِراءَتُها،</span>
<span className="v" id="v3" data-v="3" data-xref="المَزامِير 27:14; العِبرانِيّين 10:36–37; حِزْقِيال 12:25"> لأنَّ الرُّؤيا مَرهونةٌ بِوقتِها، وعِندَما يَحينُ وقتُها تَجيءُ ولا تكذِبُ. إنْ أبطَأَتْ فاَنْتَظِرْها، فهيَ لا بُدَّ أنْ تَجيءَ ولا تتَأخَّرَ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﺭﻭﻣﻴﺔ 1:17; العِبرانِيّين 10:38; غَلاطِيَّة 3:11–12"> وهذا نَصُّها: مَنْ كانَت نفْسُهُ شِرِّيرةً لا ينجو. أمَّا البارُّ فبِإيمانِهِ يحيا“.</span>
</div>
</div>

## الويل للأشرار

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="إشَعْياء 5:11–12; الأمثال 27:20; الأمثال 20:1"> الثَّروةُ غادِرةٌ، والرَّجُلُ الجَشِعُ مُتَكبِّرٌ لا يَهدَأُ. يُوَسِّعُ حُلْقومَهُ كالهاويةِ ويكونُ كالموتِ لا يشبعُ. يجمعُ إليهِ كُلَّ الأُمَمِ ويَضُمُّ جميعَ الشُّعوبِ.</span>
<span className="v" id="v6" data-v="6" data-xref="العَدَد 23:7; ميخا 2:4; إرْمِيا 50:13"> ألا يتَّخِذُهُ هؤلاءِ كُلُّهُم مثَلًا وهُزأَةً وأُحدوثةً ويقولونَ: ”وَيلٌ لِمَنْ يُكثِرُ مِمَّا لا يَحِقُّ لَه، وإلى متى يُرهِقُ الشَّعبَ بِالضَّرائِبِ؟</span>
<span className="v" id="v7" data-v="7" data-xref="الأمثال 29:1; إشَعْياء 47:11; إرْمِيا 8:17"> ألا يقومونَ علَيهِ بَغتَةً ويَرفُضونَ دَفْعَ ضَرائِبِهِم ويُبْطِلونَها، فيكونُ مَوضِعَ شَماتةٍ لهُم؟</span>
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 33:1; حَبَقُّوق 2:17; إرْمِيا 27:7"> سَلَبْتَ أمَمًا كثيرينَ، والآنَ فجميعُ بَقيَّةِ الشُّعوبِ يَسلُبونَكَ. أمَا سَفكْتَ دِماءَ البشَرِ وأنزَلْتَ الجَورَ بِأهلِ الأرضِ ومُدُنِها جميعًا؟</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="إرْمِيا 49:16; إرْمِيا 22:13–19; إشَعْياء 28:15"> وَيلٌ لِمَنْ يكسِبُ مَكسَبًا شرِّيرًا لِبَيتِهِ ويَظُنُّ أنَّهُ بِذلِكَ يَجعَلُهُ كالعُشِّ في العَلاءِ، فيَسلَمُ مِنْ قبضَةِ الشَّرِّ.</span>
<span className="v" id="v10" data-v="10" data-xref="ناحُوم 1:14; المُلُوكِ الثّانِي 9:26; إرْمِيا 36:31"> جلَبْتَ البُؤسَ على بَيتِكَ، وقضَيتَ على شُعوبٍ كثيرينَ، فتَسبَّبتَ في خسارَةِ نفْسِكَ.</span>
<span className="v" id="v11" data-v="11" data-xref="لُوقا 19:40; يَشُوع 24:27; أيُّوب 31:38–40"> حتّى الحِجارةُ تَصرخُ علَيكَ مِنَ الحائِطِ فتُرَدِّدُ صَدى صَرْخَتِها رَوافِدُ الخشَبِ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="ميخا 3:10; دانيال 4:27–31; ناحُوم 3:1"> وَيلٌ لِمَنْ يَبني مدينةً بِالدِّماءِ ويُؤسِّسُها بِالإِثْم.</span>
<span className="v" id="v13" data-v="13" data-xref="إرْمِيا 51:58; إشَعْياء 50:11; الأمثال 21:30"> أمَا هيَ مشيئةُ الرّبِّ القديرِ أنْ يذهبَ تَعَبُ الشُّعوبِ إلى النَّارِ وجَهْدُ الأُمَمِ إلى الباطِلِ؟</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 11:9; المَزامِير 22:27; المَزامِير 72:19"> لكنَّ الأرضَ ستَمتَلِئُ بِمعرِفةِ مَجدِ الرّبِّ كما تَغمُرُ المياهُ البحرَ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="إرْمِيا 51:7; التَّكوين 9:22; رُؤيا يُوحَنّا 18:3"> وَيلٌ لِمَنْ يسقي جيرانَهُ مِنْ كأسِ غضَبِهِ ويُسكِرُهُم ليَنظُرَ إلى عَوراتِهِم.</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 47:3; ناحُوم 3:5–6; إشَعْياء 51:21–23"> شَبِعْتَ هَوانًا بدَلَ المَجدِ، فاَشرَبْ أنتَ أيضًا لِتَظهَرَ عَورتُكَ، فكأسُ يَمينِ الرّبِّ تَدورُ علَيكَ وقَيْءُ الهَوانِ يُغَطِّي مَجدَكَ.</span>
<span className="v" id="v17" data-v="17" data-xref="حَبَقُّوق 2:8; المَزامِير 55:23; زَكَريّا 11:1"> جَورُكَ على لبنانَ يَغمُرُكَ، بل حتّى هَلاكُ بَهائِمِه يُرعِبُكَ. أما سَفكْتَ دِماءَ البشَرِ وظَلَمْتَ أهلَ الأرضِ ومُدُنَها جميعًا؟</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 10:8; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:2; إشَعْياء 44:9–10"> ماذا يَنفَعُ صَنَمٌ مَنحوتٌ حتّى يَنحَتَهُ صانِعُهُ، أو تِمثالٌ مَسبوكٌ حتّى يصنَعَهُ؟ أيَتَّكِلُ علَيهِ وهوَ أبكَمُ ولا يُوحي إلَّا بِالكذِبِ؟</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="المَزامِير 135:17; إرْمِيا 10:4; المُلُوكِ الأوَّلُ 18:26–29"> وَيلٌ لِمَنْ يقولُ لِصَنَمٍ مِنَ الخشَبِ اَسْتَيقِظْ، أو لِتِمثالٍ مِنَ الحجَرِ الصَّامِتِ: اَنتَبِه. أفي قُدْرَتِهِ أنْ يُريَكَ شيئًا؟ رُبَّما كانَ مَطلِيًّا بِالذَّهبِ والفِضَّةِ، ولكنْ لا روحَ فيهِ.</span>
<span className="v" id="v20" data-v="20" data-xref="زَكَريّا 2:13; صَفَنْيا 1:7; المَزامِير 46:10"> أمَّا الرّبُّ فهوَ في هَيكلِهِ المُقَدَّسِ، فاَسْكُتوا أمامَ وجهِهِ يا جميعَ أهلِ الأرضِ.</span>
</div>
