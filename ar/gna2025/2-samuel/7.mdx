## نبوءة ناثان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الأمثال 16:7; لُوقا 1:74–75; المُلُوكِ الأوَّلُ 5:4"> ولمَّا سكنَ المَلِكُ في قصرِه وأراحَهُ الرّبُّ مِنْ جميعِ أعدائِهِ</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الأوّلُ 16:1; صموئيلَ الثّانِي 5:11; حَجَّي 1:4"> قالَ لناثانَ النَّبيِّ: ”أُنظُرْ. أنا مُقيمٌ في بيتٍ مِنْ أرزٍ، وتابوتُ العَهدِ مُقيمٌ في خَيمةٍ“.</span>
<span className="v" id="v3" data-v="3" data-xref="أخبارِ الأيّامِ الأوّلُ 22:7; المُلُوكِ الأوَّلُ 8:17–18; المُلُوكِ الثّانِي 4:27"> فأجابَهُ ناثانُ: ”إذهَبْ واَفعَلْ كُلَّ ما في بالِكَ لأنَّ الرّبَّ معَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="أخبارِ الأيّامِ الأوّلُ 17:3; العَدَد 12:6; عامُوس 3:7"> ولكنَّ الرّبَّ قالَ لناثانَ في تلكَ اللَّيلةِ:</span>
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الأوّلُ 23:3–32; المُلُوكِ الأوَّلُ 5:3–4; المُلُوكِ الأوَّلُ 8:16–19"> ”إذهَبْ وقُلْ لِعبدي داوُدَ: هذا ما يقولُ الرّبُّ: أأنتَ تَبني لي بَيتًا لِسُكنايَ؟</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الأوَّلُ 8:16; يَشُوع 18:1; الخُرُوج 40:18–19"> ما سَكنتُ بَيتًا مِنْ يومِ أخرجتُ بَني إِسرائيلَ مِنْ مِصْرَ حتّى الآنَ، بل في خَيمةٍ كُنتُ أنتَقِلُ معَهُم على الدَّوامِ،</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الأوّلُ 17:6; صموئيلَ الثّانِي 5:2; اللّاويِّين 26:11–12"> وفي كُلِّ اَرتحالي معَ جميعِ بَني إِسرائيلَ لم أسألْ أحدًا مِنْ رُؤسائِهِمِ الّذينَ أمرتُهُم أنْ يَرعَوا بَني إِسرائيلَ شعبي: لِماذا لم تُقِمْ لي بَيتًا مِنَ الأرْزِ؟</span>
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 78:70–71; صموئيلَ الثّانِي 6:21; صموئيلَ الثّانِي 12:7"> فقُلْ لِعبدي داوُدَ: هذا ما يقولُ الرّبُّ القديرُ: أنا أخَذتُكَ مِنَ الحظيرةِ، مِنْ وراءِ الغنَمِ، لتكونَ رئيسًا على شعبي بَني إِسرائيلَ،</span>
<span className="v" id="v9" data-v="9" data-xref="صموئيلَ الثّانِي 5:10; أخبارِ الأيّامِ الأوّلُ 17:8; صَمُوئيلَ الأوَّلُ 18:14"> وكُنتُ معَكَ حيثُما سِرتَ، وأبَدْتُ جميعَ أعدائِكَ وسأُقيمُ لكَ اَسمًا عظيمًا كأسماءِ العُظَماءِ في الأرضِ.</span>
<span className="v" id="v10" data-v="10" data-xref="عامُوس 9:15; أخبارِ الأيّامِ الأوّلُ 17:9; إرْمِيا 24:6"> وسأجعَلُ مكانًا لشعبي بَني إِسرائيلَ، وفيهِ أغرُسُهُ، فيثبُتُ في مكانِهِ فلا يتَزَعزَعُ مِنْ بَعدُ، ولا يعودُ بَنو الشَّرِّ يضطَهِدونَهُ كما كانوا مِنْ قَبلُ،</span>
<span className="v" id="v11" data-v="11" data-xref="صموئيلَ الثّانِي 7:27; صموئيلَ الثّانِي 7:1; أخبارِ الأيّامِ الأوّلُ 17:10"> يومَ أقَمتُ قُضاةً علَيهِ. سأُريحُكَ مِنْ جميعِ أعدائِكَ. وأنا الرّبُّ أُخبِرُكَ أنِّي سأُقيمُ لكَ ذُرِّيَّةً.</span>
<span className="v" id="v12" data-v="12" data-xref="المُلُوكِ الأوَّلُ 8:20; اعمال الرسل 2:30; اعمال الرسل 13:36"> وإذا اَنتهَتْ أيّامُكَ ورقدتَ معَ آبائِكَ، أقمتُ خلَفًا لكَ مِنْ نسلِكَ الّذي يخرُجُ مِنْ صُلبِكَ وثبَّتُّ مُلكَهُ،</span>
<span className="v" id="v13" data-v="13" data-xref="لُوقا 1:31–33; المُلُوكِ الأوَّلُ 8:19; صموئيلَ الثّانِي 7:16"> فهوَ يَبني بَيتًا لاَسمي وأنا أُثَبِّتُ عرشَ مُلْكِه إلى الأبدِ.</span>
<span className="v" id="v14" data-v="14" data-xref="العِبرانِيّين 1:5; رُؤيا يُوحَنّا 3:19; أخبارِ الأيّامِ الأوّلُ 17:13"> أنا أكونُ لَه أبًا وهوَ يكونُ لي اَبنًا، وإذا فعَلَ الشَّرَّ أُؤدِّبُهُ بِعصًا كالّتي يستخدِمُها النَّاسُ وبها يَضرِبونَ.</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 15:23; صَمُوئيلَ الأوَّلُ 15:28; المَزامِير 89:28"> وأمَّا رحمَتي فلا أنزعُها عَنهُ كما نَزعتُها عَنْ شاوُلَ الّذي أزلْتُهُ مِنْ أمامِ وجهِكَ،</span>
<span className="v" id="v16" data-v="16" data-xref="لُوقا 1:32–33; المَزامِير 89:36–37; صموئيلَ الثّانِي 7:13"> بل يكونُ بَيتُكَ ومُلكُكَ ثابِتَينِ على الدَّوامِ أمامَ وجهي، وعرشُكَ يكونُ راسخًا إلى الأبدِ“.</span>
<span className="v" id="v17" data-v="17" data-xref="أخبارِ الأيّامِ الأوّلُ 17:15; اعمال الرسل 20:27; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:3"> فكَلَّمَ ناثانُ داوُدَ بِجميعِ هذا الكلامِ وهذِهِ الرُّؤيا كُلِّها.</span>
</div>
</div>

## صلاة داود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="المَزامِير 8:4; أخبارِ الأيّامِ الأوّلُ 17:16; صَمُوئيلَ الأوَّلُ 18:18"> فدخَلَ المَلِكُ داوُدُ وجلَسَ أمامَ الرّبِّ وقالَ: ”مَنْ أنا يا سيِّدي الرّبُّ، وما بَيتي الّذي أنتَسِبُ إليهِ حتّى أوصَلتَني إلى هُنا.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 55:8–9; أخبارِ الأيّامِ الأوّلُ 17:17; المَزامِير 36:7"> وكانَ هذا كُلُّه قليلًا في عينَيكَ يا سيِّدي الرّبُّ، فدَعوتَني أيضًا، أنا عبدُكَ، أنْ يدومَ نسلي إلى زمانٍ طويلٍ، وهذا خِلافًا لِسُنَّةِ البشَرِ أيُّها السَّيِّدُ الرّبُّ.</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 21:17; صَمُوئيلَ الأوَّلُ 16:7; المَزامِير 139:1"> فبماذا أعودُ أُكلِّمُكَ وأنتَ عرَفتَني أيُّها السَّيِّدُ الرّبُّ؟</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 115:1; المَزامِير 138:2; أفَسُس 1:9"> فمِنْ أجلِ وعدِكَ ومَحبَّةِ قلبِكَ فعلتَ هذا الأمرَ العظيمَ وأعلَمتَني بهِ.</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 86:10; الخُرُوج 15:11; المَزامِير 86:8"> فما أعظَمَكَ أيُّها الرّبُّ الإلهُ. أنتَ لا مثيلَ لكَ ولا إلهَ سِواكَ، كما سَمِعنا بآذانِنا.</span>
<span className="v" id="v23" data-v="23" data-xref="التَّثْنِيَة 9:26; التَّثْنِيَة 10:21; التَّثْنِيَة 15:15"> وأيَّةُ أمَّةٍ في الأرضِ مِثلُ شعبِكَ بَني إِسرائيلَ الّذينَ اَختَرتَهُم لِنَفسِكَ شعبًا وجعَلتَ لهُم اَسمًا، وعمِلتَ لهُم تِلكَ الأعمالَ العظيمةَ الهائلَةَ، حيثُ طردتَ مِنْ أمامِهِمِ الشُّعوبَ وآلِهتَهُم،</span>
<span className="v" id="v24" data-v="24" data-xref="التَّثْنِيَة 26:18; رسالَةُ بُطرُسَ الأُولَى 2:10; التَّكوين 17:7"> وثبَّتَّ شعبَكَ بَني إِسرائيلَ شعبًا لكَ إلى الأبدِ، وأنتَ يا ربُّ صرتَ لهُم إلهًا</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 119:49; حِزْقِيال 36:37; التَّكوين 32:12"> والآنَ يا ربُّ أقِمْ إلى الأبدِ وعدَك الّذي وعدتَني أنا ونسلي. واَفعَلْ كما قُلتَ</span>
<span className="v" id="v26" data-v="26" data-xref="أخبارِ الأيّامِ الأوّلُ 17:23–24; المَزامِير 72:18–19; ﻣﺘﻰ 6:9"> ليَعظُمَ اَسمُكَ إلى الأبدِ ويُقالَ: الرّبُّ القديرُ إلهٌ على إِسرائيلَ. وليكُنْ بَيتُ عبدِكَ داوُدَ ثابِتًا أمامَكَ،</span>
<span className="v" id="v27" data-v="27" data-xref="أخبارِ الأيّامِ الأوّلُ 17:25–26; المَزامِير 10:17; راعُوث 4:4"> لأنَّكَ أنتَ، أيُّها الرّبُّ القديرُ إلهُ إِسرائيلَ، أعلَنتَ ذلِكَ على مَسمَعي حينَ قُلتَ: أبني لكَ بَيتًا. لذلِكَ تشَجَّعَ قلبي، أنا عبدُكَ أنْ أُصلِّيَ إليكَ هذِهِ الصَّلاةَ.</span>
<span className="v" id="v28" data-v="28" data-xref="يُوحَنّا 17:17; الخُرُوج 34:6; العَدَد 23:19"> والآنَ أيُّها الرّبُّ الإلهُ، أنتَ هوَ اللهُ وكلامُكَ حَقٌّ، وأنتَ الّذي وعَدْتَني هذا الوعدَ الحسَنَ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 6:24–26; أخبارِ الأيّامِ الأوّلُ 17:27; المَزامِير 115:12–15"> فالآنَ تحَنَّنْ وبارِكْ نسلي لِيكونَ أمامَكَ على الدَّوامِ، لأنَّكَ أنتَ يا سيِّدي الرّبُّ تَكلَّمتَ، ومِنْ برَكتِكَ يتبارَكُ نسلُ عبدِكَ داوُدَ إلى الأبدِ“.</span>
</div>
