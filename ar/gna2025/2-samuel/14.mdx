## رجوع أبشالوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صموئيلَ الثّانِي 13:39; صموئيلَ الثّانِي 2:18; صموئيلَ الثّانِي 18:33"> وعرفَ يوآبُ بنُ صرويَّةَ أنَّ داوُدَ اَشتاقَ إلى أبشالومَ،</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الثّانِي 11:6; عامُوس 1:1; راعُوث 3:3"> فأرسَلَ إلى تَقوعَ وجاءَ مِنْ هُناكَ باَمرأةٍ حكيمةٍ وقالَ لها: ”تظاهري بالحُزنِ، واَلبَسي ثيابَ الحِدادِ ولا تَتزيَّني، بل كُوني كاَمرأةٍ تَنوحُ مِنْ أيّامٍ عديدةٍ على مَيْتٍ،</span>
<span className="v" id="v3" data-v="3" data-xref="صموئيلَ الثّانِي 14:19; الخُرُوج 4:15; إشَعْياء 51:16"> ورُوحي إلى المَلِكِ وكلِّميهِ بِما أقولُهُ لكِ“. وقالَ لها يوآبُ ما تقولُ للمَلِكِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="صموئيلَ الثّانِي 1:2; المُلُوكِ الثّانِي 6:26–28; صَمُوئيلَ الأوَّلُ 25:23"> فذهبَتِ المَرأةُ إلى المَلِكِ واَنحَنَت إلى الأرضِ وقالَت: ”ساعِدْني أيُّها المَلِكُ“.</span>
<span className="v" id="v5" data-v="5" data-xref="القُضاة 9:8–15; صموئيلَ الثّانِي 12:1–3"> فسألَها: ”ما حاجتُكِ؟“ فأجابَت: ”أنا أرملَةٌ توفِّيَ زوجي.</span>
<span className="v" id="v6" data-v="6" data-xref="التَّكوين 4:8; التَّثْنِيَة 22:26–27; الخُرُوج 2:13"> وكانَ لي يا سيِّدي اَبنانِ تشاجَرا في البَرِّيَّةِ، ولم يكُنْ مَنْ يَفصِلُ بَينَهُما، فضربَ أحدُهُما الآخَرَ وقتَلَهُ.</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 35:19; ﻣﺘﻰ 21:38; صموئيلَ الثّانِي 12:17"> فهجَمَ جميعُ العَشيرةِ عليَّ وقالوا: سلِّمي إلينا قاتِلَ أخيهِ لِنَقتُلَهُ اَنتِقامًا لَه، فإذا فَعَلوا ذلِكَ يُهلِكونَ الوارِثَ أيضًا ويُطفِئونَ جمرَةَ أملي الّتي بَقيَت، ولا يَترُكونَ لِزَوجي اَسمًا ولا نسلًا على وجهِ الأرضِ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الأمثال 18:13; إشَعْياء 11:3–4; أيُّوب 29:16"> فقالَ المَلِكُ للمرأةِ: ”عُودي إلى بَيتِكِ، فأنا أقومُ بالمطلوبِ“.</span>
<span className="v" id="v9" data-v="9" data-xref="صَمُوئيلَ الأوَّلُ 25:24; المُلُوكِ الأوَّلُ 2:33; ﻣﺘﻰ 27:25"> فقالَتِ المرأةُ: ”عليَّ اللَّومُ يا سيِّدي المَلِكُ وعلى بَيتِ أبي في كُلِّ ما تفعَلُ، وليكُنِ المَلِكُ وعرشُهُ بَريئَينِ“.</span>
<span className="v" id="v10" data-v="10"> فقالَ لها المَلِكُ: ”إذا تَهدَّدكِ أحدٌ، فَجيئي بهِ إليَّ، فلا يتعَرَّضْ لكِ مِنْ بَعدُ“.</span>
<span className="v" id="v11" data-v="11" data-xref="صَمُوئيلَ الأوَّلُ 14:45; ﻣﺘﻰ 10:30; العَدَد 35:19"> فقالَت لَه: ”أكِّدْ لي عونَكَ، واَجعَلِ الرّبَّ إلهَكَ شاهِدًا أيُّها المَلِكُ، فلا يرتكِبُ الّذي يثأرُ لدَمِ اَبني جريمةً أُخرى ويُهلِكُ اَبني الآخرَ“. فقالَ لها: ”حَيٌّ هوَ الرّبُّ، لا تسقُطُ شعرةٌ مِنْ رأسِ اَبنِكِ على الأرضِ“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 18:27; صَمُوئيلَ الأوَّلُ 25:24; اعمال الرسل 26:1"> فقالَتِ المَرأةُ: ”دَعني يا سيِّدي المَلِكُ أقولُ شيئًا آخرَ“. قالَ: ”تكلَّمي“.</span>
<span className="v" id="v13" data-v="13" data-xref="صموئيلَ الثّانِي 13:37–38; صموئيلَ الثّانِي 12:7; القُضاة 20:2"> فقالَتِ المَرأةُ: ”لماذا نوَيتَ شرًّا مُماثِلًا على شعبِ اللهِ؟ فأنتَ بِما حكَمتَ لي، أيُّها المَلِكُ، حكَمتَ على نفْسِكَ لأنَّكَ لا ترُدُّ اَبنَكَ الّذي نَفيتَهُ.</span>
<span className="v" id="v14" data-v="14" data-xref="أيُّوب 30:23; العَدَد 35:28; العِبرانِيّين 9:27"> لا بُدَّ أنْ نموتَ كُلُّنا ونكونَ كماءٍ مَصبوبٍ على الأرضِ، ولا يجمَعُهُ شيءٌ لكنَّ اللهَ لا يُريدُ أنْ يَقضيَ على حياةِ أحدٍ، ولا هوَ يُريدُ أنْ يظَلَّ منفِيٌّ بعيدًا عَنهُ في مَنفاهُ.</span>
<span className="v" id="v15" data-v="15"> والآنَ فأنا جِئتُ لأكلِّمَ المَلِكَ سيِّدي بِهذا الأمرِ لأنَّ الشَّعبَ هَدَّدني، فقُلتُ في نفْسي: أُكلِّمُ المَلِكَ لعلَّهُ يُساعدُني.</span>
<span className="v" id="v16" data-v="16" data-xref="صَمُوئيلَ الأوَّلُ 26:19"> فالمَلِكُ يسمعُ لي ويُنقِذُني مِنْ يدِ الرَّجلِ الّذي يُريدُ أنْ يُزيلَني أنا واَبنيَ مِنَ الأرضِ الّتي أورَثَها اللهُ شعبَهُ.</span>
<span className="v" id="v17" data-v="17" data-xref="صَمُوئيلَ الأوَّلُ 29:9; صموئيلَ الثّانِي 19:27; صموئيلَ الثّانِي 14:20"> وقُلتُ: ليكُنْ وعدُ سيِّدي المَلِكِ عزاءً لي، لأنَّكَ يا سيِّدي كمَلاكِ اللهِ في فَهمِ الخيرِ والشَّرِّ، والرّبُّ إلهُكَ يكونُ معَكَ“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="إرْمِيا 38:25; صَمُوئيلَ الأوَّلُ 3:17–18; إرْمِيا 38:14"> فأجابَها المَلِكُ: ”لا تكتُمي عنِّي شيئًا مِمَّا أسألُكِ“. فقالتِ المَرأةُ: ”ليتكَلَّمْ سيِّدي المَلِكُ“.</span>
<span className="v" id="v19" data-v="19" data-xref="صموئيلَ الثّانِي 14:3; صَمُوئيلَ الأوَّلُ 1:26; صَمُوئيلَ الأوَّلُ 20:3"> فقالَ: ”هل ليوآبَ يَدٌ في هذا كُلِّه؟“ فأجابَت ”وحياتِكَ يا سيِّدي المَلِكُ لا أحيدُ يَمينًا ولا يَسارًا عَنْ قولِ الحقيقةِ لكَ. نعم، يوآبُ قائدُ جيشِكَ هوَ الّذي أمرَني ولقَّنَني كُلَّ هذا الكلامِ.</span>
<span className="v" id="v20" data-v="20" data-xref="صموئيلَ الثّانِي 14:17; صموئيلَ الثّانِي 19:27; أيُّوب 32:21–22"> وهوَ لَجأَ إلى هذِهِ الطَّريقةِ لتسويةِ هذا الأمرِ، ولسيِّدي حِكمَةٌ كحِكمةِ ملاكِ اللهِ في فَهمِ جميعِ ما يَحدُثُ في الأرضِ“.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="صموئيلَ الثّانِي 14:11; صَمُوئيلَ الأوَّلُ 14:39; مَرْقُس 6:26"> وبَعدَ ذلِكَ قالَ المَلِكُ ليوآبَ: ”ها أنا فاعلٌ ما تُريدُ. فاَذهبْ ورُدَّ الفتى أبشالومَ“.</span>
<span className="v" id="v22" data-v="22" data-xref="راعُوث 2:2; أيُّوب 31:20; أيُّوب 29:11"> فاَنحَنى يوآبُ بوجهِهِ إلى الأرضِ. ودَعا للمَلِكِ وقالَ: ”اليومَ علِمتُ أنِّي أنعَمُ برِضاكَ يا سيِّدي المَلِكُ، لأنَّكَ فعلتَ ما قُلتُ لكَ“.</span>
<span className="v" id="v23" data-v="23" data-xref="صموئيلَ الثّانِي 3:3; صموئيلَ الثّانِي 13:37–38"> وذهَبَ يوآبُ إلى جَشورَ وجاءَ بأبشالومَ إلى أورُشليمَ.</span>
<span className="v" id="v24" data-v="24" data-xref="رُؤيا يُوحَنّا 22:4; التَّكوين 43:3; صموئيلَ الثّانِي 3:13"> فقالَ المَلِكُ: ”لينصرِفْ أبشالومُ إلى بَيتِهِ ولا يَرى وجهي“. فاَنصرَفَ وما قابَلَهُ المَلِكُ.</span>
</div>
</div>

## داود يصالح أبشالوم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 1:6; التَّثْنِيَة 28:35; أيُّوب 2:7"> ولم يكنْ في جميعِ بَني إِسرائيلَ رَجلٌ مَشهورٌ بجَمالِهِ كأبشالومَ. فما كانَ فيهِ عيبٌ مِنْ أسفلِ قدمِهِ إلى قِمَّةِ رأسِهِ.</span>
<span className="v" id="v26" data-v="26" data-xref="حِزْقِيال 44:20; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:14; صموئيلَ الثّانِي 18:9"> وكانَ إذا حَلقَ شعرَ رأسِهِ في آخرِ كُلِّ سنَةٍ لأنَّهُ كانَ يثقُلُ علَيهِ، بلغَ وزنُ هذا الشَّعرِ مئتي مِثقالٍ بالوزنِ الرَّسميِّ.</span>
<span className="v" id="v27" data-v="27" data-xref="صموئيلَ الثّانِي 18:18; صموئيلَ الثّانِي 13:1; إشَعْياء 14:22"> ووُلِدَ لأبشالومَ ثَلاثَةُ بَنينَ واَبنَةٌ واحدةٌ سمَّاها تامار وكانَت جميلةَ المنظَرِ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28"> وأقامَ أبشالومُ بأورُشليمَ سنَتينِ وما قابَلَهُ المَلِكُ.</span>
<span className="v" id="v29" data-v="29" data-xref="صموئيلَ الثّانِي 14:30–31; أسْتِير 1:12; ﻣﺘﻰ 22:3"> فأرسَلَ يتوسَّطُ يوآبَ ليحقِّقَ هذِهِ المُقابلَةَ، فلم يَجئْ إليهِ. فأرسَلَ يتوسَّطُه ثانيةً، فرفَضَ.</span>
<span className="v" id="v30" data-v="30" data-xref="القُضاة 15:4–5; المُلُوكِ الثّانِي 10:6–7; المُلُوكِ الثّانِي 9:33"> فقالَ لِعبيدِهِ: ”أُنظُروا. حقلُ يوآبَ بجانِبِ حقلي وهوَ مزروعٌ شعيرًا، فاَذهَبوا وأحرقوهُ بالنَّارِ“. فنفَّذوا الأمرَ.<br />فجاءَ عبيدُ يوآبَ إلى سيّدِهِم، وثيابُهُم ممزَّقَةٌ وقالوا: ”أحرقَ عبيدُ أبشالومَ حقلَكَ بالنَّارِ“.</span>
<span className="v" id="v31" data-v="31"> فأسرَعَ يوآبُ إلى أبشالومَ في بَيتِهِ وسألَهُ: ”لماذا أحرَقَ عبيدُكَ حقلي بالنَّارِ؟“</span>
<span className="v" id="v32" data-v="32" data-xref="صَمُوئيلَ الأوَّلُ 20:8; الخُرُوج 14:12; الأمثال 28:13"> فأجابَهُ: ”لأنَّكَ لم تَحضُرْ حينَ أرسلتُ وراءَكَ، لأطلُبَ مِنكَ أنْ تذهبَ إلى المَلِكِ وتسألَهُ: لماذا جاءَ بي مِنْ جَشورَ؟ كانَ خيرًا لي لو بقيتُ هُناكَ. والآنَ ساعِدْني دَعْني أقابِلُ المَلِكَ، فإنْ كُنتُ مُذنبًا في شيءٍ فليَقتُلْني“.</span>
<span className="v" id="v33" data-v="33" data-xref="التَّكوين 33:4; لُوقا 15:20; التَّكوين 27:26"> فذهَبَ يوآبُ إلى المَلِكِ وأخبرَهُ بالأمرِ، فاَستَدعى أبشالومَ. فدخَلَ على المَلِكِ واَنحَنى بوجهِه إلى الأرضِ أمامَه، فقبَّلهُ المَلِكُ.</span>
</div>
