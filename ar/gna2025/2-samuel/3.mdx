## بنو داود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المُلُوكِ الأوَّلُ 14:30; الأمثال 4:18–19; أيُّوب 8:7"> وطالَتِ الحربُ بَينَ أتباعِ شاوُلَ وأتباعِ داوُدَ، وكانَ أتباعُ داوُدَ يَتقَوَّونَ وأتباعُ شاوُلَ يَضعفونَ.</span>
<span className="v" id="v2" data-v="2" data-xref="أخبارِ الأيّامِ الأوّلُ 3:1–4; التَّكوين 49:3–4; صموئيلَ الثّانِي 13:1–29"> ووُلِدَ لِداوُدَ بَنونَ في حبرونَ، وكانَ بِكرُهُ أمنونَ مِنْ أخينوعَمَ اليَزرَعيليَّةِ،</span>
<span className="v" id="v3" data-v="3" data-xref="صَمُوئيلَ الأوَّلُ 27:8; أخبارِ الأيّامِ الأوّلُ 3:1; التَّثْنِيَة 3:14"> والثَّاني كيلأبَ مِنْ أبيجايِلَ أرمَلَةِ نابالَ الكرمليّ، والثَّالثُ أبشالومَ بنَ معكَةَ بنتِ تَلمايَ ملِكِ جَشورَ،</span>
<span className="v" id="v4" data-v="4" data-xref="المُلُوكِ الأوَّلُ 2:13–25; المُلُوكِ الأوَّلُ 1:5–18"> والرَّابعُ أدونيَّا بنَ حَجِّيتَ، والخامسُ شَفَطْيَا بنَ أبيطالَ،</span>
<span className="v" id="v5" data-v="5"> والسَّادسُ يَثْرَعامُ مِنْ عَجلَةَ زوجَةِ داوُدَ. هؤُلاءِ وُلِدوا لِداوُدَ في حبرونَ.</span>
</div>
</div>

## أبنير يلتحق بداود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="صموئيلَ الثّانِي 2:8–9; أخبارِ الأيّامِ الثّانِي 25:8; إشَعْياء 8:9–10"> وفيما الحربُ مُستمرَّةٌ بَينَ أتباعِ شاوُلَ وأتباعِ داوُدَ، كانَ أبنيرُ يزدادُ قوَّةً في جماعةِ شاوُلَ.</span>
<span className="v" id="v7" data-v="7" data-xref="صموئيلَ الثّانِي 21:8–11; المُلُوكِ الأوَّلُ 2:21–22; صموئيلَ الثّانِي 12:8"> وكانَ لِشاوُلَ جاريَةٌ اَسمُها رِصفَةُ بنتُ أيَّةَ، فقالَ إيشبوشثُ لأبنيرَ:</span>
<span className="v" id="v8" data-v="8" data-xref="صموئيلَ الثّانِي 9:8; صموئيلَ الثّانِي 16:9; مَرْقُس 6:18–19"> ”لماذا تدخُلُ على جاريَةِ أبي؟“ فغَضِبَ أبنيرُ جدًّا لكلامِ إيشبوشثَ وقالَ: ”أتظنُّ أنِّي عميلٌ ليَهوذا؟ إلى هذا اليومِ وأنا وَفيٌّ لبيتِ شاوُلَ أبيكَ وإلى إخوَتِه وأصحابِه، ولم أُسلِمْكَ إلى يدِ داوُدَ، وأنتَ تَشُكُّ بي بسبَبِ اَمرأةٍ؟</span>
<span className="v" id="v9" data-v="9" data-xref="صَمُوئيلَ الأوَّلُ 15:28; المُلُوكِ الأوَّلُ 19:2; راعُوث 1:17"> ويلٌ لي مِنَ اللهِ إنْ لم أُحقِّقْ لداوُدَ ما حلفَ الرّبُّ لَه</span>
<span className="v" id="v10" data-v="10" data-xref="القُضاة 20:1; صَمُوئيلَ الأوَّلُ 3:20; المُلُوكِ الأوَّلُ 4:25"> مِنْ نَقلِ المَملكةِ مِنْ بَيتِ شاوُلَ وإقامةِ عرشِ داوُدَ على بَني إِسرائيلَ ومِنهُم بَيتُ يَهوذا، مِنْ دانَ إلى بئرَ سَبْعَ“.</span>
<span className="v" id="v11" data-v="11"> فلم يقدِرْ إيشبوشثُ أنْ يُجيبَ أبنيرَ بكلِمةٍ واحدةٍ لِخوفِهِ مِنهُ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="أخبارِ الأيّامِ الأوّلُ 12:38–40; صموئيلَ الثّانِي 3:21; لُوقا 16:5–8"> وللحالِ أرسلَ أبنيرُ رُسُلًا إلى داوُدَ يقولونَ: ”لِمَنِ الأرضُ؟ لِنَتَّفِقْ، فأساعِدَكَ على اَسترجاعِ جميعِ إِسرائيلَ“.</span>
<span className="v" id="v13" data-v="13" data-xref="التَّكوين 43:3; صَمُوئيلَ الأوَّلُ 19:11–17; التَّكوين 44:23"> فأجابَ داوُدُ: ”حسَنٌ. أتَّفقُ معَكَ، ولكِنِّي أطلبُ مِنكَ أنْ تأتيَ بِميكالَ اَبنةِ شاوُلَ متى جِئتَ لِتَراني“.</span>
<span className="v" id="v14" data-v="14" data-xref="صَمُوئيلَ الأوَّلُ 18:27; صَمُوئيلَ الأوَّلُ 18:25; صموئيلَ الثّانِي 2:10"> وأرسَلَ داوُدُ إلى إيشبوشثَ بنِ شاوُلَ يقولُ: ”رُدَّ لي زَوجتي ميكالَ الّتي أخذتُها بمئةِ غُلفَةٍ مِنْ الفِلسطيِّينَ“.</span>
<span className="v" id="v15" data-v="15" data-xref="صَمُوئيلَ الأوَّلُ 25:44"> فأرسَلَ إيشبوشَثُ واَسترَدَّها مِنْ عِندِ زَوجِها فَلْطيئيلَ بنِ لايشَ.</span>
<span className="v" id="v16" data-v="16" data-xref="صموئيلَ الثّانِي 16:5; صموئيلَ الثّانِي 19:16; المُلُوكِ الأوَّلُ 2:8"> فرافَقَها زَوجُها وهوَ يبكي إلى بَحوريمَ فقالَ لَه أبنيرُ: ”إرجِعْ“، فرَجَعَ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17"> وقالَ أبنيرُ لشُيوخِ بَيتِ إِسرائيلَ: ”كُنتُم مِنْ قَبلُ تَطلبونَ داوُدَ مَلِكًا علَيكُم،</span>
<span className="v" id="v18" data-v="18" data-xref="صَمُوئيلَ الأوَّلُ 15:28; المَزامِير 89:19–23; صَمُوئيلَ الأوَّلُ 9:16"> فاَفعَلوا هذا الآنَ لأنَّ الرّبَّ قالَ لَه: ”على يدِ داوُدَ عبدي أخَلِّصُ شعبي إِسرائيلَ مِنْ أيدي الفِلسطيِّينَ ومِنْ أيدي جميعِ أعدائِهِم“.</span>
<span className="v" id="v19" data-v="19" data-xref="أخبارِ الأيّامِ الأوّلُ 12:29; صَمُوئيلَ الأوَّلُ 10:20–21; المَزامِير 68:27"> وتكلَّمَ أبنيرُ أيضًا على مسامِعِ بَيتِ بنيامينَ، ثُمَّ ذهَبَ ليُخبِرَ داوُدَ في حبرونَ بِما اَتَّفقَ علَيهِ جميعُ بَني إِسرائيلَ، ومِنهُم بَيتُ بنيامّينَ.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 31:54; التَّكوين 26:30; أسْتِير 1:3"> فجاءَ أبنيرُ إلى داوُدَ في حبرونَ ومعَهُ عِشرونَ رجُلًا، فأقامَ داوُدُ مأدبَةً لَه ولِرِجالِه.</span>
<span className="v" id="v21" data-v="21" data-xref="صموئيلَ الثّانِي 3:12; المُلُوكِ الأوَّلُ 11:37; صموئيلَ الثّانِي 3:10"> فقالَ أبنيرُ لداوُدَ: ”أقومُ الآنَ وأذهبُ لأجمَعَ لكَ يا سيِّدي المَلِكُ جميعَ بَيتِ إِسرائيلَ، فيُعاهِدونَكَ وتملِكُ علَيهِم وتنالُ ما يشتهي قلبُكَ“. فصرَفَ داوُدُ أبنيرَ، فذهبَ بأمانٍ.</span>
</div>
</div>

## مقتل أبنير ودفنه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22"> ثُمَّ رجَعَ أتباعُ داوُدَ ويوآب منَ الغَزوِ ومَعَهُم غنيمَةٌ عظيمَةٌ، ولم يكُنْ أبنيرُ عِندَ داوُدَ في حبرونَ لأنَّهُ كان صرفَهُ وذهَبَ بأمانٍ.</span>
<span className="v" id="v23" data-v="23"> فلمَّا حضرَ يوآبُ معَ رِجالِه قيلَ لَه: ”جاءَ أبنيرُ بنُ نيرَ إلى المَلِكِ وصرفَهُ، فذهبَ بأمانٍ“.</span>
<span className="v" id="v24" data-v="24" data-xref="صموئيلَ الثّانِي 3:8; صموئيلَ الثّانِي 19:5–7; يُوحَنّا 18:35"> فدخلَ يوآبُ على المَلِكِ وقالَ: ”ماذا فعلتَ؟ أتى أبنيرُ إليكَ فلِماذا صَرَفتَه فذهبَ بأمانٍ؟</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 28:6; إشَعْياء 37:28; العَدَد 27:17"> ألا تعرِفُ أنَّه أتى ليخدعَكَ، وليعرِفَ كُلَّ ما تقومُ بهِ وما تفعلُهُ؟“.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="الأمثال 26:23–26; الأمثال 27:4–6"> وخرجَ يوآبُ مِنْ عِندِ داوُدَ وبعثَ رُسُلًا في طلبِ أبنيرَ، فلحِقوا به وردُّوهُ مِنْ بئرَ سيرَةَ مِنْ دونِ أنْ يعلَمَ داوُدُ.</span>
<span className="v" id="v27" data-v="27" data-xref="المُلُوكِ الأوَّلُ 2:5; صموئيلَ الثّانِي 20:9–10; المُلُوكِ الأوَّلُ 2:32"> فلمَّا رجعَ أبنيرُ إلى حبرونَ، اَنفرَدَ بهِ يوآبُ إلى وسَطِ البابِ، كأنَّما يُريدُ أن يكلِّمَهُ، وضربَهُ هُناكَ في بَطنِه اَنتقامًا لِعسائيلَ أخيهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="العَدَد 35:33; الخُرُوج 21:12; التَّكوين 9:6"> فسمِعَ داوُدُ بَعدَ ذلِكَ فقالَ: ”يشهدُ الرّبُّ أنِّي ومَملكَتي بريئانِ مِنْ دمِ أبنيرَ بنِ نيرَ،</span>
<span className="v" id="v29" data-v="29" data-xref="صموئيلَ الثّانِي 1:16; المَزامِير 109:8–19; اللّاويِّين 13:44–46"> ليكُنْ دمُهُ على رأسِ يوآبَ وعلى جميعِ بَيتِ أبيهِ، فلا يَنقطِعُ منهُ مريضٌ بالسَّيلانِ وأبرصُ ومتوكِّئٌ على عُكَّازٍ وساقِطٌ بالسَّيفِ وجائعٌ إلى الخُبزِ“.</span>
<span className="v" id="v30" data-v="30" data-xref="الأمثال 28:17; اعمال الرسل 28:4; صموئيلَ الثّانِي 2:19–23"> وهكذا قُتِلَ أبنيرُ بيَدِ يوآبَ وأبيشايَ أخيهِ، لأنَّه قتلَ عسائيلَ أخاهُما بِجَبعونَ في الحربِ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 37:34; صموئيلَ الثّانِي 1:11; صموئيلَ الثّانِي 1:2"> وقالَ داوُدُ ليوآبَ ولجميعِ الّذينَ معَهُ: ”مزِّقوا ثِيابَكُم واَلبَسوا المُسوحَ ونوحوا على أبنيرَ“. ومشى داوُدُ المَلِكُ وراءَ النَّعشِ.</span>
<span className="v" id="v32" data-v="32" data-xref="الأمثال 24:17; أيُّوب 31:28; صَمُوئيلَ الأوَّلُ 30:4"> ودفَنوا أبنيرَ بحَبرونَ، فرفَعَ المَلِكُ صوتَهُ وبكى على قبرِ أبنيرَ، وبكى جميعُ الشَّعبِ.</span>
<span className="v" id="v33" data-v="33" data-xref="صموئيلَ الثّانِي 1:17; أخبارِ الأيّامِ الثّانِي 35:25; لُوقا 12:19–20"> ورثى المَلِكُ أبنيرَ. فقالَ: ”أتموتُ يا أبنيرُ بهذِهِ البساطةِ؟</span>
<span className="v" id="v34" data-v="34" data-xref="القُضاة 16:21; صموئيلَ الثّانِي 1:12; أيُّوب 24:14"> ما كانَ قيدٌ في يدَيكَ، ولا في رجلَيكَ، بل كمَنْ يسقطُ أمامَ المُجرِمينَ سَقطتَ“. وظلَّ الشَّعبُ يَبكونَه.</span>
<span className="v" id="v35" data-v="35" data-xref="صموئيلَ الثّانِي 12:17; صموئيلَ الثّانِي 1:12; راعُوث 1:17"> وحاوَلوا أنْ يُقَدِّموا لداوُدَ طَعامًا، وكانَ نهارٌ بَعدُ، فأقسَمَ وقالَ: ”ويلٌ لي مِنَ اللهِ إنْ ذُقتُ خبزًا أو شيئًا آخَرَ قبلَ أنْ تَغرُبَ الشَّمسُ“.</span>
<span className="v" id="v36" data-v="36" data-xref="مَرْقُس 7:37; المَزامِير 62:9; مَرْقُس 15:11–13"> فعلِمَ جميعُ الشَّعبِ بذلِكَ واَستَحسَنوه، ككُلِّ ما كانَ يَفعلُهُ داوُدُ،</span>
<span className="v" id="v37" data-v="37"> وأيقَنوا في ذلِكَ اليومِ أنَّ المَلِكَ لم يكُنْ لَه يَدٌ في مَقتَلِ أبنيرَ بنِ نيرَ.</span>
<span className="v" id="v38" data-v="38" data-xref="أيُّوب 32:9; صموئيلَ الثّانِي 2:8; صموئيلَ الثّانِي 3:12"> وقالَ المَلِكُ لِحاشيَتِهِ: ”ألا تَعلَمونَ أنَّ رئيسًا وعظيمًا سقطَ اليومَ في أرضِ إِسرائيلَ؟</span>
<span className="v" id="v39" data-v="39" data-xref="المُلُوكِ الأوَّلُ 2:33–34; المَزامِير 101:8; المَزامِير 28:4"> وأنا اليومَ ضعيفٌ، بَعدُ، ولو مُسِحتُ مَلِكًا، وبَنو صرويَّةَ هؤلاءِ هُم أشَدُّ منِّي. عاقبَ الرّبُّ فاعِلَ الشَّرِّ بحَسَبِ شرِّهِ“.</span>
</div>
</div>
