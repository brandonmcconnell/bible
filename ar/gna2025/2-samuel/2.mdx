## داود ملك على يهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="صَمُوئيلَ الأوَّلُ 30:31; صَمُوئيلَ الأوَّلُ 23:4; التَّكوين 32:2"> وبَعدَ ذلِكَ سألَ داوُدُ الرّبَّ: ”هل أصعَدُ إلى إحدى مُدُنِ يَهوذا؟“ فأجابَهُ الرّبُّ: ”إصعَدْ“. فعادَ داوُدُ وسألَهُ: ”إلى أينَ؟“ أجابَ: ”إلى حَبرونَ“.</span>
<span className="v" id="v2" data-v="2" data-xref="صَمُوئيلَ الأوَّلُ 25:42–43; صَمُوئيلَ الأوَّلُ 30:5; لُوقا 22:28–29"> فصعِدَ داوُدُ إلى هُناكَ معَ اَمرَأتَيهِ أخينوعَمَ اليَزرَعيليَّةِ وأبيجايِلَ أرمَلَةِ نابالَ الكرمليّ.</span>
<span className="v" id="v3" data-v="3" data-xref="صَمُوئيلَ الأوَّلُ 30:1; أخبارِ الأيّامِ الأوّلُ 12:1–7; صَمُوئيلَ الأوَّلُ 27:2–3"> واَصطَحَب داوُدُ الرِّجالَ الّذينَ معَهُ، كُلَّ واحدٍ وعائلَتَهُ، فأقاموا في جِوارِ حَبرونَ.</span>
<span className="v" id="v4" data-v="4" data-xref="صموئيلَ الثّانِي 5:5; صَمُوئيلَ الأوَّلُ 31:11–13; صموئيلَ الثّانِي 5:3"> وجاءَ رِجالُ يَهوذا ومسَحوا هُنالِكَ داوُدَ مَلِكًا على بيتِ يَهوذا.<br />وقيلَ لداوُدَ إنَّ أهلَ يابيشَ في جلعادَ هُمُ الّذينَ دفَنوا شاوُلَ</span>
<span className="v" id="v5" data-v="5" data-xref="صَمُوئيلَ الأوَّلُ 23:21; المَزامِير 115:15; راعُوث 2:20"> فأرسَلَ رُسُلًا يقولونَ لهُم: ”بارككُمُ الرّبُّ لأنَّكُم أظهَرتُم هذا الوفاءَ لِسيِّدكُم شاوُلَ ودَفنتُموهُ.</span>
<span className="v" id="v6" data-v="6" data-xref="الأمثال 14:22; الخُرُوج 34:6; ﻣﺘﻰ 5:44"> والآنَ لِيُحسِنِ الرّبُّ إليكُم ويَرأفْ بِكُم، وأنا أيضًا أُجازيكُم خَيرًا على هذا العمَلِ.</span>
<span className="v" id="v7" data-v="7" data-xref="أفَسُس 6:10; صموئيلَ الثّانِي 10:12; التَّكوين 15:1"> فقَوُّوا قُلوبَكُم وكونوا أشدَّاءَ. ماتَ شاوُلُ سيِّدُكُم، ولكِنَّ بيتَ يَهوذا مَسحوني مَلِكًا علَيهِم“.</span>
</div>
</div>

## إيشبوشث ملك على إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="صَمُوئيلَ الأوَّلُ 14:50; التَّكوين 32:2; صموئيلَ الثّانِي 3:7–8"> وأخذَ أبنيرُ بنُ نيرَ، قائدُ جيشِ شاوُلَ، إيشبوشثَ بنَ شاوُلَ وعبرَ بهِ الأردُنَّ إلى مَحنايِمَ،</span>
<span className="v" id="v9" data-v="9" data-xref="يَشُوع 19:18; يَشُوع 13:8–11; العَدَد 1:40"> وأقامَهُ مَلِكًا على بَني جلعادَ وأشيرَ ويِزرَعيلَ وأفرايمَ وبنيامينَ، وعلى جميعِ بَني إِسرائيلَ.</span>
<span className="v" id="v10" data-v="10"> وكانَ إيشبوشثُ اَبنَ أربعينَ سنَةً حينَ مَلكَ على بَني إِسرائيلَ، ودامَ ملْكُهُ سَنتينِ. وأمَّا بيتُ يَهوذا فتبِعوا داوُدَ.</span>
<span className="v" id="v11" data-v="11" data-xref="المُلُوكِ الأوَّلُ 2:11; صموئيلَ الثّانِي 5:4–5; أخبارِ الأيّامِ الأوّلُ 3:4"> وكانَ عددُ الأيّامِ الّتي ملَكَ فيها داوُدُ بِحَبرونَ على بيتِ يَهوذا سبْعَ سنينَ وسِتَّةَ أشهرٍ.</span>
</div>
</div>

## الحرب بين إسرائيل ويهوذا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="يَشُوع 18:25; يَشُوع 10:2; يَشُوع 9:3"> وسارَ أبنيرُ بنُ نيرَ وأتباعُ إيشبوشثَ بنِ شاوُلَ مِنْ مَحنايِمَ إلى جِبعونَ،</span>
<span className="v" id="v13" data-v="13" data-xref="صموئيلَ الثّانِي 8:16; أخبارِ الأيّامِ الأوّلُ 2:16; صموئيلَ الثّانِي 20:23"> وسارَ يوآبُ اَبنُ صرويَّةَ وأتباعُ داوُدَ، فاَلتَقَوا جميعًا عِندَ بِركَةِ جبعونَ. فأقامَ أُولئِكَ على جانبِ البِركَةِ مِنْ هُناكَ، وهؤلاءِ على جانبِ البِركَة مِنْ هُنا.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="صموئيلَ الثّانِي 2:17; الأمثال 10:23; الأمثال 17:14"> فقالَ أبنيرُ ليوآبَ: ”لِيَنهَضْ فِتيانٌ منَّا ومِنكُم إلى البِرازِ أمامَنا“. فقالَ يوآبُ: ”لِيَنهَضوا“</span>
<span className="v" id="v15" data-v="15"> فنهَضَ للبِرازِ اَثنا عشَرَ مِنْ بَني بنيامينَ التَّابِعينَ لإيشبوشثَ بنِ شاوُلَ، واَثنا عشَرَ مِنْ أتباعِ داوُدَ،</span>
<span className="v" id="v16" data-v="16"> وأخذَ كُلُّ واحدٍ برأسِ خصمِهِ وطَعنَهُ بالسَّيفِ في جَنبِه فسقَطوا جميعًا. فدُعيَ ذلِكَ المكانُ حِلقَثَ هَصُّوريمَ، وهوَ في جِبعونَ.</span>
<span className="v" id="v17" data-v="17" data-xref="صموئيلَ الثّانِي 3:1"> وكانَ قتالٌ شديدٌ في ذلِكَ اليومِ، فاَنهزَمَ أبنيرُ ورِجالُ إِسرائيلَ مِنْ وجهِ أتباعِ داوُدَ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="أخبارِ الأيّامِ الأوّلُ 12:8; حَبَقُّوق 3:19; نَشِيدِ الأنشاد 8:14"> وكانَ هُناكَ بَنو صرويَّةَ الثَّلاثَةُ: يوآبُ وأبيشايُ وعسائيلُ، وكانَ عسائيلُ خفيفَ الرِّجلَينِ كأنَّهُ غزالٌ مِنْ غُزلانِ الصَّحراءِ.</span>
<span className="v" id="v19" data-v="19" data-xref="صموئيلَ الثّانِي 2:21; يَشُوع 1:7; الأمثال 4:27"> فطارَدَ أبنيرَ ولم يَحدْ عنهُ يَمينًا ولا شمالًا.</span>
<span className="v" id="v20" data-v="20"> فاَلتفَتَ أبنيرُ إليهِ وقالَ: ”أعسائيلُ أنتَ؟“ فأجابَهُ: ”أنا هوَ“.</span>
<span className="v" id="v21" data-v="21" data-xref="القُضاة 14:19"> فقالَ لَه أبنيرُ: ”حِدْ عنِّي يَمينًا أو شمالًا، واَقبِضْ على أحدِ الجُنودِ وخُذْ ما معَهُ“. فرفضَ عسائيلُ أنْ يَحيدَ عنهُ.</span>
<span className="v" id="v22" data-v="22" data-xref="صموئيلَ الثّانِي 3:27; الأمثال 29:1; المُلُوكِ الثّانِي 14:10–12"> فعادَ أبنيرُ وقالَ لَه: ”إرتَدَّ عنِّي لماذا تُجبِرُني على قَتلِكَ؟ وكيفَ أرفَعُ عينيَّ إلى يوآبَ أخيكَ؟“</span>
<span className="v" id="v23" data-v="23" data-xref="صموئيلَ الثّانِي 3:27; صموئيلَ الثّانِي 4:6; صموئيلَ الثّانِي 5:6"> فرفضَ أنْ يرتَدَّ، فطَعنَهُ أبنيرُ في بطنِهِ فخرَجَ الرُّمحُ مِنْ ظهرِهِ فسقطَ وماتَ في مكانِهِ. وكُلُّ مَنْ جاءَ ذلِكَ المكانَ الّذي سقطَ فيهِ عسائيلُ كانَ الجُمودُ يُصيبُه.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24"> فجَدَّ يوآبُ وأبيشايُ وراءَ أبنيرَ، فغابَت لهُما الشَّمسُ عِندَ تَلَّةِ أمَّةَ شَرقيَّ جيحَ، على طريقِ برِّيَّةِ جبعونَ.</span>
<span className="v" id="v25" data-v="25"> واَنضَمَّ بَنو بنيامينَ إلى أبنيرَ، وتمَركَزوا على التَّلَّةِ المذكورةِ.</span>
<span className="v" id="v26" data-v="26" data-xref="إرْمِيا 46:14; إرْمِيا 46:10; اعمال الرسل 7:26"> فنادى أبنيرُ يوآبَ وقالَ: ”أنبقى طَعامًا للحربِ إلى الأبدِ؟ ألا تعلَمُ أنَّ في نِهايةِ هذا الأمرِ مَرارةً؟ فحتّى متى لا تأمُرُ رِجالَكَ أنْ يَرجِعوا عَنْ بَني قومِهِم؟“</span>
<span className="v" id="v27" data-v="27" data-xref="الأمثال 17:14; صموئيلَ الثّانِي 2:14; الأمثال 15:1"> فأجابَهُ يوآبُ: ”حَيٌّ هوَ اللهُ. لولا كلامُكَ هذا لما عادَ رِجالي عَنْ بَني قومِهِم إلى الصَّباحِ“.</span>
<span className="v" id="v28" data-v="28"> ثُمَّ نفَخَ في البوقِ فتوقَّفَ جميعُ رجالِه عن مُطارَدةِ رِجالِ إِسرائيلَ وعَنْ مُقاتَلَتِهِم.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="صموئيلَ الثّانِي 2:8; نَشِيدِ الأنشاد 2:17"> فسارَ أبنيرُ ورِجالُهُ في غَورِ الأردُنِّ كُلَّ ذلِكَ اللَّيلِ، وعبَروا الأردُنَّ ومشَوا صباحًا حتّى بَلَغوا مَحنايمَ.</span>
<span className="v" id="v30" data-v="30"> ورجَعَ يوآبُ مِنْ مُطارَدةِ أبنيرَ وجمَعَ رِجالَهُ الّذينَ لداوُدَ، فوجَدَ أنَّهُ فقدَ تِسعَةَ عشَرَ رَجُلًا وعسائيلَ</span>
<span className="v" id="v31" data-v="31" data-xref="صموئيلَ الثّانِي 3:1; المُلُوكِ الأوَّلُ 20:11"> وقتَلَ رجالُ داوُدَ مِنْ بَني بنيامينَ ومِنْ رِجالِ أبنيرَ ثَلاثَ مِئةٍ وسِتّينَ رجُلًا.</span>
<span className="v" id="v32" data-v="32" data-xref="أخبارِ الأيّامِ الأوّلُ 2:13–16; صَمُوئيلَ الأوَّلُ 17:58; أخبارِ الأيّامِ الثّانِي 21:1"> ثُمَّ حمَلوا عسائيلَ ودفَنوهُ في قبرِ أبيهِ في بيتَ لحمَ. وسارَ يوآبُ ورِجالُهُ اللَّيلَ كُلَّه حتّى وصَلوا حبرونَ في الصَّباحِ.</span>
</div>
</div>
