## السّلوك المسيحي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﺭﻭﻣﻴﺔ 8:6; ﻣﺘﻰ 6:33; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:18"> وإنْ كُنتُم قُمتُم معَ المَسيحِ، فاَسعَوا إلى الأُمورِ الّتي في السَّماءِ حَيثُ المَسيحُ جالِسٌ عَنْ يَمينِ اللهِ.</span>
<span className="v" id="v2" data-v="2" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:15–17; أخبارِ الأيّامِ الأوّلُ 22:19; كُولُوسِي 3:1"> إِهتمّوا بالأُمورِ الّتي في السَّماءِ، لا بالأُمورِ الّتي في الأرضِ،</span>
<span className="v" id="v3" data-v="3" data-xref="غَلاطِيَّة 2:20; ﺭﻭﻣﻴﺔ 6:2; كُولُوسِي 2:20"> لأنَّكُم مُتُّم وحَياتُكُم مُستَتِرَةٌ معَ المَسيحِ في اللهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="يُوحَنّا 11:25; رسالَةُ يُوحَنّا الأُولَى 3:2; غَلاطِيَّة 2:20"> فمَتى ظَهَرَ المَسيحُ الّذي هوَ حَياتُكُم، تَظهَرونَ أنتُم أيضًا معَهُ في مَجدِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="ﺭﻭﻣﻴﺔ 8:13; غَلاطِيَّة 5:19–21; غَلاطِيَّة 5:24"> أميتوا، إذًا، ما هوَ أرضِيٌّ فيكُم كالزِّنى والفِسقِ والهَوى والشَّهوَةِ الرَّديئَةِ والفجورِ، فهوَ عِبادَةُ الأوثانِ،</span>
<span className="v" id="v6" data-v="6" data-xref="أفَسُس 5:6; ﺭﻭﻣﻴﺔ 1:18; رسالَةُ بُطرُسَ الأُولَى 1:14"> وتِلكَ أُمورٌ تَجلِبُ غضَبَ اللهِ على أبناءِ المَعصيةِ.</span>
<span className="v" id="v7" data-v="7" data-xref="أفَسُس 2:2; رسالَةُ بُطرُسَ الأُولَى 4:3–4; ﺭﻭﻣﻴﺔ 6:19–20"> كذلِكَ كانَت حالُكُم فيما مَضى حينَ كُنتُم تَعيشونَ فيها.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="أفَسُس 4:29; أفَسُس 4:22; يَعقُوب 1:20–21"> أمَّا الآنَ فتَخَلَّصوا مِنْ كُلِّ ما فيهِ غَضَبٌ ونَقمَةٌ وخُبثٌ وشَتيمَةٌ. لا تَتَلفَّظوا بِالكَلامِ البَذيءِ،</span>
<span className="v" id="v9" data-v="9" data-xref="أفَسُس 4:22; أفَسُس 4:25; اللّاويِّين 19:11"> ولا يكذِبْ بَعضُكُم على بَعضٍ، لأنَّكُم خَلَعتُمُ الإنسانَ القَديمَ وكُلَّ أعمالِهِ،</span>
<span className="v" id="v10" data-v="10" data-xref="أفَسُس 4:23–24; ﺭﻭﻣﻴﺔ 12:2; أفَسُس 2:10"> ولَبِستُمُ الإنسانَ الجَديدَ الّذي يتَجَدَّدُ في المَعرِفَةِ على صُورَةِ خالِقِهِ.</span>
<span className="v" id="v11" data-v="11" data-xref="غَلاطِيَّة 3:28–29; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:13; ﺭﻭﻣﻴﺔ 10:12"> فلا يَبقى هُناكَ يَهودِيٌّ أو غَيرُ يَهودِيٍّ، ولا مَختونٌ أو غَيرُ مَختونٍ، ولا أعجَمِيٌّ أو بَربَرِيٌّ، ولا عَبدٌ أو حُرٌّ، بَلِ المَسيحُ الّذي هوَ كُلُّ شيءٍ وفي كُلِّ شيءٍ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="أفَسُس 4:32; أفَسُس 4:2; رسالَةُ يُوحَنّا الأُولَى 4:19"> وأنتُم الّذينَ اَختارَهُمُ اللهُ فقَدَّسَهُم وأحَبَّهُم، اَلبَسُوا عَواطِفَ الحَنانِ والرّأْفَةِ والتَّواضُعِ والوَداعَةِ والصَّبرِ.</span>
<span className="v" id="v13" data-v="13" data-xref="أفَسُس 4:32; مَرْقُس 11:25; يَعقُوب 2:13"> اَحتَمِلوا بَعضُكُم بَعضًا، ولْيُسامِحْ بَعضُكُم بَعضًا إذا كانَت لأَحَدٍ شَكوى مِنَ الآخَرِ. فكما سامَحَكُمُ الرَّبُّ، سامِحوا أنتُم أيضًا.</span>
<span className="v" id="v14" data-v="14" data-xref="رسالَةُ بُطرُسَ الأُولَى 4:8; أفَسُس 5:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 13:1–13"> واَلبَسُوا فَوقَ هذا كُلِّهِ المَحبَّةَ، فهِيَ رِباطُ الكَمالِ.</span>
<span className="v" id="v15" data-v="15" data-xref="فِيلِبِّي 4:6–7; يُوحَنّا 14:27; إشَعْياء 26:3"> ولْيَملِكْ في قُلوبِكُم سلامُ المَسيحِ، فإلَيهِ دَعاكُمُ اللهُ لِتَصيروا جَسَدًا واحدًا. كونوا شاكِرينَ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="المَزامِير 119:11; أفَسُس 5:19; يُوحَنّا 15:7"> لِتَحِلَّ في قُلوبِكُم كَلِمَةُ المَسيحِ بِكُلِّ غِناها لِتُعَلِّموا وتُنَبِّهوا بَعضُكُم بَعضًا بِكُلِّ حِكمَةٍ. رَتِّلوا المَزاميرَ والتَّسابيحَ والأناشيدَ الرُّوحِيَّةَ شاكِرينَ اللهَ مِنْ أعماقِ قُلوبِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:31; كُولُوسِي 3:23; الأمثال 3:6"> ومَهما يكُنْ لكُم مِنْ قَولٍ أو فِعلٍ، فلْيكُنْ باَسمِ الرَّبِّ يَسوعَ، حامِدينَ بِه اللهَ الآبَ.</span>
</div>
</div>

## وصايا اجتماعية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="أفَسُس 5:22–9; تِيطُس 2:4–5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 11:3"> أيَّتُها النِّساءُ اَخضَعْنَ لأزواجِكُنَّ كما يَليقُ في الرَّبِّ.</span>
<span className="v" id="v19" data-v="19" data-xref="رسالَةُ بُطرُسَ الأُولَى 3:7; التَّكوين 2:23–24; كُولُوسِي 3:21"> أيُّها الرِّجالُ، أحِبُّوا نِساءَكُم ولا تكونوا قُساةً علَيهِنَّ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 20:12; أفَسُس 6:1–3; الأمثال 6:20"> أيُّها البَنونَ، أطيعوا والِديكُم في كُلِّ شيءٍ لأنَّ هذا يُرضي الرَّبَّ.</span>
<span className="v" id="v21" data-v="21" data-xref="الأمثال 4:1–4; أفَسُس 6:4; المَزامِير 103:13"> أيُّها الآباءُ، لا تُغيظوا أبناءَكُم لِئَلَّا يَيأسوا.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="أفَسُس 6:5–7; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:1–2; غَلاطِيَّة 1:10"> أيُّها العَبيدُ، أطيعوا في كُلِّ شَيءٍ سادَتكُم في هذِهِ الدُّنيا، لا بِخدمَةِ العَينِ كمَنْ يُرضي النَّاسَ، بَلْ بِنَقاوَةِ القَلبِ ومَخافَةِ الرَّبِّ.</span>
<span className="v" id="v23" data-v="23" data-xref="كُولُوسِي 3:17; أفَسُس 6:6–7; ﺭﻭﻣﻴﺔ 14:8"> ومَهما تَعمَلوا فاَعمَلوهُ مِنْ كُلِّ قُلوبِكُم كأنَّهُ لِلرَّبِّ لا لِلنَّاسِ،</span>
<span className="v" id="v24" data-v="24" data-xref="أفَسُس 6:8; غَلاطِيَّة 1:10; اعمال الرسل 20:32"> عالِمينَ أنَّ الرَّبَّ سَيُكافِئُكُم بِميراثِهِ، فأنتُم تَخدِمونَ الرَّبَّ المَسيحَ.</span>
<span className="v" id="v25" data-v="25" data-xref="اعمال الرسل 10:34; ﺭﻭﻣﻴﺔ 2:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:10"> أمَّا الّذي يَعمَلُ الشَّرَّ فسَيَنالُ جَزاءَ عَمَلِهِ، ولا مُحاباة.</span>
</div>
