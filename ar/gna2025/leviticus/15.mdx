## شريعة ما يفرزه الجسد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="عامُوس 3:7; اللّاويِّين 13:1; اللّاويِّين 11:1"> وكلَّمَ الرّبُّ موسى وهرونَ فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="العَدَد 5:2; صموئيلَ الثّانِي 3:29; اللّاويِّين 22:4"> ”قُولا لِبَني إِسرائيلَ: مَنْ كانَ بِجسدِهِ سَيَلانٌ فهوَ نَجِسٌ</span>
<span className="v" id="v3" data-v="3" data-xref="حِزْقِيال 23:20; حِزْقِيال 16:26; اللّاويِّين 12:3"> سَواءٌ سالَ مِنْ جسَدِهِ أوِ اَحْتقَنَ جسَدُهُ بهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="تِيطُس 1:15; أفَسُس 5:11; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:33"> كُلُّ فِراشٍ يستَلقي علَيهِ يكونُ نَجِسًا، وكُلُّ ما يَجلِسُ علَيهِ مِنَ الأمتِعَةِ يكونُ نَجِسًا.</span>
<span className="v" id="v5" data-v="5" data-xref="اللّاويِّين 11:25; اللّاويِّين 16:26; اللّاويِّين 17:15"> ومَنْ مَسَّ فِراشَهُ، فليَغسِلْ ثيابَهُ ويَستَحِمَّ بِالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v6" data-v="6" data-xref="يَعقُوب 4:8; إشَعْياء 1:16"> ومَنْ جلَسَ على ما يَجلِسُ علَيهِ صاحبُ السَّيَلانِ، فليَغسِلْ ثيابَهُ ويَستَحِمَّ بِالماءِ ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v7" data-v="7"> ومَنْ لَمَسَ جسَدَ صاحبِ السَّيَلانِ، فليَغسِلْ ثيابَهُ ويَستَحِمَّ بالماءِ ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v8" data-v="8" data-xref="يَهُوذا 1:4; رسالَةُ بُطرُسَ الثّانيَة 2:1–3; إشَعْياء 1:16"> وإنْ بصَقَ صاحبُ السَّيلانِ على الطَّاهرِ، فليَغسِلِ الطَّاهرُ ثيابَهُ ويَستَحِمَّ بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 31:34"> كُلُّ ما يركَبُ علَيهِ صاحبُ السَّيلانِ يكونُ نَجِسًا.</span>
<span className="v" id="v10" data-v="10" data-xref="يَعقُوب 4:8; المَزامِير 26:6; اللّاويِّين 15:8"> فمَنْ لمَسَ شيئًا مِمَّا كانَ تحتَهُ يكونُ نَجِسًا إلى المَغيبِ. ومَنْ حمَلَ شيئًا مِنهُ يَغسِلُ ثيابَهُ ويَستَحِمُّ بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v11" data-v="11"> وكلُّ مَنْ لمَسَهُ صاحبُ السَّيَلانِ بيدَينِ غيرِ مَغسولَتَينِ بالماءِ، فليَغسِلْ ثيابَهُ ويستحمَّ بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v12" data-v="12" data-xref="اللّاويِّين 6:28; اللّاويِّين 11:32–33; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:1"> وإذا لمَسَ صاحبُ السَّيَلانِ وِعاءَ خزَفٍ فليُكْسَرْ، أو وِعاءَ خشَبٍ فليُغْسَلْ بالماءِ.</span>
<span className="v" id="v13" data-v="13" data-xref="اللّاويِّين 14:8; اللّاويِّين 8:33; اللّاويِّين 15:28"> وإذا طَهُرَ مِنْ سَيلانِه ينتَظِرُ سَبعَةَ أيّامٍ، ثُمَّ يَغسِلُ ثيابَه ويَستَحمُّ بماءٍ حارٍّ.</span>
<span className="v" id="v14" data-v="14" data-xref="اللّاويِّين 12:8; اللّاويِّين 12:6; العِبرانِيّين 10:10"> وفي اليومِ الثَّامنِ يأخُذُ لَه يَمامتَينِ أو فَرخَي حمامٍ ويجيءُ إلى أمامِ الرّبِّ، إلى بابِ خَيمةِ الاجتماعِ، ويدفعُهُما إلى الكاهنِ.</span>
<span className="v" id="v15" data-v="15" data-xref="اللّاويِّين 14:30–31; اللّاويِّين 4:26; ﻣﺘﻰ 3:17"> فيُقرِّبُ الكاهنُ أحدَهُما ذبيحةَ خطيئةٍ والآخرَ مُحرَقةً، ويُكفِّرُ عَنهُ أمامَ الرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="التَّثْنِيَة 23:10–11; اللّاويِّين 22:4; اللّاويِّين 15:5"> ”وإذا خرَجَت مِنْ رَجُلٍ نُطفَةُ مُضاجَعةٍ، فليَغسِلْ جميعَ بدَنِه بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v17" data-v="17"> وأيُّ ثوبٍ أو جِلْدٍ أصابَهُ مِنها شيءٌ، فليُغسَلْ بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v18" data-v="18" data-xref="اللّاويِّين 15:5; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:3–5; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:12"> وإذا ضاجعَ رَجُلٌ امرأةً بنُطفَةٍ، فلْيسَتحِمَّا بالماءِ ويكونانِ نجِسَينِ إلى المَغيبِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="اللّاويِّين 12:2; ﻣﺘﻰ 15:19; مَرْقُس 5:25"> ”وإذا كانَ باَمرأةٍ سَيلانُ دَمٍ مِنْ جسَدِها كعادةِ النِّساءِ، فسَبعَةُ أيّامٍ تكونُ في طَمْثِها، وكُلُّ مَنْ لمَسَها يكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v20" data-v="20" data-xref="اللّاويِّين 15:4–9; الأمثال 2:16–19; الأمثال 6:35"> وجميعُ ما تَضطَجِعُ أو تجلِسُ علَيهِ في طَمْثِها يكونُ نَجِسًا،</span>
<span className="v" id="v21" data-v="21" data-xref="رُؤيا يُوحَنّا 7:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1; إشَعْياء 22:14"> وكُلُّ مَنْ لمَسَ فِراشَها يَغسِلُ ثيابَهُ ويَستَحِمُّ بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v22" data-v="22"> مَنْ لمَسَ شيئًا مِمَّا تجلِسُ علَيهِ يغسِلُ ثيابَهُ ويَستَحِمُّ بالماءِ، ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v23" data-v="23"> وإنْ كانَ على فِراشِها أو على ما هيَ جالِسةٌ علَيهِ شيءٌ، فَمَنْ لمَسَهُ يكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v24" data-v="24" data-xref="اللّاويِّين 20:18; رسالَةُ بُطرُسَ الأُولَى 2:11; اللّاويِّين 18:19"> وإنْ ضاجَعَها رَجُلٌ فصارَ طَمْثُها علَيهِ يكونُ نَجِسًا سَبعةَ أيّامٍ، وكُلُّ فِراشٍ يَستَلقي علَيهِ يكونُ نَجِسًا.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="مَرْقُس 5:25; ﻣﺘﻰ 9:20; لُوقا 8:43"> ”وإذا سالَ دَمُ امرأةٍ أيّامًا كثيرةً في غيرِ وقتِ طَمْثِها أو بَعدَهُ، فلْتَكُنْ في جميعِ أيّامِ سَيَلانِها نَجِسةً كما في أيّامِ طَمْثِها.</span>
<span className="v" id="v26" data-v="26"> فكُلُّ فِراشٍ تَستَلقي علَيهِ وكُلُّ ما تجلِسُ علَيهِ يكونُ نَجِسًا.</span>
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 15:21; العِبرانِيّين 10:22; اللّاويِّين 17:15–16"> وكلُّ مَنْ لمَسَ شيئًا مِنها يكونُ نَجِسًا، فيَغسِلُ ثيابَهُ ويَستَحِمُّ بالماءِ ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v28" data-v="28" data-xref="اللّاويِّين 15:13–15; غَلاطِيَّة 3:13; أفَسُس 1:6–7"> وإذا طَهُرَت مِنْ سَيَلانِها فلتَنتَظِرْ سَبعَةَ أيّامٍ، ثُمَّ تَطهُرُ.</span>
<span className="v" id="v29" data-v="29"> وفي اليومِ الثَّامنِ تأخُذُ لها يَمامتَينِ أو فَرخَي حمامٍ وتجيءُ بهما إلى الكاهنِ، إلى بابِ خَيمةِ الاجتِماعِ،</span>
<span className="v" id="v30" data-v="30"> فيُقَرِّبُ الكاهنُ أحدَهُما للرّبِّ ذبيحةَ خطيئةٍ والآخَرَ مُحرَقةً، ويُكفِّرُ عَنها الكاهنُ أمامَ الرّبِّ سَيَلانَ نَجاستِها.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="العَدَد 19:20; العَدَد 19:13; حِزْقِيال 5:11"> ”هكذا تُجنِّبانِ بَني إِسرائيلَ النَّجاسةَ لِئلَّا يموتوا في نَجاستِهم بِتَنجيسِهِم مَسكِني الّذي بَينَهم.</span>
</div>
<div className="bp">
<span className="v" id="v32" data-v="32" data-xref="اللّاويِّين 11:46; اللّاويِّين 13:59; العَدَد 6:13"> ”تِلكَ شريعةُ مَنْ بهِ سَيَلانٌ ومَنْ يَخرُجُ مِنْهُ نُطفَةُ مُضاجَعةٍ فَيَتَنَجَّسُ بِهِما،</span>
<span className="v" id="v33" data-v="33" data-xref="اللّاويِّين 15:19; اللّاويِّين 20:18; اللّاويِّين 15:24–25"> وشريعةُ المرأةِ في أيّامِ طَمْثِها ومَنْ بِهِ سَيَلانٌ، ذَكَرًا أو أُنثى، وشريعةُ الرَّجُلِ الّذي يضاجِعُ نَجِسَةً“.</span>
</div>
