## ذبيحة الخطيئة سهوًا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="المَزامِير 19:12; العِبرانِيّين 9:7; اللّاويِّين 4:27"> ”قُلْ لِبَني إِسرائيلَ: إنْ خَطِئَ أحدٌ سَهوًا في شيءٍ أو بَعضِهِ مِمَّا نَهى الرّبُّ عَنْ فِعلِهِ،</span>
<span className="v" id="v3" data-v="3" data-xref="العِبرانِيّين 7:27–28; اللّاويِّين 4:14; اللّاويِّين 9:2"> فإنْ كانَ كاهنًا مَمسوحًا خَطِئَ الشَّعبُ بِسبَبِهِ فعلَيهِ أنْ يُقرِّبَ عَنْ خَطيئتِهِ التي اَرْتَكَبَها عِجْلًا مِنَ البقَرِ صحيحًا ذبيحةَ خطيئةٍ للرّبِّ.</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 1:3–4; دانيال 9:26; إشَعْياء 53:6"> يجيءُ بالعِجْلِ إلى بابِ خَيمةِ الاجتِماعِ ويضَعُ يدَهُ على رأسِهِ ويذبَحُهُ أمامَ الرّبِّ.</span>
<span className="v" id="v5" data-v="5" data-xref="اللّاويِّين 16:14; العَدَد 19:4; اللّاويِّين 4:16–17"> ويأخُذُ مِنْ دَمِ العِجْلِ ويدخُلُ بِهِ خَيمةَ الاجتماعِ،</span>
<span className="v" id="v6" data-v="6" data-xref="اللّاويِّين 16:14; اللّاويِّين 14:18; اللّاويِّين 26:28"> فيَغمِسُ إصبَعَهُ فيهِ ويَرُشُّ مِنهُ سَبْعَ مرَّاتٍ أمامَ الرّبِّ تِجاهَ حجابِ القُدسِ.</span>
<span className="v" id="v7" data-v="7" data-xref="اللّاويِّين 8:15; اللّاويِّين 9:9; اللّاويِّين 16:18"> ويضَعُ مِنَ الدَّمِ على قرونِ مذبحِ البَخورِ العَطِر الّذي في خَيمةِ الاجتِماعِ أمامَ الرّبِّ، وسائِرُ دَمِ العِجْلِ يسكُبُهُ عِندَ قاعدةِ مذبَحِ المُحرِقَةِ الّذي عِندَ بابِ خَيمةِ الاجتِماعِ.</span>
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 3:3–5; اللّاويِّين 4:26; اللّاويِّين 4:31"> وينزعُ مِنَ العِجْل جميعَ شَحْمِهِ، مِمَّا يُغطِّي الأمعاءَ وما علَيها،</span>
<span className="v" id="v9" data-v="9" data-xref="اللّاويِّين 3:4"> والكليتَينِ والشَّحْمَ الّذي علَيهما عِندَ الخاصرتَينِ، والكَبِدَ بَعدَ نَزْعِ زيادتِهِ معَ الكليتَينِ،</span>
<span className="v" id="v10" data-v="10"> كما تُنزَعُ مِنْ ثورِ ذبيحةِ السَّلامةِ. ويوقِدُ الكاهنُ هذا كُلَّه على مذبحِ المُحرقةِ.</span>
<span className="v" id="v11" data-v="11" data-xref="العَدَد 19:5; الخُرُوج 29:14; اللّاويِّين 9:8–11"> وأمَّا جِلْدُ العِجْلِ وجميعُ لَحمِه معَ رأسهِ وأكارِعهِ وأمعائِهِ ومَصارينِهِ،</span>
<span className="v" id="v12" data-v="12" data-xref="العِبرانِيّين 13:11; الخُرُوج 29:14; اللّاويِّين 16:27"> فيُخرِجُها إلى خارجِ المَحلَّةِ، إلى موضِعٍ نظيفٍ حيثُ يُطرَحُ الرَّمادُ، ويحرقُها على حطَبٍ بالنَّار.</span>
</div>
</div>

## ذبيحة الجماعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="اللّاويِّين 5:17; اللّاويِّين 5:2–5; هُوشَع 5:15"> ”وإنْ خَطِئَت جماعةُ بَني إِسرائيلَ كُلُّها سَهوًا وعمِلوا أيَّةَ واحدةٍ مِمَّا نَهى الرّبُّ عَنْ فِعلِهِ وأثِموا،</span>
<span className="v" id="v14" data-v="14" data-xref="اللّاويِّين 4:23; اللّاويِّين 4:3; اللّاويِّين 4:28"> ثُمَّ عُرِفَتِ الخطيئةُ الّتي اَرْتَكَبُوها، فَعلَيهِم أنْ يُقَرِّبوا للرّبِّ عِجْلًا مِنَ البقَرِ ذبيحةَ خطيئةٍ. يُحضِرونَه أمامَ خَيمةِ الاجتِماعِ.</span>
<span className="v" id="v15" data-v="15" data-xref="اللّاويِّين 1:4; اللّاويِّين 16:21; اللّاويِّين 8:14"> فيضَعُ شُيوخُ الجماعةِ أيديَهم على رأسِ العِجْلِ أمامَ الرّبِّ ويذبحونَه.</span>
<span className="v" id="v16" data-v="16" data-xref="اللّاويِّين 4:5–12; العِبرانِيّين 9:12–14"> ويأخذُ الكاهنُ المَمسوحُ مِنْ دَمِ العِجْلِ إلى خَيمةِ الاجتِماعِ،</span>
<span className="v" id="v17" data-v="17" data-xref="اللّاويِّين 4:6–7"> فيَغمسُ إصبعَهُ فيهِ ويرشُّ مِنهُ سَبْعَ مرَّاتٍ أمامَ الرّبِّ تُجاهَ الحِجابِ.</span>
<span className="v" id="v18" data-v="18" data-xref="اللّاويِّين 4:7"> ويضَعُ مِنهُ على قرونِ المذبحِ الّذي أمامَ الرّبِّ في خَيمةِ الاجتماعِ، وسائرُ الدَّمِ يسكُبُه عِندَ قاعدةِ مذبحِ المُحرقةِ الّذي عندَ بابِ الخَيمةِ</span>
<span className="v" id="v19" data-v="19" data-xref="اللّاويِّين 5:6; اللّاويِّين 4:26; العَدَد 15:25"> وينزعُ جميعَ شَحْمِهِ مِنهُ ويُوقِدُه على المذبَحِ،</span>
<span className="v" id="v20" data-v="20" data-xref="العَدَد 15:25; العِبرانِيّين 10:10–12; اللّاويِّين 12:8"> ويفعَلُ بهِ كما فعَلَ بعجْلِ ذبيحةِ الخطيئةِ. ويُكفِّرُ عنهُمُ الكاهنُ، فيسامحُهُمُ الرّبُّ.</span>
<span className="v" id="v21" data-v="21" data-xref="اللّاويِّين 16:15; ﻣﺘﻰ 20:28; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:21"> ويخرِجُ الكاهنُ العِجْلَ إلى خارجِ المحَلَّةِ ويحرقُهُ كما أحرقَ العِجْلَ الأوَّلَ. هوَ ذبيحةُ تكفيرٍ عَنْ خطيئةِ الجماعةِ.</span>
</div>
</div>

## ذبيحة الرئيس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اللّاويِّين 4:2; اللّاويِّين 4:13; العَدَد 31:13"> ”وإنْ خَطِئَ رئيسٌ وعمِلَ سَهوًا أيَّةَ واحدةٍ مِمَّا نَهى الرّبُّ عَنْ فِعلِه وأثِمَ،</span>
<span className="v" id="v23" data-v="23" data-xref="اللّاويِّين 4:14; ﺭﻭﻣﻴﺔ 8:3; العَدَد 7:22"> ثُمَّ جاءَ مَنْ يُنَبِّهُهُ على خطيئتِهِ الّتي اَرْتَكبها، فعلَيهِ أنْ يجيءَ بقربانِهِ تَيسًا مِنَ المعَزِ ذَكَرًا صحيحًا.</span>
<span className="v" id="v24" data-v="24" data-xref="اللّاويِّين 3:13; اللّاويِّين 7:2; اللّاويِّين 4:3–35"> يضَعُ يدَهُ على رأسِهِ ويذبَحُهُ في موضِعِ ذَبْحِ المُحرَقةِ أمامَ الرّبِّ هوَ ذبيحةُ خطيئةٍ.</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 4:7; اللّاويِّين 4:18; اللّاويِّين 4:30"> فيأخذُ الكاهنُ مِنْ دَمِ الذَّبيحةِ هذه بإصبعِهِ ويضَعُ على قرونِ مذبحِ المُحرَقةِ، ويَصُبُّ الدَّمَ عِندَ قاعدةِ المذبَحِ.</span>
<span className="v" id="v26" data-v="26" data-xref="اللّاويِّين 4:20; اللّاويِّين 4:31; اللّاويِّين 4:35"> ويوقِدُ جميعَ شَحْمِهِ على المذبَحِ، كما يفعَلُ بشَحْمِ ذبيحةِ السَّلامةِ. ويُكَفِّرُ عَنهُ الكاهنُ خطيئتَهُ فيُسامِحُهُ الرّبُّ.</span>
</div>
</div>

## ذبيحة العامة من الناس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 4:2; العَدَد 15:27; اللّاويِّين 4:13"> ”وإنْ خَطِئَ أحدٌ مِنْ عامَّةِ النَّاسِ سَهوًا وعمِلَ واحدةً مِمَّا نَهى الرّبُّ عَنْ فِعلِهِ وأثِمَ،</span>
<span className="v" id="v28" data-v="28" data-xref="اللّاويِّين 4:23; اللّاويِّين 4:3; اللّاويِّين 4:32"> ثُمَّ جاءَ مَنْ نَبَّهَهُ على خطيئتِهِ الّتي اَرْتَكبها، فليَجِئْ بِقُربانِهِ عَنْزًا مِنَ المعَزِ صحيحةً.</span>
<span className="v" id="v29" data-v="29" data-xref="اللّاويِّين 4:4; اللّاويِّين 4:24; اللّاويِّين 1:4"> يضَعُ يدَهُ على رأسِها ويذبَحُها ذبيحةَ خطيئةٍ في موضِعِ المُحرَقةِ.</span>
<span className="v" id="v30" data-v="30" data-xref="اللّاويِّين 4:7; اللّاويِّين 4:25; اللّاويِّين 4:34"> فيأخُذُ الكاهنُ مِنْ دَمِها بإصبعِهِ ويضَعُ على قرونِ مذبَحِ المُحرَقةِ، وسائرُ دَمِها يسكبُهُ عِندَ قاعدةِ المذبَحِ.</span>
<span className="v" id="v31" data-v="31" data-xref="اللّاويِّين 4:26; اللّاويِّين 4:35; اللّاويِّين 1:9"> وينزعُ جميعَ شَحْمِها، كما ينزعُ الشَّحْمَ مِنْ ذبيحةِ السَّلامةِ، ويوقِدُهُ على المذبَحِ رائحةَ رضًى للرّبِّ. ويُكَفِّرُ عَنهُ الكاهنُ خطيئتَهُ فيسامِحُهُ الرّبُّ.</span>
<span className="v" id="v32" data-v="32" data-xref="اللّاويِّين 4:28; رسالَةُ بُطرُسَ الأُولَى 3:18; رسالَةُ بُطرُسَ الأُولَى 2:22"> وإنْ هوَ جاءَ بقربانِهِ مِنَ الضأْنِ ذبيحةَ خطيئةٍ، فلتكنْ أنثى صحيحةً.</span>
<span className="v" id="v33" data-v="33" data-xref="اللّاويِّين 4:29–31; اللّاويِّين 4:4"> يضَعُ يدَهُ على رأسِها ويذبحُها ذبيحةَ خطيئةٍ في موضِعِ ذبْحِ المُحرَقةِ.</span>
<span className="v" id="v34" data-v="34" data-xref="اللّاويِّين 4:30; اللّاويِّين 4:25; اللّاويِّين 4:7"> فيأخذُ الكاهنُ مِنْ دَمِ الذَّبيحةِ هذِهِ بإصبَعِهِ ويرُشُّ على قرونِ مذبَحِ المُحرَقةِ، وسائرُ الدَّمِ يسكبُهُ عِندَ قاعدةِ المذبَحِ.</span>
<span className="v" id="v35" data-v="35" data-xref="اللّاويِّين 4:26; اللّاويِّين 4:20; اللّاويِّين 14:18"> وجميعُ شَحْمها ينزعُهُ، كما ينزعُ شَحْمَ ضأْنِ ذبيحةِ السَّلامةِ، ويوقِدُهُ على المذبَحِ معَ وقائدِ الرّبِّ، ويُكَفِّرُ عَنهُ الكاهنُ خطيئتَهُ فيسامِحُهُ الرّبُّ.</span>
