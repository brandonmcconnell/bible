## شريعة بلوى البرص

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وكلَّمَ الرّبُّ موسى وهرونَ فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 24:8; اللّاويِّين 14:56; اللّاويِّين 14:3"> ”مَنْ كانَ في جِلْدِ بَدَنِهِ نُتوءٌ أو طَفْحةٌ جِلْديَّةٌ أو لَمعَةٌ تُشيرُ إلى بلوى البرَصِ، فلْيَحضُرْ إلى هرونَ الكاهنِ أو إلى واحدٍ مِنْ بَنيهِ الكهَنةِ.</span>
<span className="v" id="v3" data-v="3" data-xref="ﺭﻭﻣﻴﺔ 3:19–20; اعمال الرسل 20:28; مَلاخِي 2:7"> فيفحَصَ الكاهنُ الدَّاءَ في جِلْدِ البدَنِ، فإنْ كانَ في موضِعِ الدَّاءِ شَعْرٌ أبيضُ، ومَنظرُ الدَّاءِ أعمقُ مِنْ جِلْدِ بَدَنِهِ، فهوَ بَلوى البرَصِ. فإذا رآهُ الكاهنُ كذلِكَ، يحكُمُ بِنَجاسَتِه.</span>
<span className="v" id="v4" data-v="4" data-xref="العَدَد 12:15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 4:5; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 5:24"> وإنْ كانَ الدَّاءُ لَمعَةً بيضاءَ في جِلْدِ بدَنِهِ، لا مَنظرُها أعمَقُ مِنَ الجِلْدِ ولا شَعْرُها أبيضُ، يحجُزُهُ الكاهنُ سَبعَةَ أيّامٍ.</span>
<span className="v" id="v5" data-v="5"> ثُمَّ يفحَصُهُ في اليومِ السَّابِعِ، فإذا رأى أنَّ الدَّاءَ لم يَتَفَشَّ في الجِلْدِ، يحجُزُهُ سَبعَةَ أيّامٍ أُخَرَ.</span>
<span className="v" id="v6" data-v="6" data-xref="اللّاويِّين 11:25; اللّاويِّين 14:8; التَّثْنِيَة 32:5"> ثُمَّ يفحَصُهُ في اليومِ السَّابعِ ثانيةً، فإنْ كانَ الدَّاءُ داكِنَ اللَّونِ وغيرَ مُتَفَشٍّ في الجِلْدِ، يحكُمُ بِطهارَتِه. هيَ طَفحَةٌ جِلْديَّةٌ، فيَغسِلُ ثيابَه ويَطهُرُ.</span>
<span className="v" id="v7" data-v="7" data-xref="المَزامِير 38:3; اللّاويِّين 13:35–36; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:16–17"> وإنْ تَفَشَّتِ الطَّفحَةُ في جِلْدِهِ، بَعدَما عَرَضَ نفْسَه على الكاهنِ لأجلِ تَطهيرِهِ، يحضُرُ إلى الكاهنِ ثانيةً.</span>
<span className="v" id="v8" data-v="8" data-xref="ﻣﺘﻰ 15:7–8; فِيلِبِّي 3:18–19; اعمال الرسل 8:21"> فإذا رأى الكاهنُ أنَّ الطَّفحَةَ تَفَشَّت في الجِلْدِ، يحكُمُ بِنَجاستِهِ، فهيَ برَصٌ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9"> ”وإذا أصابَت أحدًا بَلوى برَصٍ فأُحضِرَ إلى الكاهنِ</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الثّانِي 5:27; أخبارِ الأيّامِ الثّانِي 26:19–20; اللّاويِّين 13:14–15"> ففَحَصَهُ الكاهنُ فرأى في جِلْدِهِ نُتوءًا أبيضَ جعَلَ الشَّعْرَ أبيضَ وكانَ في النُّتوءِ لَحمٌ حيٌّ،</span>
<span className="v" id="v11" data-v="11"> فهوَ برَصٌ مُزمِنٌ في جِلْدِ بدَنِهِ. فيَحكُمُ الكاهنُ بِنَجاستهِ ولا يحجُزُهُ لأنَّهُ نَجِسٌ.</span>
<span className="v" id="v12" data-v="12" data-xref="رسالَةُ يُوحَنّا الأُولَى 1:8–10; أيُّوب 42:6; ﺭﻭﻣﻴﺔ 7:14"> وإنْ خرَجَ البرَصُ في البدَنِ وغطَّى المُبتَلَى بهِ مِنْ رأسِهِ إلى قدَمَيهِ، ممَّا يقَعُ تَحتَ بَصَرِ الكاهنِ،</span>
<span className="v" id="v13" data-v="13" data-xref="يُوحَنّا 9:41; إشَعْياء 64:6"> وفَحَصَهُ الكاهنُ فرأى البرَصَ غطَّى جميعَ بدَنِهِ، يحكُمُ بِطهارَتِهِ لأنَّهُ اَنْقَلبَ كُلُّهُ أبيضَ، فهوَ طاهرٌ.</span>
<span className="v" id="v14" data-v="14"> وإنْ جاءَ يومٌ ظهَرَ في البدَنِ لَحمٌ حيٌّ، فيكونُ نَجِسًا.</span>
<span className="v" id="v15" data-v="15"> فإذا رأى الكاهنُ اللَّحمَ الحيَّ، يحكُمُ بِنَجاستِهِ. فاللَّحمُ الحيُّ نَجِسٌ لأنَّه برَصٌ.</span>
<span className="v" id="v16" data-v="16" data-xref="غَلاطِيَّة 1:14–16; ﺭﻭﻣﻴﺔ 7:14–24; فِيلِبِّي 3:6–8"> وإنْ رجَعَ اللَّحمُ الحيُّ فاَبيضَّ يَجيءُ الرَّجُلُ إلى الكاهنِ.</span>
<span className="v" id="v17" data-v="17"> فإذا فحَصَهُ الكاهنُ فرأى أنَّ البَلوى اَنقَلَبَت بيضاءَ، يحكُمُ بِطهارتِهِ، فهوَ طاهرٌ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 9:9; إشَعْياء 38:21; المَزامِير 38:3–7"> ”وإذا كانَ أحدٌ في جِلْدِ بدَنِهِ دُملةٌ فبرِئَت،</span>
<span className="v" id="v19" data-v="19" data-xref="اللّاويِّين 13:24"> فصارَ في موضِعِ الدُّملةِ نُتوءٌ أبيضُ أو لَمعَةٌ بيضاءُ تَميلُ إلى حُمرَةٍ، فلتُعْرَضْ على الكاهنِ.</span>
<span className="v" id="v20" data-v="20" data-xref="يُوحَنّا 5:14; اللّاويِّين 13:3; رسالَةُ بُطرُسَ الثّانيَة 2:20"> فإذا فَحَصها الكاهنُ فرأى مَنظرَها أعمقَ مِنَ الجِلْدِ وشَعْرَها أبيضَ، يحكُمُ بِنَجاسَتِهِ. فهيَ بَلوى برَصٍ طَلَعَت في الدُّملةِ.</span>
<span className="v" id="v21" data-v="21" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:5"> وإنْ فحَصَها الكاهنُ فما رأى فيها شَعْرًا أبيضَ، ولا وجَدَها أعمَقَ مِنَ الجِلْدِ، وهيَ داكِنَةُ اللَّونِ، يحجُزُهُ سَبعةَ أيّامٍ.</span>
<span className="v" id="v22" data-v="22"> فإنْ هيَ تَفَشَّت في الجِلْدِ يحكُمُ الكاهنُ بِنجاستِهِ، فهيَ بَلوى.</span>
<span className="v" id="v23" data-v="23" data-xref="غَلاطِيَّة 6:1; أيُّوب 34:31–32; ﻣﺘﻰ 26:75"> ولكِنْ إنْ وَقَفَتِ اللَّمعَةُ مكانَها ولم تَتَفَشَّ، فهيَ أثَرُ الدُّملَةِ. فيَحكُمُ الكاهنُ بِطهارتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 3:24"> ”وإذا كانَ أحدٌ في جِلْدِ بدَنِهِ كَيُّ نارٍ، وكانَ أثرُ الكَيِّ لَمعَةً بيضاءَ أو بيضاءَ تَميلُ إلى حُمرةٍ،</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 13:18–20; اللّاويِّين 13:4"> فيَفحَصُها الكاهنُ. فإنْ كانَ الشَّعْرُ أبيضَ في اللَّمعَةِ، وكانَ مَنظَرُها أعمقَ مِنَ الجِلْدِ، فهذا برَصٌ طَلَعَ في الكَيِّ. فيحكُمُ الكاهنُ بنجاستِهِ. فهوَ بَلوى برَصٍ.</span>
<span className="v" id="v26" data-v="26" data-xref="اللّاويِّين 13:23; اللّاويِّين 13:4–5"> ولكنْ إذا فحَصَها الكاهنُ فما رأى في اللَّمعَةِ شَعْرًا أبيضَ، ولا وجَدَها أعمقَ مِنَ الجِلْدِ، وهيَ داكِنَةُ اللَّونِ، يَحجُزُه سَبعَةَ أيّامٍ.</span>
<span className="v" id="v27" data-v="27"> ثُمَّ يَفحَصُهُ الكاهنُ في اليومِ السَّابعِ، فإنْ كانت تَفَشَّت في الجِلْدِ يَحكُمُ بِنجاستِهِ، فهوَ بَلوى برَصٍ.</span>
<span className="v" id="v28" data-v="28"> وإنْ وقَفَتِ اللَّمعَةُ في مكانِها ولم تَتَفَشَّ في الجِلْدِ وكانت داكِنةَ اللَّونِ، فهيَ نُتوءُ الكَيِّ. فيحكُمُ الكاهنُ بِطهارتِهِ، فهيَ أثَرُ الكَيِّ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="إشَعْياء 9:15; ﻣﺘﻰ 13:14–15; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 4:3–4"> ”وإذا كانت بِرَجُلٍ أوِ اَمرأةٍ بَلوى في رأسهِ أو ذَقنِهِ،</span>
<span className="v" id="v30" data-v="30" data-xref="اللّاويِّين 13:34–37; اللّاويِّين 14:54"> فليَفحَصِ الكاهنُ البَلوى. فإنْ كانَ مَنظَرُها أعمقَ مِنَ الجِلْدِ، وفيها شَعْرٌ أشقرُ دقيقٌ، يحكُمُ الكاهنُ بِنجاستِهِ، فهوَ قَرَعٌ، أي برَصُ الرَّأسِ أو الذَّقنِ.</span>
<span className="v" id="v31" data-v="31"> فإنْ فحَصَها الكاهنُ ولم يَجِدْ مَنظَرَها أعمقَ مِنَ الجِلْدِ، ولا فيها شَعْرٌ أسودُ، يحجُزُ المُبْتَلَى بالقَرَعِ سَبعَةَ أيّامٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="اللّاويِّين 13:30; ﻣﺘﻰ 23:5; لُوقا 18:9–12"> ثُمَّ يَفحَصُهُ في اليومِ السَّابعِ، فإنْ كانَ القَرَعُ غيرَ مُتَفَشٍّ، ولا فيه شَعْرٌ أشقرُ، ولا مَنظَرُهُ أعمقُ مِنَ الجِلْد،</span>
<span className="v" id="v33" data-v="33"> يَحلِقُ الرَّجُلُ شَعْرَه ولا يَحلِقُ موضِعَ القَرَعِ، ويحجُزُهُ الكاهنُ سَبعَةَ أيّامٍ أُخَرَ.</span>
<span className="v" id="v34" data-v="34" data-xref="يَهُوذا 1:22; رسالَةُ يُوحَنّا الأُولَى 4:1; رُؤيا يُوحَنّا 2:2"> ثُمَّ يَفحَصُه في اليومِ السَّابعِ، فإنْ كانَ القَرَعُ غيرَ مُتَفَشٍّ في الجِلْدِ، ولا مَنظَرُهُ أعمقَ مِنَ الجِلْدِ، يحكُمُ بِطهارتِهِ، فيَغسِلُ ثيابَهُ ويَطهُرُ.</span>
<span className="v" id="v35" data-v="35" data-xref="اللّاويِّين 13:27; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:16–17; اللّاويِّين 13:7"> ولكنْ إنْ تَفَشَّى القَرَعُ في جِلْدِ الرَّجُلِ بَعدَما طَهُرَ،</span>
<span className="v" id="v36" data-v="36"> ففَحَصَهُ الكاهنُ فرأى ذلِكَ، فلا يُفَتِّشُ عَنِ الشَّعْرِ الأشقرِ لأنَّ الرَّجُلَ نَجِسٌ.</span>
<span className="v" id="v37" data-v="37"> ولكنْ إنْ رأى أنَّ القَرَعَ وَقَفَ ونَبَتَ فيهِ شَعْرٌ أسودُ، فيكونُ بَرَأ والرَّجُلُ طاهرٌ. فيحكُمُ الكاهنُ بِطهارتِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38"> ”وإذا كانت لِرَجُلٍ أوِ اَمرأةٍ لُمَعٌ بيضٌ في جِلْدِ بدَنِهِ،</span>
<span className="v" id="v39" data-v="39" data-xref="ﺭﻭﻣﻴﺔ 7:22–25; يَعقُوب 3:2; الجامِعَة 7:20"> فيَفحَصُهُ الكاهنُ. فإذا كانتِ اللُّمَعُ داكِنةَ اللَّونِ تَميلُ إلى بياضٍ، فهوَ بَهَقٌ خرَجَ في الجِلْدِ، والرَّجُلُ طاهرٌ.</span>
</div>
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="غَلاطِيَّة 4:13; ﺭﻭﻣﻴﺔ 8:10; نَشِيدِ الأنشاد 5:11"> ”مَنْ سقَطَ شَعْرُ رأسِهِ فهوَ أقرعُ، وهو طاهرٌ.</span>
<span className="v" id="v41" data-v="41"> وإنْ كانَ سُقوطُ شَعْرِهِ مِنْ جِهةِ وجهِهِ فهوَ أصلَعُ، وهوَ طاهرٌ.</span>
<span className="v" id="v42" data-v="42"> وإنْ كانَ في الحَصَصِ أوِ الصَّلْعَةِ بَلوى بيضاءُ تَميلُ إلى حُمرةٍ، فهوَ برَصٌ خرَجَ في قُرعتِهِ أو في صَلْعَتِهِ.</span>
<span className="v" id="v43" data-v="43"> فيفحَصُهُ الكاهنُ، فإنْ كانَ نُتوءُ البَلوى أبيضَ مائِلًا إلى حُمرةٍ، كَمَنظَرِ برَصِ جِلْدِ البدَنِ،</span>
<span className="v" id="v44" data-v="44" data-xref="رسالَةُ بُطرُسَ الثّانيَة 2:1–2; ﻣﺘﻰ 6:23; إشَعْياء 1:5"> فالرَّجُلُ أبرَصُ، وهوَ نَجِسٌ. فيحكُمُ الكاهنُ بِنجاستِهِ مِنَ البَلوى في رأسِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v45" data-v="45" data-xref="ميخا 3:7; اللّاويِّين 10:6; حِزْقِيال 24:17"> ”والأبرَصُ الّذي بهِ البَلوى يَلبَسُ ثيابَهُ مَشقُوقةً، ويكشِفُ رأسَه، ويُغَطِّي شاربَيهِ ويُنادي: نَجِسٌ، نَجِسٌ.</span>
<span className="v" id="v46" data-v="46" data-xref="المُلُوكِ الثّانِي 7:3; المُلُوكِ الثّانِي 15:5; أخبارِ الأيّامِ الثّانِي 26:21"> ما دامت بهِ البَلوى يكونُ نَجِسًا، ويسكُنُ مُنفرِدًا وفي خارِجِ المَحلَّةِ.</span>
</div>

## شريعة بلوى العفن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v47" data-v="47" data-xref="يَهُوذا 1:23; ﺭﻭﻣﻴﺔ 13:12; حِزْقِيال 16:16"> ”وإذا كانت بَلوى العَفَنِ في ثوبٍ مِنْ صُوفٍ أو كَتَّانٍ،</span>
<span className="v" id="v48" data-v="48" data-xref="اللّاويِّين 13:51; رُؤيا يُوحَنّا 3:4; التَّثْنِيَة 8:11"> أو في ثوبٍ سَداهُ أو لُحمَتُهُ مِنْ كتَّانٍ أو صُوفٍ، أو في جِلْدٍ أو كُلِّ ما يُصنَعُ مِنَ الجِلْدِ،</span>
<span className="v" id="v49" data-v="49"> وكانَ مَنظَرُها لا يَميلُ إلى خُضرةٍ أو حُمرةٍ، فذلِكَ هوَ بَلوى العَفَنِ ويُعرَضُ على الكاهنِ.</span>
<span className="v" id="v50" data-v="50" data-xref="حِزْقِيال 44:23"> فيفحَصُ الكاهنُ البَلوى ويُقفِلُ على المَتاعِ الّذي أصابَتهُ البَلوى سَبعَةَ أيّامٍ.</span>
