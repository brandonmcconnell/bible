## شريعة ذبيحة الإثم

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اللّاويِّين 6:17; اللّاويِّين 5:14–7; اللّاويِّين 6:25"> ”وهذِهِ شريعةُ ذبيحةِ الإثْمِ، وهيَ قُربانٌ مُقَدَّسٌ كُلَّ التَّقديسِ:</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 1:5; اللّاويِّين 6:25; اللّاويِّين 1:11"> في موضِعِ ذَبْحِ المُحرَقةِ يذبَحُ الكاهنُ ذبيحةَ الإثْمِ ويَرُشُّ دَمَها على أربعَةِ جوانبِ المذبَحِ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 29:13; اللّاويِّين 3:15–16; المَزامِير 51:6"> ويُقَرِّبُ مِنها جميعَ شَحْمِها: الألْيَةَ والشَّحْمَ الّذي يُغَطِّي الأمعاءَ،</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 3:4"> والكلْيتَينِ والشَّحْمَ الّذي علَيهما عِندَ الخاصِرتَينِ، والكَبِدَ بَعدَ نَزْعِ زيادَتِهِ معَ الكلْيتَينِ.</span>
<span className="v" id="v5" data-v="5" data-xref="اللّاويِّين 1:13; غَلاطِيَّة 5:24; رسالَةُ بُطرُسَ الأُولَى 4:1–2"> ويُوقِدُ الكاهنُ الذَّبيحةَ على المذبَحِ وقيدةً للرّبِّ. هيَ ذبيحةُ إثْمٍ.</span>
<span className="v" id="v6" data-v="6" data-xref="اللّاويِّين 6:29; العَدَد 18:9–10; اللّاويِّين 2:3"> كُلُّ ذَكَرٍ مِنَ الكهَنةِ يأكُلُ مِنها في موضِعٍ مُقَدَّسٍ هيَ قُربانٌ مُقَدَّسٌ كُلَّ التَّقديس.</span>
</div>
</div>

## نصيب الكاهن

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="اللّاويِّين 6:25–26; اللّاويِّين 14:13; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 9:13"> ”لِذبيحةِ الإثْمِ وذبيحةِ الخطيئةِ شريعةٌ واحدةٌ، والكاهنُ الّذي يُكَفِّرُ بها تكونُ مِنْ نَصيبهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 4:11; الخُرُوج 29:14; ﺭﻭﻣﻴﺔ 13:14"> والكاهنُ الّذي يُقَرِّبُ مُحرَقةَ أحدٍ مِنَ النَّاسِ، يكونُ جِلْدُها مِنْ نَصيبِهِ بَعدَ تَقريبِها.</span>
<span className="v" id="v9" data-v="9" data-xref="العَدَد 18:9; حِزْقِيال 44:29; اللّاويِّين 2:10"> كُلُّ تقدِمةٍ مِمَّا تُخبَزُ في التَنُّورِ أو تُقلَى في مِقلاةٍ أو تُخبَزُ على الصَّاجِ، تكونُ للكاهنِ الّذي يُقَرِّبُها.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:14; الخُرُوج 16:18"> وكُلُّ تقدِمةٍ مَلتوتةٍ بِزيتٍ أو جافةٍ تكونُ لجميعِ بَني هرونَ بالتَّساوي.</span>
</div>
</div>

## شريعة ذبيحة السلامة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="اللّاويِّين 22:18–21; حِزْقِيال 45:15; اللّاويِّين 3:1–17"> ”وهذِهِ شريعةُ ذبيحةِ السَّلامةِ الّتي تُقَرَّبُ للرّبِّ:</span>
<span className="v" id="v12" data-v="12" data-xref="اللّاويِّين 2:4; العَدَد 6:15; رسالَةُ بُطرُسَ الأُولَى 2:5"> إذا قُدِّمَت للشُّكرِ يُقَرَّبُ معَها قُربانٌ مِنْ أقراصِ فطيرٍ مَلتُوتةٍ بِزيتٍ، ورِقاقِ فطيرٍ مَمسوحةٍ بِزيتٍ، ودقيقٍ مَدلوكٍ أقراصًا مَلتوتةً بِزيتٍ.</span>
<span className="v" id="v13" data-v="13" data-xref="عامُوس 4:5; اللّاويِّين 23:17; ﻣﺘﻰ 13:33"> معَ أقراصِ خُبزِ خَميرٍ يُقدَّمُ هذا القُربانُ معَ ذبيحةِ السَّلامةِ الّتي للشُّكرِ.</span>
<span className="v" id="v14" data-v="14" data-xref="العَدَد 18:19; الخُرُوج 29:27–28; العَدَد 18:24–32"> ويُرفَعُ واحدٌ مِنْ كُلِّ ذلكَ ويُقَرَّبُ قُربانًا للربِّ، والباقي يكونُ للكاهنِ الّذي يَرُشُّ دمَ ذبيحةِ السَّلامةِ.</span>
<span className="v" id="v15" data-v="15" data-xref="اللّاويِّين 22:29–30"> ولَحمُ ذبيحةِ السَّلامةِ الّتي للشُّكرِ يُؤكَلُ في يومِ تَقريبِهِ، لا يَبقَ مِنهُ إلى الغَدِ.</span>
<span className="v" id="v16" data-v="16" data-xref="اللّاويِّين 19:5–8; التَّثْنِيَة 12:6; المَزامِير 116:14"> وإنْ كانتِ الذَّبيحةُ نَذْرًا أو تَبَرُّعًا، فلتُؤكَلْ في يومِ تقريبِها، وما فضَلَ مِنها يُؤكَلُ في الغَدِ.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 12:10; هُوشَع 6:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:4"> وما بقيَ مِنها إلى اليومِ الثَّالثِ يُحرَقُ بالنَّارِ.</span>
<span className="v" id="v18" data-v="18" data-xref="اللّاويِّين 19:7–8; اللّاويِّين 11:41; العَدَد 18:27"> وإنْ أكلَ أحدٌ مِنها في اليومِ الثَّالِثِ، فلا يَرضى عَنها الرّبُّ، والّذي قَرَّبَها لا تُحسَبُ لَه، بل تكونُ رِجْسًا، ومَنْ أكلَ مِنها تحَمَّلَ عاقِبةَ إثْمِهِ.</span>
<span className="v" id="v19" data-v="19" data-xref="لُوقا 11:41; تِيطُس 1:15; اعمال الرسل 10:15–16"> وإذا مسَّ لَحمُها شيئًا نَجِسًا، فلا يُؤكَلُ بل يُحرَقُ بالنَّارِ، ولَحمُها يأكلُهُ مَنْ كانَ طاهرًا.</span>
</div>
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="اللّاويِّين 15:2–33; التَّكوين 17:14; اللّاويِّين 22:3–7"> ”كُلُّ مَنْ أكلَ لَحمًا مِنْ ذبيحةِ السَّلامةِ الّتي للرّبِّ وكانَ نَجِسًا أقطَعُهُ مِنْ شعبِهِ.</span>
<span className="v" id="v21" data-v="21" data-xref="اللّاويِّين 12:1–59; اللّاويِّين 11:20; حِزْقِيال 4:14"> وكُلُّ مَنْ لامَسَ شيئًا نَجِسًا لإنسانٍ أو بَهيمةٍ، أو أيَّةِ رجاسةٍ ما نَجِسةٍ، ثُمَّ أكلَ مِنْ ذبيحةِ السَّلامةِ الّتي للرّبِّ أقطَعُه مِنْ شعبِهِ“.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v23" data-v="23" data-xref="اللّاويِّين 3:16–17; اللّاويِّين 17:6; صَمُوئيلَ الأوَّلُ 2:15–17"> ”قُلْ لِبني إسرائيلَ: كُلُّ شَحْمٍ مِنْ بقَرٍ أو ضَأنٍ أو معَزٍ لا تأكُلُوهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 22:31; اللّاويِّين 17:15; اللّاويِّين 22:8"> وشَحْمُ الحيوانِ المَيْتِ والفريسةِ تستعمِلونَه في أيَّةِ صَنعَةٍ، ولكنْ لا تَأكُلُوه.</span>
<span className="v" id="v25" data-v="25"> مَنْ أكلَ شَحْمًا مِنَ البهيمةِ الّتي يُقَرَّبُ مِنها وقِيدةً للرّبِّ أقطَعُهُ مِنْ شعبِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="التَّكوين 9:4; اللّاويِّين 3:17; اعمال الرسل 15:20"> كُلُّ دَمٍ مِنَ الطَّيرِ والبَهائِمِ لا تأكُلُوه أينَما سَكَنتُم.</span>
<span className="v" id="v27" data-v="27" data-xref="العِبرانِيّين 10:29; اللّاويِّين 7:20–21; اللّاويِّين 7:25"> ومَنْ أكلَ شيئًا مِنَ الدَّمِ أقطَعُه مِنْ شعبِه“.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28"> وكلَّمَ الرّبُّ موسى فقالَ:</span>
<span className="v" id="v29" data-v="29" data-xref="كُولُوسِي 1:20; رسالَةُ يُوحَنّا الأُولَى 1:7; اللّاويِّين 3:1–17"> ”قُلْ لِبني إِسرائيلَ: مَنْ قرَّبَ ذبيحةَ سلامةٍ لِلرّبِّ، فَليُقَرِّبْ مِنها للرّبِّ</span>
<span className="v" id="v30" data-v="30" data-xref="العَدَد 6:20; اللّاويِّين 3:9; اللّاويِّين 3:14"> وقيدةً مِنَ الشَّحْمِ المُلتَصِقِ بالصَّدْرِ تحمِلُها يَداهُ معَ الصَّدْرِ، فيُحَرِّكُ الصَّدْرَ تحريكًا أمامَ الرّبِّ لِيُكَرِّسَهُ</span>
<span className="v" id="v31" data-v="31" data-xref="اللّاويِّين 3:5; اللّاويِّين 3:11; اللّاويِّين 3:16"> وأمَّا الشَّحْمُ فيُوقِدُه الكاهنُ على المذبَحِ. ويكونُ الصَّدْرُ لِهرونَ وبَنيهِ الكهَنةِ.</span>
<span className="v" id="v32" data-v="32" data-xref="العَدَد 6:20; اللّاويِّين 7:34; اللّاويِّين 9:21"> وأعطُوا الكَتِفَ اليُمنى لِمَنْ مِنهم</span>
<span className="v" id="v33" data-v="33" data-xref="اللّاويِّين 26:1–46; اللّاويِّين 6:1–30; اللّاويِّين 7:3"> قرِّبْ دَمَ ذبيحةِ السَّلامةِ وشحْمَها</span>
<span className="v" id="v34" data-v="34" data-xref="العَدَد 18:18–19; اللّاويِّين 10:14–15; الخُرُوج 29:9"> فيكونُ أنِّي أخذتُ الصَّدْرَ والكَتِفَ اليُمنى مِنْ ذبائِحِ سلامةِ بَني إِسرائيلَ وأعطيتُها لِهرونَ الكاهنِ ولبَنيهِ. هذِهِ فريضةٌ لِبني إِسرائيلَ إلى الأبدِ“.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="رسالَةُ يُوحَنّا الأُولَى 2:20; رسالَةُ يُوحَنّا الأُولَى 2:27; العَدَد 18:7–19"> تِلكَ حِصَّةُ هرونَ وبَنيهِ مِنَ الذَّبائِحِ الّتي توقَدُ للرّبِّ يومَ تقديمِهِم ليكونوا كهَنةً لَه،</span>
<span className="v" id="v36" data-v="36" data-xref="اللّاويِّين 8:12; اللّاويِّين 8:30; الخُرُوج 40:13–15"> أمرَ الرّبُّ بِها فريضَةً أبديَّةً على مَمَرِّ أجيالِهِم،</span>
<span className="v" id="v37" data-v="37" data-xref="الخُرُوج 29:1; اللّاويِّين 1:1–6; اللّاويِّين 6:9–18"> وتلكَ شريعةُ المُحرَقةِ والتَّقدمةِ وذبيحةِ الخطيئةِ وذبيحةِ الإثْمِ وتكريسِ الكهَنةِ وذبيحةِ السَّلامةِ،</span>
<span className="v" id="v38" data-v="38" data-xref="اللّاويِّين 1:1–2"> أمرَ الرّبُّ بها موسى في جبَلِ سيناءَ يومَ أمرَ بَني إِسرائيلَ بأنْ يُقرِّبوا قرابينَهُم للرّبِّ في بريَّةِ سيناءَ“.</span>
</div>
