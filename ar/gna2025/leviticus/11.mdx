## شريعة الطاهر والنجس من الحيوان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ الرّبُّ لموسى وهرونَ:</span>
<span className="v" id="v2" data-v="2" data-xref="ﻣﺘﻰ 15:11; اللّاويِّين 11:11; العِبرانِيّين 9:10"> ”قُولا لبَني إِسرائيلَ هذِهِ هيَ الحيواناتُ الّتي تأكُلونَها مِنْ جميعِ البَهائِمِ الّتي على الأرضِ:</span>
<span className="v" id="v3" data-v="3" data-xref="الأمثال 2:10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:17; التَّثْنِيَة 6:6–7"> جميعُ ما هوَ مشقوقُ الظِّفْرِ ويجتَرُّ مِنَ البَهائِمِ.</span>
<span className="v" id="v4" data-v="4"> أمَّا الحيواناتُ الّتي تجتَرُّ، وأظفارُها غيرُ مشقوقةٍ أو الّتي لا تَجتَرُّ وأظفارُها مشقوقةٌ، فلا تأكلوها لأنَّها نَجِسٌ لكُم: الجَمَلُ لأنَّهُ يجتَرُّ ولكنَّه غيرُ مشقوقِ الظِّفْرِ،</span>
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 104:18; الأمثال 30:26; ﻣﺘﻰ 7:26"> والغَرغورُ لأنَّهُ يجتَرُّ ولكنَّه غيرُ مشقوقِ الظِّفْرِ،</span>
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 14:7"> والأرنَبُ لأنَّهُ يجتَرُّ ولكِنَّه غيرُ مشقوقِ الظِّفْرِ،</span>
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 66:17; إشَعْياء 65:4; إشَعْياء 66:3"> والخِنزيرُ لأنَّهُ مشقوقُ الظِّفْرِ ولكنَّه لا يجتَرُّ.</span>
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 52:11; كُولُوسِي 2:16; العِبرانِيّين 9:10"> مِنْ لَحمِها لا تأكلوا ولا تَمَسُّوها مَيتَةً، فهيَ نَجِسةٌ لكُم.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="غَلاطِيَّة 5:6; يَعقُوب 2:18; اعمال الرسل 20:21"> ”وهذا ما تأكلونَه مِنْ جميعِ ما في الماءِ: جميعُ ما لَه زعانِفُ وقُشورٌ في البِحارِ والأنهارِ.</span>
<span className="v" id="v10" data-v="10" data-xref="الأمثال 29:27; اللّاويِّين 7:18; المَزامِير 139:21–22"> وما عدا ذلِكَ مِمَّا لا زعانِفَ وقشورَ لَه مِنْ جميعِ ما يزحفُ في الماءِ مِنَ الحيوانِ، فهوَ رِجْسٌ لكُم.</span>
<span className="v" id="v11" data-v="11"> مِنْ لَحمِهِ لا تأكلوا، وإنْ كانَ جُثَّةً فتَجَنَّبُوه لأنَّه رِجْسٌ.</span>
<span className="v" id="v12" data-v="12"> كُلُّ ما لا زعانِفَ وقُشورَ لَه مِمَّا في الماءِ، فهو رجْسٌ لكُم.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="أيُّوب 39:27–30; إرْمِيا 4:22; تِيطُس 3:3"> ”وهذا ما تَتَجَنَّبونَه مِنَ الطَّيرِ ولا تأكلونَه لأنَّه رِجْسٌ: النَّسْرُ والأنوقُ والعِقابُ،</span>
<span className="v" id="v14" data-v="14"> والحَدأُ والصَّدى بأصنافِها،</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 12:24; المُلُوكِ الأوَّلُ 17:6; الأمثال 30:17"> وجميعُ الغُربانِ بأصنافِها</span>
<span className="v" id="v16" data-v="16" data-xref="إشَعْياء 34:11–15; أفَسُس 5:7–11; أفَسُس 2:2–3"> والنَّعامُ والخطَّافُ والسَّأفُ والبازيُّ بأصنافِه،</span>
<span className="v" id="v17" data-v="17"> والبُومُ والزَمُّجُ والباشَقُ،</span>
<span className="v" id="v18" data-v="18"> والشَّاهينُ والقُوقُ والرَّخَمُ،</span>
<span className="v" id="v19" data-v="19"> والصَّقْرُ والبَبَّغاءُ بأصنافِهِ، والهُدهُدُ، والخفَّاشُ،</span>
<span className="v" id="v20" data-v="20" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:10; فِيلِبِّي 3:18–19; اللّاويِّين 11:23"> ولا تأكلوا الحشَراتِ المُجَنَّحَةَ الّتي تمشي على أربَعٍ، فهوَ رِجْسٌ لكُم،</span>
<span className="v" id="v21" data-v="21"> ما عدا الّتي لها رِجْلانِ أطولُ مِنْ يَدَيها تقفزُ بهما على الأرضِ،</span>
<span className="v" id="v22" data-v="22" data-xref="ﻣﺘﻰ 3:4; مَرْقُس 1:6; الخُرُوج 10:4–5"> كالجَرادِ بأصنافِهِ، والدَّبَى بأصنافِه، والحَرجُوانِ بأصنافِهِ، والجُنْدبِ بأصنافِهِ.</span>
<span className="v" id="v23" data-v="23"> وأمَّا سائِرُ الطَّيرِ الّذي يَدِبُّ ولَه أربَعُ أرجُلٍ، فهو رِجْسٌ لكُم.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:33; أفَسُس 2:1–3; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:17"> مِنْ هذِهِ تتَنَجَّسونَ، وكُلُّ مَنْ مَسَّ جُثَّتَها يكونُ نَجِسًا إلى المَغيبِ،</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 15:5; اللّاويِّين 14:8; العَدَد 31:24"> وكُلُّ مَنْ حمَلَ جُثَّتَها يغسِلُ ثيابَه ويكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v26" data-v="26"> كُلُّ حيوانٍ لَه ظِفْرٌ غيرُ مَشقوقٍ وكُلُّ ما لا يجترُّ، فهوَ نَجِسٌ لكُم. كُلُّ مَنْ مَسَّ جُثَّتَه يكونُ نَجِسًا.</span>
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 11:23; اللّاويِّين 11:20"> وكُلُّ ما يسعى على راحَتَيه مِنْ جميعِ الوحوشِ ويمشي على أربَعٍ فهوَ نَجِسٌ لكُم. وكُلُّ مَنْ مَسَّ جُثَّتَه يكونُ نَجِسًا إلى المَغيبِ،</span>
<span className="v" id="v28" data-v="28" data-xref="اللّاويِّين 11:24–25"> وكُلُّ مَنْ حمَلَ جُثَّتَه يَغسِلُ ثيابَهُ ويكونُ نَجِسًا إلى المَغيبِ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="اللّاويِّين 11:41–42; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 3:2–5; المَزامِير 10:3"> ”وهذا هوَ النَّجِسُ لكُم مِمَّا يدِبُّ على الأرضِ: الخِلْدُ والفأرُ والضَّبُّ بأصنافِها،</span>
<span className="v" id="v30" data-v="30"> والورَلُ والحِرْذَونُ والعِظاءةُ والحِرباءُ وسامٌّ أبرصُ.</span>
<span className="v" id="v31" data-v="31" data-xref="اللّاويِّين 11:24–25; اللّاويِّين 11:8"> هذِهِ نَجِسةٌ لكُم مِنْ جميع ما يَدِبُّ. مَنْ مسَّ جُثَّتَها يكونُ نجِسًا إلى المَغيبِ،</span>
<span className="v" id="v32" data-v="32" data-xref="اللّاويِّين 15:12; تِيطُس 3:5; تِيطُس 2:14"> وكُلُّ ما وُضِعَ علَيه شيءٌ مِنها بَعدَ موتِها يكونُ نَجِسًا: أوعيةُ الخشَبِ، والثِّيابُ، والجِلْدُ، والمَسْحُ وكُلُّ وِعاءٍ مُستعمَلٍ، يُنقَعُ في الماءِ ويكونُ نَجِسًا إلى المَغيبِ ثُمَّ يُطهَّرُ.</span>
<span className="v" id="v33" data-v="33" data-xref="اللّاويِّين 6:28; اللّاويِّين 15:12; اللّاويِّين 14:45"> وكُلُّ وِعاءِ خَزَفٍ مِنها وقعَ شيءٌ في وسَطهِ يكونُ كُلُّ ما في داخلِهِ نَجِسًا، فاَكسِروهُ.</span>
<span className="v" id="v34" data-v="34" data-xref="تِيطُس 1:15; الأمثال 21:4; الأمثال 21:27"> فإنْ كانَ طَعامًا يُؤكلُ، فهوَ نَجِسٌ ولو غُسِلَ بالماءِ، وإنْ كانَ شَرابًا يُشرَبُ، فهوَ نَجِسٌ مهما يَكُنْ وِعاؤُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="اللّاويِّين 15:12; اللّاويِّين 6:28; اللّاويِّين 11:33"> ”وكُلُّ ما وقَعَ علَيهِ مِنْ جُثَّتِها يكونُ نَجِسًا، فإنْ كانَ تَنُّورًا أو مَوقدةً فاَهدموهُما. هيَ نَجِسةٌ، فَنَجِسةً تكونُ لكُم.</span>
<span className="v" id="v36" data-v="36" data-xref="زَكَريّا 13:1; يُوحَنّا 4:14"> أمَّا النَّبْعُ والبِئْرُ وكُلُّ مُجتمعِ ماءٍ، فيكونُ طاهرًا. ولكنْ إذا مَسَّها وهي جُثَّةٌ فيكونُ نجسًا.</span>
<span className="v" id="v37" data-v="37" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:37; رسالَةُ بُطرُسَ الأُولَى 1:23; رسالَةُ يُوحَنّا الأُولَى 5:18"> وإنْ وقَعَ شيءٌ مِنْها وهي جثَّةٌ على أيِّ بِذرٍ يُزرَعُ، فهوَ طاهرٌ.</span>
<span className="v" id="v38" data-v="38"> فإنْ كانَ على البَذرِ ماءٌ ووقَعَ شيءٌ مِنها علَيهِ وهيَ جُثَّةٌ، فهوَ نَجِسٌ لكُم.</span>
</div>
<div className="bp">
<span className="v" id="v39" data-v="39" data-xref="اللّاويِّين 11:40; اللّاويِّين 11:24; اللّاويِّين 15:7"> ”وإذا ماتَ حيوانٌ مِمَّا يَحِلُّ لكُم أكلُهُ، فإنْ مَسَّهُ أحدٌ يكونُ نَجِسًا إلى المَغيبِ.</span>
<span className="v" id="v40" data-v="40" data-xref="اللّاويِّين 22:8; حِزْقِيال 44:31; التَّثْنِيَة 14:21"> ومَنْ أكلَ مِنْ جُثَّتِهِ يَغسِلُ ثيابَه ويكونُ نَجِسًا إلى المَغيب. ومَنْ حمَلَ جُثَّتَه يَغسِلُ ثيابَه ويكونُ نَجِسًا إلى المَغيبِ.</span>
</div>
<div className="bp">
<span className="v" id="v41" data-v="41" data-xref="اللّاويِّين 11:29; اللّاويِّين 11:20; اللّاويِّين 11:23"> ”وجميعُ الحشَراتِ الّتي تدِبُّ على الأرضِ هي رجسٌ لا يُؤكَلُ.</span>
<span className="v" id="v42" data-v="42" data-xref="تِيطُس 1:12; إشَعْياء 65:25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 11:3"> ولا تأكلوا كُلَّ ما حَبا على بَطنِه وعلى أربَعٍ، وكُلَّ ما كثُرَت أرجلُهُ، فهوَ رِجْسٌ.</span>
<span className="v" id="v43" data-v="43" data-xref="اللّاويِّين 20:25; اللّاويِّين 11:41–42"> لا تُدَنِّسوا أنفُسَكُم بشيءٍ مِنهُ ولا تتنجَّسوا بهِ، فتكونوا نَجِسينَ.</span>
</div>
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="الخُرُوج 20:2; رسالَةُ بُطرُسَ الأُولَى 1:15–16; اللّاويِّين 19:2"> ”أنا الرّبُّ إلهُكُم، فتَقَدَّسوا وكونوا قدِّيسينَ لأنِّي أنا قدُّوسٌ. ولا تُنَجِّسوا أنفُسَكُم بشيءٍ مِنَ الحشَراتِ الّتي تدِبُّ مُتَحرِّكةً على الأرضِ.</span>
<span className="v" id="v45" data-v="45" data-xref="اللّاويِّين 11:44; الخُرُوج 20:2; الخُرُوج 6:7"> فأنا الرّبُّ الّذي أخرَجَكُم مِنْ أرضِ مِصْرَ لأكونَ إلهًا لكُم، فكونوا قدِّيسينَ لأنِّي أنا قدُّوسٌ.</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="اللّاويِّين 14:54; اللّاويِّين 15:32; حِزْقِيال 43:12"> هذِهِ شريعةُ البَهائمِ والطَّيرِ وجميعِ المخلوقاتِ الحَيَّةِ مِمَّا يتَحرَّكُ في الماءِ ومِمَّا يدِّبُّ على الأرضِ.</span>
<span className="v" id="v47" data-v="47" data-xref="اللّاويِّين 10:10; حِزْقِيال 44:23; ﺭﻭﻣﻴﺔ 14:13–23"> بها تُميِّزونَ بَينَ النَّجِسِ والطَّاهرِ، وبَينَ الحيوانِ الّذي يُؤكَلُ والّذي لا يُؤكَلُ“.</span>
</div>
