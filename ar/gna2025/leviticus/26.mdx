## جزاء الطاعة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اللّاويِّين 19:4; التَّثْنِيَة 27:15; العَدَد 33:52"> ”لا تصنَعوا لكُم أوثانًا ولا تَماثيلَ مَنحوتةً، ولا تُقيموا نُصْبًا ولا أحجارًا مُزَخرَفةً في أرضِكُم لِتَسجدوا لها، لأنِّي أنا الرّبُّ إلهُكُم.</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 19:30"> اَحتفِظوا لي بأيّامِ السَّبْتِ وتَهَيَّبوا معبَدي المُقَدَّسَ. أنا الرّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 1:19; التَّثْنِيَة 11:13–15; التَّثْنِيَة 28:1–14"> ”إنْ سلَكتُم في فرائِضي وحَفِظتُم وصايايَ وعَمِلتُم بِها</span>
<span className="v" id="v4" data-v="4" data-xref="زَكَريّا 8:12; المَزامِير 67:6; اللّاويِّين 25:21"> أنزَلتُ المطَرَ علَيكُم في حينِهِ، فتُخرِجُ الأرضُ غِلالَها وشجَرُ الحَقلِ ثَمرَهُ</span>
<span className="v" id="v5" data-v="5" data-xref="عامُوس 9:13; يُوئيل 2:26; يُوئيل 2:19"> ويتَّصلُ أوانُ دَرْسِ الحِنطةِ بقطافِ العِنَبِ، وقطافُ العِنَبِ بالزَّرعِ، وتأكلونَ طَعامَكُم وتشبعونَ وتُقيمونَ في أرضِكُم آمِنينَ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="إشَعْياء 35:9; صَفَنْيا 3:13; المَزامِير 147:14"> ”وأُلقي السَّلامَ في الأرضِ، فتنامونَ ولا مَنْ يُرعِبُكُم وأُزيلُ الوحوشَ الضَّاريَةَ مِنَ الأرضِ، وحربٌ لا تَعبرُ أرضَكُم.</span>
<span className="v" id="v7" data-v="7"> وتطردونَ أعداءَكُم، فيَسقطونَ أمامَكُم بالسَّيفِ.</span>
<span className="v" id="v8" data-v="8" data-xref="يَشُوع 23:10; التَّثْنِيَة 28:7; التَّثْنِيَة 32:30"> تطردُ الخمسَةُ مِنكُم مئةً، والمئةُ عشَرةَ آلافٍ، ويسقطُ أعداؤُكُم أمامَكُم بالسَّيفِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="التَّكوين 17:6–7; نَحَمْيا 9:23; حِزْقِيال 16:62"> ”وألتَفِتُ إليكُم وأُنمِّيكُم وأُكثِّرُكُم وأُقيمُ عَهدي لكُم.</span>
<span className="v" id="v10" data-v="10" data-xref="اللّاويِّين 25:22; المُلُوكِ الثّانِي 19:29; لُوقا 12:17"> وتأكلونَ مِنْ قديمِ غِلالِكُم إلى أنْ تُخرِجوه لِتَخزُنوا الجديدَ.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 25:8; رُؤيا يُوحَنّا 21:3; الخُرُوج 29:45"> وأجعَلُ مَسكِني فيما بَينَكُم ولا أنبِذُكُم.</span>
<span className="v" id="v12" data-v="12" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:16; إرْمِيا 30:22; إرْمِيا 7:23"> وأسيرُ فيما بَينَكُم، وأكونُ لكُم إلهًا، وأنتُم تكونونَ لي شعبًا.</span>
<span className="v" id="v13" data-v="13" data-xref="حِزْقِيال 34:27; اللّاويِّين 25:38; المَزامِير 116:16"> أنا الرّبُّ إلهُكُم الّذي أخرجَكُم مِنْ أرضِ المِصْريِّينَ لِئلَّا تكونوا عبيدًا لهُم، وحطَّمَ أغلالَ نيرِكُم وجعَلَكُم تسيرونَ مَرفُوعي الرَّأسِ.</span>
</div>
</div>

## جزاء العصيان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="التَّثْنِيَة 28:15–68; اللّاويِّين 26:18; إرْمِيا 17:27"> ”وإنْ كنتُم لا تسمَعونَ لي ولا تعمَلونَ بجميعِ هذِهِ الوصايا،</span>
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الثّانِي 17:15; التَّثْنِيَة 31:16; زَكَريّا 7:11–13"> ورفَضْتُم فَرائضي، ونبَذْتُم أحكامي، ونقَضْتُم عَهدي ولم تعمَلوا بجميعِ وصايايَ،</span>
<span className="v" id="v16" data-v="16" data-xref="أيُّوب 31:8; إرْمِيا 5:17; ميخا 6:15"> فأنا أيضًا أعامِلُكُم بالمِثْلِ، فأجلبُ علَيكُم رُعبًا، وداءً عُضَالًا، وحُمَّى تُفني العينينِ وتُتلِفُ النَّفْسَ، وتزرعونَ زرعَكُم باطلًا فيأكلُهُ أعداؤُكُم.</span>
<span className="v" id="v17" data-v="17" data-xref="اللّاويِّين 17:10; الأمثال 28:1; المَزامِير 53:5"> وأميلُ بوجهي علَيكُم، فتَنهزِمونَ مِنْ أمامِ أعدائِكُم، ويتسلَّطُ علَيكُم مُبغضوكُم، وتهربونَ ولا مَنْ يطردُكُم.</span>
<span className="v" id="v18" data-v="18" data-xref="اللّاويِّين 26:21; اللّاويِّين 26:28; اللّاويِّين 26:24"> وإنْ كنتُم بَعدَ هذا كُلِّه لا تسمعونَ لي، زِدتُكُم تأديبًا على خطاياكُم سَبعةَ أضعافٍ،</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 28:23; المُلُوكِ الأوَّلُ 17:1; دانيال 4:37"> فأُحطِّمُ جاهَكُم وعِزَّكُم، وأجعَلُ سَماءَكُم كالحديدِ وأرضَكُم كالنُّحاسِ</span>
<span className="v" id="v20" data-v="20" data-xref="المَزامِير 127:1; التَّثْنِيَة 11:17; إشَعْياء 49:4"> وتستَنفِدونَ قِواكُم عَبَثًا، ولا تُعطي أرضُكُم غَلَّتَها، ولا يُخرِجُ شجَرُ الأرضِ ثَمرَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="اللّاويِّين 26:18; اللّاويِّين 26:27"> ”وإنْ تَمادَيتُم في مُخالفتي وما سَمِعْتُم لي، زِدْتُكم سَبعةَ أضعافٍ مِنَ الضَّرَباتِ على خطاياكُم</span>
<span className="v" id="v22" data-v="22" data-xref="حِزْقِيال 14:15; التَّثْنِيَة 32:24; القُضاة 5:6"> وأطلقتُ علَيكُم وحُوشَ البريَّةِ، فَتُفقِدُكُم أولادَكُم وتُهلِكُ بهائِمَكُم وتُقلِّلُ عددَكُم حتّى تُقفِرَ طُرقاتُكُم.</span>
</div>
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="إرْمِيا 5:3; إرْمِيا 2:30; عامُوس 4:6–12"> ”وإنْ كانَ هذا لا يُؤدِّبُكُم وتَمادَيتُم في مُخالفتي</span>
<span className="v" id="v24" data-v="24" data-xref="المَزامِير 18:26; صموئيلَ الثّانِي 22:27; أيُّوب 9:4"> تَمادَيتُ أنا أيضًا في مُخالفتِكُم وضَرَبْتُكُم سَبعةَ أضعافٍ على خطاياكُم</span>
<span className="v" id="v25" data-v="25" data-xref="العَدَد 14:12; حِزْقِيال 5:17; إرْمِيا 29:17–18"> فأجلبُ علَيكُم حربًا تَنتَقِمُ للعَهدِ الّذي بَيني وبَينَكُم، فتَتَجَمَّعونَ إلى مُدنِكُم هاربينَ وأُرسِلُ الوباءَ فيما بَينَكُم وتُسَلَّمُونَ إلى أيدي أعدائِكُم.</span>
<span className="v" id="v26" data-v="26" data-xref="ميخا 6:14; إشَعْياء 3:1; المَزامِير 105:16"> وأقطعُ عَنكُم كفايتَكُم مِنَ الخُبزِ، وهوَ قِوامُ حياتِكُم، فتَخبُزُ عَشْرُ نِساءٍ الخُبزَ في تنُّورٍ واحدٍ ويُوزِّعْنَه بالتَّقتيرِ، وتأكلونَ ولا تشبعونَ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 26:24; اللّاويِّين 26:21"> ”وإنْ كنتُم بهذا كُلِّه لا تسمعونَ لي وتَماديتُم في مُخالفتي،</span>
<span className="v" id="v28" data-v="28" data-xref="إشَعْياء 59:18; إرْمِيا 21:5; إشَعْياء 66:15"> تَمادَيتُ أنا أيضًا في مُخالفتِكُم ساخِطًا وأدَّبتُكُم سَبعةَ أضعافٍ على خطاياكُم،</span>
<span className="v" id="v29" data-v="29" data-xref="حِزْقِيال 5:10; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:10; ﻣﺘﻰ 24:19"> فتأكلونَ لُحومَ بَنيكم وبَناتِكُم،</span>
<span className="v" id="v30" data-v="30" data-xref="المُلُوكِ الثّانِي 23:20; إشَعْياء 27:9; حِزْقِيال 6:13"> وأهدِمُ مَعابِدَكُمُ العاليةَ، وأُحطِّمُ مذابحَ بَخورِكُم، وأُلقي جُثَثَكُم على جُثَثِ أصنامِكُم وتَعافُكُم نفْسي.</span>
<span className="v" id="v31" data-v="31" data-xref="نَحَمْيا 2:3; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:10; العِبرانِيّين 10:26"> وأجعَلُ مُدُنَكُم خَرابًا وأماكِنَكُمُ المُقدَّسةَ قَفرًا، ولا أشتَمُّ رائحةَ مُحرَقاتِكُم لمُراضاتي.</span>
<span className="v" id="v32" data-v="32" data-xref="إرْمِيا 9:11; إرْمِيا 19:8; إرْمِيا 18:16"> وأتركُ الأرضَ قفرًا، فينذَهِلُ لها أعداؤُكُمُ المُقيمونَ فيها.</span>
<span className="v" id="v33" data-v="33" data-xref="زَكَريّا 7:14; حِزْقِيال 20:23; التَّثْنِيَة 28:64–66"> وأُشتِّتُكُم فيما بَينَ الأُمَمِ، وأشهَرُ وراءَكُم سيفًا، فتصيرُ أرضُكُم قَفرًا ومُدنُكُم خَرابًا.</span>
<span className="v" id="v34" data-v="34" data-xref="أخبارِ الأيّامِ الثّانِي 36:21; اللّاويِّين 26:43; اللّاويِّين 25:2–4"> فتنعمُ الأرضُ براحتِها طُولَ أيّامِ خُلُوِّها مِنكُم، وأنتُم في أرضِ أعدائِكُم. نعم، ترتاحُ الأرضُ وتنعمُ براحتِها</span>
<span className="v" id="v35" data-v="35" data-xref="إشَعْياء 24:5–6; ﺭﻭﻣﻴﺔ 8:22"> طُولَ أيّامِ خُلُوِّها مِنكُم. فهيَ قَلَّما اَرتاحت في أيّامِ سُبوتِها وأنتُم مُقيمونَ بِها.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="حِزْقِيال 21:7; إشَعْياء 30:17; اللّاويِّين 26:17"> ”والباقونَ مِنكُم أُلقي الجَبانةَ في قُلوبِهِم، في أراضي أعدائِهِم، حتّى إنَّ صوتَ ورقةٍ مُتحرِّكةٍ يَهزمُهُم، فيَهربونَ هرَبَهُم مِنَ السَّيفِ ولا مَنْ يطردُهُم.</span>
<span className="v" id="v37" data-v="37" data-xref="يَشُوع 7:12–13; القُضاة 2:14; إرْمِيا 37:10"> ويَعثُرُ الواحدُ مِنهمُ بالآخَرِ، كمَنْ يهرُبُ مِنْ أمامِ السَّيفِ ولا مَنْ يطردُهُ، ولا يقومُ لكم قائمةٌ في وجوهِ أعدائِكُم.</span>
<span className="v" id="v38" data-v="38" data-xref="التَّثْنِيَة 28:48; إشَعْياء 27:13; إرْمِيا 42:17–18"> وتَبيدونَ بَينَ الأممِ وتبتلِعُكُم أرضُ أعدائِكُم.</span>
<span className="v" id="v39" data-v="39" data-xref="حِزْقِيال 4:17; حِزْقِيال 24:23; حِزْقِيال 33:10"> والباقونَ مِنكُم يسيرونَ إلى الزَّوالِ عِقابًا على آثامِهِم وآثامِ آبائِهِم في أراضي أعدائِكُم،</span>
<span className="v" id="v40" data-v="40" data-xref="الأمثال 28:13; نَحَمْيا 9:2–5; رسالَةُ يُوحَنّا الأُولَى 1:8–10"> حتّى إذا اَعترفوا بآثامِهِم وآثامِ آبائِهِم، وبخيانتِهِم لي وتَماديهِم في مُخالفتي،</span>
<span className="v" id="v41" data-v="41" data-xref="حِزْقِيال 44:7; أخبارِ الأيّامِ الثّانِي 32:26; أخبارِ الأيّامِ الثّانِي 12:6–7"> بحيثُ تَماديتُ أنا أيضًا في مُخالفتِهِم، فأدخلتُهُم أرضَ أعدائِهِم حتّى تخضعَ قلوبُهُمُ الكافرةُ ويتحَمَّلوا جَزاءَ آثامِهِم</span>
<span className="v" id="v42" data-v="42" data-xref="المَزامِير 106:45; الخُرُوج 6:5; حِزْقِيال 16:60"> ذكَرْتُ عَهدي معَ يَعقوبَ وإسحَقَ وإبراهيمَ وذكَرْتُ الأرضَ</span>
<span className="v" id="v43" data-v="43" data-xref="ﺭﻭﻣﻴﺔ 8:7; اللّاويِّين 26:15; يُوحَنّا 7:7"> الّتي أخليتُها مِنهُم لتنعَمَ براحتِها في خُلوِّها مِنهُم، وليَتَحَمَّلوا جَزاءَ آثامِهِم لأنَّهُم رفَضُوا أحكامي ونبذوا فرائضي.</span>
</div>
<div className="bp">
<span className="v" id="v44" data-v="44" data-xref="ﺭﻭﻣﻴﺔ 11:2; نَحَمْيا 9:31; التَّثْنِيَة 4:29–31"> ”وإلى ذلِكَ الحينِ لا أرفُضُهُم كُلَّ الرَّفْضِ وهُم في أرضِ أعدائِهِم، ولا أنبذُهُم بحيثُ أَفنيهِم وأنقضُ عَهدي معَهُم، فأنا الرّبُّ إلهُهُم.</span>
<span className="v" id="v45" data-v="45" data-xref="اللّاويِّين 25:38; حِزْقِيال 20:22; حِزْقِيال 20:9"> بل أذكرُ عَهدي للأوَّلينَ الّذينَ أخرَجْتُهُم مِنْ أرضِ مِصْرَ، على مشهدٍ مِنَ الأُمَمِ، لأكونَ لهُم إلهًا. أنا الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v46" data-v="46" data-xref="اللّاويِّين 27:34; التَّثْنِيَة 12:1; التَّثْنِيَة 6:1"> تلكَ هيَ الفرائِضُ والأحكامُ والشَّرائعُ الّتي وضعَها الرّبُّ لبَني إِسرائيلَ، في جبَلِ سيناءَ، على يدِ موسى.</span>
</div>
