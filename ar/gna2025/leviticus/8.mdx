## شريعة التكريس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ الرّبُّ لِموسى:</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 28:2–4; الخُرُوج 39:41; العِبرانِيّين 7:27"> ”خُذْ هرونَ وبَنيهِ معَهُ، والثِّيابَ وزيتَ المَسْحِ وعِجْلَ ذبيحةِ الخطيئةِ والكَبشَينِ وسَلَّةَ الفطيرِ،</span>
<span className="v" id="v3" data-v="3" data-xref="أخبارِ الأيّامِ الثّانِي 30:13; اعمال الرسل 2:1; نَحَمْيا 8:1"> واَجْمَعْ بَني إِسرائيلَ كُلَّهم إلى بابِ خَيمَةِ الاجتِماعِ“.</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 39:26; الخُرُوج 39:7; الخُرُوج 39:5"> فعَمِلَ موسى كما أمرَ الرّبُّ. ولمّا اَجْتمَعتِ الجماعةُ إلى بابِ الخَيمةِ،</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 29:4–37"> قالَ لهُم موسى: ”هذا ما أمرَني الرّبُّ أنْ أعمَلَهُ“،</span>
<span className="v" id="v6" data-v="6" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:11; أفَسُس 5:26; الخُرُوج 29:4"> وقدَّمَ هرونَ وبَنيه وغسَلَهُم بالماءِ،</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 28:4; إشَعْياء 61:10; غَلاطِيَّة 3:27"> ثُمَّ ألبَسَ هرونَ القميصَ وشَدَّهُ بالحِزامِ، ثُمَّ ألبَسَه الجُبَّةَ والأفودَ وشدَّ الأفودَ بِزُنَّارٍ،</span>
<span className="v" id="v8" data-v="8" data-xref="عَزْرا 2:63; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:8; إشَعْياء 59:17"> ووَضعَ علَيهِ الصُّدْرةَ وفيها الأُوريمُوالتُّمّيمُ.</span>
<span className="v" id="v9" data-v="9" data-xref="فِيلِبِّي 2:9–11; زَكَريّا 6:11–14; الخُرُوج 28:36–38"> وَضعَ العِمامةَ على رأسِ هرونَ وعلَيها مِنَ الأمامِ صَفيحةُ الذَّهبِ، وهوَ التَّاجُ المُقَدَّسُ، كما أمرَ الرّبُّ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 40:9–11; اللّاويِّين 8:2; الخُرُوج 30:23–29"> وأخذَ موسى زيتَ المَسْحِ ومسَحَ المسكِنَ وجميعَ ما فيهِ وقَدَّسَهُ.</span>
<span className="v" id="v11" data-v="11" data-xref="تِيطُس 3:6; الخُرُوج 30:29; الخُرُوج 16:19"> ورَشَّ مِنهُ على المذبَحِ سَبْعَ مرَّاتٍ، ومسَحَ المذبَحَ وجميعَ أدواتِهِ والمِغسَلَةِ ومقعَدَها وقدَّسَها.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 30:30; المَزامِير 133:2; الخُرُوج 29:7"> وصبَّ مِنْ زيتِ المَسْحِ على رأسِ هرونَ ومسَحَهُ وقدَّسَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الخُرُوج 29:8–9; رُؤيا يُوحَنّا 5:10; رُؤيا يُوحَنّا 1:6"> ثُمَّ قدَّمَ موسى بَني هرونَ وألبَسَهُم أقمِصَةً، وشَدَّهم بأحزِمةٍ، وعصَبَ لهُم قلانِسَ، كما أمرَ الرّبُّ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="حِزْقِيال 43:19; المَزامِير 66:15; رسالَةُ بُطرُسَ الأُولَى 3:18"> وقدَّمَ موسى عِجْلَ ذبيحةِ الخطيئةِ، فوضَعَ هرونُ وبَنوهُ أيديَهُم على رأسِهِ.</span>
<span className="v" id="v15" data-v="15" data-xref="اللّاويِّين 4:7; العِبرانِيّين 9:18–23; حِزْقِيال 43:19–27"> وذبَحَهُ موسى وأخذَ الدَّمَ وجعَلَهُ على قُرونِ المذبَحِ مِنْ كُلِّ جِهةٍ بإصبَعِهِ، فطَهَّرهُ وصَبَّ الدَّمَ عِندَ قاعِدَتِهِ وقَدَّسَهُ تكفيرًا عَنه.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 29:13; اللّاويِّين 3:3–5; اللّاويِّين 4:8–9"> وأخذَ موسى جميعَ الشَّحْمِ الّذي على الأمعاءِ وزياداتِ الكَبِدِ والكليَتَينِ وشَحْمَهُما وأوقدَ ذلِكَ على المذبَحِ.</span>
<span className="v" id="v17" data-v="17" data-xref="اللّاويِّين 4:11–12; الخُرُوج 29:14; اللّاويِّين 6:30"> وأحرقَ جِلْدَ العِجْلِ ولَحمَهُ ورَوثَه بالنَّارِ في خارجِ المحَلَّةِ كما أمرَ الرّبُّ.</span>
<span className="v" id="v18" data-v="18" data-xref="اللّاويِّين 8:2; اللّاويِّين 1:4–13; الخُرُوج 29:15–18"> ثُمَّ قدَّمَ كَبْشَ المُحرَقةِ، فوضعَ هرونُ وبَنوهُ أيديَهُم على رأسِهِ.</span>
<span className="v" id="v19" data-v="19"> وذبَحَهُ موسى ورشَّ الدَّمَ على المذبَحِ مِنْ كُلِّ جِهةٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="اللّاويِّين 1:8"> وقطَّعَ موسى الكَبْشَ قِطَعًا وأوقدَ الرَّأسَ والقِطَعَ والشَّحْمَ.</span>
<span className="v" id="v21" data-v="21" data-xref="أفَسُس 5:2; الخُرُوج 29:18; اللّاويِّين 1:17"> وغسَلَ الأمعاءَ والأكارِعَ بالماءِ وأوقدَ الكَبْشَ كُلَّهُ على المذبَحِ. هوَ مُحرَقةٌ وقيدةٌ تُرضي رائحَتُها الرّبَّ، كما أمرَ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="اللّاويِّين 8:2; اللّاويِّين 8:29; الخُرُوج 29:19–31"> ثُمَّ قدَّمَ الكَبْشَ الثَّاني، كَبْشَ التَّكريسِ، فوضعَ هرونُ وبَنوهُ أيديَهُم على رأسِهِ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 29:20; اللّاويِّين 14:14; ﺭﻭﻣﻴﺔ 12:1"> وذبحَهُ موسى وأخذَ مِنْ دَمِهِ ووضعَ على شَحْمةِ أذُنِ هرونَ اليُمنى، وعلى إبهامِ يدِهِ اليُمنى، وعلى إبهام رجلِهِ اليُمنى.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="العِبرانِيّين 9:18–22"> ثُمَّ قَدَّمَ بني هرونَ ووَضعَ مِنَ الدَّمِ على شَحْماتِ آذانِهِمِ اليُمنى، وعلى أباهمِ أيديهِمِ اليُمنى وأباهمِ أرجُلِهِمِ اليُمنى، ورَشَّ الدَّمَ على المذبَحِ مِنْ كُلِّ جِهةٍ.</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 3:9; اللّاويِّين 3:3–5; إشَعْياء 53:10"> وأخذَ الشَّحْمَ والألْيةَ، وجميعَ الشَّحْمِ الّذي على الأمعاءِ، وزياداتِ الكَبِدِ، والكليتَينِ وشَحْمَهُما، والكَتِفَ اليُمنى.</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 29:23; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 2:5; اعمال الرسل 5:12"> وأخذَ مِنْ سلَّةِ الفطيرِ الّتي أمامَ الرّبِّ قُرصَ فطيرٍ وقرصَ خبزٍ بزيتٍ ورُقاقةً ووضعَها على الشُّحومِ والكَتِفِ اليمنى.</span>
<span className="v" id="v27" data-v="27" data-xref="إرْمِيا 30:21; العِبرانِيّين 9:14; الخُرُوج 29:24–37"> ووضعَ موسى هذا كُلَّه على راحتَي هرونَ، وعلى راحاتِ بَنيه، وحرَّكَه تَحْريكًا أمامَ الرّبِّ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 29:25; المَزامِير 22:13–14; العِبرانِيّين 10:14–22"> ثُمَّ أخذَهُ عَنْ راحاتِهِم وأوقدَهُ على المذبَحِ فوقَ المُحرَقةِ. هوَ قُربانُ تكريسٍ ووقيدةٌ تُرضي رائحَتُها الرّبَّ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:31; رسالَةُ بُطرُسَ الأُولَى 4:11; اللّاويِّين 7:30–34"> ثُمَّ أخذَ موسى صَدْرَ الذَّبيحةِ وحرَّكَهُ تَحْريكًا أمامَ الرّبِّ، وكانَ هذا حِصَّةَ موسى مِنْ كَبْشِ التَّكريسِ، كما أمرَ الرّبُّ موسى.</span>
<span className="v" id="v30" data-v="30" data-xref="الخُرُوج 29:21; العَدَد 3:3; الخُرُوج 30:30"> وأخذَ موسى مِنْ زيتِ المَسْحِ، ومِنَ الدَّمِ الّذي على المذبَحِ، فرَشَّ على هرونَ وثيابِهِ، وعلى بَنيهِ وثيابِهِم، وقَدَّسَ هرونَ وثيابَهُ وبَنيهِ وثيابَهُم.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 29:31–32; غَلاطِيَّة 2:20; صَمُوئيلَ الأوَّلُ 2:13–17"> وقالَ موسى لِهرونَ وبَنيهِ: ”أُطبُخوا اللَّحمَ عِندَ بابِ خَيمةِ الاجتماعِ وكُلُوهُ هُناكَ معَ الخُبزِ الّذي في سلَّةِ التَّكريسِ، كما أمرَ الرّبُّ وقالَ ”هرونُ وبَنوه يأكلونَهُ“،</span>
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 29:34; الخُرُوج 12:10; العِبرانِيّين 3:13–14"> وما فضَلَ مِنَ اللَّحمِ والخُبزِ، فاَحرِقوه بالنَّارِ.</span>
<span className="v" id="v33" data-v="33" data-xref="الخُرُوج 29:35; العَدَد 19:12; حِزْقِيال 43:25–27"> ولا تخرُجوا مِنْ عِندِ بابِ خَيمةِ الاجتِماعِ حتى تمضيَ سَبعةُ أيّامٍ ويكونَ مِنْ بَعدِها تكريسُكُم.</span>
<span className="v" id="v34" data-v="34" data-xref="العِبرانِيّين 7:27; العِبرانِيّين 7:16; العِبرانِيّين 10:11–12"> فأعملُ بِكُم مِثلما عَمِلْتُ اليومَ تكفيرًا عنكُم، كما أمرَ الرّبُّ.</span>
<span className="v" id="v35" data-v="35" data-xref="العَدَد 3:7; التَّثْنِيَة 11:1; المُلُوكِ الأوَّلُ 2:3"> فأقيمُوا عِندَ بابِ خَيمةِ الاجتِماعِ نهارًا وليلًا سَبعةَ أيّامٍ لِتَحرُسوا شعائِرَ الرّبِّ فلا تموتُونَ. هكذا أمرَ الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="التَّثْنِيَة 12:32; التَّثْنِيَة 4:2; الخُرُوج 40:16"> فعَمِل هرونُ وبَنوهُ بجميعِ ما أمرَ بهِ الرّبُّ على لِسانِ موسى.</span>
</div>
