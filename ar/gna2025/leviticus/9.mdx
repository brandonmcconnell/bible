## هرون يقرب الذبائح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="ﻣﺘﻰ 28:1; حِزْقِيال 43:26–27; اللّاويِّين 15:14"> ودَعا موسى في اليومِ الثَّامنِ هرونَ وبَنيهِ وشُيوخَ بَني إِسرائيلَ،</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 4:3; العِبرانِيّين 10:10–14; اللّاويِّين 8:14"> وقالَ لِهرونَ: ”خُذْ عِجْلًا مِنَ البقَرِ لِذبيحةِ الخطيئةِ، وكَبْشًا لَمُحرَقةٍ صحيحَينِ وقدِّمْهُما قربانًا أمامَ الرّبِّ.</span>
<span className="v" id="v3" data-v="3" data-xref="عَزْرا 6:17; رُؤيا يُوحَنّا 5:9; تِيطُس 2:14"> وقُلْ لِبَني إِسرائيلَ: خُذُوا تَيسًا مِنَ المعَزِ لذبيحةِ خطيئةٍ، وعِجْلًا وخروفًا حَوليَّينِ صحيحَينِ لِمُحرَقةٍ،</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 29:43; حِزْقِيال 43:2; اللّاويِّين 9:23"> وثورًا وكَبْشًا لِذبيحةِ سلامةٍ أمامَ الرّبِّ، وتقدِمةً مَلتُوتَةً بِزيتٍ، فالرّبُّ اليومَ يتراءَى لكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الثّانِي 5:2–3; الخُرُوج 19:17; نَحَمْيا 8:1"> فأخذَ بَنو إِسرائيلَ ما أمرَ موسى إلى بابِ خَيمةِ الاجتِماعِ واَقتَربُوا جميعًا ووَقَفُوا أمامَ الرّبِّ.</span>
<span className="v" id="v6" data-v="6" data-xref="اللّاويِّين 9:23; حِزْقِيال 43:2; الخُرُوج 24:16"> فقالَ موسى: ”هذا ما أمرَ بهِ الرّبُّ، تعمَلُونَهُ فيَتَراءى لكم مَجدُهُ“.</span>
<span className="v" id="v7" data-v="7" data-xref="العِبرانِيّين 5:3; العِبرانِيّين 5:1; اللّاويِّين 4:3"> ثُمَّ قالَ لِهرونَ: ”إقْتَرِبْ إلى المذبَحِ واَعْمَلْ للرّبِّ ذبيحةَ تكفيرٍ عَنْ خطيئَتِكَ ومُحرَقةً تُكفِّرُ بِها عَنْ نَفسِكَ وعَنِ الشَّعبِ. وقرِّبْ إلى الرّبِّ ما يُكَفِّرُ عَنِ الشَّعبِ بِحسَبِ ما أمرَ الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 4:1–12; اللّاويِّين 1:4–5; اللّاويِّين 4:29"> فاَقْتَرَبَ هرونُ إلى المذبَحِ وذبَحَ عِجْلَ ذبيحةِ الخطيئَةِ الّذي لهُ.</span>
<span className="v" id="v9" data-v="9" data-xref="اللّاويِّين 4:6–7; اللّاويِّين 8:15; اللّاويِّين 4:25"> وقَدَّمَ لَه بَنو هرونَ الدَّمَ، فغَمسَ إصبعَهُ في الدَّمِ وجعَلَ على قُرونِ المذبَحِ، وصَبَّ الدَّمَ على قاعِدتِهِ.</span>
<span className="v" id="v10" data-v="10" data-xref="اللّاويِّين 8:16–17; الأمثال 23:26; اللّاويِّين 3:9–11"> وأوقدَ الشَّحْمَ والكليَتَينِ وزيادةَ الكَبِدِ على المذبَحِ، كما أمرَ الرّبُّ موسى.</span>
<span className="v" id="v11" data-v="11" data-xref="اللّاويِّين 8:17; اللّاويِّين 4:11–12; اللّاويِّين 16:27–28"> وأمَّا اللَّحمُ والجِلْدُ فأحرَقَهُما بِنارٍ في خارجِ المحَلَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="أفَسُس 5:2; أفَسُس 5:25–27; اللّاويِّين 1:1–17"> ثُمَّ ذبَحَ المُحرَقةَ، وناولَهُ بَنو هرونَ الدَّمَ، فرَشَّهُ على أربعَةِ جوانبِ المذبَحِ.</span>
<span className="v" id="v13" data-v="13"> وناولُوهُ المُحرَقةَ بِقِطَعِها والرَّأسَ، فأوقدَها على المذبَحِ.</span>
<span className="v" id="v14" data-v="14" data-xref="اللّاويِّين 8:21"> وغسَلَ الأمعاءَ والأكارِعَ وأوقدَها فوقَ المُحرقَةِ على المذبَحِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="اللّاويِّين 4:27–31; العِبرانِيّين 5:3; اللّاويِّين 9:3"> وقدَّمَ هرونُ قُربانَ الشَّعبِ، فأخذَ تَيْسَ ذبيحةِ الخطيئةِ الّذي للشَّعبِ وذبَحَهُ وقدَّمَهُ ذبيحةَ خطيئةٍ كالعِجْلِ الأوَّلِ.</span>
<span className="v" id="v16" data-v="16" data-xref="اللّاويِّين 9:12–14; اللّاويِّين 8:18–21; العِبرانِيّين 10:1–22"> ثُمَّ قرَّبَ المُحرَقةَ، كما جرَتِ العادةُ.</span>
<span className="v" id="v17" data-v="17" data-xref="اللّاويِّين 2:1–2; يُوحَنّا 6:53; اللّاويِّين 9:1"> وقرَّبَ التَّقدِمةَ، فمَلأَ قبضَتَهُ مِنها وأوقدَها على المذبَحِ معَ مُحرَقةِ الصَّباحِ.</span>
<span className="v" id="v18" data-v="18" data-xref="كُولُوسِي 1:20; ﺭﻭﻣﻴﺔ 5:1; أفَسُس 2:14–17"> وذبَحَ الثَّورَ والكَبْشَ ذبيحةَ سلامةٍ للشَّعبِ، وناولَهُ بَنو هرونَ الدَّمَ، فرَشَّهُ على أربعَةِ جَوانبِ المذبَحِ.</span>
<span className="v" id="v19" data-v="19" data-xref="اللّاويِّين 3:16; اللّاويِّين 9:10; اللّاويِّين 3:9"> وناولُوهُ الشَّحْمَ مِنَ الثَّورِ ومِنَ الكَبْشِ، معَ ألْيتِهِ وما يُغطِّي الأمعاءَ والكليتَينِ والزَّائِدَ مِنَ الكَبِدِ.</span>
<span className="v" id="v20" data-v="20" data-xref="اللّاويِّين 3:14–17; اللّاويِّين 7:29–34"> ووضعوا الشَّحْمَ على الصَّدْرَينِ، فأوقدَ هرونُ الشَّحْمَ على المذبَحِ.</span>
<span className="v" id="v21" data-v="21" data-xref="اللّاويِّين 7:30–34; اللّاويِّين 7:24; الخُرُوج 29:24"> وأمَّا الصَّدْرانَ والسَّاقُ اليُمنى، فحَرَّكَها هرونُ تَحْريكًا أمامَ الرّبِّ، كما قالَ لَه موسى.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 21:5; لُوقا 24:50; العَدَد 6:23–27"> ولمَّا أكملَ هرونُ تقريبَ ذبيحةِ الخطيئةِ، والمُحرَقةِ، وذبيحةِ السَّلامةِ، رفَعَ يَدهُ نحوَ الشَّعبِ وبارَكَهُم ونزَلَ عَنِ المذبَحِ.</span>
<span className="v" id="v23" data-v="23" data-xref="اللّاويِّين 9:6; العَدَد 16:19; لُوقا 1:21–22"> ودخَلَ موسى وهرونُ إلى خَيمةِ الاجتِماعِ، ثُمَّ خرجا وباركا الشَّعبَ. فتَراءى مَجْدُ الرّبِّ لِكُلِّ الشَّعبِ.</span>
<span className="v" id="v24" data-v="24" data-xref="القُضاة 6:21; المُلُوكِ الأوَّلُ 18:38–39; أخبارِ الأيّامِ الأوّلُ 21:26"> وخرَجَتْ نارٌ مِنْ عِندِ الرّبِّ وأحرقَتِ المُحرَقةَ والشَّحْمَ على المذبَحِ. فلمَّا رأى الشَّعبُ ذلِكَ هتَفوا ووقعوا على وجُوهِهِم ساجِدينَ.</span>
</div>
