## الرب يملك على صهيون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="إشَعْياء 2:1–4; إرْمِيا 3:17; اعمال الرسل 2:17"> وفي الآتي مِنَ الأيّامِ<br />يكونُ جبَلُ بَيتِ الرّبِّ<br />في رأسِ كُلِّ الجبالِ<br />يرتفعُ فَوقَ التِّلالِ<br />وتجري إليهِ الشُّعوبُ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="إشَعْياء 2:3; إرْمِيا 31:6; اعمال الرسل 1:8"> يقولُ أمَمٌ كثيرونَ:<br />لنصعَدْ إلى جبَلِ الرّبِّ،<br />إلى بَيتِ إلهِ يَعقوبَ،<br />حَيثُ الرّبُّ يُرينا طُرُقَهَ<br />فنسلُكَ فيها جميعًا.<br />فَمِنْ صِهيَونَ تخرُجُ الشَّريعةُ،<br />مِنْ أُورُشليمَ كَلِمةُ الرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 2:4; يُوئيل 3:9–16; ﻣﺘﻰ 25:31–32"> يحكُمُ بَينَ شعوبٍ كثيرينَ<br />ويؤدِّبُ الأمَمَ الأقوياءَ،<br />البعيدينَ منهُم والقريبينَ،<br />فيَضربونَ سيوفَهُم سِكَكًا<br />ورِماحَهُم مناجِلَ،<br />فلا ترفعُ أمَّةٌ على أمَّةٍ سيفًا<br />ولا يتَعَلَّمونَ الحربَ مِنْ بَعدُ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="زَكَريّا 3:10; المُلُوكِ الأوَّلُ 4:25; حِزْقِيال 39:26"> ويُقيمُ كلُّ واحدٍ تَحتَ كَرمتِهِ<br />وتَحتَ تينتِهِ ولا مَنْ يُرعِبُهُ.<br />فَمُ الرّبِّ القديرِ تكلَّمَ.</span>
</div>
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="زَكَريّا 10:12; يَشُوع 24:15; كُولُوسِي 3:17"> ويسيرُ جميعُ الشُّعوبِ،<br />كُلُّ واحدٍ باَسمِ إلهِهِ،<br />ونسيرُ نحنُ باَسمِ الرّبِّ<br />إلهِنا إلى آخرِ الدَّهرِ.</span>
</div>
</div>

## العودة من السبي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="صَفَنْيا 3:19; حِزْقِيال 34:12–17; إشَعْياء 35:3–6"> في ذلِكَ اليومِ يقولُ الرّبُّ،<br />أجمعُ البعيدينَ مِنَ السبْي<br />وألمُّ المُشرَّدينَ مِنْ شعبي<br />وأرعى الباقينَ مِنهُم.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="دانيال 7:14; إشَعْياء 24:23; ميخا 2:12"> أجعَلُ العُرْجَ يَمشونَ<br />والجياعَ أمَّةً عظيمةً<br />فيملِكُ الرّبُّ في جبَلِ صِهيَونَ<br />مِنَ الآنَ وإلى الأبدِ.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 1:26; رُؤيا يُوحَنّا 22:5; زَكَريّا 9:10"> وأنتَ يا بُرجَ القطيعِ<br />يا جبَلَ بِنتِ صِهيَونَ،<br />إليكَ يأتي الحُكْمُ<br />ويَعودُ المُلكُ كما مِنْ قَبلُ<br />إلى مدينةِ أُورُشليمَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="إرْمِيا 8:19; إشَعْياء 3:1–7; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 4:20"> والآنَ لماذا تصرُخينَ<br />وتتوجَّعينَ كالّتي تَلِدُ؟<br />أمَا فيكِ مَنْ يَملِكُ؟<br />أبادَ عَنكِ المُشيرُ؟</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إشَعْياء 48:20; إشَعْياء 52:9–12; المُلُوكِ الثّانِي 20:18"> تلَوّي يا بِنتَ صِهيَونَ<br />وتوجَّعي كالّتي تلِدُ،<br />فالآنَ تخرُجينَ مِنَ المدينةِ،<br />وفي البرِّيَّةِ تسكُنينَ،<br />لتذهبي إلى بابلَ مِنْ بَعدُ.<br />هُناكَ يُنقِذُكِ الرّبُّ،<br />ومِنْ أيدي الأعداءِ يَفتديكِ.</span>
</div>
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="عُوبَدْيا 1:12; إشَعْياء 5:25–30; ميخا 7:10"> أُمَمٌ كثيرونَ اَجتمعوا علَيكِ<br />وهُم يقولونَ طامِعينَ:<br />”لِنَلتَفِتْ إلى صِهيَونَ،<br />وَلنَرفَعْ عُيونَنا إليها“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="إشَعْياء 55:8; إرْمِيا 29:11; ﺭﻭﻣﻴﺔ 11:33–34"> هُم لا يعرفونَ ما أعدَّهُ الرّبُّ<br />ولا يتَبَيَّنُونَ مَشورتَهُ،<br />فيَحزِمُهُم جميعًا إلى البيادِرِ<br />كما تُحزَمُ أكداسُ الحنطةِ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 41:15–16; دانيال 2:44; زَكَريّا 9:13–15"> فدُوسي علَيهِم يا بنتَ صِهيَونَ<br />كثورٍ قرناهُ مِنْ حديدٍ،<br />ومِنْ نحاسٍ جُعِلَتْ أظلافُهُ.<br />هكذا تَسحقينَ شُعوبًا كثيرينَ،<br />فأجعلُ غنائِمَهُم وما يَملِكونَ<br />حَلالًا لي أنا الرّبُّ<br />سيِّدُ كُلِّ الأرضِ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14"> والآنَ جُدَّ الجِدُّ يا بنتَ الجدودِ!<br />يُضيِّقونَ علَينا الحِصارَ<br />ويَضربونَ قاضي بَني إِسرائيلَ<br />على خَدِّهِ بالقضيبِ.</span>
