## العجل الذهبي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اعمال الرسل 7:40; التَّثْنِيَة 9:9; الخُرُوج 24:18"> ولمَّا رأى الشَّعبُ أنَّ موسى أبطأَ في النُّزولِ مِنَ الجبَلِ اَجتَمَعوا على هرونَ وقالوا لَه: ”قُمِ اَصنَعْ لنا آلِهَةً تسيرُ أمامَنا. فهذا الرَّجلُ موسى الّذي أخرَجَنا مِنْ أرضِ مِصْرَ لا نعرِفُ ماذا أصابَهُ“.</span>
<span className="v" id="v2" data-v="2" data-xref="القُضاة 8:24–27; الخُرُوج 12:35–36; حِزْقِيال 16:17"> فقالَ لهُم هرونُ: ”إنزَعوا حَلَقَ الذَّهَبِ الّتي في آذانِ نِسائِكُم وبَنيكُم وبناتِكُم وجيئوني بِها“.</span>
<span className="v" id="v3" data-v="3" data-xref="إشَعْياء 46:6; إشَعْياء 40:19–20; إرْمِيا 10:9"> فنزَعَ جميعُ الشَّعبِ حَلَقَ الذَّهَبِ الّتي في آذانِ نِسائِهِم وجاؤوا بِها إلى هرونَ.</span>
<span className="v" id="v4" data-v="4" data-xref="نَحَمْيا 9:18; التَّثْنِيَة 9:16; المُلُوكِ الأوَّلُ 12:28"> فأخذَها منْ أيديهِم وأذابَها وسَكبَها في صَنَمٍ على صورَةِ عِجْلٍ. فقالَ الشَّعبُ: ”هذِهِ آلِهتُكُم يا بَني إِسرائيلَ، آلِهتُكُمُ الّتي أخرَجتْكُم مِنْ أرضِ مِصْرَ“.</span>
<span className="v" id="v5" data-v="5" data-xref="المُلُوكِ الثّانِي 10:20; اللّاويِّين 23:2; اللّاويِّين 23:37"> فلمَّا رأى هرونُ ذلِكَ بَنى أمامَ الصَّنَمِ مذبَحًا ونادى وقالَ: ”غدًا عيدٌ للرّبِّ“.</span>
<span className="v" id="v6" data-v="6" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:7; العَدَد 25:2; اعمال الرسل 7:41–42"> فبكَّروا في الصَّباحِ وأصعَدوا مُحرقاتٍ وقدَّموا ذبائِحَ سلامَةٍ وجلسوا يأكلُونَ ويشرَبونَ، ثمَّ قاموا يمرَحونَ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 32:11; هُوشَع 9:9; القُضاة 2:19"> فقالَ الرّبُّ لِموسى: ”قُمِ اَنْزِلْ. فسَدَ شعبُكَ الّذينَ أخرَجْتُهُم مِنْ أرضِ مِصْرَ.</span>
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الأوَّلُ 12:28; الخُرُوج 20:23; الخُرُوج 20:3–4"> حادوا سريعًا عَنِ الطَّريقِ الّذي أمَرتُهُم بِسُلوكِهِ، فصنَعوا لهُم عِجْلًا مَسبوكًا وسجَدوا لَه وقَدَّموا الذَّبائِحَ وقالوا: ”هذِهِ آلِهتُكُم يا بَني إِسرائيلَ، آلِهتُكُمُ الّتي أخرَجَتْكُم منْ أَرضِ مِصْرَ“.</span>
<span className="v" id="v9" data-v="9" data-xref="اعمال الرسل 7:51; التَّثْنِيَة 9:13; إشَعْياء 48:4"> وقالَ الرّبُّ لِموسى: ”رأيتُ هؤلاءِ الشَّعبَ، فإذا هُم شعبٌ قُساةُ الرِّقابِ.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 9:14; العَدَد 14:12; التَّثْنِيَة 9:19"> والآنَ دَعْ غضَبي يشتَدُّ علَيهِم فأُفنيَهُم وأجعَلَكَ أنتَ أُمَّةً عظيمَةً“.</span>
</div>
</div>

## صلاة موسى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v11" data-v="11" data-xref="التَّثْنِيَة 9:18–20; التَّثْنِيَة 9:26–29; المَزامِير 106:23"> فتَضَرَّعَ موسى إلى الرّبِّ إلهِهِ وقالَ: ”يا ربُّ، لماذا يشتَدُّ غضَبُكَ على شعبِكَ الّذينَ أخرَجْتَهُم مِنْ أرضِ مِصْرَ بِقوَّةٍ عظيمَةٍ ويَدٍ قديرَةٍ؟</span>
<span className="v" id="v12" data-v="12" data-xref="العَدَد 14:13–16; التَّثْنِيَة 9:28; يَشُوع 7:9"> أفَلا يقولُ المِصْريُّونَ إنَّ إِلهَهُم أخرَجَهُم مِنْ هُنا بِسُوءِ نِيَّةٍ، لِيَقتُلَهُم في الجِبالِ ويُفنِيَهُم عَنْ وجهِ الأرضِ؟ إرجِعْ عَنْ شِدَّةِ غضَبِكَ وَعُدْ عنِ الإساءَةِ إلى شعبِكَ.</span>
<span className="v" id="v13" data-v="13" data-xref="العِبرانِيّين 6:13; التَّكوين 12:7; التَّكوين 22:16"> واَذكُر إبراهيمَ وإسحَقَ ويعقوبَ عبيدَكَ الّذينَ أقسمْتَ لهُم بِذاتِكَ وقُلتَ لهُم إنِّي أُكثِرُ نسلَكُم كنجومِ السَّماءِ وأُعطيكُم جميعَ هذِهِ الأرضِ الّتي وعَدتُكم بِها، فتَرِثونَها إلى الأبدِ“.</span>
<span className="v" id="v14" data-v="14" data-xref="ﻳﻮﻧﺎﻥ 3:10; المَزامِير 106:45; إرْمِيا 26:13"> فعادَ الرّبُّ عَنِ السُّوءِ الّذي قالَ إنَّه سَيُنزِلُهُ بِشعبِهِ.</span>
</div>
</div>

## موسى يكسر لوحي الوصايا

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="التَّثْنِيَة 9:15; الخُرُوج 40:20; التَّثْنِيَة 5:22"> ثُمَّ اَنصرفَ موسى ونزَلَ مِنَ الجبَلِ ولوحا الوصايا في يَدِهِ، وهُما لوحانِ مكتوبانِ على كُلِّ جانبٍ مِنهُما،</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 31:18; العِبرانِيّين 8:10; التَّثْنِيَة 10:1"> صنعَهُما اللهُ ونقَشَ كِتابَتَهُ علَيهِما.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 24:13; الخُرُوج 17:9; يَشُوع 6:20"> وسَمِعَ يشوعُ صوتَ صياحِ الشَّعبِ فقالَ لموسى: ”صوتُ حربٍ في المَحلَّةِ“.</span>
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 15:1–18; دانيال 5:4; دانيال 5:23"> فأجابَه موسى: ”ما هذا صياحُ نَصرٍ ولا صياحُ هزيمَةٍ، بل صوتُ غِناءٍ أنا سامِعٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 9:16–17; إرْمِيا 31:32; ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 5:15"> فلمَّا اَقترَبَ موسى مِنَ المَحلَّةِ رأى العِجْلَ والرَّقصَ، فاَشتَدَّ غضَبُهُ ورمى بِاللَّوحَينِ مِنْ يَدَيهِ وكسَّرَهُما في أسفَلِ الجبَلِ.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 9:21; المُلُوكِ الثّانِي 23:6; المُلُوكِ الثّانِي 23:15"> ثُمَّ أخذَ العِجْلَ الّذي صنَعوهُ، فأحرَقَهُ بِالنَّارِ وطَحَنَهُ حتّى صارَ ناعِمًا، وذَرَّاهُ على وجهِ الماءِ وسقى بَني إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 20:9; يَشُوع 7:19–26; المُلُوكِ الأوَّلُ 21:22"> وقالَ موسى لِهرونَ: ”ماذا صنَعَ بِكَ هؤلاءِ الشَّعبُ حتّى جلَبْتَ علَيهِم خطيئةً عظيمةً؟“</span>
<span className="v" id="v22" data-v="22" data-xref="التَّثْنِيَة 9:24; الخُرُوج 15:24; الخُرُوج 16:20"> فأجابَهُ هرونُ: ”لا يشْتَدَّ غضَبُكَ يا سيِّدي. أنتَ تعرِفُ الشَّعبَ أنَّهم مِنَ الرُّعَاعِ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 32:1–4; الخُرُوج 32:8"> قالوا لي: إصنَعْ لنا آلِهةً تسيرُ أمامَنا، فهذا الرَّجلُ موسى الّذي أخرَجَنا مِنْ أرضِ مِصْرَ لا نعلَمُ ماذا أصابَهُ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 32:4; لُوقا 10:29; ﺭﻭﻣﻴﺔ 3:10"> فقُلتُ لهُم: مَنْ لَه ذهَبٌ فليَنزَعْهُ. فجاؤُوني بهِ، فطَرَحْتُه في النَّارِ، فخَرَجَ هذا العِجْلُ“.</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 47:3; ميخا 1:11; رُؤيا يُوحَنّا 16:15"> ولمَّا رأى موسى أنَّ الشَّعبَ خرَجوا على هرونَ فتَركَهُم يُمْعِنُونَ في غَيِّهِم،</span>
<span className="v" id="v26" data-v="26" data-xref="المُلُوكِ الثّانِي 9:32; ﻣﺘﻰ 12:30; صموئيلَ الثّانِي 20:11"> وقفَ على بابِ المَحلَّةِ وقالَ ”مَنْ مِنكُم لِلرّبِّ فَلْيَجِئْ إليَّ“. فاَجْتَمعَ إليهِ جميعُ بَني لاوي،</span>
<span className="v" id="v27" data-v="27" data-xref="العَدَد 25:5; لُوقا 14:26; التَّثْنِيَة 33:8–9"> فقالَ لهُم: ”قالَ الرّبُّ إلهُ إِسرائيلَ: على كُلِّ واحدٍ مِنكُم أنْ يحمِلَ سيفَه ويَطوفَ المَحلَّةِ مِنْ بابٍ إلى بابٍ ويَقتُلَ أخاهُ وصديقَه وجارَهُ“.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 33:9; مَلاخِي 2:4–6; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:8"> ففعَلَ بَنو لاوي كما أمرَ موسى، فسَقطَ مِنَ الشَّعبِ في ذلِكَ اليومِ ثَلاثَةُ آلافِ رَجُلٍ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 25:11–13; ﻣﺘﻰ 10:37; زَكَريّا 13:3"> وقالَ لهُم موسى: ”اليومَ كَرَّسْتُم نُفوسَكُم لِلرّبِّ، كُلُّ واحدٍ على حِسابِ اَبْنِه وأخيهِ، فمَنَحكُمُ الرّبُّ بَرَكَتَهُ“.</span>
</div>
</div>

## موسى يصلي للرب ثانية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="صَمُوئيلَ الأوَّلُ 12:20; العَدَد 25:13; المُلُوكِ الثّانِي 17:21"> وفي الغدِ قالَ موسى للشَّعبِ: ”خَطئتُم خطيئةً عظيمةً. والآنَ أصعَدُ إلى الرّبِّ لَعلِّي أُكَفِّرُ خطيئَتَكُم“.</span>
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 20:23; دانيال 9:11; دانيال 9:8"> ورَجعَ موسى إلى الرّبِّ وقالَ: ”يا ربُّ، خَطِئَ هؤلاءِ الشَّعبُ خطيئةً عظيمَةً وصنَعوا لهُم آلِهةً مِنْ ذهَبٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="المَزامِير 69:28; ﺭﻭﻣﻴﺔ 9:3; رُؤيا يُوحَنّا 21:27"> فإمَّا تغفِرُ خطيئَتَهُم أو تَمحوني مِنْ كِتابِكَ الّذي كتَبتَهُ“.</span>
<span className="v" id="v33" data-v="33" data-xref="التَّثْنِيَة 29:20; حِزْقِيال 18:4; رُؤيا يُوحَنّا 20:12"> فقالَ الرّبُّ لِموسى: ”لا أمحو مِنْ كِتابي إلَّا الّذي خَطِئَ إليَّ.</span>
<span className="v" id="v34" data-v="34" data-xref="الخُرُوج 23:20; التَّثْنِيَة 32:35; ﻣﺘﻰ 23:35"> والآنَ تذهَبُ وتقودُ الشَّعبَ إلى حَيثُ قُلتُ لكَ، وها هوَ ملاكي يسيرُ أمامَكَ. ولكِنْ يجيءُ يومٌ أُعاقِبُهُم فيهِ على خطيئَتِهِم“.</span>
<span className="v" id="v35" data-v="35" data-xref="اعمال الرسل 7:41; الخُرُوج 32:25; صموئيلَ الثّانِي 12:9–10"> وبَطَشَ الرّبُّ بالشَّعبِ لأنَّهُم عبَدوا العِجْلَ الّذي صنعهُ هرونُ.</span>
</div>
