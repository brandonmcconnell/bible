## الضربة الخامسة: موت المواشي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 8:1; الخُرُوج 10:3; الخُرُوج 8:20"> وقالَ الرّبُّ لموسى: ”أُدخُلْ على فِرعَونَ وقُلْ لَه إنَّ الرّبَّ إلهَ العِبرانيِّينَ يقولُ لكَ أطلِقْ شعبي ليعبُدوني.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 8:2; الخُرُوج 4:23; اللّاويِّين 26:27–28"> فإنْ رفَضْتَ أنْ تُطلِقَهُم وأصرَرْتَ على اَحْتِجازِهِم،</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 7:4; اعمال الرسل 13:11; صَمُوئيلَ الأوَّلُ 6:9"> فها يَدي أنا الرّبُّ تَضرِبُ مَواشيَكَ الّتي في البرِّيَّةِ: الخَيلَ والحميرَ والجِمالَ والبقَرَ والغنَمَ بوباءٍ مُميتٍ.</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 8:22; الخُرُوج 10:23; مَلاخِي 3:18"> وأميِّزُ بَينَ مَواشي إِسرائيلَ ومَواشي المِصْريِّينَ، فلا يموتُ شيءٌ مِنْ جميعِ ما هوَ لبَني إِسرائيلَ“.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 8:23; الخُرُوج 10:4; إرْمِيا 28:16–17"> وعيَّنَ الرّبُ يومَ غدٍ موعِدًا يُنَفِّذُ فيهِ هذا الأمرَ في أرضِ مِصْرَ.</span>
<span className="v" id="v6" data-v="6" data-xref="المَزامِير 78:48; الخُرُوج 9:25; المَزامِير 78:50"> فنفَّذَ الرّبُّ هذا الأمرَ في الغدِ، فماتَت مَواشي المِصْريِّينَ كُلُّها، وأمَّا مَواشي بَني إِسرائيلَ فما ماتَ منها واحِدٌ.</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 7:14; الخُرُوج 8:32; إشَعْياء 48:4"> واَسْتَخبَرَ فِرعَونُ عمَّا جرَى، فعلِمَ أنَّ مَواشي إِسرائيلَ لم يمُتْ مِنها واحدٌ. ولكنَّ قلبَهُ بَقيَ قاسيًا فلم يُطلِقْ شعبَ إِسرائيلَ مِنْ مِصْرَ.</span>
</div>
</div>

## الضربة السادسة: القروح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 8:16"> فقالَ الرّبُّ لموسى وهرونَ: ”خُذا مِلْءَ كفَّيكُما مِنْ رَمادِ الأتُونِ، وليَذُرَّهُ موسى إلى السَّماءِ أمامَ عينَي فِرعَونَ،</span>
<span className="v" id="v9" data-v="9" data-xref="رُؤيا يُوحَنّا 16:2; التَّثْنِيَة 28:27; أيُّوب 2:7"> فيصيرَ غُبارًا على جميعِ أرضِ مِصْرَ ويُخرِجَ قُروحًا وبُثورًا في النَّاسِ والبَهائمِ كُلِّها“.</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 28:27"> فأخذَ موسى وهرونُ مِنْ رمادِ الأتُونِ ووقَفا أمامَ فِرعَونَ، وذَرَّاهُ موسى إلى السَّماءِ فصارَ قُروحًا وبُثورًا في النَّاسِ والبَهائِمِ.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 7:11–12; الخُرُوج 8:18–19; رُؤيا يُوحَنّا 16:2"> ولم يقدِرِ السَّحرَةُ أنْ يقِفُوا أمامَ موسى بسببِ القُروحِ الّتي كانَت في السَّحَرَةِ وفي جميعِ المِصْريِّينَ.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 4:21; الخُرُوج 7:13–14; المَزامِير 81:11–12"> وقَسَّى الرّبُّ قلبَ فِرعَونَ، فلم يسمعْ لموسى وهرونَ، كما قالَ الرّبُّ لموسى.</span>
</div>
</div>

## الضربة السابعة: البرد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="الخُرُوج 8:20; الخُرُوج 7:15; الخُرُوج 9:1"> وقالَ الرّبُّ لموسى: ”بَكِّرْ في الغدِ وَقِفْ أمامَ فِرعَونَ وقُلْ لَه: يقولُ لكَ إلهُ العِبرانيِّينَ أطلِقْ شعبي ليعبُدوني،</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 8:10; اللّاويِّين 26:28; أخبارِ الأيّامِ الأوّلُ 17:20"> وإلَّا فأنا في هذِهِ المرَّةِ أُنزِلُ جميعَ ضرَباتي علَيكَ أنتَ وعلى رِجالِكَ وشعبِكَ، فتَعرِفَ أنْ لا نظيرَ لي في الأرضِ.</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 3:20; الخُرُوج 9:3; الخُرُوج 9:16"> وإنْ كُنتُ حتّى الآنَ لم أمُدَّ يَدي وأضرِبْكَ أنتَ وشعبَكَ بالوباءِ لتزولَ مِنَ الأرضِ،</span>
<span className="v" id="v16" data-v="16" data-xref="ﺭﻭﻣﻴﺔ 9:17; الأمثال 16:4; المَزامِير 83:17–18"> فلأنِّي أردتُ أنْ أُبقِيَكَ لتَرى قُدرَتي ويشتَهِرَ اَسمي في كُلِّ الأرضِ.</span>
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 37:29; إشَعْياء 10:15; إشَعْياء 37:23–24"> وما دُمتَ تُقاومُ شعبي ولا تُطلِقُهم مِنْ بِلادِكَ،</span>
<span className="v" id="v18" data-v="18" data-xref="المُلُوكِ الأوَّلُ 20:6; المَزامِير 83:15; الخُرُوج 9:22–25"> فسأُمطِرُ غدًا في مِثلِ هذا الوقتِ بَرَدًا عظيمًا ما عرَفَت مِصْرُ لَه مَثيلًا مِنْ يومِ تأسيسِها إلى الآنَ.</span>
<span className="v" id="v19" data-v="19" data-xref="حَبَقُّوق 3:2"> فأسرِعْ واَجمَعْ مَواشيَكَ وكُلَّ ما لكَ في البرِّيَّةِ إلى الحَظائِرِ. فكُلُّ إنسانٍ أو بَهيمةٍ في البرِّيَّةِ لا يلجأُ إلى الحَظائرِ يَنزِلُ علَيهِ البَرَدُ فيموتُ“.</span>
<span className="v" id="v20" data-v="20" data-xref="الأمثال 13:13; الأمثال 22:3; العِبرانِيّين 11:7"> فكانَ أنَّ الّذينَ خافوا كلامَ الرّبِّ مِنْ رِجالِ فِرعَونَ هربوا بعبيدِهِم ومَواشيهِم إلى البيوتِ،</span>
<span className="v" id="v21" data-v="21" data-xref="حِزْقِيال 40:4; أيُّوب 34:14; أخبارِ الأيّامِ الأوّلُ 22:19"> وأمَّا الّذينَ لم يُبالُوا بِكلامِ الرّبِّ فترَكوا عبيدَهُم ومَواشيَهُم في البرِّيَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="رُؤيا يُوحَنّا 16:21; الخُرُوج 7:19; الخُرُوج 8:5"> وقالَ الرّبُّ لموسى: ”مُدَّ يَدَكَ نَحوَ السَّماءِ فينزِلَ بَرَدٌ في جميعِ أرضِ مِصْرَ على النَّاسِ والبَهائِمِ وأعشابِ البرِّيَّةِ كُلِّها“.</span>
<span className="v" id="v23" data-v="23" data-xref="المَزامِير 18:13; يَشُوع 10:11; إشَعْياء 30:30"> فمَدَّ موسى عصاهُ نحوَ السَّماءِ، فأرعَدَ الرّبُّ وأبرَقَ نارًا جَرَت على الأرضِ. وأمطرَ الرّبُّ بَرَدًا</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 9:23; ﻣﺘﻰ 24:21; الخُرُوج 10:6"> فكانَ البَرَدُ والنَّارُ بَينَ البَرَدِ شيئًا عظيمًا لم يكُنْ لَه مَثيلٌ في أرضِ مِصْرَ مُنذُ سكَنهَا بشَرٌ.</span>
<span className="v" id="v25" data-v="25" data-xref="المَزامِير 78:47; المَزامِير 105:32–33"> وضَربَ البَرَدُ في جميعِ أرضِ مِصْرَ كُلَّ ما في البرِّيَّةِ مِنَ النَّاسِ والبهائِمِ والأعشابِ وكسَّرَ كُلَّ شجَرَةٍ فيها.</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 11:7; الخُرُوج 9:4; الخُرُوج 10:23"> أمَّا أرضُ جاسانَ الّتي فيها بَنو إِسرائيلَ فلم ينزِلْ علَيها البَرَدُ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="ﻣﺮﺍﺛﻲ ﺇﺭﻣﻴﺎ 1:18; المَزامِير 129:4; الخُرُوج 10:16"> فاَستَدعى فِرعَونُ موسى وهرونَ وقالَ لهُما: ”خَطِئتُ هذِهِ المرَّةَ. الرّبُّ عادلٌ وأنا وشعبي أشرارٌ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 8:8; الخُرُوج 10:17; الخُرُوج 8:28"> تَشفَّعا إلى الرّبِّ أنْ يُوقِفَ الرُّعودَ والبرَدَ، فأُطلِقَكُم مِنْ مِصْرَ ولا تُقيمونَ فيها بَعدُ“.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 10:26; المُلُوكِ الأوَّلُ 8:38; إشَعْياء 1:15"> فقالَ لَه موسى: ”إذا خَرجْتُ مِنَ المدينةِ أبسُطُ يَدي إلى الرّبِّ، فتتوقَّفُ الرُّعودُ ولا ينزِلُ البرَدُ، لتَعرِفَ أنَّ الأرضَ للرّبِّ.</span>
<span className="v" id="v30" data-v="30" data-xref="إشَعْياء 26:10; الأمثال 16:6; إشَعْياء 63:17"> أمَّا أنتَ ورجالُكَ، فأعرِفُ أنَّكم لا تخافونَ الرّبَّ الإلهَ بَعدُ“.</span>
<span className="v" id="v31" data-v="31" data-xref="راعُوث 1:22; راعُوث 2:23; حَبَقُّوق 3:17"> وكانَ البرَدُ ضرَبَ الكتَّانَ والشَّعيرَ، لأنَّ الشَّعيرَ كانَ بَعْدُ سُنبُلًا والكتَّانَ بِزْرًا،</span>
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 10:22"> وأمَّا الحِنطةُ والقَطانيُّ فما تَلِفَت لأنَّها تتأخَّرُ في النُّضوجِ.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="الخُرُوج 9:29; الخُرُوج 8:12; يَعقُوب 5:17–18"> وخرَجَ موسى مِنْ عِندِ فِرعَونَ ومِنَ المدينةِ وبسطَ يَديهِ إلى الرّبِّ، فتَوقَّفَتِ الرُّعودُ والبرَدُ وما عادَ المَطرُ يَنصبُّ على الأرضِ.</span>
<span className="v" id="v34" data-v="34" data-xref="الخُرُوج 7:14; أخبارِ الأيّامِ الثّانِي 36:13; الخُرُوج 4:21"> ولمَّا رأى فِرعَونُ أنَّ المَطرَ توقَّفَ والبرَدَ والرُّعودَ أيضًا، عادَ إلى خطيئَتِهِ، فقسَّى قلبَهُ هوَ ورِجالُه.</span>
<span className="v" id="v35" data-v="35" data-xref="الخُرُوج 4:21"> فلمْ يُطلِقْ بَني إِسرائيلَ مِنْ مِصْرَ، كما قالَ الرّبُّ.</span>
</div>
