<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 34:32; الخُرُوج 31:1–11; ﺭﻭﻣﻴﺔ 2:13"> وجمَعَ موسى بَني إِسرائيلَ كُلَّهم وقالَ لهُم: ”هذِهِ هيَ الوصايا الّتي أمرَ الرّبُّ أن تعمَلوا بِها:</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 23:3; لُوقا 13:14–15; الخُرُوج 20:9–10"> في سِتَّةِ أيّامٍ تعمَلونَ، واليومُ السَّابعُ يكونُ لكُم سبْتًا مُقَدَّسًا وعُطلَةً لي أنا الرّبّ. كُلُّ مَنْ عمِلَ فيه عمَلًا يُقتَلُ.</span>

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 25:1–2"> وقالَ موسى لِجميعِ بَني إِسرائيلَ: ”هذا ما أمرَ الرّبُّ بهِ:</span>
<span className="v" id="v5" data-v="5" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:7; المَزامِير 110:3; الخُرُوج 25:2–7"> تَأخُذونَ مِنْ عِندِكُم تَقدِمَةً لِلرّبِّ. كُلُّ مَنْ كانَت نفْسُهُ سَخيَّةً يَجيءُ بِتَقدِمَةٍ لِلرّبِّ مِنْ ذهَبٍ وفِضَّةٍ ونُحاسٍ</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 26:7–14; الخُرُوج 26:31; الخُرُوج 28:15"> ونسيجٍ بنَفْسَجيٍّ وأُرجُوانيٍّ وقِرمِزيِّ اللَّونِ وشَعْرِ مَعَزٍ،</span>
<span className="v" id="v7" data-v="7"> وجُلودِ كِباشٍ مَصبوغَةٍ بِالحُمرَةِ وجُلودِ مَعَزٍ وخشَبِ سَنْطٍ،</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 27:20; الخُرُوج 25:1–40; الخُرُوج 30:23"> وزيتٍ لِلإنارَةِ وأطيابٍ لِزيتِ المَسْحِ ولِلبَخورِ العَطِرِ،</span>

<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 26:1–37; الخُرُوج 36:8–34; الخُرُوج 31:7–9"> المَسكِنَ وخَيمتَهُ وغِطاءَه ومَشابِكَهُ وألواحَهُ وعَوارِضَهُ وأعمِدَتَهُ وقواعِدَهُ،</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 25:10–22; الخُرُوج 26:31–33; الخُرُوج 37:1–9"> والتَّابوتَ وقضيبَيهِ، والغِطاءَ والحِجابَ،</span>

<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 27:1–8; الخُرُوج 30:18–21; الخُرُوج 38:1–8"> ومذبَحَ المُحرَقاتِ ووِجاقَةَ النُّحاسِ وقضيبَيه وجميعَ أدَواتِه، والمَغسلَةَ وقاعِدَتَها،</span>
<span className="v" id="v17" data-v="17" data-xref="صموئيلَ الثّانِي 7:2; الخُرُوج 38:9–20; الخُرُوج 27:9–19"> وستائِرَ الرِّواقِ وأعمِدَتَهُ وقواعِدَهُ، وسِتارَةَ مَدخلِ الرِّواقِ،</span>
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 27:19"> وأوتادَ المَسكِنِ والرِّواقِ وحِبالَهُما،</span>

## تقدمات بني إسرائيل

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20"> فخَرجَ جماعَةُ بَني إِسرائيلَ مِنْ عِندِ موسى،</span>
<span className="v" id="v21" data-v="21" data-xref="الخُرُوج 25:2; الخُرُوج 35:26; أخبارِ الأيّامِ الأوّلُ 29:14"> ثُمَّ جاءَ كُلُّ مَنْ حرَّكَهُ قلبُهُ وسَخَتْ نفْسُهُ بِتقدِمَةٍ للرّبِّ مِنْ أجلِ بِناءِ خَيمَةِ الإجتماعِ. وجاؤوا بِكُلِّ ما يَلزَمُ لِلعِبادَةِ فيها ولِلثِّيابِ المُقَدَّسَةِ.</span>
<span className="v" id="v22" data-v="22" data-xref="نَحَمْيا 7:70–72; ﻣﺘﻰ 2:11; عَزْرا 2:68–69"> وجاءَ الرِّجالُ والنِّساءُ مِمَّنْ سَخَتْ نُفوسُهُم بِأساوِرَ وحَلَقٍ وخَواتِمَ وقلائِدَ وما إلى ذلِكَ مِنْ أشياءَ ذهَبيةٍ وقدَّمُوها لِلرّبِّ.</span>
<span className="v" id="v23" data-v="23" data-xref="أخبارِ الأيّامِ الأوّلُ 29:8; الخُرُوج 25:2–7; الخُرُوج 35:6–10"> وكُلُّ مَنْ كانَ عِندَه نَسيجٌ بَنَفْسَجيٌّ وأُرجُوانيٌّ وقِرمِزيُّ اللَّونِ جاءَ بهِ، وجاءَ بِشَعْرِ مَعَزٍ وجُلودِ كِباشٍ مَصبوغَةٍ بِالحُمرَةِ وجُلودٍ بَنَفْسَجيَّةٍ.</span>
<span className="v" id="v24" data-v="24" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:12"> وكلُّ مَنْ كانَ عِندَه تقدِمَةٌ مِنْ فِضَّةٍ ونُحاسٍ جاءَ بِها. وكُلُّ مَنْ كانَ عِندَه خشَبُ سَنْطٍ يَلزَمُ لِلعمَلِ جاءَ بِهِ.</span>
<span className="v" id="v25" data-v="25" data-xref="ﺭﻭﻣﻴﺔ 16:12; لُوقا 8:2–3; الأمثال 14:1"> وكُلُّ اَمرأةٍ ماهرَةٍ غزَلَت بِيَدِها وجاءَت بِغَزْلٍ مِنَ النَّسيجِ البَنَفْسَجيِّ والأُرجُوانيِّ والقِرمِزيِّ اللَّونِ.</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 35:21; الخُرُوج 36:8; الخُرُوج 35:29"> وكُلُّ اَمرأةٍ حرَّكَها قلبُها إلى العمَلِ بِمَهارَةٍ غزَلَت شَعْرَ مَعَزٍ.</span>
<span className="v" id="v27" data-v="27" data-xref="عَزْرا 2:68; أخبارِ الأيّامِ الأوّلُ 29:6; الخُرُوج 35:9"> والأعيانُ جاؤوا بِحجَارةِ العَقيقِ وحجارَةِ التَّرصيعِ لِلأَفودِ وللصُّدْرَةِ،</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 30:23–38; الخُرُوج 35:8"> وبالطِّيْبِ والزَّيتِ للمَنارَةِ والمَسْحِ والبَخورِ العَطِرِ.</span>
<span className="v" id="v29" data-v="29" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:7; الخُرُوج 35:21–22; الخُرُوج 35:4"> كُلُّ رَجُلٍ أوِ اَمرأةٍ مِنْ بَني إِسرائيلَ سَخَتْ نفْسُهُ جاءَ بشيءٍ لِلعمَلِ الّذي أمرَ الرّبُّ بأنْ يعمَلَهُ الشَّعبُ على يَدِ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="الخُرُوج 31:1–6; يَعقُوب 1:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:11"> وقالَ موسى لِبَني إِسرائيلَ: ”الرّبُّ اَختارَ بَصَلْئيلَ بنَ أُوري بنَ حُورٍ مِنْ سِبْطِ يَهوذا</span>

<span className="v" id="v32" data-v="32"> لاَبتداعِ أشكالٍ يصنَعُها مِنَ الذَّهَبِ والفِضَّةِ والنُّحاسِ،</span>

<span className="v" id="v35" data-v="35" data-xref="الخُرُوج 35:31; المُلُوكِ الأوَّلُ 7:14; إشَعْياء 28:26"> ومَلأَ قلبَيهِما حِكمَةً لِيَصنَعا كُلَّ صَنعَةٍ يُتقِنُها النَّجَّارُ والحائِكُ الماهِرُ والمُطرِّزُ في النَّسيجِ البَنَفْسَجيِّ والأُرجُوانيِّ والقِرمِزيِّ اللَّونِ، وغيرِ ذلِكَ مِمَّا يُتقِنُه الحائِكُ وكُلُّ صاحِبِ صَنعَةٍ وفَنٍّ مِنَ الفُنونِ“.</span>
