<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 12:1; الخُرُوج 13:1"> وقالَ الرّبُّ لموسى:</span>
<span className="v" id="v2" data-v="2" data-xref="العَدَد 33:7–8; إرْمِيا 44:1; الخُرُوج 13:17–18"> ”قُلْ لبَني إِسرائيلَ أنْ يرجعوا وينزِلوا أمامَ فَمِ الحيروثِ بَينَ مَجدُلَ والبحرِ، تُجاهَ بعْلَ صفونَ على البحرِ.</span>
<span className="v" id="v3" data-v="3" data-xref="اعمال الرسل 4:28; صَمُوئيلَ الأوَّلُ 23:23; المَزامِير 71:11"> فيظُنَّ فِرعَونُ أنَّ بَني إِسرائيلَ تائِهونَ في الأرضِ وأنَّ البرِّيَّةَ اَنغلقت علَيهِم.</span>
<span className="v" id="v4" data-v="4" data-xref="ﺭﻭﻣﻴﺔ 9:22–23; الخُرُوج 7:5; ﺭﻭﻣﻴﺔ 9:17"> وأُقسِّي قلبَ فِرعَونَ فيتبعُهُم، وأُجهِزُ على فِرعَونَ وجميعِ جُنودهِ، ويَعلمُ المِصْريُّونَ أنِّي أنا الرّبُّ“. ففعَلَ بَنو إسرائيلَ كما قيلَ لهُم.</span>
</div>

## المصريون يتبعون الإسرائيليين

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v5" data-v="5" data-xref="المَزامِير 105:25; الخُرُوج 12:33; لُوقا 11:24–26"> فلمَّا سَمِعَ فِرعَونُ مَلِكُ مِصْرَ بأنَّ بَني إِسرائيلَ هربوا اَنقلبَ رأيُهُ ورأيُ رِجالهِ علَيهِم وقالوا: ”ماذا عَمِلْنا، فأطلَقْنا بَني إسرائيلَ مِنْ خِدمتِنا“.</span>
<span className="v" id="v6" data-v="6"> فشدَّ فِرعَونُ مركبتَهُ وأخذَ جيشَهُ معَهُ،</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 15:4; القُضاة 4:3; المَزامِير 68:17"> واَختارَ سِتَّ مئةِ مركبةٍ مِنْ مركباتهِ وأخذَ معَها جميعَ مركباتِ مِصْرَ وعلَيها قادتُها.</span>
<span className="v" id="v8" data-v="8" data-xref="اعمال الرسل 13:17; العَدَد 33:3; التَّثْنِيَة 26:8"> وقَسَّى الرّبُّ قلبَ فِرعَونَ مَلكِ مِصْرَ، حتّى تَبِعَ بَني إِسرائيلَ، وهُم خارجونَ بقوَّةٍ عظيمةٍ،</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 15:9; يَشُوع 24:6; الخُرُوج 14:2"> فلحِقَهُمُ المِصْريُّونَ بجميعِ مركباتِ فِرعَونَ وفُرسانهِ وجُنودهِ وهُم نازلونَ عِندَ البحرِ قربَ فَمِ الحيروثِ، أمامَ بعْلَ صفونَ.</span>
<span className="v" id="v10" data-v="10" data-xref="يَشُوع 24:7; المَزامِير 34:17; نَحَمْيا 9:9"> ولمَّا اَقتربَ فِرعَونُ رفعَ بَنو إِسرائيلَ عُيونَهُم، فرأوا المِصْريِّينَ يتبَعونَهم. فخافوا جدًّا وصرخوا إلى الرّبِّ.</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 106:7–8; العَدَد 14:1–4; الخُرُوج 15:23–24"> وقالوا لموسى: ”أمَا في مِصْرَ قُبورٌ، فأخذْتَنا لنموتَ في هذهِ البرِّيَّةِ؟ ماذا عمِلتَ بنا، فأخرجْتَنا مِنْ مِصْرَ؟</span>
<span className="v" id="v12" data-v="12" data-xref="ﻳﻮﻧﺎﻥ 4:3; الخُرُوج 6:9; ﻳﻮﻧﺎﻥ 4:8"> أما قُلنا لكَ في مِصْرَ؛ دَعْنا نخدِمُ المِصْريِّينَ، فخِدْمَتُنا لهُم خَيرٌ مِنْ أنْ نموتَ في البرِّيَّةِ؟“</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الثّانِي 20:17; أخبارِ الأيّامِ الثّانِي 20:15; الخُرُوج 14:30"> فقالَ لهُم موسى: ”لا تخافوا. قِفوا واَنظروا اليومَ خلاصَ الرّبِّ. سترَونَ المِصْريِّينَ اليومَ ولن تعودوا ترَونَهم إلى الأبدِ.</span>
<span className="v" id="v14" data-v="14" data-xref="التَّثْنِيَة 3:22; التَّثْنِيَة 20:4; أخبارِ الأيّامِ الثّانِي 20:17"> الرّبُّ يُحاربُ عَنكُم، وأنتُم لا تُحرِّكونَ ساكِنًا“.</span>
</div>
</div>

## عبور البحر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="يَشُوع 7:10; عَزْرا 10:4–5; نَحَمْيا 9:9"> فقالَ الرّبُّ لموسى: ”ما بالُكَ تصرُخُ إليَّ؟ قُلْ لبَني إِسرائيلَ ليرحَلوا.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 7:19; الخُرُوج 4:20; الخُرُوج 4:2"> وأنتَ إرفعْ عصاكَ ومُدَّ يَدَكَ على البحرِ فينشقَّ ليَدخُلَ بَنو إِسرائيلَ في وسَطِ البحرِ على الأرضِ اليابسةِ.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 14:8; الخُرُوج 14:4; حِزْقِيال 34:11"> وأنا أُقسِّي قُلوبَ المِصْريِّينَ فيدخُلونَ وراءَهُم، فأُجهِزُ على فِرعَونَ وجميعِ جُنودهِ ومركباتهِ وفُرسانهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 7:5; الخُرُوج 14:4; الخُرُوج 7:17"> فيعلمَ المِصْريُّونَ متى أجهَزْتُ علَيهِم أنِّي أنا الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 63:9; الخُرُوج 32:34; الخُرُوج 23:20–21"> فاَنتقلَ ملاكُ اللهِ مِنْ أمامِ جيشِ بَني إِسرائيلَ وسارَ وراءَهُم واَنتقلَ عَمودُ السَّحابِ أيضًا ووقفَ وراءَهم</span>
<span className="v" id="v20" data-v="20" data-xref="إشَعْياء 8:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 2:15–16; الأمثال 4:18–19"> بحيثُ دخلَ بَينَ جيشِ المِصْريِّينَ وجيشِ بَني إسرائيلَ، فكانَ مِنْ جانبِ أولئكَ مُظلِمًا ومِنْ جانبِ هؤلاءِ يُنيرُ اللَّيلَ حتّى لا يقتربَ أحدُ الفريقَينِ مِنَ الآخرِ طُولَ اللَّيلِ.</span>
<span className="v" id="v21" data-v="21" data-xref="إشَعْياء 63:12; الخُرُوج 15:8; المَزامِير 66:6"> ومَدَّ موسى يَدَهُ على البحرِ، فأرسلَ الرّبُّ على البحرِ ريحًا شرقيَّةً عاصفةً طُولَ اللَّيلِ حتّى أيبسَ ما بَينَ مِياههِ فاَنشقَّتِ المياهُ</span>
<span className="v" id="v22" data-v="22" data-xref="العِبرانِيّين 11:29; المَزامِير 66:6; الخُرُوج 15:19"> ودخَلَ بَنو إِسرائيلَ في وسَطِ البحرِ على الأرضِ اليابسةِ. والماءُ لهُم سورٌ عَنْ يَمينِهم وعَنْ يسارِهِم.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 15:19; الخُرُوج 14:17; الخُرُوج 15:9"> وتَبِعَهُمُ المِصْريُّونَ بجميعِ خَيلِ فِرعَونَ ومركباتِهِ وفُرسانهِ ودخلوا إلى وسَطِ البحرِ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 13:21; المَزامِير 77:16–19; الخُرُوج 14:25"> وأشرفَ الرّبُّ عِندَ طُلوعِ الصُّبحِ على جيشِ المِصْريِّينَ مِنْ عَمودِ النَّارِ والسَّحابِ، فأوقعَ الفوضَى في صُفوفِهم</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 14:14; التَّثْنِيَة 3:22; أيُّوب 20:24"> وعطَّلَ دواليبَ مركباتِهِم فساقُوها بصعوبةٍ، فقالوا: ”نهربُ مِنْ أمامِ بَني إِسرائيلَ، لأنَّ الرّبَّ يُقاتلُنا عَنهُم“.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 14:16; الخُرُوج 8:5; يَعقُوب 2:13"> فقالَ الرّبُّ لموسى: ”مُدَّ يَدَكَ على البحرِ فيرتدَّ الماءُ على المِصْريِّينَ وعلى مركباتِهِم وفرسانِهِم“.</span>
<span className="v" id="v27" data-v="27" data-xref="يَشُوع 4:18; المَزامِير 78:53; العِبرانِيّين 11:29"> فمَدَّ موسى يدَهُ على البحرِ، فاَرتدَّ البحرُ عِندَ طُلوعِ الصُّبحِ إلى ما كانَ علَيهِ. فواجَهَهُ المِصْريُّونَ وهُم هاربونَ. فطرحَهُمُ الرّبُّ في وسَطِ البحرِ.</span>
<span className="v" id="v28" data-v="28" data-xref="المَزامِير 78:53; المَزامِير 106:9–11; حَبَقُّوق 3:13"> ورجَعتِ المِياهُ فغطَّتِ المركباتِ والفُرسانَ وجميعَ جيشِ فِرعَونَ الّذينَ دخلوا وراءَ بَني إسرائيلَ في البحرِ، وما بقيَ مِنهُم أحدٌ.</span>
<span className="v" id="v29" data-v="29" data-xref="الخُرُوج 14:22; المَزامِير 77:19–20; إشَعْياء 43:2"> ومشى بَنو إِسرائيلَ على الأرضِ اليابسةِ في وسَطِ البحرِ، والماءُ لهُم سورٌ عَنْ يَمينِهِم وعَنْ شمالِهِم.</span>
<span className="v" id="v30" data-v="30" data-xref="المَزامِير 106:8; المَزامِير 106:10; المَزامِير 92:9–11"> وخلَّصَ الرّبُّ في ذلِكَ اليومِ بَني إِسرائيلَ مِنْ أيدي المِصْريِّينَ، ورأى بَنو إِسرائيلَ المِصْريِّينَ أمواتًا على شاطئِ البحرِ.</span>
<span className="v" id="v31" data-v="31" data-xref="يُوحَنّا 2:11; يُوحَنّا 11:45; يُوحَنّا 2:23–25"> وشاهدَ بَنو إِسرائيلَ الفِعْلَ العظيمَ الّذي فعَلَهُ الرّبُّ بالمِصْريِّينَ، فخافوا الرّبَّ وآمنوا بهِ وبموسى عبدِهِ.</span>
</div>
