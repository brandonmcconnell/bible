## تدشين خيمة الإجتماع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ الرّبُّ لموسى:</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 26:30; الخُرُوج 13:4; العَدَد 7:1"> ”في اليومِ الأوَّلِ مِنَ الشَّهرِ الأوَّلِ تنصُبُ مَسكِنَ خَيمَةِ الإجتماعِ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 35:12; العَدَد 4:5; رُؤيا يُوحَنّا 11:19"> ضَعْ فيهِ تابوتَ العَهدِ واَسْتُرِ التَّابوتَ بالحِجابِ.</span>
<span className="v" id="v4" data-v="4" data-xref="اللّاويِّين 24:8; الخُرُوج 40:22–25; الخُرُوج 37:10–24"> وتُدخِلُ المائِدَةَ وتُرَتِّبُ علَيها أوانيَها. وتُدخِلُ المَنارَةَ وعلَيها سُرُجُها.</span>
<span className="v" id="v5" data-v="5" data-xref="العِبرانِيّين 10:19–22; العِبرانِيّين 9:24; رسالَةُ يُوحَنّا الأُولَى 2:1–2"> وتضَعُ مذبَحَ الذَّهبِ لِلبَخورِ أمامَ تابوتِ العَهدِ. وتُعَلِّقُ سِتارَةَ بابِ المَسكِنِ.</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 40:29; رسالَةُ يُوحَنّا الأُولَى 4:9–10; العِبرانِيّين 13:10"> وتَضعُ مذبَحَ المُحرَقاتِ أمامَ بابِ مَسكِنِ خَيمةِ الإجتماعِ،</span>
<span className="v" id="v7" data-v="7" data-xref="زَكَريّا 13:1; رسالَةُ يُوحَنّا الأُولَى 1:7; المَزامِير 26:6"> وضَعِ المَغسلَةَ بَينَ غِطاءِ الخَيمةِ والمذبَحِ على أنْ يكونَ فيها ماءٌ.</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 40:33; أفَسُس 4:11–12; ﻣﺘﻰ 16:18"> واَنْصُبْ رِواقَ المَسكِنِ حَولَ هذا كُلِّهِ، وتُعَلِّقُ سِتارَةً لِبابِ الرِّواقِ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="اللّاويِّين 8:10; العَدَد 7:1; ﻣﺘﻰ 3:16"> ”وتأخُذُ زيتَ المَسْحِ وتَمسَحُ المَسكِنَ وجميعَ ما فيهِ وتُكَرِّسُهُ هوَ وجميعَ أثاثِهِ فيصير مُقَدَّسًا.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 29:36–37; لُوقا 1:35; إشَعْياء 61:1"> وتمسَحُ مذبَحَ المُحرَقاتِ وجميعَ أدَواتِه وتُكَرِّسُ المذبَحَ، فيكون مُقَدَّسًا كُلَّ التَّقديسِ.</span>
<span className="v" id="v11" data-v="11"> وتَمسَحُ المَغسلَةَ ومَقعَدَها وتُكَرِّسُهُما.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="اللّاويِّين 8:1–13; اللّاويِّين 9:1–24; الخُرُوج 29:1–35"> ”وتُقَدِّمُ هرونَ وبَنيهِ إلى بابِ خَيمةِ الإجتماعِ وتَغسِلُهُم بِالماءِ،</span>
<span className="v" id="v13" data-v="13" data-xref="الخُرُوج 28:41; رسالَةُ يُوحَنّا الأُولَى 2:20; رسالَةُ يُوحَنّا الأُولَى 2:27"> وتُلبِسُ هرونَ الثِّيابَ المُقَدَّسَةَ وتمسَحُهُ وتُكَرِّسُهُ ليكونَ لي كاهنًا.</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 61:10; ﺭﻭﻣﻴﺔ 8:30; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:30"> وتُقَدِّمُ بَنيهِ وتُلبِسُهُم قُمصانًا</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 25:13; العِبرانِيّين 5:1–14; العِبرانِيّين 8:1–10"> وتمسَحُهُم كما مَسَحْتَ أباهُم ليكونوا لي كهَنَةً، وليكونَ لهُم هذا المَسْحُ كهنوتًا أبديًّا مدَى أجيالِهم“.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="التَّثْنِيَة 4:1; الخُرُوج 40:17–32; التَّثْنِيَة 12:32"> فعَمِلَ موسى بِجميعِ ما أمرَهُ الرّبُّ بهِ.</span>
<span className="v" id="v17" data-v="17" data-xref="العَدَد 7:1; الخُرُوج 40:1–2; العَدَد 9:1"> فلمَّا جاءَ اليومُ الأوَّلُ مِنَ الشَّهرِ الأوَّلِ في السَّنَةِ التَّاليَةِ نَصَبَ موسى المَسكِنَ.</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 37:27–28; رسالَةُ بُطرُسَ الأُولَى 1:5; اللّاويِّين 26:11"> فوضَعَ قواعِدَهُ وَرَكَّبَ علَيها ألواحَهُ وعَوارِضَهُ وأقامَ أعمِدَتَهُ.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 26:1–14; الخُرُوج 36:8–19"> ثُمَّ مَدَّ الخَيمةَ فَوقَ المَسكِنِ ووضَعَ الغِطاءَ علَيهِ مِنْ فَوقُ، كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 16:34; التَّثْنِيَة 10:5; العِبرانِيّين 10:19–21"> وأخذَ لوحَي الوصايا فوَضَعَهُما في تابوتِ العَهدِ وأدخَلَ القضيبَينِ في حلَقَاتِهِ وألقى علَيهِ غِطاءَهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="الخُرُوج 26:33; الخُرُوج 40:3; الخُرُوج 35:12"> ثُمَّ حَملَ تابوتَ العَهدِ إلى المَسكِنِ وعَلَّقَ الحِجابَ وسَتَرَهُ، كما أمرَهُ الرّبُّ،</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 26:35; يُوحَنّا 6:53–57; أفَسُس 3:8"> ووضَعَ المائِدَةَ في جانِبِ خَيمةِ الإجتماعِ إلى جِهةِ الشَّمالِ في خارِجِ الحِجابِ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 25:30; الخُرُوج 40:4; العِبرانِيّين 9:2"> ورَتَّبَ علَيها خُبزَ التَّقدِمَةِ أمامَ الرّبِّ، كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 26:35; رُؤيا يُوحَنّا 2:5; يُوحَنّا 1:1"> ووضَعَ المَنارَةَ في جانِبِ خَيمةِ الإجتماعِ إلى جِهَةِ الجَنوبِ،</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 40:4; الخُرُوج 25:37; رُؤيا يُوحَنّا 4:5"> وأصعَدَ السُّرُجَ أمامَ الرّبِّ، كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 40:5; العِبرانِيّين 7:25; يُوحَنّا 11:42"> ثُمَّ وضَعَ مذبَحَ الذَّهَبِ في خَيمةِ الإجتماعِ أمامَ الحِجابِ</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 30:7"> وبَخَّرَ علَيهِ بِبَخورٍ عَطِرٍ، كما أمرَهُ الرّبُّ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 40:5; العِبرانِيّين 10:19–20; أفَسُس 2:18"> وعَلَّقَ سِتارَةَ البابِ على الخَيمةِ،</span>
<span className="v" id="v29" data-v="29" data-xref="الخُرُوج 40:6; العِبرانِيّين 13:5–6; العِبرانِيّين 13:10"> ووضَعَ مذبَحَ المُحرَقَةِ عِندَ بابِ خَيمةِ الإجتماعِ وأصعَدَ علَيهِ المُحرَقَةَ والتَّقدِمَةَ، كما أمرَهُ الرّبُّ.</span>
</div>
<div className="bp">
<span className="v" id="v30" data-v="30" data-xref="الخُرُوج 40:7; الخُرُوج 38:8; حِزْقِيال 36:25"> ووضَعَ موسى المَغسلَةَ بَينَ خَيمَةِ الإجتماعِ والمذبَحِ وجعَلَ فيها ماءً لِلغَسْلِ،</span>
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 30:19–20; المَزامِير 26:6; يُوحَنّا 13:10"> لِيَغسِلَ مِنهُ موسى وهرونُ وبَنوهُ أيديَهُم وأرجُلَهُم</span>
<span className="v" id="v32" data-v="32" data-xref="المَزامِير 73:19; الخُرُوج 30:19–20; الخُرُوج 40:19"> عِندَ دُخولِهِم خَيمةَ الإجتماعِ وعِندَ اَقتِرابِهِم إلى المذبَحِ، كما أمرَ الرّبُّ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v33" data-v="33" data-xref="الخُرُوج 40:8; الخُرُوج 27:9–16; يُوحَنّا 17:4"> ونَصَبَ موسى الرِّواقَ حَولَ الخَيمةِ والمذبَحِ وعَلَّقَ سِتارَةَ بابِ الرِّواقِ وأكمَلَ العمَلَ كُلَّه.</span>
</div>
</div>

## السحاب فوق الخيمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v34" data-v="34" data-xref="رُؤيا يُوحَنّا 15:8; العَدَد 9:15–23; أخبارِ الأيّامِ الثّانِي 7:2"> ثُمَّ غَطَّى السَّحابُ خَيمةَ الإجتماعِ ومَلأَ مَجدُ الرّبِّ المَسكِنَ،</span>
<span className="v" id="v35" data-v="35" data-xref="المُلُوكِ الأوَّلُ 8:11; اللّاويِّين 16:2; أخبارِ الأيّامِ الثّانِي 7:2"> فلم يقدِرْ موسى أنْ يدخُلَ إليهِ.</span>
<span className="v" id="v36" data-v="36" data-xref="نَحَمْيا 9:19; العَدَد 9:17; الخُرُوج 13:21–22"> وكانَ إذا اَرْتَفَعَ السَّحابُ عَنِ المَسكِنِ يُتابِعُ بَنو إِسرائيلَ سَفَرَهُم،</span>
<span className="v" id="v37" data-v="37" data-xref="العَدَد 9:19–22; المَزامِير 31:15"> وإلَّا لَزِموا مكانَهُم إلى أنْ يرتَفِعَ.</span>
</div>
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="المَزامِير 78:14; العَدَد 9:15; الخُرُوج 13:21"> فسَحابُ الرّبِّ كانَ على المَسكنِ نهارًا، وكانَتِ النَّارُ في السَّحابِ ليلًا أمامَ عُيونِ بَني إِسرائيلَ في جميعِ مَراحِلِ سَفَرِهِم.</span>
</div>
