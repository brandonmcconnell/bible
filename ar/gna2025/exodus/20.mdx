## الوصايا العشر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّثْنِيَة 5:22; التَّثْنِيَة 5:4; التَّثْنِيَة 4:36"> وتكَلَّمَ الرّبُّ فقالَ:</span>
<span className="v" id="v2" data-v="2" data-xref="التَّثْنِيَة 5:6; هُوشَع 13:4; اللّاويِّين 26:13"> ”أنا الرّبُّ إلهُكَ الّذي أخرَجَكَ مِنْ أرض مِصْرَ، مِنْ دارِ العُبوديَّةِ.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 5:7; ﻣﺘﻰ 4:10; المَزامِير 81:9"> لا يَكُنْ لكَ آلهةٌ سِوايَ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 5:8; اللّاويِّين 26:1; التَّثْنِيَة 4:15–19"> لا تصنَعْ لكَ تِمثَالًا مَنحوتًا ولا صورَةَ شيءٍ مِمَّا في السَّماءِ مِنْ فَوقُ، ولا مِمَّا في الأرضِ مِنْ تحتُ، ولا مِمَّا في المياهِ مِنْ تَحتِ الأرضِ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 5:9; العَدَد 14:18; التَّثْنِيَة 4:24"> لا تسجُدْ لها ولا تَعبُدْها، لأنِّي أنا الرّبُّ إلهُكَ إلهٌ غيورٌ أعاقِبُ ذنوبَ الآباءِ في الأبناءِ إلى الجيلِ الثَّالثِ والرَّابعِ مِمَّنْ يُبغِضونَني،</span>
<span className="v" id="v6" data-v="6" data-xref="التَّثْنِيَة 7:9; يُوحَنّا 14:21; رسالَةُ يُوحَنّا الأُولَى 5:3"> وأرحَمُ إلى ألوفِ الأجيالِ مَنْ يُحِبُّونَني ويعمَلونَ بوصايايَ.</span>
</div>
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="التَّثْنِيَة 5:11; اللّاويِّين 19:12; يَعقُوب 5:12"> لا تَحلِفْ باَسْمِ الرّبِّ إلهِكَ باطلًا، لأنَّ الرّبَّ لا يُبرِّرُ مَنْ يَحلِفُ باَسْمِهِ باطِلًا.</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 26:2; اللّاويِّين 19:3; الخُرُوج 31:13–16"> أُذكُرْ يومَ السَّبتِ وكَرِّسْهُ لي.</span>
<span className="v" id="v9" data-v="9" data-xref="لُوقا 13:14; الخُرُوج 34:21; الخُرُوج 23:12"> في ستَّةِ أيّامٍ تعملُ وتُنجِزُ جميعَ أعمالِكَ،</span>
<span className="v" id="v10" data-v="10" data-xref="التَّثْنِيَة 5:14–15; الخُرُوج 34:21; العَدَد 15:32–36"> واليومُ السَّابِعُ سَبتٌ للرّبِّ إلهكَ. لا تَقُمْ فيهِ بعمَلٍ ما، أنتَ واَبنُكَ واَبنَتُكَ وعبدُكَ وجاريَتُكَ وبَهيمَتُكَ ونزيلُكَ الّذي في داخلِ أبوابِكَ،</span>
<span className="v" id="v11" data-v="11" data-xref="التَّكوين 1:1–2; التَّكوين 2:2–3; الخُرُوج 31:17"> لأنَّ الرّبَّ في ستَّةِ أيّامٍ خلَقَ السَّماواتِ والأرضَ والبحرَ وجميعَ ما فيها، وفي اليومِ السَّابعِ اَستراحَ. ولذلِكَ بارَكَ الرّبُّ يومَ السَّبتِ وكَرَّسَهُ لَه.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="أفَسُس 6:1–3; كُولُوسِي 3:20; التَّثْنِيَة 5:16"> أكرِمْ أبَاكَ وأُمَّكَ لِيَطولَ عُمرُكَ في الأرضِ الّتي يُعطيكَ الرّبُّ إلهُكَ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="ﺭﻭﻣﻴﺔ 13:9; التَّثْنِيَة 5:17; التَّكوين 9:5–6"> لا تقتُلْ.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="العِبرانِيّين 13:4; التَّثْنِيَة 5:18; ﺭﻭﻣﻴﺔ 7:2–3"> لا تزْنِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="ﺭﻭﻣﻴﺔ 13:9; أفَسُس 4:28; اللّاويِّين 19:11"> لا تسرِقْ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الأمثال 19:5; ﻣﺘﻰ 19:18; المَزامِير 15:3"> لا تشهَدْ على غيرِكَ شَهادَةَ زُورٍ.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="ﺭﻭﻣﻴﺔ 13:9; لُوقا 12:15; العِبرانِيّين 13:5"> لا تشتَهِ بيتَ غيرِكَ. لا تشتَهِ اَمرَأةَ غيرِكَ ولا عبدَهُ ولا جاريَتَهُ ولا ثَورَهُ ولا حِمارَهُ ولا شيئًا مِمَّا لهُ“.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 19:16–18; إرْمِيا 23:23; المَزامِير 139:7–8"> وكانَ جميعُ الشَّعبِ يُشاهِدونَ الرُّعودَ والبُروقَ وصوتَ البوقِ والجبَلَ يصعَدُ مِنهُ الدُّخانُ، فاَضْطَرَبوا ووَقَفوا على بُعدٍ.</span>
<span className="v" id="v19" data-v="19" data-xref="التَّثْنِيَة 18:16; اعمال الرسل 7:38; غَلاطِيَّة 3:19"> وقالوا لِموسى: ”كَلِّمْنا أنتَ فنَسمَعَ، ولا يُكَلِّمْنا اللهُ لِئلَّا نموتَ“.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 13:3; الأمثال 3:7; صَمُوئيلَ الأوَّلُ 12:20"> فقالَ لهُم موسى: ”لا تخافوا. جاءَ اللهُ لِيَمْتَحِنَكُم ولِتَكونَ مخافَتُهُ نُصْبَ عُيونِكُم لِئلَّا تَخطَأوا“.</span>
<span className="v" id="v21" data-v="21" data-xref="المُلُوكِ الأوَّلُ 8:12; المَزامِير 97:2; المَزامِير 18:9"> فوقفَ الشَّعبُ على بُعدٍ، وتقدَّمَ موسى إلى الضَّبَابِ الّذي كانَ فيهِ اللهُ.</span>
</div>

## بناء المذبح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="نَحَمْيا 9:13; التَّثْنِيَة 4:36; العِبرانِيّين 12:25–26"> فقالَ الرّبُّ لموسى: ”قُلْ هذا لبَني إِسرائيلَ: شاهَدتُمْ أنِّي مِنَ السَّماءِ خاطَبتُكُم.</span>
<span className="v" id="v23" data-v="23" data-xref="حِزْقِيال 20:39; صَفَنْيا 1:5; المُلُوكِ الثّانِي 17:33"> لا تصنَعوا لَكُم آلهَةً مِنْ فِضَّةٍ أو ذهَبٍ لِتَعبُدوها معَ عِبادَتِكُم لي.</span>
<span className="v" id="v24" data-v="24" data-xref="التَّثْنِيَة 12:5; أخبارِ الأيّامِ الثّانِي 6:6; المَزامِير 132:13–14"> إبنوا لي مذبَحًا مِنْ تُرابٍ. واَذْبَحُوا علَيهِ مُحرَقاتِكُم وذبائِحَ السَّلامَةِ مِنْ غنَمِكُم وبقَرِكُم. وأنا إلى كُلِّ مَوضِعٍ يُذكَرُ فيهِ اَسْمي أجيءُ وأُبارِكُكُم.</span>
<span className="v" id="v25" data-v="25" data-xref="يَشُوع 8:31; التَّثْنِيَة 27:5–6"> وإنْ بَنَيتُم لي مذبَحًا مِنْ حِجارَةٍ فلا تَنحَتوها، لأنَّكُم إنْ رفَعتُم إزميلًا علَيها دَنَّستُموهَا.</span>
<span className="v" id="v26" data-v="26" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:16; الجامِعَة 5:1; اللّاويِّين 10:3"> ولا تصعَدوا إلى مذبَحي على دَرَجٍ لِئلَّا تَنكَشِفَ عَورَتُكُم علَيهِ“.</span>
