## التبرّع لصنع المقدس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1"> وقالَ الرّبُّ لِموسى:</span>
<span className="v" id="v2" data-v="2" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 9:7; نَحَمْيا 11:2; عَزْرا 2:68"> ”قُلْ لِبَني إِسرائيلَ أنْ يُقدِّموا لي تقدِمَةً، كُلُّ واحدٍ وما تَسخو بِهِ نفْسُهُ.</span>
<span className="v" id="v3" data-v="3" data-xref="التَّثْنِيَة 8:9; أيُّوب 28:2"> وهذِهِ هيَ التَّقدِمَةُ الّتي تقبَلُها مِنهُم: ذهَبٌ وفِضَّةٌ ونُحاسٌ،</span>
<span className="v" id="v4" data-v="4" data-xref="رُؤيا يُوحَنّا 19:8; التَّكوين 41:42; حِزْقِيال 16:10"> ونسيجٌ بَنَفْسَجيٌّ وأُرجوانيٌّ وقِرْمِزيُّ اللَّونِ وكتَّانٌ وشَعْرُ مَعَزٍ</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 26:37–1; الخُرُوج 26:14–15; الخُرُوج 36:20"> وجُلودُ كِباشٍ مَصبوغةٌ بالحُمْرةِ وجُلودٌ بَنَفْسَجيّةُ اللَّونِ وخشَبُ سَنْطٍ،</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 27:20; الخُرُوج 25:37; الخُرُوج 40:24–25"> وزيتٌ للإنارَةِ وأطيابٌ لِزيتِ المِسْحِ والبَخورِ العَطِرِ،</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 28:4; الخُرُوج 28:6–30"> وحِجارَةُ عَقيقٍ وحِجارَةُ ترصيعٍ لِلأُفودِ والصُّدْرَةِ،</span>
<span className="v" id="v8" data-v="8" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 6:16; المُلُوكِ الأوَّلُ 6:13; رُؤيا يُوحَنّا 21:3"> فيَصنعونَ لي مَسكِنًا مُقدَّسًا لأسكُنَ فيما بَينَهُم</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 25:40; العِبرانِيّين 8:5; اعمال الرسل 7:44"> ويكونُ المَسكِنُ وجميعُ أثاثِهِ على المِثالِ الّذي أنا أُريكَ.</span>
</div>
</div>

## المسكن المقدس وآثاثه: التابوت

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="العِبرانِيّين 9:4; الخُرُوج 37:1–3; التَّثْنِيَة 10:1–3"> ”تصنَعُ تابوتًا مِنْ خشَبِ السَّنْطِ، طولُهُ ذِراعانِ ونِصفٌ وعَرضُهُ ذِراعٌ ونِصفٌ وسُمكُهُ ذِراعٌ ونِصفٌ.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 25:24; الخُرُوج 30:3; أخبارِ الأيّامِ الثّانِي 3:4"> فتُغَشِّيهِ بِذهَبٍ خالصٍ مِنْ داخلٍ ومِنْ خارجٍ وتَصنعُ علَيهِ حِليَةً مِنْ ذهَبٍ تُحيطُ بهِ.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 27:7; الخُرُوج 25:26; الخُرُوج 25:15"> وتصوغُ لَه أربَعَ حَلَقَاتٍ مِنْ ذهَبٍ وتجعَلُها على أربَعِ قوائمِهِ، حَلَقَتينِ مِنْ جانِبِه الواحدِ وحَلَقَتينِ مِنْ جانِبِه الآخَرِ.</span>
<span className="v" id="v13" data-v="13" data-xref="أخبارِ الأيّامِ الأوّلُ 15:15; العَدَد 4:11; الخُرُوج 27:6"> وتصنعُ قضيبَينِ مِنْ خَشَبِ السَّنْطِ وتُغشِّيهِما بِذهَبٍ،</span>
<span className="v" id="v14" data-v="14"> وتُدخِلُ القضيبَينِ في الحَلَقاتِ على جانِبَي التَّابُوتِ لِيُحمَلَ بِهِما.</span>
<span className="v" id="v15" data-v="15" data-xref="المُلُوكِ الأوَّلُ 8:8; أخبارِ الأيّامِ الثّانِي 5:9"> ويبقى القضيبانِ في الحَلَقاتِ، لا يَزولانِ مِنها.</span>
<span className="v" id="v16" data-v="16" data-xref="التَّثْنِيَة 31:26; المُلُوكِ الأوَّلُ 8:9; العِبرانِيّين 9:4"> وتضَعُ في التَّابوتِ لوحَي الوصايا الّتي أُعطيكَ.</span>
<span className="v" id="v17" data-v="17" data-xref="العِبرانِيّين 9:5; الخُرُوج 37:6; العِبرانِيّين 4:16"> وتصنعُ غِطاءً للتَّابوتِ مِنْ ذهَبٍ خالصٍ، طُولُه ذِراعانِ ونِصفٌ وعَرضُهُ ذِراعٌ ونِصفٌ.</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 10:20; العِبرانِيّين 9:5; المُلُوكِ الأوَّلُ 8:6–7"> وتصنعُ على طرَفَي الغِطاءِ كَرُوبَينِ مِنْ ذهَبٍ مَطروقٍ،</span>
<span className="v" id="v19" data-v="19"> فيكونُ على كُلِّ طرَفٍ كَروبٌ مُثْبَتٌ كجُزءٍ مِنَ الغِطاءِ.</span>
<span className="v" id="v20" data-v="20" data-xref="أخبارِ الأيّامِ الأوّلُ 28:18; المُلُوكِ الأوَّلُ 8:7; العِبرانِيّين 9:5"> ويكونُ الكَروبانِ باسِطَينِ أجنِحَتَهُما إلى فوقُ على الغِطاءِ ووجهُ الواحدِ إلى الآخَرِ وإلى الغِطاءِ.</span>
<span className="v" id="v21" data-v="21" data-xref="الخُرُوج 26:34; ﺭﻭﻣﻴﺔ 10:4; الخُرُوج 25:16–17"> وتجعَلُ الغِطاءَ على التَّابوتِ مِنْ فَوقُ، وفي التَّابوتِ تضَعُ لوحَي الوصايا الّتي أُعطيكَ.</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 29:42–43; العَدَد 7:89; إشَعْياء 37:16"> فأجتَمِعُ بِكَ هُناكَ وأُكَلِّمُكَ مِنْ فَوقِ الغِطاءِ، مِنْ بَينِ الكَروبَينِ اللّذَينِ على تابوتِ العَهدِ، بِجميعِ ما أوصيكَ بهِ إلى بَني إِسرائيلَ.</span>
</div>
</div>

## مائدة خبز التقدمة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v23" data-v="23" data-xref="العِبرانِيّين 9:2; الخُرُوج 37:10–16; المُلُوكِ الأوَّلُ 7:48"> ”وتصنعُ مائدَةً مِنْ خشَبِ السَّنطِ، طولُها ذِراعانِ وعَرضُها ذِراعٌ وسُمكُها ذِراعٌ ونِصفٌ.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 25:11; المُلُوكِ الأوَّلُ 6:20–22"> وتُغَشِّيها بِذهَبٍ خالصٍ، وتصوغُ لها حِليَةً مِنْ ذهَبٍ تُحيطُ بِها.</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 37:2; الخُرُوج 30:3"> وتصنعُ لها حافَةً مِقدارَ شِبرٍ مِنْ حَولِها وحِليَةً مِنْ ذهَبٍ على مُحيطِها.</span>
<span className="v" id="v26" data-v="26"> وتصوغُ لها أربَعَ حَلَقَاتٍ مِنْ ذهَبٍ لِزوايا قوائمِها الأربَعِ،</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 25:14; الخُرُوج 25:28"> على أنْ تكونَ الحَلَقاتُ عِندَ الحافَةِ مكانًا لِقَضيبَينِ لِحَملِ المائِدَةِ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 25:14; اعمال الرسل 9:15; العَدَد 10:17"> وتصنعُ القضيبَينِ مِنْ خَشَبِ السَّنطِ وتُغَشِّيهِما بِذهَبٍ.</span>
<span className="v" id="v29" data-v="29" data-xref="العَدَد 4:7; الخُرُوج 37:16; رُؤيا يُوحَنّا 3:20"> وتصنعُ لِلمائدَةِ صُحونًا ومَجامِرَ وكُؤوسًا وجاماتٍ لِلسَّكْبِ مِنْ ذهَبٍ خالصٍ.</span>
<span className="v" id="v30" data-v="30" data-xref="اللّاويِّين 24:5–9; ﻣﺘﻰ 12:4; الخُرُوج 39:36"> وتُقَدِّمُ على المائدَةِ خُبزًا أمامي دائمًا.</span>
</div>
</div>

## المنارة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="المُلُوكِ الأوَّلُ 7:49; زَكَريّا 4:2; الخُرُوج 37:17–24"> ”وتصنعُ مَنارَةً مَطروقَةً مِنْ ذهَبٍ خالصٍ، هيَ وقاعِدَتُها وساقُها. وتكونُ كاساتُها، بِما في ذلِكَ عُقَدُها وأوراقُهَا قِطعَةً واحدةً مِنها.</span>
<span className="v" id="v32" data-v="32"> ويتَفَرَّعُ مِنْ جانبَي المَنارَةِ سِتُّ شُعَبٍ، ثلاثٌ مِنْ جانِبِها الواحدِ وثلاثٌ مِنْ جانِبِها الآخَرِ.</span>
<span className="v" id="v33" data-v="33" data-xref="العَدَد 17:4–8; إرْمِيا 1:11–12; زَكَريّا 4:3"> ويكونُ لِكُلِّ شُعبَةٍ مِنَ الشُّعَبِ السِتِّ ثَلاثُ كاساتٍ على شكلِ زهرَةِ اللوزِ بِعُقدَتِها وأوراقِها.</span>
</div>
<div className="bp">
<span className="v" id="v34" data-v="34"> ”وفي عَمودِ المَنارَةِ ذاتِها أربَعُ كاساتٍ على شكلِ زهرَةِ اللَّوزِ بِعُقدَتِها وأوراقِها،</span>
<span className="v" id="v35" data-v="35"> وعُقدَةٌ تَحتَ كُلِّ شُعبَتَينِ مِنَ الشُّعَبِ السِتِّ المُتَفَرِّعَةِ مِنْ عَمودِ المَنارةِ.</span>
</div>
<div className="bp">
<span className="v" id="v36" data-v="36" data-xref="الخُرُوج 25:18; المُلُوكِ الأوَّلُ 10:16–17; أخبارِ الأيّامِ الثّانِي 9:15"> ”وتكونُ العُقَدُ والشُّعَبُ كُلُّها قِطعَةً واحدةً مَطروقَةً مِنْ ذهَبٍ خالصٍ.</span>
<span className="v" id="v37" data-v="37" data-xref="العَدَد 8:2; رُؤيا يُوحَنّا 1:20–1; اللّاويِّين 24:2–4"> وتصنعُ سُرُجَها سَبْعًا، وتجعَلُها علَيها لِتُضيءَ على جِهَةِ وجهِها.</span>
<span className="v" id="v38" data-v="38" data-xref="المُلُوكِ الثّانِي 12:13; إرْمِيا 52:18; المُلُوكِ الثّانِي 25:14"> وتكونُ مَلاقِطُها ومَنافِضُها مِنْ ذهَبٍ خالصٍ.</span>
<span className="v" id="v39" data-v="39"> مِنْ وَزْنَةِ ذهَبٍ خالصٍ تصنَعُ المِسْرَجةَ معَ جميعِ هذِهِ الأدَواتِ.</span>
<span className="v" id="v40" data-v="40" data-xref="العِبرانِيّين 8:5; الخُرُوج 26:30; اعمال الرسل 7:44"> واَحرِصْ على أنْ تصنعَها على مِثالِ ما أرَيتُكَ في الجبَلِ“.</span>
</div>
