## العدل ومعاملة العدو

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="المَزامِير 101:5; الأمثال 19:5; التَّثْنِيَة 19:16–21"> ”لا تنقلْ خَبرًا كاذِبًا، ولا تضَعْ يدَكَ بِيَدِ الشِّرِّيرِ لِشَهادَةِ زُورٍ.</span>
<span className="v" id="v2" data-v="2" data-xref="الأمثال 1:10–11; الأمثال 1:15; لُوقا 23:23–24"> لا تتبَعِ الكثرَةَ إلى السُّوءِ، ولا تُسَايِرْها في الدَّعاوي خِلافًا لِلحَقِّ،</span>
<span className="v" id="v3" data-v="3" data-xref="يَعقُوب 3:17; اللّاويِّين 19:15; المَزامِير 82:2–3"> ولا تهدُرْ حَقَّ المِسكينِ في دعواهُ.</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 22:1–4; الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 5:15; ﻣﺘﻰ 5:44"> إذا لَقِيتَ ثَورَ عَدُوِّكَ أو حمارَهُ شارِدًا، فَرُدَّهُ إليهِ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 22:4"> وإذا رأيتَ حمارَ مَنْ يُبغِضُكَ رازِحًا تحتَ حِمْلِهِ، فلا تمتَنِعْ عَن مُساعَدَتِهِ ورَفْعِ الحِمْلِ معَهُ.</span>
</div>
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="مَلاخِي 3:5; إرْمِيا 5:28; اللّاويِّين 19:15"> ”لا تسكُتْ عَن إنصافِ المِسكينِ في دعواهُ.</span>
<span className="v" id="v7" data-v="7" data-xref="أفَسُس 4:25; الخُرُوج 34:7; الخُرُوج 23:1"> اَبتَعِدْ عنِ الحُكْمِ في الدَّعاوي الكاذِبَةِ، ولا تحكُمْ بِالموتِ على البَريءِ والصِّدِّيقِ، ولا تُصدِّقِ الشِّرِّيرَ.</span>
<span className="v" id="v8" data-v="8" data-xref="التَّثْنِيَة 16:19; الأمثال 17:23; إشَعْياء 5:23"> لا تأخُذْ رَشوَةً، فالرَّشوَةُ تُعمي أبصارَ القُضاةِ وتُكَذِّبُ أقوالَ الصَّادِقينَ.</span>
</div>
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="التَّثْنِيَة 27:19; الخُرُوج 22:21; التَّثْنِيَة 24:14–18"> ”لا تُضايِقِ الغريبَ. فأنتُم تعرِفونَ حقيقةَ ما يشعُرُ بِه الغريبُ، لأنَّكُم كُنتُم غُرَباءَ في أرضِ مِصْرَ“.</span>
</div>
</div>

## السنة السابعة واليوم السابع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="اللّاويِّين 25:3–4"> ”إزرَعْ أرضَكَ واَجمَعْ غَلَّتها سِتَّ سنينَ،</span>
<span className="v" id="v11" data-v="11" data-xref="اللّاويِّين 26:34–35; اللّاويِّين 25:2–7; اللّاويِّين 25:11–12"> وفي السَّابِعَةِ اَترُكْهَا بُورًا، فيَأكُلَ مِنها بُؤساءُ شعبِكَ، وما فَضَلَ عَنهُم يأكُلُه وَحشُ البرِّيَّةِ. وكذلِكَ تفعَلُ بِكرمِكَ وزيتونِكَ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="لُوقا 13:14; الخُرُوج 34:21; الخُرُوج 35:3"> ”في سِتَّةِ أيّامٍ تعمَلُ عمَلكَ، وفي اليومِ السَّابعِ تستَريحُ، لِيَستريحَ ثَورُكَ وحِمارُكَ ويتَنَفَّسَ الصُّعَداءَ عبدُكَ والغريبُ الّذي يعمَلُ عِندَكَ.</span>
</div>
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="يَشُوع 23:7; زَكَريّا 13:2; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 4:16"> ”جميعُ ما قُلتُهُ لكُم اَعمَلوا بهِ، واَسمَ إلهٍ آخَرَ لا تذكُروا ولا يُسمَعْ مِنْ أفواهِكُم“.</span>
</div>

## الأعياد الكبرى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="التَّثْنِيَة 16:16; الخُرُوج 34:22–24; اللّاويِّين 23:34"> ”عَيِّدوا لي ثلاثَ مرَّاتٍ في السَّنةِ.</span>
<span className="v" id="v15" data-v="15" data-xref="لُوقا 22:7; الخُرُوج 34:20; العَدَد 28:16–25"> عَيِّدوا عيدَ الفطيرِ، وكُلوا فطيرًا سبْعَةَ أيّامٍ كما أمرْتُكُم، في وقتٍ مُعَيَّنٍ مِنْ شهرِ أبيبَ، لأنَّكُم فيهِ خرَجْتُم مِنْ مِصْرَ، ولا تحضُروا أمامي ولا شيءَ لي في أيديكُم</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 34:22; الخُرُوج 22:29; اللّاويِّين 23:9–21"> وعَيِّدوا عيدَ حَصادِ بَواكيرِ غَلَّاتِكُمُ الّتي تزرَعونَها في الحقلِ، وعيدَ جمْعِ غَلَّاتِكُم مِنَ الحقلِ عِندَ نِهايَةِ السَّنةِ.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّثْنِيَة 16:16; الخُرُوج 34:23; الخُرُوج 23:14"> وليَحضُرْ جميعُ الذُّكورِ هذِهِ الأعيادَ الثَّلاثةَ لِيَعبُدوني أنا الرّبُّ الإلهُ.</span>
</div>
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 34:25; اللّاويِّين 2:11; التَّثْنِيَة 16:4"> ”لا تَسفِكوا دَمَ ذبيحَتي عَشيّةَ الفِصْحِ ما دامَ الخَميرُ موجودًا، ولا تَحتَفِظوا بِشَحْمِها طُولَ اللَّيلِ إلى الصَّباحِ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 34:26; التَّثْنِيَة 14:21; التَّثْنِيَة 26:10"> إحمِلوا أوائِلَ بَواكيرِ أرضِكُم إلى بيتِ الرّبِّ إلهِكُم. لا تطبُخوا الجَديَ بِلَبَنِ أُمِّهِ“.</span>
</div>
</div>

## الإستعداد لدخول أرض كنعان

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 33:14; المَزامِير 91:11; الخُرُوج 32:34"> ”ها أنا سأُرسِلُ أمامَكُم ملاكًا يحفَظُكُم في الطَّريقِ ويَجيءُ بِكُم إلى المكانِ الّذي أعدَدْتُهُ.</span>
<span className="v" id="v21" data-v="21" data-xref="المَزامِير 78:56; المَزامِير 78:40; العِبرانِيّين 12:25"> فاَنْتَبِهوا لَه واَسْتَمِعوا إلى صوتِهِ ولا تَتَمَرَّدوا علَيهِ، لأنَّهُ لا يَصفَحُ عَن ذُنوبِكُم، لأنَّهُ يعمَلُ باَسْمي.</span>
<span className="v" id="v22" data-v="22" data-xref="التَّكوين 12:3; التَّثْنِيَة 30:7; زَكَريّا 2:8"> فإنِ اَسْتَمَعتُم إلى صوتِهِ وعَمِلتُم بِكُلِّ ما أتكلَّمُ بهِ، عادَيتُ مَنْ يُعاديكُم وضايَقْتُ مَنْ يُضايقُكُم.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 23:20; إشَعْياء 5:13; التَّكوين 15:19–21"> ويسيرُ ملاكي أمامَكُم ويُدخِلُكُم أرضَ الأموريِّينَ والحِثِّيِّينَ والفِرزِّيِّينَ والكنعانيِّينَ والحوِّيِّينَ واليَبوسيِّينَ جميعًا، بعدَ أنْ أُزيلَهُم.</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 20:5; التَّثْنِيَة 12:30–31; العَدَد 33:52"> لا تسجُدوا لآلِهَتِهِم ولا تعبُدوها. لا تعمَلوا كأعمالِهِم، بل أزيلوهُم وحَطِّموا أصنامَهُم.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 7:15; الخُرُوج 15:26; يَشُوع 22:5"> اَعْبُدوا الرّبَّ إلهَكُم، فيُبارِكَكُم في خُبزِكُم ومائِكُم ويرفَعَ الأمراضَ مِنْ بَينكُم.</span>
<span className="v" id="v26" data-v="26" data-xref="التَّثْنِيَة 7:14; أيُّوب 5:26; التَّثْنِيَة 28:4"> لا مُسقِطَ ولا عاقِرَ تكونُ في أرضِكُم، وعدَدُ أيّامِ حياتِكُم أُكْمِلُهُ.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="التَّثْنِيَة 2:25; التَّكوين 35:5; التَّثْنِيَة 11:25"> ”والرُّعبُ مِنّي أُرسِلُهُ أمامَكُم، وأهزِمُ جميعَ الأُمَمِ التي تُواجِهونَها، وأجعَلُ جميعَ أعدائِكُم يُولّونَ مُدْبِرينَ.</span>
<span className="v" id="v28" data-v="28" data-xref="التَّثْنِيَة 7:20; يَشُوع 24:11–12"> وأُرسِلُ الذُعْرَ أمامَكُم فتَطرُدونَ الحوِّيِّينَ والكنعَانيِّينَ والحِثِّيِّينَ مِنْ وُجوهِكُم.</span>
<span className="v" id="v29" data-v="29" data-xref="التَّثْنِيَة 7:22; يَشُوع 17:12–13; القُضاة 3:1–4"> لا أطرُدُهُم مِنْ وُجوهِكُم في سنَةٍ واحدةٍ لِئلَّا تصيرَ الأرضُ قَفْرًا، فتَكثُرَ علَيكُم وُحوشُ البرِّيَّةِ.</span>
<span className="v" id="v30" data-v="30"> ولكنِّي أطرُدُهُم قليلًا قليلًا مِنْ أمامِكُمْ إلى أنْ يَكثُرَ عدَدُكُم وتَملِكونَ الأرضَ.</span>
<span className="v" id="v31" data-v="31" data-xref="التَّكوين 15:18; التَّثْنِيَة 11:24; يَشُوع 21:44"> وأجعَلُ حُدودَ أرضِكُم مِنَ البحرِ الأحمرِ جَنوبًا إلى البحرِ المُتوسّطِ غَربًا، ومِنَ الصَّحراءِ شَرقًا إلى نهرِ الفُراتِ شَمالًا، وأُسلِمُ إلى أيديكُم سُكَّانَ الأرضِ فتَطرُدونَهُم مِنْ أمامِ وُجوهِكُم.</span>
<span className="v" id="v32" data-v="32" data-xref="التَّثْنِيَة 7:2; الخُرُوج 34:12; الخُرُوج 34:15"> لا تقطَعُوا لهُم ولا لآلِهَتِهِم عَهدًا.</span>
<span className="v" id="v33" data-v="33" data-xref="المَزامِير 106:36; التَّثْنِيَة 7:16; الخُرُوج 34:12"> ولا يُقيموا في أرضِكُم لِئلَّا يجعلوكُم تَخطَأونَ إليَّ، فتَعبُدونَ آلِهَتَهُم ويكُونُ ذلِكَ لَكُم شَرَكًا“.</span>
</div>
