## ثياب الكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="العَدَد 18:7; العِبرانِيّين 5:1–5; العَدَد 17:2–9"> ”وخُذْ مِنْ بَني إِسرائيلَ هرونَ أخاكَ وبَنيهِ نادابَ وأبيهو وألِعازَرَ وإيثامارَ لِيكونوا كهَنةً لي.</span>
<span className="v" id="v2" data-v="2" data-xref="اللّاويِّين 8:30; الخُرُوج 31:10; رُؤيا يُوحَنّا 19:8"> واَصْنَعْ ثيابًا مُقَدَّسَةً لِهرونَ أخيكَ، لِلكرامَةِ والجَلالِ.</span>
<span className="v" id="v3" data-v="3" data-xref="أفَسُس 1:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:7–11; إشَعْياء 11:2"> وقُلْ لِكُلِّ حكيمٍ مِمَّنْ مَلأْتُ قلوبَهُم بِروحِ الحِكمَةِ أنْ يصنَعوا ثيابَ هرونَ لِتكريسِهِ كاهنًا لي.</span>
<span className="v" id="v4" data-v="4" data-xref="أفَسُس 6:14; رُؤيا يُوحَنّا 9:17; إشَعْياء 59:17"> وهذِهِ هيَ الثِّيابُ الّتي يصنَعونَها: صُدْرَةٌ وأفُودٌ وجُبَّةٌ وقميصٌ مُطَرَّزٌ وعِمامَةٌ وحِزامٌ، فيصنَعونَ ثيابًا مُقَدَّسَةً لِهرونَ أخيكَ وبَنيه لِيكونوا كهَنةً لي.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 25:3–4; الخُرُوج 39:2–3"> ويستعمِلونَ في صُنْعِها الذَّهَبَ والنَّسيجَ البَنَفْسَجيَّ والأُرجُوانيَّ والقِرمِزيَّ اللَّونِ، والكتَّانَ.</span>
</div>

## الأفود

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 39:2–7; الخُرُوج 26:1"> ”يصنَعونَ الأَفودَ مِنْ ذهَبٍ ونسيجٍ بَنَفْسَجيٍّ وأُرجوانيٍّ وقِرمِزيِّ اللَّونِ وكتَّانٍ مَبرومٍ بِيَدِ نَسَّاجٍ ماهرٍ.</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 39:4"> ويكونُ لَه في طَرَفَيهِ كَتِفانِ مَوصولتانِ لِيَتَّصِلَ.</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 28:27–28; اللّاويِّين 8:7; الخُرُوج 29:5"> والزُنَّارُ الّذي يَشُدُّ الأَفودَ يكونُ مِثلَه، مِنْ ذهَبٍ ونسيجٍ بَنَفْسَجيٍّ وأُرجُوانيٍّ وقِرمِزيِّ اللَّونِ وكتَّانٍ مَبرومٍ.</span>
<span className="v" id="v9" data-v="9" data-xref="حِزْقِيال 28:13; أخبارِ الأيّامِ الثّانِي 2:7; إشَعْياء 49:16"> وخُذْ حجَرَي عَقيقٍ يَمانيٍّ واَنْقُشْ علَيهِما أسماءَ أسباطِ بَني إِسرائيلَ،</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 1:1–4; التَّكوين 43:33"> سِتَّةٌ مِنها على الحجَرِ الواحدِ والسِّتَّةُ الأسماءِ الباقيَةُ على الحجَرِ الآخَرِ حسَبَ مَواليدِهِم.</span>
<span className="v" id="v11" data-v="11" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:19; رُؤيا يُوحَنّا 7:2; إرْمِيا 22:24"> بِمَهارَةِ الصَّائِغِ ونَقَّاشِ الخَواتِمِ تَنقُشُ الحجَرَينِ على حسَبِ أسماءِ أسباط بَني إِسرائيلَ وتُحيطُهما بِطَوقٍ مِنْ ذهَبٍ</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 28:29; يَشُوع 4:7; الخُرُوج 39:6–7"> وتضَعُهما على كَتفَي الأَفودِ، ذِكْرًا لِأسباطِ بَني إِسرائيلَ، فيَحمِلُ هرونُ أسماءَهُم أمامَ الرّبِّ على كَتفَيه.</span>
<span className="v" id="v13" data-v="13"> وتصنعُ طَوقَينِ مِنْ ذهَبٍ،</span>
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 39:15; الخُرُوج 39:17–18; المُلُوكِ الأوَّلُ 7:17"> وسِلسِلتَينِ مِنْ ذهَبٍ خالصٍ، مَجدولتَينِ جَدْلَ الضَّفائِرِ، وتُعَلِّقُهما بالطَّوقَينِ.</span>
</div>
</div>

## صدرة القضاء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 28:30; الخُرُوج 26:1; الخُرُوج 28:4"> ”وتصنعُ صُدْرَةَ قضاءٍ بِيَدِ نَسَّاجٍ ماهِرٍ كَصُنْعَةِ الأَفودِ، مِنْ ذهَبٍ تصنعُها ومِنْ نسيجٍ بَنَفْسَجيٍّ وَأُرجُوانيٍّ وقِرمِزيِّ اللَّونِ وكتَّانٍ مَبرومٍ.</span>
<span className="v" id="v16" data-v="16"> تكونُ مُرَبَّعةً مَثْنِيَّةً، طُولُها شِبْرٌ وعَرْضُها شِبْرٌ.</span>
<span className="v" id="v17" data-v="17" data-xref="حِزْقِيال 28:13; الخُرُوج 39:10–21; الأمثال 8:11"> وتُرَصِّعُ فيها أربعَةَ صُفوفٍ مِنَ الحِجارَةِ الكريمةِ. الصَّفُّ الأوَّلُ عَقيقٌ أحمَرُ وياقُوتٌ أصفَرُ وزُمُرُّدٌ.</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 27:16; أيُّوب 28:6; نَشِيدِ الأنشاد 5:14"> والصَّفُّ الثَّاني بَهْرَمانٌ وياقوتٌ أزرقُ وعَقيقٌ أبيضُ.</span>
<span className="v" id="v19" data-v="19" data-xref="إشَعْياء 54:12; الخُرُوج 39:12"> والصَّفُّ الثَّالِثُ ياقوتٌ زَعْفَرانيٌّ وياقوتٌ أحمَرُ وجَمَشْتٌ.</span>
<span className="v" id="v20" data-v="20" data-xref="حِزْقِيال 1:16; رُؤيا يُوحَنّا 4:3; دانيال 10:6"> والصَّفُّ الرَّابعُ زَبَرْجَدٌ وعَقيقٌ يَمانيٌّ ويَشْبٌ. وتكونُ الحجارةُ مُحاطَةً بِذهَبٍ في ترصيعِها.</span>
<span className="v" id="v21" data-v="21" data-xref="رُؤيا يُوحَنّا 21:12; رُؤيا يُوحَنّا 7:4–8; المُلُوكِ الأوَّلُ 18:31"> وهيَ بِعَدَدِ أسماءِ أسباطِ بَني إِسرائيلَ، إثني عشَرَ حجَرًا، يُنقَشُ على كُلِّ حجَرٍ اَسمُهُ كَنَقْشِ الخاتَمِ.</span>
<span className="v" id="v22" data-v="22"> وتصنعُ لِلصُّدْرَةِ سَلاسِلَ مَجدُولَةً جَدْلَ الضَّفَائِرِ مِنَ الذَّهَبِ الخالصِ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 25:11–15"> وتصنعُ لها حلقَتَينِ مِنْ ذهَبٍ في طَرَفَيهَا،</span>
<span className="v" id="v24" data-v="24"> وتُدخِلُ سِلسِلتَيِّ الذَّهَبِ في الحلقَتَينِ</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 39:15; الخُرُوج 39:4; الخُرُوج 28:14"> وتُعَلِّقُ طَرَفَيهِما بِالطَّوقَينِ وتجعَلُهُما على كَتفَي الأَفودِ في مُقَدِّمِهِ.</span>
<span className="v" id="v26" data-v="26"> وتصنعُ حلَقَتَينِ مِنْ ذهَبٍ في طَرَفَي الصُّدْرَةِ في حاشيَتِها الّتي إلى جِهَةِ الأَفودِ مِنْ داخلٍ.</span>
<span className="v" id="v27" data-v="27"> وتصنعُ حلَقَتَينِ مِنْ ذهَبٍ على كَتِفَي الأَفودِ مِنْ أسفَلُ في مُقَدِّمِهِ عِندَ وَصْلَتِهِ فَوقَ زُنَّارِ الأَفودِ.</span>
<span className="v" id="v28" data-v="28" data-xref="العَدَد 15:38; الخُرُوج 28:31; الخُرُوج 39:30–31"> وتحْبِكُ الصُّدْرَةَ مِنْ حلَقَتَيها إلى حلَقَتَي الأَفودِ بِخيطٍ مِنْ نسيجٍ بنَفْسَجيِّ اللَّونِ حتّى تصيرَ على زُنَّارِ الأَفودِ ولا تنفَصِلُ عَنِ الأَفودِ.</span>
<span className="v" id="v29" data-v="29" data-xref="الخُرُوج 28:12; الخُرُوج 28:30; إرْمِيا 30:21"> فيَحمِلُ هرونُ أسماءَ أسباطِ بَني إِسرائيلَ في صُدْرَةِ القضاءِ على صَدرِهِ، عِندَ دُخولِه القُدْسَ، ذِكْرًا أمامَ الرّبِّ دائمًا.</span>
<span className="v" id="v30" data-v="30" data-xref="نَحَمْيا 7:65; عَزْرا 2:63; العَدَد 27:21"> وتجعَلُ على صُدرَةِ القضاءِ الأُوريمَ والتُّمّيمَ، فتكونُ على صَدْرِ هرونَ عِندَ دُخولِه أمامَ الربِّ. ويحمِلُ هرونُ على صَدرِه أمامَ الرّبِّ دائمًا ما قضى بهِ الرّبُّ على أسباطِ بَني إِسرائيلَ.</span>
</div>

## الجبة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="اللّاويِّين 8:7; الخُرُوج 28:4; الخُرُوج 39:22–26"> ”وتصنعُ جُبَّةَ الأفَودِ كُلَّها مِنْ نسيجٍ بَنَفْسَجيِّ اللَّونِ.</span>
<span className="v" id="v32" data-v="32" data-xref="نَحَمْيا 4:16; يُوحَنّا 19:23–24; الخُرُوج 39:28"> ويكونُ لها في وسَطِها فَتحَةٌ لِلرأسِ تُحيطُ بِها حاشيةٌ تُحَاكُ كَفتحَةِ الدِّرعِ فلا تتَمزَّقُ.</span>
<span className="v" id="v33" data-v="33" data-xref="الخُرُوج 39:24–26; زَكَريّا 14:20; المُلُوكِ الأوَّلُ 7:18"> وتصنعُ لِأذيالِها رُمَّاناتٍ مِنْ نسيجٍ بنَفْسَجيٍّ وأُرجُوانيٍّ وقِرمِزيِّ اللَّونِ وأجراسًا مِنْ ذهَبٍ فيما بَينَ الرُّمَّاناتِ مِنْ حَولِها:</span>
<span className="v" id="v34" data-v="34" data-xref="نَشِيدِ الأنشاد 8:2; يُوحَنّا 15:4–8; كُولُوسِي 1:10"> جَرَسُ ذهَبٍ وَرُمَّانَةٌ بَعدَه مِنْ أوَّلِ أذيالِ الجُبَّةِ إلى آخِرِها.</span>
<span className="v" id="v35" data-v="35" data-xref="العِبرانِيّين 9:12; اللّاويِّين 16:2"> فيلبَسُها هرونُ عِندَ الخِدمَةِ لِيَسمعَ النَّاسُ صوتَ أجراسِها عِندَ دُخولِهِ المَقْدِسَ أمامَ اللهِ وعِندَ خُروجِهِ مِنهُ لِئلَّا يموتَ.</span>

<div className="bp">

<span className="v" id="v37" data-v="37" data-xref="الخُرُوج 29:6; الخُرُوج 28:4; زَكَريّا 3:5"> وتصنعُهُ على خيطٍ مِنْ نسيجٍ بَنَفْسجيِّ اللَّونِ، فيكونُ على العِمامَةِ مِنْ مُقَدِّمِها.</span>
<span className="v" id="v38" data-v="38" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:24; العَدَد 18:1; اللّاويِّين 10:17"> ويكونُ على جَبهةِ هرونَ، فيَحمِلُ هرونُ عاقِبَةَ أيِّ قُصورٍ في تقديمِ القرابينِ التي يُكَرِّسُها لي بَنو إِسرائيلَ مِنْ عَطاياهُمُ المُقَدَّسَةِ. ويبقى التَّاجُ المُقَدَّسُ على جَبهتِهِ عَلامَةَ الرِّضى عَنهُم أمامَ اللهِ.</span>
<span className="v" id="v39" data-v="39" data-xref="المَزامِير 45:14"> وتصنعُ القميصَ المُخَرَّمَ مِنْ كَتَّانٍ، والعِمامَةَ مِنْ كَتَّانٍ، والحِزامَ تصنعُهُ بِيَدِ مُطرِّزٍ.</span>

## ثياب الكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v40" data-v="40" data-xref="الخُرُوج 39:41; الخُرُوج 28:4; الخُرُوج 28:2"> ”ولِبَني هرونَ تصنعُ قُمصانًا وأحزِمَةً وقلانِسَ لِلمَهابَةِ والجَلالِ.</span>
<span className="v" id="v41" data-v="41" data-xref="الخُرُوج 29:7; اللّاويِّين 8:1–36; الخُرُوج 40:15"> وهذِهِ يلبَسُها هرونُ وبَنوهُ معَهُ، وتمسَحُهُم وتُقَلِّدُهُم وظيفَتَهُم وتُكَرِّسُهُم لِيكونوا كهَنةً لي.</span>
<span className="v" id="v42" data-v="42" data-xref="حِزْقِيال 44:18; اللّاويِّين 6:10; اللّاويِّين 16:4"> وتصنعُ لهُم سَراويلَ مِنَ الكَتَّانِ لِتُغَطِّيَ العُرْيَ مِنْ أبدانِهِم ما بَينَ الحَقْوَينِ والفَخْذَينِ.</span>
<span className="v" id="v43" data-v="43" data-xref="الخُرُوج 27:21; الخُرُوج 20:26; اللّاويِّين 5:17"> ويلبَسُ هرونُ وبَنوهُ هذِهِ السَّراويلَ، عِندَ دُخولِهِم خَيمَةَ الإجتماعِ وعِندَ اَقتِرابِهِم مِنَ المذبَحِ لِيَخدِموا في القُدْسِ، لِئلَّا يحمِلوا عاقِبَةَ أيِّ عُرْيٍ في أبدانِهِم فيَموتوا. هذِهِ فريضةٌ لِهرونَ ولِنسلِهِ مِنْ بَعدِهِ إلى الأبدِ“.</span>
