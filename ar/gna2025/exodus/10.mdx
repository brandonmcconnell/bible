## الضربة الثامنة: الجراد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 4:21; الخُرُوج 7:13–14; الخُرُوج 3:20"> وقالَ الرّبُّ لموسى: ”أُدخُلْ على فِرعَونَ، فأنا الّذي قَسَّى قلبَهُ وقلوبَ رِجالِهِ لأصنَعَ مُعجِزاتي هذِهِ بَينَهُم،</span>
<span className="v" id="v2" data-v="2" data-xref="يُوئيل 1:3; التَّثْنِيَة 4:9; التَّثْنِيَة 6:20–22"> ولِتَرويَ على مَسمَعِ اَبنِكَ واَبنِ اَبنِكَ ما فعَلْتُ بالمِصْريِّينَ، وما صنعْتُ بَينَهُم مِنَ المُعجزاتِ، وتعلَمُوا أنِّي أنا الرّبُّ“.</span>
</div>
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:6; المُلُوكِ الأوَّلُ 21:29; يَعقُوب 4:10"> فدخَلَ موسى وهرونُ على فِرعَونَ وقالا لَه: ”يقولُ لكَ الرّبُّ إلهُ العِبرانيِّينَ إلى متى ترفُضُ أنْ تخضَعَ لي؟ أطلِقْ شعبي ليَعبُدوني،</span>
<span className="v" id="v4" data-v="4" data-xref="رُؤيا يُوحَنّا 9:3; يُوئيل 2:25; الخُرُوج 11:4–5"> وإنْ رفَضْتَ أنْ تُطلِقَهُم، أجلِبُ الجَرادَ غدًا على أرضِكَ.</span>
<span className="v" id="v5" data-v="5" data-xref="يُوئيل 1:4; الخُرُوج 9:32; يُوئيل 2:25"> فيُغطِّيَ الجَرادُ وجهَ الأرضِ حتّى لا يقدِرَ أحدٌ أنْ يراها، ويأكُلَ البقيَّةَ الَّتي سَلِمَت مِنَ البرَدِ وجميعَ الشَّجَرِ النَّابتِ لكُم في البرِّيَّةِ.</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 8:21; الخُرُوج 8:3; الخُرُوج 11:6"> ويَملأَ الجَرادُ بُيوتَكَ وبُيوتَ جميعِ رِجالِكَ وجميعِ المِصْريِّينَ. فلا آباؤُكَ ولا آباءُ آبائِكَ رَأوا مِثلَهُ مُنذُ يومَ وُجِدوا على الأرضِ إلى هذا اليومِ“. وأدارَ موسى ظهرَهُ وخرَجَ مِنْ عِندِ فِرعَونَ.</span>
<span className="v" id="v7" data-v="7" data-xref="صَمُوئيلَ الأوَّلُ 18:21; الخُرُوج 23:33; يَشُوع 23:13"> فقالَ رِجالُ فِرعَونَ لسيِّدِهِم: ”إلى متى نَقعُ في حبائِلِ هذا الرَّجُلِ؟ أطلِقْ هؤلاءِ القومَ مِنْ مِصْرَ ليعبُدوا الرّبَّ إلهَهُم. أما علِمْتَ بَعدُ أنَّ مِصْرَ خَرِبَت؟“</span>
</div>
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 10:24; الخُرُوج 12:31; الخُرُوج 8:8"> فأُعيدَ موسى وهرونُ إلى فِرعَونَ فقالَ لهُما: ”اَذهَبوا اَعبُدُوا الرّبَّ إلهَكُم. ولكِنْ مَنْ هُمُ الّذينَ يذهَبونَ؟“</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 5:1; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 5:7–8; التَّثْنِيَة 31:12–13"> فأجَابَهُ موسى: ”نذهَبُ بشَبابِنا وشُيوخِنا وبَنينا وبَناتِنا وغنَمِنا وبقَرِنا، لأنَّنا سنُعَيِّدُ للرّبِّ“.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 13:21; أخبارِ الأيّامِ الثّانِي 32:15; الخُرُوج 12:30–31"> فقالَ لهُما: ”أُقسِمُ بالرّبِّ لا أُطلِقُكُم معَ عيالِكُم. فأنا أرى الشَّرَّ على وجوهِكُم.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 10:28; المَزامِير 119:69; الخُرُوج 5:4"> لا، فليَذهَبِ الرِّجالُ مِنكُم وحدَهُم، فيَعبُدونَ الرّبَّ كما تَطلُبونَ“. وطرَدَهُما فِرعَونُ مِنْ أمامِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 7:19; الخُرُوج 10:4–5"> فقالَ الرّبُّ لموسى: ”مُدَّ يَدَكَ على أرضِ مِصْرَ لِيَجيءَ الجَرادُ ويصعَدَ علَيها ويأكُلَ جميعَ عُشْبِ الأرضِ وكُلَّ ما تركَهُ البرَدُ“.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻳﻮﻧﺎﻥ 1:4; المَزامِير 78:26; المَزامِير 148:8"> فمَدَّ موسى عصاهُ على أرضِ مِصْرَ. وسيَّرَ الرّبُّ ريحًا شرقيَّةً على الأرضِ طولَ ذلِكَ النَّهارِ وطولَ اللَّيلِ. وعِندَ الصُّبحِ حمَلَتِ الرِّيحُ الشرقيَّةُ الجَرادَ،</span>
<span className="v" id="v14" data-v="14" data-xref="المَزامِير 78:46; يُوئيل 1:2–4; المَزامِير 105:34–35"> فصعِدَ على جميعِ أرضِ مِصْرَ ونزلَ فيها بأعدادٍ كثيرةٍ لا مَثيلَ لها مِنْ قَبلُ ولن يكونَ لها مَثيلٌ مِنْ بَعدُ.</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 10:5; المَزامِير 78:46; يُوئيل 1:6–7"> فغطَّى كُلَّ وجهِ الأرضِ حتى أظلَمَتِ الدُّنيا. وأكَلَ الجَرادُ جميعَ عُشْبِها وكُلَّ ما تَرَكهُ البرَدُ مِنْ ثَمرِ الشَّجَرِ، فما بقيَ شيءٌ مِنَ الخُضرَةِ في الشَّجَرِ ولا في عُشْبِ البرِّيَّةِ في جميعِ أرضِ مِصْرَ.</span>
</div>
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 9:27; صموئيلَ الثّانِي 19:20; صَمُوئيلَ الأوَّلُ 15:30"> فسارعَ فِرعَونُ ودعا موسى وهرونَ وقالَ لهُما: ”خَطِئتُ إلى الرّبِّ إلهِكُما وإليكُما.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 8:8; الخُرُوج 9:28; المُلُوكِ الأوَّلُ 13:6"> فاَصْفَحا عَنْ خطيئَتي لِآخرِ مرَّةٍ، وتشفَّعا إلى الرّبِّ إلهِكُما أنْ يرفَعَ عنِّي هذِهِ الضَرَباتِ المُميتَةَ“.</span>
<span className="v" id="v18" data-v="18" data-xref="لُوقا 6:28; الخُرُوج 8:28–30; الخُرُوج 8:9"> فخَرجَ موسى مِنْ عِندِ فِرعَونَ وتشفَّعَ فيهِ إلى الرّبِّ،</span>
<span className="v" id="v19" data-v="19" data-xref="يُوئيل 2:20; الخُرُوج 13:18; الخُرُوج 15:4"> فَردَّ الرّبُّ ريحًا غربيَّةً عاصفةً، فحَملتِ الجرادَ وطرحَتْهُ في البحرِ الأحمرِ، فما بقِيَت جَرادةٌ واحدةٌ في كُلِّ أرضِ مِصْرَ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 4:21; الخُرُوج 11:10; التَّثْنِيَة 2:30"> وقَسَّى الرّبُّ قلبَ فِرعَونَ، فلمْ يُطلِقْ بَني إِسرائيلَ مِنْ تلكَ الأرضِ.</span>
</div>
</div>

## الضربة التاسعة: الظلام

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="رُؤيا يُوحَنّا 16:10–11; يَهُوذا 1:6; لُوقا 23:44"> وقالَ الرّبُّ لموسى: ”مُدَّ يَدَكَ نحوَ السَّماءِ فيكونَ على أرضِ مِصْرَ ظلامٌ كثيفٌ تَلمَسُهُ اليَدُ“.</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 105:28; رُؤيا يُوحَنّا 16:10; الخُرُوج 20:21"> فمَدَّ موسى يدَهُ نحوَ السَّماءِ، فخيَّمَ ظلامٌ حالِكٌ على جميعِ أرضِ مِصْرَ ثلاثةَ أيّامٍ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 8:22; إشَعْياء 42:16; رسالَةُ بُطرُسَ الأُولَى 2:9"> فما كانَ الواحدُ يُبصِرُ الآخرَ، ولا كان أحدٌ يقومُ مِنْ مكانِهِ ثلاثةَ أيّامٍ. أمَّا بَنو إسرائيلَ فكانَ لهُم نورٌ في مَساكِنِهم.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 10:8–10; الخُرُوج 8:28; التَّكوين 34:23"> فاَستَدعى فِرعَونُ موسى وقالَ لَه: ”إذهبوا اَعبُدوا الرّبَّ، أنتُم ونِساؤُكُم وأطفالُكُم، وأمَّا غنَمُكُم وبقَرُكُم فتَتركونَها هُنا في مِصْرَ“.</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 9:22; اللّاويِّين 16:9; الخُرُوج 29:1–46"> فقالَ لَه موسى: ”إذًا، فأعطِنا ذبائِحَ ومُحرَقاتٍ نُقَدِّمُها للرّبِّ إلهنا.</span>
<span className="v" id="v26" data-v="26" data-xref="إشَعْياء 23:18; اعمال الرسل 2:44–45; الأمثال 3:9"> كلَّا. لا نرضَى إلَّا أنْ تذهَبَ كُلُّ مَواشينا معَنا. لا يبقى مِنها هنا رأسٌ واحدٌ لِأنَّنا مِنْها نأخُذُ ما نعبُدُ بهِ الرّبَّ إلهَنا. ونحنُ لا نعرِفُ أيًّا نختارُ مِنها لنَعبُدَ الرّبَّ إلهَنا حتّى نصِلَ إلى هُناكَ“.</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 10:20; الخُرُوج 4:21; الخُرُوج 14:8"> وقسَّى الرّبُّ قلبَ فِرعَونَ. فأبى أنْ يُطلِقَ بَني إِسرائيلَ.</span>
</div>
<div className="bp">
<span className="v" id="v28" data-v="28" data-xref="أخبارِ الأيّامِ الثّانِي 25:16; أخبارِ الأيّامِ الثّانِي 16:10; عامُوس 7:13"> وقالَ فِرعَونُ لموسى: ”إذهَبْ عنِّي! إيَّاكَ أنْ تنظُرَ إلى وجهي بَعدَ اليومِ. فيومَ تنظُرُ إلى وجهي تموتُ“.</span>
<span className="v" id="v29" data-v="29" data-xref="العِبرانِيّين 11:27; الخُرُوج 12:30–31; الخُرُوج 11:4–8"> فقالَ لَه موسى: ”حسَنًا قُلتَ. لن أرى وجهَكَ بَعدَ اليومِ“.</span>
</div>
</div>
