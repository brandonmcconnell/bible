## تقديس هرون وبنيه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="اللّاويِّين 4:3; مَلاخِي 1:13–14; الخُرُوج 12:5"> ”وهذا ما تصنعُه لِهرونَ وبَنيهِ حتّى تُكَرِّسَهُم كهَنةً لي. تأخُذُ عِجْلًا مِنَ البقَرِ وكَبشَينِ صحيحَينِ</span>
<span className="v" id="v2" data-v="2" data-xref="العَدَد 6:15; اللّاويِّين 2:15; اللّاويِّين 7:12"> وخُبزَ فطيرٍ وأقراصَ فطيرٍ مَلتوتَةً بِزيتٍ، ورِقاقَ فطيرٍ مَدهونَةً بِزيتٍ، على أنْ تكونَ مِنْ دقيقِ الحِنطَةِ.</span>
<span className="v" id="v3" data-v="3" data-xref="العَدَد 6:17; اللّاويِّين 8:26; اللّاويِّين 8:2"> وتضَعُ هذِهِ كُلَّها في سَلَّةٍ وقدِّمْها في السَّلَّةِ معَ العِجْل والكَبشَينِ.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 40:12; العِبرانِيّين 10:22; الخُرُوج 26:36"> ”وَتُقدِّمُ هرونَ وبَنيهِ إلى بابِ خَيمةِ الإجتماعِ وتَغسِلُهُم بالماءِ،</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 28:2–8; اللّاويِّين 8:7–8"> وتأخُذُ الثِّيابَ وتُلبِسُ هرونَ القميصَ والجُبَّةَ والأَفودَ والصُّدْرَةَ، وتَشُدُّهُ بِزُنَّارِ الأفودِ،</span>
<span className="v" id="v6" data-v="6" data-xref="اللّاويِّين 8:9; الخُرُوج 28:36–39"> وتضَعُ العِمامَةَ على رأسِه والتَّاجَ المُقَدَّسَ على العِمامَةِ.</span>
<span className="v" id="v7" data-v="7" data-xref="العَدَد 35:25; اللّاويِّين 21:10; يُوحَنّا 3:34"> وتأخُذُ مِنْ زيتِ المِسْحِ وتصُبُّ على رأسِهِ وتمسَحُهُ.</span>
<span className="v" id="v8" data-v="8" data-xref="اللّاويِّين 8:13; الخُرُوج 28:40"> وتُقَدِّمُ بَنيهِ وتُلبِسُهُم قُمصانًا،</span>
<span className="v" id="v9" data-v="9" data-xref="العَدَد 18:7; اللّاويِّين 8:22–28; التَّثْنِيَة 18:5"> وتشُدُّهُم بِالأحزِمَةِ وتُلبِسُهُم قلانِسَ، فيكونُ لهُمُ الكهَنوتُ فريضَةً إلى الأبدِ. هكذا تُكَرِّسُ لي هرونَ وبَنيهِ.</span>
</div>
</div>

## الذبائح

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="اللّاويِّين 1:4; اللّاويِّين 8:14; الخُرُوج 29:15"> ”وتُقَدِّمُ العِجْلَ أمامَ خَيمةِ الإجتماعِ، فيضَعُ هرونُ وبَنوهُ أيديَهُم على رأسِه.</span>
<span className="v" id="v11" data-v="11" data-xref="اللّاويِّين 9:8; اللّاويِّين 9:12; الخُرُوج 29:4"> واَذبَحِ العِجْلَ أمامَ الرّبِّ عِندَ بابِ خَيمةِ الإجتماعِ.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 27:2; اللّاويِّين 9:9; اللّاويِّين 8:15"> وتأخُذُ مِنْ دَمِ العِجْل وترُشُّ على قُرونِ المذبَحِ بِإصبَعِكَ، وتصُبُّ سائِرَ الدَّمِ على أسفلِ المذبَحِ.</span>
<span className="v" id="v13" data-v="13" data-xref="اللّاويِّين 3:3–4; صَمُوئيلَ الأوَّلُ 2:16; اللّاويِّين 9:10"> وتأخُذُ جميعَ الشَّحْمِ الّذي يُغَطِّي الأمعاءَ وزِياداتِ الكَبِدِ والكليَتَينِ والشَّحْمِ الّذي علَيهِما وتحرُقُها على المذبَحِ.</span>
<span className="v" id="v14" data-v="14" data-xref="اللّاويِّين 4:11–12; اللّاويِّين 4:21; الخُرُوج 30:10"> وأمَّا لَحمُ العِجْلِ وجِلدُهُ ورَوثُهُ، فتَحرُقُها بالنَّارِ في خارجِ المَحلَّةِ، لأنَّ هذِهِ ذبيحَةُ تكفيرٍ عَنْ خطيئَةِ الكهَنةِ.</span>
</div>
<div className="bp">
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 29:19; الخُرُوج 29:3; اللّاويِّين 8:18–21"> ”وتأخُذُ أحدَ الكَبشَينِ، فيضَعُ هرونُ وبَنوهُ أيديَهم على رأْسِهِ،</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 29:11–12"> وتذبَحُه وترُشُّ دَمَهُ على المذبَحِ مِنْ كُلِّ جِهَةٍ.</span>
<span className="v" id="v17" data-v="17" data-xref="إرْمِيا 4:14; اللّاويِّين 1:13; اللّاويِّين 1:9"> وتُقَطِّعُ الكَبشَ قِطَعًا وتَغسِلُ جَوفَهُ وأكارِعَه وتُضيفُها إلى قِطَعِهِ ورأسِهِ.</span>
<span className="v" id="v18" data-v="18" data-xref="التَّكوين 8:21; فِيلِبِّي 4:18; التَّكوين 22:7"> وتحرُقُ الكَبشَ كُلَّه على المذبَحِ، لأنَّ هذِهِ مُحرَقَةٌ تُرضي رائِحَتُها الرّبَّ وذبيحَةٌ تُحرَقُ لهُ.</span>
</div>
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 29:3; اللّاويِّين 8:22–29"> ”وتأخذُ الكَبشَ الآخَرَ، فيضَعُ هرونُ وبَنوهُ أيديَهم على رأسِهِ،</span>
<span className="v" id="v20" data-v="20" data-xref="إشَعْياء 50:5; العِبرانِيّين 10:22; اللّاويِّين 14:16"> وتذبَحُهُ وتأخُذُ مِنْ دَمِهِ وتضَعُ على شَحْمَةِ أُذُنِ هرونَ، وعلى شَحماتِ آذانِ بَنيهِ اليُمنى، وعلى أباهيمِ أيديهِم وأرجُلِهِم اليُمنى وترُشُّ الدَّمَ على المذبَحِ مِنْ كُلِّ جِهَةٍ.</span>
<span className="v" id="v21" data-v="21" data-xref="العِبرانِيّين 9:22; الخُرُوج 29:7; الخُرُوج 29:1"> وتأخُذُ مِنَ الدَّمِ الّذي على المذبَحِ، ومِنْ زيتِ المَسْحِ، وترُشُّ على هرونَ وثيابِهِ، وعلى بَنيهِ وثيابِهِم، فيتقَدَّس هوَ وثيابُهُ وبَنوهُ وثيابُهُم.</span>
</div>
</div>

## تكريس الكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 29:13; اللّاويِّين 9:21; اللّاويِّين 9:19"> ”وتأخُذُ مِنَ الكَبْشِ الشَّحْمَ والأَليَةَ والشَّحمَ الّذي يُغَطِّي الأَمعاءَ، وزياداتِ الكَبِدِ والكِليَتَينِ والشَّحمَ الّذي علَيهِما، والكَتِفَ اليُمنى، لأنَّهُ كَبْشُ تكريسِ الكهَنةِ،</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 29:2–3; اللّاويِّين 8:26"> ورغيفًا واحدًا مِنَ الخُبزِ، وقُرْصًا واحدًا مِنَ الخُبزِ بِزيتٍ، ورِقاقَةً واحدةً مِنْ سَلَّةِ الفطيرِ الّتي أمامَ الرّبِّ.</span>
<span className="v" id="v24" data-v="24" data-xref="اللّاويِّين 7:30; اللّاويِّين 8:27; اللّاويِّين 9:21"> وتضَعُ كُلَّ هذا على كَفَّيْ هرونَ وعلى أكُفِّ بَنيهِ وحَرِّكْهُ تكرارًا أمامَ الرّبِّ.</span>
<span className="v" id="v25" data-v="25" data-xref="اللّاويِّين 8:28; الخُرُوج 29:41; اللّاويِّين 3:16"> ثُمَّ تأخُذُهُ مِنْ أيديهِم، وترُشُّهُ على المذبَحِ فَوقَ المُحرَقَةِ، فتُرضي رَائحَتُهُ الرّبَّ ويكونُ ذبيحةً تُحرَقُ لِلرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v26" data-v="26" data-xref="اللّاويِّين 8:29; اللّاويِّين 7:31; اللّاويِّين 7:34"> ”وتأخذُ الصَّدْرَ مِنْ كَبْشِ التَّكريسِ الّذي لِهرونَ وحَرِّكْهُ تكرارًا أمامَ الرّبِّ وهوَ يكونُ لكَ نصيبًا.</span>
<span className="v" id="v27" data-v="27" data-xref="اللّاويِّين 10:15; العَدَد 18:11; التَّثْنِيَة 18:3"> وتُكَرِّسُ لي صَدْرَ التَّحريكِ هذا والفَخْذَ الّتي رفَعْتَها جانبًا مِنْ كَبْشِ التَّكرِيسِ الّذي لِهرونَ وبَنيهِ.</span>
<span className="v" id="v28" data-v="28" data-xref="العَدَد 15:19–20; اللّاويِّين 7:11–38; اللّاويِّين 10:14–15"> فيكونانِ لِهرونَ وبَنيهِ فريضةً أبديَّةً مِنْ بَني إِسرائيلَ، يَرفعانِها مِنْ ذبائحِ السَّلامةِ الّتي يُقَرِّبونَها لِلرّبِّ.</span>
</div>
<div className="bp">
<span className="v" id="v29" data-v="29" data-xref="العَدَد 18:8; الخُرُوج 28:3–4; الخُرُوج 30:30"> ”والثِّيابُ المُقَدَّسَةُ الّتي لِهرونَ تكونُ لبَنيهِ مِنْ بَعدِهِ، يُمسَحونَ فيها ويُكرَّسونَ.</span>
<span className="v" id="v30" data-v="30" data-xref="العَدَد 20:28; اللّاويِّين 13:5; اعمال الرسل 20:6–7"> ويَلبَسُها سَبْعَةَ أيّامٍ كُلُّ مَنْ يخلِفُ هرونَ مِنْ بَنيهِ، ويدخُلُ خَيمةَ الإجتماعِ لِيَخدُمَ في القُدْسِ.</span>
</div>
</div>

## الطعام المقدس

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="اللّاويِّين 8:31; صَمُوئيلَ الأوَّلُ 2:13; حِزْقِيال 46:20–24"> ”أمَّا كَبْشُ التَّكريسِ فتأخذُهُ وتطبُخُ لَحمَه في مكانٍ مُقَدَّسٍ.</span>
<span className="v" id="v32" data-v="32" data-xref="ﻣﺘﻰ 12:4; الخُرُوج 29:23; الخُرُوج 29:2–3"> فيأكُل هرونُ وبَنوهُ لَحمَ الكَبْشِ والخُبزَ الّذي في السَّلَّةِ عِندَ بابِ خَيمةِ الإجتماعِ.</span>
<span className="v" id="v33" data-v="33" data-xref="العَدَد 18:4; اللّاويِّين 10:13–18; المَزامِير 22:26"> يأكلونَ هذا الّذي كانَ كَفَّارَةً لهُم لِتَكريسِهِم وتقديسِهِم، ولا يأكُلُ مِنهُ أحدٌ مِنْ عامَةِ الشَّعبِ لأنَّهُ مُقَدَّسٌ لي.</span>
<span className="v" id="v34" data-v="34" data-xref="الخُرُوج 12:10; اللّاويِّين 8:32; الخُرُوج 29:22"> وإنْ بقيَ شيءٌ مِنْ لَحمِ التَّكريسِ أو مِنَ الخُبزِ إلى الصَّباحِ يُحرَقُ بالنَّارِ. لا يُؤكَلُ لأنَّهُ مُقَدَّسٌ.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="الخُرُوج 29:30; اللّاويِّين 14:8–11; اللّاويِّين 8:4–36"> ”تصنعُ هذا كُلَّه لِهرونَ وبَنيهِ كما أمَرْتُكَ بهِ. في سَبْعةِ أيّامِ تكريسِهِم.</span>
</div>
</div>

## الذبيحة اليومية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v38" data-v="38" data-xref="أخبارِ الأيّامِ الثّانِي 2:4; أخبارِ الأيّامِ الأوّلُ 16:40; العَدَد 28:3–8"> ”وهذا ما تُقرِّبُهُ على المذبَحِ: خروفانِ اَبنا سنَةٍ في كُلِّ يومٍ مِنَ الأيّامِ بغيرِ اَنقطاعٍ.</span>
<span className="v" id="v39" data-v="39" data-xref="حِزْقِيال 46:13–15; المُلُوكِ الثّانِي 16:15; المَزامِير 5:3"> أحدُهُما تُقرِّبُه في الصَّباحِ، والآخَرُ تُقرِّبُه في العَشيَّةِ.</span>
<span className="v" id="v40" data-v="40" data-xref="فِيلِبِّي 2:17; حِزْقِيال 46:14; يُوئيل 2:14"> وتُقرِّبُ معَ الخروفِ الواحدِ عُشْرَ قُفَةٍ مِنَ الدَّقيقِ مَلتوتًا بِرُبعِ هَينٍ مِنْ زيتِ زيتونٍ مَعصورٍ ورُبعَ هَينٍ مِنْ سكيبِ الخَمرِ.</span>
<span className="v" id="v41" data-v="41" data-xref="حِزْقِيال 46:13–15; المُلُوكِ الأوَّلُ 18:29; المَزامِير 141:2"> وتُقرِّبُ الخروفَ الآخَرَ في العَشيَّةِ وتصنعُهُ كتقدِمَةِ الصَّباحِ وكسكيبِ خَمرِها، فَتُرضي رَائِحَتُهُ الرّبَّ ويكونُ ذبيحَةً تُحرَقُ للرّبِّ.</span>
<span className="v" id="v42" data-v="42" data-xref="الخُرُوج 25:22"> ويكونُ ذلِكَ مُحرَقَةً دائمةً مدَى أجيالِكُم، عِندَ بابِ خَيمةِ الإجتماعِ أمامَ الرّبِّ، حَيثُ أجتمعُ بكَ لأكلِّمَكَ.</span>
<span className="v" id="v43" data-v="43" data-xref="المُلُوكِ الأوَّلُ 8:11; أخبارِ الأيّامِ الثّانِي 7:1–3; مَلاخِي 3:1"> هُناكَ أجتمعُ بِبَني إِسرائيلَ فيتقدَّسُ المكانُ بِمَجدي.</span>
