## موسى يصنع المعجزات

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 3:18; الخُرُوج 2:14; إرْمِيا 1:6"> فقالَ موسى لِلرّبِّ: ”هُم لا يُصَدِّقونَني ولا يسمَعُونَ لِكلامي، بل يقولونَ: لم يظهَرْ لكَ الرّبُّ“.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 4:17; الخُرُوج 4:20; إشَعْياء 11:4"> فأجابَهُ الرّبُّ: ”ما هذِهِ الّتي في يَدِكَ؟“ قالَ: ”عصًا“.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 7:10–15; عامُوس 5:19; الخُرُوج 4:17"> قالَ: ”ألقِها على الأرضِ“. فألقاها على الأرضِ فصارَت حَيَّةً. فهَربَ موسى مِنْ وجهِها.</span>
<span className="v" id="v4" data-v="4" data-xref="اعمال الرسل 28:3–6; التَّكوين 22:1–2; المَزامِير 91:13"> فقالَ لَه الرّبُّ: ”مُدَّ يَدَكَ وأمسِكْ ذنَبَها“. فمَدَّ موسى يَدَهُ فأمسَكَها، فعادَت عصًا في يَدِهِ.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 19:9; التَّكوين 48:3; الخُرُوج 3:15"> وقالَ لَه الرّبُّ: ”تفعَلُ هذِهِ المُعجِزَةَ ليُصَدِّقوا أنَّ الرّبَّ ظَهرَ لكَ، وهوَ إلهُ آبائِهم، إلهُ إبراهيمَ وإسحَقَ ويعقوبَ“.</span>
<span className="v" id="v6" data-v="6" data-xref="المُلُوكِ الثّانِي 5:27; العَدَد 12:10"> وقالَ لَه الرّبُّ أيضًا: ”أدخِلْ يَدَكَ في جَيبِكَ“. فأدخَلَ يَدَهُ في جَيبِهِ ثُمَّ أخرَجَها، فإذا هيَ بَرصاءُ كالثَّلجِ.</span>
<span className="v" id="v7" data-v="7" data-xref="المُلُوكِ الثّانِي 5:14; التَّثْنِيَة 32:39; ﻣﺘﻰ 8:3"> فقالَ لَه الرّبُّ: ”رُدَّ يَدَك إلى جَيبِكَ“. فرَدَّ يَدَهُ إلى جَيبِهِ ثُمَّ أخرَجَها فَعادَت كسائِرِ بَدَنِه.</span>
<span className="v" id="v8" data-v="8" data-xref="إشَعْياء 28:10; التَّثْنِيَة 32:39; الخُرُوج 4:30–31"> قالَ لَه: ”إنْ كانوا لا يُصَدِّقونَكَ ولا يَقتَنِعونَ بِالمُعجِزَةِ الأُولى، فبِالمُعجِزَةِ الثَّانيةِ يقتَنِعونَ.</span>
<span className="v" id="v9" data-v="9" data-xref="الخُرُوج 7:19–25; الخُرُوج 1:22; ﻣﺘﻰ 7:2"> وإنْ كانُوا لا يُصَدِّقونَ هاتَينِ المُعجِزَتَينِ ولا يسمعونَ لِكلامِكَ، فَخُذْ مِنْ ماءِ النَّهرِ واَسْكُبْ على الأرضِ، فيَصيرَ الماءُ الّذي تأخُذُه مِنَ النَّهرِ دَمًا“.</span>
</div>
</div>

## هرون ينطق باَسم موسى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="إرْمِيا 1:6; الخُرُوج 6:12; اعمال الرسل 7:22"> فقال موسى للرّبِّ: ”يا ربُّ! ما كُنتُ يومًا رَجُلًا فصيحًا. لا بالأمسِ ولا مِنْ يومَ كَلَّمْتَني أنا عبدَكَ بل أنا بطيءُ النُّطْقِ وثقيلُ اللِّسانِ“.</span>
<span className="v" id="v11" data-v="11" data-xref="عامُوس 3:6; المَزامِير 146:8; حِزْقِيال 3:26–27"> فقالَ لَه الرّبُّ: ”مَنِ الذي خلَقَ للإنسانِ فَمًا؟ ومَنِ الّذي خلَقَ الأخرسَ أوِ الأصمَّ أوِ البَصيرَ أوِ الأعمى؟ أما هوَ أنا الرّبُّ؟</span>
<span className="v" id="v12" data-v="12" data-xref="إرْمِيا 1:9; لُوقا 12:11–12; ﻣﺘﻰ 10:19–20"> فاَذهَبْ وأنا أُعينُكَ على الكلامِ وأُعَلِّمُكَ ما تقولُ“.</span>
<span className="v" id="v13" data-v="13" data-xref="ﻳﻮﻧﺎﻥ 1:3; الخُرُوج 4:1; ﻣﺘﻰ 13:41"> فقالَ موسى: ”يا ربُّ! أرسِلْ أحدًا غَيري“.</span>
<span className="v" id="v14" data-v="14" data-xref="مَرْقُس 14:13–15; الخُرُوج 4:27; أخبارِ الأيّامِ الأوّلُ 21:7"> فغَضِبَ الرّبُّ على موسى غضَبًا شديدًا وقالَ لَه: ”أعرِفُ هرونَ اللَّاويَّ أخاكَ أنَّه فصيحُ اللِّسانِ وها هوَ الآنَ خارِجٌ لِلقائِكَ وحينَ يراكَ يفرَحُ في قلبِه.</span>
<span className="v" id="v15" data-v="15" data-xref="إشَعْياء 51:16; التَّثْنِيَة 18:18; لُوقا 21:15"> فَكلِّمْهُ أنتَ بِما تُريدُ أنْ يَنطِقَ بِهِ، وأنا أُعينُكُما على ما تقولانِه وأُعلِّمُكُما وأُريكُما ما تعمَلانِه.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 7:1–2; المَزامِير 82:6; يُوحَنّا 10:34–35"> هوَ يُخاطِبُ الشَّعبَ عَنكَ ويَنطِقُ باَسمِكَ، وأنتَ تكونُ لَه كأنَّكَ اللهُ يُوحي إليه.</span>
<span className="v" id="v17" data-v="17" data-xref="الخُرُوج 4:2; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:27; الخُرُوج 7:9–20"> وخُذْ بِيَدِكَ هذِهِ العصا، فبِها تصنَعُ المُعجِزاتِ“.</span>
</div>
</div>

## موسى يرجع إلى مصر

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 3:1; صَمُوئيلَ الأوَّلُ 1:17; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:1"> فرَجَعَ موسى إلى يَثرونَ حَميِّهِ وقالَ لَه: ”دَعْني أرجِعُ إلى بَني قومي الّذينَ في مِصْرَ لأرى هل هُم أحياءُ بَعدُ“. فأجابَهُ يَثرونُ: ”إذهَبْ بِسلامٍ“.</span>
<span className="v" id="v19" data-v="19" data-xref="ﻣﺘﻰ 2:20; الخُرُوج 2:23; الخُرُوج 2:15"> وقالَ الرّبُّ لِموسى في مِدْيانَ: ”إرجِعْ إلى مِصْرَ، لأنَّ جميعَ الّذينَ يُريدونَ قَتلَكَ ماتوا“.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 17:9; العَدَد 20:8–9; الخُرُوج 18:3–4"> فأخذَ موسى اَمرأتَه ووَلدَيهِ وأركبَهُم على الحميرِ ورَجعَ إلى أرضِ مِصْرَ. وأخذَ موسى عصا اللهِ بِيَدِهِ.</span>
</div>
<div className="bp">
<span className="v" id="v21" data-v="21" data-xref="ﺭﻭﻣﻴﺔ 9:18; إشَعْياء 63:17; الخُرُوج 14:8"> وقالَ الرّبُّ لموسى: ”أنتَ راجِعٌ إلى مِصْرَ، فاَصْنَعْ أمامَ فِرعَونَ جميعَ العَجائبِ الّتي جَعَلْتُكَ قادِرًا على صُنْعِها وأنا أُقسِّي قلبَه فلا يُطلِقُ شعبَ إِسرائيلَ مِنَ البلادِ.</span>
<span className="v" id="v22" data-v="22" data-xref="هُوشَع 11:1; ﺭﻭﻣﻴﺔ 9:4; إرْمِيا 31:9"> وقُلْ لِفِرعَونَ هذا ما قالَ الرّبُّ: ”إِسرائيلُ اَبْني البِكْرُ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 11:5; الخُرُوج 12:29; المَزامِير 135:8"> أقولُ لكَ: أطلِقِ اَبْني مِنْ مِصْرَ لِيعبُدَني وإنْ رفَضْتَ أنْ تُطلِقَهُ أقتُلِ ابنَكَ البِكْرَ“.</span>
</div>
</div>

## موسى يختن اَبنه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="التَّكوين 17:14; العَدَد 22:22–23; هُوشَع 13:8"> وبينما موسى في الطَّريقِ لاقاهُ الرّبُّ في مكانٍ لِلمَبيتِ وحاولَ أنْ يُميتَه.</span>
<span className="v" id="v25" data-v="25" data-xref="يَشُوع 5:2–3; التَّكوين 17:14; صموئيلَ الثّانِي 16:7"> فأخذَت صَفُّورَةُ اَمرأتُه صَوَّانَةً فختَنَتِ اَبنَها ومسَّت بِها رِجلَي موسى وقالت: ”أنتَ الآنَ عريسُ دَمٍ لي“.</span>
<span className="v" id="v26" data-v="26"> فعَفَّ عَنهُ الربُّعِندَما قالت: ”عريسُ دَمٍ لي“، مِنْ أجلِ الخِتانِ.</span>
</div>
</div>

## لقاء موسى وهرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 3:1; الخُرُوج 4:14–16; المُلُوكِ الأوَّلُ 19:8"> وقالَ الرّبُّ لهَرونَ: ”إذهَبْ إلى البرِّيَّةِ لِلِقاءِ موسى“. فذَهبَ ولَقِيهُ في جبَلِ اللهِ فقبَّلهُ.</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 4:8–9; الخُرُوج 4:15–16; ﻣﺘﻰ 21:29"> فأخبَر موسى هرونَ بجميعِ ما كلَّمَهُ بِهِ الرّبُّ حينَ أرسَلَه، وبجميعِ المُعجِزاتِ الّتي أمَرَهُ بها.</span>
<span className="v" id="v29" data-v="29" data-xref="الخُرُوج 3:16; الخُرُوج 24:11; الخُرُوج 24:1"> فذَهبَ موسى وهرونُ وجمعا شُيوخَ بَني إِسرائِيلَ كُلَّهُم.</span>
<span className="v" id="v30" data-v="30" data-xref="الخُرُوج 4:16"> وخاطَبَهُم هرونُ بجميعِ ما كلَّمَ الرّبُّ بهِ موسى. ثُمَّ صنَعَ موسى المُعجزاتِ أمامَ عُيونِ الشَّعبِ</span>
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 3:18; الخُرُوج 2:25; المَزامِير 106:12–13"> فآمَنوا. ولمَّا سمِعوا أنَّ الرّبَّ تفقَّدَ بَني إِسرائيلَ ورأى ما يُعانُونَه مِنَ الذُلِّ، ركَعوا ساجِدينَ.</span>
