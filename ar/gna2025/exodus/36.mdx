## تقدمات كثيرة من الشعب

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="كُولُوسِي 4:17; العِبرانِيّين 5:4; الخُرُوج 31:6"> ودَعا موسى بَصَلْئيلَ وأهوليآبَ وكُلَّ صانِعٍ ماهِرٍ أودَعَ الرّبُّ حِكمةً في قلبِهِ وكُلَّ مَنْ حرَّكَهُ قَلبُهُ أنْ يتَقَدَّمَ ويبدأَ العمَلَ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 35:29; الخُرُوج 35:27; الخُرُوج 35:5–21"> فتَسَلَّموا مِنْ موسى جميعَ التَّقدِماتِ الّتي جاءَ بِها بَنو إِسرائيلَ لِبِناءِ الخَيمةِ المُقَدَّسةِ وتجهيزِها. وما زالَ الشَّعبُ يَجيئونَ إلى موسى في كُلِّ صباحٍ بِشيءٍ يتَبرَّعونَ بِهِ.</span>
<span className="v" id="v4" data-v="4" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 3:10; لُوقا 12:42; ﻣﺘﻰ 24:45"> فأقبَلَ جميعُ المَهَرَةِ الّذينَ يَقومونَ بِبِناءِ الخَيمةِ وتجهيزِها، كُلٌّ مِنْهُم حسَبَ حِرْفَتِهِ،</span>
<span className="v" id="v5" data-v="5" data-xref="أخبارِ الأيّامِ الثّانِي 31:6–10; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 8:2–3; أخبارِ الأيّامِ الثّانِي 24:14"> وقالوا لِموسى: ”الشَّعبُ يتَبرَّعُ بِأكثَرَ مِمَّا يحتاجُ إليهِ العمَلُ الّذي أمَرَ الرّبُّ بهِ“.</span>
<span className="v" id="v6" data-v="6"> فأمرَ موسى أنْ يُذاعَ في المَحلَّةِ أنْ لا يتَبرَّعَ رَجُلٌ ولا اَمْرَأةٌ بَعدُ بشيءٍ لِلخَيمةِ المُقَدَّسةِ، فاَمْتَنَعَ الشَّعبُ عَنِ التَبرُّعِ.</span>
<span className="v" id="v7" data-v="7" data-xref="أخبارِ الأيّامِ الثّانِي 31:10"> وكانَ ما تبَرَّعوا بهِ أكثَرَ مِنْ كافٍ لإتمامِ العمَلِ.</span>
</div>
</div>

## كيف صنعت خيمة الإجتماع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="المُلُوكِ الأوَّلُ 6:23; الخُرُوج 26:1–37; حِزْقِيال 1:5–28"> فصنعَ جميعُ المَهَرَةِ مِنَ الصُّنَّاعِ مَسكِنَ الرّبِّ عَشْرُ شِقَقٍ مِنْ كتَّانٍ مَبرومٍ ونسيجٍ بَنَفْسَجيٍّ وأُرجوانيٍّ وقِرمِزيِّ اللَّونِ مُطرَّزٌ كُلُّهُ بكروبيمَ بِيَدِ نَسَّاجٍ ماهرٍ،</span>
<span className="v" id="v9" data-v="9"> طولُ كُلِّ شِقَّةٍ ثمانٍ وعِشرونَ ذِراعًا في عَرْضِ أربَعِ أذرُعٍ، والقياسُ واحدٌ لِكُلِّ الشِّقَقِ.</span>
<span className="v" id="v10" data-v="10" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:27; الخُرُوج 26:3; المَزامِير 122:3"> ووَصَلوا خَمْسًا مِنَ الشِّقَقِ، بَعضها بِبَعضٍ، وكذلِكَ الشِّقَقُ الخَمْسُ الأُخرى.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 26:4"> وعمِلوا عُرًى مِنْ نسيجٍ بَنَفْسَجيِّ اللَّونِ على حاشيَةِ الشِّقَّةِ المُتطَرِّفَةِ مِنَ المُوَصَّلِ الواحدِ.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 26:5; الخُرُوج 26:10"> وصَنَعوا خَمسينَ عُروَةً في حاشيَةِ الشِّقَّةِ الواحدَةِ، وخَمسينَ عُروَةً في طَرَفِ الشِّقَّةِ الّتي تُقابِلُها في المُوَصَّلِ الآخَرِ.</span>
<span className="v" id="v13" data-v="13" data-xref="رسالَةُ بُطرُسَ الأُولَى 2:4–5; أفَسُس 2:20–22; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:20"> وصَنَعوا خَمْسينَ مَشبَكًا مِنَ الذَّهَبِ وضَمُّوا المُوَصَّلينِ بالمَشابِكِ، فصارَ المَسكِنُ واحدًا.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 26:7–13"> وصَنَعوا خَيمةً على المَسكنِ، إحدى عشْرَةَ شِقَّةً مِنْ شَعْرِ مَعَزٍ،</span>
<span className="v" id="v15" data-v="15"> طولُ الواحدةِ ثَلاثونَ ذِراعًا في عَرْضِ أربعِ أذرُعٍ، والقياسُ وَاحِدٌ لِلإحدى عَشْرَةَ شِقَّةً.</span>
<span className="v" id="v16" data-v="16"> وضَمُّوا خَمْسَ شِقَقٍ على حِدَةٍ، وستَّ شِقَقٍ على حِدَةٍ.</span>
<span className="v" id="v17" data-v="17"> وصَنَعوا خمسينَ عُروةً على حاشيَةِ الشِّقَّةِ المُتطَرِّفَةِ في كُلٍّ مِنَ المُوَصَّلَينِ.</span>
<span className="v" id="v18" data-v="18"> وصَنَعوا خَمسينَ مَشبَكًا مِنْ نُحاسٍ لِضَمِّ المُوَصَّلَينِ حتّى تصيرَ الخَيمةُ واحدَةً.</span>
<span className="v" id="v19" data-v="19" data-xref="الخُرُوج 26:14"> وصَنَعوا فَوقَ الخَيمةِ غِطاءَينِ آخرَينِ، أحدُهُما مِنْ جُلودِ كِباشٍ مَصبوغَةٍ بِالحُمرَةِ، والآخَرُ مِنْ جُلودٍ بَنَفسَجيَّةِ اللَّونِ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 25:5; العَدَد 25:1; الخُرُوج 40:18–19"> وصَنَعوا ألواحًا قَائمَةً مِنْ خشَبِ السَّنْطِ،</span>
<span className="v" id="v21" data-v="21"> طولُ الواحدِ مِنها عشْرُ أذرُعٍ في عَرْضِ ذِراعٍ ونِصْفٍ.</span>
<span className="v" id="v22" data-v="22"> وعَمِلوا لِكُلِّ لوحٍ مِنها رِجْلَينِ مُتَقابِلَتَينِ، إحداهُما بِإزاءِ الأُخرى، لِضَمِّ الألواحِ كُلِّها بَعضها إلى بَعضٍ.</span>
<span className="v" id="v23" data-v="23"> فكانَ لِلمَسكِنِ عِشرونُ لوحًا مِنْ جِهَةِ الجَنوبِ</span>
<span className="v" id="v24" data-v="24"> وأربعونَ قاعِدَةً تَحتَها مِنْ فِضَّةٍ، قاعِدتانِ تَحتَ كُلِّ لوحٍ،</span>
<span className="v" id="v25" data-v="25"> وعِشرونَ لوحًا مِنْ جِهَةِ الشَّمالِ.</span>
<span className="v" id="v26" data-v="26"> وأربعونَ قاعدَةً تَحتَها مِنْ فِضَّةٍ، قاعِدتانِ قاعِدتانِ تَحتَ كُلِّ لوحٍ.</span>
<span className="v" id="v27" data-v="27" data-xref="الخُرُوج 26:22; الخُرُوج 26:27"> وصَنَعوا في مُؤخَّرِ الخَيمةِ جِهَةَ الغربِ سِتَّةَ ألواحٍ،</span>
<span className="v" id="v28" data-v="28"> وفي زَاويَتَيها في المُؤخَّرِ لوحَينِ</span>
<span className="v" id="v29" data-v="29" data-xref="اعمال الرسل 4:32; أفَسُس 3:18–19; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:10"> مُنفَصِلَينِ مِنْ أسفَلُ ومُتَّحِدَينِ في أعلاهُما عِندَ الحَلقَةِ الأُولى. كذلِكَ كانا لِلزَّاويَتَينِ.</span>
<span className="v" id="v30" data-v="30" data-xref="الخُرُوج 26:25"> فكانَت هُناكَ ثمانيَةُ ألواحٍ قواعِدُها السِّتَّ عشْرةَ مِنْ فِضَّةٍ، قاعِدَتانِ قاعِدَتانِ تَحتَ كُلِّ لوحٍ.</span>
</div>
<div className="bp">
<span className="v" id="v31" data-v="31" data-xref="الخُرُوج 26:26–29; الخُرُوج 25:28; الخُرُوج 30:5"> وصَنَعوا عَوارِضَ مِنْ خشَبِ السَّنْطِ، خمْسًا لألواحِ الجَانِبِ الواحدِ مِنَ المَسكِنِ،</span>
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 26:26"> وخَمْسًا لألواحِ الجانِبِ الآخَرِ، وخَمْسًا لألواحِ المَسكِنِ في المُؤخَّرِ جِهَةَ الغَربِ.</span>
<span className="v" id="v33" data-v="33"> وصَنَعوا العارِضَةَ الوُسطَى في وسَطِ الألواحِ مِنَ الطَّرَفِ إلى الطَّرَفِ.</span>
<span className="v" id="v34" data-v="34"> وغَشَّوا الألواحَ بِذهَبٍ، وصَنَعوا لها حلَقَاتٍ مِنْ ذهَبْ مكانًا لِلعَوارِضِ وغَشَّوا العَوارِضَ بِذهَبٍ.</span>
</div>
<div className="bp">
<span className="v" id="v35" data-v="35" data-xref="ﻣﺘﻰ 27:51; العِبرانِيّين 10:20; الخُرُوج 40:21"> وصَنَعوا حِجابًا مِنْ نسيجٍ بَنَفْسَجيٍّ وأُرْجوانيٍّ وقِرمِزيِّ اللَّونِ وكتَّانٍ مَبرومٍ مُطَرَّزٍ بِكروبيمَ، بِيَدِ نَسَّاجٍ ماهرٍ.</span>
<span className="v" id="v36" data-v="36" data-xref="إرْمِيا 1:18"> وصَنَعوا لِلحِجابِ أربَعَةَ أعمِدَةٍ مِنْ خشَبِ السَّنْطِ وغَشَّوها ذهَبًا، وعَقاقيفُها مِنْ ذهَبٍ. وصاغوا لِلأعمِدَةِ أربَعَ قواعِدَ مِنْ فِضَّةٍ.</span>
<span className="v" id="v37" data-v="37" data-xref="الخُرُوج 26:36–37; الخُرُوج 40:28"> صَنَعوا سِتارَةً لبابِ المَسكِنِ مِنْ نسيجٍ بَنَفْسَجيٍّ وأُرجوانِيٍّ وقِرمِزيِّ اللَّونِ وكتَّانٍ مَبرومٍ مُطَرَّزٍ.</span>
<span className="v" id="v38" data-v="38" data-xref="الخُرُوج 27:10"> وصَنَعوا لِلِّستارَةِ خَمْسَةَ أعمِدَةٍ بِعَقَاقيفِها وغَشَّوا رُؤُوسَها وقُضبانَها بِالذَّهَبِ، وصَنَعوا لها خَمْسَ قواعِدَ مِنْ نُحاسٍ.</span>
