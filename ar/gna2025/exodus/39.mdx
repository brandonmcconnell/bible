## ثياب الكهنة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 31:10; الخُرُوج 35:23; الخُرُوج 35:19"> ومِنَ النَّسيجِ البَنَفْسَجيِّ والأُرجُوانيِّ والقِرمِزيِّ اللَّونِ صنعَوا ثيابًا فاخِرَةً يَلبَسُها الكهنَةُ لِلخِدمَةِ في القُدْسِ، كما صنعَوا ثيابَ هرونَ المُقَدَّسةَ على نحوِ ما أمرَ الرّبُّ موسى.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 28:6–12; اللّاويِّين 8:7; الخُرُوج 25:7"> وصنعَوا الأَفودَ مِنْ ذهَبٍ ونسيجٍ بَنَفْسَجيٍّ وأُرجوانيٍّ وقِرمِزيِّ اللَّونِ وكَتَّانٍ مَبرومٍ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 36:8; الخُرُوج 26:1"> وطَرَّقوا الذَّهَبَ صَفائِحَ وقَدُّوها خُيوطًا، لِيَصنَعوها بِيَدِ نَسَّاجٍ ماهرٍ في وسَطِ النَّسيجِ البَنَفْسَجيِّ والأُرجُوانيِّ والقِرمِزيِّ اللَّونِ والكَتَّانِ المَبرومِ،</span>
<span className="v" id="v4" data-v="4"> وصنعَوا لَه في طرَفَيهِ كَتِفَينِ مَوصُولَتَينِ لِيَتَّصِلَ.</span>
<span className="v" id="v5" data-v="5" data-xref="رُؤيا يُوحَنّا 1:13; ﻣﺘﻰ 28:20; إشَعْياء 11:5"> وكانَ الزُنَّارُ الّذي يَشُدُّ الأَفودَ مَصنوعًا مِنْ ذهَبٍ ونَسيجٍ بَنَفسَجيٍّ وأُرجوانيٍّ وقِرمزيِّ اللَّونِ وكَتَّانٍ مَبرومٍ، على نحوِ ما أمرَ الرّبُّ موسى.</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 28:9; حِزْقِيال 28:13; أيُّوب 28:16"> وعَمِلوا حجَرَي العَقيقِ اليَماني وأحاطوهُما بِطَوقَينِ مِنَ الذَّهَبِ نُقِشَت علَيهما كَنَقْشِ الخاتَمِ أسماءُ أسباطِ بَني إِسرائيلَ،</span>
<span className="v" id="v7" data-v="7" data-xref="يَشُوع 4:7; الخُرُوج 28:12; الخُرُوج 28:29"> وجعَلوهُما على كَتفَي الأَفودِ لِتَذكيرِ أسباطِ بَني إِسرائيلَ، كما أمرَ الرّبُّ موسى.</span>
</div>
</div>

## صدرة القضاء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 28:4; اللّاويِّين 8:8–9; إشَعْياء 59:17"> وصنَعوا الصُّدْرَةَ بِيَدِ نَسَّاجٍ ماهِرٍ كصَنعَةِ الأَفودِ، مِنْ ذهَبٍ ومِنْ نَسيجٍ بَنَفْسَجيٍّ وأُرجُوانيٍّ وقِرمِزيِّ اللَّونِ وكَتَّانٍ مَبرومٍ.</span>
<span className="v" id="v9" data-v="9"> صنَعوها مُرَبَّعَةً مَثنيَّةً، طُولُها شِبْرٌ وعَرضُها شِبْرٌ.</span>
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 28:21; الخُرُوج 28:16–17; رُؤيا يُوحَنّا 21:19–21"> ورَصَّعوا فِيها أربعَةَ صُفوفٍ مِنَ الحِجارَةِ الكريمَةِ، الصفُّ الأوَّلُ عَقيقٌ أحمَرُ وياقوتٌ أصفَرُ وزُمُرُّدٌ،</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 28:18; حِزْقِيال 28:13"> والصَّفُّ الثَّاني بَهرَمانٌ وياقوتٌ أزرَقُ وعَقيقٌ أبيَضُ،</span>
<span className="v" id="v12" data-v="12"> والصَّفُّ الثَّالثُ ياقوتٌ زَعفَرانيٌّ وياقوتٌ أحمَرُ وجَمَشْتُ،</span>
<span className="v" id="v13" data-v="13"> والصَّفُّ الرَّابِعُ زَبَرْجَدٌ وعَقيقٌ يَمانيٌّ ويَشْبٌ يُحيطُ بِها ذهَبٌ في تَرصيعِها.</span>
<span className="v" id="v14" data-v="14" data-xref="رُؤيا يُوحَنّا 21:12"> وكانَت هذِهِ الحِجارَةُ بِعَدَدِ أسماءِ أسباطِ بَني إِسرائيلَ اَثني عشَرَ حجَرًا، يُنقَشُ على كُلِّ حجَرٍ اَسمُهُ كَنَقشِ الخاتَمِ.</span>
<span className="v" id="v15" data-v="15" data-xref="يُوحَنّا 10:28; يُوحَنّا 17:12; رسالَةُ بُطرُسَ الأُولَى 1:5"> وصنَعوا لِلصُّدْرَةِ سَلاسِلَ مَجدولَةً جَدْلَ الضَّفائِرِ مِنَ الذَّهَبِ الخالصِ.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 25:12"> وصنَعوا طَوقَينِ مِنَ الذَّهبِ وحَلقَتَينِ مِنَ الذَّهبِ وجعَلوا الحلَقَتَينِ في طَرفَيِّ الصُّدْرَةِ،</span>
<span className="v" id="v17" data-v="17"> وعلَّقوا ضفيرَتيِّ الذَّهَبِ على الحلَقتَينِ اللَّتينِ في طَرفَيِّ الصُّدْرَةِ</span>
<span className="v" id="v18" data-v="18" data-xref="الخُرُوج 28:14; نَشِيدِ الأنشاد 1:10"> وعلَّقوا طَرفَيِّ الصُّدْرَةِ بِالطَّوقَينِ وجعَلوهُما على كَتِفَي الأَفودِ مِنَ الأمامِ.</span>
<span className="v" id="v19" data-v="19"> وصنَعوا حلَقَتَينِ مِنْ ذهَبٍ في طَرفَيِّ الصُّدْرَةِ في حاشيَتِها الّتي إلى جِهَةِ الأَفودِ مِنْ داخِلٍ.</span>
<span className="v" id="v20" data-v="20" data-xref="الخُرُوج 26:3"> وصنَعوا حلَقَتَينِ مِنْ ذهَبٍ على كَتِفَي الأَفودِ مِنْ أسفَلُ في مُقَدَّمَتِهِ عِندَ وَصْلَتِهِ فَوقَ زُنَّارِ الأَفودِ،</span>
<span className="v" id="v21" data-v="21" data-xref="ﻣﺘﻰ 16:24; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:25; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 1:27"> وحبَكوا الصُّدْرَةَ مِنْ حلَقَتَيها إلى حلَقَتَيِّ الأَفودِ بِخَيطٍ مِنْ نسيجٍ بَنَفْسَجيِّ اللَّونِ لِتَصيرَ على زُنَّارِ الأَفودِ ولا تنفَصِلُ عن الأَفودِ، كما أمرَ الرّبُّ موسى.</span>
</div>
</div>

## الجبّة وما إليها

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 28:31–35"> وصنَعوا جُبَّةَ الأَفودِ بِيَدِ حائكٍ، كُلُّها مِنْ نسيجٍ بَنَفْسَجيِّ اللَّونِ</span>
<span className="v" id="v23" data-v="23"> وكانَ لها في وسَطِها فُتْحَةٌ لِلرأسِ تُحيطُ بِها حاشيَةٌ كَفُتْحَةِ الدِّرْعِ فلا تتَمَزَّقُ</span>
<span className="v" id="v24" data-v="24" data-xref="الخُرُوج 28:33; غَلاطِيَّة 5:22"> وصنَعوا لأذيالِها رُمَّاناتٍ مِنْ نسيجٍ بَنَفْسَجيٍّ وأُرجُوانيٍّ وقِرمِزيِّ اللَّونِ وكَتَّانٍ مَبرومٍ.</span>
<span className="v" id="v25" data-v="25" data-xref="الخُرُوج 28:33–34; نَشِيدِ الأنشاد 4:13; المَزامِير 89:15"> وصنَعوا أجراسًا مِنْ ذهَبٍ خالصٍ فيما بَينَ الرُّماناتِ في ذَيلِ الجُبَّةِ مِنْ حَولِها:</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 28:34; نَشِيدِ الأنشاد 4:13; نَشِيدِ الأنشاد 6:7"> جَرَسُ ذهَبٍ وَرُمَّانَةٌ بَعدَهُ مِنْ أوَّلِ أذيالِ الجُبَّةِ إلى آخِرِها، كما أمرَ الرّبُّ موسى.</span>
<span className="v" id="v27" data-v="27" data-xref="حِزْقِيال 44:18; رسالَةُ بُطرُسَ الأُولَى 1:13; إشَعْياء 61:10"> وصنَعوا القُمصانَ مِنْ كَتَّانٍ صَنعَةَ حائِكٍ لِهرونَ وبَنيهِ،</span>
<span className="v" id="v28" data-v="28" data-xref="الخُرُوج 28:4; حِزْقِيال 44:18; الخُرُوج 28:39"> والعِمامَةَ مِنْ كَتَّانٍ، وعَصائِبَ القَلانِسِ مِنْ كَتَّانٍ، والسَّراويلَ مِنْ كَتَّانٍ مَبرومٍ،</span>
<span className="v" id="v29" data-v="29" data-xref="الخُرُوج 28:39"> والحِزامَ مِنْ كَتَّانٍ مَبرومٍ ونسيجٍ بَنَفْسَجيٍّ وأُرجوانيٍّ وقِرمِزيِّ اللَّونِ صَنعَةَ مُطَرِّزٍ، كما أمرَ الرّبُّ موسى.</span>
<span className="v" id="v30" data-v="30" data-xref="رُؤيا يُوحَنّا 5:10; العِبرانِيّين 7:26; العِبرانِيّين 1:3"> وصنَعوا التَّاجَ المُقَدَّسَ مِنْ ذهَبٍ خالصٍ ونَذَروهُ لِلرّبِّ وكَتَبوا علَيهِ كِتابَةً كَنَقْشِ الخَاتَمِ: مُقَدَّسٌ لِلرّبِّ.</span>
<span className="v" id="v31" data-v="31"> ووضَعوا علَيهِ خَيطًا مِنْ نَسيجٍ بَنَفْسَجيِّ اللَّونِ، فيكونُ على العِمامَةِ مِنْ مُقَدِّمِها، كما أمرَ الرّبُّ موسى.</span>
<span className="v" id="v32" data-v="32" data-xref="الخُرُوج 39:42–43; ﻣﺘﻰ 28:20; الخُرُوج 25:40"> فكَمُلَ العمَلُ كُلُّه في مَسكِنِ خَيمةِ الإجتماعِ، وبِذلِكَ صَنعَ بَنو إِسرائيلَ جميعَ ما أمرَ الرّبُّ بهِ موسى.</span>
<span className="v" id="v33" data-v="33" data-xref="الخُرُوج 35:11–19; الخُرُوج 31:7–11; الخُرُوج 36:1–38"> وجاؤوا بِالمَسكِنِ إلى موسى: الخَيمَةِ وجميعِ أدَواتِها مِنْ مَشابِكَ وألواحٍ وعَوارِضَ وأعمِدَةٍ وقواعِدَ،</span>
<span className="v" id="v34" data-v="34"> والغِطاءِ الّذي مِنْ جُلودِ الكِباشِ المَصبوغَةِ بالحُمرَةِ، والغِطاءِ الّذي مِنَ الجُلودِ البَنَفْسَجيَّةِ اللَّونِ، والحِجاب</span>
<span className="v" id="v35" data-v="35" data-xref="العِبرانِيّين 9:5; الخُرُوج 25:17; العِبرانِيّين 9:8"> وتابوتِ العَهدِ وعَوارِضِهِ وغِشائِهِ،</span>
<span className="v" id="v36" data-v="36" data-xref="الخُرُوج 25:30; المُلُوكِ الأوَّلُ 7:48"> والمائِدَةِ وجميعِ أوانيها وخُبزِ التَّقدِمَةِ،</span>
<span className="v" id="v37" data-v="37" data-xref="فِيلِبِّي 2:15; الخُرُوج 27:21; ﻣﺘﻰ 5:14–16"> والمَنارَةِ المُقَدَّسَةِ وعلَيها سُرُجُها ومعَها جميعُ أدَواتِها وزيتُ إنارَتِها.</span>
<span className="v" id="v38" data-v="38" data-xref="الخُرُوج 30:7; الخُرُوج 37:29; الخُرُوج 30:3"> ومذبَحِ الذَّهَبِ وزيتِ المَسْحِ والبَخورِ العَطِرِ وسِتارَةِ بابِ الخَيمةِ،</span>
<span className="v" id="v39" data-v="39" data-xref="المُلُوكِ الأوَّلُ 8:64; الخُرُوج 38:30"> ومذبَحِ النُّحاسِ وشبَكَةِ النُّحاسِ الّتي لَه وقضيبَيهِ وجميعِ أدَواتِه، والمَغسلَةِ ومَقعَدِها،</span>
<span className="v" id="v40" data-v="40" data-xref="الخُرُوج 27:9–16"> وستائِرِ رواقِ المَسكِنِ وأعمِدَتِهِ وقواعِدِهِ وسِتارَةِ بابِهِ وحِبالِهِ وأوتادِهِ وسائِرِ الأدَواتِ الّتي تُستَعمَلُ في مَسكِنِ خَيمةِ الإجتماعِ،</span>
<span className="v" id="v41" data-v="41" data-xref="الخُرُوج 28:2; الخُرُوج 31:10; الخُرُوج 39:1"> وثِيابِ الكهَنَةِ الفاخِرَةِ لِلخِدمَةِ في القُدْسِ، وثيابِ الكاهنِ المُقَدَّسَةِ وثيابِ بَنيهِ الكهنَةِ.</span>
</div>
<div className="bp">
<span className="v" id="v42" data-v="42" data-xref="الخُرُوج 35:10; الخُرُوج 23:21–22; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 2:15"> وهكذا أكمَلَ بَنو إِسرائيلَ هذا العمَلَ على نحوِ ما أمرَ الرّبُّ بهِ موسى.</span>
<span className="v" id="v43" data-v="43" data-xref="العَدَد 6:23–27; اللّاويِّين 9:22–23; صموئيلَ الثّانِي 6:18"> فلمَّا نظَرَ إليهِ موسى رأى أنَّهم صنَعوا على نحوِ ما أمرَ الرّبُّ بهِ. فبارَكَهُم موسى.</span>
