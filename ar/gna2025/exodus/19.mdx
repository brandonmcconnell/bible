## بنو إسرائيل في سيناء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 16:1; الخُرُوج 12:6; اللّاويِّين 23:16–18"> وفي الشَّهرِ الثَّالثِ لخُروجِ بَني إِسرائيلَ منْ أرضِ مِصْرَ جاؤوا في أحدِ الأيّامِ إلى برِّيَّةِ سيناءَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 3:1; الخُرُوج 17:1; الخُرُوج 3:12"> جاؤوا منْ رَفيديمَ ونزَلوا هُناكَ في البرِّيَّةِ تُجاهَ الجبَلِ.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 3:4; اعمال الرسل 7:38; الخُرُوج 20:21"> وصعِدَ موسى إلى الجبَلِ لمُلاقاةِ اللهِ. فناداهُ الرّبُّ مِنَ الجبَلِ وقالَ لَه: ”قُلْ لبَيتِ يعقوبَ بَني إِسرائيلَ:</span>
<span className="v" id="v4" data-v="4" data-xref="التَّثْنِيَة 32:11–12; إشَعْياء 63:9; إشَعْياء 40:31"> رأيتُم ما فعلْتُ بالمِصْريِّينَ وكيفَ حَمَلتُكُم على أجنِحَةِ النُّسورِ وجِئتُ بكُم إليَّ.</span>
<span className="v" id="v5" data-v="5" data-xref="التَّثْنِيَة 14:2; التَّثْنِيَة 7:6; التَّثْنِيَة 26:18"> والآنَ إنْ سمِعْتُم كلامي وحَفِظتُم عَهْدي، فإنَّكُم تكونونَ شعبيَ الخاصَّ بَينَ جميعِ الشُّعوبِ. فالأرضُ كُلُّها لي،</span>
<span className="v" id="v6" data-v="6" data-xref="رُؤيا يُوحَنّا 5:10; رسالَةُ بُطرُسَ الأُولَى 2:9; رسالَةُ بُطرُسَ الأُولَى 2:5"> وأنتُم تكونونَ لي مَملَكةَ كهَنةٍ وأُمَّةً مُقَدَّسَةً. هذا هوَ الكلامُ الّذي تقولُه لبَني إِسرائيلَ“.</span>
<span className="v" id="v7" data-v="7" data-xref="الخُرُوج 4:29–30; الخُرُوج 3:16; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 15:1"> فدعا موسى شُيوخَ الشَّعبِ وألقَى على مَسامِعِهِم جميعَ هذا الكلامِ الّذي أمرَهُ الرّبُّ بِهِ.</span>
<span className="v" id="v8" data-v="8" data-xref="الخُرُوج 24:7; الخُرُوج 24:3; يَشُوع 24:24"> فأجابوا كُلُّهُم: ”جميع ما تكلّمَ بهِ الرّبُّ نعمَلُ بهِ“. فنقَلَ موسى إلى الرّبِّ جوابَ الشَّعبِ.</span>
</div>
</div>

## التمهيد للعهد

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v9" data-v="9" data-xref="التَّثْنِيَة 4:36; الخُرُوج 19:16; الخُرُوج 24:15–16"> وقالَ الرّبُّ لموسى: ”أجيءُ إليكَ في سحابَةٍ كثيفَةٍ ليسمَعَني الشَّعبُ حينَ أُخاطِبُكَ فيُؤمِنوا بِكَ إلى الأبدِ“. ولمَّا نقلَ موسى إلى الرّبِّ جوابَ الشَّعبِ،</span>
<span className="v" id="v10" data-v="10" data-xref="العِبرانِيّين 10:22; التَّكوين 35:2; اللّاويِّين 11:44–45"> قالَ لَه الرّبُّ: ”إذهبْ إلى الشَّعبِ وقُلْ لهُم: طَهِّروا نُفوسَكُمُ اليومَ وغدًا، واَغسِلوا ثيابَكُم،</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 19:16; الخُرُوج 34:5; التَّثْنِيَة 33:2"> واَستَعِدُّوا لليومِ الثَّالِثِ. ففي اليومِ الثَّالِثِ أنزِلُ أمامَكُم على جبَلِ سيناءَ.</span>
<span className="v" id="v12" data-v="12" data-xref="العِبرانِيّين 12:20–21; يَشُوع 3:4; الخُرُوج 19:21"> فاَرسُمْ للشَّعبِ حَدًّا حَولَ الجبَلِ وقُلْ لهُم: ”إيَّاكُم أنْ تصعَدوا الجبَلَ أو تَمَسُّوا طرَفَهُ، فكُلُّ مَنْ مَسَّ الجَبلَ</span>
<span className="v" id="v13" data-v="13" data-xref="الرِّسالَةُ الأُولَى إلَى تَسالُونِيكِي 4:16; العِبرانِيّين 12:20; الخُرُوج 19:16"> يُقتَلُ قَتلًا مِنْ دونِ أنْ تَلمَسَهُ يَدُ أحدٍ يُرجَمُ رَجْمًا أو يُرمى بالسِّهامِ، سَواءٌ أكانَ بهيمَةً أمِ انسانًا فلا يحيا، أمَّا عِندَ صوتِ البوقِ، فيجوزُ للبعضِ أنْ يصعَدوا“.</span>
</div>
<div className="bp">
<span className="v" id="v14" data-v="14" data-xref="الخُرُوج 19:10"> فنَزَلَ موسى إلى الشَّعبِ وأمرَهُم أنْ يُطَهِّروا نُفوسَهم ويَغسِلوا ثيابَهُم</span>
<span className="v" id="v15" data-v="15" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 7:5; صَمُوئيلَ الأوَّلُ 21:4–5; الخُرُوج 19:11"> وقالَ: ”كونوا مُستَعِدِّينَ لليومِ الثَّالِثِ ولا تَقرَبُوا اَمْرَأةً“.</span>
</div>
</div>

## التجلي على جبل سيناء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v16" data-v="16" data-xref="العِبرانِيّين 12:18–19; المَزامِير 18:11–14; العِبرانِيّين 12:21"> وحدَثَ في اليومِ الثَّالِثِ عِندَ الصَّباحِ أنْ كانَت رُعودٌ وبُروقٌ وسَحابٌ كثيفٌ على جبَلِ سيناءَ، وصوتُ بوقٍ شديدٌ جدًّا، فاَرْتَعدَ جميعُ الّذينَ في المَحلَّةِ.</span>
<span className="v" id="v17" data-v="17" data-xref="التَّثْنِيَة 4:10; التَّثْنِيَة 5:5"> فأخرَجَهُم موسى مِنَ المَحلَّةِ لِمُلاقاةِ اللهِ، فوقَفوا عندَ أسفَلِ الجبَلِ،</span>
<span className="v" id="v18" data-v="18" data-xref="القُضاة 5:5; الخُرُوج 3:2; المَزامِير 144:5"> والجبَلُ يلُفُّهُ دُخانٌ، لأنَّ الرّبَّ نَزلَ علَيهِ بالنَّارِ، فتصاعدَ دُخانُهُ كدُخانِ الأتونِ واَهتَزَّ الجبَلُ اَهتِزَازًا شديدًا.</span>
<span className="v" id="v19" data-v="19" data-xref="المَزامِير 81:7; الخُرُوج 19:16; نَحَمْيا 9:13"> وكانَ صوتُ البوقِ يرتَفِعُ جدًّا وموسى يتكلّمُ واللهُ يُجيبُهُ بقصفِ الرَّعْدِ.</span>
<span className="v" id="v20" data-v="20" data-xref="نَحَمْيا 9:13; الخُرُوج 34:2; الخُرُوج 19:11"> ونَزلَ الرّبُّ على رأسِ جبَلِ سيناءَ ونادى موسى إلى رأسِ الجبَلِ فصَعِدَ.</span>
<span className="v" id="v21" data-v="21" data-xref="صَمُوئيلَ الأوَّلُ 6:19; الخُرُوج 3:5; العِبرانِيّين 12:28–29"> فقالَ الرّبُّ لموسى: ”إنزِلْ إلى الشَّعبِ وأنذِرْهُم أنْ لا يُجاوِزوا الحَدَّ المَرسومَ حَولَ الجبَلِ ليَنظروا إليَّ فيَهلِكَ مِنهُم كثيرونَ.</span>
<span className="v" id="v22" data-v="22" data-xref="الخُرُوج 24:5; اعمال الرسل 5:10; أخبارِ الأيّامِ الأوّلُ 13:9–11"> وقُلْ للكهَنةِ الذينَ يتَقَدَّمونَ إليَّ أنْ يَتَطَهَّروا لِئلَّا أبطُشَ بِهِم“.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 19:12; يَشُوع 3:4–5"> فقالَ موسى للرّبِّ: ”لا يقدِرُ الشَّعبُ أنْ يَصعَدوا إلى جبَلِ سيناءَ لأنَّكَ أنتَ هَدَّدْتَنا وقُلتَ: ”أرسُمْ حَدًّا حَولَ الجبَلِ وحرِّمْ علَيهِم أنْ يُجَاوِزُوهُ“.</span>
<span className="v" id="v24" data-v="24" data-xref="العِبرانِيّين 4:16; العِبرانِيّين 12:18–25; العِبرانِيّين 12:29"> فقالَ لَه الرّبُّ: ”إنزِلْ واَصْعَدْ وهرونُ معَكَ، ولكِنْ لا تدَعِ الكهَنةَ والشَّعبَ يُجاوِزونَ حَدَّهُم لِيَصعَدُوا إليَّ لِئلَّا أبطُشَ بِهِم“.</span>
<span className="v" id="v25" data-v="25"> فنزَلَ موسى إلى الشَّعبِ وكلَّمَهُم بِهذا الكلامِ.</span>
</div>
