## لقاء موسى ويثرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 2:16; المَزامِير 77:14–15; الخُرُوج 3:1"> وسمِعَ يَثْرونُ كاهنُ مِديانَ وحَمُو موسى بكلِّ ما عمِلَ الرّبُّ لموسى ولإِسرائيلَ شعبِهِ حينَ أخرجَهُم مِنْ مِصْرَ.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 2:21; الخُرُوج 4:25–26"> وكانَ يَثْرونُ اَسترجَعَ اَبنَتَهُ صَفُّورةَ اَمرأةَ موسى</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 2:22; اعمال الرسل 7:29; المَزامِير 39:12"> واَبنَيها اللّذَينِ سَمَّى موسى أحدَهُما جَرشومَ وقالَ: ”لأنِّي كنتُ نزيلًا في أرضٍ غريبَةٍ“،</span>
<span className="v" id="v4" data-v="4" data-xref="العِبرانِيّين 13:6; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 4:17; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 1:8–10"> وسَمَّى الآخرَ أليعازَرَ وقالَ: ”لأنَّ إلهَ أبي نصَرَني وأنقَذَني مِنْ سيفِ فِرعَونَ“.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 3:1; الخُرُوج 3:12; الخُرُوج 24:16–17"> فجاءَ يَثْرونُ بهؤلاءِ إلى موسى في البرِّيَّةِ حيثُ كانَ نازلًا عِندَ جبَلِ اللهِ.</span>
<span className="v" id="v6" data-v="6"> ولمَّا قيلَ لموسى: ”ها حَمُوكَ يَثْرونُ قادِمٌ إليكَ معَ اَمرَأتِكَ واَبنَيكَ،</span>
<span className="v" id="v7" data-v="7" data-xref="التَّكوين 29:13; التَّكوين 14:17; صموئيلَ الثّانِي 11:7"> خرجَ موسى للقائِهِ وسجدَ وقبَّلهُ. واَستَفسَرَ كُلُّ واحدٍ مِنهُما عَنْ سلامَةِ الآخَرِ ودَخَلا إلى الخَيمَةِ.</span>
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 145:4–12; نَحَمْيا 9:9–15; المَزامِير 107:2"> وقَصَّ موسى على حَميّهِ جميعَ ما فعلَ الرّبُّ بفِرعَونَ والمِصْريِّينَ منْ أجلِ بَني إِسرائيلَ، وجميعَ ما نالَهُم مِنَ المَشقَّةِ في الطَّريقِ، وكيفَ أنقَذَهُمُ الرّبُّ.</span>
<span className="v" id="v9" data-v="9" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 12:26; ﺭﻭﻣﻴﺔ 12:15; إشَعْياء 66:10"> فسُرَّ يَثْرونُ بما عمِلَه الرّبُّ لبَني إِسرائيلَ مِنْ إحسانٍ فائقٍ، حينَ أنقذَهُم مِنْ أيدي المِصْريِّينَ،</span>
<span className="v" id="v10" data-v="10" data-xref="التَّكوين 14:20; صموئيلَ الثّانِي 18:28; لُوقا 1:68"> وقالَ يَثْرونُ: ”تَباركَ الرّبُّ الّذي نجَّاكُم مِنْ أيدي المِصْريِّينَ ومِنْ يَدِ فِرعَونَ. تَباركَ الرّبُّ الّذي نجَّى شعبَهُ مِنْ تَحتِ أيدي المِصْريِّينَ،</span>
<span className="v" id="v11" data-v="11" data-xref="المَزامِير 95:3; الخُرُوج 15:11; أخبارِ الأيّامِ الثّانِي 2:5"> الآنَ علِمتُ أنَّ الرّبَّ أعظمُ مِنْ جميعِ الآلهَةِ، لأنَّه صنَعَ هذا بالمِصْريِّينَ حينَ طغَوا علَيكُم“.</span>
</div>
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="التَّكوين 31:54; التَّثْنِيَة 12:7; الخُرُوج 24:11"> وقدَّمَ يَثْرونُ مُحرقَةً وذبائِحَ للهِ. وجاءَ هرونُ وجميعُ شُيوخِ بَني إِسرائيلَ لِيأكلوا معَهُ أمامَ مذبَحِ اللهِ.</span>
</div>
</div>

## تعيين القضاة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="إشَعْياء 16:5; أيُّوب 29:7; القُضاة 5:10"> وجلسَ موسى في اليومِ الثَّاني ليَقضيَ للشَّعبِ، فوقفَ الشَّعبُ أمامَهُ مِنَ الصَّباحِ إلى المساءِ.</span>
<span className="v" id="v14" data-v="14"> فلمَّا رأى يَثْرونُ كُلَّ ما كانَ على موسى أنْ يفعَلَه لِلشَّعبِ قالَ لَه: ”ما هذا؟ وما بالُكَ جالسًا وحدَكَ لِتَقضيَ لجميعِ الشَّعبِ وهُمْ واقِفونَ أمامَكَ مِنَ الصَّباحِ إلى المساءِ؟“</span>
<span className="v" id="v15" data-v="15" data-xref="العَدَد 15:34; العَدَد 27:5; العَدَد 9:8"> فأجابَهُ موسى: ”يجيءُ إليَّ الشَّعبُّ ليعلَموا إرادةَ اللهِ.</span>
<span className="v" id="v16" data-v="16" data-xref="الخُرُوج 24:14; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:1; صموئيلَ الثّانِي 15:3"> إذا كانَت لهُم دعوى يجيئونَ فأقضي بَينَ الفريقَينِ وأُعَرِّفُهُم فرائِضَ اللهِ وشرائِعَهُ“.</span>
<span className="v" id="v17" data-v="17"> فقالَ يَثْرونُ لموسى: ”لا خيرَ في ما تعمَلُ.</span>
<span className="v" id="v18" data-v="18" data-xref="العَدَد 11:14–17; اعمال الرسل 6:1–4; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 12:15"> فأنتَ تُتعِبُ نفْسَكَ وتُتعِبُ هذا الشَّعبَ معَكَ، لأنَّ الحِمْلَ أثقَلُ منْ أنْ تَحمِلَهُ وحدَكَ.</span>
<span className="v" id="v19" data-v="19" data-xref="العَدَد 27:5; التَّثْنِيَة 5:5; الخُرُوج 20:19"> والآنَ اَسْمَعْ نصيحَتي فيكونَ اللهُ معَكَ: كُنْ أنتَ وسيطًا للشَّعبِ عِندَ اللهِ ترفعُ إليهِ دعاوِيَهُم،</span>
<span className="v" id="v20" data-v="20" data-xref="التَّثْنِيَة 5:1; المَزامِير 143:8; التَّثْنِيَة 1:18"> وتُعلِنُ لهُمُ الفَرائِضَ والشَّرائِعَ، وتُعَرِّفُهُمُ الطَّريقَ الّذي يسلُكونَهُ والنَّهْجَ الّذي ينهَجونَه.</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 6:3; التَّثْنِيَة 1:13–17; أخبارِ الأيّامِ الثّانِي 19:5–10"> واَختَرْ مِنَ الشَّعبِ كُلِّهِ رِجَالًا أكْفَاءَ يخافونَ اللهَ وأُمنَاءَ يكرَهونَ الرَّشوةَ، ووَلِّهِم على الشَّعبِ رؤساءَ ألوفٍ ومئاتٍ وخَماسينَ وعَشَراتٍ</span>
<span className="v" id="v22" data-v="22" data-xref="العَدَد 11:17; العَدَد 27:2; العَدَد 15:33"> يَقضونَ للشَّعبِ في كُلِّ وقتٍ ويرفعونَ إليكَ كُلَّ أمرٍ خطيرٍ ويحكمونَ هُم في كُلِّ أمرٍ صغيرٍ، فيحمِلونَ الحِمْلَ معَكَ ويُخَفِّفونَ عَنكَ.</span>
<span className="v" id="v23" data-v="23" data-xref="صَمُوئيلَ الأوَّلُ 8:6–7; الخُرُوج 18:18; صَمُوئيلَ الأوَّلُ 8:22"> فإنْ أنتَ عمِلْتَ هكذا وأمَرَكَ اللهُ بهِ، أمكنَكَ الاحتِمالُ، وجميعُ هذا الشَّعبِ ينصَرفونَ إلى بُيوتِهِم بسلامٍ“.</span>
</div>
<div className="bp">
<span className="v" id="v24" data-v="24" data-xref="عَزْرا 10:2; الخُرُوج 18:19; الخُرُوج 18:2–5"> فسمِعَ موسى مِنْ يَثْرونَ حَميِّهِ وعمِلَ بكُلِّ ما قالَ لَه.</span>
<span className="v" id="v25" data-v="25" data-xref="التَّثْنِيَة 1:15; الخُرُوج 18:21; اعمال الرسل 6:5"> فاَخْتارَ رِجالًا أكْفَاءَ مِنْ جميعِ بَني إِسرائيلَ وَوَلَّاهُم على الشَّعبِ رؤساءَ ألوفٍ ومئاتٍ وخَماسينَ وعَشَراتٍ.</span>
<span className="v" id="v26" data-v="26" data-xref="الخُرُوج 18:22; الخُرُوج 18:14–15; التَّثْنِيَة 17:8"> فكانوا يَقضونَ للشَّعبِ في كُلِّ وقتٍ، وكُلِّ دعوى صعبةٍ يرفعونَهَا إلى موسى، وكلِّ دعوى بسيطةٍ يحكُمونَ هُم فِيها.</span>
</div>
<div className="bp">
<span className="v" id="v27" data-v="27" data-xref="العَدَد 10:29–30; القُضاة 19:9; التَّكوين 31:55"> ثمَّ ودَّعَ يَثْرونُ موسى ورجَعَ إلى أرضِهِ.</span>
</div>
