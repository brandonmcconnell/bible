## موسى في حضرة فرعون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="الخُرُوج 10:9; اعمال الرسل 4:29; ﻣﺘﻰ 10:28"> وبَعدَ ذلِكَ دخَلَ موسى وهرونُ على فِرعَونَ وقالا لَه: ”قالَ الرّبُّ إلهُ إِسرائيلَ: أطلِقْ شعبي لِيُعَيِّدُوا لي في البرِّيَّةِ“.</span>
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 3:19; أيُّوب 21:15; المُلُوكِ الثّانِي 18:35"> فقالَ فِرعَونُ: ”مَنْ هوَ الرّبُّ حتّى أسمَعَ لَه وأُطلِقَ بَني إِسرائيلَ؟ لا أعرِفُ الرّبَّ ولا أُطلِقُ بَني إِسرائيلَ“.</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 3:18; زَكَريّا 14:16–19; حِزْقِيال 6:11"> فقالا: ”إلهُ العِبرانيِّينَ قابَلَنا، فدَعْنا الآنَ نسيرُ مَسيرَةَ ثلاثةِ أيّامٍ في البرِّيَّةِ ونُقَدِّمُ ذبيحةً للرّبِّ إلهِنا، لِئلَّا يُعاقِبَنا بِوباءٍ أو بِحَدِّ السَّيفِ“.</span>
<span className="v" id="v4" data-v="4" data-xref="الخُرُوج 1:11; اعمال الرسل 24:5; عامُوس 7:10"> فقالَ لهُما مَلِكُ مِصْرَ: ”لماذا يا موسى وهرونُ تُعَطِّلانِ شعبَكمُا عَنْ أعمالِهِم؟ اَذهَبَا وأعيداهُم إلى أشغالِهِم“.</span>
<span className="v" id="v5" data-v="5" data-xref="الخُرُوج 1:7–11; الأمثال 14:28"> وقالَ فِرعَونُ: ”البلادُ تضيقُ الآنَ بِهِم، فكيفَ إذا أرَحْتُماهُم مِنْ أثقالِهم؟“</span>
</div>
</div>

## فرعون يأمر المسخّرين بالقساوة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 5:10; الخُرُوج 5:19; التَّثْنِيَة 16:18"> وفي ذلِكَ اليومِ قالَ فِرعَونُ لِمُسَخِّري شعبِ إِسرائيلَ ورُقَبائِهِم:</span>
<span className="v" id="v7" data-v="7" data-xref="القُضاة 19:19; التَّكوين 24:25"> ”لا تُعطُوا الشَّعبَ بَعدَ اليومِ تِبْنًا لِيصنَعوا اللِّبْنَ، بل ليذهَبوا هُم ويجمَعوا لهُم تِبْنًا.</span>
<span className="v" id="v8" data-v="8" data-xref="المَزامِير 106:41"> واَفْرِضوا علَيهِم مِقدارَ اللِّبْنِ الّذي كانُوا يصنَعونَهُ مِنْ قَبلُ ولا تُنقِصوا مِنهُ شيئًا. هُم مُتكاسِلونَ، ولِذلِكَ يصرُخونَ ويقولونَ: دَعْنا نذهَبُ ونُقَدِّمُ ذبيحَةً لإلهِنا.</span>
<span className="v" id="v9" data-v="9" data-xref="المُلُوكِ الثّانِي 18:20; إرْمِيا 43:2; أيُّوب 16:3"> أرهِقوا هؤلاءِ القومَ بالعَملِ، فيَلتَهوا بهِ عَنِ الكلامِ الباطِلِ“.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="الخُرُوج 1:11; الخُرُوج 3:7; الأمثال 29:12"> فخَرجَ مُسَخِّرُو الشَّعبِ ورُقَبَاؤُهُم وقالوا لهُم: ”قالَ فِرعَونُ: لا أُعطِيكُم تِبْنًا.</span>
<span className="v" id="v11" data-v="11" data-xref="الخُرُوج 5:13–14"> اَذهَبوا أنتُم واَجْمَعُوا لكُم تِبْنًا مِنْ حيثُ تجِدونَ، ومِقدارُ ما كُنتُم تصنعونَ مِنَ اللِّبْنِ لا يَنقُصْ مِنهُ شيءٌ“.</span>
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 15:7; إشَعْياء 5:24; إشَعْياء 47:14"> فتفَرَّقَ الشَّعبُ في كُلِّ أرضِ مِصْرَ لِيجمَعُوا قَشًّا عِوضَ التِّبْنِ،</span>
<span className="v" id="v13" data-v="13"> والمُسَخِّرونَ يستَعجِلونَهم ويقولونَ: ”اَصْنَعوا مِنَ اللِّبْنِ مِقدارَ ما كُنتُم تصنعونَ وقتَ تَزويدِكُم بالتِّبْنِ“.</span>
<span className="v" id="v14" data-v="14" data-xref="إشَعْياء 10:24"> وكانَ المُسَخِّرونَ يضرِبونَ رُقباءَ بَني إِسرائيلَ الَّذينَ ولَّاهُم علَيهِم فِرعَونُ ويقولونَ لهُم: ”ما بالُكُم قصَّرْتُم أمسِ واليومَ في صُنعِ اللِّبْنِ عَمّا قَبلُ؟“</span>
</div>
</div>

## الرقباء يشكون أمرهم لفرعون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v15" data-v="15"> فجاءَ رُقباءُ بَني إِسرائيلَ إلى فِرعَونَ صارِخينَ وقالوا لَه: ”لماذا تفعَلُ بِنا، نحنُ عبيدَكَ هكذا؟</span>
<span className="v" id="v16" data-v="16"> لا نُعطى تِبْنًا ويقالُ لنا: اَصْنَعوا لِبْنًا! ونُضرَبُ ونُقهَرُ ونحنُ عبيدُكَ ورعاياكَ“.</span>
<span className="v" id="v17" data-v="17" data-xref="ﻣﺘﻰ 26:8; يُوحَنّا 6:27; الخُرُوج 5:8"> فأجابهُم فِرعَونُ: ”أنتُم مُتكاسِلونَ، ولِذلِكَ تقولونَ: دَعْنا نذهبُ ونُقَدِّمُ ذبيحةً للرّبِّ!</span>
<span className="v" id="v18" data-v="18" data-xref="حِزْقِيال 18:18; دانيال 2:9–13"> فاَنْصَرِفوا إلى عَملِكُم، وتِبْنٌ لا يُعطى لكُم، ومِقدارُ اللِّبْنِ ذاتُهُ تُقَدِّمونَهُ“.</span>
</div>
</div>

## الرقباء يلومون موسى وهرون

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v19" data-v="19" data-xref="الجامِعَة 5:8; الجامِعَة 4:1; التَّثْنِيَة 32:36"> فرأى رُقباءُ بَني إِسرائيلَ أنَّهُم في أسوإ حالٍ عِندَما قيلَ لهُم: ”لا يَنقُصْ مِنْ مِقدارِ اللِّبْنِ شيءٌ، بل تُقَدِّمونَه كُلَّ يومٍ في يومِهِ“.</span>
<span className="v" id="v20" data-v="20"> وصادَفوا موسى وهرونَ وهُما ينتظِرانِ خُروجَهُم مِنْ عِندِ فِرعَونَ،</span>
<span className="v" id="v21" data-v="21" data-xref="التَّكوين 34:30; صَمُوئيلَ الأوَّلُ 13:4; أخبارِ الأيّامِ الأوّلُ 19:6"> فقالوا لهُما: ”الرّبُّ يرى ويحكُمُ علَيكُما. لأنَّكُما أفسَدْتُما سُمعَتَنا عِندَ فِرعَونَ ورِجالِه، وأعطَيتُماهُم حُجَّةً لِيَقتلونا“.</span>
</div>
</div>

## الحوار بين الرّبّ وموسى

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="إرْمِيا 20:7; المُلُوكِ الأوَّلُ 19:10; العَدَد 11:14–15"> فرَجَعَ موسى إلى الرّبِّ وقالَ: ”يا ربُّ، لماذا أسأْتَ إلى شعبِكَ لماذا أرسَلتَني؟</span>
<span className="v" id="v23" data-v="23" data-xref="إرْمِيا 11:21; العِبرانِيّين 10:36–37; إشَعْياء 28:16"> فمُنذُ دخَلْتُ على فِرعَونَ لأتكلَّمَ باَسمِكَ، أخذَ يُسيءُ إليهِم وأنتَ لا تعمَلُ شيئًا لإنقاذِهِم“.</span>
