## الرحيل عن سيناء

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="التَّكوين 12:7; الخُرُوج 32:13; الخُرُوج 32:7"> وقالَ الرّبُّ لِموسى: ”قُمْ مِنْ هُنا واَصْعَدْ، أنتَ والشَّعبُ الّذينَ أخرَجْتُهُم مِنْ أرضِ مِصْرَ، إلى الأرضِ الّتي أقسَمْتُ أن أُعطيَها لِنسلِ إبراهيمَ وإسحَقَ ويعقوبَ.</span>
</div>
<div className="bp">
<span className="v" id="v2" data-v="2" data-xref="الخُرُوج 32:34; يَشُوع 24:11; الخُرُوج 23:27–31"> وأنا أُرسِلُ أمامَكَ مَلاكًا وأطرُدُ الكنعانيِّينَ والأموريِّينَ والحِثِّيِّينَ والفِرزِّيِّينَ والحوِّيِّينَ واليَبوسيِّينَ</span>
<span className="v" id="v3" data-v="3" data-xref="الخُرُوج 32:9–10; الخُرُوج 3:8; اعمال الرسل 7:51"> مِنْ تِلكَ الأرضِ الّتي تَفيضُ لَبنًا وعسَلًا. وأمَّا أنا فلا أصعَدُ إليها معَكُم لِئلَّا أُفنيَكُم في الطَّريقِ، لأنَّكُم شعبٌ قُساةُ الرِّقابِ“.</span>
</div>
<div className="bp">
<span className="v" id="v4" data-v="4" data-xref="العَدَد 14:39; العَدَد 14:1; حِزْقِيال 26:16"> فلمَّا سَمِعَ الشَّعبُّ هذا الخَبرَ السَّيِّئَ حَزِنوا ولم يلبَسْ أحدٌ مِنهُم زينَتَهُ.</span>
<span className="v" id="v5" data-v="5" data-xref="العَدَد 16:21; العَدَد 16:45–46; الخُرُوج 33:3–4"> لأنَّ الرّبَّ قالَ لِموسى: ”قُلْ لِبَني إِسرائيلَ: أنتُم شعبٌ قُساةُ الرِّقابِ، فإذا صَعِدْتُ معَكُم، ولو لحظةً واحدةً، أُفنيكُم. والآنَ، فاَنزَعُوا عَنكُم زينَتَكُم وسأرى ما أفعلُ بِكُم“.</span>
<span className="v" id="v6" data-v="6" data-xref="الخُرُوج 33:4; الخُرُوج 32:3; إرْمِيا 2:19"> فنَزعَ بَنو إِسرائيلَ زينَتَهُم بَعدَما ترَكوا جبَلَ حوريبَ.</span>
</div>
</div>

## خيمة الإجتماع

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v7" data-v="7" data-xref="إشَعْياء 55:6–7; الخُرُوج 29:42–43; الأمثال 15:29"> وكانَ كُلَّما نزلَ بَنو إِسرائيلَ في مكانٍ، أخذَ موسى الخَيمةَ المُقدَّسةَ ونصَبَها في خارجِ المَحلَّةِ على بُعْدٍ مِنها. وسَمَّاها خَيمةَ الإجتماعِ، فكانَ كُلُّ مَنْ طَلَبَ مَشورةَ الرّبِّ يخرُجُ إليها.</span>
<span className="v" id="v8" data-v="8" data-xref="العَدَد 16:27"> وكانَ موسى إذا خرَجَ إليها يقومُ كُلُّ واحدٍ مِنَ الشَّعبِ ويقفُ على بابِ خَيمتِهِ وينظرُ إلى موسى حتّى يدخُلَ الخَيمةَ.</span>
<span className="v" id="v9" data-v="9" data-xref="المَزامِير 99:7; الخُرُوج 31:18; الخُرُوج 13:21–22"> وكانَ موسى إذا دخَلَ الخَيمَةَ ينزِلُ عَمودُ السَّحابِ ويقفُ على بابِ الخَيمةِ، ويُكَلِّمُ الرّبُّ موسى.</span>
<span className="v" id="v10" data-v="10" data-xref="المُلُوكِ الأوَّلُ 8:22; المُلُوكِ الأوَّلُ 8:14; الخُرُوج 4:31"> وكانَ إذا رأى الشَّعبُ عَمودَ السَّحابِ واقفًا على بابِ الخَيمةِ يقومونَ ويسجُدونَ، كُلُّ واحدٍ على بابِ خَيمتِه.</span>
<span className="v" id="v11" data-v="11" data-xref="العَدَد 12:8; التَّثْنِيَة 34:10; الخُرُوج 24:13"> ويُكَلِّمُ الرّبُّ موسى وجهًا إلى وجهٍ كما يُكَلِّمُ الإنسانُ صاحبَه. وإذا رجَعَ إلى المَحلَّةِ كانَ خادِمُه يَشوعُ بنُ نُونٍ الشَّابُّ لا يُفارِقُ الخَيمةَ.</span>
</div>
</div>

## الرب مع شعبه

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v12" data-v="12" data-xref="الخُرُوج 33:17; إرْمِيا 1:5; يُوحَنّا 10:14–15"> وقالَ موسى لِلرّبِّ: ”قُلتَ لي أصعِدْ هؤُلاءِ الشَّعبَ إلى تلكَ الأرضِ ولم تُخبِرْني مَنْ تُرسِلُ معي، وقُلتَ لي عرَفْتُكَ بِاَسْمِكَ ورَضيتُ عَنكَ.</span>
<span className="v" id="v13" data-v="13" data-xref="المَزامِير 119:33; المَزامِير 25:4; المَزامِير 27:11"> فالآنَ إنْ كُنتَ رَضيتَ عنِّي، فأرِني طريقَكَ حتّى أعرِفَكَ وأحتَفِظَ بِرِضاكَ. ألا ترى أنَّ هذِهِ الأمَّةَ هيَ شعبُكَ“.</span>
<span className="v" id="v14" data-v="14" data-xref="يَشُوع 21:44; ﻣﺘﻰ 11:28; يَشُوع 1:5"> فقالَ لَه الرّبُّ: ”أنا أسيرُ معَكَ وأهديكَ“.</span>
<span className="v" id="v15" data-v="15" data-xref="الخُرُوج 34:9; المَزامِير 4:6; الخُرُوج 33:3"> فأجابَ موسى: ”إنْ كُنتَ لا تسيرُ معي، فلا تُصعِدْنا مِنْ هُنا.</span>
<span className="v" id="v16" data-v="16" data-xref="التَّثْنِيَة 4:7; العَدَد 14:14; المُلُوكِ الأوَّلُ 8:53"> كيفَ لأَحدٍ أنْ يعرِفَ أنَّكَ راضٍ عنِّي وعَنْ شعبِكَ إنْ كُنتَ لا تسيرُ معَنا، فنتَميَّزَ أنا وشعبُكَ عَنْ كُلِّ شعبٍ على وجهِ الأرضِ؟“</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="إشَعْياء 65:24; رسالَةُ يُوحَنّا الأُولَى 5:14–15; الخُرُوج 33:12"> فقالَ الرّبُّ لِموسى: ”وهذا الّذي قُلْتَهُ أفعَلُهُ، لأنِّي رَضيتُ عَنكَ وعرَفتُك باَسْمِكَ“.</span>
<span className="v" id="v18" data-v="18" data-xref="رُؤيا يُوحَنّا 21:23; الخُرُوج 33:20; تِيطُس 2:13"> فقالَ موسى: ”أرِني مَجدَكَ“.</span>
<span className="v" id="v19" data-v="19" data-xref="ﺭﻭﻣﻴﺔ 9:15–18; أفَسُس 1:6–8; إرْمِيا 31:14"> فقالَ الرّبُّ: ”سأعرِضُ كُلَّ جَلالي أمامَكَ وأُنادي باَسمي أنا الرّبُّ على مسمَعِكَ، وأتَحَنَّنُ على مَنْ أتَحَنَّنُ وأرحَمُ مَنْ أرحَمُ“.</span>
<span className="v" id="v20" data-v="20" data-xref="التَّكوين 32:30; يُوحَنّا 1:18; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:16"> وقالَ: ”أمَّا وجهي فلا تقدِرُ أنْ تراهُ، لأنَّ الّذي يراني لا يعيشُ“.</span>
<span className="v" id="v21" data-v="21" data-xref="التَّثْنِيَة 5:31; زَكَريّا 3:7; يَشُوع 20:4"> وقالَ الرّبُّ: ”هُنا مكانٌ بِجانبي تقِفُ فيهِ على الصَّخرةِ،</span>
<span className="v" id="v22" data-v="22" data-xref="المَزامِير 91:1; إشَعْياء 2:21; المَزامِير 91:4"> وحينَ يمُرُّ مَجدي أجعَلُكَ في فَجوَةِ الصَّخرةِ وأُغَطِّيكَ بِيَدي حتّى أمُرَّ.</span>
<span className="v" id="v23" data-v="23" data-xref="الخُرُوج 33:20; يُوحَنّا 1:18; الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 6:16"> ثُمَّ أُزيحُ يَدي، فتنظُرُ ظَهري. وأما وجهي، فلا تراهُ“.</span>
</div>
