## تحية

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v1" data-v="1" data-xref="يَعقُوب 1:1; اعمال الرسل 2:5–11; العِبرانِيّين 11:13"> مِنْ بُطرُسَ، رَسولِ يَسوعَ المَسيحِ، إلى المُختارينَ المُتَغرِّبينَ المُشَتَّتينَ في بنطُسَ وغَلاطِيَّةَ وكَبَّدُوكِيَّةَ وآسيةَ وبيثينيَّةَ،</span>
<span className="v" id="v2" data-v="2" data-xref="الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:13; أفَسُس 1:4–5; العِبرانِيّين 12:24"> إلى الّذينَ اَختارَهُمُ اللهُ الآبُ بِسابِقِ عِلمِهِ وقَدَّسَهُم بالرُّوحِ ليُطيعوا يَسوعَ المَسيحَ ويَتَطَهَّروا بِرَشِّ دَمِه.<br />علَيكُم وافِرُ النِّعمَةِ والسَّلامِ.</span>

## رجاء حي

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v3" data-v="3" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:23; رسالَةُ يُوحَنّا الأُولَى 5:4; ﺭﻭﻣﻴﺔ 15:13"> تَبارَكَ اللهُ أبو رَبِّنا يَسوعَ المَسيحِ لأنَّهُ شَمَلَنا بِفائِقِ رَحمَتِهِ، فوَلَدَنا بِقيامَةِ يَسوعَ المَسيحِ مِنْ بَينِ الأمواتِ وِلادَةً ثانِيَةً لِرَجاءٍ حَيٍّ</span>
<span className="v" id="v4" data-v="4" data-xref="رُؤيا يُوحَنّا 21:27; العِبرانِيّين 9:15; رسالَةُ بُطرُسَ الأُولَى 5:4"> ولِميراثٍ لا يَفسُدُ ولا يتَدَنَّسُ ولا يَضمَحِلُّ، مَحفوظٍ لكُم في السَّماواتِ،</span>
<span className="v" id="v5" data-v="5" data-xref="فِيلِبِّي 1:6; أفَسُس 2:8; الرِّسالَةُ الثّانيَةُ إلَى تَسالُونِيكِي 2:13–14"> أنتُمُ الّذينَ بِالإيمانِ تَحرُسُكُم قُدرَةُ اللهِ لِخَلاصٍ سينكشِفُ في اليومِ الأخيرِ،</span>
<span className="v" id="v6" data-v="6" data-xref="يَعقُوب 1:2; رسالَةُ بُطرُسَ الأُولَى 5:10; ﺭﻭﻣﻴﺔ 12:12"> بِه تَبتَهِجونَ، معَ أنَّكم لا بُدَّ أنْ تَحزَنوا حينًا بِما يُصيبُكمُ الآنَ مِنْ أنواعِ المِحَنِ</span>
<span className="v" id="v7" data-v="7" data-xref="أيُّوب 23:10; إشَعْياء 48:10; رسالَةُ بُطرُسَ الأُولَى 4:12"> الّتي تَمتَحِنُ إيمانَكُم كما تَمتَحِنُ النَّارُ الذَّهَبَ، وهوَ أثمَنُ مِنَ الذَّهَبِ الفاني، فيكونُ أهلًا لِلمَديحِ والمَجدِ والإكرامِ يومَ ظُهورِ يَسوعَ المَسيحِ.</span>
<span className="v" id="v8" data-v="8" data-xref="يُوحَنّا 20:29; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 5:7; ﺭﻭﻣﻴﺔ 15:13"> أنتُم تُحبُّونَهُ وما رأيتُموهُ، وتُؤْمِنونَ بِه ولا تَرَوْنَهُ الآنَ، فتَفرَحونَ فَرَحًا مَجيدًا لا يوصَفُ،</span>
<span className="v" id="v9" data-v="9" data-xref="ﺭﻭﻣﻴﺔ 6:22; يَعقُوب 1:21; العِبرانِيّين 11:13"> واثِقينَ بِبُلوغِ غايَةِ إيمانِكُم وهِيَ خَلاصُ نُفوسِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v10" data-v="10" data-xref="ﻣﺘﻰ 13:17; لُوقا 10:24; لُوقا 24:25–27"> عَنْ هذا الخَلاصِ فَتَّشَ الأنبياءُ وبَحَثوا، فأنبَأُوا بالنِّعمَةِ الّتي نِلتموها.</span>
<span className="v" id="v11" data-v="11" data-xref="إشَعْياء 52:13–14; رسالَةُ بُطرُسَ الثّانيَة 1:21; لُوقا 24:25–27"> وحاوَلوا أنْ يَعرِفوا الوَقتَ وكيفَ تَجيءُ هذِهِ النِّعمَةُ الّتي دَلَّ علَيها رُوحُ المَسيحِ فيهِم، حينَ شَهِدَ مِنْ قَبلُ بآلامِ المَسيحِ وما يَتلوها مِنْ مَجدٍ.</span>
<span className="v" id="v12" data-v="12" data-xref="يُوحَنّا 15:26; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 2:10; أفَسُس 3:10"> واَنكشَفَ لهُم أنَّهُم كانوا يَعمَلونَ، لا مِنْ أجلِهِم، بَلْ مِنْ أجلِكُم، لِهذِهِ الأُمورِ الّتي أعلَنَها الآنَ لكُمُ الّذينَ بَشَّروكُم بِها، يُؤَيِّدُهُمُ الرُّوحُ القُدُسُ المُرسَلُ مِنَ السَّماءِ، والمَلائِكَةُ يَتَمنَّونَ أنْ يَنظُروا إلَيها.</span>
</div>
</div>

## دعوة إلى حياة القداسة

<div className="bt" dir="rtl">
<div className="bp">
<span className="v" id="v13" data-v="13" data-xref="رسالَةُ بُطرُسَ الأُولَى 5:8; لُوقا 12:35; رسالَةُ بُطرُسَ الأُولَى 4:7"> لذلِكَ هَيِّئوا عُقولَكُم وتَنَبَّهوا واَجعَلوا كُلَّ رَجائِكُم في النِّعمَةِ الّتي تَجيئُكُم عِندَ ظُهورِ يَسوعَ المَسيحِ.</span>
<span className="v" id="v14" data-v="14" data-xref="ﺭﻭﻣﻴﺔ 12:2; تِيطُس 3:3–5; رسالَةُ بُطرُسَ الأُولَى 4:2–3"> وكأبناءٍ طائِعينَ، لا تَتبَعوا شَهَواتِكم ذاتَها الّتي تَبِعتُموها في أيّامِ جَهالَتِكُم،</span>
<span className="v" id="v15" data-v="15" data-xref="ﻣﺘﻰ 5:48; ﻛﻮﺭﻧﺜﻮﺱ ﺍﻟﺜﺎﻧﻴﺔ 7:1; العِبرانِيّين 12:14"> بَلْ كونوا قِدِّيسينَ في كُلِّ ما تَعمَلونَ، لأنَّ اللهَ الّذي دَعاكُم قُدُّوسٌ.</span>
<span className="v" id="v16" data-v="16" data-xref="اللّاويِّين 11:44; اللّاويِّين 19:2; اللّاويِّين 20:7"> فالكِتابُ يَقولُ: ”كونوا قِدِّيسينَ لأنِّي أنا قُدُّوسٌ“.</span>
</div>
<div className="bp">
<span className="v" id="v17" data-v="17" data-xref="اعمال الرسل 10:34–35; أخبارِ الأيّامِ الثّانِي 19:7; ﻣﺘﻰ 6:9"> وإذا كُنتُم تَدعُونَ اللهَ أبًا، وهوَ الّذي يَدينُ مِنْ غَيرِ مُحاباةٍ كُلَّ واحدٍ على قَدْرِ أعمالِهِ، فَعيشوا مُدَّةَ غُربَتِكُم في مخافَتِهِ،</span>
<span className="v" id="v18" data-v="18" data-xref="ﻛﻮﺭﻧﺜﻮﺱ ﺍﻻﻭﻝ 6:20; تِيطُس 2:14; المَزامِير 49:7–8"> عارِفينَ أنَّهُ اَفتَداكُم مِنْ سِيرَتِكُمُ الباطِلَةِ الّتي ورِثتُموها عَنْ آبائِكُم، لا بِالفاني مِنَ الفِضَّةِ أوِ الذَّهَبِ،</span>
<span className="v" id="v19" data-v="19" data-xref="رسالَةُ يُوحَنّا الأُولَى 1:7; رسالَةُ بُطرُسَ الأُولَى 2:22–24; يُوحَنّا 1:29"> بَلْ بِدَمٍ كَريمٍ، دمِ الحمَلِ الّذي لا عَيبَ فيهِ ولا دنَسَ، دَمِ المَسيحِ.</span>
<span className="v" id="v20" data-v="20" data-xref="أفَسُس 1:4; الرِّسالَةُ الثّانيَةُ إلَى تِيمُوثاوُس 1:9–10; العِبرانِيّين 9:26"> وكانَ اللهُ اَختارَهُ قَبلَ إنشاءِ العالَمِ، ثُمَّ تَجَلَّى مِنْ أجلِكُم في الأزمِنَةِ الأخيرَةِ،</span>
<span className="v" id="v21" data-v="21" data-xref="اعمال الرسل 2:24; العِبرانِيّين 2:9; يُوحَنّا 12:44"> وهوَ الّذي جعَلَكُم تُؤمِنونَ باللهِ الّذي أقامَهُ مِنْ بَينِ الأمواتِ ووهَبَهُ المَجدَ، فأصبَحَ اللهُ غايَةَ إيمانِكُم ورَجائِكُم.</span>
</div>
<div className="bp">
<span className="v" id="v22" data-v="22" data-xref="الرِّسالَةُ الأُولَى إلَى تِيمُوثاوُس 1:5; العِبرانِيّين 13:1; ﺭﻭﻣﻴﺔ 12:9–10"> والآنَ، بَعدَما طَهَّرتُم نُفوسَكُم بإطاعَةِ الحَقِّ وصِرتُم تُحبُّونَ إخوَتَكُم حُبًّا صادِقًا، أحِبُّوا بَعضُكم بَعضًا حُبًّا طاهِرًا مِنْ صَميمِ القَلبِ.</span>
<span className="v" id="v23" data-v="23" data-xref="رسالَةُ بُطرُسَ الأُولَى 1:3; العِبرانِيّين 4:12; يُوحَنّا 1:13"> فأنتُم وُلِدتُم وِلادَةً ثانِيَةً، لا مِنْ زَرعٍ يَفنى، بَلْ مِنْ زَرع لا يَفنى، وهوَ كَلِمةُ اللهِ الحَيَّةُ الباقِيَةُ.</span>
<span className="v" id="v24" data-v="24" data-xref="إشَعْياء 40:6–8; المَزامِير 103:15; يَعقُوب 4:14"> فالكِتابُ يقولُ:<br />”كُلُّ بَشَرٍ كَالعُشبِ<br />وكُلُّ مَجدهِ كَزَهرِ العُشبِ.<br />العُشبُ يَيبَسُ وزَهرُهُ يَسقُطُ،</span>
</div>
<div className="bp">
<span className="v" id="v25" data-v="25" data-xref="إشَعْياء 40:8; ﻣﺘﻰ 5:18; المَزامِير 119:89"> وكلامُ اللهِ يَبقى إلى الأبَدِ“.<br />هذا هوَ الكلامُ الّذي بَشَّرناكُم بهِ.</span>
