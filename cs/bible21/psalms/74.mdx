<div className="bp poetry">

</div>
<div className="bp poetry">

</div>
<div className="bp poetry">

<span className="v" id="v15" data-v="15" data-xref="Numbers 20:11; Isaiah 48:21; Psalms 105:41"> Ty jsi dal průchod pramenným vodám,<br /><span className="pl pl-2">mohutné řeky ty jsi vysušil.</span></span>

</div>

</div>
<div className="bp poetry">
